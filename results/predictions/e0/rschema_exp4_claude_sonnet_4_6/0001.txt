-- Schema for database: rschema_0001

CREATE TABLE `airplane` (
  `airplane_number` varchar(20) NOT NULL,
  `model` varchar(100) NOT NULL,
  `passenger_capacity` int(11) NOT NULL,
  PRIMARY KEY (`airplane_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pilot` (
  `pilot_id` varchar(20) NOT NULL,
  `name` varchar(100) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`pilot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `flight_number` varchar(20) NOT NULL,
  `departure_time` datetime NOT NULL,
  `arrival_time` datetime NOT NULL,
  `origin` varchar(100) NOT NULL,
  `destination` varchar(100) NOT NULL,
  `airplane_number` varchar(20) NOT NULL,
  `captain_id` varchar(20) NOT NULL,
  `first_officer_id` varchar(20) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`flight_number`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`airplane_number`) REFERENCES `airplane` (`airplane_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_2` FOREIGN KEY (`captain_id`) REFERENCES `pilot` (`pilot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_3` FOREIGN KEY (`first_officer_id`) REFERENCES `pilot` (`pilot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
