-- Schema for database: rschema_0003

CREATE TABLE `warehouse` (
  `warehouse_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_no` varchar(50) NOT NULL,
  `address` varchar(255) NOT NULL,
  `capacity` decimal(10,2) NOT NULL,
  PRIMARY KEY (`warehouse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `worker` (
  `worker_id` int(11) NOT NULL AUTO_INCREMENT,
  `worker_no` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `phone` varchar(20) NOT NULL,
  PRIMARY KEY (`worker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inbound_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_id` int(11) NOT NULL,
  `worker_id` int(11) NOT NULL,
  `goods_name` varchar(255) NOT NULL,
  `quantity` decimal(10,2) NOT NULL,
  `inbound_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `inbound_task_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `inbound_task_ibfk_2` FOREIGN KEY (`worker_id`) REFERENCES `worker` (`worker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `outbound_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_id` int(11) NOT NULL,
  `worker_id` int(11) NOT NULL,
  `goods_name` varchar(255) NOT NULL,
  `quantity` decimal(10,2) NOT NULL,
  `outbound_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `outbound_task_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `outbound_task_ibfk_2` FOREIGN KEY (`worker_id`) REFERENCES `worker` (`worker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouse_monitor` (
  `monitor_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_id` int(11) NOT NULL,
  `current_usage` decimal(10,2) NOT NULL DEFAULT '0.00',
  `status` varchar(50) NOT NULL DEFAULT 'normal',
  `recorded_at` datetime NOT NULL,
  PRIMARY KEY (`monitor_id`),
  CONSTRAINT `warehouse_monitor_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `worker_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `worker_id` int(11) NOT NULL,
  `evaluation_period` varchar(50) NOT NULL,
  `inbound_count` int(11) NOT NULL DEFAULT '0',
  `outbound_count` int(11) NOT NULL DEFAULT '0',
  `total_quantity` decimal(10,2) NOT NULL DEFAULT '0.00',
  `score` decimal(5,2) DEFAULT NULL,
  `evaluated_at` datetime NOT NULL,
  PRIMARY KEY (`performance_id`),
  CONSTRAINT `worker_performance_ibfk_1` FOREIGN KEY (`worker_id`) REFERENCES `worker` (`worker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
