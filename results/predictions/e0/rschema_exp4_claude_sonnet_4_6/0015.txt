-- Schema for database: rschema_0015

CREATE TABLE `operator` (
  `employee_id` varchar(20) NOT NULL,
  `name` varchar(100) NOT NULL,
  `gender` char(1) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plot` (
  `plot_id` varchar(20) NOT NULL,
  `area` decimal(10,2) DEFAULT NULL,
  `soil_type` varchar(100) DEFAULT NULL,
  `geographical_location` varchar(255) DEFAULT NULL,
  `pest_disease_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pest_disease` (
  `pest_disease_id` varchar(20) NOT NULL,
  `name` varchar(100) NOT NULL,
  `damage_degree` varchar(50) DEFAULT NULL,
  `prevention_method` varchar(500) DEFAULT NULL,
  `occurrence_season` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`pest_disease_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `plot_id` varchar(20) NOT NULL,
  `pest_disease_id` varchar(20) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `monitoring_result` varchar(500) DEFAULT NULL,
  `employee_id` varchar(20) NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `monitoring_record_ibfk_1` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_2` FOREIGN KEY (`pest_disease_id`) REFERENCES `pest_disease` (`pest_disease_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_3` FOREIGN KEY (`employee_id`) REFERENCES `operator` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
