-- Schema for database: rschema_0016

CREATE TABLE `farm` (
  `farm_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(500) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`farm_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL AUTO_INCREMENT,
  `farm_id` int(11) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  `soil_type` varchar(100) DEFAULT NULL,
  `current_status` varchar(100) NOT NULL DEFAULT 'idle',
  PRIMARY KEY (`plot_id`),
  CONSTRAINT `plot_ibfk_1` FOREIGN KEY (`farm_id`) REFERENCES `farm` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `planting_plan` (
  `plan_id` int(11) NOT NULL AUTO_INCREMENT,
  `farm_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `crop_type` varchar(255) NOT NULL,
  `planting_time` datetime NOT NULL,
  `expected_harvest_time` datetime NOT NULL,
  `planting_quantity` decimal(10,2) NOT NULL,
  `operator` varchar(255) NOT NULL,
  PRIMARY KEY (`plan_id`),
  CONSTRAINT `planting_plan_ibfk_1` FOREIGN KEY (`farm_id`) REFERENCES `farm` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `planting_plan_ibfk_2` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
