-- Schema for database: rschema_0022

CREATE TABLE `personnel` (
  `employee_id` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `gender` char(6) NOT NULL,
  `position` varchar(100) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `document` (
  `document_id` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `category` varchar(100) NOT NULL,
  `creation_time` datetime NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`document_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_record` (
  `approval_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_id` varchar(50) NOT NULL,
  `approver_id` varchar(50) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_comments` text DEFAULT NULL,
  PRIMARY KEY (`approval_id`),
  CONSTRAINT `approval_record_ibfk_1` FOREIGN KEY (`document_id`) REFERENCES `document` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `approval_record_ibfk_2` FOREIGN KEY (`approver_id`) REFERENCES `personnel` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `archiving_record` (
  `archiving_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_id` varchar(50) NOT NULL,
  `archivist_id` varchar(50) NOT NULL,
  `archiving_time` datetime NOT NULL,
  `archiving_location` varchar(255) NOT NULL,
  PRIMARY KEY (`archiving_id`),
  UNIQUE KEY `document_id` (`document_id`),
  CONSTRAINT `archiving_record_ibfk_1` FOREIGN KEY (`document_id`) REFERENCES `document` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `archiving_record_ibfk_2` FOREIGN KEY (`archivist_id`) REFERENCES `personnel` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `query_record` (
  `query_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_id` varchar(50) NOT NULL,
  `query_personnel_id` varchar(50) NOT NULL,
  `query_time` datetime NOT NULL,
  `query_result` text DEFAULT NULL,
  PRIMARY KEY (`query_id`),
  CONSTRAINT `query_record_ibfk_1` FOREIGN KEY (`document_id`) REFERENCES `document` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `query_record_ibfk_2` FOREIGN KEY (`query_personnel_id`) REFERENCES `personnel` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
