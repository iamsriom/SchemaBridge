-- Schema for database: rschema_0028

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `manager` varchar(255) DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_name` varchar(255) NOT NULL,
  `gender` char(6) DEFAULT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timber` (
  `timber_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_name` varchar(255) NOT NULL,
  `timber_type` varchar(255) DEFAULT NULL,
  `purchase_price` decimal(10,2) DEFAULT NULL,
  `processing_cost` decimal(10,2) DEFAULT NULL,
  `selling_price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`timber_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `supplier` (
  `supplier_id` int(11) NOT NULL AUTO_INCREMENT,
  `supplier_name` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`supplier_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `procurement` (
  `procurement_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_id` int(11) NOT NULL,
  `quantity` decimal(10,2) NOT NULL,
  `purchaser_id` int(11) NOT NULL,
  `supplier_id` int(11) NOT NULL,
  `purchase_time` datetime NOT NULL,
  PRIMARY KEY (`procurement_id`),
  CONSTRAINT `procurement_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `procurement_ibfk_2` FOREIGN KEY (`purchaser_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `procurement_ibfk_3` FOREIGN KEY (`supplier_id`) REFERENCES `supplier` (`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing` (
  `processing_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_id` int(11) NOT NULL,
  `quantity` decimal(10,2) NOT NULL,
  `processor_id` int(11) NOT NULL,
  `processing_time` datetime NOT NULL,
  PRIMARY KEY (`processing_id`),
  CONSTRAINT `processing_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `processing_ibfk_2` FOREIGN KEY (`processor_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `sales_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_id` int(11) NOT NULL,
  `quantity` decimal(10,2) NOT NULL,
  `salesperson_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  PRIMARY KEY (`sales_id`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`salesperson_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_3` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluation` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `evaluation_period` varchar(255) DEFAULT NULL,
  `score` decimal(5,2) DEFAULT NULL,
  `remarks` varchar(255) DEFAULT NULL,
  `evaluation_time` datetime DEFAULT NULL,
  PRIMARY KEY (`evaluation_id`),
  CONSTRAINT `performance_evaluation_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
