-- Schema for database: rschema_0032

CREATE TABLE `rental_location` (
  `location_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_name` varchar(255) NOT NULL,
  `address` varchar(500) DEFAULT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `manager_name` varchar(255) DEFAULT NULL,
  `total_tasks_completed` int(11) DEFAULT NULL,
  `customer_satisfaction_score` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`location_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_id` int(11) NOT NULL,
  `make` varchar(100) NOT NULL,
  `model` varchar(100) NOT NULL,
  `year` int(11) DEFAULT NULL,
  `license_plate` varchar(50) DEFAULT NULL,
  `vehicle_type` varchar(100) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `rental_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone` varchar(50) NOT NULL,
  `rental_date` datetime NOT NULL,
  `rental_duration_days` int(11) NOT NULL,
  `rental_cost` decimal(10,2) NOT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `rental_task_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `rental_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_task_ibfk_2` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_completion` (
  `completion_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_id` int(11) NOT NULL,
  `completion_date` datetime DEFAULT NULL,
  `vehicle_condition` varchar(255) DEFAULT NULL,
  `customer_feedback` text DEFAULT NULL,
  `satisfaction_rating` int(11) DEFAULT NULL,
  PRIMARY KEY (`completion_id`),
  CONSTRAINT `rental_completion_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `rental_task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `location_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_id` int(11) NOT NULL,
  `assessment_period` varchar(50) DEFAULT NULL,
  `tasks_completed` int(11) DEFAULT NULL,
  `average_satisfaction` decimal(5,2) DEFAULT NULL,
  `assessment_date` datetime DEFAULT NULL,
  PRIMARY KEY (`performance_id`),
  CONSTRAINT `location_performance_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `rental_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
