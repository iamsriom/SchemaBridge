-- Schema for database: rschema_0043

CREATE TABLE `loan_application` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_number` varchar(100) NOT NULL,
  `loan_amount` decimal(15,2) NOT NULL,
  `loan_term` int(11) NOT NULL,
  `approval_status` varchar(50) NOT NULL DEFAULT 'pending',
  `applicant_name` varchar(100) NOT NULL,
  `applicant_phone` varchar(20) NOT NULL,
  `applicant_id_number` varchar(50) NOT NULL,
  `application_time` datetime NOT NULL,
  PRIMARY KEY (`application_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approver` (
  `approver_id` int(11) NOT NULL AUTO_INCREMENT,
  `approver_name` varchar(100) NOT NULL,
  `contact_info` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`approver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval` (
  `approval_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_id` int(11) NOT NULL,
  `approver_id` int(11) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_result` varchar(50) NOT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`approval_id`),
  CONSTRAINT `loan_approval_ibfk_1` FOREIGN KEY (`application_id`) REFERENCES `loan_application` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_ibfk_2` FOREIGN KEY (`approver_id`) REFERENCES `approver` (`approver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
