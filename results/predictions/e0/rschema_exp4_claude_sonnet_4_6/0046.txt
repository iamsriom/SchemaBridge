-- Schema for database: rschema_0046

CREATE TABLE `members` (
  `member_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `current_points` int(11) NOT NULL DEFAULT '0',
  PRIMARY KEY (`member_id`),
  UNIQUE KEY `phone_number` (`phone_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shopping_records` (
  `shopping_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `shopping_time` datetime NOT NULL,
  `total_price` decimal(10,2) NOT NULL,
  `points_change` int(11) NOT NULL,
  PRIMARY KEY (`shopping_id`),
  KEY `member_id` (`member_id`),
  CONSTRAINT `shopping_records_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `redemption_records` (
  `redemption_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `redemption_time` datetime NOT NULL,
  `product_name` varchar(200) NOT NULL,
  `redemption_quantity` int(11) NOT NULL,
  `points_consumed` int(11) NOT NULL,
  PRIMARY KEY (`redemption_id`),
  KEY `member_id` (`member_id`),
  CONSTRAINT `redemption_records_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
