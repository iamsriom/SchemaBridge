-- Schema for database: rschema_0051

CREATE TABLE `resource` (
  `resource_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `resource_type` varchar(100) NOT NULL,
  PRIMARY KEY (`resource_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(50) NOT NULL,
  `position` varchar(100) NOT NULL,
  `personnel_role` varchar(50) NOT NULL COMMENT 'e.g. monitoring or maintenance',
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring` (
  `monitoring_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `monitoring_point` varchar(255) NOT NULL,
  `water_quality_indicator` varchar(255) DEFAULT NULL,
  `water_volume` decimal(15,3) DEFAULT NULL,
  `pollution_level` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`monitoring_id`),
  CONSTRAINT `monitoring_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintenance_cost` decimal(15,2) DEFAULT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `maintenance_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
