-- Schema for database: rschema_0059
CREATE TABLE `handler` (
  `handler_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`handler_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fault` (
  `fault_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` text DEFAULT NULL,
  `occurrence_date` datetime NOT NULL,
  `resolution_date` datetime DEFAULT NULL,
  `handler_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`fault_id`),
  CONSTRAINT `fault_ibfk_1` FOREIGN KEY (`handler_id`) REFERENCES `handler` (`handler_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fault_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
