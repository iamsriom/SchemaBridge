-- Schema for database: rschema_0060

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  `project_leader` int(11) NOT NULL,
  `budget` decimal(15,2) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `researcher` (
  `employee_number` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `position` varchar(100) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_number`),
  CONSTRAINT `researcher_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

ALTER TABLE `project` ADD CONSTRAINT `project_ibfk_1` FOREIGN KEY (`project_leader`) REFERENCES `researcher` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE;

CREATE TABLE `research_result` (
  `result_id` int(11) NOT NULL AUTO_INCREMENT,
  `result_name` varchar(255) NOT NULL,
  `publication_date` datetime NOT NULL,
  `researcher_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`result_id`),
  CONSTRAINT `research_result_ibfk_1` FOREIGN KEY (`researcher_id`) REFERENCES `researcher` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `research_result_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fund_usage` (
  `fund_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `purpose` varchar(255) NOT NULL,
  `amount` decimal(15,2) NOT NULL,
  `usage_date` datetime NOT NULL,
  PRIMARY KEY (`fund_id`),
  CONSTRAINT `fund_usage_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
