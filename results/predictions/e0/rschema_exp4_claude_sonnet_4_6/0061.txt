-- Schema for database: rschema_0061
CREATE TABLE `patent` (
  `patent_number` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`patent_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventor` (
  `work_number` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `gender` char(6) NOT NULL,
  `telephone_number` varchar(20) NOT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`work_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patent_application` (
  `application_number` varchar(50) NOT NULL,
  `patent_number` varchar(50) NOT NULL,
  `work_number` varchar(50) NOT NULL,
  `application_date` datetime NOT NULL,
  PRIMARY KEY (`application_number`),
  CONSTRAINT `patent_application_ibfk_1` FOREIGN KEY (`patent_number`) REFERENCES `patent` (`patent_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patent_application_ibfk_2` FOREIGN KEY (`work_number`) REFERENCES `inventor` (`work_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
