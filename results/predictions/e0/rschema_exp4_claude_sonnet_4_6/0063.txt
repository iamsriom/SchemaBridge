-- Schema for database: rschema_0063

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `achievement` (
  `achievement_id` varchar(50) NOT NULL,
  `achievement_name` varchar(255) NOT NULL,
  `rd_date` datetime NOT NULL,
  `technical_field` varchar(255) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`achievement_id`),
  CONSTRAINT `achievement_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` varchar(50) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `model` varchar(100) NOT NULL,
  `barcode` varchar(100) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `transformation_date` datetime NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `achievement_transformation` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `achievement_id` varchar(50) NOT NULL,
  `product_id` varchar(50) NOT NULL,
  `transformation_date` datetime NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `achievement_transformation_ibfk_1` FOREIGN KEY (`achievement_id`) REFERENCES `achievement` (`achievement_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `achievement_transformation_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
