-- Schema for database: rschema_0064
CREATE TABLE `project` (
  `project_no` varchar(20) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `application_unit` varchar(255) NOT NULL,
  PRIMARY KEY (`project_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expert` (
  `expert_no` varchar(20) NOT NULL,
  `expert_name` varchar(100) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `title` varchar(100) NOT NULL,
  PRIMARY KEY (`expert_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `project_no` varchar(20) NOT NULL,
  `expert_no` varchar(20) NOT NULL,
  `review_date` datetime NOT NULL,
  `review_opinion` text DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `review_record_ibfk_1` FOREIGN KEY (`project_no`) REFERENCES `project` (`project_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `review_record_ibfk_2` FOREIGN KEY (`expert_no`) REFERENCES `expert` (`expert_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
