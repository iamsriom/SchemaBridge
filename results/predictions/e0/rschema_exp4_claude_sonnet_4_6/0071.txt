-- Schema for database: rschema_0071

CREATE TABLE `mine` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_number` varchar(50) NOT NULL,
  `mine_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(100) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_officer` (
  `officer_id` int(11) NOT NULL AUTO_INCREMENT,
  `officer_number` varchar(50) NOT NULL,
  `officer_name` varchar(100) NOT NULL,
  `gender` char(6) NOT NULL,
  `telephone` varchar(20) DEFAULT NULL,
  `position` varchar(100) NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`officer_id`),
  CONSTRAINT `safety_officer_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspection` (
  `inspection_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_id` int(11) NOT NULL,
  `officer_id` int(11) NOT NULL,
  `inspection_date` datetime NOT NULL,
  `inspection_content` text NOT NULL,
  `inspection_result` varchar(255) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  CONSTRAINT `safety_inspection_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `safety_inspection_ibfk_2` FOREIGN KEY (`officer_id`) REFERENCES `safety_officer` (`officer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
