-- Schema for database: rschema_0075

CREATE TABLE `analyst` (
  `analyst_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_hire` datetime NOT NULL,
  `contact_number` varchar(50) NOT NULL,
  PRIMARY KEY (`analyst_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stock` (
  `stock_code` char(10) NOT NULL,
  `stock_name` varchar(255) NOT NULL,
  `current_price` decimal(15,2) NOT NULL,
  PRIMARY KEY (`stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `analysis` (
  `analysis_id` int(11) NOT NULL,
  `analyst_id` int(11) NOT NULL,
  `stock_code` char(10) NOT NULL,
  `analysis_date` datetime NOT NULL,
  `results` varchar(255) NOT NULL,
  `reasoning` text NOT NULL,
  PRIMARY KEY (`analysis_id`),
  CONSTRAINT `analysis_ibfk_1` FOREIGN KEY (`analyst_id`) REFERENCES `analyst` (`analyst_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `analysis_ibfk_2` FOREIGN KEY (`stock_code`) REFERENCES `stock` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stock_price_history` (
  `history_id` int(11) NOT NULL,
  `stock_code` char(10) NOT NULL,
  `price_date` datetime NOT NULL,
  `closing_price` decimal(15,2) NOT NULL,
  `opening_price` decimal(15,2) NOT NULL,
  `highest_price` decimal(15,2) NOT NULL,
  `lowest_price` decimal(15,2) NOT NULL,
  `trading_volume` bigint(20) NOT NULL,
  PRIMARY KEY (`history_id`),
  CONSTRAINT `stock_price_history_ibfk_1` FOREIGN KEY (`stock_code`) REFERENCES `stock` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
