-- Schema for database: rschema_0076

CREATE TABLE `manufacturer` (
  `manufacturer_id` int(11) NOT NULL,
  `manufacturer_name` varchar(255) NOT NULL,
  `manufacturer_country` varchar(255) NOT NULL,
  PRIMARY KEY (`manufacturer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beer` (
  `beer_id` int(11) NOT NULL,
  `beer_name` varchar(255) NOT NULL,
  `manufacturer_id` int(11) NOT NULL,
  `beer_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`beer_id`),
  CONSTRAINT `beer_ibfk_1` FOREIGN KEY (`manufacturer_id`) REFERENCES `manufacturer` (`manufacturer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bar` (
  `bar_id` int(11) NOT NULL,
  `bar_name` varchar(255) NOT NULL,
  `bar_address` varchar(255) NOT NULL,
  `bar_phone` varchar(50) NOT NULL,
  PRIMARY KEY (`bar_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drinker` (
  `drinker_id` int(11) NOT NULL,
  `drinker_name` varchar(255) NOT NULL,
  `drinker_gender` char(6) NOT NULL,
  `drinker_age` int(11) NOT NULL,
  `drinker_phone` varchar(50) NOT NULL,
  PRIMARY KEY (`drinker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `sales_id` int(11) NOT NULL,
  `bar_id` int(11) NOT NULL,
  `beer_id` int(11) NOT NULL,
  `drinker_id` int(11) NOT NULL,
  `sales_date` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`sales_id`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`bar_id`) REFERENCES `bar` (`bar_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`beer_id`) REFERENCES `beer` (`beer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_3` FOREIGN KEY (`drinker_id`) REFERENCES `drinker` (`drinker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
