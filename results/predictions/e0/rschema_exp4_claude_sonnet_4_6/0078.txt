-- Schema for database: rschema_0078

CREATE TABLE `author` (
  `author_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `introduction` text DEFAULT NULL,
  PRIMARY KEY (`author_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `article` (
  `article_id` int(11) NOT NULL AUTO_INCREMENT,
  `article_title` varchar(500) NOT NULL,
  `article_content` longtext DEFAULT NULL,
  `creation_time` datetime NOT NULL,
  `total_words` int(11) DEFAULT NULL,
  `read_volume` int(11) DEFAULT 0,
  `like_volume` int(11) DEFAULT 0,
  `collection_volume` int(11) DEFAULT 0,
  `comment_volume` int(11) DEFAULT 0,
  `last_modified_time` datetime DEFAULT NULL,
  `last_modified_author_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`article_id`),
  CONSTRAINT `article_ibfk_1` FOREIGN KEY (`last_modified_author_id`) REFERENCES `author` (`author_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `article_author` (
  `article_id` int(11) NOT NULL,
  `author_id` int(11) NOT NULL,
  `author_order` int(11) NOT NULL,
  `responsible_part` varchar(500) DEFAULT NULL,
  `role` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`article_id`, `author_id`),
  CONSTRAINT `article_author_ibfk_1` FOREIGN KEY (`article_id`) REFERENCES `article` (`article_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `article_author_ibfk_2` FOREIGN KEY (`author_id`) REFERENCES `author` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `article_modification` (
  `modification_id` int(11) NOT NULL AUTO_INCREMENT,
  `article_id` int(11) NOT NULL,
  `author_id` int(11) NOT NULL,
  `modification_start_time` datetime NOT NULL,
  `modification_save_time` datetime DEFAULT NULL,
  `modification_notes` text DEFAULT NULL,
  PRIMARY KEY (`modification_id`),
  CONSTRAINT `article_modification_ibfk_1` FOREIGN KEY (`article_id`) REFERENCES `article` (`article_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `article_modification_ibfk_2` FOREIGN KEY (`author_id`) REFERENCES `author` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
