-- Schema for database: rschema_0096

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `hair_color` varchar(100) DEFAULT NULL,
  `appointment_time` datetime NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hairstylist` (
  `hairstylist_id` int(11) NOT NULL AUTO_INCREMENT,
  `hairstylist_name` varchar(255) NOT NULL,
  `expertise` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`hairstylist_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `appointment` (
  `appointment_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `hairstylist_id` int(11) NOT NULL,
  `appointment_time` datetime NOT NULL,
  PRIMARY KEY (`appointment_id`),
  CONSTRAINT `appointment_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `appointment_ibfk_2` FOREIGN KEY (`hairstylist_id`) REFERENCES `hairstylist` (`hairstylist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_record` (
  `service_id` int(11) NOT NULL AUTO_INCREMENT,
  `appointment_id` int(11) NOT NULL,
  `service_time` datetime NOT NULL,
  `service_content` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`service_id`),
  CONSTRAINT `service_record_ibfk_1` FOREIGN KEY (`appointment_id`) REFERENCES `appointment` (`appointment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
