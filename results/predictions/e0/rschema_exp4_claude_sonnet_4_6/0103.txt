-- Schema for database: rschema_0103

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` date NOT NULL,
  `hire_date` date NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_code` char(6) NOT NULL,
  `department_name` varchar(100) NOT NULL,
  `manager_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`department_code`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`manager_id`) REFERENCES `employee` (`employee_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position` (
  `position_id` int(11) NOT NULL AUTO_INCREMENT,
  `position_name` varchar(100) NOT NULL,
  `department_code` char(6) NOT NULL,
  PRIMARY KEY (`position_id`),
  CONSTRAINT `position_ibfk_1` FOREIGN KEY (`department_code`) REFERENCES `department` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_department` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `department_code` char(6) NOT NULL,
  `position_id` int(11) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `employee_department_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_department_ibfk_2` FOREIGN KEY (`department_code`) REFERENCES `department` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_department_ibfk_3` FOREIGN KEY (`position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position_change_history` (
  `change_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `original_department_code` char(6) NOT NULL,
  `original_position_id` int(11) NOT NULL,
  `current_department_code` char(6) NOT NULL,
  `current_position_id` int(11) NOT NULL,
  `entry_date` date NOT NULL,
  `departure_date` date DEFAULT NULL,
  PRIMARY KEY (`change_id`),
  CONSTRAINT `position_change_history_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `position_change_history_ibfk_2` FOREIGN KEY (`original_department_code`) REFERENCES `department` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `position_change_history_ibfk_3` FOREIGN KEY (`original_position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `position_change_history_ibfk_4` FOREIGN KEY (`current_department_code`) REFERENCES `department` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `position_change_history_ibfk_5` FOREIGN KEY (`current_position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_standard` (
  `standard_id` int(11) NOT NULL AUTO_INCREMENT,
  `standard_description` varchar(255) NOT NULL,
  `standard_value` decimal(10,2) NOT NULL,
  PRIMARY KEY (`standard_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_payment` (
  `payment_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `standard_id` int(11) NOT NULL,
  `payment_date` date NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `department_code` char(6) NOT NULL,
  PRIMARY KEY (`payment_id`),
  CONSTRAINT `salary_payment_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `salary_payment_ibfk_2` FOREIGN KEY (`standard_id`) REFERENCES `salary_standard` (`standard_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `salary_payment_ibfk_3` FOREIGN KEY (`department_code`) REFERENCES `department` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
