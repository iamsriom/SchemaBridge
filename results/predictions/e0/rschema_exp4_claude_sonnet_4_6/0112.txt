-- Schema for database: rschema_0112

CREATE TABLE `offices` (
  `office_code` varchar(10) NOT NULL,
  `city` varchar(50) NOT NULL,
  `street_address` varchar(255) NOT NULL,
  `postal_code` varchar(15) NOT NULL,
  `country` varchar(50) NOT NULL,
  `phone` varchar(50) NOT NULL,
  `fax` varchar(50) DEFAULT NULL,
  `manager_employee_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`office_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_number` int(11) NOT NULL,
  `last_name` varchar(50) NOT NULL,
  `first_name` varchar(50) NOT NULL,
  `job_title` varchar(50) NOT NULL,
  `hire_date` date NOT NULL,
  `office_phone` varchar(50) DEFAULT NULL,
  `email` varchar(100) NOT NULL,
  `reports_to` int(11) DEFAULT NULL,
  `office_code` varchar(10) NOT NULL,
  `salary` decimal(10,2) DEFAULT NULL,
  `commission_rate` decimal(5,4) DEFAULT NULL,
  PRIMARY KEY (`employee_number`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`reports_to`) REFERENCES `employees` (`employee_number`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `employees_ibfk_2` FOREIGN KEY (`office_code`) REFERENCES `offices` (`office_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

ALTER TABLE `offices` ADD CONSTRAINT `offices_ibfk_1` FOREIGN KEY (`manager_employee_number`) REFERENCES `employees` (`employee_number`) ON DELETE SET NULL ON UPDATE CASCADE;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL,
  `customer_name` varchar(100) NOT NULL,
  `contact_last_name` varchar(50) NOT NULL,
  `contact_first_name` varchar(50) NOT NULL,
  `contact_job_title` varchar(50) DEFAULT NULL,
  `contact_phone` varchar(50) NOT NULL,
  `address_line1` varchar(255) NOT NULL,
  `address_line2` varchar(255) DEFAULT NULL,
  `city` varchar(50) NOT NULL,
  `state` varchar(50) DEFAULT NULL,
  `postal_code` varchar(15) DEFAULT NULL,
  `country` varchar(50) NOT NULL,
  `sales_rep_employee_number` int(11) DEFAULT NULL,
  `credit_limit` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`customer_id`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`sales_rep_employee_number`) REFERENCES `employees` (`employee_number`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `payments` (
  `customer_id` int(11) NOT NULL,
  `check_number` varchar(50) NOT NULL,
  `payment_date` date NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `check_clear_date` date DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `check_number`),
  CONSTRAINT `payments_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product_lines` (
  `product_line` varchar(50) NOT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`product_line`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
  `product_code` varchar(15) NOT NULL,
  `product_name` varchar(100) NOT NULL,
  `product_line` varchar(50) NOT NULL,
  `product_scale` varchar(10) NOT NULL,
  `product_vendor` varchar(100) NOT NULL,
  `product_description` text NOT NULL,
  `quantity_in_stock` int(11) NOT NULL,
  `buy_price` decimal(10,2) NOT NULL,
  `msrp` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_code`),
  CONSTRAINT `products_ibfk_1` FOREIGN KEY (`product_line`) REFERENCES `product_lines` (`product_line`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
  `order_number` int(11) NOT NULL,
  `order_date` date NOT NULL,
  `required_date` date NOT NULL,
  `shipped_date` date DEFAULT NULL,
  `status` varchar(15) NOT NULL,
  `comments` text DEFAULT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`order_number`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order_details` (
  `order_number` int(11) NOT NULL,
  `product_code` varchar(15) NOT NULL,
  `quantity_ordered` int(11) NOT NULL,
  `price_each` decimal(10,2) NOT NULL,
  `order_line_number` smallint(6) NOT NULL,
  PRIMARY KEY (`order_number`, `product_code`),
  CONSTRAINT `order_details_ibfk_1` FOREIGN KEY (`order_number`) REFERENCES `orders` (`order_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_details_ibfk_2` FOREIGN KEY (`product_code`) REFERENCES `products` (`product_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_performance` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_number` int(11) NOT NULL,
  `fiscal_year` int(4) NOT NULL,
  `total_sales` decimal(15,2) NOT NULL,
  `sales_target` decimal(15,2) DEFAULT NULL,
  `performance_change` decimal(10,4) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_sales_performance` (`employee_number`, `fiscal_year`),
  CONSTRAINT `sales_performance_ibfk_1` FOREIGN KEY (`employee_number`) REFERENCES `employees` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
