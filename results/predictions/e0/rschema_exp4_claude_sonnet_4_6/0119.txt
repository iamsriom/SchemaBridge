-- Schema for database: rschema_0119

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `indicators` (
  `indicator_id` int(11) NOT NULL AUTO_INCREMENT,
  `indicator_name` varchar(255) NOT NULL,
  `indicator_target` decimal(15,4) NOT NULL,
  `indicator_description` text DEFAULT NULL,
  `calculation_method` varchar(255) DEFAULT NULL,
  `unit` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`indicator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_indicator_assignments` (
  `assignment_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `indicator_id` int(11) NOT NULL,
  `allocation_time` datetime NOT NULL,
  PRIMARY KEY (`assignment_id`),
  CONSTRAINT `user_indicator_assignments_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_indicator_assignments_ibfk_2` FOREIGN KEY (`indicator_id`) REFERENCES `indicators` (`indicator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `indicator_data_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `indicator_id` int(11) NOT NULL,
  `submission_time` datetime NOT NULL,
  `remarks` text DEFAULT NULL,
  `improvement_over_last` decimal(15,4) DEFAULT NULL,
  `recorded_value` decimal(15,4) NOT NULL,
  `review_status` varchar(50) DEFAULT 'pending',
  `reviewed_at` datetime DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `indicator_data_records_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `indicator_data_records_ibfk_2` FOREIGN KEY (`indicator_id`) REFERENCES `indicators` (`indicator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
