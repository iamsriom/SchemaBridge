-- Schema for database: rschema_0121

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `full_name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password_hash` varchar(255) NOT NULL,
  `registration_date` datetime NOT NULL,
  `last_login_date` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `username` (`username`),
  UNIQUE KEY `email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authenticated_users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `full_name` varchar(255) NOT NULL,
  `ring` int(11) NOT NULL,
  `authentication_time` datetime NOT NULL,
  `signature` text DEFAULT NULL,
  `issuer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_migrations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `old_user_id` int(11) NOT NULL,
  `new_user_id` int(11) NOT NULL,
  `migration_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `user_migrations_ibfk_1` FOREIGN KEY (`new_user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authenticated_consumers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `type` varchar(100) NOT NULL,
  `user_id` int(11) NOT NULL,
  `config` text DEFAULT NULL,
  `permissions` text DEFAULT NULL,
  `signature` text DEFAULT NULL,
  `issuer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `authenticated_consumers_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sessions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `session_id` varchar(255) NOT NULL,
  `consumer_id` int(11) NOT NULL,
  `expiration_time` datetime NOT NULL,
  `mfa_status` tinyint(1) NOT NULL DEFAULT 0,
  `signature` text DEFAULT NULL,
  `issuer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `session_id` (`session_id`),
  CONSTRAINT `sessions_ibfk_1` FOREIGN KEY (`consumer_id`) REFERENCES `authenticated_consumers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `services` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `service_name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `url` varchar(255) DEFAULT NULL,
  `config` text DEFAULT NULL,
  `last_heartbeat_time` datetime DEFAULT NULL,
  `monitoring_status` varchar(100) DEFAULT NULL,
  `signature` text DEFAULT NULL,
  `issuer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `worker_nodes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `node_name` varchar(255) NOT NULL,
  `status` varchar(100) NOT NULL,
  `service_id` int(11) NOT NULL,
  `operating_system` varchar(255) DEFAULT NULL,
  `architecture` varchar(100) DEFAULT NULL,
  `last_heartbeat_time` datetime DEFAULT NULL,
  `signature` text DEFAULT NULL,
  `issuer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `worker_nodes_ibfk_1` FOREIGN KEY (`service_id`) REFERENCES `services` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
