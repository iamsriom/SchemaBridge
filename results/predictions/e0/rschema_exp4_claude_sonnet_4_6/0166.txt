-- Schema for database: rschema_0166

CREATE TABLE `patients` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` varchar(50) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `medical_records` text DEFAULT NULL,
  PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctors` (
  `doctor_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`doctor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_images` (
  `image_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `image_type` varchar(100) NOT NULL,
  `film_time` datetime NOT NULL,
  `image_data` longblob DEFAULT NULL,
  `image_resolution` varchar(100) DEFAULT NULL,
  `image_enhancement_information` text DEFAULT NULL,
  PRIMARY KEY (`image_id`),
  CONSTRAINT `medical_images_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patients` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `image_operations` (
  `operation_id` int(11) NOT NULL AUTO_INCREMENT,
  `image_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `operation_content` text DEFAULT NULL,
  `operation_time` datetime NOT NULL,
  PRIMARY KEY (`operation_id`),
  CONSTRAINT `image_operations_ibfk_1` FOREIGN KEY (`image_id`) REFERENCES `medical_images` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `image_operations_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `image_uploads` (
  `upload_id` int(11) NOT NULL AUTO_INCREMENT,
  `image_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `upload_time` datetime NOT NULL,
  `server_path` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`upload_id`),
  CONSTRAINT `image_uploads_ibfk_1` FOREIGN KEY (`image_id`) REFERENCES `medical_images` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `image_uploads_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `image_shares` (
  `share_id` int(11) NOT NULL AUTO_INCREMENT,
  `image_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `sharing_time` datetime NOT NULL,
  `target_institution` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`share_id`),
  CONSTRAINT `image_shares_ibfk_1` FOREIGN KEY (`image_id`) REFERENCES `medical_images` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `image_shares_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
