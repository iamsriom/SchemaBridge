CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `equipment_type` varchar(100) NOT NULL,
  `manufacturer` varchar(255) DEFAULT NULL,
  `installation_date` datetime DEFAULT NULL,
  `health_indicator` decimal(5,2) DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `recorded_at` datetime NOT NULL,
  `temperature` decimal(8,2) DEFAULT NULL,
  `vibration` decimal(8,4) DEFAULT NULL,
  `voltage` decimal(8,2) DEFAULT NULL,
  `current` decimal(8,2) DEFAULT NULL,
  `pressure` decimal(8,2) DEFAULT NULL,
  `humidity` decimal(5,2) DEFAULT NULL,
  `operational_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`state_id`),
  CONSTRAINT `operational_state_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
