-- Schema for database: rschema_0171

CREATE TABLE `parking_space` (
  `space_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_number` varchar(50) NOT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'Vacant',
  PRIMARY KEY (`space_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `entrance_exit` (
  `gate_id` int(11) NOT NULL AUTO_INCREMENT,
  `gate_number` varchar(50) NOT NULL,
  `location` varchar(100) NOT NULL,
  PRIMARY KEY (`gate_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `ticket_number` varchar(100) NOT NULL,
  `vehicle_model` varchar(100) NOT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `space_id` int(11) NOT NULL,
  `entrance_gate_id` int(11) NOT NULL,
  `exit_gate_id` int(11) DEFAULT NULL,
  `entry_time` datetime NOT NULL,
  `exit_time` datetime DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `parking_record_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_2` FOREIGN KEY (`space_id`) REFERENCES `parking_space` (`space_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_3` FOREIGN KEY (`entrance_gate_id`) REFERENCES `entrance_exit` (`gate_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_4` FOREIGN KEY (`exit_gate_id`) REFERENCES `entrance_exit` (`gate_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
