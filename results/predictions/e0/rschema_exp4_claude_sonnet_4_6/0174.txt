-- Schema for database: rschema_0174

CREATE TABLE `patient` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_record_number` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `id_card_number` varchar(18) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `admission_time` datetime NOT NULL,
  PRIMARY KEY (`patient_id`),
  UNIQUE KEY `medical_record_number` (`medical_record_number`),
  UNIQUE KEY `id_card_number` (`id_card_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_sign_type` (
  `vital_sign_type_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `unit` varchar(50) DEFAULT NULL,
  `normal_lower_limit` decimal(10,2) DEFAULT NULL,
  `normal_upper_limit` decimal(10,2) DEFAULT NULL,
  `description` varchar(255) DEFAULT NULL,
  `updated_by` int(11) DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`vital_sign_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor` (
  `doctor_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `department` varchar(100) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`doctor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `nurse` (
  `nurse_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `department` varchar(100) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`nurse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_sign_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `vital_sign_type_id` int(11) NOT NULL,
  `value` decimal(10,2) NOT NULL,
  `collected_at` datetime NOT NULL,
  `collector_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  KEY `patient_id` (`patient_id`),
  KEY `vital_sign_type_id` (`vital_sign_type_id`),
  CONSTRAINT `vital_sign_record_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `vital_sign_record_ibfk_2` FOREIGN KEY (`vital_sign_type_id`) REFERENCES `vital_sign_type` (`vital_sign_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_sign_log` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `record_id` int(11) NOT NULL,
  `patient_id` int(11) NOT NULL,
  `vital_sign_type_id` int(11) NOT NULL,
  `formatted_value` varchar(255) NOT NULL,
  `logged_at` datetime NOT NULL,
  `is_abnormal` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`log_id`),
  KEY `record_id` (`record_id`),
  KEY `patient_id` (`patient_id`),
  KEY `vital_sign_type_id` (`vital_sign_type_id`),
  CONSTRAINT `vital_sign_log_ibfk_1` FOREIGN KEY (`record_id`) REFERENCES `vital_sign_record` (`record_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `vital_sign_log_ibfk_2` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `vital_sign_log_ibfk_3` FOREIGN KEY (`vital_sign_type_id`) REFERENCES `vital_sign_type` (`vital_sign_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warning_message` (
  `warning_id` int(11) NOT NULL AUTO_INCREMENT,
  `record_id` int(11) NOT NULL,
  `patient_id` int(11) NOT NULL,
  `vital_sign_type_id` int(11) NOT NULL,
  `message` varchar(500) NOT NULL,
  `warning_time` datetime NOT NULL,
  `is_resolved` tinyint(1) NOT NULL DEFAULT 0,

  KEY `vital_sign_type_id` (`vital_sign_type_id`),
  CONSTRAINT `warning_message_ibfk_1` FOREIGN KEY (`record_id`) REFERENCES `vital_sign_record` (`record_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `warning_message_ibfk_2` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `warning_message_ibfk_3` FOREIGN KEY (`vital_sign_type_id`) REFERENCES `vital_sign_type` (`vital_sign_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_record` (
  `medical_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `record_id` int(11) DEFAULT NULL,
  `doctor_id` int(11) DEFAULT NULL,
  `diagnosis_time` datetime NOT NULL,
  `diagnosis_content` text NOT NULL,
  PRIMARY KEY (`medical_record_id`),
  KEY `patient_id` (`patient_id`),
  KEY `record_id` (`record_id`),
  KEY `doctor_id` (`doctor_id`),
  CONSTRAINT `medical_record_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `medical_record_ibfk_2` FOREIGN KEY (`record_id`) REFERENCES `vital_sign_record` (`record_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `medical_record_ibfk_3` FOREIGN KEY (`doctor_id`) REFERENCES `doctor` (`doctor_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `treatment_recommendation` (
  `treatment_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `medical_record_id` int(11) DEFAULT NULL,
  `record_id` int(11) DEFAULT NULL,
  `recommendation_content` text NOT NULL,
  `generated_at` datetime NOT NULL,
  PRIMARY KEY (`treatment_id`),
  KEY `patient_id` (`patient_id`),
  KEY `medical_record_id` (`medical_record_id`),
  KEY `record_id` (`record_id`),
  CONSTRAINT `treatment_recommendation_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `treatment_recommendation_ibfk_2` FOREIGN KEY (`medical_record_id`) REFERENCES `medical_record` (`medical_record_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `treatment_recommendation_ibfk_3` FOREIGN KEY (`record_id`) REFERENCES `vital_sign_record` (`record_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
