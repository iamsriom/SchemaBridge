-- Schema for database: rschema_0177

CREATE TABLE `abstract_file` (
  `FileID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Type` varchar(50) NOT NULL,
  `CreateTime` datetime NOT NULL,
  `UpdateTime` datetime NOT NULL,
  PRIMARY KEY (`FileID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `folder` (
  `FolderID` int(11) NOT NULL,
  `FileID` int(11) NOT NULL,
  `ChildList` varchar(1024) DEFAULT NULL,
  PRIMARY KEY (`FolderID`),
  CONSTRAINT `folder_ibfk_1` FOREIGN KEY (`FileID`) REFERENCES `abstract_file` (`FileID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `file` (
  `FileID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Type` varchar(50) NOT NULL,
  `FileSize` bigint(20) NOT NULL,
  `CreateTime` datetime NOT NULL,
  `UpdateTime` datetime NOT NULL,
  PRIMARY KEY (`FileID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
