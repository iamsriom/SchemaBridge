-- Schema for database: rschema_0183

CREATE TABLE `fishing_tool` (
  `tool_id` int(11) NOT NULL AUTO_INCREMENT,
  `tool_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  PRIMARY KEY (`tool_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tool_usage` (
  `usage_id` int(11) NOT NULL AUTO_INCREMENT,
  `tool_id` int(11) NOT NULL,
  `usage_time` datetime NOT NULL,
  `user` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`usage_id`),
  CONSTRAINT `tool_usage_ibfk_1` FOREIGN KEY (`tool_id`) REFERENCES `fishing_tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tool_maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `tool_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintenance_person` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `tool_maintenance_ibfk_1` FOREIGN KEY (`tool_id`) REFERENCES `fishing_tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tool_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `tool_id` int(11) NOT NULL,
  `usage_frequency` int(11) NOT NULL DEFAULT '0',
  `maintenance_count` int(11) NOT NULL DEFAULT '0',
  PRIMARY KEY (`performance_id`),
  CONSTRAINT `tool_performance_ibfk_1` FOREIGN KEY (`tool_id`) REFERENCES `fishing_tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
