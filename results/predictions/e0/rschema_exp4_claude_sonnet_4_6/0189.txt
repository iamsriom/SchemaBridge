-- Schema for database: rschema_0189

CREATE TABLE `repair_station` (
  `station_id` int(11) NOT NULL AUTO_INCREMENT,
  `station_name` varchar(255) NOT NULL,
  `station_address` varchar(255) DEFAULT NULL,
  `station_phone` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technician` (
  `technician_id` int(11) NOT NULL AUTO_INCREMENT,
  `station_id` int(11) NOT NULL,
  `technician_name` varchar(255) NOT NULL,
  `technician_contact_number` varchar(50) DEFAULT NULL,
  `technician_skill_level` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`technician_id`),
  CONSTRAINT `technician_ibfk_1` FOREIGN KEY (`station_id`) REFERENCES `repair_station` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate_number` varchar(50) NOT NULL,
  `vehicle_model` varchar(255) DEFAULT NULL,
  `vehicle_owner_name` varchar(255) DEFAULT NULL,
  `owner_contact_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`),
  UNIQUE KEY `license_plate_number` (`license_plate_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `station_id` int(11) NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(255) DEFAULT NULL,
  `repair_duration` decimal(10,2) DEFAULT NULL,
  `repair_cost` decimal(10,2) DEFAULT NULL,
  `repair_effect` varchar(255) DEFAULT NULL,
  `owner_evaluation` varchar(500) DEFAULT NULL,
  `owner_satisfaction_score` int(11) DEFAULT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `repair_task_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_task_ibfk_2` FOREIGN KEY (`station_id`) REFERENCES `repair_station` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_technician` (
  `task_technician_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_id` int(11) NOT NULL,
  `technician_id` int(11) NOT NULL,
  PRIMARY KEY (`task_technician_id`),
  CONSTRAINT `task_technician_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `repair_task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_technician_ibfk_2` FOREIGN KEY (`technician_id`) REFERENCES `technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technician_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `technician_id` int(11) NOT NULL,
  `evaluation_period` varchar(100) DEFAULT NULL,
  `tasks_completed` int(11) DEFAULT NULL,
  `average_satisfaction_score` decimal(5,2) DEFAULT NULL,
  `performance_rating` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`performance_id`),
  CONSTRAINT `technician_performance_ibfk_1` FOREIGN KEY (`technician_id`) REFERENCES `technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
