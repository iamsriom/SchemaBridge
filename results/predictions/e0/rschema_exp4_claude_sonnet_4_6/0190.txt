-- Schema for database: rschema_0190

CREATE TABLE `workstation` (
  `workstation_id` int(11) NOT NULL AUTO_INCREMENT,
  `workstation_name` varchar(100) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`workstation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technician` (
  `technician_id` int(11) NOT NULL AUTO_INCREMENT,
  `workstation_id` int(11) NOT NULL,
  `technician_name` varchar(100) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(150) DEFAULT NULL,
  `hire_date` datetime DEFAULT NULL,
  `completed_tasks` int(11) NOT NULL DEFAULT '0',
  `avg_satisfaction_rating` decimal(3,2) DEFAULT NULL,
  PRIMARY KEY (`technician_id`),
  CONSTRAINT `technician_ibfk_1` FOREIGN KEY (`workstation_id`) REFERENCES `workstation` (`workstation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(100) NOT NULL,
  `customer_phone` varchar(20) NOT NULL,
  `email` varchar(150) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `technician_id` int(11) NOT NULL,
  `workstation_id` int(11) NOT NULL,
  `product_name` varchar(150) NOT NULL,
  `model_number` varchar(100) DEFAULT NULL,
  `fault_description` text DEFAULT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(100) DEFAULT NULL,
  `repair_duration` decimal(6,2) DEFAULT NULL,
  `repair_cost` decimal(10,2) DEFAULT NULL,
  `repair_result` varchar(255) DEFAULT NULL,
  `status` varchar(50) NOT NULL DEFAULT 'pending',
  PRIMARY KEY (`task_id`),
  CONSTRAINT `repair_task_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_task_ibfk_2` FOREIGN KEY (`technician_id`) REFERENCES `technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_task_ibfk_3` FOREIGN KEY (`workstation_id`) REFERENCES `workstation` (`workstation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_feedback` (
  `feedback_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `satisfaction_rating` int(11) DEFAULT NULL,
  `feedback_comment` text DEFAULT NULL,
  `feedback_date` datetime DEFAULT NULL,
  PRIMARY KEY (`feedback_id`),
  CONSTRAINT `customer_feedback_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `repair_task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_feedback_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technician_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `technician_id` int(11) NOT NULL,
  `evaluation_period` varchar(50) NOT NULL,
  `completed_tasks_count` int(11) NOT NULL DEFAULT '0',
  `avg_customer_satisfaction` decimal(3,2) DEFAULT NULL,
  `evaluation_date` datetime DEFAULT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`performance_id`),
  CONSTRAINT `technician_performance_ibfk_1` FOREIGN KEY (`technician_id`) REFERENCES `technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
