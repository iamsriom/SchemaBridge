-- Schema for database: rschema_0199

CREATE TABLE `store` (
  `store_id` int(11) NOT NULL AUTO_INCREMENT,
  `store_number` varchar(50) NOT NULL,
  `store_name` varchar(255) NOT NULL,
  `address` varchar(500) DEFAULT NULL,
  `manager` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`store_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_number` varchar(50) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `model` varchar(100) DEFAULT NULL,
  `barcode` varchar(100) DEFAULT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `daily_sales` (
  `sales_id` int(11) NOT NULL AUTO_INCREMENT,
  `store_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_amount` decimal(10,2) NOT NULL,
  `salesperson` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`sales_id`),
  CONSTRAINT `daily_sales_ibfk_1` FOREIGN KEY (`store_id`) REFERENCES `store` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `daily_sales_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_appraisal` (
  `appraisal_id` int(11) NOT NULL AUTO_INCREMENT,
  `store_id` int(11) NOT NULL,
  `appraisal_date` datetime NOT NULL,
  `sales_score` decimal(5,2) DEFAULT NULL,
  `customer_satisfaction` decimal(5,2) DEFAULT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`appraisal_id`),
  CONSTRAINT `performance_appraisal_ibfk_1` FOREIGN KEY (`store_id`) REFERENCES `store` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
