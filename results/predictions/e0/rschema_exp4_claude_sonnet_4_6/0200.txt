-- Schema for database: rschema_0200

CREATE TABLE `customer` (
  `customer_no` varchar(20) NOT NULL,
  `customer_name` varchar(100) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_person` varchar(50) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`customer_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `order_no` varchar(20) NOT NULL,
  `customer_no` varchar(20) NOT NULL,
  `order_time` datetime NOT NULL,
  `product_no` varchar(20) NOT NULL,
  `quantity` int(11) NOT NULL,
  `total_price` decimal(15,2) NOT NULL,
  PRIMARY KEY (`order_no`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`customer_no`) REFERENCES `customer` (`customer_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_rating` (
  `rating_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_no` varchar(20) NOT NULL,
  `historical_order_count` int(11) NOT NULL DEFAULT 0,
  `total_order_amount` decimal(15,2) NOT NULL DEFAULT 0.00,
  `credit_score` decimal(5,2) DEFAULT NULL,
  `rating_time` datetime NOT NULL,
  PRIMARY KEY (`rating_id`),
  CONSTRAINT `credit_rating_ibfk_1` FOREIGN KEY (`customer_no`) REFERENCES `customer` (`customer_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
