-- Schema for database: rschema_0203

CREATE TABLE `store` (
  `store_id` int(11) NOT NULL AUTO_INCREMENT,
  `store_number` varchar(50) NOT NULL,
  `store_name` varchar(255) NOT NULL,
  `store_address` varchar(500) NOT NULL,
  PRIMARY KEY (`store_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_number` varchar(50) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `product_model` varchar(100) DEFAULT NULL,
  `product_barcode` varchar(100) NOT NULL,
  `product_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `promotion_activity` (
  `activity_id` int(11) NOT NULL AUTO_INCREMENT,
  `activity_number` varchar(50) NOT NULL,
  `activity_name` varchar(255) NOT NULL,
  `activity_start_time` datetime NOT NULL,
  `activity_end_time` datetime NOT NULL,
  `discount` decimal(5,2) NOT NULL,
  PRIMARY KEY (`activity_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `activity_store` (
  `activity_store_id` int(11) NOT NULL AUTO_INCREMENT,
  `activity_id` int(11) NOT NULL,
  `store_id` int(11) NOT NULL,
  PRIMARY KEY (`activity_store_id`),
  CONSTRAINT `activity_store_ibfk_1` FOREIGN KEY (`activity_id`) REFERENCES `promotion_activity` (`activity_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `activity_store_ibfk_2` FOREIGN KEY (`store_id`) REFERENCES `store` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `activity_product` (
  `activity_product_id` int(11) NOT NULL AUTO_INCREMENT,
  `activity_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  PRIMARY KEY (`activity_product_id`),
  CONSTRAINT `activity_product_ibfk_1` FOREIGN KEY (`activity_id`) REFERENCES `promotion_activity` (`activity_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `activity_product_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `promotion_effect` (
  `effect_id` int(11) NOT NULL AUTO_INCREMENT,
  `activity_id` int(11) NOT NULL,
  `store_id` int(11) NOT NULL,
  `sales_amount` decimal(15,2) NOT NULL,
  `sales_volume` int(11) NOT NULL,
  `participating_members` int(11) NOT NULL,
  PRIMARY KEY (`effect_id`),
  CONSTRAINT `promotion_effect_ibfk_1` FOREIGN KEY (`activity_id`) REFERENCES `promotion_activity` (`activity_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `promotion_effect_ibfk_2` FOREIGN KEY (`store_id`) REFERENCES `store` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `promotion_evaluation` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `activity_id` int(11) NOT NULL,
  `store_id` int(11) NOT NULL,
  `sales_growth_rate` decimal(8,4) DEFAULT NULL,
  `customer_participation` decimal(8,4) DEFAULT NULL,
  `evaluation_time` datetime NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  CONSTRAINT `promotion_evaluation_ibfk_1` FOREIGN KEY (`activity_id`) REFERENCES `promotion_activity` (`activity_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `promotion_evaluation_ibfk_2` FOREIGN KEY (`store_id`) REFERENCES `store` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
