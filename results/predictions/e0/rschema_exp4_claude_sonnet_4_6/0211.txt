-- Schema for database: rschema_0211

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  `client_name` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultant` (
  `consultant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` datetime DEFAULT NULL,
  `gender` char(6) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `professional_field` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`consultant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consulting_service_record` (
  `service_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `consultant_id` int(11) NOT NULL,
  `service_date` datetime NOT NULL,
  `service_content` varchar(1000) DEFAULT NULL,
  PRIMARY KEY (`service_id`),
  CONSTRAINT `consulting_service_record_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `consulting_service_record_ibfk_2` FOREIGN KEY (`consultant_id`) REFERENCES `consultant` (`consultant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
