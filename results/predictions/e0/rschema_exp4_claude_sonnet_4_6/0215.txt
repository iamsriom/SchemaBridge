-- Schema for database: rschema_0215

CREATE TABLE `mine` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_administrator` (
  `admin_id` int(11) NOT NULL AUTO_INCREMENT,
  `admin_name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `position` varchar(255) NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`admin_id`),
  CONSTRAINT `material_administrator_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_purchase` (
  `purchase_id` int(11) NOT NULL AUTO_INCREMENT,
  `purchase_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `supplier` varchar(255) NOT NULL,
  `purchase_cost` decimal(15,2) NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`purchase_id`),
  CONSTRAINT `material_purchase_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_distribution` (
  `distribution_id` int(11) NOT NULL AUTO_INCREMENT,
  `distribution_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `distributor_id` int(11) NOT NULL,
  `recipient` varchar(255) NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`distribution_id`),
  CONSTRAINT `material_distribution_ibfk_1` FOREIGN KEY (`distributor_id`) REFERENCES `material_administrator` (`admin_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_distribution_ibfk_2` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
