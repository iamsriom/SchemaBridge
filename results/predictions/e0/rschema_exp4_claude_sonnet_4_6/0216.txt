-- Schema for database: rschema_0216

CREATE TABLE `card_type` (
  `card_type_code` varchar(20) NOT NULL,
  `card_type_name` varchar(100) NOT NULL,
  `annual_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`card_type_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `id_number` varchar(50) NOT NULL,
  `contact_number` varchar(20) NOT NULL,
  PRIMARY KEY (`customer_id`),
  UNIQUE KEY `id_number` (`id_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_card` (
  `card_number` varchar(20) NOT NULL,
  `card_type_code` varchar(20) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `account_opening_date` datetime NOT NULL,
  `credit_limit` decimal(15,2) NOT NULL,
  PRIMARY KEY (`card_number`),
  CONSTRAINT `credit_card_ibfk_1` FOREIGN KEY (`card_type_code`) REFERENCES `card_type` (`card_type_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `credit_card_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `card_number` varchar(20) NOT NULL,
  `amount` decimal(15,2) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_location` varchar(255) NOT NULL,
  `transaction_type` varchar(50) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  CONSTRAINT `transaction_ibfk_1` FOREIGN KEY (`card_number`) REFERENCES `credit_card` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repayment` (
  `repayment_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `card_number` varchar(20) NOT NULL,
  `repayment_amount` decimal(15,2) NOT NULL,
  `repayment_time` datetime NOT NULL,
  PRIMARY KEY (`repayment_id`),
  CONSTRAINT `repayment_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repayment_ibfk_2` FOREIGN KEY (`card_number`) REFERENCES `credit_card` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
