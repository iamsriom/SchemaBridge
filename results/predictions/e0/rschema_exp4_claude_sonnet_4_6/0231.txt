-- Schema for database: rschema_0231

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone` varchar(50) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `experience_records` (
  `experience_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` text DEFAULT NULL,
  `code_content` text DEFAULT NULL,
  `creator_user_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `modification_time` datetime DEFAULT NULL,
  `status` char(1) NOT NULL DEFAULT 'F',
  PRIMARY KEY (`experience_id`),
  CONSTRAINT `experience_records_ibfk_1` FOREIGN KEY (`creator_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=latin1;
