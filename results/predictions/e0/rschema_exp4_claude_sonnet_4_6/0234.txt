  KEY `idx_primary_category` (`primary_category`),

  KEY `idx_primary_category_code` (`primary_category_code`),
  KEY `idx_secondary_category_code` (`secondary_category_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
