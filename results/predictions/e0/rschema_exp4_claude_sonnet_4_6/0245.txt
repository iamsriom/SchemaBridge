-- Schema for database: rschema_0245

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL,
  `password` varchar(255) NOT NULL,
  `nickname` varchar(50) DEFAULT NULL,
  `avatar` varchar(255) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `register_time` datetime NOT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `status` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `category` (
  `category_id` int(11) NOT NULL AUTO_INCREMENT,
  `category_name` varchar(50) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `sort_order` int(11) DEFAULT NULL,
  PRIMARY KEY (`category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `novel` (
  `novel_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(100) NOT NULL,
  `author` varchar(50) NOT NULL,
  `category_id` int(11) DEFAULT NULL,
  `cover_image` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `total_chapters` int(11) DEFAULT '0',
  `total_words` int(11) DEFAULT '0',
  `status` tinyint(1) NOT NULL DEFAULT '1' COMMENT '0:ongoing,1:completed',
  `publish_time` datetime DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  `view_count` int(11) DEFAULT '0',
  `collect_count` int(11) DEFAULT '0',
  `rank_score` decimal(10,2) DEFAULT '0.00',
  PRIMARY KEY (`novel_id`),
  CONSTRAINT `novel_ibfk_1` FOREIGN KEY (`category_id`) REFERENCES `category` (`category_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chapter` (
  `chapter_id` int(11) NOT NULL AUTO_INCREMENT,
  `novel_id` int(11) NOT NULL,
  `chapter_number` int(11) NOT NULL,
  `chapter_title` varchar(100) NOT NULL,
  `content` longtext DEFAULT NULL,
  `word_count` int(11) DEFAULT '0',
  `publish_time` datetime DEFAULT NULL,
  `is_free` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`chapter_id`),
  CONSTRAINT `chapter_ibfk_1` FOREIGN KEY (`novel_id`) REFERENCES `novel` (`novel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bookshelf` (
  `bookshelf_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `novel_id` int(11) NOT NULL,
  `add_time` datetime NOT NULL,
  `last_read_chapter_id` int(11) DEFAULT NULL,
  `last_read_time` datetime DEFAULT NULL,
  PRIMARY KEY (`bookshelf_id`),
  CONSTRAINT `bookshelf_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bookshelf_ibfk_2` FOREIGN KEY (`novel_id`) REFERENCES `novel` (`novel_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bookshelf_ibfk_3` FOREIGN KEY (`last_read_chapter_id`) REFERENCES `chapter` (`chapter_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `read_history` (
  `history_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `novel_id` int(11) NOT NULL,
  `chapter_id` int(11) NOT NULL,
  `read_time` datetime NOT NULL,
  `read_progress` decimal(5,2) DEFAULT '0.00' COMMENT 'percentage of chapter read',
  PRIMARY KEY (`history_id`),
  CONSTRAINT `read_history_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `read_history_ibfk_2` FOREIGN KEY (`novel_id`) REFERENCES `novel` (`novel_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `read_history_ibfk_3` FOREIGN KEY (`chapter_id`) REFERENCES `chapter` (`chapter_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review` (
  `review_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `novel_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `rating` tinyint(1) DEFAULT NULL COMMENT '1-5 stars',
  `like_count` int(11) DEFAULT '0',
  `create_time` datetime NOT NULL,
  `update_time` datetime DEFAULT NULL,
  `status` tinyint(1) NOT NULL DEFAULT '1' COMMENT '0:hidden,1:visible',
  PRIMARY KEY (`review_id`),
  CONSTRAINT `review_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `review_ibfk_2` FOREIGN KEY (`novel_id`) REFERENCES `novel` (`novel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ranking` (
  `ranking_id` int(11) NOT NULL AUTO_INCREMENT,
  `ranking_type` varchar(50) NOT NULL COMMENT 'e.g. weekly,monthly,all_time',
  `novel_id` int(11) NOT NULL,
  `rank_position` int(11) NOT NULL,
  `score` decimal(10,2) DEFAULT '0.00',
  `update_time` datetime NOT NULL,
  PRIMARY KEY (`ranking_id`),
  CONSTRAINT `ranking_ibfk_1` FOREIGN KEY (`novel_id`) REFERENCES `novel` (`novel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `search_log` (
  `search_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) DEFAULT NULL,
  `keyword` varchar(100) NOT NULL,
  `search_time` datetime NOT NULL,
  `result_count` int(11) DEFAULT '0',
  PRIMARY KEY (`search_id`),
  CONSTRAINT `search_log_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
