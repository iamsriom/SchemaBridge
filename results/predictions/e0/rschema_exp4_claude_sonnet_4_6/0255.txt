) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `report_id` int(11) NOT NULL AUTO_INCREMENT,

  `report_type` varchar(100) NOT NULL,
  `report_date` datetime NOT NULL,

  `completion_status` varchar(50) DEFAULT NULL,
  `metal_recovery_rate` decimal(10,4) DEFAULT NULL,
  `grade` decimal(10,4) DEFAULT NULL,

  CONSTRAINT `production_report_ibfk_1` FOREIGN KEY (`process_id`) REFERENCES `process` (`process_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
