-- Schema for database: rschema_0257

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `customer_type` varchar(50) NOT NULL,
  `id_number` varchar(50) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `booking_date` datetime DEFAULT NULL,
  `check_in_time` datetime DEFAULT NULL,
  `check_out_time` datetime DEFAULT NULL,
  `discount_rate` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `flight_id` int(11) NOT NULL AUTO_INCREMENT,
  `flight_name` varchar(255) NOT NULL,
  `flight_date` datetime NOT NULL,
  `airline_name` varchar(255) NOT NULL,
  `departure_location` varchar(255) NOT NULL,
  `departure_time` datetime NOT NULL,
  `destination` varchar(255) NOT NULL,
  `arrival_time` datetime NOT NULL,
  PRIMARY KEY (`flight_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_booking` (
  `booking_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `flight_id` int(11) NOT NULL,
  `discount_rate` decimal(5,2) DEFAULT NULL,
  `ticket_order_number` varchar(100) NOT NULL,
  PRIMARY KEY (`booking_id`),
  CONSTRAINT `flight_booking_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_booking_ibfk_2` FOREIGN KEY (`flight_id`) REFERENCES `flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_order` (
  `order_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `flight_id` int(11) NOT NULL,
  `ticket_order_number` varchar(100) NOT NULL,
  `discount_rate` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `customer_order_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_order_ibfk_2` FOREIGN KEY (`flight_id`) REFERENCES `flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
