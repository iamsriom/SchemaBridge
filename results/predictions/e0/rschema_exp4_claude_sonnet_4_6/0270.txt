-- Schema for database: rschema_0270

CREATE TABLE `program` (
  `program_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`program_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_staff` (
  `staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(100) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviewer` (
  `reviewer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`reviewer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production` (
  `production_id` int(11) NOT NULL AUTO_INCREMENT,
  `program_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  `reviewer_id` int(11) NOT NULL,
  `review_result` varchar(255) DEFAULT NULL,
  `review_time` datetime DEFAULT NULL,
  PRIMARY KEY (`production_id`),
  CONSTRAINT `production_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `program` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_ibfk_2` FOREIGN KEY (`reviewer_id`) REFERENCES `reviewer` (`reviewer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_staff_assignment` (
  `assignment_id` int(11) NOT NULL AUTO_INCREMENT,
  `production_id` int(11) NOT NULL,
  `staff_id` int(11) NOT NULL,
  `assignment_role` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`assignment_id`),
  CONSTRAINT `production_staff_assignment_ibfk_1` FOREIGN KEY (`production_id`) REFERENCES `production` (`production_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_staff_assignment_ibfk_2` FOREIGN KEY (`staff_id`) REFERENCES `production_staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
