-- Schema for database: rschema_0275

CREATE TABLE `ecosystem` (
  `system_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `current_status` varchar(100) NOT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operator` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `system_id` int(11) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `temperature` decimal(5,2) DEFAULT NULL,
  `humidity` decimal(5,2) DEFAULT NULL,
  `soil_moisture` decimal(5,2) DEFAULT NULL,
  `plant_growth` decimal(5,2) DEFAULT NULL,
  `employee_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `monitoring_record_ibfk_1` FOREIGN KEY (`system_id`) REFERENCES `ecosystem` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `operator` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
