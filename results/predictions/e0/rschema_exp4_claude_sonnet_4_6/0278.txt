-- Schema for database: rschema_0278

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `project_manager` varchar(100) NOT NULL,
  `construction_start_date` date NOT NULL,
  `expected_completion_date` date NOT NULL,
  `actual_completion_date` date DEFAULT NULL,
  `project_budget` decimal(15,2) NOT NULL,
  `project_location` varchar(500) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspection` (
  `inspection_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `inspection_date` date NOT NULL,
  `inspection_content` text NOT NULL,
  `inspection_result` varchar(255) NOT NULL,
  `inspector` varchar(100) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  CONSTRAINT `safety_inspection_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tracking_record` (
  `tracking_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `tracking_date` date NOT NULL,
  `tracking_content` text NOT NULL,
  `tracking_result` varchar(255) NOT NULL,
  PRIMARY KEY (`tracking_record_id`),
  CONSTRAINT `tracking_record_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
