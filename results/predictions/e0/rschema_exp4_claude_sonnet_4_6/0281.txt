-- Schema for database: rschema_0281

CREATE TABLE `exhibition` (
  `exhibition_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`exhibition_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `curator` (
  `curator_id` int(11) NOT NULL AUTO_INCREMENT,
  `exhibition_id` int(11) NOT NULL,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone` varchar(50) DEFAULT NULL,
  `bio` text DEFAULT NULL,
  PRIMARY KEY (`curator_id`),
  CONSTRAINT `curator_ibfk_1` FOREIGN KEY (`exhibition_id`) REFERENCES `exhibition` (`exhibition_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exhibit` (
  `exhibit_id` int(11) NOT NULL AUTO_INCREMENT,
  `exhibition_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `artist_name` varchar(255) DEFAULT NULL,
  `medium` varchar(255) DEFAULT NULL,
  `year_created` int(11) DEFAULT NULL,
  PRIMARY KEY (`exhibit_id`),
  CONSTRAINT `exhibit_ibfk_1` FOREIGN KEY (`exhibition_id`) REFERENCES `exhibition` (`exhibition_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
