-- Schema for database: rschema_0287

CREATE TABLE `park` (
  `park_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  `responsible_person` varchar(100) NOT NULL,
  PRIMARY KEY (`park_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plant_species` (
  `species_id` int(11) NOT NULL AUTO_INCREMENT,
  `species_name` varchar(255) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`species_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plant_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `park_id` int(11) NOT NULL,
  `species_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `growth_status` varchar(255) DEFAULT NULL,
  `record_time` datetime NOT NULL,
  `recorder` varchar(100) NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `plant_record_ibfk_1` FOREIGN KEY (`park_id`) REFERENCES `park` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `plant_record_ibfk_2` FOREIGN KEY (`species_id`) REFERENCES `plant_species` (`species_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_activity` (
  `activity_id` int(11) NOT NULL AUTO_INCREMENT,
  `park_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(500) NOT NULL,
  `maintenance_personnel` varchar(100) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`activity_id`),
  CONSTRAINT `maintenance_activity_ibfk_1` FOREIGN KEY (`park_id`) REFERENCES `park` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
