-- Schema for database: rschema_0297
CREATE TABLE `document_type` (
  `document_type_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `version` varchar(50) NOT NULL,
  PRIMARY KEY (`document_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status_category` (
  `status_category_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_type_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `version` varchar(50) NOT NULL,
  PRIMARY KEY (`status_category_id`),
  CONSTRAINT `status_category_ibfk_1` FOREIGN KEY (`document_type_id`) REFERENCES `document_type` (`document_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status` (
  `status_id` int(11) NOT NULL AUTO_INCREMENT,
  `status_category_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `sequence_number` int(11) NOT NULL,
  PRIMARY KEY (`status_id`),
  CONSTRAINT `status_ibfk_1` FOREIGN KEY (`status_category_id`) REFERENCES `status_category` (`status_category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
