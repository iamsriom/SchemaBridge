-- Schema for database: rschema_0298

CREATE TABLE `department` (
  `dept_no` char(6) NOT NULL,
  `dept_name` varchar(255) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`dept_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_grade` (
  `grade_level` int(11) NOT NULL,
  `min_salary` decimal(10,2) NOT NULL,
  `max_salary` decimal(10,2) NOT NULL,
  PRIMARY KEY (`grade_level`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `emp_id` int(11) NOT NULL AUTO_INCREMENT,
  `emp_name` varchar(255) NOT NULL,
  `position` varchar(255) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `salary_grade` int(11) DEFAULT NULL,
  `supervisor_id` int(11) DEFAULT NULL,
  `dept_no` char(6) DEFAULT NULL,
  PRIMARY KEY (`emp_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`salary_grade`) REFERENCES `salary_grade` (`grade_level`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`supervisor_id`) REFERENCES `employee` (`emp_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_3` FOREIGN KEY (`dept_no`) REFERENCES `department` (`dept_no`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
