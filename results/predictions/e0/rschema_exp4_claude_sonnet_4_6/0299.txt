-- Schema for database: rschema_0299

CREATE TABLE `city` (
  `city_id` int(11) NOT NULL AUTO_INCREMENT,
  `city_name` varchar(255) NOT NULL,
  `longitude` decimal(9,6) NOT NULL,
  `latitude` decimal(9,6) NOT NULL,
  `country` varchar(100) NOT NULL,
  `population` int(11) DEFAULT NULL,
  `timezone` varchar(100) NOT NULL,
  PRIMARY KEY (`city_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather_type` (
  `weather_type_id` int(11) NOT NULL AUTO_INCREMENT,
  `main_description` varchar(255) NOT NULL,
  `detailed_description` varchar(255) NOT NULL,
  `icon` varchar(100) NOT NULL,
  PRIMARY KEY (`weather_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `daily_forecast` (
  `forecast_id` int(11) NOT NULL AUTO_INCREMENT,
  `city_id` int(11) NOT NULL,
  `weather_type_id` int(11) NOT NULL,
  `forecast_date` date NOT NULL,
  `sunrise` datetime NOT NULL,
  `sunset` datetime NOT NULL,
  `temp_morning` decimal(5,2) DEFAULT NULL,
  `temp_day` decimal(5,2) DEFAULT NULL,
  `temp_evening` decimal(5,2) DEFAULT NULL,
  `temp_night` decimal(5,2) DEFAULT NULL,
  `temp_min` decimal(5,2) DEFAULT NULL,
  `temp_max` decimal(5,2) DEFAULT NULL,
  `feels_like_morning` decimal(5,2) DEFAULT NULL,
  `feels_like_day` decimal(5,2) DEFAULT NULL,
  `feels_like_evening` decimal(5,2) DEFAULT NULL,
  `feels_like_night` decimal(5,2) DEFAULT NULL,
  `pressure` int(11) DEFAULT NULL,
  `humidity` int(11) DEFAULT NULL,
  `wind_speed` decimal(6,2) DEFAULT NULL,
  `wind_direction` int(11) DEFAULT NULL,
  `cloud_cover` int(11) DEFAULT NULL,
  `precipitation_probability` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`forecast_id`),
  CONSTRAINT `daily_forecast_ibfk_1` FOREIGN KEY (`city_id`) REFERENCES `city` (`city_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `daily_forecast_ibfk_2` FOREIGN KEY (`weather_type_id`) REFERENCES `weather_type` (`weather_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
