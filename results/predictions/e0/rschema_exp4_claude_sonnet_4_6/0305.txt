-- Schema for database: rschema_0305

CREATE TABLE `users` (
  `account_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `verification_status` tinyint(1) NOT NULL DEFAULT 0,
  `account_status` varchar(50) NOT NULL DEFAULT 'active',
  PRIMARY KEY (`account_id`),
  UNIQUE KEY `email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `follows` (
  `follow_id` int(11) NOT NULL AUTO_INCREMENT,
  `follower_account_id` int(11) NOT NULL,
  `followee_account_id` int(11) NOT NULL,
  PRIMARY KEY (`follow_id`),
  CONSTRAINT `follows_ibfk_1` FOREIGN KEY (`follower_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `follows_ibfk_2` FOREIGN KEY (`followee_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `posts` (
  `post_id` int(11) NOT NULL AUTO_INCREMENT,
  `publisher_account_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `publish_time` datetime NOT NULL,
  `attachment_link` varchar(512) DEFAULT NULL,
  PRIMARY KEY (`post_id`),
  CONSTRAINT `posts_ibfk_1` FOREIGN KEY (`publisher_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notifications` (
  `notification_id` int(11) NOT NULL AUTO_INCREMENT,
  `notification_type` varchar(100) NOT NULL,
  `recipient_account_id` int(11) NOT NULL,
  `related_post_id` int(11) DEFAULT NULL,
  `related_user_id` int(11) DEFAULT NULL,
  `notification_time` datetime NOT NULL,
  PRIMARY KEY (`notification_id`),
  CONSTRAINT `notifications_ibfk_1` FOREIGN KEY (`recipient_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `notifications_ibfk_2` FOREIGN KEY (`related_post_id`) REFERENCES `posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `notifications_ibfk_3` FOREIGN KEY (`related_user_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
