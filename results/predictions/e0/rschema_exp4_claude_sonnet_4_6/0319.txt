-- Schema for database: rschema_0319

CREATE TABLE `region` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `regional_code` varchar(50) NOT NULL,
  `city_code` varchar(20) DEFAULT NULL,
  `center_longitude` decimal(10,6) DEFAULT NULL,
  `center_latitude` decimal(10,6) DEFAULT NULL,
  `regional_level` int(11) DEFAULT NULL,
  `merged_name` varchar(255) DEFAULT NULL,
  `regional_name` varchar(100) NOT NULL,
  `parent_regional_id` int(11) DEFAULT NULL,
  `pinyin` varchar(255) DEFAULT NULL,
  `abbreviation` varchar(50) DEFAULT NULL,
  `postal_code` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `region_ibfk_1` FOREIGN KEY (`parent_regional_id`) REFERENCES `region` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
