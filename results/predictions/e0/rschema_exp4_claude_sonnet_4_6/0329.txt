-- Schema for database: rschema_0329

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `balance` decimal(15,2) NOT NULL DEFAULT '0.00',
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stocks` (
  `stock_code` char(6) NOT NULL,
  `stock_price` decimal(10,2) NOT NULL,
  `issue_date` datetime DEFAULT NULL,
  `company_name` varchar(255) NOT NULL,
  PRIMARY KEY (`stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_logs` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `log_date` datetime NOT NULL,
  `record_value` decimal(15,2) NOT NULL DEFAULT '0.00',
  PRIMARY KEY (`log_id`),
  CONSTRAINT `user_logs_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `portfolios` (
  `portfolio_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `stock_code` char(6) NOT NULL,
  `quantity` int(11) NOT NULL DEFAULT '0',
  `earnings` decimal(15,2) NOT NULL DEFAULT '0.00',
  PRIMARY KEY (`portfolio_id`),
  CONSTRAINT `portfolios_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `portfolios_ibfk_2` FOREIGN KEY (`stock_code`) REFERENCES `stocks` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hall_of_fame` (
  `fame_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `earnings` decimal(15,2) NOT NULL DEFAULT '0.00',
  `recorded_at` datetime DEFAULT NULL,
  PRIMARY KEY (`fame_id`),
  CONSTRAINT `hall_of_fame_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
