-- Schema for database: rschema_0335
CREATE TABLE `OSS_BUCKET` (
  `ID` varchar(64) NOT NULL,
  `NAME` varchar(255) NOT NULL,
  `CREATE_TIME` datetime DEFAULT NULL,
  `DESCRIPTION` varchar(500) DEFAULT NULL,
  `USER_ID` varchar(64) DEFAULT NULL,
  `TENANT_ID` varchar(64) DEFAULT NULL,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `OSS_ACCESS` (
  `ID` varchar(64) NOT NULL,
  `BUCKET_ID` varchar(64) NOT NULL,
  `TYPE` varchar(64) DEFAULT NULL,
  `ACCESS_KEY` varchar(255) DEFAULT NULL,
  `SECRET_KEY` varchar(255) DEFAULT NULL,
  `STATUS` varchar(32) DEFAULT NULL,
  `CREATE_TIME` datetime DEFAULT NULL,
  `DESCRIPTION` varchar(500) DEFAULT NULL,
  `USER_ID` varchar(64) DEFAULT NULL,
  `TENANT_ID` varchar(64) DEFAULT NULL,
  PRIMARY KEY (`ID`),
  CONSTRAINT `OSS_ACCESS_ibfk_1` FOREIGN KEY (`BUCKET_ID`) REFERENCES `OSS_BUCKET` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
