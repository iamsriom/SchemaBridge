-- Schema for database: rschema_0340

CREATE TABLE `application` (
  `app_id` int(11) NOT NULL AUTO_INCREMENT,
  `app_name` varchar(255) NOT NULL,
  PRIMARY KEY (`app_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country` (
  `country_id` int(11) NOT NULL AUTO_INCREMENT,
  `country_name` varchar(255) NOT NULL,
  `iso_code` char(3) NOT NULL,
  PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `language` (
  `language_id` int(11) NOT NULL AUTO_INCREMENT,
  `language_name` varchar(255) NOT NULL,
  `iso_code` char(6) NOT NULL,
  PRIMARY KEY (`language_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `region` (
  `region_id` int(11) NOT NULL AUTO_INCREMENT,
  `region_name` varchar(255) NOT NULL,
  `iso_code` char(6) NOT NULL,
  `country_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`region_id`),
  CONSTRAINT `region_ibfk_1` FOREIGN KEY (`country_id`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_file` (
  `file_id` int(11) NOT NULL AUTO_INCREMENT,
  `file_path` varchar(512) NOT NULL,
  `base_name` varchar(255) NOT NULL,
  `version` varchar(50) NOT NULL,
  `app_id` int(11) NOT NULL,
  `language_id` int(11) NOT NULL,
  `region_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`file_id`),
  CONSTRAINT `resource_file_ibfk_1` FOREIGN KEY (`app_id`) REFERENCES `application` (`app_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resource_file_ibfk_2` FOREIGN KEY (`language_id`) REFERENCES `language` (`language_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resource_file_ibfk_3` FOREIGN KEY (`region_id`) REFERENCES `region` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_key` (
  `key_id` int(11) NOT NULL AUTO_INCREMENT,
  `key_name` varchar(255) NOT NULL,
  `key_version` varchar(50) NOT NULL,
  `file_id` int(11) NOT NULL,
  PRIMARY KEY (`key_id`),
  CONSTRAINT `resource_key_ibfk_1` FOREIGN KEY (`file_id`) REFERENCES `resource_file` (`file_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_value` (
  `value_id` int(11) NOT NULL AUTO_INCREMENT,
  `value_content` text NOT NULL,
  `value_version` varchar(50) NOT NULL,
  `key_id` int(11) NOT NULL,
  PRIMARY KEY (`value_id`),
  CONSTRAINT `resource_value_ibfk_1` FOREIGN KEY (`key_id`) REFERENCES `resource_key` (`key_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
