-- Schema for database: rschema_0348

CREATE TABLE `plugin_type` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `type_name` varchar(255) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sql_statement` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `sql_text` text NOT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plugin_execution` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `plugin_version` varchar(50) NOT NULL,
  `plugin_type_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `sql_statement_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_start_time` (`start_time`),
  KEY `idx_end_time` (`end_time`),
  KEY `idx_plugin_type_id` (`plugin_type_id`),
  KEY `idx_sql_statement_id` (`sql_statement_id`),
  CONSTRAINT `plugin_execution_ibfk_1` FOREIGN KEY (`plugin_type_id`) REFERENCES `plugin_type` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `plugin_execution_ibfk_2` FOREIGN KEY (`sql_statement_id`) REFERENCES `sql_statement` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
