  `workflow_id` int(11) NOT NULL AUTO_INCREMENT,

  `workflow_type` varchar(100) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  `is_deleted` tinyint(1) NOT NULL DEFAULT 0,
