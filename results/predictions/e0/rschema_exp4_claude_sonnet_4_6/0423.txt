CREATE TABLE `course_basic` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_name` varchar(255) NOT NULL,
  `course_description` text DEFAULT NULL,
  `cover_image` varchar(500) DEFAULT NULL,
  `price` decimal(10,2) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_detail` (
  `course_id` int(11) NOT NULL,
  `lesson_count` int(11) DEFAULT NULL,
  `total_duration` int(11) DEFAULT NULL,

  `language` varchar(50) DEFAULT NULL,
  `prerequisites` text DEFAULT NULL,
  `learning_objectives` text DEFAULT NULL,
  PRIMARY KEY (`course_id`),
  CONSTRAINT `course_detail_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `course_basic` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
