-- Schema for database: rschema_0425

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role` varchar(20) NOT NULL COMMENT 'admin, dormitory_manager, student, maintenance_staff',
  `real_name` varchar(50) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  `status` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `building` (
  `building_id` int(11) NOT NULL AUTO_INCREMENT,
  `building_name` varchar(100) NOT NULL,
  `building_type` varchar(50) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `manager_id` int(11) DEFAULT NULL COMMENT 'responsible dormitory manager user_id',
  PRIMARY KEY (`building_id`),
  CONSTRAINT `building_ibfk_1` FOREIGN KEY (`manager_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitory` (
  `dormitory_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_name` varchar(100) NOT NULL,
  `dormitory_type` varchar(50) DEFAULT NULL,
  `bed_count` int(11) NOT NULL DEFAULT '0',
  `building_id` int(11) NOT NULL,
  `facilities` text DEFAULT NULL,
  `manager_id` int(11) DEFAULT NULL COMMENT 'responsible dormitory manager user_id',
  PRIMARY KEY (`dormitory_id`),
  CONSTRAINT `dormitory_ibfk_1` FOREIGN KEY (`building_id`) REFERENCES `building` (`building_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dormitory_ibfk_2` FOREIGN KEY (`manager_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student` (
  `student_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) DEFAULT NULL,
  `student_no` varchar(30) NOT NULL,
  `real_name` varchar(50) NOT NULL,
  `gender` char(1) DEFAULT NULL COMMENT 'M or F',
  `dormitory_id` int(11) DEFAULT NULL,
  `bed_no` varchar(10) DEFAULT NULL,
  `class_name` varchar(100) DEFAULT NULL,
  `department` varchar(100) DEFAULT NULL,
  `enrollment_year` int(11) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`student_id`),
  UNIQUE KEY `student_no` (`student_no`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `student_ibfk_2` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory` (`dormitory_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `application` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_type` varchar(50) NOT NULL COMMENT 'bed_change, repair, absence, checkout, etc.',
  `student_id` int(11) NOT NULL,
  `content` text DEFAULT NULL,
  `apply_time` datetime NOT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'pending' COMMENT 'pending, approved, rejected, processing, completed',
  `admin_opinion` text DEFAULT NULL,
  `processed_by` int(11) DEFAULT NULL COMMENT 'user_id of the processor',
  `process_time` datetime DEFAULT NULL,
  PRIMARY KEY (`application_id`),
  CONSTRAINT `application_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `application_ibfk_2` FOREIGN KEY (`processed_by`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `absence_record` (
  `absence_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_id` int(11) NOT NULL,
  `application_id` int(11) DEFAULT NULL,
  `leave_start` datetime NOT NULL,
  `leave_end` datetime DEFAULT NULL,
  `reason` text DEFAULT NULL,
  `manager_id` int(11) DEFAULT NULL COMMENT 'dormitory manager who handled',
  `status` varchar(20) NOT NULL DEFAULT 'active',
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`absence_id`),
  CONSTRAINT `absence_record_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `absence_record_ibfk_2` FOREIGN KEY (`application_id`) REFERENCES `application` (`application_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `absence_record_ibfk_3` FOREIGN KEY (`manager_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_id` int(11) NOT NULL,
  `application_id` int(11) DEFAULT NULL,
  `fault_description` text NOT NULL,
  `repair_time` datetime DEFAULT NULL,
  `maintenance_staff_id` int(11) DEFAULT NULL,
  `maintenance_situation` text DEFAULT NULL,
  `completion_time` datetime DEFAULT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'pending' COMMENT 'pending, in_progress, completed',
  PRIMARY KEY (`record_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_2` FOREIGN KEY (`application_id`) REFERENCES `application` (`application_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_3` FOREIGN KEY (`maintenance_staff_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcement` (
  `announcement_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `release_time` datetime NOT NULL,
  `admin_id` int(11) NOT NULL COMMENT 'user_id of the administrator who released',
  `status` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`announcement_id`),
  CONSTRAINT `announcement_ibfk_1` FOREIGN KEY (`admin_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
