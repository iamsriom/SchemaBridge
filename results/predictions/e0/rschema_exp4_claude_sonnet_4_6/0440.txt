-- Schema for database: rschema_0440

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(100) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `account_status` varchar(50) NOT NULL DEFAULT 'active',
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `devices` (
  `device_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `device_model` varchar(255) DEFAULT NULL,
  `person_in_charge` int(11) DEFAULT NULL,
  `group_name` varchar(255) DEFAULT NULL,
  `type` varchar(100) DEFAULT NULL,
  `usage_location` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `ip_address` varchar(50) DEFAULT NULL,
  `device_image` varchar(500) DEFAULT NULL,
  `qr_code` varchar(500) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`device_id`),
  UNIQUE KEY `qr_code` (`qr_code`),
  CONSTRAINT `devices_ibfk_1` FOREIGN KEY (`person_in_charge`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_status` (
  `status_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `current_status` varchar(100) NOT NULL,
  `last_update_time` datetime NOT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`status_id`),
  CONSTRAINT `device_status_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_inventory` (
  `inventory_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `inventory_person_id` int(11) NOT NULL,
  `inventory_time` datetime NOT NULL,
  `device_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`inventory_id`),
  CONSTRAINT `device_inventory_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `device_inventory_ibfk_2` FOREIGN KEY (`inventory_person_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_requests` (
  `request_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `requester_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text DEFAULT NULL,
  `handling_status` varchar(100) NOT NULL DEFAULT 'pending',
  `handling_personnel` int(11) DEFAULT NULL,
  `handling_time` datetime DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`request_id`),
  CONSTRAINT `maintenance_requests_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_requests_ibfk_2` FOREIGN KEY (`requester_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_requests_ibfk_3` FOREIGN KEY (`handling_personnel`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `request_id` int(11) DEFAULT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text DEFAULT NULL,
  `maintenance_result` text DEFAULT NULL,
  `user_satisfaction` varchar(50) DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `maintenance_records_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_records_ibfk_2` FOREIGN KEY (`request_id`) REFERENCES `maintenance_requests` (`request_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `maintenance_records_ibfk_3` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
