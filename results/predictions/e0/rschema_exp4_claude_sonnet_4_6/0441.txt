-- Schema for database: rschema_0441

CREATE TABLE `roles` (
  `role_id` int(11) NOT NULL AUTO_INCREMENT,
  `role_name` varchar(50) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`role_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_name` varchar(100) NOT NULL,
  `role_id` int(11) NOT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `users_ibfk_1` FOREIGN KEY (`role_id`) REFERENCES `roles` (`role_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `budget` decimal(18,2) NOT NULL,
  `remaining_funds` decimal(18,2) NOT NULL,
  `responsible_user_id` int(11) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`project_id`),
  CONSTRAINT `projects_ibfk_1` FOREIGN KEY (`responsible_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fund_transactions` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `transaction_amount` decimal(18,2) NOT NULL,
  `purpose_description` varchar(500) DEFAULT NULL,
  `transaction_operator` int(11) NOT NULL,
  `transaction_type` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `fund_transactions_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fund_transactions_ibfk_2` FOREIGN KEY (`transaction_operator`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_records` (
  `approval_id` int(11) NOT NULL AUTO_INCREMENT,
  `record_id` int(11) NOT NULL,
  `approver_id` int(11) NOT NULL,
  `approval_status` varchar(50) NOT NULL,
  `approval_time` datetime DEFAULT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`approval_id`),
  CONSTRAINT `approval_records_ibfk_1` FOREIGN KEY (`record_id`) REFERENCES `fund_transactions` (`record_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `approval_records_ibfk_2` FOREIGN KEY (`approver_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fund_analysis_reports` (
  `report_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `report_name` varchar(255) NOT NULL,
  `generated_by` int(11) NOT NULL,
  `generated_at` datetime DEFAULT NULL,
  `report_content` text DEFAULT NULL,
  `query_dimension` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`report_id`),
  CONSTRAINT `fund_analysis_reports_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fund_analysis_reports_ibfk_2` FOREIGN KEY (`generated_by`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fund_warnings` (
  `warning_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `warning_type` varchar(100) NOT NULL,
  `warning_message` varchar(500) DEFAULT NULL,
  `warning_time` datetime DEFAULT NULL,
  `is_resolved` tinyint(1) DEFAULT NULL,
  PRIMARY KEY (`warning_id`),
  CONSTRAINT `fund_warnings_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
