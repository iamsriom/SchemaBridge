-- Schema for database: rschema_0447

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `associated_unit` varchar(255) DEFAULT NULL,
  `role` varchar(100) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `financial_document` (
  `document_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_title` varchar(255) NOT NULL,
  `document_number` varchar(100) NOT NULL,
  `issuing_unit` varchar(255) DEFAULT NULL,
  `issue_date` datetime DEFAULT NULL,
  `document_category` varchar(100) DEFAULT NULL,
  `document_level` varchar(50) DEFAULT NULL,
  `validity_status` tinyint(1) NOT NULL DEFAULT 1,
  `remarks` text DEFAULT NULL,
  `document_content` longtext DEFAULT NULL,
  `attachment_path` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`document_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `document_operation` (
  `operation_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_id` int(11) NOT NULL,
  `admin_id` int(11) NOT NULL,
  `operation_time` datetime NOT NULL,
  `operation_type` varchar(100) NOT NULL,
  PRIMARY KEY (`operation_id`),
  CONSTRAINT `document_operation_ibfk_1` FOREIGN KEY (`document_id`) REFERENCES `financial_document` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `document_operation_ibfk_2` FOREIGN KEY (`admin_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notification` (
  `notification_id` int(11) NOT NULL AUTO_INCREMENT,
  `publisher_id` int(11) NOT NULL,
  `notification_content` text NOT NULL,
  `issue_date` datetime NOT NULL,
  `notification_status` varchar(50) NOT NULL DEFAULT 'active',
  PRIMARY KEY (`notification_id`),
  CONSTRAINT `notification_ibfk_1` FOREIGN KEY (`publisher_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `web_system_config` (
  `config_id` int(11) NOT NULL AUTO_INCREMENT,
  `system_name` varchar(255) NOT NULL,
  `database_directory` varchar(500) NOT NULL,
  `data_type` varchar(100) DEFAULT NULL,
  `file_upload_directory` varchar(500) DEFAULT NULL,
  `admin_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`config_id`),
  CONSTRAINT `web_system_config_ibfk_1` FOREIGN KEY (`admin_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
