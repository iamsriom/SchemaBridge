-- Schema for database: rschema_0448

CREATE TABLE `unit` (
  `unit_id` int(11) NOT NULL AUTO_INCREMENT,
  `unit_name` varchar(255) NOT NULL,
  `unit_description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`unit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(100) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `role` enum('Student','Teacher','Laboratory Administrator','Experimental Center Director Administrator','System Administrator') NOT NULL,
  `unit_id` int(11) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `user_ibfk_1` FOREIGN KEY (`unit_id`) REFERENCES `unit` (`unit_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `model` varchar(100) DEFAULT NULL,
  `category` varchar(100) DEFAULT NULL,
  `manufacturer` varchar(255) DEFAULT NULL,
  `status` enum('inactive','in use','active') NOT NULL DEFAULT 'active',
  `acquisition_time` datetime DEFAULT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_usage_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `usage_time` datetime NOT NULL,
  `purpose` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `equipment_usage_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `equipment_usage_record_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `experimental_course` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_name` varchar(255) NOT NULL,
  `class_time` varchar(255) DEFAULT NULL,
  `class_location` varchar(255) DEFAULT NULL,
  `teacher_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`course_id`),
  CONSTRAINT `experimental_course_ibfk_1` FOREIGN KEY (`teacher_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_student` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `course_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `course_student_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `experimental_course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_student_ibfk_2` FOREIGN KEY (`student_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_course` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `equipment_course_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `equipment_course_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `experimental_course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consumable` (
  `consumable_id` int(11) NOT NULL AUTO_INCREMENT,
  `consumable_name` varchar(255) NOT NULL,
  `category` varchar(100) DEFAULT NULL,
  `inventory_quantity` int(11) NOT NULL DEFAULT 0,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`consumable_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consumable_transaction` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `consumable_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `transaction_type` enum('purchase','storage','issuance','return') NOT NULL,
  `quantity` int(11) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`transaction_id`),
  CONSTRAINT `consumable_transaction_ibfk_1` FOREIGN KEY (`consumable_id`) REFERENCES `consumable` (`consumable_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `consumable_transaction_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hazardous_chemical` (
  `chemical_id` int(11) NOT NULL AUTO_INCREMENT,
  `chemical_name` varchar(255) NOT NULL,
  `inventory_quantity` int(11) NOT NULL DEFAULT 0,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`chemical_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hazardous_chemical_transaction` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `chemical_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `transaction_type` enum('purchase','storage','issuance','return') NOT NULL,
  `quantity` int(11) NOT NULL,
  `issuance_time` datetime NOT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`transaction_id`),
  CONSTRAINT `hazardous_chemical_transaction_ibfk_1` FOREIGN KEY (`chemical_id`) REFERENCES `hazardous_chemical` (`chemical_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `hazardous_chemical_transaction_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
