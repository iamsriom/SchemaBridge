-- Schema for database: rschema_0449

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(100) NOT NULL,
  `department_function` varchar(500) DEFAULT NULL,
  `department_planning` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position` (
  `position_id` int(11) NOT NULL AUTO_INCREMENT,
  `position_name` varchar(100) NOT NULL,
  `position_responsibilities` varchar(500) DEFAULT NULL,
  `position_requirements` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`position_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_name` varchar(100) NOT NULL,
  `employee_gender` char(1) DEFAULT NULL,
  `employee_date_of_birth` datetime DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  `employee_entry_date` datetime DEFAULT NULL,
  `employee_resignation_date` datetime DEFAULT NULL,
  `position_id` int(11) DEFAULT NULL,
  `employee_contact` varchar(50) DEFAULT NULL,
  `employee_email` varchar(100) DEFAULT NULL,
  `employee_status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `training` (
  `training_id` int(11) NOT NULL AUTO_INCREMENT,
  `training_time` datetime DEFAULT NULL,
  `training_location` varchar(200) DEFAULT NULL,
  `training_content` varchar(1000) DEFAULT NULL,
  `trainer` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`training_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_position_change` (
  `employee_id` int(11) NOT NULL,
  `position_id` int(11) NOT NULL,
  `change_date` datetime NOT NULL,
  `change_reason` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`employee_id`, `position_id`, `change_date`),
  CONSTRAINT `employee_position_change_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_position_change_ibfk_2` FOREIGN KEY (`position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
