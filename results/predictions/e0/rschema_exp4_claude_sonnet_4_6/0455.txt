-- Schema for database: rschema_0455

CREATE TABLE `parking_space` (
  `parking_space_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_number` varchar(50) NOT NULL,
  `is_occupied` tinyint(1) NOT NULL DEFAULT 0,
  `vehicle_id` varchar(50) DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`parking_space_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `entry` (
  `entry_id` int(11) NOT NULL AUTO_INCREMENT,
  `entry_point_name` varchar(100) NOT NULL,
  `vehicle_id` varchar(50) DEFAULT NULL,
  `parking_space_id` int(11) DEFAULT NULL,
  `has_entered` tinyint(1) NOT NULL DEFAULT 0,
  `entry_time` datetime DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`entry_id`),
  CONSTRAINT `entry_ibfk_1` FOREIGN KEY (`parking_space_id`) REFERENCES `parking_space` (`parking_space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exit` (
  `exit_id` int(11) NOT NULL AUTO_INCREMENT,
  `exit_point_name` varchar(100) NOT NULL,
  `vehicle_id` varchar(50) DEFAULT NULL,
  `parking_space_id` int(11) DEFAULT NULL,
  `has_exited` tinyint(1) NOT NULL DEFAULT 0,
  `exit_time` datetime DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`exit_id`),
  CONSTRAINT `exit_ibfk_1` FOREIGN KEY (`parking_space_id`) REFERENCES `parking_space` (`parking_space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
