CREATE TABLE `owner` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `workplace` varchar(255) DEFAULT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`owner_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `house` (
  `house_number` varchar(20) NOT NULL,
  `floor_area` decimal(10,2) NOT NULL,
  `owner_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`house_number`),
  CONSTRAINT `house_ibfk_1` FOREIGN KEY (`owner_id`) REFERENCES `owner` (`owner_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(100) NOT NULL,
  `department_phone` varchar(20) DEFAULT NULL,
  `supervisor_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  `position` varchar(50) DEFAULT NULL,
  `password` varchar(255) NOT NULL,
  `department_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

ALTER TABLE `department` ADD CONSTRAINT `department_ibfk_1` FOREIGN KEY (`supervisor_id`) REFERENCES `employee` (`employee_id`) ON DELETE SET NULL ON UPDATE CASCADE;

CREATE TABLE `fee_collection` (
  `fee_id` int(11) NOT NULL AUTO_INCREMENT,
  `house_number` varchar(20) NOT NULL,
  `fee_type` enum('property_management_fee','sanitation_fee','water_fee','electricity_fee') NOT NULL,
  `quantity` decimal(10,4) NOT NULL,
  `fee_amount` decimal(10,2) NOT NULL,
  `collection_month` char(7) NOT NULL,
  `collected_by` int(11) DEFAULT NULL,
  `collection_date` datetime DEFAULT NULL,
  PRIMARY KEY (`fee_id`),
  CONSTRAINT `fee_collection_ibfk_1` FOREIGN KEY (`house_number`) REFERENCES `house` (`house_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fee_collection_ibfk_2` FOREIGN KEY (`collected_by`) REFERENCES `employee` (`employee_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fee_receipt` (
  `receipt_id` int(11) NOT NULL AUTO_INCREMENT,
  `fee_id` int(11) NOT NULL,
  `receipt_date` datetime NOT NULL,
  `total_amount` decimal(10,2) NOT NULL,
  `issued_by` int(11) DEFAULT NULL,
  PRIMARY KEY (`receipt_id`),
  CONSTRAINT `fee_receipt_ibfk_1` FOREIGN KEY (`fee_id`) REFERENCES `fee_collection` (`fee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fee_receipt_ibfk_2` FOREIGN KEY (`issued_by`) REFERENCES `employee` (`employee_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `accounts_receivable` (
  `ar_id` int(11) NOT NULL AUTO_INCREMENT,
  `house_number` varchar(20) NOT NULL,
  `fee_type` enum('property_management_fee','sanitation_fee','water_fee','electricity_fee') NOT NULL,
  `collection_month` char(7) NOT NULL,
  `amount_due` decimal(10,2) NOT NULL,
  `amount_paid` decimal(10,2) DEFAULT NULL,
  `status` varchar(20) DEFAULT NULL,
  `fee_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`ar_id`),
  CONSTRAINT `accounts_receivable_ibfk_1` FOREIGN KEY (`house_number`) REFERENCES `house` (`house_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `accounts_receivable_ibfk_2` FOREIGN KEY (`fee_id`) REFERENCES `fee_collection` (`fee_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
