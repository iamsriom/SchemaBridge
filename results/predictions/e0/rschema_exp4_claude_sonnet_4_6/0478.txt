-- Schema for database: rschema_0478

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `phone_number` varchar(20) NOT NULL,
  `name` varchar(100) NOT NULL,
  `credit` decimal(10,2) DEFAULT NULL,
  `status` varchar(20) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bicycle` (
  `bicycle_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) DEFAULT NULL,
  `status` varchar(20) NOT NULL,
  `unlock_code` varchar(50) NOT NULL,
  PRIMARY KEY (`bicycle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `billing_rule` (
  `rule_id` int(11) NOT NULL AUTO_INCREMENT,
  `billing_method` varchar(50) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  `minimum_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`rule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trip` (
  `trip_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `bicycle_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `cost` decimal(10,2) DEFAULT NULL,
  `payment_status` varchar(20) NOT NULL,
  `rule_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`trip_id`),
  CONSTRAINT `trip_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trip_ibfk_2` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycle` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trip_ibfk_3` FOREIGN KEY (`rule_id`) REFERENCES `billing_rule` (`rule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fault_record` (
  `fault_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `bicycle_id` int(11) NOT NULL,
  `fault_description` text NOT NULL,
  `report_time` datetime NOT NULL,
  PRIMARY KEY (`fault_id`),
  CONSTRAINT `fault_record_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fault_record_ibfk_2` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycle` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
