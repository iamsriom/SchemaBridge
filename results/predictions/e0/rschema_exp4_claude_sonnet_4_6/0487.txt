-- Schema for database: rschema_0487

CREATE TABLE `kodeliste` (
  `kodeliste_id` int(11) NOT NULL AUTO_INCREMENT,
  `kode` varchar(50) NOT NULL,
  `navn` varchar(255) NOT NULL,
  `beskrivelse` varchar(1000) DEFAULT NULL,
  `aktiv` tinyint(1) NOT NULL DEFAULT 1,
  `opprettet_av` varchar(100) NOT NULL,
  `opprettet_dato` datetime NOT NULL,
  `endret_av` varchar(100) DEFAULT NULL,
  `endret_dato` datetime DEFAULT NULL,
  PRIMARY KEY (`kodeliste_id`),
  UNIQUE KEY `uq_kodeliste_kode` (`kode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `koderelasjon` (
  `koderelasjon_id` int(11) NOT NULL AUTO_INCREMENT,
  `kilde_kodeliste_id` int(11) NOT NULL,
  `maal_kodeliste_id` int(11) NOT NULL,
  `kilde_kode` varchar(50) NOT NULL,
  `maal_kode` varchar(50) NOT NULL,
  `gyldig_fra` datetime NOT NULL,
  `gyldig_til` datetime DEFAULT NULL,
  `opprettet_av` varchar(100) NOT NULL,
  `opprettet_dato` datetime NOT NULL,
  `endret_av` varchar(100) DEFAULT NULL,
  `endret_dato` datetime DEFAULT NULL,
  PRIMARY KEY (`koderelasjon_id`),
  UNIQUE KEY `uq_koderelasjon` (`kilde_kodeliste_id`, `maal_kodeliste_id`, `kilde_kode`, `maal_kode`),
  CONSTRAINT `koderelasjon_ibfk_1` FOREIGN KEY (`kilde_kodeliste_id`) REFERENCES `kodeliste` (`kodeliste_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `koderelasjon_ibfk_2` FOREIGN KEY (`maal_kodeliste_id`) REFERENCES `kodeliste` (`kodeliste_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
