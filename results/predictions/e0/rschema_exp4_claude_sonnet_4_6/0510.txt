-- Schema for database: rschema_0510

CREATE TABLE `parking_space` (
  `space_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_number` varchar(20) NOT NULL,
  `floor` varchar(10) DEFAULT NULL,
  `zone` varchar(10) DEFAULT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'available',
  `space_type` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`space_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_space_owner` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_id` int(11) NOT NULL,
  `owner_name` varchar(100) NOT NULL,
  `license_plate` varchar(20) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  `purchase_date` datetime DEFAULT NULL,
  PRIMARY KEY (`owner_id`),
  CONSTRAINT `parking_space_owner_ibfk_1` FOREIGN KEY (`space_id`) REFERENCES `parking_space` (`space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_entry_exit` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_id` int(11) NOT NULL,
  `license_plate` varchar(20) NOT NULL,
  `entry_time` datetime NOT NULL,
  `exit_time` datetime DEFAULT NULL,
  `fee` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `vehicle_entry_exit_ibfk_1` FOREIGN KEY (`space_id`) REFERENCES `parking_space` (`space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
