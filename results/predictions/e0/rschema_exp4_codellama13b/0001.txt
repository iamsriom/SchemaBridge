-- Schema for database: rschema_0001

CREATE TABLE airplanes (
  airplane_number INT(11) NOT NULL,
  model VARCHAR(255) NOT NULL,
  passenger_capacity INT(11) NOT NULL,
  PRIMARY KEY (airplane_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE flights (
  flight_number INT(11) NOT NULL,
  departure_time DATETIME NOT NULL,
  arrival_time DATETIME NOT NULL,
  origin VARCHAR(255) NOT NULL,
  destination VARCHAR(255) NOT NULL,
  airplane_number INT(11) NOT NULL,
  captain_id INT(11) NOT NULL,
  first_officer_id INT(11) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  PRIMARY KEY (flight_number),
  FOREIGN KEY (airplane_number) REFERENCES airplanes (airplane_number) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (captain_id) REFERENCES captains (captain_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (first_officer_id) REFERENCES first_officers (first_officer_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE captains (
  captain_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (captain_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE first_officers (
  first_officer_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (first_officer_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
