-- Schema for database: rschema_0003

CREATE TABLE warehouses (
  warehouse_id INT(11) NOT NULL AUTO_INCREMENT,
  warehouse_number VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  capacity INT(11) NOT NULL,
  PRIMARY KEY (warehouse_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE loaders (
  loader_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (loader_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE unloaders (
  unloader_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (unloader_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inbound_tasks (
  task_id INT(11) NOT NULL AUTO_INCREMENT,
  warehouse_id INT(11) NOT NULL,
  loader_id INT(11) NOT NULL,
  goods_name VARCHAR(255) NOT NULL,
  quantity INT(11) NOT NULL,
  time DATETIME NOT NULL,
  PRIMARY KEY (task_id),
  FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (loader_id) REFERENCES loaders(loader_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE outbound_tasks (
  task_id INT(11) NOT NULL AUTO_INCREMENT,
  warehouse_id INT(11) NOT NULL,
  unloader_id INT(11) NOT NULL,
  goods_name VARCHAR(255) NOT NULL,
  quantity INT(11) NOT NULL,
  time DATETIME NOT NULL,
  PRIMARY KEY (task_id),
  FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (unloader_id) REFERENCES unloaders(unloader_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
