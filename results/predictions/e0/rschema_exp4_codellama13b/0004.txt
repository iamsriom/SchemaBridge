-- Schema for database: rschema_0004

CREATE TABLE `customers` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_plans` (
`plan_id` int(11) NOT NULL AUTO_INCREMENT,
`plan_number` varchar(255) NOT NULL,
`plan_name` varchar(255) NOT NULL,
`price` decimal(10,2) NOT NULL,
`included_services` varchar(255) NOT NULL,
PRIMARY KEY (`plan_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subscriptions` (
`subscription_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_id` int(11) NOT NULL,
`plan_id` int(11) NOT NULL,
`subscription_time` datetime NOT NULL,
`cancellation_time` datetime DEFAULT NULL,
`operating_personnel` varchar(255) NOT NULL,
PRIMARY KEY (`subscription_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`plan_id`) REFERENCES `service_plans` (`plan_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
