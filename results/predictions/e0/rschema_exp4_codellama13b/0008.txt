-- Schema for database: rschema_0008

CREATE TABLE equipment (
  equipment_id int(11) NOT NULL,
  name varchar(255) NOT NULL,
  model varchar(255) NOT NULL,
  usage_status varchar(255) NOT NULL,
  PRIMARY KEY (equipment_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE personnel (
  personnel_id int(11) NOT NULL,
  name varchar(255) NOT NULL,
  telephone varchar(255) NOT NULL,
  email varchar(255) NOT NULL,
  PRIMARY KEY (personnel_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE maintenance_records (
  maintenance_record_id int(11) NOT NULL,
  equipment_id int(11) NOT NULL,
  maintenance_time datetime NOT NULL,
  maintenance_results varchar(255) NOT NULL,
  personnel_id int(11) NOT NULL,
  PRIMARY KEY (maintenance_record_id),
  FOREIGN KEY (equipment_id) REFERENCES equipment (equipment_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (personnel_id) REFERENCES personnel (personnel_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
