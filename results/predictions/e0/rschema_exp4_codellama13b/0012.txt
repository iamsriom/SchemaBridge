-- Schema for database: rschema_0012

CREATE TABLE warehouses (
  warehouse_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  capacity INT(11) NOT NULL,
  current_inventory INT(11) NOT NULL,
  PRIMARY KEY (warehouse_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE products (
  product_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  specification VARCHAR(255) NOT NULL,
  date_of_storage DATE NOT NULL,
  date_of_dispatch DATE NOT NULL,
  quantity_stored INT(11) NOT NULL,
  quantity_dispatched INT(11) NOT NULL,
  current_inventory INT(11) NOT NULL,
  PRIMARY KEY (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE storage_records (
  storage_record_id INT(11) NOT NULL AUTO_INCREMENT,
  warehouse_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  storage_date DATE NOT NULL,
  quantity_stored INT(11) NOT NULL,
  operator VARCHAR(255) NOT NULL,
  PRIMARY KEY (storage_record_id),
  FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE dispatch_records (
  dispatch_record_id INT(11) NOT NULL AUTO_INCREMENT,
  warehouse_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  dispatch_date DATE NOT NULL,
  quantity_dispatched INT(11) NOT NULL,
  operator VARCHAR(255) NOT NULL,
  PRIMARY KEY (dispatch_record_id),
  FOREIGN KEY (warehouse_id) REFERENCES warehouses(warehouse_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
