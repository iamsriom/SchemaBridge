-- Schema for database: rschema_0013

CREATE TABLE weather_stations (
  weather_station_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  person_in_charge VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  PRIMARY KEY (weather_station_id)
);

CREATE TABLE weather_data (
  weather_data_id INT(11) NOT NULL AUTO_INCREMENT,
  weather_station_id INT(11) NOT NULL,
  recording_date DATETIME NOT NULL,
  temperature DECIMAL(10,2) NOT NULL,
  humidity DECIMAL(10,2) NOT NULL,
  precipitation DECIMAL(10,2) NOT NULL,
  wind_speed DECIMAL(10,2) NOT NULL,
  wind_direction VARCHAR(255) NOT NULL,
  PRIMARY KEY (weather_data_id),
  FOREIGN KEY (weather_station_id) REFERENCES weather_stations (weather_station_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE farm_subscriptions (
  farm_subscription_id INT(11) NOT NULL AUTO_INCREMENT,
  weather_station_id INT(11) NOT NULL,
  subscription_date DATETIME NOT NULL,
  unsubscription_date DATETIME,
  PRIMARY KEY (farm_subscription_id),
  FOREIGN KEY (weather_station_id) REFERENCES weather_stations (weather_station_id) ON DELETE CASCADE ON UPDATE CASCADE
);
