-- Schema for database: rschema_0014

CREATE TABLE `irrigation_systems` (
  `system_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `installation_date` datetime NOT NULL,
  `maintenance_records` text NOT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plots` (
  `plot_id` int(11) NOT NULL,
  `area` int(11) NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `irrigation_status` varchar(255) NOT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `irrigation_records` (
  `system_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `irrigation_date` datetime NOT NULL,
  `irrigation_duration` int(11) NOT NULL,
  `irrigation_amount` int(11) NOT NULL,
  PRIMARY KEY (`system_id`, `plot_id`, `irrigation_date`),
  FOREIGN KEY (`system_id`) REFERENCES `irrigation_systems` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
