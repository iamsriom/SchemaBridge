-- Schema for database: rschema_0015

CREATE TABLE `plots` (
`plot_id` int(11) NOT NULL,
`area` int(11) NOT NULL,
`soil_type` varchar(255) NOT NULL,
`geographical_location` varchar(255) NOT NULL,
`current_pest_and_disease_status` varchar(255) NOT NULL,
PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pests_and_diseases` (
`pest_and_disease_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`degree_of_damage` varchar(255) NOT NULL,
`prevention_and_control_methods` varchar(255) NOT NULL,
`season_of_occurrence` varchar(255) NOT NULL,
PRIMARY KEY (`pest_and_disease_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_records` (
`plot_id` int(11) NOT NULL,
`pest_and_disease_id` int(11) NOT NULL,
`monitoring_date` datetime NOT NULL,
`monitoring_results` varchar(255) NOT NULL,
`operator_id` int(11) NOT NULL,
PRIMARY KEY (`plot_id`, `pest_and_disease_id`, `monitoring_date`),
FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`pest_and_disease_id`) REFERENCES `pests_and_diseases` (`pest_and_disease_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`operator_id`) REFERENCES `operators` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operators` (
`operator_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`age` int(11) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
