-- Schema for database: rschema_0017

CREATE TABLE workshops (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE workers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  contact_number VARCHAR(255) NOT NULL,
  workshop_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (workshop_id) REFERENCES workshops(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE processing_records (
  id INT(11) NOT NULL AUTO_INCREMENT,
  worker_id INT(11) NOT NULL,
  raw_materials VARCHAR(255) NOT NULL,
  processing_time DATETIME NOT NULL,
  product_name VARCHAR(255) NOT NULL,
  production_quantity INT(11) NOT NULL,
  product_quality_rating INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (worker_id) REFERENCES workers(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE raw_materials (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  workshop_id INT(11) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (workshop_id) REFERENCES workshops(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE production_metrics (
  id INT(11) NOT NULL AUTO_INCREMENT,
  workshop_id INT(11) NOT NULL,
  monthly_total_processing_volume INT(11) NOT NULL,
  product_qualification_rate INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (workshop_id) REFERENCES workshops(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
