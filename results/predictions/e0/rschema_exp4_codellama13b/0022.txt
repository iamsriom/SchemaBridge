-- Schema for database: rschema_0022

CREATE TABLE documents (
  document_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  category VARCHAR(255) NOT NULL,
  creation_time DATETIME NOT NULL,
  status VARCHAR(255) NOT NULL,
  PRIMARY KEY (document_id)
);

CREATE TABLE approval_records (
  document_id INT(11) NOT NULL,
  approver VARCHAR(255) NOT NULL,
  approval_time DATETIME NOT NULL,
  approval_comments VARCHAR(255) NOT NULL,
  PRIMARY KEY (document_id, approver),
  FOREIGN KEY (document_id) REFERENCES documents(document_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE archiving_records (
  document_id INT(11) NOT NULL,
  archivist VARCHAR(255) NOT NULL,
  archiving_time DATETIME NOT NULL,
  archiving_location VARCHAR(255) NOT NULL,
  PRIMARY KEY (document_id),
  FOREIGN KEY (document_id) REFERENCES documents(document_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE query_records (
  document_id INT(11) NOT NULL,
  query_personnel VARCHAR(255) NOT NULL,
  query_time DATETIME NOT NULL,
  query_results VARCHAR(255) NOT NULL,
  PRIMARY KEY (document_id, query_personnel),
  FOREIGN KEY (document_id) REFERENCES documents(document_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE document_management_personnel (
  employee_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  PRIMARY KEY (employee_id)
);
