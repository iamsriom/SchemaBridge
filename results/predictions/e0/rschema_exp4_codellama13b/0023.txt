-- Schema for database: rschema_0023

CREATE TABLE donors (
  donor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(10) NOT NULL,
  dob DATE NOT NULL,
  contact_info VARCHAR(255) NOT NULL,
  PRIMARY KEY (donor_id)
);

CREATE TABLE recipients (
  recipient_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(10) NOT NULL,
  dob DATE NOT NULL,
  contact_info VARCHAR(255) NOT NULL,
  PRIMARY KEY (recipient_id)
);

CREATE TABLE donation_records (
  donation_id INT(11) NOT NULL AUTO_INCREMENT,
  donor_id INT(11) NOT NULL,
  recipient_id INT(11) NOT NULL,
  donation_time DATETIME NOT NULL,
  PRIMARY KEY (donation_id),
  FOREIGN KEY (donor_id) REFERENCES donors(donor_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (recipient_id) REFERENCES recipients(recipient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE donated_items (
  item_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  quantity INT(11) NOT NULL,
  description VARCHAR(255) NOT NULL,
  PRIMARY KEY (item_id)
);

CREATE TABLE donor_donation_count (
  donor_id INT(11) NOT NULL,
  donation_count INT(11) NOT NULL,
  PRIMARY KEY (donor_id),
  FOREIGN KEY (donor_id) REFERENCES donors(donor_id) ON DELETE CASCADE ON UPDATE CASCADE
);
