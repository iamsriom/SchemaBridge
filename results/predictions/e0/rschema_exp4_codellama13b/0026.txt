-- Schema for database: rschema_0026

CREATE TABLE `departments` (
`department_id` int(11) NOT NULL AUTO_INCREMENT,
`department_name` varchar(255) NOT NULL,
`manager_name` varchar(255) NOT NULL,
`contact_info` varchar(255) NOT NULL,
PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
`employee_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `textiles` (
`textile_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`production_date` datetime NOT NULL,
`price` int(11) NOT NULL,
PRIMARY KEY (`textile_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `productions` (
`production_id` int(11) NOT NULL AUTO_INCREMENT,
`textile_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`producer` varchar(255) NOT NULL,
`production_time` datetime NOT NULL,
PRIMARY KEY (`production_id`),
FOREIGN KEY (`textile_id`) REFERENCES `textiles` (`textile_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventories` (
`inventory_id` int(11) NOT NULL AUTO_INCREMENT,
`textile_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`stock_in_time` datetime NOT NULL,
PRIMARY KEY (`inventory_id`),
FOREIGN KEY (`textile_id`) REFERENCES `textiles` (`textile_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
`sale_id` int(11) NOT NULL AUTO_INCREMENT,
`textile_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`sales_personnel` varchar(255) NOT NULL,
`customer` varchar(255) NOT NULL,
`sales_time` datetime NOT NULL,
PRIMARY KEY (`sale_id`),
FOREIGN KEY (`textile_id`) REFERENCES `textiles` (`textile_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `evaluations` (
`evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
`department_id` int(11) NOT NULL,
`employee_id` int(11) NOT NULL,
`evaluation_date` datetime NOT NULL,
`performance_score` int(11) NOT NULL,
PRIMARY KEY (`evaluation_id`),
FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
