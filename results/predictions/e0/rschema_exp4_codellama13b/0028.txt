-- Schema for database: rschema_0028

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timber` (
  `timber_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `purchase_price` decimal(10,2) NOT NULL,
  `processing_cost` decimal(10,2) NOT NULL,
  `selling_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`timber_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `procurement` (
  `procurement_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `purchaser` varchar(255) NOT NULL,
  `supplier` varchar(255) NOT NULL,
  `purchase_time` datetime NOT NULL,
  PRIMARY KEY (`procurement_id`),
  FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing` (
  `processing_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `processor` varchar(255) NOT NULL,
  `processing_time` datetime NOT NULL,
  PRIMARY KEY (`processing_id`),
  FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `sales_id` int(11) NOT NULL AUTO_INCREMENT,
  `timber_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `salesperson` varchar(255) NOT NULL,
  `customer` varchar(255) NOT NULL,
  `sales_time` datetime NOT NULL,
  PRIMARY KEY (`sales_id`),
  FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluation` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `evaluation_time` datetime NOT NULL,
  `evaluation_result` varchar(255) NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
