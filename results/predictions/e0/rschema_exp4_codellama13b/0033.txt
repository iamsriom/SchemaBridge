-- Schema for database: rschema_0033

CREATE TABLE `products` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`model` varchar(255) NOT NULL,
`price` decimal(10,2) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_teams` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`team_id` int(11) NOT NULL,
`team_name` varchar(255) NOT NULL,
`manager` varchar(255) NOT NULL,
`contact_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`team_id`) REFERENCES `sales_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_personnel` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`team_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`team_id`) REFERENCES `sales_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_tasks` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`customer_name` varchar(255) NOT NULL,
`customer_phone_number` varchar(255) NOT NULL,
`product_name` varchar(255) NOT NULL,
`product_model` varchar(255) NOT NULL,
`product_price` decimal(10,2) NOT NULL,
`sales_date` datetime NOT NULL,
`quantity` int(11) NOT NULL,
`sales_cost` decimal(10,2) NOT NULL,
`team_id` int(11) NOT NULL,
`personnel_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`team_id`) REFERENCES `sales_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`personnel_id`) REFERENCES `sales_personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_feedback` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`customer_name` varchar(255) NOT NULL,
`customer_phone_number` varchar(255) NOT NULL,
`product_name` varchar(255) NOT NULL,
`product_model` varchar(255) NOT NULL,
`product_price` decimal(10,2) NOT NULL,
`sales_date` datetime NOT NULL,
`quantity` int(11) NOT NULL,
`sales_cost` decimal(10,2) NOT NULL,
`rating` int(11) NOT NULL,
`feedback` varchar(255) NOT NULL,
`team_id` int(11) NOT NULL,
`personnel_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`team_id`) REFERENCES `sales_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`personnel_id`) REFERENCES `sales_personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
