-- Schema for database: rschema_0034

CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL,
  `equipment_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `purchase_price` decimal(10,2) NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `project_description` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_usage` (
  `equipment_usage_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `equipment_id` int(11) NOT NULL,
  `usage_date` datetime NOT NULL,
  `usage_time` time NOT NULL,
  `user` varchar(255) NOT NULL,
  PRIMARY KEY (`equipment_usage_id`),
  FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
