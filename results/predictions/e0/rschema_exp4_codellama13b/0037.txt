-- Schema for database: rschema_0037

CREATE TABLE `projects` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`project_manager` varchar(255) NOT NULL,
`start_date` datetime NOT NULL,
`expected_completion_date` datetime NOT NULL,
`actual_completion_date` datetime NOT NULL,
`project_budget` decimal(10,2) NOT NULL,
`project_location` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspectors` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspections` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`project_id` int(11) NOT NULL,
`inspector_id` int(11) NOT NULL,
`inspection_date` datetime NOT NULL,
`inspection_content` varchar(255) NOT NULL,
`inspection_results` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`project_id`) REFERENCES `projects` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`inspector_id`) REFERENCES `inspectors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
