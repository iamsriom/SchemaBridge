-- Schema for database: rschema_0039

CREATE TABLE `projects` (
`project_id` int(11) NOT NULL,
`project_name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`start_date` datetime NOT NULL,
`expected_completion_date` datetime NOT NULL,
`actual_completion_date` datetime DEFAULT NULL,
`project_manager_id` int(11) NOT NULL,
PRIMARY KEY (`project_id`),
FOREIGN KEY (`project_manager_id`) REFERENCES `project_managers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `buildings` (
`building_id` int(11) NOT NULL,
`project_id` int(11) NOT NULL,
`number_of_floors` int(11) NOT NULL,
`area` int(11) NOT NULL,
`price` int(11) NOT NULL,
PRIMARY KEY (`building_id`),
FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
`sale_id` int(11) NOT NULL,
`building_id` int(11) NOT NULL,
`sales_person_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
`sales_status` varchar(255) NOT NULL,
PRIMARY KEY (`sale_id`),
FOREIGN KEY (`building_id`) REFERENCES `buildings` (`building_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`sales_person_id`) REFERENCES `sales_people` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_people` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_managers` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
