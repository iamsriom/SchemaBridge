-- Schema for database: rschema_0040

CREATE TABLE `properties` (
`property_id` int(11) NOT NULL AUTO_INCREMENT,
`address` varchar(255) NOT NULL,
`area` int(11) NOT NULL,
`sales_price` int(11) NOT NULL,
`sales_status` varchar(255) NOT NULL,
PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salespeople` (
`salesperson_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
PRIMARY KEY (`salesperson_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_contracts` (
`contract_id` int(11) NOT NULL AUTO_INCREMENT,
`property_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
`contract_date` datetime NOT NULL,
`contract_price` int(11) NOT NULL,
`signing_time` datetime NOT NULL,
`contract_content` text NOT NULL,
PRIMARY KEY (`contract_id`),
FOREIGN KEY (`property_id`) REFERENCES `properties` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_tracking` (
`salesperson_id` int(11) NOT NULL,
`contract_id` int(11) NOT NULL,
`execution_status` varchar(255) NOT NULL,
PRIMARY KEY (`salesperson_id`, `contract_id`),
FOREIGN KEY (`salesperson_id`) REFERENCES `salespeople` (`salesperson_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`contract_id`) REFERENCES `sales_contracts` (`contract_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
