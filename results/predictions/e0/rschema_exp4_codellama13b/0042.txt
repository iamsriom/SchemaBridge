-- Schema for database: rschema_0042

CREATE TABLE `community` (
  `community_id` int(11) NOT NULL,
  `community_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `property_manager_id` int(11) NOT NULL,
  PRIMARY KEY (`community_id`),
  FOREIGN KEY (`property_manager_id`) REFERENCES `property_manager` (`property_manager_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resident` (
  `resident_id` int(11) NOT NULL,
  `community_id` int(11) NOT NULL,
  `resident_name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  `building_number` int(11) NOT NULL,
  PRIMARY KEY (`resident_id`),
  FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`building_number`) REFERENCES `building` (`building_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `building` (
  `building_number` int(11) NOT NULL,
  `building_name` varchar(255) NOT NULL,
  `building_location` varchar(255) NOT NULL,
  PRIMARY KEY (`building_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_order` (
  `maintenance_order_number` int(11) NOT NULL,
  `community_id` int(11) NOT NULL,
  `resident_id` int(11) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_order_number`),
  FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`resident_id`) REFERENCES `resident` (`resident_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `maintenance_personnel` (`maintenance_personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `maintenance_personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property_manager` (
  `property_manager_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`property_manager_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
