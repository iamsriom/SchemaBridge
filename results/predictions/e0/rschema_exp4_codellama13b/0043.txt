-- Schema for database: rschema_0043

CREATE TABLE `loan_applications` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `property_number` varchar(255) NOT NULL,
  `loan_amount` int(11) NOT NULL,
  `loan_term` int(11) NOT NULL,
  `approval_status` varchar(255) NOT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `applicant_phone_number` varchar(255) NOT NULL,
  `applicant_id_number` varchar(255) NOT NULL,
  `application_time` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval_tracking` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `approval_personnel` varchar(255) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_result` varchar(255) NOT NULL,
  `loan_application_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`loan_application_id`) REFERENCES `loan_applications` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_results` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `approver` varchar(255) NOT NULL,
  `application_number` int(11) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_result` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`loan_application_id`) REFERENCES `loan_applications` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
