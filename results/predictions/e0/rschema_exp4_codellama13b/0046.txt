-- Schema for database: rschema_0046

CREATE TABLE `members` (
  `member_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `current_points` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shopping_records` (
  `shopping_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  `shopping_time` datetime NOT NULL,
  `total_price` decimal(10,2) NOT NULL,
  `points_change` int(11) NOT NULL,
  PRIMARY KEY (`shopping_id`),
  FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `point_redemption_records` (
  `redemption_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  `redemption_time` datetime NOT NULL,
  `redeemed_product_name` varchar(255) NOT NULL,
  `redemption_quantity` int(11) NOT NULL,
  `points_consumed` int(11) NOT NULL,
  PRIMARY KEY (`redemption_id`),
  FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shopping_records_by_member` (
  `member_id` int(11) NOT NULL,
  `shopping_record_id` int(11) NOT NULL,
  PRIMARY KEY (`member_id`, `shopping_record_id`),
  FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`shopping_record_id`) REFERENCES `shopping_records` (`shopping_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `point_redemption_records_by_member` (
  `member_id` int(11) NOT NULL,
  `redemption_record_id` int(11) NOT NULL,
  PRIMARY KEY (`member_id`, `redemption_record_id`),
  FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`redemption_record_id`) REFERENCES `point_redemption_records` (`redemption_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
