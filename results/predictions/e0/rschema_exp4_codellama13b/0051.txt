-- Schema for database: rschema_0051

CREATE TABLE `reservoirs` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rivers` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_activities` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`reservoir_id` int(11) NOT NULL,
`river_id` int(11) NOT NULL,
`monitoring_time` datetime NOT NULL,
`monitoring_point` varchar(255) NOT NULL,
`water_quality_indicators` varchar(255) NOT NULL,
`water_volume_data` varchar(255) NOT NULL,
`pollution_level` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`reservoir_id`) REFERENCES `reservoirs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`river_id`) REFERENCES `rivers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_activities` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`reservoir_id` int(11) NOT NULL,
`river_id` int(11) NOT NULL,
`maintenance_time` datetime NOT NULL,
`maintenance_content` varchar(255) NOT NULL,
`maintenance_costs` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`reservoir_id`) REFERENCES `reservoirs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`river_id`) REFERENCES `rivers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel_monitoring_activities` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`personnel_id` int(11) NOT NULL,
`monitoring_activity_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`monitoring_activity_id`) REFERENCES `monitoring_activities` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel_maintenance_activities` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`personnel_id` int(11) NOT NULL,
`maintenance_activity_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`maintenance_activity_id`) REFERENCES `maintenance_activities` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
