-- Schema for database: rschema_0052

CREATE TABLE land_resources (
  id INT(11) NOT NULL AUTO_INCREMENT,
  location VARCHAR(255) NOT NULL,
  area INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE projects (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  purpose VARCHAR(255) NOT NULL,
  responsible_person VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE project_land_usage (
  id INT(11) NOT NULL AUTO_INCREMENT,
  project_id INT(11) NOT NULL,
  land_area INT(11) NOT NULL,
  years_of_usage INT(11) NOT NULL,
  current_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE land_changes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  land_resource_id INT(11) NOT NULL,
  change_type VARCHAR(255) NOT NULL,
  reason VARCHAR(255) NOT NULL,
  cost INT(11) NOT NULL,
  time_of_change DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (land_resource_id) REFERENCES land_resources(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE land_usage_records (
  id INT(11) NOT NULL AUTO_INCREMENT,
  project_id INT(11) NOT NULL,
  land_area INT(11) NOT NULL,
  years_of_usage INT(11) NOT NULL,
  current_status VARCHAR(255) NOT NULL,
  personnel VARCHAR(255) NOT NULL,
  timing DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
