-- Schema for database: rschema_0061

CREATE TABLE patents (
  patent_id INT(11) NOT NULL AUTO_INCREMENT,
  patent_number VARCHAR(255) NOT NULL,
  patent_name VARCHAR(255) NOT NULL,
  application_date DATETIME NOT NULL,
  status VARCHAR(255) NOT NULL,
  PRIMARY KEY (patent_id),
  UNIQUE KEY patent_number (patent_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inventors (
  inventor_id INT(11) NOT NULL AUTO_INCREMENT,
  work_number VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  gender VARCHAR(255) NOT NULL,
  telephone_number VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  PRIMARY KEY (inventor_id),
  UNIQUE KEY work_number (work_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE patent_applications (
  application_id INT(11) NOT NULL AUTO_INCREMENT,
  application_number VARCHAR(255) NOT NULL,
  patent_id INT(11) NOT NULL,
  inventor_id INT(11) NOT NULL,
  application_date DATETIME NOT NULL,
  PRIMARY KEY (application_id),
  UNIQUE KEY application_number (application_number),
  FOREIGN KEY (patent_id) REFERENCES patents (patent_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (inventor_id) REFERENCES inventors (inventor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
