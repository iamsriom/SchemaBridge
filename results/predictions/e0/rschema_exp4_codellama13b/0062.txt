-- Schema for database: rschema_0062

CREATE TABLE `documents` (
`document_id` int(11) NOT NULL,
`title` varchar(255) NOT NULL,
`creation_date` datetime NOT NULL,
`version_number` int(11) NOT NULL,
PRIMARY KEY (`document_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authors` (
`author_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`date_of_birth` datetime NOT NULL,
`gender` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`author_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `document_versions` (
`modification_id` int(11) NOT NULL,
`document_id` int(11) NOT NULL,
`version_number` int(11) NOT NULL,
`modification_date` datetime NOT NULL,
`modifying_author` int(11) NOT NULL,
`modification_content` varchar(255) NOT NULL,
PRIMARY KEY (`modification_id`),
FOREIGN KEY (`document_id`) REFERENCES `documents` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`modifying_author`) REFERENCES `authors` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
