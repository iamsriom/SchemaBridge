-- Schema for database: rschema_0063

CREATE TABLE `scientific_achievements` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `achievement_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `research_and_development_date` datetime NOT NULL,
  `technical_field` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `product_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `transformation_date` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `achievement_transformation_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `achievement_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `transformation_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`achievement_id`) REFERENCES `scientific_achievements` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`product_id`) REFERENCES `products` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
