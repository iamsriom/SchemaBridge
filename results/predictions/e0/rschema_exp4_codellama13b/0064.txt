-- Schema for database: rschema_0064

CREATE TABLE project (
  project_id INT(11) NOT NULL AUTO_INCREMENT,
  project_number VARCHAR(255) NOT NULL,
  project_name VARCHAR(255) NOT NULL,
  application_date DATETIME NOT NULL,
  application_unit VARCHAR(255) NOT NULL,
  PRIMARY KEY (project_id),
  UNIQUE KEY project_number (project_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE expert (
  expert_id INT(11) NOT NULL AUTO_INCREMENT,
  expert_number VARCHAR(255) NOT NULL,
  expert_name VARCHAR(255) NOT NULL,
  expert_unit VARCHAR(255) NOT NULL,
  expert_title VARCHAR(255) NOT NULL,
  PRIMARY KEY (expert_id),
  UNIQUE KEY expert_number (expert_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE project_review (
  project_review_id INT(11) NOT NULL AUTO_INCREMENT,
  project_id INT(11) NOT NULL,
  expert_id INT(11) NOT NULL,
  review_date DATETIME NOT NULL,
  review_opinion VARCHAR(255) NOT NULL,
  PRIMARY KEY (project_review_id),
  FOREIGN KEY (project_id) REFERENCES project (project_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (expert_id) REFERENCES expert (expert_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
