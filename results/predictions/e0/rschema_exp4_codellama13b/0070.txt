-- Schema for database: rschema_0070

CREATE TABLE `mines` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ores` (
  `ore_id` int(11) NOT NULL AUTO_INCREMENT,
  `ore_name` varchar(255) NOT NULL,
  `characteristics` varchar(255) NOT NULL,
  PRIMARY KEY (`ore_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_plants` (
  `processing_plant_id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_plant_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`processing_plant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ore_outputs` (
  `ore_output_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_id` int(11) NOT NULL,
  `ore_id` int(11) NOT NULL,
  `output_date` datetime NOT NULL,
  `output_quantity` int(11) NOT NULL,
  PRIMARY KEY (`ore_output_id`),
  FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ore_id`) REFERENCES `ores` (`ore_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ore_processings` (
  `ore_processing_id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_plant_id` int(11) NOT NULL,
  `ore_id` int(11) NOT NULL,
  `processing_date` datetime NOT NULL,
  `processing_quantity` int(11) NOT NULL,
  `processing_cost` int(11) NOT NULL,
  PRIMARY KEY (`ore_processing_id`),
  FOREIGN KEY (`processing_plant_id`) REFERENCES `processing_plants` (`processing_plant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ore_id`) REFERENCES `ores` (`ore_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mine_employees` (
  `mine_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  PRIMARY KEY (`mine_id`, `employee_id`),
  FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_plant_employees` (
  `processing_plant_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  PRIMARY KEY (`processing_plant_id`, `employee_id`),
  FOREIGN KEY (`processing_plant_id`) REFERENCES `processing_plants` (`processing_plant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
