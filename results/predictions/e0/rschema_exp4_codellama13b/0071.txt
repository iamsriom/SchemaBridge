-- Schema for database: rschema_0071

CREATE TABLE `safety_officers` (
`safety_officer_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`telephone_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`safety_officer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mines` (
`mine_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`person_in_charge` varchar(255) NOT NULL,
PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspections` (
`inspection_id` int(11) NOT NULL,
`inspection_date` datetime NOT NULL,
`inspection_content` varchar(255) NOT NULL,
`inspection_results` varchar(255) NOT NULL,
`inspector_id` int(11) NOT NULL,
`mine_id` int(11) NOT NULL,
PRIMARY KEY (`inspection_id`),
FOREIGN KEY (`inspector_id`) REFERENCES `safety_officers` (`safety_officer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
