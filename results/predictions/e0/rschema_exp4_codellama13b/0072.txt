-- Schema for database: rschema_0072

CREATE TABLE `pipelines` (
  `pipeline_id` int(11) NOT NULL,
  `starting_point` varchar(255) NOT NULL,
  `ending_point` varchar(255) NOT NULL,
  `length` int(11) NOT NULL,
  `maintenance_records` varchar(255) NOT NULL,
  PRIMARY KEY (`pipeline_id`),
  FOREIGN KEY (`pipeline_id`) REFERENCES `fields` (`field_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fields` (
  `field_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`field_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_records` (
  `maintenance_id` int(11) NOT NULL,
  `pipeline_id` int(11) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_cost` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`pipeline_id`) REFERENCES `pipelines` (`pipeline_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
