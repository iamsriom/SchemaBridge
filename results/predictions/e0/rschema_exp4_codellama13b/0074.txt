-- Schema for database: rschema_0074

CREATE TABLE agents (
  agent_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  hire_date DATE NOT NULL,
  contact_phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (agent_id)
);

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  id_card_number VARCHAR(255) NOT NULL,
  contact_phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
);

CREATE TABLE policies (
  policy_id INT(11) NOT NULL AUTO_INCREMENT,
  insurance_type VARCHAR(255) NOT NULL,
  insured_amount DECIMAL(10,2) NOT NULL,
  insurance_term VARCHAR(255) NOT NULL,
  premium DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (policy_id)
);

CREATE TABLE payments (
  payment_id INT(11) NOT NULL AUTO_INCREMENT,
  payment_amount DECIMAL(10,2) NOT NULL,
  payment_date DATE NOT NULL,
  payment_method VARCHAR(255) NOT NULL,
  PRIMARY KEY (payment_id)
);

CREATE TABLE medical_records (
  medical_record_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  medical_record_type VARCHAR(255) NOT NULL,
  medical_record_date DATE NOT NULL,
  PRIMARY KEY (medical_record_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE claim_records (
  claim_record_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  claim_record_type VARCHAR(255) NOT NULL,
  claim_record_date DATE NOT NULL,
  PRIMARY KEY (claim_record_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE agent_customer (
  agent_id INT(11) NOT NULL,
  customer_id INT(11) NOT NULL,
  PRIMARY KEY (agent_id, customer_id),
  FOREIGN KEY (agent_id) REFERENCES agents (agent_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE policy_customer (
  policy_id INT(11) NOT NULL,
  customer_id INT(11) NOT NULL,
  PRIMARY KEY (policy_id, customer_id),
  FOREIGN KEY (policy_id) REFERENCES policies (policy_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE payment_policy (
  payment_id INT(11) NOT NULL,
  policy_id INT(11) NOT NULL,
  PRIMARY KEY (payment_id, policy_id),
  FOREIGN KEY (payment_id) REFERENCES payments (payment_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (policy_id) REFERENCES policies (policy_id) ON DELETE CASCADE ON UPDATE CASCADE
);
