-- Schema for database: rschema_0075

CREATE TABLE analysts (
  analyst_id int(11) NOT NULL,
  name varchar(255) NOT NULL,
  date_of_hire datetime NOT NULL,
  contact_number varchar(255) NOT NULL,
  PRIMARY KEY (analyst_id)
);

CREATE TABLE stocks (
  stock_code int(11) NOT NULL,
  stock_name varchar(255) NOT NULL,
  current_price decimal(10,2) NOT NULL,
  PRIMARY KEY (stock_code)
);

CREATE TABLE analyst_stocks (
  analyst_id int(11) NOT NULL,
  stock_code int(11) NOT NULL,
  date_of_analysis datetime NOT NULL,
  results varchar(255) NOT NULL,
  reasoning varchar(255) NOT NULL,
  PRIMARY KEY (analyst_id, stock_code, date_of_analysis),
  FOREIGN KEY (analyst_id) REFERENCES analysts (analyst_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (stock_code) REFERENCES stocks (stock_code) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE historical_prices (
  stock_code int(11) NOT NULL,
  date datetime NOT NULL,
  closing_price decimal(10,2) NOT NULL,
  opening_price decimal(10,2) NOT NULL,
  highest_price decimal(10,2) NOT NULL,
  lowest_price decimal(10,2) NOT NULL,
  trading_volume int(11) NOT NULL,
  PRIMARY KEY (stock_code, date),
  FOREIGN KEY (stock_code) REFERENCES stocks (stock_code) ON DELETE CASCADE ON UPDATE CASCADE
);
