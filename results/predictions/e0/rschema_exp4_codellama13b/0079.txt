-- Schema for database: rschema_0079

CREATE TABLE `stocks` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`code` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`market` varchar(255) NOT NULL,
`owner` varchar(255) NOT NULL,
`created_at` datetime NOT NULL,
`updated_at` datetime NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `code` (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `funds` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`code` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`market` varchar(255) NOT NULL,
`owner` varchar(255) NOT NULL,
`created_at` datetime NOT NULL,
`updated_at` datetime NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `code` (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `holdings` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`stock_id` int(11) NOT NULL,
`fund_id` int(11) NOT NULL,
`amount` decimal(10,2) NOT NULL,
`shares` int(11) NOT NULL,
`last_trading_date` datetime NOT NULL,
`created_at` datetime NOT NULL,
`updated_at` datetime NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`stock_id`) REFERENCES `stocks` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`fund_id`) REFERENCES `funds` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trading_orders` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`out_id` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`created_at` datetime NOT NULL,
`trading_date` datetime NOT NULL,
`amount` decimal(10,2) NOT NULL,
`quantity` int(11) NOT NULL,
`service_fee` decimal(10,2) NOT NULL,
`status` varchar(255) NOT NULL,
`settled` tinyint(1) NOT NULL,
`created_at` datetime NOT NULL,
`updated_at` datetime NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `out_id` (`out_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`trading_order_id` int(11) NOT NULL,
`amount` decimal(10,2) NOT NULL,
`quantity` int(11) NOT NULL,
`service_fee` decimal(10,2) NOT NULL,
`created_at` datetime NOT NULL,
`updated_at` datetime NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`trading_order_id`) REFERENCES `trading_orders` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `statistics` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`market` varchar(255) NOT NULL,
`category` varchar(255) NOT NULL,
`subcategory` varchar(255) NOT NULL,
`period` varchar(255) NOT NULL,
`amount` decimal(10,2) NOT NULL,
`quantity` int(11) NOT NULL,
`created_at` datetime NOT NULL,
`updated_at` datetime NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
