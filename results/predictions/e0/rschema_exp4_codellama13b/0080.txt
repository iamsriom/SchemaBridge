-- Schema for database: rschema_0080

CREATE TABLE customers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE repair_workers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE parts (
  id INT(11) NOT NULL AUTO_INCREMENT,
  serial_number VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  supplier VARCHAR(255) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE repair_tasks (
  id INT(11) NOT NULL AUTO_INCREMENT,
  vehicle_model VARCHAR(255) NOT NULL,
  task_status VARCHAR(255) NOT NULL,
  customer_id INT(11) NOT NULL,
  repair_worker_id INT(11) NOT NULL,
  task_start_date DATETIME NOT NULL,
  completion_date DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (customer_id) REFERENCES customers (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (repair_worker_id) REFERENCES repair_workers (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE parts_needed (
  id INT(11) NOT NULL AUTO_INCREMENT,
  repair_task_id INT(11) NOT NULL,
  part_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (repair_task_id) REFERENCES repair_tasks (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (part_id) REFERENCES parts (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
