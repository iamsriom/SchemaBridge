-- Schema for database: rschema_0084

CREATE TABLE land_parcels (
  land_id INT(11) NOT NULL,
  land_location VARCHAR(255) NOT NULL,
  total_land_area INT(11) NOT NULL,
  PRIMARY KEY (land_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE projects (
  project_id INT(11) NOT NULL,
  project_name VARCHAR(255) NOT NULL,
  project_location VARCHAR(255) NOT NULL,
  purpose VARCHAR(255) NOT NULL,
  person_in_charge VARCHAR(255) NOT NULL,
  PRIMARY KEY (project_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE land_usage_records (
  land_usage_id INT(11) NOT NULL,
  land_id INT(11) NOT NULL,
  project_id INT(11) NOT NULL,
  used_land_area INT(11) NOT NULL,
  usage_duration INT(11) NOT NULL,
  current_status VARCHAR(255) NOT NULL,
  record_time DATETIME NOT NULL,
  recording_personnel VARCHAR(255) NOT NULL,
  PRIMARY KEY (land_usage_id),
  FOREIGN KEY (land_id) REFERENCES land_parcels (land_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (project_id) REFERENCES projects (project_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
