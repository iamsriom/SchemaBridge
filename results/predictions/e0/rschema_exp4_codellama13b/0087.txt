-- Schema for database: rschema_0087

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `user_description` varchar(255) DEFAULT NULL,
  `gender` varchar(255) DEFAULT NULL,
  `interest_area` varchar(255) DEFAULT NULL,
  `num_followers` int(11) NOT NULL DEFAULT 0,
  `num_following` int(11) NOT NULL DEFAULT 0,
  `user_creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `podcasts` (
  `podcast_id` int(11) NOT NULL AUTO_INCREMENT,
  `publisher_id` int(11) NOT NULL,
  `podcast_file_path` varchar(255) NOT NULL,
  `podcast_type` varchar(255) NOT NULL,
  `podcast_name` varchar(255) NOT NULL,
  `cover_image_path` varchar(255) NOT NULL,
  `num_visits` int(11) NOT NULL DEFAULT 0,
  `num_likes` int(11) NOT NULL DEFAULT 0,
  `creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `num_favorites` int(11) NOT NULL DEFAULT 0,
  `last_update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`podcast_id`),
  FOREIGN KEY (`publisher_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `playlists` (
  `playlist_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `playlist_name` varchar(255) NOT NULL,
  `num_visits` int(11) NOT NULL DEFAULT 0,
  `num_likes` int(11) NOT NULL DEFAULT 0,
  `creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `num_favorites` int(11) NOT NULL DEFAULT 0,
  `last_update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`playlist_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `podcast_playlist` (
  `podcast_id` int(11) NOT NULL,
  `playlist_id` int(11) NOT NULL,
  PRIMARY KEY (`podcast_id`, `playlist_id`),
  FOREIGN KEY (`podcast_id`) REFERENCES `podcasts` (`podcast_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`playlist_id`) REFERENCES `playlists` (`playlist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `podcast_user` (
  `podcast_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`podcast_id`, `user_id`),
  FOREIGN KEY (`podcast_id`) REFERENCES `podcasts` (`podcast_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `playlist_user` (
  `playlist_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`playlist_id`, `user_id`),
  FOREIGN KEY (`playlist_id`) REFERENCES `playlists` (`playlist_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
