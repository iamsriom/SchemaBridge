-- Schema for database: rschema_0092

CREATE TABLE users (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  identity_type VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  identity_status VARCHAR(255) NOT NULL,
  creation_time DATETIME NOT NULL,
  expiration_time DATETIME NOT NULL,
  superior_id INT(11) NOT NULL,
  PRIMARY KEY (user_id),
  FOREIGN KEY (superior_id) REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_hierarchy_adjustments (
  adjustment_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  adjustment_time DATETIME NOT NULL,
  previous_superior_id INT(11) NOT NULL,
  new_superior_id INT(11) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (adjustment_id),
  FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE operators (
  operator_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  permission VARCHAR(255) NOT NULL,
  PRIMARY KEY (operator_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE operator_permissions (
  operator_id INT(11) NOT NULL,
  user_id INT(11) NOT NULL,
  permission VARCHAR(255) NOT NULL,
  PRIMARY KEY (operator_id, user_id),
  FOREIGN KEY (operator_id) REFERENCES operators(operator_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE change_audit (
  change_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  adjustment_time DATETIME NOT NULL,
  previous_superior_id INT(11) NOT NULL,
  new_superior_id INT(11) NOT NULL,
  operator_id INT(11) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (change_id),
  FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (operator_id) REFERENCES operators(operator_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
