-- Schema for database: rschema_0101

CREATE TABLE `employees` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`date_of_birth` datetime NOT NULL,
`phone_number` varchar(255) NOT NULL,
`id_number` varchar(255) NOT NULL,
`place_of_origin` varchar(255) NOT NULL,
`age` int(11) NOT NULL,
`department` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`level` varchar(255) NOT NULL,
`contract_type` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`responsibilities` varchar(255) NOT NULL,
`head_id` int(11) NOT NULL,
`department_location` varchar(255) NOT NULL,
`total_people` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendance` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`employee_id` int(11) NOT NULL,
`clock_in_time` datetime NOT NULL,
`clock_in_type` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department_transfers` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`employee_id` int(11) NOT NULL,
`department_id` int(11) NOT NULL,
`transfer_time` datetime NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluations` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`employee_id` int(11) NOT NULL,
`evaluation_content` varchar(255) NOT NULL,
`evaluation_result` varchar(255) NOT NULL,
`evaluator` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
