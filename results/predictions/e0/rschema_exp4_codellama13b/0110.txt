-- Schema for database: rschema_0110

CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  email VARCHAR(255) DEFAULT NULL,
  phone VARCHAR(255) DEFAULT NULL,
  dob DATE DEFAULT NULL,
  avatar VARCHAR(255) DEFAULT NULL,
  created_at DATETIME NOT NULL,
  updated_at DATETIME DEFAULT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY username (username),
  UNIQUE KEY email (email),
  UNIQUE KEY phone (phone)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE updates (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  content TEXT NOT NULL,
  content_type VARCHAR(255) NOT NULL,
  created_at DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE followers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  follower_id INT(11) NOT NULL,
  followed_id INT(11) NOT NULL,
  follow_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (follower_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (followed_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE comments (
  id INT(11) NOT NULL AUTO_INCREMENT,
  update_id INT(11) NOT NULL,
  commenter_id INT(11) NOT NULL,
  content TEXT NOT NULL,
  comment_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (update_id) REFERENCES updates (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (commenter_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
