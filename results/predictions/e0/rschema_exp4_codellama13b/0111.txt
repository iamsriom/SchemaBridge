-- Schema for database: rschema_0111

CREATE TABLE `user` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
`registration_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`user_id`),
UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit` (
`credit_number` int(11) NOT NULL AUTO_INCREMENT,
`credit_type` varchar(255) NOT NULL,
`application_date` datetime NOT NULL,
`monthly_factor` decimal(10,2) NOT NULL,
`number_of_applications` int(11) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
`deletion_time` datetime NULL,
`user_id` int(11) NOT NULL,
`employee_id` int(11) NOT NULL,
`company_id` int(11) NOT NULL,
PRIMARY KEY (`credit_number`),
UNIQUE KEY `credit_number` (`credit_number`),
KEY `user_id` (`user_id`),
KEY `employee_id` (`employee_id`),
KEY `company_id` (`company_id`),
CONSTRAINT `credit_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `credit_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `credit_ibfk_3` FOREIGN KEY (`company_id`) REFERENCES `company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_record_status` (
`credit_record_status_id` int(11) NOT NULL AUTO_INCREMENT,
`credit_record_status` varchar(255) NOT NULL,
`credit_record_status_description` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`credit_record_status_id`),
UNIQUE KEY `credit_record_status` (`credit_record_status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `company` (
`company_id` int(11) NOT NULL AUTO_INCREMENT,
`company_name` varchar(255) NOT NULL,
`company_address` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`company_id`),
UNIQUE KEY `company_name` (`company_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
`employee_id` int(11) NOT NULL AUTO_INCREMENT,
`employee_name` varchar(255) NOT NULL,
`employee_position` varchar(255) NOT NULL,
`affiliated_company_id` int(11) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`employee_id`),
UNIQUE KEY `employee_name` (`employee_name`),
KEY `affiliated_company_id` (`affiliated_company_id`),
CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`affiliated_company_id`) REFERENCES `company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
