-- Schema for database: rschema_0112

CREATE TABLE `offices` (
`office_id` int(11) NOT NULL AUTO_INCREMENT,
`city` varchar(255) NOT NULL,
`street_address` varchar(255) NOT NULL,
`postal_code` varchar(255) NOT NULL,
`country` varchar(255) NOT NULL,
`office_contact_number` varchar(255) NOT NULL,
`office_fax_number` varchar(255) NOT NULL,
`manager_id` int(11) NOT NULL,
PRIMARY KEY (`office_id`),
FOREIGN KEY (`manager_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
`employee_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`hire_date` datetime NOT NULL,
`employee_number` varchar(255) NOT NULL,
`office_phone` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`supervisor_id` int(11) NOT NULL,
`base_salary` int(11) NOT NULL,
`commission_rate` int(11) NOT NULL,
PRIMARY KEY (`employee_id`),
FOREIGN KEY (`supervisor_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`company_name` varchar(255) NOT NULL,
`customer_id` varchar(255) NOT NULL,
`contact_person` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`street_address` varchar(255) NOT NULL,
`city` varchar(255) NOT NULL,
`state_province` varchar(255) NOT NULL,
`postal_code` varchar(255) NOT NULL,
`country` varchar(255) NOT NULL,
`sales_rep_id` int(11) NOT NULL,
`credit_limit` int(11) NOT NULL,
PRIMARY KEY (`customer_id`),
FOREIGN KEY (`sales_rep_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
`product_id` int(11) NOT NULL AUTO_INCREMENT,
`product_name` varchar(255) NOT NULL,
`model` varchar(255) NOT NULL,
`specifications` varchar(255) NOT NULL,
`description` varchar(255) NOT NULL,
`cost_price` int(11) NOT NULL,
`suggested_retail_price` int(11) NOT NULL,
`inventory_quantity` int(11) NOT NULL,
`supplier_id` int(11) NOT NULL,
`brand` varchar(255) NOT NULL,
PRIMARY KEY (`product_id`),
FOREIGN KEY (`supplier_id`) REFERENCES `suppliers` (`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `suppliers` (
`supplier_id` int(11) NOT NULL AUTO_INCREMENT,
`supplier_name` varchar(255) NOT NULL,
`contact_person` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`street_address` varchar(255) NOT NULL,
`city` varchar(255) NOT NULL,
`state_province` varchar(255) NOT NULL,
`postal_code` varchar(255) NOT NULL,
`country` varchar(255) NOT NULL,
PRIMARY KEY (`supplier_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
`order_id` int(11) NOT NULL AUTO_INCREMENT,
`order_date` datetime NOT NULL,
`requested_delivery_date` datetime NOT NULL,
`actual_shipping_date` datetime NOT NULL,
`customer_id` int(11) NOT NULL,
`order_status` varchar(255) NOT NULL,
`special_instructions` varchar(255) NOT NULL,
`notes` varchar(255) NOT NULL,
PRIMARY KEY (`order_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order_items` (
`order_item_id` int(11) NOT NULL AUTO_INCREMENT,
`order_id` int(11) NOT NULL,
`product_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`unit_price` int(11) NOT NULL,
`line_number` int(11) NOT NULL,
PRIMARY KEY (`order_item_id`),
FOREIGN KEY (`order_id`) REFERENCES `orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `payments` (
`payment_id` int(11) NOT NULL AUTO_INCREMENT,
`check_number` varchar(255) NOT NULL,
`payment_date` datetime NOT NULL,
`invoice_amount` int(11) NOT NULL,
`paying_customer_id` int(11) NOT NULL,
`clearing_date` datetime NOT NULL,
PRIMARY KEY (`payment_id`),
FOREIGN KEY (`paying_customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_performance` (
`salesperson_id` int(11) NOT NULL,
`fiscal_year` int(11) NOT NULL,
`total_sales` int(11) NOT NULL,
PRIMARY KEY (`salesperson_id`, `fiscal_year`),
FOREIGN KEY (`salesperson_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `accounts_receivable` (
`customer_id` int(11) NOT NULL,
`accounts_receivable` int(11) NOT NULL,
PRIMARY KEY (`customer_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
