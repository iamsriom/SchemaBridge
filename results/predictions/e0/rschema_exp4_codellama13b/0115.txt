-- Schema for database: rschema_0115

CREATE TABLE `units` (
`unit_id` int(11) NOT NULL AUTO_INCREMENT,
`unit_name` varchar(255) NOT NULL,
`benchmark_testing_type_id` int(11) NOT NULL,
PRIMARY KEY (`unit_id`),
FOREIGN KEY (`benchmark_testing_type_id`) REFERENCES `benchmark_testing_types` (`benchmark_testing_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `benchmark_testing_types` (
`benchmark_testing_type_id` int(11) NOT NULL AUTO_INCREMENT,
`benchmark_testing_type_name` varchar(255) NOT NULL,
PRIMARY KEY (`benchmark_testing_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
