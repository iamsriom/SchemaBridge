-- Schema for database: rschema_0116

CREATE TABLE `customer` (
`customer_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_route` (
`flight_route_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`route` varchar(255) NOT NULL,
`time` datetime NOT NULL,
`airline_id` int(11) NOT NULL,
PRIMARY KEY (`flight_route_id`),
FOREIGN KEY (`airline_id`) REFERENCES `airline` (`airline_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `evaluation` (
`evaluation_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
`flight_route_id` int(11) NOT NULL,
`content` varchar(255) NOT NULL,
`score` int(11) NOT NULL,
PRIMARY KEY (`evaluation_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`flight_route_id`) REFERENCES `flight_route` (`flight_route_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bank_account` (
`account_id` int(11) NOT NULL,
`account_number` varchar(255) NOT NULL,
`balance` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
PRIMARY KEY (`account_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
UNIQUE KEY `account_number` (`account_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airline` (
`airline_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`airline_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
