-- Schema for database: rschema_0118

CREATE TABLE `employees` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`age` int(11) NOT NULL,
`date_of_birth` datetime NOT NULL,
`contact_information` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`entry_time` datetime NOT NULL,
`salary` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`department_head` varchar(255) NOT NULL,
`department_responsibilities` varchar(255) NOT NULL,
`department_location` varchar(255) NOT NULL,
`total_number_of_department_personnel` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_department_association` (
`employee_id` int(11) NOT NULL,
`department_id` int(11) NOT NULL,
PRIMARY KEY (`employee_id`, `department_id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendance` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`employee_id` int(11) NOT NULL,
`attendance_type` varchar(255) NOT NULL,
`attendance_time` datetime NOT NULL,
`leave_type` varchar(255) NOT NULL,
`leave_time` datetime NOT NULL,
`overtime_type` varchar(255) NOT NULL,
`overtime_time` datetime NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authority_management` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`employee_id` int(11) NOT NULL,
`authority_level` varchar(255) NOT NULL,
`authority_description` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `enterprise_goal_setting` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`goal_name` varchar(255) NOT NULL,
`goal_description` varchar(255) NOT NULL,
`goal_status` varchar(255) NOT NULL,
`goal_progress` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_performance_appraisal` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`employee_id` int(11) NOT NULL,
`performance_appraisal_type` varchar(255) NOT NULL,
`performance_appraisal_description` varchar(255) NOT NULL,
`performance_appraisal_status` varchar(255) NOT NULL,
`performance_appraisal_progress` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
