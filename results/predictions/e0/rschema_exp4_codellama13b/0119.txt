-- Schema for database: rschema_0119
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `indicators` (
  `indicator_id` int(11) NOT NULL AUTO_INCREMENT,
  `indicator_name` varchar(255) NOT NULL,
  `indicator_target` int(11) NOT NULL,
  `indicator_description` varchar(255) NOT NULL,
  `calculation_method` varchar(255) NOT NULL,
  `unit` varchar(255) NOT NULL,
  PRIMARY KEY (`indicator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_indicator_association` (
  `user_id` int(11) NOT NULL,
  `indicator_id` int(11) NOT NULL,
  `allocation_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`, `indicator_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`indicator_id`) REFERENCES `indicators` (`indicator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `indicator_data_record` (
  `user_id` int(11) NOT NULL,
  `indicator_id` int(11) NOT NULL,
  `submission_record_time` datetime NOT NULL,
  `remarks` varchar(255) NOT NULL,
  `improvement_relative_to_last_record` int(11) NOT NULL,
  `recorded_indicator_value` int(11) NOT NULL,
  PRIMARY KEY (`user_id`, `indicator_id`, `submission_record_time`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`indicator_id`) REFERENCES `indicators` (`indicator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
