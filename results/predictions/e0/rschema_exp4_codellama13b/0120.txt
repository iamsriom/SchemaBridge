-- Schema for database: rschema_0120

CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  is_activated TINYINT(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  UNIQUE KEY username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE characters (
  id INT(11) NOT NULL AUTO_INCREMENT,
  character_name VARCHAR(255) NOT NULL,
  character_id INT(11) NOT NULL,
  corporation_id INT(11) NOT NULL,
  is_active TINYINT(1) NOT NULL DEFAULT 0,
  is_default TINYINT(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  FOREIGN KEY (corporation_id) REFERENCES corporations (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE corporations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  corporation_name VARCHAR(255) NOT NULL,
  ticker VARCHAR(255) NOT NULL,
  ceo_id INT(11) NOT NULL,
  api_key VARCHAR(255) NULL,
  PRIMARY KEY (id),
  UNIQUE KEY ticker (ticker)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE login_attempts (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  source_address VARCHAR(255) NOT NULL,
  browser_agent VARCHAR(255) NOT NULL,
  login_result VARCHAR(255) NOT NULL,
  timestamp DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (username) REFERENCES users (username) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE third_party_applications (
  id INT(11) NOT NULL AUTO_INCREMENT,
  application_name VARCHAR(255) NOT NULL,
  maintainer VARCHAR(255) NOT NULL,
  key VARCHAR(255) NOT NULL,
  callback_url VARCHAR(255) NOT NULL,
  is_active TINYINT(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  UNIQUE KEY application_name (application_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
