-- Schema for database: rschema_0121

CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  full_name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  registration_date DATETIME NOT NULL,
  last_login_date DATETIME NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY username (username),
  UNIQUE KEY email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_registrations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  registration_date DATETIME NOT NULL,
  last_login_date DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_authentications (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  authentication_level VARCHAR(255) NOT NULL,
  authentication_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_migrations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  old_user_id INT(11) NOT NULL,
  new_user_id INT(11) NOT NULL,
  migration_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (old_user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (new_user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE authenticated_consumers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE consumer_configurations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  consumer_id INT(11) NOT NULL,
  configuration VARCHAR(255) NOT NULL,
  permissions VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (consumer_id) REFERENCES authenticated_consumers (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sessions (
  id INT(11) NOT NULL AUTO_INCREMENT,
  consumer_id INT(11) NOT NULL,
  expiration_time DATETIME NOT NULL,
  mfa_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (consumer_id) REFERENCES authenticated_consumers (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE services (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  url VARCHAR(255) NOT NULL,
  configuration VARCHAR(255) NOT NULL,
  last_heartbeat_time DATETIME NOT NULL,
  monitoring_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE worker_nodes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  service_id INT(11) NOT NULL,
  node_name VARCHAR(255) NOT NULL,
  status VARCHAR(255) NOT NULL,
  operating_system VARCHAR(255) NOT NULL,
  architecture VARCHAR(255) NOT NULL,
  last_heartbeat_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (service_id) REFERENCES services (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
