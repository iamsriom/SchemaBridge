-- Schema for database: rschema_0122

CREATE TABLE `conversation` (
`conversation_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
`organization_id` int(11) NOT NULL,
`creation_time` datetime NOT NULL,
`last_update_time` datetime NOT NULL,
`base_id` int(11) NOT NULL,
PRIMARY KEY (`conversation_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`organization_id`) REFERENCES `organization` (`organization_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inquiry` (
`inquiry_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
`status` varchar(255) NOT NULL,
`category` varchar(255) NOT NULL,
`content` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
PRIMARY KEY (`inquiry_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organization` (
`organization_id` int(11) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`avatar` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`description` varchar(255) NOT NULL,
`homepage_link` varchar(255) NOT NULL,
`image_link` varchar(255) NOT NULL,
PRIMARY KEY (`organization_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
`customer_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conversation_customer` (
`conversation_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
PRIMARY KEY (`conversation_id`, `customer_id`),
FOREIGN KEY (`conversation_id`) REFERENCES `conversation` (`conversation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inquiry_customer` (
`inquiry_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
PRIMARY KEY (`inquiry_id`, `customer_id`),
FOREIGN KEY (`inquiry_id`) REFERENCES `inquiry` (`inquiry_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conversation_organization` (
`conversation_id` int(11) NOT NULL,
`organization_id` int(11) NOT NULL,
PRIMARY KEY (`conversation_id`, `organization_id`),
FOREIGN KEY (`conversation_id`) REFERENCES `conversation` (`conversation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`organization_id`) REFERENCES `organization` (`organization_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inquiry_organization` (
`inquiry_id` int(11) NOT NULL,
`organization_id` int(11) NOT NULL,
PRIMARY KEY (`inquiry_id`, `organization_id`),
FOREIGN KEY (`inquiry_id`) REFERENCES `inquiry` (`inquiry_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`organization_id`) REFERENCES `organization` (`organization_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `base_record` (
`base_id` int(11) NOT NULL,
`encrypted_information` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`base_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conversation_base_record` (
`conversation_id` int(11) NOT NULL,
`base_id` int(11) NOT NULL,
PRIMARY KEY (`conversation_id`, `base_id`),
FOREIGN KEY (`conversation_id`) REFERENCES `conversation` (`conversation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`base_id`) REFERENCES `base_record` (`base_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
