-- Schema for database: rschema_0126

CREATE TABLE `deceased` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`unique_id` varchar(255) NOT NULL,
`inspector_id` int(11) NOT NULL,
`death_time` datetime NOT NULL,
`last_modified` datetime NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`inspector_id`) REFERENCES `inspectors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspectors` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`unique_id` varchar(255) NOT NULL,
`department` varchar(255) NOT NULL,
`last_modified` datetime NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `death_records` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`deceased_id` int(11) NOT NULL,
`inspector_id` int(11) NOT NULL,
`death_time` datetime NOT NULL,
`last_modified` datetime NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`deceased_id`) REFERENCES `deceased` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`inspector_id`) REFERENCES `inspectors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `death_record_stats` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`deceased_id` int(11) NOT NULL,
`inspector_id` int(11) NOT NULL,
`death_time` datetime NOT NULL,
`last_modified` datetime NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`deceased_id`) REFERENCES `deceased` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`inspector_id`) REFERENCES `inspectors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
