-- Schema for database: rschema_0128

CREATE TABLE task_execution (
  execution_id int(11) NOT NULL AUTO_INCREMENT,
  start_time datetime NOT NULL,
  end_time datetime DEFAULT NULL,
  task_name varchar(255) NOT NULL,
  exit_code int(11) DEFAULT NULL,
  exit_message varchar(255) DEFAULT NULL,
  error_message varchar(255) DEFAULT NULL,
  last_update_time datetime NOT NULL,
  external_execution_id int(11) DEFAULT NULL,
  parent_task_execution_id int(11) DEFAULT NULL,
  PRIMARY KEY (execution_id),
  FOREIGN KEY (external_execution_id) REFERENCES external_execution (execution_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (parent_task_execution_id) REFERENCES task_execution (execution_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE task_parameter (
  execution_id int(11) NOT NULL,
  parameter_name varchar(255) NOT NULL,
  parameter_value varchar(255) NOT NULL,
  PRIMARY KEY (execution_id, parameter_name),
  FOREIGN KEY (execution_id) REFERENCES task_execution (execution_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE batch_execution (
  batch_id int(11) NOT NULL AUTO_INCREMENT,
  batch_name varchar(255) NOT NULL,
  start_time datetime NOT NULL,
  end_time datetime DEFAULT NULL,
  PRIMARY KEY (batch_id),
  FOREIGN KEY (batch_id) REFERENCES task_execution (execution_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE batch_task (
  batch_id int(11) NOT NULL,
  task_execution_id int(11) NOT NULL,
  PRIMARY KEY (batch_id, task_execution_id),
  FOREIGN KEY (batch_id) REFERENCES batch_execution (batch_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (task_execution_id) REFERENCES task_execution (execution_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
