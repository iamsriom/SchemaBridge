-- Schema for database: rschema_0130

CREATE TABLE clients (
  id INT(11) NOT NULL AUTO_INCREMENT,
  corporate_tax_id VARCHAR(255) NOT NULL,
  corporate_name VARCHAR(255) NOT NULL,
  postal_code VARCHAR(255) NOT NULL,
  corporate_size VARCHAR(255) NOT NULL,
  corporate_type VARCHAR(255) NOT NULL,
  person_name VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  personal_tax_id VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (corporate_tax_id, person_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE projects (
  id INT(11) NOT NULL AUTO_INCREMENT,
  project_value DECIMAL(10,2) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  project_status VARCHAR(255) NOT NULL,
  service_type VARCHAR(255) NOT NULL,
  is_digital BOOLEAN NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE project_members (
  id INT(11) NOT NULL AUTO_INCREMENT,
  project_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  domain VARCHAR(255) NOT NULL,
  date_of_hire DATE NOT NULL,
  date_of_departure DATE NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE client_projects (
  id INT(11) NOT NULL AUTO_INCREMENT,
  client_id INT(11) NOT NULL,
  project_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (client_id) REFERENCES clients(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
