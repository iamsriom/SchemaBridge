-- Schema for database: rschema_0133

CREATE TABLE `books` (
`book_id` int(11) NOT NULL,
`title` varchar(255) NOT NULL,
`author` varchar(255) NOT NULL,
`publisher` varchar(255) NOT NULL,
`publication_date` datetime NOT NULL,
`price` decimal(10,2) NOT NULL,
`book_type` int(11) NOT NULL,
`content_description` text NOT NULL,
PRIMARY KEY (`book_id`),
FOREIGN KEY (`book_type`) REFERENCES `book_category` (`book_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `book_category` (
`book_type_id` int(11) NOT NULL,
`type_name` varchar(255) NOT NULL,
`description` text NOT NULL,
PRIMARY KEY (`book_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `readers` (
`reader_id` int(11) NOT NULL,
`reader_type` int(11) NOT NULL,
`contact_number` varchar(255) NOT NULL,
`credit_score` int(11) NOT NULL,
`description` text NOT NULL,
PRIMARY KEY (`reader_id`),
FOREIGN KEY (`reader_type`) REFERENCES `reader_category` (`reader_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reader_category` (
`reader_type_id` int(11) NOT NULL,
`type_name` varchar(255) NOT NULL,
`description` text NOT NULL,
PRIMARY KEY (`reader_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `borrowing` (
`book_id` int(11) NOT NULL,
`reader_id` int(11) NOT NULL,
`borrow_date` datetime NOT NULL,
`return_date` datetime NOT NULL,
`renewal_count` int(11) NOT NULL,
`fine` decimal(10,2) NOT NULL,
PRIMARY KEY (`book_id`, `reader_id`),
FOREIGN KEY (`book_id`) REFERENCES `books` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`reader_id`) REFERENCES `readers` (`reader_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
