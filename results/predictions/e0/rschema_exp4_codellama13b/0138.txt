-- Schema for database: rschema_0138

CREATE TABLE `employees` (
`employee_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`performance` varchar(255) NOT NULL,
PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stores` (
`store_id` int(11) NOT NULL,
`store_name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
PRIMARY KEY (`store_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
`product_id` int(11) NOT NULL,
`product_name` varchar(255) NOT NULL,
`specifications` varchar(255) NOT NULL,
`unit_price` int(11) NOT NULL,
PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employment_information` (
`employee_id` int(11) NOT NULL,
`employment_period` varchar(255) NOT NULL,
`monthly_salary` int(11) NOT NULL,
PRIMARY KEY (`employee_id`),
FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_information` (
`product_id` int(11) NOT NULL,
`monthly_sales_volume` int(11) NOT NULL,
PRIMARY KEY (`product_id`),
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
