-- Schema for database: rschema_0139

CREATE TABLE `departments` (
`department_id` int(11) NOT NULL AUTO_INCREMENT,
`department_name` varchar(255) NOT NULL,
`floor` int(11) NOT NULL,
PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctors` (
`doctor_id` int(11) NOT NULL AUTO_INCREMENT,
`doctor_number` varchar(255) NOT NULL,
`doctor_name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`title` varchar(255) NOT NULL,
PRIMARY KEY (`doctor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patients` (
`patient_id` int(11) NOT NULL AUTO_INCREMENT,
`patient_name` varchar(255) NOT NULL,
`age` int(11) NOT NULL,
`gender` varchar(255) NOT NULL,
PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `diagnoses` (
`diagnosis_id` int(11) NOT NULL AUTO_INCREMENT,
`doctor_id` int(11) NOT NULL,
`patient_id` int(11) NOT NULL,
`diagnosis_date` datetime NOT NULL,
`diagnosis_description` varchar(255) NOT NULL,
PRIMARY KEY (`diagnosis_id`),
FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`patient_id`) REFERENCES `patients` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `prescriptions` (
`prescription_id` int(11) NOT NULL AUTO_INCREMENT,
`doctor_id` int(11) NOT NULL,
`patient_id` int(11) NOT NULL,
`prescription_date` datetime NOT NULL,
`prescription_description` varchar(255) NOT NULL,
PRIMARY KEY (`prescription_id`),
FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`patient_id`) REFERENCES `patients` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drugs` (
`drug_id` int(11) NOT NULL AUTO_INCREMENT,
`drug_name` varchar(255) NOT NULL,
`price` int(11) NOT NULL,
`manufacturer` varchar(255) NOT NULL,
PRIMARY KEY (`drug_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drug_lists` (
`drug_list_id` int(11) NOT NULL AUTO_INCREMENT,
`prescription_id` int(11) NOT NULL,
`drug_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
PRIMARY KEY (`drug_list_id`),
FOREIGN KEY (`prescription_id`) REFERENCES `prescriptions` (`prescription_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`drug_id`) REFERENCES `drugs` (`drug_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
