-- Schema for database: rschema_0142

CREATE TABLE coaches (
  coach_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  age INT(11) NOT NULL,
  PRIMARY KEY (coach_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE teams (
  team_name VARCHAR(255) NOT NULL,
  representative_region VARCHAR(255) NOT NULL,
  establishment_date DATE NOT NULL,
  PRIMARY KEY (team_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE players (
  player_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  weight INT(11) NOT NULL,
  height INT(11) NOT NULL,
  joining_date DATE NOT NULL,
  PRIMARY KEY (player_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE coach_team_relationship (
  coach_id INT(11) NOT NULL,
  team_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (coach_id, team_name),
  FOREIGN KEY (coach_id) REFERENCES coaches (coach_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (team_name) REFERENCES teams (team_name) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE player_team_relationship (
  player_id INT(11) NOT NULL,
  team_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (player_id, team_name),
  FOREIGN KEY (player_id) REFERENCES players (player_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (team_name) REFERENCES teams (team_name) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
