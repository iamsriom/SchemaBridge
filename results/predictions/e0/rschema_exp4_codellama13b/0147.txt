-- Schema for database: rschema_0147

CREATE TABLE `class` (
`class_id` int(11) NOT NULL,
`class_name` varchar(255) NOT NULL,
`department_name` varchar(255) NOT NULL,
`grade` int(11) NOT NULL,
PRIMARY KEY (`class_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member` (
`member_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`age` int(11) NOT NULL,
`workplace` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcement` (
`announcement_id` int(11) NOT NULL,
`announcement_time` datetime NOT NULL,
`announcement_content` varchar(255) NOT NULL,
PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `message` (
`message_id` int(11) NOT NULL,
`sender_id` int(11) NOT NULL,
`receiver_id` int(11) NOT NULL,
`message_content` varchar(255) NOT NULL,
`send_time` datetime NOT NULL,
`read_status` varchar(255) NOT NULL,
PRIMARY KEY (`message_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `log` (
`log_id` int(11) NOT NULL,
`log_content` varchar(255) NOT NULL,
`time` datetime NOT NULL,
`mood` varchar(255) NOT NULL,
`member_id` int(11) NOT NULL,
PRIMARY KEY (`log_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `leave` (
`leave_id` int(11) NOT NULL,
`message_id` int(11) NOT NULL,
`message_time` datetime NOT NULL,
`message_content` varchar(255) NOT NULL,
`message_count` int(11) NOT NULL,
PRIMARY KEY (`leave_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `application` (
`application_id` int(11) NOT NULL,
`application_time` datetime NOT NULL,
`approval_status` varchar(255) NOT NULL,
`application_content` varchar(255) NOT NULL,
PRIMARY KEY (`application_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
