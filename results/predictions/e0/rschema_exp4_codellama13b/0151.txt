-- Schema for database: rschema_0151

CREATE TABLE `customers` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`id_number` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `savings_banks` (
`savings_bank_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
PRIMARY KEY (`savings_bank_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `deposits` (
`deposit_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_id` int(11) NOT NULL,
`savings_bank_id` int(11) NOT NULL,
`deposit_amount` decimal(10,2) NOT NULL,
`expiration_date` datetime NOT NULL,
`password` varchar(255) NOT NULL,
`interest_rate` decimal(10,2) NOT NULL,
`type` varchar(255) NOT NULL,
PRIMARY KEY (`deposit_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`savings_bank_id`) REFERENCES `savings_banks` (`savings_bank_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `withdrawals` (
`withdrawal_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_id` int(11) NOT NULL,
`savings_bank_id` int(11) NOT NULL,
`withdrawal_amount` decimal(10,2) NOT NULL,
`expiration_date` datetime NOT NULL,
`password` varchar(255) NOT NULL,
`interest_rate` decimal(10,2) NOT NULL,
`type` varchar(255) NOT NULL,
PRIMARY KEY (`withdrawal_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`savings_bank_id`) REFERENCES `savings_banks` (`savings_bank_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
