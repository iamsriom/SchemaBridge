-- Schema for database: rschema_0157

CREATE TABLE `users` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`student_number` varchar(255) NOT NULL,
`student_name` varchar(255) NOT NULL,
`student_gender` varchar(255) NOT NULL,
`student_class` varchar(255) NOT NULL,
`student_branch` varchar(255) NOT NULL,
`student_dormitory` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `schools` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`school_number` varchar(255) NOT NULL,
`school_name` varchar(255) NOT NULL,
`school_introduction` varchar(255) NOT NULL,
`school_facilities` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `branches` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`branch_number` varchar(255) NOT NULL,
`branch_name` varchar(255) NOT NULL,
`branch_introduction` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classes` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`class_number` varchar(255) NOT NULL,
`class_name` varchar(255) NOT NULL,
`class_introduction` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teachers` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`teacher_number` varchar(255) NOT NULL,
`teacher_name` varchar(255) NOT NULL,
`teacher_title` varchar(255) NOT NULL,
`teacher_profile` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courses` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`course_number` varchar(255) NOT NULL,
`course_name` varchar(255) NOT NULL,
`course_description` varchar(255) NOT NULL,
`course_credits` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitories` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`dormitory_number` varchar(255) NOT NULL,
`dormitory_building` varchar(255) NOT NULL,
`dormitory_room_number` varchar(255) NOT NULL,
`dormitory_beds` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_schedules` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`course_id` int(11) NOT NULL,
`student_id` int(11) NOT NULL,
`schedule_date` datetime NOT NULL,
`schedule_time` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`course_id`) REFERENCES `courses` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`student_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrators` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`administrator_name` varchar(255) NOT NULL,
`administrator_password` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
