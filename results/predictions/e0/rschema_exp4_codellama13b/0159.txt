-- Schema for database: rschema_0159

CREATE TABLE `routes` (
`route_id` int(11) NOT NULL AUTO_INCREMENT,
`route_number` varchar(255) NOT NULL,
`start_point` varchar(255) NOT NULL,
`end_point` varchar(255) NOT NULL,
`passing_stations` varchar(255) NOT NULL,
PRIMARY KEY (`route_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
`vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
`vehicle_number` varchar(255) NOT NULL,
`vehicle_type` varchar(255) NOT NULL,
`vehicle_status` varchar(255) NOT NULL,
PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `allocations` (
`allocation_id` int(11) NOT NULL AUTO_INCREMENT,
`route_id` int(11) NOT NULL,
`vehicle_id` int(11) NOT NULL,
`driver_id` int(11) NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
PRIMARY KEY (`allocation_id`),
FOREIGN KEY (`route_id`) REFERENCES `routes` (`route_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`driver_id`) REFERENCES `drivers` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drivers` (
`driver_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`id_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`driver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcements` (
`announcement_id` int(11) NOT NULL AUTO_INCREMENT,
`title` varchar(255) NOT NULL,
`content` varchar(255) NOT NULL,
`release_time` datetime NOT NULL,
PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `passengers` (
`passenger_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
`registration_information` varchar(255) NOT NULL,
PRIMARY KEY (`passenger_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mobile_terminals` (
`mobile_terminal_id` int(11) NOT NULL AUTO_INCREMENT,
`app_id` varchar(255) NOT NULL,
`app_name` varchar(255) NOT NULL,
`app_version` varchar(255) NOT NULL,
PRIMARY KEY (`mobile_terminal_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `statistics` (
`statistics_id` int(11) NOT NULL AUTO_INCREMENT,
`route_id` int(11) NOT NULL,
`vehicle_id` int(11) NOT NULL,
`driver_id` int(11) NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
`passenger_count` int(11) NOT NULL,
PRIMARY KEY (`statistics_id`),
FOREIGN KEY (`route_id`) REFERENCES `routes` (`route_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`driver_id`) REFERENCES `drivers` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
