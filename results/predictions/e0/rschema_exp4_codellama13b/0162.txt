CREATE TABLE `tower_base_stations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `collection_frequency` int(11) NOT NULL,
  `sensor_type` varchar(255) NOT NULL,
  `sensor_id` int(11) NOT NULL,
  `installation_location` varchar(255) NOT NULL,
  `collection_frequency` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sensor_data` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `tower_base_station_id` int(11) NOT NULL,
  `sensor_id` int(11) NOT NULL,
  `data` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`tower_base_station_id`) REFERENCES `tower_base_stations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `humidity` float NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`tower_base_station_id`) REFERENCES `tower_base_stations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
