-- Schema for database: rschema_0163

CREATE TABLE `region` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`area` int(11) NOT NULL,
`soil_type` varchar(255) NOT NULL,
`person_in_charge_id` int(11) NOT NULL,
`soil_ph` float NOT NULL,
`organic_matter_content` float NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`person_in_charge_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `variety` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`suitable_climate` varchar(255) NOT NULL,
`maturity_cycle` int(11) NOT NULL,
`suitable_soil` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`working_hours` int(11) NOT NULL,
`telephone_number` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `planting_record` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`plot_id` int(11) NOT NULL,
`variety_id` int(11) NOT NULL,
`planting_time` datetime NOT NULL,
`planting_quantity` int(11) NOT NULL,
`planting_method` varchar(255) NOT NULL,
`planter_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`plot_id`) REFERENCES `region` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`variety_id`) REFERENCES `variety` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`planter_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fertilization_record` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`person_in_charge_id` int(11) NOT NULL,
`plot_id` int(11) NOT NULL,
`fertilization_time` datetime NOT NULL,
`fertilization_type` varchar(255) NOT NULL,
`fertilization_amount` float NOT NULL,
`fertilization_method` varchar(255) NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`person_in_charge_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`plot_id`) REFERENCES `region` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
