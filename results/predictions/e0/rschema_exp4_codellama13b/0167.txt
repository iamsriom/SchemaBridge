-- Schema for database: rschema_0167

CREATE TABLE `natural_disaster_data` (
`data_id` int(11) NOT NULL,
`data_type` varchar(255) NOT NULL,
`data_source` varchar(255) NOT NULL,
`upload_time` datetime NOT NULL,
`data_content` text NOT NULL,
`data_format` varchar(255) NOT NULL,
`layer_type` varchar(255) NOT NULL,
PRIMARY KEY (`data_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regional_data` (
`region_id` int(11) NOT NULL,
`region_name` varchar(255) NOT NULL,
`administrative_division` varchar(255) NOT NULL,
`latitude` decimal(10,6) NOT NULL,
`longitude` decimal(10,6) NOT NULL,
`area` decimal(10,2) NOT NULL,
`disaster_risk_level` int(11) NOT NULL,
PRIMARY KEY (`region_id`),
FOREIGN KEY (`region_id`) REFERENCES `natural_disaster_data` (`data_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department_data` (
`department_id` int(11) NOT NULL,
`department_name` varchar(255) NOT NULL,
`person_in_charge` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
`functional_scope` varchar(255) NOT NULL,
PRIMARY KEY (`department_id`),
FOREIGN KEY (`department_id`) REFERENCES `regional_data` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_data` (
`user_id` int(11) NOT NULL,
`username` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
`permission_level` int(11) NOT NULL,
PRIMARY KEY (`user_id`),
FOREIGN KEY (`user_id`) REFERENCES `department_data` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
