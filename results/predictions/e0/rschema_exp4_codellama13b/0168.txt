-- Schema for database: rschema_0168

CREATE TABLE equipment (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  manufacturer VARCHAR(255) NOT NULL,
  installation_date DATE NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE operational_state (
  id INT(11) NOT NULL AUTO_INCREMENT,
  equipment_id INT(11) NOT NULL,
  timestamp DATETIME NOT NULL,
  temperature FLOAT(11) NOT NULL,
  vibration FLOAT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (equipment_id) REFERENCES equipment (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
