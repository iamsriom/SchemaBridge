-- Schema for database: rschema_0172

CREATE TABLE `patient_vital_signs` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `temperature` decimal(10,2) NOT NULL,
  `blood_pressure` decimal(10,2) NOT NULL,
  `heart_rate` int(11) NOT NULL,
  `other_data` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`log_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alerts` (
  `alert_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `abnormal_vital_signs` varchar(255) NOT NULL,
  `alert_time` datetime NOT NULL,
  `alert_content` varchar(255) NOT NULL,
  PRIMARY KEY (`alert_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor_normal_ranges` (
  `doctor_id` int(11) NOT NULL,
  `vital_sign` varchar(255) NOT NULL,
  `normal_range` decimal(10,2) NOT NULL,
  PRIMARY KEY (`doctor_id`, `vital_sign`),
  FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patients` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctors` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specialty` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
