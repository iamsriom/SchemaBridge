-- Schema for database: rschema_0174

CREATE TABLE `patients` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_record_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `id_card_number` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `admission_time` datetime NOT NULL,
  PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_signs` (
  `vital_sign_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `vital_sign` varchar(255) NOT NULL,
  `value` varchar(255) NOT NULL,
  `normal_range_lower` varchar(255) NOT NULL,
  `normal_range_upper` varchar(255) NOT NULL,
  PRIMARY KEY (`vital_sign_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_records` (
  `medical_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `diagnosis_time` datetime NOT NULL,
  `diagnosis_content` varchar(255) NOT NULL,
  PRIMARY KEY (`medical_record_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `treatment_recommendations` (
  `treatment_recommendation_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `treatment_recommendation` varchar(255) NOT NULL,
  PRIMARY KEY (`treatment_recommendation_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
