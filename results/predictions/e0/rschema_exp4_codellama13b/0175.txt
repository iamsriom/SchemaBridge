-- Schema for database: rschema_0175

CREATE TABLE alumni (
  alumni_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  employer VARCHAR(255) NOT NULL,
  job_title VARCHAR(255) NOT NULL,
  department VARCHAR(255) NOT NULL,
  class VARCHAR(255) NOT NULL,
  enrollment_year INT(11) NOT NULL,
  id_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (alumni_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE donation (
  donation_id INT(11) NOT NULL AUTO_INCREMENT,
  donating_alumni_id INT(11) NOT NULL,
  donation_time DATETIME NOT NULL,
  donation_amount DECIMAL(10,2) NOT NULL,
  donation_type VARCHAR(255) NOT NULL,
  donation_usage VARCHAR(255) NOT NULL,
  PRIMARY KEY (donation_id),
  FOREIGN KEY (donating_alumni_id) REFERENCES alumni(alumni_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE project (
  project_id INT(11) NOT NULL AUTO_INCREMENT,
  project_name VARCHAR(255) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  PRIMARY KEY (project_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE project_beneficiary (
  project_id INT(11) NOT NULL,
  beneficiary_unit_id INT(11) NOT NULL,
  benefit_percentage DECIMAL(10,2) NOT NULL,
  benefit_amount DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (project_id, beneficiary_unit_id),
  FOREIGN KEY (project_id) REFERENCES project(project_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (beneficiary_unit_id) REFERENCES beneficiary_unit(beneficiary_unit_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE beneficiary_unit (
  beneficiary_unit_id INT(11) NOT NULL AUTO_INCREMENT,
  beneficiary_unit_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (beneficiary_unit_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
