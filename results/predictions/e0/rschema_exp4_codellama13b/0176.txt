-- Schema for database: rschema_0176

CREATE TABLE `flights` (
  `flight_id` int(11) NOT NULL,
  `airline` varchar(255) NOT NULL,
  `departure_city` varchar(255) NOT NULL,
  `departure_time` datetime NOT NULL,
  `destination` varchar(255) NOT NULL,
  `arrival_time` datetime NOT NULL,
  `ticket_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`flight_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `discounts` (
  `flight_id` int(11) NOT NULL,
  `discount_start_date` datetime NOT NULL,
  `discount_end_date` datetime NOT NULL,
  `discount_rate` decimal(10,2) NOT NULL,
  PRIMARY KEY (`flight_id`),
  FOREIGN KEY (`flight_id`) REFERENCES `flights` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `passengers` (
  `passenger_id` int(11) NOT NULL,
  `passenger_name` varchar(255) NOT NULL,
  `passenger_gender` varchar(255) NOT NULL,
  `passenger_dob` datetime NOT NULL,
  `passenger_phone` varchar(255) NOT NULL,
  `vip_discount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`passenger_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ticket_orders` (
  `ticket_order_number` int(11) NOT NULL,
  `passenger_id` int(11) NOT NULL,
  `flight_id` int(11) NOT NULL,
  `travel_date` datetime NOT NULL,
  `purchase_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`ticket_order_number`),
  FOREIGN KEY (`passenger_id`) REFERENCES `passengers` (`passenger_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`flight_id`) REFERENCES `flights` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
