-- Schema for database: rschema_0181

CREATE TABLE trucks (
  id INT(11) NOT NULL AUTO_INCREMENT,
  license_plate VARCHAR(255) NOT NULL,
  model VARCHAR(255) NOT NULL,
  load_capacity INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE tasks (
  id INT(11) NOT NULL AUTO_INCREMENT,
  task_number VARCHAR(255) NOT NULL,
  origin VARCHAR(255) NOT NULL,
  destination VARCHAR(255) NOT NULL,
  cargo_name VARCHAR(255) NOT NULL,
  weight INT(11) NOT NULL,
  transportation_time INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE employees (
  id INT(11) NOT NULL AUTO_INCREMENT,
  employee_id VARCHAR(255) NOT NULL,
  employee_name VARCHAR(255) NOT NULL,
  employee_phone_number VARCHAR(255) NOT NULL,
  employee_role VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE task_assignments (
  id INT(11) NOT NULL AUTO_INCREMENT,
  task_id INT(11) NOT NULL,
  truck_id INT(11) NOT NULL,
  driver_id INT(11) NOT NULL,
  assistant_id INT(11) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT task_ibfk_1 FOREIGN KEY (task_id) REFERENCES tasks (id) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT truck_ibfk_1 FOREIGN KEY (truck_id) REFERENCES trucks (id) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT driver_ibfk_1 FOREIGN KEY (driver_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT assistant_ibfk_1 FOREIGN KEY (assistant_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE task_logs (
  id INT(11) NOT NULL AUTO_INCREMENT,
  task_id INT(11) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT task_ibfk_2 FOREIGN KEY (task_id) REFERENCES tasks (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
