-- Schema for database: rschema_0185

CREATE TABLE `proposals` (
  `proposal_id` int(11) NOT NULL,
  `proposal_content` varchar(255) NOT NULL,
  `proposal_time` datetime NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`proposal_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `proposers` (
  `proposer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `party_affiliation` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`proposer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `handlers` (
  `handler_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `party_affiliation` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`handler_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `proposal_handlings` (
  `proposal_id` int(11) NOT NULL,
  `handling_time` datetime NOT NULL,
  `handler_id` int(11) NOT NULL,
  `handling_opinion` varchar(255) NOT NULL,
  PRIMARY KEY (`proposal_id`, `handling_time`),
  FOREIGN KEY (`proposal_id`) REFERENCES `proposals` (`proposal_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`handler_id`) REFERENCES `handlers` (`handler_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `proposer_members` (
  `proposer_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`proposer_id`, `member_id`),
  FOREIGN KEY (`proposer_id`) REFERENCES `proposers` (`proposer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `handling_members` (
  `handler_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`handler_id`, `member_id`),
  FOREIGN KEY (`handler_id`) REFERENCES `handlers` (`handler_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
