-- Schema for database: rschema_0186

CREATE TABLE `insured_info` (
  `insured_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `identity_card_number` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`insured_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `social_security_type` (
  `type_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `payment_details` (
  `payment_id` int(11) NOT NULL,
  `insured_id` int(11) NOT NULL,
  `type_id` int(11) NOT NULL,
  `payment_time` datetime NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `payer` varchar(255) NOT NULL,
  PRIMARY KEY (`payment_id`),
  FOREIGN KEY (`insured_id`) REFERENCES `insured_info` (`insured_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`type_id`) REFERENCES `social_security_type` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `receipt_details` (
  `receipt_id` int(11) NOT NULL,
  `insured_id` int(11) NOT NULL,
  `type_id` int(11) NOT NULL,
  `receipt_time` datetime NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `recipient` varchar(255) NOT NULL,
  PRIMARY KEY (`receipt_id`),
  FOREIGN KEY (`insured_id`) REFERENCES `insured_info` (`insured_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`type_id`) REFERENCES `social_security_type` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
