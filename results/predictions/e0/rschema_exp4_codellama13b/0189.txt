-- Schema for database: rschema_0189

CREATE TABLE `technicians` (
  `technician_id` int(11) NOT NULL,
  `technician_name` varchar(255) NOT NULL,
  `technician_contact_number` varchar(255) NOT NULL,
  `technician_skill_level` varchar(255) NOT NULL,
  PRIMARY KEY (`technician_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_tasks` (
  `task_id` int(11) NOT NULL,
  `vehicle_license_plate_number` varchar(255) NOT NULL,
  `vehicle_model` varchar(255) NOT NULL,
  `vehicle_owner_name` varchar(255) NOT NULL,
  `vehicle_owner_contact_number` varchar(255) NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(255) NOT NULL,
  `repair_duration` int(11) NOT NULL,
  `repair_cost` int(11) NOT NULL,
  `repair_effect` varchar(255) NOT NULL,
  `owner_evaluation` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`technician_id`) REFERENCES `technicians` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluation` (
  `technician_id` int(11) NOT NULL,
  `number_of_repair_tasks_completed` int(11) NOT NULL,
  `satisfaction_of_vehicle_owners` int(11) NOT NULL,
  PRIMARY KEY (`technician_id`),
  FOREIGN KEY (`technician_id`) REFERENCES `technicians` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
