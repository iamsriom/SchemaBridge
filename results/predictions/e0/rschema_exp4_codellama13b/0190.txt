-- Schema for database: rschema_0190

CREATE TABLE `repair_tasks` (
  `task_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone_number` varchar(255) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `model_number` varchar(255) NOT NULL,
  `fault_description` varchar(255) NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(255) NOT NULL,
  `repair_duration` int(11) NOT NULL,
  `repair_cost` int(11) NOT NULL,
  `repair_results` varchar(255) NOT NULL,
  `customer_feedback` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`task_id`) REFERENCES `repair_technicians` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_technicians` (
  `technician_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `completed_tasks` int(11) NOT NULL,
  `customer_satisfaction` int(11) NOT NULL,
  PRIMARY KEY (`technician_id`),
  FOREIGN KEY (`task_id`) REFERENCES `repair_tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `workstations` (
  `workstation_id` int(11) NOT NULL,
  `technician_id` int(11) NOT NULL,
  `workstation_name` varchar(255) NOT NULL,
  PRIMARY KEY (`workstation_id`),
  FOREIGN KEY (`technician_id`) REFERENCES `repair_technicians` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  PRIMARY KEY (`repair_result_id`)

CREATE TABLE `customer_feedback` (
