-- Schema for database: rschema_0191

CREATE TABLE `sales_teams` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_name` varchar(255) NOT NULL,
  `team_lead` varchar(255) NOT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_telephone` varchar(255) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `sales_date` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_cost` decimal(10,2) NOT NULL,
  `evaluation` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`team_id`) REFERENCES `sales_teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `evaluation` decimal(10,2) NOT NULL,
  PRIMARY KEY (`personnel_id`),
  FOREIGN KEY (`team_id`) REFERENCES `sales_teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_evaluations` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_id` int(11) NOT NULL,
  `evaluation` varchar(255) NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  FOREIGN KEY (`task_id`) REFERENCES `sales_tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
