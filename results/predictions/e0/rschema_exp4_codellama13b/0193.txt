-- Schema for database: rschema_0193

CREATE TABLE `materials` (
`material_id` int(11) NOT NULL AUTO_INCREMENT,
`material_number` varchar(255) NOT NULL,
`material_name` varchar(255) NOT NULL,
`specification` varchar(255) NOT NULL,
`unit_price` decimal(10,2) NOT NULL,
`inventory` int(11) NOT NULL,
PRIMARY KEY (`material_id`),
UNIQUE KEY `material_number` (`material_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
`project_id` int(11) NOT NULL AUTO_INCREMENT,
`project_name` varchar(255) NOT NULL,
`project_description` varchar(255) NOT NULL,
PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_materials` (
`project_id` int(11) NOT NULL,
`material_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`date_used` datetime NOT NULL,
`department_using` varchar(255) NOT NULL,
`employee_using` varchar(255) NOT NULL,
PRIMARY KEY (`project_id`, `material_id`),
FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`material_id`) REFERENCES `materials` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_usage` (
`material_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`date_used` datetime NOT NULL,
`department_using` varchar(255) NOT NULL,
`employee_using` varchar(255) NOT NULL,
PRIMARY KEY (`material_id`, `date_used`),
FOREIGN KEY (`material_id`) REFERENCES `materials` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory_management` (
`material_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`date_updated` datetime NOT NULL,
PRIMARY KEY (`material_id`),
FOREIGN KEY (`material_id`) REFERENCES `materials` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
