-- Schema for database: rschema_0206

CREATE TABLE `areas` (
  `area_id` int(11) NOT NULL AUTO_INCREMENT,
  `area_number` int(11) NOT NULL,
  `area_name` varchar(255) NOT NULL,
  `geographic_location` varchar(255) NOT NULL,
  `administrator` varchar(255) NOT NULL,
  PRIMARY KEY (`area_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plant_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `area_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `recorder` varchar(255) NOT NULL,
  `plant_species` varchar(255) NOT NULL,
  `plant_number` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`area_id`) REFERENCES `areas` (`area_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `animal_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `area_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `recorder` varchar(255) NOT NULL,
  `animal_species` varchar(255) NOT NULL,
  `animal_number` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`area_id`) REFERENCES `areas` (`area_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `environmental_changes` (
  `change_id` int(11) NOT NULL AUTO_INCREMENT,
  `area_id` int(11) NOT NULL,
  `temperature` int(11) NOT NULL,
  `humidity` int(11) NOT NULL,
  `rainfall` int(11) NOT NULL,
  `change_time` datetime NOT NULL,
  `change_location` varchar(255) NOT NULL,
  PRIMARY KEY (`change_id`),
  FOREIGN KEY (`area_id`) REFERENCES `areas` (`area_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
