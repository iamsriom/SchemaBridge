-- Schema for database: rschema_0207

CREATE TABLE `facilities` (
`facility_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`responsible_person` varchar(255) NOT NULL,
`creation_date` datetime NOT NULL,
PRIMARY KEY (`facility_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_details` (
`facility_id` int(11) NOT NULL,
`maintenance_time` datetime NOT NULL,
`maintenance_content` varchar(255) NOT NULL,
`maintenance_personnel` varchar(255) NOT NULL,
`maintenance_costs` int(11) NOT NULL,
PRIMARY KEY (`facility_id`, `maintenance_time`),
FOREIGN KEY (`facility_id`) REFERENCES `facilities` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `usage_details` (
`facility_id` int(11) NOT NULL,
`usage_time` datetime NOT NULL,
`usage_frequency` int(11) NOT NULL,
`user_feedback` varchar(255) NOT NULL,
PRIMARY KEY (`facility_id`, `usage_time`),
FOREIGN KEY (`facility_id`) REFERENCES `facilities` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
`user_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_interactions` (
`user_id` int(11) NOT NULL,
`facility_id` int(11) NOT NULL,
`interaction_time` datetime NOT NULL,
PRIMARY KEY (`user_id`, `facility_id`, `interaction_time`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`facility_id`) REFERENCES `facilities` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
