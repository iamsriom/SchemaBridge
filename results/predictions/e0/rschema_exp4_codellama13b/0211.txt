-- Schema for database: rschema_0211

CREATE TABLE `consulting_service_projects` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  `client_name` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultants` (
  `consultant_id` int(11) NOT NULL,
  `consultant_name` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `gender` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `professional_field` varchar(255) NOT NULL,
  PRIMARY KEY (`consultant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consulting_service_records` (
  `service_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `consultant_id` int(11) NOT NULL,
  `service_date` datetime NOT NULL,
  `service_content` varchar(255) NOT NULL,
  PRIMARY KEY (`service_id`),
  FOREIGN KEY (`project_id`) REFERENCES `consulting_service_projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`consultant_id`) REFERENCES `consultants` (`consultant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
