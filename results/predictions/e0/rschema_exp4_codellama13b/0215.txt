-- Schema for database: rschema_0215

CREATE TABLE `mines` (
`mine_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`person_in_charge` varchar(255) NOT NULL,
PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_administrators` (
`administrator_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`administrator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_purchases` (
`purchase_id` int(11) NOT NULL,
`purchase_date` datetime NOT NULL,
`material_name` varchar(255) NOT NULL,
`quantity` int(11) NOT NULL,
`supplier` varchar(255) NOT NULL,
`purchase_cost` decimal(10,2) NOT NULL,
`mine_id` int(11) NOT NULL,
PRIMARY KEY (`purchase_id`),
FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_distributions` (
`distribution_id` int(11) NOT NULL,
`distribution_date` datetime NOT NULL,
`material_name` varchar(255) NOT NULL,
`quantity` int(11) NOT NULL,
`distributor` varchar(255) NOT NULL,
`recipient` varchar(255) NOT NULL,
`mine_id` int(11) NOT NULL,
PRIMARY KEY (`distribution_id`),
FOREIGN KEY (`distributor`) REFERENCES `material_administrators` (`administrator_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
