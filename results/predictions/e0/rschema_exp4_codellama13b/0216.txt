-- Schema for database: rschema_0216

CREATE TABLE `card_types` (
  `card_type_code` int(11) NOT NULL,
  `card_type_name` varchar(255) NOT NULL,
  `annual_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`card_type_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `id_number` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_cards` (
  `card_number` varchar(255) NOT NULL,
  `card_type_code` int(11) NOT NULL,
  `account_opening_date` datetime NOT NULL,
  `credit_limit` decimal(10,2) NOT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`card_number`),
  FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `transaction_id` int(11) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_location` varchar(255) NOT NULL,
  `transaction_type` varchar(255) NOT NULL,
  `card_number` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repayments` (
  `repayment_id` int(11) NOT NULL,
  `repayment_amount` decimal(10,2) NOT NULL,
  `repayment_time` datetime NOT NULL,
  `card_number` varchar(255) NOT NULL,
  PRIMARY KEY (`repayment_id`),
  FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
