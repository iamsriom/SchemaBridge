-- Schema for database: rschema_0217

CREATE TABLE course (
  course_id INT(11) NOT NULL,
  teacher VARCHAR(255) NOT NULL,
  class_hours INT(11) NOT NULL,
  description VARCHAR(255) NOT NULL,
  schedule VARCHAR(255) NOT NULL,
  PRIMARY KEY (course_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE student (
  student_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  grade INT(11) NOT NULL,
  class VARCHAR(255) NOT NULL,
  PRIMARY KEY (student_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE course_enrollment (
  student_id INT(11) NOT NULL,
  course_id INT(11) NOT NULL,
  PRIMARY KEY (student_id, course_id),
  FOREIGN KEY (student_id) REFERENCES student(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (course_id) REFERENCES course(course_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user (
  user_id INT(11) NOT NULL,
  email VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  username VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE system_settings (
  system_name VARCHAR(255) NOT NULL,
  theme VARCHAR(255) NOT NULL,
  page_configuration VARCHAR(255) NOT NULL,
  version VARCHAR(255) NOT NULL,
  PRIMARY KEY (system_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE tool_management (
  tool_id INT(11) NOT NULL,
  asset_number VARCHAR(255) NOT NULL,
  holder VARCHAR(255) NOT NULL,
  PRIMARY KEY (tool_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE tool_loan (
  record_id INT(11) NOT NULL,
  borrower VARCHAR(255) NOT NULL,
  asset_number VARCHAR(255) NOT NULL,
  time VARCHAR(255) NOT NULL,
  PRIMARY KEY (record_id),
  FOREIGN KEY (asset_number) REFERENCES tool_management(asset_number) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
