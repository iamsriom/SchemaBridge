-- Schema for database: rschema_0218

CREATE TABLE manufacturers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  person_in_charge VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE printer_models (
  id INT(11) NOT NULL AUTO_INCREMENT,
  manufacturer_id INT(11) NOT NULL,
  model_name VARCHAR(255) NOT NULL,
  standard_driver_config VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (manufacturer_id) REFERENCES manufacturers (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE printers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  model_id INT(11) NOT NULL,
  location VARCHAR(255) NOT NULL,
  hostname VARCHAR(255) NOT NULL,
  personalized_driver_settings VARCHAR(255) NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (model_id) REFERENCES printer_models (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
