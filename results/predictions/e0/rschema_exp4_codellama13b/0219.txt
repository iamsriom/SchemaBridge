-- Schema for database: rschema_0219

CREATE TABLE `card` (
  `card_id` int(11) NOT NULL,
  `card_name` varchar(255) NOT NULL,
  `card_type` varchar(255) NOT NULL,
  `card_artist` varchar(255) NOT NULL,
  `card_effect` varchar(255) NOT NULL,
  `card_power` int(11) NOT NULL,
  `card_intelligence` int(11) NOT NULL,
  `card_endurance` int(11) NOT NULL,
  PRIMARY KEY (`card_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `card_collection` (
  `collection_id` int(11) NOT NULL,
  `collection_name` varchar(255) NOT NULL,
  PRIMARY KEY (`collection_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `card_collection_association` (
  `card_id` int(11) NOT NULL,
  `collection_id` int(11) NOT NULL,
  `collection_number` int(11) NOT NULL,
  `card_rarity` varchar(255) NOT NULL,
  PRIMARY KEY (`card_id`, `collection_id`),
  FOREIGN KEY (`card_id`) REFERENCES `card` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`collection_id`) REFERENCES `card_collection` (`collection_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_exchange_card` (
  `user_1_id` int(11) NOT NULL,
  `user_2_id` int(11) NOT NULL,
  `exchange_time` datetime NOT NULL,
  `exchanged_card_id` int(11) NOT NULL,
  PRIMARY KEY (`user_1_id`, `user_2_id`, `exchange_time`),
  FOREIGN KEY (`user_1_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_2_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`exchanged_card_id`) REFERENCES `card` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
