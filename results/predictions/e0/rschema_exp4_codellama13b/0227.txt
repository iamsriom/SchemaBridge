-- Schema for database: rschema_0227

CREATE TABLE `regions` (
`region_id` int(11) NOT NULL AUTO_INCREMENT,
`region_name` varchar(255) NOT NULL,
`region_description` varchar(255) NOT NULL,
PRIMARY KEY (`region_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `countries` (
`country_id` int(11) NOT NULL AUTO_INCREMENT,
`country_name` varchar(255) NOT NULL,
`country_description` varchar(255) NOT NULL,
`region_id` int(11) NOT NULL,
PRIMARY KEY (`country_id`),
FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
`product_id` int(11) NOT NULL AUTO_INCREMENT,
`product_name` varchar(255) NOT NULL,
`manufacturer` varchar(255) NOT NULL,
`brand` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`size` varchar(255) NOT NULL,
`retail_price` decimal(10,2) NOT NULL,
PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `suppliers` (
`supplier_id` int(11) NOT NULL AUTO_INCREMENT,
`supplier_name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`country_id` int(11) NOT NULL,
`contact_phone_number` varchar(255) NOT NULL,
`account_balance` decimal(10,2) NOT NULL,
PRIMARY KEY (`supplier_id`),
FOREIGN KEY (`country_id`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product_supplier_relationships` (
`product_id` int(11) NOT NULL,
`supplier_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`supply_cost` decimal(10,2) NOT NULL,
PRIMARY KEY (`product_id`, `supplier_id`),
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`supplier_id`) REFERENCES `suppliers` (`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`country_id` int(11) NOT NULL,
`contact_phone_number` varchar(255) NOT NULL,
`account_balance` decimal(10,2) NOT NULL,
`market_segment` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`),
FOREIGN KEY (`country_id`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
`order_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_id` int(11) NOT NULL,
`order_status` varchar(255) NOT NULL,
`total_price` decimal(10,2) NOT NULL,
`order_date` datetime NOT NULL,
`priority` varchar(255) NOT NULL,
`handler` varchar(255) NOT NULL,
PRIMARY KEY (`order_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order_products` (
`order_id` int(11) NOT NULL,
`product_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`extended_price` decimal(10,2) NOT NULL,
`discount` decimal(10,2) NOT NULL,
`tax` decimal(10,2) NOT NULL,
`return_indicator` varchar(255) NOT NULL,
`status` varchar(255) NOT NULL,
`shipment_date` datetime NOT NULL,
`committed_date` datetime NOT NULL,
`receipt_date` datetime NOT NULL,
`shipping_instructions` varchar(255) NOT NULL,
`shipping_method` varchar(255) NOT NULL,
PRIMARY KEY (`order_id`, `product_id`),
FOREIGN KEY (`order_id`) REFERENCES `orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
