-- Schema for database: rschema_0228

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_name VARCHAR(255) NOT NULL,
  customer_address VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
);

CREATE TABLE magazines (
  magazine_name VARCHAR(255) NOT NULL,
  publisher VARCHAR(255) NOT NULL,
  theme VARCHAR(255) NOT NULL,
  publication_frequency VARCHAR(255) NOT NULL,
  website VARCHAR(255) NOT NULL,
  PRIMARY KEY (magazine_name, publisher)
);

CREATE TABLE pricing (
  pricing_id INT(11) NOT NULL AUTO_INCREMENT,
  magazine_name VARCHAR(255) NOT NULL,
  publisher VARCHAR(255) NOT NULL,
  subscription_package VARCHAR(255) NOT NULL,
  subscription_period VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (pricing_id),
  FOREIGN KEY (magazine_name, publisher) REFERENCES magazines (magazine_name, publisher) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE customer_subscribes_to_magazine (
  customer_id INT(11) NOT NULL,
  pricing_id INT(11) NOT NULL,
  creation_time DATETIME NOT NULL,
  subscription_start_date DATETIME NOT NULL,
  PRIMARY KEY (customer_id, pricing_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (pricing_id) REFERENCES pricing (pricing_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE magazine_to_pricing (
  pricing_id INT(11) NOT NULL,
  magazine_name VARCHAR(255) NOT NULL,
  publisher VARCHAR(255) NOT NULL,
  subscription_package VARCHAR(255) NOT NULL,
  subscription_period VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (pricing_id),
  FOREIGN KEY (magazine_name, publisher) REFERENCES magazines (magazine_name, publisher) ON DELETE CASCADE ON UPDATE CASCADE
);
