-- Schema for database: rschema_0231

CREATE TABLE users (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  contact_email VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  registration_time DATETIME NOT NULL,
  PRIMARY KEY (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE experiences (
  experience_id INT(11) NOT NULL AUTO_INCREMENT,
  description TEXT NOT NULL,
  code_content TEXT NOT NULL,
  creator_user_id INT(11) NOT NULL,
  creation_time DATETIME NOT NULL,
  modification_time DATETIME NOT NULL,
  status CHAR(1) NOT NULL DEFAULT 'F',
  PRIMARY KEY (experience_id),
  FOREIGN KEY (creator_user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_activity_log (
  user_id INT(11) NOT NULL,
  experience_id INT(11) NOT NULL,
  creation_time DATETIME NOT NULL,
  modification_time DATETIME NOT NULL,
  PRIMARY KEY (user_id, experience_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (experience_id) REFERENCES experiences (experience_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE status (
  status_id INT(11) NOT NULL AUTO_INCREMENT,
  status_value CHAR(1) NOT NULL,
  PRIMARY KEY (status_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE experience_status (
  experience_id INT(11) NOT NULL,
  status_id INT(11) NOT NULL,
  PRIMARY KEY (experience_id, status_id),
  FOREIGN KEY (experience_id) REFERENCES experiences (experience_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (status_id) REFERENCES status (status_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
