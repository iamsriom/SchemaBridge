-- Schema for database: rschema_0236

CREATE TABLE cameras (
  id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  rtsp_stream_address VARCHAR(255) NOT NULL,
  real_time_video_stream_path VARCHAR(255) NOT NULL,
  camera_analysis_time DATETIME NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE video_recordings (
  id INT(11) NOT NULL AUTO_INCREMENT,
  video_file_path VARCHAR(255) NOT NULL,
  cover_path VARCHAR(255) NOT NULL,
  start_time DATETIME NOT NULL,
  duration INT(11) NOT NULL,
  camera_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (camera_id) REFERENCES cameras (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE alarms (
  id INT(11) NOT NULL AUTO_INCREMENT,
  alarm_type VARCHAR(255) NOT NULL,
  level VARCHAR(255) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  video_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (video_id) REFERENCES video_recordings (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE events (
  id INT(11) NOT NULL AUTO_INCREMENT,
  event_type VARCHAR(255) NOT NULL,
  occurrence_time DATETIME NOT NULL,
  coordinates VARCHAR(255) NOT NULL,
  dimensions VARCHAR(255) NOT NULL,
  video_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (video_id) REFERENCES video_recordings (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE real_time_monitoring (
  id INT(11) NOT NULL AUTO_INCREMENT,
  rtsp_stream_address VARCHAR(255) NOT NULL,
  camera_analysis_time DATETIME NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE historical_query (
  id INT(11) NOT NULL AUTO_INCREMENT,
  event_frequency_statistics VARCHAR(255) NOT NULL,
  alarm_type_analysis VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE security_management (
  id INT(11) NOT NULL AUTO_INCREMENT,
  event_trace VARCHAR(255) NOT NULL,
  potential_security_risk VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
