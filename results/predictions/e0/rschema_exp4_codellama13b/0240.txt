-- Schema for database: rschema_0240

CREATE TABLE ships (
  ship_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  fleet_id INT(11) NOT NULL,
  construction_date DATETIME NOT NULL,
  service_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (ship_id),
  FOREIGN KEY (fleet_id) REFERENCES fleets (fleet_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE crew_members (
  crew_member_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  fleet_id INT(11) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  length_of_service INT(11) NOT NULL,
  PRIMARY KEY (crew_member_id),
  FOREIGN KEY (fleet_id) REFERENCES fleets (fleet_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE fleets (
  fleet_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  commander_id INT(11) NOT NULL,
  base VARCHAR(255) NOT NULL,
  PRIMARY KEY (fleet_id),
  FOREIGN KEY (commander_id) REFERENCES crew_members (crew_member_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE fleet_mission_execution (
  crew_member_id INT(11) NOT NULL,
  fleet_id INT(11) NOT NULL,
  execution_time DATETIME NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  is_crew_member_primary_responsible BOOLEAN NOT NULL,
  PRIMARY KEY (crew_member_id, fleet_id),
  FOREIGN KEY (crew_member_id) REFERENCES crew_members (crew_member_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (fleet_id) REFERENCES fleets (fleet_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE ship_supply (
  responsible_supply_crew_member_id INT(11) NOT NULL,
  ship_id INT(11) NOT NULL,
  supply_time DATETIME NOT NULL,
  supply_items VARCHAR(255) NOT NULL,
  supply_quantity INT(11) NOT NULL,
  PRIMARY KEY (responsible_supply_crew_member_id, ship_id),
  FOREIGN KEY (responsible_supply_crew_member_id) REFERENCES crew_members (crew_member_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (ship_id) REFERENCES ships (ship_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
