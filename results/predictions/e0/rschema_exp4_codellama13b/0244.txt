-- Schema for database: rschema_0244

CREATE TABLE `sites` (
`site_id` int(11) NOT NULL AUTO_INCREMENT,
`site_name` varchar(255) NOT NULL,
`site_address` varchar(255) NOT NULL,
`site_type` varchar(255) NOT NULL,
PRIMARY KEY (`site_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bus_routes` (
`route_id` int(11) NOT NULL AUTO_INCREMENT,
`route_name` varchar(255) NOT NULL,
`route_type` varchar(255) NOT NULL,
`route_description` varchar(255) NOT NULL,
PRIMARY KEY (`route_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stations` (
`station_id` int(11) NOT NULL AUTO_INCREMENT,
`station_name` varchar(255) NOT NULL,
`station_address` varchar(255) NOT NULL,
`station_type` varchar(255) NOT NULL,
PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transfer_routes` (
`transfer_id` int(11) NOT NULL AUTO_INCREMENT,
`start_station_id` int(11) NOT NULL,
`end_station_id` int(11) NOT NULL,
`transfer_time` datetime NOT NULL,
`transfer_type` varchar(255) NOT NULL,
PRIMARY KEY (`transfer_id`),
FOREIGN KEY (`start_station_id`) REFERENCES `stations` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`end_station_id`) REFERENCES `stations` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_accounts` (
`account_id` int(11) NOT NULL AUTO_INCREMENT,
`account_number` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
PRIMARY KEY (`account_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_roles` (
`role_id` int(11) NOT NULL AUTO_INCREMENT,
`role_name` varchar(255) NOT NULL,
PRIMARY KEY (`role_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_role_assignments` (
`assignment_id` int(11) NOT NULL AUTO_INCREMENT,
`account_id` int(11) NOT NULL,
`role_id` int(11) NOT NULL,
PRIMARY KEY (`assignment_id`),
FOREIGN KEY (`account_id`) REFERENCES `user_accounts` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`role_id`) REFERENCES `user_roles` (`role_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `questions` (
`question_id` int(11) NOT NULL AUTO_INCREMENT,
`question_text` varchar(255) NOT NULL,
`question_date` datetime NOT NULL,
`question_status` varchar(255) NOT NULL,
PRIMARY KEY (`question_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `answers` (
`answer_id` int(11) NOT NULL AUTO_INCREMENT,
`question_id` int(11) NOT NULL,
`answer_text` varchar(255) NOT NULL,
`answer_date` datetime NOT NULL,
PRIMARY KEY (`answer_id`),
FOREIGN KEY (`question_id`) REFERENCES `questions` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcements` (
`announcement_id` int(11) NOT NULL AUTO_INCREMENT,
`announcement_text` varchar(255) NOT NULL,
`announcement_date` datetime NOT NULL,
PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_feedback` (
`feedback_id` int(11) NOT NULL AUTO_INCREMENT,
`user_id` int(11) NOT NULL,
`feedback_text` varchar(255) NOT NULL,
`feedback_date` datetime NOT NULL,
PRIMARY KEY (`feedback_id`),
FOREIGN KEY (`user_id`) REFERENCES `user_accounts` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
