-- Schema for database: rschema_0246
CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE crews (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE films (
  id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  release_date DATE NOT NULL,
  genre VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY title (title)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE tv_series (
  id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  release_date DATE NOT NULL,
  genre VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY title (title)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reviews (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  review TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE shares (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE rankings (
  id INT(11) NOT NULL AUTO_INCREMENT,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  ranking INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE search_history (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  search_term VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE crew_information (
  id INT(11) NOT NULL AUTO_INCREMENT,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  crew_type VARCHAR(255) NOT NULL,
  crew_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE news (
  id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  content TEXT NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE news_columns (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE news_column_relations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  news_id INT(11) NOT NULL,
  column_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (news_id) REFERENCES news (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (column_id) REFERENCES news_columns (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_and_tv_information (
  id INT(11) NOT NULL AUTO_INCREMENT,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  title VARCHAR(255) NOT NULL,
  release_date DATE NOT NULL,
  genre VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE comment_information (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  comment TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_and_tv_ranking (
  id INT(11) NOT NULL AUTO_INCREMENT,
  film_id INT(11) NOT NULL,
  tv_series_id INT(11) NOT NULL,
  ranking INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (film_id) REFERENCES films (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tv_series_id) REFERENCES tv_series (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=Inno
