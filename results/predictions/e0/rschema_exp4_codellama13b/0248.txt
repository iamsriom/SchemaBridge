-- Schema for database: rschema_0248

CREATE TABLE users (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  registration_time DATETIME NOT NULL,
  last_login_time DATETIME NOT NULL,
  role VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id),
  UNIQUE KEY username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE books (
  book_id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  author VARCHAR(255) NOT NULL,
  description TEXT NOT NULL,
  PRIMARY KEY (book_id),
  UNIQUE KEY title (title)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE articles (
  article_id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  content TEXT NOT NULL,
  author VARCHAR(255) NOT NULL,
  publish_time DATETIME NOT NULL,
  PRIMARY KEY (article_id),
  UNIQUE KEY title (title)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE activities (
  activity_id INT(11) NOT NULL AUTO_INCREMENT,
  activity_name VARCHAR(255) NOT NULL,
  activity_time DATETIME NOT NULL,
  description TEXT NOT NULL,
  PRIMARY KEY (activity_id),
  UNIQUE KEY activity_name (activity_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE video_interviews (
  video_id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  video_link VARCHAR(255) NOT NULL,
  publish_time DATETIME NOT NULL,
  PRIMARY KEY (video_id),
  UNIQUE KEY title (title)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_activities (
  user_id INT(11) NOT NULL,
  activity_id INT(11) NOT NULL,
  PRIMARY KEY (user_id, activity_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (activity_id) REFERENCES activities (activity_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_book_recommendations (
  user_id INT(11) NOT NULL,
  book_id INT(11) NOT NULL,
  PRIMARY KEY (user_id, book_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (book_id) REFERENCES books (book_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_article_recommendations (
  user_id INT(11) NOT NULL,
  article_id INT(11) NOT NULL,
  PRIMARY KEY (user_id, article_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (article_id) REFERENCES articles (article_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_video_interviews (
  user_id INT(11) NOT NULL,
  video_id INT(11) NOT NULL,
  PRIMARY KEY (user_id, video_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (video_id) REFERENCES video_interviews (video_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE lottery_winners (
  user_id INT(11) NOT NULL,
  award VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
