-- Schema for database: rschema_0255

CREATE TABLE production_statistics (
  report_type VARCHAR(255) NOT NULL,
  date DATE NOT NULL,
  production_data VARCHAR(255) NOT NULL,
  production_indicators VARCHAR(255) NOT NULL,
  completion_status VARCHAR(255) NOT NULL,
  metal_recovery_rate VARCHAR(255) NOT NULL,
  grade VARCHAR(255) NOT NULL,
  PRIMARY KEY (report_type, date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE equipment_management (
  equipment_id INT(11) NOT NULL AUTO_INCREMENT,
  equipment_name VARCHAR(255) NOT NULL,
  equipment_model VARCHAR(255) NOT NULL,
  main_category VARCHAR(255) NOT NULL,
  subcategory VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  operation_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (equipment_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE equipment_inspection (
  inspection_id INT(11) NOT NULL AUTO_INCREMENT,
  inspection_plan VARCHAR(255) NOT NULL,
  inspection_progress VARCHAR(255) NOT NULL,
  inspector_id INT(11) NOT NULL,
  PRIMARY KEY (inspection_id),
  FOREIGN KEY (inspector_id) REFERENCES equipment_management (equipment_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE equipment_failure (
  failure_id INT(11) NOT NULL AUTO_INCREMENT,
  reporter_id INT(11) NOT NULL,
  failure_type VARCHAR(255) NOT NULL,
  failure_time DATETIME NOT NULL,
  processing_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (failure_id),
  FOREIGN KEY (reporter_id) REFERENCES equipment_management (equipment_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE equipment_repair (
  repair_id INT(11) NOT NULL AUTO_INCREMENT,
  repair_details VARCHAR(255) NOT NULL,
  repair_progress VARCHAR(255) NOT NULL,
  repair_time DATETIME NOT NULL,
  PRIMARY KEY (repair_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE quality_management (
  metal_id INT(11) NOT NULL AUTO_INCREMENT,
  metal_recovery_rate VARCHAR(255) NOT NULL,
  incoming_warehouse_sample VARCHAR(255) NOT NULL,
  metal_grade VARCHAR(255) NOT NULL,
  recovery_rate_daily_report VARCHAR(255) NOT NULL,
  grade_daily_report VARCHAR(255) NOT NULL,
  PRIMARY KEY (metal_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_management (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  role VARCHAR(255) NOT NULL,
  permissions VARCHAR(255) NOT NULL,
  status VARCHAR(255) NOT NULL,
  operation_history VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE announcements (
  announcement_id INT(11) NOT NULL AUTO_INCREMENT,
  announcement_content VARCHAR(255) NOT NULL,
  release_time DATETIME NOT NULL,
  announcement_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (announcement_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE job_positions (
  position_id INT(11) NOT NULL AUTO_INCREMENT,
  position_name VARCHAR(255) NOT NULL,
  position_description VARCHAR(255) NOT NULL,
  position_responsibilities VARCHAR(255) NOT NULL,
  job_requirements VARCHAR(255) NOT NULL,
  PRIMARY KEY (position_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE position_match (
  position_id INT(11) NOT NULL,
  user_id INT(11) NOT NULL,
  match_rate VARCHAR(255) NOT NULL,
  PRIMARY KEY (position_id, user_id),
  FOREIGN KEY (position_id) REFERENCES job_positions (position_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (user_id) REFERENCES user_management (user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_application (
  application_id INT(11) NOT NULL AUTO_INCREMENT,
  position_id INT(11) NOT NULL,
  user_id INT(11) NOT NULL,
  application_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (application_id),
  FOREIGN KEY (position_id) REFERENCES job_positions (position_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (user_id) REFERENCES user_management (user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
