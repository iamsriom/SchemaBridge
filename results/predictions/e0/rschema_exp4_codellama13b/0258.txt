-- Schema for database: rschema_0258

CREATE TABLE donors (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE fundraising_events (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  start_date DATETIME NOT NULL,
  end_date DATETIME NOT NULL,
  location VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE donation_records (
  id INT(11) NOT NULL AUTO_INCREMENT,
  donor_id INT(11) NOT NULL,
  fundraising_event_id INT(11) NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  donation_date DATETIME NOT NULL,
  method VARCHAR(255) NOT NULL,
  message VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (donor_id) REFERENCES donors (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (fundraising_event_id) REFERENCES fundraising_events (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
