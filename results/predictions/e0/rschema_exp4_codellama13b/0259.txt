-- Schema for database: rschema_0259

CREATE TABLE users (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id),
  UNIQUE KEY (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE email_accounts (
  email_account_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  email_address VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  PRIMARY KEY (email_account_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE address_books (
  address_book_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  PRIMARY KEY (address_book_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE contacts (
  contact_id INT(11) NOT NULL AUTO_INCREMENT,
  address_book_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  work_address VARCHAR(255) NOT NULL,
  email_address_1 VARCHAR(255) NOT NULL,
  email_address_2 VARCHAR(255) NOT NULL,
  email_address_3 VARCHAR(255) NOT NULL,
  PRIMARY KEY (contact_id),
  FOREIGN KEY (address_book_id) REFERENCES address_books (address_book_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE emails (
  email_id INT(11) NOT NULL AUTO_INCREMENT,
  sender_address VARCHAR(255) NOT NULL,
  recipient_address VARCHAR(255) NOT NULL,
  email_status VARCHAR(255) NOT NULL,
  subject VARCHAR(255) NOT NULL,
  content TEXT NOT NULL,
  sending_time DATETIME NOT NULL,
  receiving_time DATETIME NOT NULL,
  PRIMARY KEY (email_id),
  FOREIGN KEY (sender_address) REFERENCES email_accounts (email_address) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (recipient_address) REFERENCES email_accounts (email_address) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE attachments (
  attachment_id INT(11) NOT NULL AUTO_INCREMENT,
  email_id INT(11) NOT NULL,
  attachment_filename VARCHAR(255) NOT NULL,
  attachment_size INT(11) NOT NULL,
  PRIMARY KEY (attachment_id),
  FOREIGN KEY (email_id) REFERENCES emails (email_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
