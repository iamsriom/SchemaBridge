-- Schema for database: rschema_0262

CREATE TABLE `supermarkets` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`telephone` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`supermarket_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`manager` varchar(255) NOT NULL,
`contact_telephone` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`supermarket_id`) REFERENCES `supermarkets` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`department_id` int(11) NOT NULL,
`employee_number` varchar(255) NOT NULL,
`name` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`salary` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`supermarket_id` int(11) NOT NULL,
`product_number` varchar(255) NOT NULL,
`name` varchar(255) NOT NULL,
`model` varchar(255) NOT NULL,
`unit_price` int(11) NOT NULL,
`inventory_quantity` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`supermarket_id`) REFERENCES `supermarkets` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `allocations` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`product_id` int(11) NOT NULL,
`salesperson_id` int(11) NOT NULL,
`allocation_time` datetime NOT NULL,
`allocation_quantity` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`product_id`) REFERENCES `products` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`salesperson_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
