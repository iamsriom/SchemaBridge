-- Schema for database: rschema_0263

CREATE TABLE students (
  student_id INT(11) NOT NULL,
  student_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (student_id)
);

CREATE TABLE lecturers (
  lecturer_id INT(11) NOT NULL,
  lecturer_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (lecturer_id)
);

CREATE TABLE homework (
  homework_id INT(11) NOT NULL,
  student_id INT(11) NOT NULL,
  lecturer_id INT(11) NOT NULL,
  submission_date DATETIME NOT NULL,
  submission_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (homework_id),
  FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (lecturer_id) REFERENCES lecturers(lecturer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE graded_homework (
  graded_homework_id INT(11) NOT NULL,
  homework_id INT(11) NOT NULL,
  lecturer_id INT(11) NOT NULL,
  grading_date DATETIME NOT NULL,
  score INT(11) NOT NULL,
  evaluation VARCHAR(255) NOT NULL,
  PRIMARY KEY (graded_homework_id),
  FOREIGN KEY (homework_id) REFERENCES homework(homework_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (lecturer_id) REFERENCES lecturers(lecturer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE random_inspection (
  random_inspection_id INT(11) NOT NULL,
  lecturer_id INT(11) NOT NULL,
  graded_homework_id INT(11) NOT NULL,
  inspection_date DATETIME NOT NULL,
  inspection_opinion VARCHAR(255) NOT NULL,
  PRIMARY KEY (random_inspection_id),
  FOREIGN KEY (lecturer_id) REFERENCES lecturers(lecturer_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (graded_homework_id) REFERENCES graded_homework(graded_homework_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE student_information (
  student_id INT(11) NOT NULL,
  homework_id INT(11) NOT NULL,
  graded_homework_id INT(11) NOT NULL,
  score INT(11) NOT NULL,
  evaluation VARCHAR(255) NOT NULL,
  PRIMARY KEY (student_id),
  FOREIGN KEY (homework_id) REFERENCES homework(homework_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (graded_homework_id) REFERENCES graded_homework(graded_homework_id) ON DELETE CASCADE ON UPDATE CASCADE
);
