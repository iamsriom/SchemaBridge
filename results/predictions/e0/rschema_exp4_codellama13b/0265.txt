-- Schema for database: rschema_0265

CREATE TABLE chessboards (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE chess_pieces (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chessboard_id INT(11) NOT NULL,
  color VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chessboard_id) REFERENCES chessboards(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inspections (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chess_piece_id INT(11) NOT NULL,
  inspection_time DATETIME NOT NULL,
  result VARCHAR(255) NOT NULL,
  defect_type VARCHAR(255),
  PRIMARY KEY (id),
  FOREIGN KEY (chess_piece_id) REFERENCES chess_pieces(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE quality_alerts (
  id INT(11) NOT NULL AUTO_INCREMENT,
  inspection_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (inspection_id) REFERENCES inspections(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE quality_inspection_models (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chessboard_id INT(11) NOT NULL,
  model_running_time DATETIME NOT NULL,
  model_result VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chessboard_id) REFERENCES chessboards(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
