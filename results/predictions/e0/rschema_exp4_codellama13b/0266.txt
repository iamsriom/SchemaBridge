-- Schema for database: rschema_0266

CREATE TABLE pipelines (
  pipeline_id INT(11) NOT NULL,
  start_point VARCHAR(255) NOT NULL,
  end_point VARCHAR(255) NOT NULL,
  length INT(11) NOT NULL,
  diameter INT(11) NOT NULL,
  PRIMARY KEY (pipeline_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE personnel (
  personnel_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (personnel_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE maintenance_tasks (
  task_id INT(11) NOT NULL,
  pipeline_id INT(11) NOT NULL,
  personnel_id INT(11) NOT NULL,
  details TEXT NOT NULL,
  time DATETIME NOT NULL,
  PRIMARY KEY (task_id),
  FOREIGN KEY (pipeline_id) REFERENCES pipelines (pipeline_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (personnel_id) REFERENCES personnel (personnel_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE performance_evaluation (
  evaluation_id INT(11) NOT NULL,
  pipeline_id INT(11) NOT NULL,
  personnel_id INT(11) NOT NULL,
  details TEXT NOT NULL,
  time DATETIME NOT NULL,
  PRIMARY KEY (evaluation_id),
  FOREIGN KEY (pipeline_id) REFERENCES pipelines (pipeline_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (personnel_id) REFERENCES personnel (personnel_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
