-- Schema for database: rschema_0272

CREATE TABLE ranches (
  ranch_id INT(11) NOT NULL AUTO_INCREMENT,
  ranch_name VARCHAR(255) NOT NULL,
  ranch_address VARCHAR(255) NOT NULL,
  PRIMARY KEY (ranch_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE livestock (
  livestock_id INT(11) NOT NULL AUTO_INCREMENT,
  ranch_id INT(11) NOT NULL,
  livestock_name VARCHAR(255) NOT NULL,
  species VARCHAR(255) NOT NULL,
  PRIMARY KEY (livestock_id),
  FOREIGN KEY (ranch_id) REFERENCES ranches (ranch_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE health_status (
  health_status_id INT(11) NOT NULL AUTO_INCREMENT,
  livestock_id INT(11) NOT NULL,
  health_status VARCHAR(255) NOT NULL,
  recording_time DATETIME NOT NULL,
  PRIMARY KEY (health_status_id),
  FOREIGN KEY (livestock_id) REFERENCES livestock (livestock_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE vaccination_status (
  vaccination_status_id INT(11) NOT NULL AUTO_INCREMENT,
  livestock_id INT(11) NOT NULL,
  vaccination_measures VARCHAR(255) NOT NULL,
  vaccination_time DATETIME NOT NULL,
  PRIMARY KEY (vaccination_status_id),
  FOREIGN KEY (livestock_id) REFERENCES livestock (livestock_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE feed_consumption (
  feed_consumption_id INT(11) NOT NULL AUTO_INCREMENT,
  livestock_id INT(11) NOT NULL,
  consumption_time DATETIME NOT NULL,
  feed_type VARCHAR(255) NOT NULL,
  consumption_amount INT(11) NOT NULL,
  PRIMARY KEY (feed_consumption_id),
  FOREIGN KEY (livestock_id) REFERENCES livestock (livestock_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE entry_exit_records (
  entry_exit_record_id INT(11) NOT NULL AUTO_INCREMENT,
  livestock_id INT(11) NOT NULL,
  entry_exit_type VARCHAR(255) NOT NULL,
  entry_exit_time DATETIME NOT NULL,
  PRIMARY KEY (entry_exit_record_id),
  FOREIGN KEY (livestock_id) REFERENCES livestock (livestock_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
