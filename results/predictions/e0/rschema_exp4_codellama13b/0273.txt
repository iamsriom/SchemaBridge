-- Schema for database: rschema_0273

CREATE TABLE `feedback` (
`feedback_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_info` varchar(255) NOT NULL,
`feedback_content` varchar(255) NOT NULL,
`feedback_time` datetime NOT NULL,
`processing_status` varchar(255) NOT NULL,
PRIMARY KEY (`feedback_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `feedback_processing` (
`feedback_id` int(11) NOT NULL,
`processing_status` varchar(255) NOT NULL,
`processing_time` datetime NOT NULL,
PRIMARY KEY (`feedback_id`, `processing_status`),
FOREIGN KEY (`feedback_id`) REFERENCES `feedback` (`feedback_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_info` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `feedback_customer` (
`feedback_id` int(11) NOT NULL,
`customer_id` int(11) NOT NULL,
PRIMARY KEY (`feedback_id`, `customer_id`),
FOREIGN KEY (`feedback_id`) REFERENCES `feedback` (`feedback_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`customer_id`) REFERENCES `customer_info` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
