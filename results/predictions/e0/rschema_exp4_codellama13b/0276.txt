-- Schema for database: rschema_0276

CREATE TABLE `maintenance_technician` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `employment_date` datetime NOT NULL,
  `team_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`team_id`) REFERENCES `maintenance_team` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_team` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `team_leader_id` int(11) NOT NULL,
  `team_responsibilities` varchar(255) NOT NULL,
  `total_members` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`team_leader_id`) REFERENCES `maintenance_technician` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_customer` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_task` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `maintenance_time` datetime NOT NULL,
  `maintenance_duration` int(11) NOT NULL,
  `costs` decimal(10,2) NOT NULL,
  `results` varchar(255) NOT NULL,
  `customer_feedback` varchar(255) NOT NULL,
  `customer_satisfaction_level` int(11) NOT NULL,
  `feedback_resolution_results` varchar(255) NOT NULL,
  `model_and_name_of_maintenance_product` varchar(255) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `maintenance_technician_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`customer_id`) REFERENCES `maintenance_customer` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`maintenance_technician_id`) REFERENCES `maintenance_technician` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
