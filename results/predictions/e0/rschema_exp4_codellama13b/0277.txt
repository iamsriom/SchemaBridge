-- Schema for database: rschema_0277

CREATE TABLE `housekeepers` (
`staff_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
`service_type` varchar(255) NOT NULL,
`years_of_experience` int(11) NOT NULL,
`professional_skills` varchar(255) NOT NULL,
PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_teams` (
`team_id` int(11) NOT NULL,
`team_name` varchar(255) NOT NULL,
`team_leader` varchar(255) NOT NULL,
`team_size` int(11) NOT NULL,
PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_tasks` (
`task_id` int(11) NOT NULL,
`customer_name` varchar(255) NOT NULL,
`customer_phone` varchar(255) NOT NULL,
`service_date` datetime NOT NULL,
`service_address` varchar(255) NOT NULL,
`service_type` varchar(255) NOT NULL,
`service_duration` int(11) NOT NULL,
`housekeeper_id` int(11) NOT NULL,
PRIMARY KEY (`task_id`),
FOREIGN KEY (`housekeeper_id`) REFERENCES `housekeepers` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_fees` (
`task_id` int(11) NOT NULL,
`service_fee` decimal(10,2) NOT NULL,
`service_effectiveness` varchar(255) NOT NULL,
`customer_evaluation` varchar(255) NOT NULL,
PRIMARY KEY (`task_id`),
FOREIGN KEY (`task_id`) REFERENCES `service_tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `housekeepers_performance` (
`housekeeper_id` int(11) NOT NULL,
`completed_tasks` int(11) NOT NULL,
`customer_satisfaction` int(11) NOT NULL,
PRIMARY KEY (`housekeeper_id`),
FOREIGN KEY (`housekeeper_id`) REFERENCES `housekeepers` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
