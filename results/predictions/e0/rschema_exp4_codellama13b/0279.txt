-- Schema for database: rschema_0279

CREATE TABLE `properties` (
`property_id` int(11) NOT NULL,
`address` varchar(255) NOT NULL,
`area` int(11) NOT NULL,
`sale_price` int(11) NOT NULL,
`sale_date` datetime NOT NULL,
PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reports` (
`report_id` int(11) NOT NULL,
`report_date` datetime NOT NULL,
`report_type` varchar(255) NOT NULL,
`analysis_results` varchar(255) NOT NULL,
PRIMARY KEY (`report_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property_reports` (
`property_id` int(11) NOT NULL,
`report_id` int(11) NOT NULL,
PRIMARY KEY (`property_id`, `report_id`),
FOREIGN KEY (`property_id`) REFERENCES `properties` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`report_id`) REFERENCES `reports` (`report_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
