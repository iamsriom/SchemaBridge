-- Schema for database: rschema_0288

CREATE TABLE `sites` (
  `site_id` int(11) NOT NULL,
  `site_name` varchar(255) NOT NULL,
  `site_location` varchar(255) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  PRIMARY KEY (`site_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_personnel` (
  `personnel_id` int(11) NOT NULL,
  `personnel_name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `scope_of_monitoring` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `personnel_id` int(11) NOT NULL,
  `personnel_name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `scope_of_maintenance` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_sessions` (
  `session_id` int(11) NOT NULL,
  `site_id` int(11) NOT NULL,
  `air_quality` varchar(255) NOT NULL,
  `water_quality` varchar(255) NOT NULL,
  `noise_level` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  `monitoring_personnel_id` int(11) NOT NULL,
  `monitoring_result` varchar(255) NOT NULL,
  PRIMARY KEY (`session_id`),
  FOREIGN KEY (`site_id`) REFERENCES `sites` (`site_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`monitoring_personnel_id`) REFERENCES `monitoring_personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_activities` (
  `activity_id` int(11) NOT NULL,
  `site_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  `maintenance_cost` varchar(255) NOT NULL,
  PRIMARY KEY (`activity_id`),
  FOREIGN KEY (`site_id`) REFERENCES `sites` (`site_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `maintenance_personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
