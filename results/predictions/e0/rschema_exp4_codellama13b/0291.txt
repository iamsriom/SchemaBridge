-- Schema for database: rschema_0291

CREATE TABLE reagents (
  reagent_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  specifications VARCHAR(255) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  expiration_date DATETIME NOT NULL,
  PRIMARY KEY (reagent_id),
  UNIQUE KEY reagent_id_UNIQUE (reagent_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE laboratories (
  laboratory_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  person_in_charge VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  PRIMARY KEY (laboratory_id),
  UNIQUE KEY laboratory_id_UNIQUE (laboratory_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reagent_usage (
  usage_id INT(11) NOT NULL AUTO_INCREMENT,
  reagent_id INT(11) NOT NULL,
  laboratory_id INT(11) NOT NULL,
  usage_date DATETIME NOT NULL,
  user VARCHAR(255) NOT NULL,
  usage_quantity INT(11) NOT NULL,
  PRIMARY KEY (usage_id),
  UNIQUE KEY usage_id_UNIQUE (usage_id),
  FOREIGN KEY (reagent_id) REFERENCES reagents (reagent_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (laboratory_id) REFERENCES laboratories (laboratory_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reagent_inventory (
  reagent_id INT(11) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  PRIMARY KEY (reagent_id),
  FOREIGN KEY (reagent_id) REFERENCES reagents (reagent_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE laboratory_inventory (
  laboratory_id INT(11) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  PRIMARY KEY (laboratory_id),
  FOREIGN KEY (laboratory_id) REFERENCES laboratories (laboratory_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reagent_usage_history (
  usage_id INT(11) NOT NULL,
  usage_date DATETIME NOT NULL,
  PRIMARY KEY (usage_id),
  FOREIGN KEY (usage_id) REFERENCES reagent_usage (usage_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE laboratory_usage_history (
  laboratory_id INT(11) NOT NULL,
  usage_date DATETIME NOT NULL,
  PRIMARY KEY (laboratory_id),
  FOREIGN KEY (laboratory_id) REFERENCES laboratories (laboratory_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reagent_expiration_alerts (
  reagent_id INT(11) NOT NULL,
  expiration_date DATETIME NOT NULL,
  PRIMARY KEY (reagent_id),
  FOREIGN KEY (reagent_id) REFERENCES reagents (reagent_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE laboratory_expiration_alerts (
  laboratory_id INT(11) NOT NULL,
  expiration_date DATETIME NOT NULL,
  PRIMARY KEY (laboratory_id),
  FOREIGN KEY (laboratory_id) REFERENCES laboratories (laboratory_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
