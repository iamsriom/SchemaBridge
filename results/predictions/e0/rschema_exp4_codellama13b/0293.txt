-- Schema for database: rschema_0293

CREATE TABLE `equipment` (
`equipment_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`model` varchar(255) NOT NULL,
`purchase_date` datetime NOT NULL,
`other_details` varchar(255) NOT NULL,
PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mine` (
`mine_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`person_in_charge` varchar(255) NOT NULL,
PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_event` (
`maintenance_event_id` int(11) NOT NULL,
`equipment_id` int(11) NOT NULL,
`mine_id` int(11) NOT NULL,
`maintenance_date` datetime NOT NULL,
`maintenance_content` varchar(255) NOT NULL,
`personnel` varchar(255) NOT NULL,
`maintenance_cost` int(11) NOT NULL,
PRIMARY KEY (`maintenance_event_id`),
FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
`maintenance_personnel_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
PRIMARY KEY (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_usage` (
`equipment_id` int(11) NOT NULL,
`mine_id` int(11) NOT NULL,
`usage_date` datetime NOT NULL,
`usage_duration` int(11) NOT NULL,
PRIMARY KEY (`equipment_id`, `mine_id`, `usage_date`),
FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_maintenance` (
`equipment_id` int(11) NOT NULL,
`maintenance_event_id` int(11) NOT NULL,
`maintenance_date` datetime NOT NULL,
`maintenance_cost` int(11) NOT NULL,
PRIMARY KEY (`equipment_id`, `maintenance_event_id`, `maintenance_date`),
FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`maintenance_event_id`) REFERENCES `maintenance_event` (`maintenance_event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mine_maintenance` (
`mine_id` int(11) NOT NULL,
`maintenance_event_id` int(11) NOT NULL,
`maintenance_date` datetime NOT NULL,
`maintenance_cost` int(11) NOT NULL,
PRIMARY KEY (`mine_id`, `maintenance_event_id`, `maintenance_date`),
FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`maintenance_event_id`) REFERENCES `maintenance_event` (`maintenance_event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
