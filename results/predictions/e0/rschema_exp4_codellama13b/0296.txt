-- Schema for database: rschema_0296

CREATE TABLE `financial_advisors` (
  `advisor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `hire_date` datetime NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  PRIMARY KEY (`advisor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `id_number` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `risk_preference` varchar(255) NOT NULL,
  `total_assets` int(11) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `financial_products` (
  `product_code` int(11) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `product_type` varchar(255) NOT NULL,
  `product_value` int(11) NOT NULL,
  PRIMARY KEY (`product_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `investment_portfolios` (
  `portfolio_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `product_code` int(11) NOT NULL,
  `quantity_held` int(11) NOT NULL,
  PRIMARY KEY (`portfolio_id`),
  FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`product_code`) REFERENCES `financial_products` (`product_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_records` (
  `record_id` int(11) NOT NULL,
  `portfolio_id` int(11) NOT NULL,
  `product_code` int(11) NOT NULL,
  `transaction_type` varchar(255) NOT NULL,
  `transaction_quantity` int(11) NOT NULL,
  `transaction_price` int(11) NOT NULL,
  `transaction_time` datetime NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`portfolio_id`) REFERENCES `investment_portfolios` (`portfolio_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`product_code`) REFERENCES `financial_products` (`product_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
