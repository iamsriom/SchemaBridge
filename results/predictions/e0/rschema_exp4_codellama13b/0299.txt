-- Schema for database: rschema_0299

CREATE TABLE `cities` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`longitude` decimal(10,6) NOT NULL,
`latitude` decimal(10,6) NOT NULL,
`country` varchar(255) NOT NULL,
`population` int(11) NOT NULL,
`timezone` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `daily_forecasts` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`city_id` int(11) NOT NULL,
`date` date NOT NULL,
`sunrise` time NOT NULL,
`sunset` time NOT NULL,
`temperature_day` decimal(10,2) NOT NULL,
`temperature_night` decimal(10,2) NOT NULL,
`feels_like` decimal(10,2) NOT NULL,
`atmospheric_pressure` decimal(10,2) NOT NULL,
`humidity` decimal(10,2) NOT NULL,
`wind_speed` decimal(10,2) NOT NULL,
`wind_direction` decimal(10,2) NOT NULL,
`cloud_cover` decimal(10,2) NOT NULL,
`precipitation_probability` decimal(10,2) NOT NULL,
`weather_type` varchar(255) NOT NULL,
`weather_description` varchar(255) NOT NULL,
`weather_icon` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`city_id`) REFERENCES `cities` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
