-- Schema for database: rschema_0300

CREATE TABLE `material_supply_projects` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `project_number` varchar(255) NOT NULL,
  `material_code` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `charge_amount` decimal(10,2) NOT NULL,
  `created_at` datetime NOT NULL,
  `closed_at` datetime NOT NULL,
  `project_number_bill` varchar(255) NOT NULL,
  `bill_number` varchar(255) NOT NULL,
  `bill_quantity` int(11) NOT NULL,
  `bill_unit` varchar(255) NOT NULL,
  `bill_cost` decimal(10,2) NOT NULL,
  `bill_charge_amount` decimal(10,2) NOT NULL,
  `bill_created_at` datetime NOT NULL,
  `bill_modified_at` datetime NOT NULL,
  `bill_verification_errors` varchar(255) NOT NULL,
  `bill_verification_errors_handling` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`project_number`) REFERENCES `material_supply_projects` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bills` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bill_number` varchar(255) NOT NULL,
  `material_code` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `charge_amount` decimal(10,2) NOT NULL,
  `created_at` datetime NOT NULL,
  `modified_at` datetime NOT NULL,
  `verification_errors` varchar(255) NOT NULL,
  `verification_errors_handling` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`bill_number`) REFERENCES `bills` (`bill_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_supply_project_bills` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `project_number` varchar(255) NOT NULL,
  `bill_number` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`project_number`) REFERENCES `material_supply_projects` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`bill_number`) REFERENCES `bills` (`bill_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
