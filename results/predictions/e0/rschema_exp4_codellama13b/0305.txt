-- Schema for database: rschema_0305

CREATE TABLE `users` (
  `account_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `verification_status` varchar(255) NOT NULL,
  `account_status` varchar(255) NOT NULL,
  PRIMARY KEY (`account_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `follow_relationships` (
  `follower_account_id` int(11) NOT NULL,
  `followee_account_id` int(11) NOT NULL,
  PRIMARY KEY (`follower_account_id`, `followee_account_id`),
  FOREIGN KEY (`follower_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`followee_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `posts` (
  `post_id` int(11) NOT NULL,
  `publisher_account_id` int(11) NOT NULL,
  `content` varchar(255) NOT NULL,
  `publish_time` datetime NOT NULL,
  `attachment_link` varchar(255) NOT NULL,
  PRIMARY KEY (`post_id`),
  FOREIGN KEY (`publisher_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notifications` (
  `notification_id` int(11) NOT NULL,
  `notification_type` varchar(255) NOT NULL,
  `recipient_account_id` int(11) NOT NULL,
  `related_post_or_user` varchar(255) NOT NULL,
  `notification_time` datetime NOT NULL,
  PRIMARY KEY (`notification_id`),
  FOREIGN KEY (`recipient_account_id`) REFERENCES `users` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
