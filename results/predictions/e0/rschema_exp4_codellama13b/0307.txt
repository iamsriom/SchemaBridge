-- Schema for database: rschema_0307

CREATE TABLE company (
  company_id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  company_website VARCHAR(255) NOT NULL,
  company_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (company_id),
  UNIQUE KEY (company_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  is_admin TINYINT(1) NOT NULL,
  real_name VARCHAR(255) NOT NULL,
  password_type VARCHAR(255) NOT NULL,
  password_salt VARCHAR(255) NOT NULL,
  username VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  display_name_usage VARCHAR(255) NOT NULL,
  is_developer TINYINT(1) NOT NULL,
  last_login_time DATETIME NOT NULL,
  PRIMARY KEY (user_id),
  UNIQUE KEY (username),
  UNIQUE KEY (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE api_key (
  key_id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  private_key VARCHAR(255) NOT NULL,
  application_reason VARCHAR(255) NOT NULL,
  key_name VARCHAR(255) NOT NULL,
  uri VARCHAR(255) NOT NULL,
  PRIMARY KEY (key_id),
  UNIQUE KEY (key_name),
  FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE employee (
  employee_id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  employee_name VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  salary INT(11) NOT NULL,
  PRIMARY KEY (employee_id),
  FOREIGN KEY (company_id) REFERENCES company(company_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE site (
  site_id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  database_name VARCHAR(255) NOT NULL,
  is_deprecated TINYINT(1) NOT NULL,
  site_name VARCHAR(255) NOT NULL,
  hostname VARCHAR(255) NOT NULL,
  site_abbreviation VARCHAR(255) NOT NULL,
  PRIMARY KEY (site_id),
  UNIQUE KEY (database_name),
  UNIQUE KEY (site_abbreviation),
  FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE device (
  device_id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  device_name VARCHAR(255) NOT NULL,
  associated_employee_id INT(11) NOT NULL,
  PRIMARY KEY (device_id),
  FOREIGN KEY (employee_id) REFERENCES employee(employee_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
