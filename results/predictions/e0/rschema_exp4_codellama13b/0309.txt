-- Schema for database: rschema_0309

CREATE TABLE customers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  marital_status VARCHAR(255) NOT NULL,
  region_id INT(11) NOT NULL,
  income_group_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (region_id) REFERENCES regions (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (income_group_id) REFERENCES income_groups (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE regions (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE income_groups (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE vehicle_segments (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE policies (
  id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  vehicle_segment_id INT(11) NOT NULL,
  policy_type VARCHAR(255) NOT NULL,
  premium DECIMAL(10,2) NOT NULL,
  purchase_date DATE NOT NULL,
  expiration_date DATE NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (customer_id) REFERENCES customers (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (vehicle_segment_id) REFERENCES vehicle_segments (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE policy_attributes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  policy_id INT(11) NOT NULL,
  attribute_name VARCHAR(255) NOT NULL,
  attribute_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (policy_id) REFERENCES policies (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
