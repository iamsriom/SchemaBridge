-- Schema for database: rschema_0311

CREATE TABLE organizations (
  organization_id INT(11) NOT NULL AUTO_INCREMENT,
  organization_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (organization_id)
);

CREATE TABLE processes (
  process_id INT(11) NOT NULL AUTO_INCREMENT,
  parent_process_id INT(11) NULL,
  organization_id INT(11) NOT NULL,
  process_type VARCHAR(255) NOT NULL,
  log_information TEXT NOT NULL,
  resource_id INT(11) NOT NULL,
  status VARCHAR(255) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NULL,
  PRIMARY KEY (process_id),
  FOREIGN KEY (organization_id) REFERENCES organizations (organization_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (parent_process_id) REFERENCES processes (process_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE events (
  event_id INT(11) NOT NULL AUTO_INCREMENT,
  process_id INT(11) NOT NULL,
  event_type VARCHAR(255) NOT NULL,
  log_information TEXT NOT NULL,
  status VARCHAR(255) NOT NULL,
  timestamp DATETIME NOT NULL,
  PRIMARY KEY (event_id),
  FOREIGN KEY (process_id) REFERENCES processes (process_id) ON DELETE CASCADE ON UPDATE CASCADE
);
