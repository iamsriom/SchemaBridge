-- Schema for database: rschema_0315

CREATE TABLE `personnel` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `books` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`title` varchar(255) NOT NULL,
`author` varchar(255) NOT NULL,
`publisher` varchar(255) NOT NULL,
`publication_date` datetime NOT NULL,
`isbn` varchar(255) NOT NULL,
`copies` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `borrowings` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`personnel_id` int(11) NOT NULL,
`book_id` int(11) NOT NULL,
`borrow_date` datetime NOT NULL,
`return_date` datetime NOT NULL,
`returned` tinyint(1) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`book_id`) REFERENCES `books` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `returns` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`borrowing_id` int(11) NOT NULL,
`return_date` datetime NOT NULL,
`returned` tinyint(1) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`borrowing_id`) REFERENCES `borrowings` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `queries` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`query` varchar(255) NOT NULL,
`result` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `statistics` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`statistic` varchar(255) NOT NULL,
`value` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
