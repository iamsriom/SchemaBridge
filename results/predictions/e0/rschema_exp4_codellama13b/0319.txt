-- Schema for database: rschema_0319

CREATE TABLE `regional_info` (
`regional_code` int(11) NOT NULL,
`city_code` int(11) NOT NULL,
`center_coordinates` varchar(255) NOT NULL,
`regional_level` varchar(255) NOT NULL,
`merged_name` varchar(255) NOT NULL,
`regional_name` varchar(255) NOT NULL,
`parent_regional_id` int(11) NOT NULL,
`pinyin` varchar(255) NOT NULL,
`abbreviation` varchar(255) NOT NULL,
`postal_code` varchar(255) NOT NULL,
PRIMARY KEY (`regional_code`),
FOREIGN KEY (`parent_regional_id`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regional_attributes` (
`regional_code` int(11) NOT NULL,
`attribute_name` varchar(255) NOT NULL,
`attribute_value` varchar(255) NOT NULL,
PRIMARY KEY (`regional_code`, `attribute_name`),
FOREIGN KEY (`regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regional_relations` (
`regional_code` int(11) NOT NULL,
`parent_regional_code` int(11) NOT NULL,
PRIMARY KEY (`regional_code`, `parent_regional_code`),
FOREIGN KEY (`regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`parent_regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regional_hierarchy` (
`regional_code` int(11) NOT NULL,
`parent_regional_code` int(11) NOT NULL,
PRIMARY KEY (`regional_code`, `parent_regional_code`),
FOREIGN KEY (`regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`parent_regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regional_merged_name` (
`regional_code` int(11) NOT NULL,
`merged_name` varchar(255) NOT NULL,
PRIMARY KEY (`regional_code`, `merged_name`),
FOREIGN KEY (`regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regional_pinyin` (
`regional_code` int(11) NOT NULL,
`pinyin` varchar(255) NOT NULL,
PRIMARY KEY (`regional_code`, `pinyin`),
FOREIGN KEY (`regional_code`) REFERENCES `regional_info` (`regional_code`) ON DELETE CASCADE ON UPDATE CASCADE
