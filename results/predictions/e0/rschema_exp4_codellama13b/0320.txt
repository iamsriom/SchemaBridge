-- Schema for database: rschema_0320

CREATE TABLE products (
  product_id INT(11) NOT NULL AUTO_INCREMENT,
  product_name VARCHAR(255) NOT NULL,
  product_slogan VARCHAR(255) NOT NULL,
  product_description TEXT NOT NULL,
  product_category VARCHAR(255) NOT NULL,
  default_price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE styles (
  style_id INT(11) NOT NULL AUTO_INCREMENT,
  associated_product_id INT(11) NOT NULL,
  style_name VARCHAR(255) NOT NULL,
  sale_price DECIMAL(10,2) NOT NULL,
  original_price DECIMAL(10,2) NOT NULL,
  is_default_style BOOLEAN NOT NULL,
  PRIMARY KEY (style_id),
  FOREIGN KEY (associated_product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE features (
  feature_id INT(11) NOT NULL AUTO_INCREMENT,
  associated_product_id INT(11) NOT NULL,
  feature_name VARCHAR(255) NOT NULL,
  feature_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (feature_id),
  FOREIGN KEY (associated_product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE skus (
  sku_id INT(11) NOT NULL AUTO_INCREMENT,
  associated_style_id INT(11) NOT NULL,
  size VARCHAR(255) NOT NULL,
  stock_quantity INT(11) NOT NULL,
  PRIMARY KEY (sku_id),
  FOREIGN KEY (associated_style_id) REFERENCES styles (style_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE photos (
  photo_id INT(11) NOT NULL AUTO_INCREMENT,
  associated_style_id INT(11) NOT NULL,
  photo_url VARCHAR(255) NOT NULL,
  thumbnail_url VARCHAR(255) NOT NULL,
  PRIMARY KEY (photo_id),
  FOREIGN KEY (associated_style_id) REFERENCES styles (style_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
