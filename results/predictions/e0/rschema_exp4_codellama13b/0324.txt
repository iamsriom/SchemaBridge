-- Schema for database: rschema_0324

CREATE TABLE `ip_addresses` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`ip_address` varchar(255) NOT NULL,
`continent` varchar(255) NOT NULL,
`country` varchar(255) NOT NULL,
`country_code` varchar(255) NOT NULL,
`eu_member` tinyint(1) NOT NULL,
`anonymous_proxy` tinyint(1) NOT NULL,
`satellite_provider` tinyint(1) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `ip_address` (`ip_address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `geographic_locations` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`country` varchar(255) NOT NULL,
`country_code` varchar(255) NOT NULL,
`eu_member` tinyint(1) NOT NULL,
`continent` varchar(255) NOT NULL,
`latitude` decimal(10,6) NOT NULL,
`longitude` decimal(10,6) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `country` (`country`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ip_networks` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`ip_network` varchar(255) NOT NULL,
`continent` varchar(255) NOT NULL,
`country` varchar(255) NOT NULL,
`country_code` varchar(255) NOT NULL,
`eu_member` tinyint(1) NOT NULL,
`anonymous_proxy` tinyint(1) NOT NULL,
`satellite_provider` tinyint(1) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `ip_network` (`ip_network`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ip_address_geographic_location` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`ip_address_id` int(11) NOT NULL,
`geographic_location_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `ip_address_id` (`ip_address_id`, `geographic_location_id`),
CONSTRAINT `ip_address_geographic_location_ibfk_1` FOREIGN KEY (`ip_address_id`) REFERENCES `ip_addresses` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `ip_address_geographic_location_ibfk_2` FOREIGN KEY (`geographic_location_id`) REFERENCES `geographic_locations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ip_network_geographic_location` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`ip_network_id` int(11) NOT NULL,
`geographic_location_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `ip_network_id` (`ip_network_id`, `geographic_location_id`),
CONSTRAINT `ip_network_geographic_location_ibfk_1` FOREIGN KEY (`ip_network_id`) REFERENCES `ip_networks` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `ip_network_geographic_location_ibfk_2` FOREIGN KEY (`geographic_location_id`) REFERENCES `geographic_locations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
