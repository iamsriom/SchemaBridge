-- Schema for database: rschema_0331

CREATE TABLE `players` (
`player_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`contact_info` varchar(255) NOT NULL,
`account_status` varchar(255) NOT NULL,
PRIMARY KEY (`player_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `matches` (
`match_id` int(11) NOT NULL AUTO_INCREMENT,
`match_type` varchar(255) NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
`player_1_id` int(11) NOT NULL,
`player_2_id` int(11) NOT NULL,
PRIMARY KEY (`match_id`),
FOREIGN KEY (`player_1_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`player_2_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `battles` (
`battle_id` int(11) NOT NULL AUTO_INCREMENT,
`match_id` int(11) NOT NULL,
`player_1_id` int(11) NOT NULL,
`player_2_id` int(11) NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
`win_status` varchar(255) NOT NULL,
PRIMARY KEY (`battle_id`),
FOREIGN KEY (`match_id`) REFERENCES `matches` (`match_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`player_1_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`player_2_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `results` (
`result_id` int(11) NOT NULL AUTO_INCREMENT,
`match_id` int(11) NOT NULL,
`player_1_id` int(11) NOT NULL,
`player_2_id` int(11) NOT NULL,
`score` int(11) NOT NULL,
`win_status` varchar(255) NOT NULL,
PRIMARY KEY (`result_id`),
FOREIGN KEY (`match_id`) REFERENCES `matches` (`match_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`player_1_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`player_2_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
