-- Schema for database: rschema_0337

CREATE TABLE exams (
  exam_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  time DATETIME NOT NULL,
  PRIMARY KEY (exam_id)
);

CREATE TABLE practice_questions (
  question_id INT(11) NOT NULL AUTO_INCREMENT,
  exam_id INT(11) NOT NULL,
  question_content TEXT NOT NULL,
  score INT(11) NOT NULL,
  PRIMARY KEY (question_id),
  FOREIGN KEY (exam_id) REFERENCES exams(exam_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE students (
  student_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (student_id)
);

CREATE TABLE submissions (
  submission_id INT(11) NOT NULL AUTO_INCREMENT,
  exam_id INT(11) NOT NULL,
  student_id INT(11) NOT NULL,
  submission_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (submission_id),
  FOREIGN KEY (exam_id) REFERENCES exams(exam_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE answers (
  answer_id INT(11) NOT NULL AUTO_INCREMENT,
  question_id INT(11) NOT NULL,
  submission_id INT(11) NOT NULL,
  answer_content TEXT NOT NULL,
  PRIMARY KEY (answer_id),
  FOREIGN KEY (question_id) REFERENCES practice_questions(question_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (submission_id) REFERENCES submissions(submission_id) ON DELETE CASCADE ON UPDATE CASCADE
);
