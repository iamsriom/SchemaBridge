-- Schema for database: rschema_0339

CREATE TABLE projects (
  id int(11) NOT NULL AUTO_INCREMENT,
  name varchar(255) NOT NULL,
  avatar_id int(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (avatar_id) REFERENCES avatars (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE apis (
  id int(11) NOT NULL AUTO_INCREMENT,
  project_id int(11) NOT NULL,
  name varchar(255) NOT NULL,
  access_path varchar(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (project_id) REFERENCES projects (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE api_stats (
  id int(11) NOT NULL AUTO_INCREMENT,
  api_id int(11) NOT NULL,
  call_frequency int(11) NOT NULL,
  avg_response_time int(11) NOT NULL,
  p90_response_time int(11) NOT NULL,
  creation_time datetime NOT NULL,
  update_time datetime NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (api_id) REFERENCES apis (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE avatars (
  id int(11) NOT NULL AUTO_INCREMENT,
  name varchar(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
