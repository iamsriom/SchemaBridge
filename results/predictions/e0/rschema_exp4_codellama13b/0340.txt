-- Schema for database: rschema_0340

CREATE TABLE `applications` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `languages` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`iso_code` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regions` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`iso_code` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_files` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`path` varchar(255) NOT NULL,
`base_name` varchar(255) NOT NULL,
`language` varchar(255) NOT NULL,
`region` varchar(255) NOT NULL,
`version` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_keys` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`key_name` varchar(255) NOT NULL,
`key_version` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_values` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`value_content` varchar(255) NOT NULL,
`value_version` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_file_keys` (
`resource_file_id` int(11) NOT NULL,
`resource_key_id` int(11) NOT NULL,
PRIMARY KEY (`resource_file_id`, `resource_key_id`),
CONSTRAINT `resource_file_keys_ibfk_1` FOREIGN KEY (`resource_file_id`) REFERENCES `resource_files` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `resource_file_keys_ibfk_2` FOREIGN KEY (`resource_key_id`) REFERENCES `resource_keys` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_key_values` (
`resource_key_id` int(11) NOT NULL,
`resource_value_id` int(11) NOT NULL,
PRIMARY KEY (`resource_key_id`, `resource_value_id`),
CONSTRAINT `resource_key_values_ibfk_1` FOREIGN KEY (`resource_key_id`) REFERENCES `resource_keys` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `resource_key_values_ibfk_2` FOREIGN KEY (`resource_value_id`) REFERENCES `resource_values` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `countries` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`iso_code` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `region_countries` (
`region_id` int(11) NOT NULL,
`country_id` int(11) NOT NULL,
PRIMARY KEY (`region_id`, `country_id`),
CONSTRAINT `region_countries_ibfk_1` FOREIGN KEY (`region_id`) REFERENCES `regions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `region_countries_ibfk_2` FOREIGN KEY (`country_id`) REFERENCES `countries` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
