-- Schema for database: rschema_0342

CREATE TABLE cors_stations (
  station_id int(11) NOT NULL AUTO_INCREMENT,
  station_name varchar(255) NOT NULL,
  station_description varchar(255) NOT NULL,
  PRIMARY KEY (station_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE cors_networks (
  network_id int(11) NOT NULL AUTO_INCREMENT,
  network_name varchar(255) NOT NULL,
  network_description varchar(255) NOT NULL,
  PRIMARY KEY (network_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE cors_station_network_relationship (
  relationship_id int(11) NOT NULL AUTO_INCREMENT,
  station_id int(11) NOT NULL,
  network_id int(11) NOT NULL,
  effective_start_time datetime NOT NULL,
  effective_end_time datetime NOT NULL,
  PRIMARY KEY (relationship_id),
  FOREIGN KEY (station_id) REFERENCES cors_stations (station_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (network_id) REFERENCES cors_networks (network_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
