-- Schema for database: rschema_0343

CREATE TABLE personnel (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  street VARCHAR(255) NOT NULL,
  city VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE vehicles (
  vin VARCHAR(255) NOT NULL,
  manufacturer VARCHAR(255) NOT NULL,
  model VARCHAR(255) NOT NULL,
  year INT(11) NOT NULL,
  driver_id INT(11) NOT NULL,
  PRIMARY KEY (vin),
  FOREIGN KEY (driver_id) REFERENCES personnel (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE insurance (
  vin VARCHAR(255) NOT NULL,
  insurance_type_id INT(11) NOT NULL,
  insurance_company_name VARCHAR(255) NOT NULL,
  insurance_cost INT(11) NOT NULL,
  insurance_content VARCHAR(255) NOT NULL,
  purchase_date DATE NOT NULL,
  validity_period DATE NOT NULL,
  PRIMARY KEY (vin),
  FOREIGN KEY (vin) REFERENCES vehicles (vin) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE phones (
  id INT(11) NOT NULL AUTO_INCREMENT,
  number VARCHAR(255) NOT NULL,
  personnel_id INT(11) NOT NULL,
  PRIMARY KEY (id, number),
  FOREIGN KEY (personnel_id) REFERENCES personnel (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
