-- Schema for database: rschema_0344

CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  role VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY username (username),
  UNIQUE KEY email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE applications (
  id INT(11) NOT NULL AUTO_INCREMENT,
  code VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE web_services (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  creation_date DATETIME NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE error_data (
  id INT(11) NOT NULL AUTO_INCREMENT,
  error_code VARCHAR(255) NOT NULL,
  error_description VARCHAR(255) NOT NULL,
  timestamp DATETIME NOT NULL,
  return_status VARCHAR(255) NOT NULL,
  delay_time INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (web_service_id) REFERENCES web_services (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE monitoring_data (
  id INT(11) NOT NULL AUTO_INCREMENT,
  invocation_time DATETIME NOT NULL,
  return_status VARCHAR(255) NOT NULL,
  delay_time INT(11) NOT NULL,
  success_rate DECIMAL(10, 2) NOT NULL,
  call_count INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (web_service_id) REFERENCES web_services (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_application_permissions (
  user_id INT(11) NOT NULL,
  application_id INT(11) NOT NULL,
  PRIMARY KEY (user_id, application_id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (application_id) REFERENCES applications (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE error_monitoring_data (
  error_data_id INT(11) NOT NULL,
  monitoring_data_id INT(11) NOT NULL,
  PRIMARY KEY (error_data_id, monitoring_data_id),
  FOREIGN KEY (error_data_id) REFERENCES error_data (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (monitoring_data_id) REFERENCES monitoring_data (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
