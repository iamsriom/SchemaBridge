-- Schema for database: rschema_0345

CREATE TABLE courses (
  course_id INT(11) NOT NULL AUTO_INCREMENT,
  code VARCHAR(255) NOT NULL,
  total_credits INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  professor_id INT(11) NOT NULL,
  PRIMARY KEY (course_id),
  FOREIGN KEY (professor_id) REFERENCES professors(professor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE professors (
  professor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  PRIMARY KEY (professor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE subjects (
  subject_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  credits INT(11) NOT NULL,
  equivalent_subjects VARCHAR(255) NOT NULL,
  prerequisite_subjects VARCHAR(255) NOT NULL,
  PRIMARY KEY (subject_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE courses_subjects (
  course_id INT(11) NOT NULL,
  subject_id INT(11) NOT NULL,
  semester INT(11) NOT NULL,
  PRIMARY KEY (course_id, subject_id, semester),
  FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (subject_id) REFERENCES subjects(subject_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE students (
  student_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  credits_completed INT(11) NOT NULL,
  average_grade DECIMAL(3,2) NOT NULL,
  PRIMARY KEY (student_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE courses_taken (
  student_id INT(11) NOT NULL,
  course_id INT(11) NOT NULL,
  PRIMARY KEY (student_id, course_id),
  FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
