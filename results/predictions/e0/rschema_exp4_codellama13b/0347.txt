-- Schema for database: rschema_0347

CREATE TABLE `leaderboard_metadata` (
`run_id` int(11) NOT NULL,
`execution_time` datetime NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
`runtime` int(11) NOT NULL,
`run_path` varchar(255) NOT NULL,
`max_models` int(11) NOT NULL,
`model_types` varchar(255) NOT NULL,
PRIMARY KEY (`run_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `leaderboard` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`performance_metrics` varchar(255) NOT NULL,
`best_family_model` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hyperparameter` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hyperparameter_values` (
`id` int(11) NOT NULL,
`model_id` int(11) NOT NULL,
`hyperparameter_id` int(11) NOT NULL,
`value` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`model_id`) REFERENCES `leaderboard` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`hyperparameter_id`) REFERENCES `hyperparameter` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `model_run` (
`model_id` int(11) NOT NULL,
`run_id` int(11) NOT NULL,
PRIMARY KEY (`model_id`, `run_id`),
FOREIGN KEY (`model_id`) REFERENCES `leaderboard` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`run_id`) REFERENCES `leaderboard_metadata` (`run_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dataset_metadata` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`owner` varchar(255) NOT NULL,
`license` varchar(255) NOT NULL,
`visibility` varchar(255) NOT NULL,
`url` varchar(255) NOT NULL,
`creation_date` datetime NOT NULL,
`last_update_date` datetime NOT NULL,
`data_size` int(11) NOT NULL,
`num_rows` int(11) NOT NULL,
`num_cols` int(11) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `data_map` (
`dataset_id` int(11) NOT NULL,
`run_id` int(11) NOT NULL,
PRIMARY KEY (`dataset_id`, `run_id`),
FOREIGN KEY (`dataset_id`) REFERENCES `dataset_metadata` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`run_id`) REFERENCES `leaderboard_metadata` (`run_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `model_map` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hyperparameter_def_values` (
`model_type_id` int(11) NOT NULL,
`hyperparameter_id` int(11) NOT NULL,
`value` varchar(255) NOT NULL,
PRIMARY KEY (`model_type_id`, `hyperparameter_id`),
FOREIGN KEY (`model_type_id`) REFERENCES `model_map` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`hyperparameter_id`) REFERENCES `hyperparameter` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `id_map` (
`model_id` int(11) NOT NULL,
`model_type_id` int(11) NOT NULL,
PRIMARY KEY (`model_id`, `model_type_id`),
FOREIGN KEY (`model_id`) REFERENCES `leaderboard` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`model_type_id`) REFERENCES `model_map` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dataset_variable_details` (
`dataset_id` int(11) NOT NULL,
`variable_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`data_type` varchar(255) NOT NULL,
`num_unique_values` int(11) NOT NULL,
`num_null_values` int(11) NOT NULL,
PRIMARY KEY (`dataset_id`, `variable_id`),
FOREIGN KEY (`dataset_id`) REFERENCES `dataset_metadata` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tags` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tag_map` (
`dataset_id` int(11) NOT NULL,
`tag_id` int(11) NOT NULL,
PRIMARY KEY (`dataset_id`, `tag_id`),
FOREIGN KEY (`dataset_id`) REFERENCES `dataset_metadata` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`tag_id`) REFERENCES `tags` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
