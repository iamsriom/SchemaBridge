-- Schema for database: rschema_0354

CREATE TABLE `image_info` (
`image_id` int(11) NOT NULL,
`image_url` varchar(255) NOT NULL,
`image_description` varchar(255) NOT NULL,
`compressed_signature` varchar(255) NOT NULL,
PRIMARY KEY (`image_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_info` (
`user_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sending_records` (
`sending_id` int(11) NOT NULL,
`sender_id` int(11) NOT NULL,
`sending_time` datetime NOT NULL,
`image_id` int(11) NOT NULL,
`sending_link` varchar(255) NOT NULL,
PRIMARY KEY (`sending_id`),
FOREIGN KEY (`sender_id`) REFERENCES `user_info` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`image_id`) REFERENCES `image_info` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sending_link_info` (
`sending_link_id` int(11) NOT NULL,
`sending_id` int(11) NOT NULL,
`sending_link` varchar(255) NOT NULL,
PRIMARY KEY (`sending_link_id`),
FOREIGN KEY (`sending_id`) REFERENCES `sending_records` (`sending_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
