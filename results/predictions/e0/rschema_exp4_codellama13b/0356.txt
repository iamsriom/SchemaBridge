-- Schema for database: rschema_0356

CREATE TABLE work_items (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  creation_date DATETIME NOT NULL,
  update_date DATETIME NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE work_item_attributes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  work_item_id INT(11) NOT NULL,
  attribute_name VARCHAR(255) NOT NULL,
  attribute_value VARCHAR(255) NOT NULL,
  link_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (work_item_id) REFERENCES work_items(id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE work_item_links (
  id INT(11) NOT NULL AUTO_INCREMENT,
  work_item_id INT(11) NOT NULL,
  linked_work_item_id INT(11) NOT NULL,
  link_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (work_item_id) REFERENCES work_items(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (linked_work_item_id) REFERENCES work_items(id) ON DELETE CASCADE ON UPDATE CASCADE
);
