-- Schema for database: rschema_0358

CREATE TABLE `task` (
  `task_id` int(11) NOT NULL,
  `task_type` varchar(255) NOT NULL,
  `lock_info` varchar(255) NOT NULL,
  `execution_instance_id` int(11) NOT NULL,
  `process_instance_id` int(11) NOT NULL,
  `process_definition_id` int(11) NOT NULL,
  `number_of_retries` int(11) NOT NULL,
  `exception_info` varchar(255) NOT NULL,
  `expiration_time` datetime NOT NULL,
  `repeat_configuration` varchar(255) NOT NULL,
  `processor_configuration` varchar(255) NOT NULL,
  `tenant_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`execution_instance_id`) REFERENCES `execution_instance` (`execution_instance_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `execution_instance` (
  `execution_instance_id` int(11) NOT NULL,
  `process_instance_id` int(11) NOT NULL,
  `process_definition_id` int(11) NOT NULL,
  PRIMARY KEY (`execution_instance_id`),
  FOREIGN KEY (`process_instance_id`) REFERENCES `process_instance` (`process_instance_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `process_instance` (
  `process_instance_id` int(11) NOT NULL,
  `process_definition_id` int(11) NOT NULL,
  PRIMARY KEY (`process_instance_id`),
  FOREIGN KEY (`process_definition_id`) REFERENCES `process_definition` (`process_definition_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `process_definition` (
  `process_definition_id` int(11) NOT NULL,
  PRIMARY KEY (`process_definition_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_execution` (
  `task_id` int(11) NOT NULL,
  `execution_instance_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `execution_result` varchar(255) NOT NULL,
  `exception_info` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`, `execution_instance_id`),
  FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`execution_instance_id`) REFERENCES `execution_instance` (`execution_instance_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tenant` (
  `tenant_id` int(11) NOT NULL,
  PRIMARY KEY (`tenant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
