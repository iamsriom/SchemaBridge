-- Schema for database: rschema_0361

CREATE TABLE `restaurants` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`business_hours` varchar(255) NOT NULL,
`average_cost_per_person` int(11) NOT NULL,
`latitude` decimal(10, 6) NOT NULL,
`longitude` decimal(10, 6) NOT NULL,
`feature_tags` varchar(255) NOT NULL,
`rating` int(11) NOT NULL,
`number_of_reviews` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `id` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dishes` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`restaurant_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`price` decimal(10, 2) NOT NULL,
`description` varchar(255) NOT NULL,
`category_tags` varchar(255) NOT NULL,
`image` varchar(255) NOT NULL,
`satisfaction_rate` int(11) NOT NULL,
`monthly_sales` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `id` (`id`),
FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviews` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`restaurant_id` int(11) NOT NULL,
`user_id` int(11) NOT NULL,
`content` varchar(255) NOT NULL,
`rating` int(11) NOT NULL,
`dining_time` datetime NOT NULL,
`image` varchar(255) NOT NULL,
`reply_time` datetime NOT NULL,
`reply_content` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `id` (`id`),
FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `id` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
