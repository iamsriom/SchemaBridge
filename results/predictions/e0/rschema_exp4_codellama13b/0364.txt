-- Schema for database: rschema_0364

CREATE TABLE `companies` (
`company_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
`user_id` int(11) NOT NULL,
`company_id` int(11) NOT NULL,
`username` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`job_title` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`),
FOREIGN KEY (`company_id`) REFERENCES `companies` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `login_response_types` (
`type_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `login_response_records` (
`record_id` int(11) NOT NULL,
`user_id` int(11) NOT NULL,
`login_time` datetime NOT NULL,
`response_type_id` int(11) NOT NULL,
`response_label` varchar(255) NOT NULL,
PRIMARY KEY (`record_id`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`response_type_id`) REFERENCES `login_response_types` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
