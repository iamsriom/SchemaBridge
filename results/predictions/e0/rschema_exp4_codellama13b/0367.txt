-- Schema for database: rschema_0367

CREATE TABLE `users` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`birth_date` datetime NOT NULL,
`gender` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`insurance_status` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `appointments` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`examination_description` varchar(255) NOT NULL,
`examination_document` varchar(255) NOT NULL,
`institution_id` int(11) NOT NULL,
`payment_amount` decimal(10,2) NOT NULL,
`creation_date` datetime NOT NULL,
`appointment_date` datetime NOT NULL,
`status` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`institution_id`) REFERENCES `institutions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `institutions` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`opening_time` datetime NOT NULL,
`closing_time` datetime NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operators` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`institution_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`institution_id`) REFERENCES `institutions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
