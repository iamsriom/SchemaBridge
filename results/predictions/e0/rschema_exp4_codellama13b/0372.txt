-- Schema for database: rschema_0372

CREATE TABLE `leagues` (
`league_id` int(11) NOT NULL AUTO_INCREMENT,
`league_name` varchar(255) NOT NULL,
`region_id` int(11) NOT NULL,
`parent_league_id` int(11) DEFAULT NULL,
`person_in_charge` varchar(255) DEFAULT NULL,
`created_at` datetime NOT NULL,
`created_by` varchar(255) NOT NULL,
`updated_at` datetime DEFAULT NULL,
`updated_by` varchar(255) DEFAULT NULL,
`version` int(11) NOT NULL DEFAULT 1,
PRIMARY KEY (`league_id`),
FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`parent_league_id`) REFERENCES `leagues` (`league_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regions` (
`region_id` int(11) NOT NULL AUTO_INCREMENT,
`region_name` varchar(255) NOT NULL,
`created_at` datetime NOT NULL,
`created_by` varchar(255) NOT NULL,
`updated_at` datetime DEFAULT NULL,
`updated_by` varchar(255) DEFAULT NULL,
PRIMARY KEY (`region_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`created_at` datetime NOT NULL,
`created_by` varchar(255) NOT NULL,
`updated_at` datetime DEFAULT NULL,
`updated_by` varchar(255) DEFAULT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `league_history` (
`league_id` int(11) NOT NULL,
`version` int(11) NOT NULL,
`created_at` datetime NOT NULL,
`created_by` varchar(255) NOT NULL,
`updated_at` datetime DEFAULT NULL,
`updated_by` varchar(255) DEFAULT NULL,
PRIMARY KEY (`league_id`, `version`),
FOREIGN KEY (`league_id`) REFERENCES `leagues` (`league_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_leagues` (
`user_id` int(11) NOT NULL,
`league_id` int(11) NOT NULL,
`created_at` datetime NOT NULL,
`created_by` varchar(255) NOT NULL,
`updated_at` datetime DEFAULT NULL,
`updated_by` varchar(255) DEFAULT NULL,
PRIMARY KEY (`user_id`, `league_id`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`league_id`) REFERENCES `leagues` (`league_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
