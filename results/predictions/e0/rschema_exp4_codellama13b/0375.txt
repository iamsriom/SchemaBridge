-- Schema for database: rschema_0375

CREATE TABLE devices (
  device_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  ip_address VARCHAR(255) NOT NULL,
  installation_location VARCHAR(255) NOT NULL,
  installation_time DATETIME NOT NULL,
  device_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (device_id)
);

CREATE TABLE users (
  user_id INT(11) NOT NULL,
  user_account VARCHAR(255) NOT NULL,
  user_password VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id)
);

CREATE TABLE network_traffic (
  device_id INT(11) NOT NULL,
  timestamp DATETIME NOT NULL,
  upload_traffic INT(11) NOT NULL,
  download_traffic INT(11) NOT NULL,
  total_traffic INT(11) NOT NULL,
  traffic_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (device_id, timestamp),
  FOREIGN KEY (device_id) REFERENCES devices (device_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE device_failures (
  device_id INT(11) NOT NULL,
  failure_occurrence_time DATETIME NOT NULL,
  failure_type VARCHAR(255) NOT NULL,
  failure_description VARCHAR(255) NOT NULL,
  handling_status VARCHAR(255) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (device_id, failure_occurrence_time),
  FOREIGN KEY (device_id) REFERENCES devices (device_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE staff (
  staff_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  PRIMARY KEY (staff_id)
);

CREATE TABLE user_connections (
  user_id INT(11) NOT NULL,
  device_id INT(11) NOT NULL,
  connection_time DATETIME NOT NULL,
  disconnection_time DATETIME NOT NULL,
  bandwidth_used INT(11) NOT NULL,
  user_ip VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_id, device_id, connection_time),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (device_id) REFERENCES devices (device_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE device_maintenance (
  device_id INT(11) NOT NULL,
  staff_id INT(11) NOT NULL,
  maintenance_time DATETIME NOT NULL,
  maintenance_content VARCHAR(255) NOT NULL,
  maintenance_result VARCHAR(255) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (device_id, staff_id, maintenance_time),
  FOREIGN KEY (device_id) REFERENCES devices (device_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (staff_id) REFERENCES staff (staff_id) ON DELETE CASCADE ON UPDATE CASCADE
);
