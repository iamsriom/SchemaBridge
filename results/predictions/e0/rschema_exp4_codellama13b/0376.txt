-- Schema for database: rschema_0376

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
);

CREATE TABLE technicians (
  technician_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  contact_address VARCHAR(255) NOT NULL,
  PRIMARY KEY (technician_id)
);

CREATE TABLE car_models (
  model_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (model_id)
);

CREATE TABLE repair_tasks (
  task_id INT(11) NOT NULL AUTO_INCREMENT,
  model_id INT(11) NOT NULL,
  status VARCHAR(255) NOT NULL,
  customer_id INT(11) NOT NULL,
  technician_id INT(11) NOT NULL,
  task_creation_date DATETIME NOT NULL,
  completion_date DATETIME,
  PRIMARY KEY (task_id),
  FOREIGN KEY (model_id) REFERENCES car_models (model_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (technician_id) REFERENCES technicians (technician_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE orders (
  order_id INT(11) NOT NULL AUTO_INCREMENT,
  task_id INT(11) NOT NULL,
  order_creation_date DATETIME NOT NULL,
  delivery_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (order_id),
  FOREIGN KEY (task_id) REFERENCES repair_tasks (task_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE suppliers (
  supplier_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (supplier_id)
);

CREATE TABLE parts (
  part_id INT(11) NOT NULL AUTO_INCREMENT,
  serial_number VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  supplier_id INT(11) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  PRIMARY KEY (part_id),
  FOREIGN KEY (supplier_id) REFERENCES suppliers (supplier_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE order_parts (
  order_id INT(11) NOT NULL,
  part_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (order_id, part_id),
  FOREIGN KEY (order_id) REFERENCES orders (order_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (part_id) REFERENCES parts (part_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE repair_task_parts (
  task_id INT(11) NOT NULL,
  part_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (task_id, part_id),
  FOREIGN KEY (task_id) REFERENCES repair_tasks (task_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (part_id) REFERENCES parts (part_id) ON DELETE CASCADE ON UPDATE CASCADE
);
