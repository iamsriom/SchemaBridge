-- Schema for database: rschema_0384

CREATE TABLE `task` (
  `task_id` int(11) NOT NULL,
  `task_title` varchar(255) NOT NULL,
  `creation_date` datetime NOT NULL,
  `task_status` varchar(255) NOT NULL,
  `end_date` datetime NOT NULL,
  `status_reason` varchar(255) NOT NULL,
  `query_statement` varchar(255) NOT NULL,
  `database` varchar(255) NOT NULL,
  `executable_file` varchar(255) NOT NULL,
  `algorithm` varchar(255) NOT NULL,
  `options` varchar(255) NOT NULL,
  `task_owner` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_parameter` (
  `parameter_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `parameter_name` varchar(255) NOT NULL,
  `parameter_value` varchar(255) NOT NULL,
  PRIMARY KEY (`parameter_id`),
  FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_related_organism` (
  `organism_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `organism_name` varchar(255) NOT NULL,
  PRIMARY KEY (`organism_id`),
  FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_excluded_organism` (
  `organism_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `organism_name` varchar(255) NOT NULL,
  PRIMARY KEY (`organism_id`),
  FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
