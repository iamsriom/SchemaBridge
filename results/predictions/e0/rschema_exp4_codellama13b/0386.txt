-- Schema for database: rschema_0386

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `public_key` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `block` (
  `block_id` int(11) NOT NULL,
  `block_hash` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  `block_size` int(11) NOT NULL,
  `previous_block_id` int(11) NOT NULL,
  PRIMARY KEY (`block_id`),
  FOREIGN KEY (`previous_block_id`) REFERENCES `block` (`block_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_record` (
  `transaction_id` int(11) NOT NULL,
  `sender_id` int(11) NOT NULL,
  `receiver_id` int(11) NOT NULL,
  `transaction_amount` int(11) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `block_id` int(11) NOT NULL,
  `operation_type` varchar(255) NOT NULL,
  `operation_content` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  FOREIGN KEY (`sender_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`receiver_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`block_id`) REFERENCES `block` (`block_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
