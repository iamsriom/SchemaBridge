-- Schema for database: rschema_0390

CREATE TABLE consumers (
  consumer_id int(11) NOT NULL AUTO_INCREMENT,
  address varchar(255) NOT NULL,
  phone_number varchar(255) NOT NULL,
  PRIMARY KEY (consumer_id)
);

CREATE TABLE courier_companies (
  company_id int(11) NOT NULL AUTO_INCREMENT,
  company_name varchar(255) NOT NULL,
  arrival_time datetime NOT NULL,
  charging_price int(11) NOT NULL,
  PRIMARY KEY (company_id)
);

CREATE TABLE courier_tracking (
  courier_number int(11) NOT NULL AUTO_INCREMENT,
  recipient_id int(11) NOT NULL,
  recipient_name varchar(255) NOT NULL,
  recipient_phone_number varchar(255) NOT NULL,
  PRIMARY KEY (courier_number),
  FOREIGN KEY (recipient_id) REFERENCES consumers (consumer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE administrators (
  administrator_id int(11) NOT NULL AUTO_INCREMENT,
  name varchar(255) NOT NULL,
  login_password varchar(255) NOT NULL,
  courier_user_name varchar(255) NOT NULL,
  PRIMARY KEY (administrator_id)
);

CREATE TABLE partnerships (
  partner_id int(11) NOT NULL AUTO_INCREMENT,
  company_id int(11) NOT NULL,
  administrator_id int(11) NOT NULL,
  PRIMARY KEY (partner_id),
  FOREIGN KEY (company_id) REFERENCES courier_companies (company_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (administrator_id) REFERENCES administrators (administrator_id) ON DELETE CASCADE ON UPDATE CASCADE
);
