-- Schema for database: rschema_0395

CREATE TABLE `students` (
`student_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`department` varchar(255) NOT NULL,
`contact_info` varchar(255) NOT NULL,
PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meal_cards` (
`card_id` int(11) NOT NULL,
`student_id` int(11) NOT NULL,
`balance` int(11) NOT NULL,
`recharge_date` datetime NOT NULL,
`recharge_location` varchar(255) NOT NULL,
PRIMARY KEY (`card_id`),
FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consumption_records` (
`record_id` int(11) NOT NULL,
`card_id` int(11) NOT NULL,
`amount` int(11) NOT NULL,
`location` varchar(255) NOT NULL,
`time` datetime NOT NULL,
PRIMARY KEY (`record_id`),
FOREIGN KEY (`card_id`) REFERENCES `meal_cards` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consumption_statistics` (
`stat_id` int(11) NOT NULL,
`record_id` int(11) NOT NULL,
`amount` int(11) NOT NULL,
`location` varchar(255) NOT NULL,
`time` datetime NOT NULL,
PRIMARY KEY (`stat_id`),
FOREIGN KEY (`record_id`) REFERENCES `consumption_records` (`record_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consumption_analysis` (
`analysis_id` int(11) NOT NULL,
`stat_id` int(11) NOT NULL,
`amount` int(11) NOT NULL,
`location` varchar(255) NOT NULL,
`time` datetime NOT NULL,
PRIMARY KEY (`analysis_id`),
FOREIGN KEY (`stat_id`) REFERENCES `consumption_statistics` (`stat_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
