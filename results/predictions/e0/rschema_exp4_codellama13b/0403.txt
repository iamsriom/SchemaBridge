-- Schema for database: rschema_0403

CREATE TABLE `students` (
`student_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`student_number` varchar(255) NOT NULL,
`grade` varchar(255) NOT NULL,
`major` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrators` (
`administrator_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`job_number` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
PRIMARY KEY (`administrator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `books` (
`book_id` int(11) NOT NULL,
`title` varchar(255) NOT NULL,
`author` varchar(255) NOT NULL,
`publisher` varchar(255) NOT NULL,
`publication_date` datetime NOT NULL,
`category` varchar(255) NOT NULL,
`stock_quantity` int(11) NOT NULL,
`borrowed_quantity` int(11) NOT NULL,
PRIMARY KEY (`book_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `borrowing_records` (
`student_id` int(11) NOT NULL,
`borrowed_book_id` int(11) NOT NULL,
`borrow_date` datetime NOT NULL,
`return_date` datetime NOT NULL,
`review_status` varchar(255) NOT NULL,
PRIMARY KEY (`student_id`, `borrowed_book_id`),
FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`borrowed_book_id`) REFERENCES `books` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `returning_records` (
`student_id` int(11) NOT NULL,
`borrowed_book_id` int(11) NOT NULL,
`return_date` datetime NOT NULL,
`review_status` varchar(255) NOT NULL,
PRIMARY KEY (`student_id`, `borrowed_book_id`),
FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`borrowed_book_id`) REFERENCES `books` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `audit_records` (
`administrator_id` int(11) NOT NULL,
`book_id` int(11) NOT NULL,
`borrowing_status` varchar(255) NOT NULL,
`audit_date` datetime NOT NULL,
`audit_result` varchar(255) NOT NULL,
PRIMARY KEY (`administrator_id`, `book_id`),
FOREIGN KEY (`administrator_id`) REFERENCES `administrators` (`administrator_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`book_id`) REFERENCES `books` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
