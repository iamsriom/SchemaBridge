-- Schema for database: rschema_0405

CREATE TABLE drug (
  drug_id INT(11) NOT NULL AUTO_INCREMENT,
  drug_name VARCHAR(255) NOT NULL,
  drug_specifications VARCHAR(255) NOT NULL,
  usage VARCHAR(255) NOT NULL,
  PRIMARY KEY (drug_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE doctor (
  doctor_id INT(11) NOT NULL AUTO_INCREMENT,
  doctor_name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (doctor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE patient (
  patient_id INT(11) NOT NULL AUTO_INCREMENT,
  patient_name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (patient_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE prescription (
  prescription_id INT(11) NOT NULL AUTO_INCREMENT,
  medical_order VARCHAR(255) NOT NULL,
  date DATE NOT NULL,
  category VARCHAR(255) NOT NULL,
  prescribing_doctor INT(11) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (prescription_id),
  FOREIGN KEY (prescribing_doctor) REFERENCES doctor(doctor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE outpatient (
  consultation_id INT(11) NOT NULL AUTO_INCREMENT,
  patient_id INT(11) NOT NULL,
  doctor_id INT(11) NOT NULL,
  date DATE NOT NULL,
  PRIMARY KEY (consultation_id),
  FOREIGN KEY (patient_id) REFERENCES patient(patient_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (doctor_id) REFERENCES doctor(doctor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE payment (
  payment_id INT(11) NOT NULL AUTO_INCREMENT,
  prescription_id INT(11) NOT NULL,
  payment_amount DECIMAL(10,2) NOT NULL,
  discount DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (payment_id),
  FOREIGN KEY (prescription_id) REFERENCES prescription(prescription_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
