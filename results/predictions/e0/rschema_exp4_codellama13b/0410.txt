-- Schema for database: rschema_0410

CREATE TABLE `users` (
`user_id` int(11) NOT NULL,
`user_name` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
`vehicle_id` int(11) NOT NULL,
`vehicle_name` varchar(255) NOT NULL,
`vehicle_photo` varchar(255) NOT NULL,
`vehicle_status` varchar(255) NOT NULL,
`vehicle_category` varchar(255) NOT NULL,
PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance` (
`maintenance_id` int(11) NOT NULL,
`vehicle_id` int(11) NOT NULL,
`maintenance_time` datetime NOT NULL,
`maintenance_details` varchar(255) NOT NULL,
`review` varchar(255) NOT NULL,
`review_reply` varchar(255) NOT NULL,
PRIMARY KEY (`maintenance_id`),
FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departures` (
`departure_id` int(11) NOT NULL,
`vehicle_id` int(11) NOT NULL,
`departure_time` datetime NOT NULL,
`departure_notes` varchar(255) NOT NULL,
PRIMARY KEY (`departure_id`),
FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_applications` (
`application_id` int(11) NOT NULL,
`vehicle_id` int(11) NOT NULL,
`application_time` datetime NOT NULL,
`application_reason` varchar(255) NOT NULL,
`review` varchar(255) NOT NULL,
`review_reply` varchar(255) NOT NULL,
PRIMARY KEY (`application_id`),
FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_shops` (
`repair_shop_id` int(11) NOT NULL,
`shop_name` varchar(255) NOT NULL,
`contact_person` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
PRIMARY KEY (`repair_shop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcements` (
`announcement_id` int(11) NOT NULL,
`title` varchar(255) NOT NULL,
`content` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tokens` (
`token_id` int(11) NOT NULL,
`token_value` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
`expiration_time` datetime NOT NULL,
PRIMARY KEY (`token_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
