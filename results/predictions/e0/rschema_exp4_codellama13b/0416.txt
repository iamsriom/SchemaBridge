-- Schema for database: rschema_0416

CREATE TABLE `users` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`avatar` varchar(255) NOT NULL,
`nickname` varchar(255) NOT NULL,
`registration_time` datetime NOT NULL,
`role` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `companies` (
`company_id` int(11) NOT NULL AUTO_INCREMENT,
`company_name` varchar(255) NOT NULL,
`company_address` varchar(255) NOT NULL,
`company_profile` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `jobs` (
`job_id` int(11) NOT NULL AUTO_INCREMENT,
`title` varchar(255) NOT NULL,
`content` varchar(255) NOT NULL,
`requirements` varchar(255) NOT NULL,
`job_category` varchar(255) NOT NULL,
`salary` varchar(255) NOT NULL,
`education_level` varchar(255) NOT NULL,
`work_location` varchar(255) NOT NULL,
PRIMARY KEY (`job_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conversations` (
`conversation_id` int(11) NOT NULL AUTO_INCREMENT,
`message_content` varchar(255) NOT NULL,
`send_time` datetime NOT NULL,
PRIMARY KEY (`conversation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `messages` (
`message_id` int(11) NOT NULL AUTO_INCREMENT,
`message_content` varchar(255) NOT NULL,
`message_time` datetime NOT NULL,
PRIMARY KEY (`message_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_jobs` (
`user_id` int(11) NOT NULL,
`job_id` int(11) NOT NULL,
PRIMARY KEY (`user_id`, `job_id`),
CONSTRAINT `user_jobs_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `user_jobs_ibfk_2` FOREIGN KEY (`job_id`) REFERENCES `jobs` (`job_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `company_jobs` (
`company_id` int(11) NOT NULL,
`job_id` int(11) NOT NULL,
PRIMARY KEY (`company_id`, `job_id`),
CONSTRAINT `company_jobs_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `companies` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `company_jobs_ibfk_2` FOREIGN KEY (`job_id`) REFERENCES `jobs` (`job_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conversation_messages` (
`conversation_id` int(11) NOT NULL,
`message_id` int(11) NOT NULL,
PRIMARY KEY (`conversation_id`, `message_id`),
CONSTRAINT `conversation_messages_ibfk_1` FOREIGN KEY (`conversation_id`) REFERENCES `conversations` (`conversation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `conversation_messages_ibfk_2` FOREIGN KEY (`message_id`) REFERENCES `messages` (`message_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
