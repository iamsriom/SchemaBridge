-- Schema for database: rschema_0417

CREATE TABLE `users` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`avatar_link` varchar(255) NOT NULL,
`registration_date` datetime NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `posts` (
`post_id` int(11) NOT NULL AUTO_INCREMENT,
`title` varchar(255) NOT NULL,
`content` text NOT NULL,
`category_id` int(11) NOT NULL,
`posting_user_id` int(11) NOT NULL,
`posting_time` datetime NOT NULL,
`status` varchar(255) NOT NULL,
`reply_count` int(11) NOT NULL,
`like_count` int(11) NOT NULL,
`favorite_count` int(11) NOT NULL,
PRIMARY KEY (`post_id`),
FOREIGN KEY (`posting_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`category_id`) REFERENCES `categories` (`category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comments` (
`comment_id` int(11) NOT NULL AUTO_INCREMENT,
`comment_content` text NOT NULL,
`posting_user_id` int(11) NOT NULL,
`like_count` int(11) NOT NULL,
`reply_count` int(11) NOT NULL,
`posting_time` datetime NOT NULL,
PRIMARY KEY (`comment_id`),
FOREIGN KEY (`posting_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `categories` (
`category_id` int(11) NOT NULL AUTO_INCREMENT,
`category_name` varchar(255) NOT NULL,
PRIMARY KEY (`category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notifications` (
`message_number` int(11) NOT NULL AUTO_INCREMENT,
`notification_content` text NOT NULL,
`user_id` int(11) NOT NULL,
`read_status` varchar(255) NOT NULL,
PRIMARY KEY (`message_number`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
