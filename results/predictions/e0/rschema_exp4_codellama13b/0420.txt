CREATE TABLE `users` (
`user_id` int(11) NOT NULL,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`wechat_openid` varchar(255) NOT NULL,
`name` varchar(255) NOT NULL,
`qq_number` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`registration_time` datetime NOT NULL,
`account_status` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `news` (
`news_id` int(11) NOT NULL,
`title` varchar(255) NOT NULL,
`content` text NOT NULL,
`publication_time` datetime NOT NULL,
`read_count` int(11) NOT NULL,
PRIMARY KEY (`news_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

`category_id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `news_status` (
`status_id` int(11) NOT NULL,
`status_name` varchar(255) NOT NULL,
PRIMARY KEY (`status_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `videos` (
`video_id` int(11) NOT NULL,
`title` varchar(255) NOT NULL,
`description` text NOT NULL,
`cover_image` varchar(255) NOT NULL,

PRIMARY KEY (`video_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `video_categories` (
`category_id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `video_status` (

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

`instructor_id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

`instructor_id` int(11) NOT NULL,

`professional_field` varchar(255) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `associated_videos` (
`instructor_id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `collections` (
`user_id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
`user_id` int(11) NOT NULL,

`payment_status` varchar(255) NOT NULL,
`payment_time` datetime NOT NULL,
`order_status` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`, `video_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_records` (
`user_id` int(11) NOT NULL,

CREATE TABLE `learning_records` (
