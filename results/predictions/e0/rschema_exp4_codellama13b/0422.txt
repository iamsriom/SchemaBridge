-- Schema for database: rschema_0422

CREATE TABLE `users` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`user_name` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
`registration_time` datetime NOT NULL,
`last_login_time` datetime NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
`project_id` int(11) NOT NULL AUTO_INCREMENT,
`project_name` varchar(255) NOT NULL,
`project_introduction` varchar(255) NOT NULL,
`responsible_person_id` int(11) NOT NULL,
`creation_time` datetime NOT NULL,
`project_status` varchar(255) NOT NULL,
PRIMARY KEY (`project_id`),
FOREIGN KEY (`responsible_person_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_versions` (
`version_id` int(11) NOT NULL AUTO_INCREMENT,
`project_id` int(11) NOT NULL,
`version_name` varchar(255) NOT NULL,
`release_date` datetime NOT NULL,
`version_description` varchar(255) NOT NULL,
`is_stable_version` varchar(255) NOT NULL,
PRIMARY KEY (`version_id`),
FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tasks` (
`task_id` int(11) NOT NULL AUTO_INCREMENT,
`project_id` int(11) NOT NULL,
`assigned_to_user_id` int(11) NOT NULL,
`task_name` varchar(255) NOT NULL,
`task_status` varchar(255) NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
`task_description` varchar(255) NOT NULL,
PRIMARY KEY (`task_id`),
FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`assigned_to_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `code_submissions` (
`submission_id` int(11) NOT NULL AUTO_INCREMENT,
`user_id` int(11) NOT NULL,
`project_id` int(11) NOT NULL,
`submission_time` datetime NOT NULL,
`submission_description` varchar(255) NOT NULL,
`code_path` varchar(255) NOT NULL,
PRIMARY KEY (`submission_id`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
