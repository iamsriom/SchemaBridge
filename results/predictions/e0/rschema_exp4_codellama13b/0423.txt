-- Schema for database: rschema_0423

CREATE TABLE courses (
  course_id INT(11) NOT NULL AUTO_INCREMENT,
  course_name VARCHAR(255) NOT NULL,
  course_description TEXT,
  PRIMARY KEY (course_id)
);

CREATE TABLE course_details (
  course_id INT(11) NOT NULL,
  num_lessons INT(11) NOT NULL,
  total_duration INT(11) NOT NULL,
  difficulty_level VARCHAR(255) NOT NULL,
  PRIMARY KEY (course_id),
  FOREIGN KEY (course_id) REFERENCES courses (course_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE course_classifications (
  course_id INT(11) NOT NULL,
  course_category VARCHAR(255) NOT NULL,
  target_audience VARCHAR(255) NOT NULL,
  recommendation_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (course_id),
  FOREIGN KEY (course_id) REFERENCES courses (course_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE chapters (
  chapter_id INT(11) NOT NULL AUTO_INCREMENT,
  chapter_name VARCHAR(255) NOT NULL,
  chapter_description TEXT,
  PRIMARY KEY (chapter_id)
);

CREATE TABLE chapter_details (
  chapter_id INT(11) NOT NULL,
  video_url VARCHAR(255) NOT NULL,
  course_materials TEXT,
  duration INT(11) NOT NULL,
  PRIMARY KEY (chapter_id),
  FOREIGN KEY (chapter_id) REFERENCES chapters (chapter_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE chapter_sequences (
  chapter_id INT(11) NOT NULL,
  previous_chapter_id INT(11) NOT NULL,
  next_chapter_id INT(11) NOT NULL,
  chapter_hierarchy INT(11) NOT NULL,
  PRIMARY KEY (chapter_id),
  FOREIGN KEY (chapter_id) REFERENCES chapters (chapter_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (previous_chapter_id) REFERENCES chapters (chapter_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (next_chapter_id) REFERENCES chapters (chapter_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE instructors (
  instructor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  avatar VARCHAR(255) NOT NULL,
  PRIMARY KEY (instructor_id)
);

CREATE TABLE instructor_details (
  instructor_id INT(11) NOT NULL,
  teaching_experience VARCHAR(255) NOT NULL,
  professional_field VARCHAR(255) NOT NULL,
  personal_profile TEXT,
  PRIMARY KEY (instructor_id),
  FOREIGN KEY (instructor_id) REFERENCES instructors (instructor_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE instructor_courses (
  instructor_id INT(11) NOT NULL,
  course_id INT(11) NOT NULL,
  teaching_list TEXT,
  ratings TEXT,
  student_feedback TEXT,
  PRIMARY KEY (instructor_id, course_id),
  FOREIGN KEY (instructor_id) REFERENCES instructors (instructor_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (course_id) REFERENCES courses (course_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE students (
  student_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  avatar VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  age INT(11) NOT NULL,
  contact_information TEXT,
  PRIMARY KEY (student_id)
);

CREATE TABLE student_accounts (
  student_id INT(11) NOT NULL,
  registration_time DATETIME NOT NULL,
  account_status VARCHAR(255) NOT NULL,
  membership_level VARCHAR(255) NOT NULL,
  points INT(11) NOT NULL,
  PRIMARY KEY (student_id),
  FOREIGN KEY (student_id) REFERENCES students (student_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE student_progress (
  student_id INT(11) NOT NULL,
  course_id INT(11) NOT NULL,
  current_learning_chapter INT(11) NOT NULL,
  last_learning_time DATETIME NOT NULL,
  PRIMARY KEY (student_id, course_id),
  FOREIGN KEY (student_id) REFERENCES students (student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (course_id) REFERENCES courses (course_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE viewing_records (
  student_id INT(11) NOT NULL,
  chapter_id INT(11) NOT NULL,
  viewing_duration INT(11) NOT NULL,
  num_views INT(11) NOT NULL,
  completion_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (student_id, chapter_id),
  FOREIGN KEY (student_id) REFERENCES students (student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (chapter_id) REFERENCES chapters (chapter_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE learning_statistics (
  student_id INT(11) NOT NULL,
  total_learning_duration INT(11) NOT NULL,
  course_completion_rate VARCHAR(255) NOT NULL,
  PRIMARY KEY (student_id),
  FOREIGN KEY (student_id) REFERENCES students (student_id) ON DELETE CASCADE ON UPDATE CASCADE
);
