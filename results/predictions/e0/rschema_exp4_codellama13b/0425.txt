-- Schema for database: rschema_0425

CREATE TABLE `users` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `buildings` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`description` varchar(255) NOT NULL,
`responsible_manager` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitories` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`number_of_beds` int(11) NOT NULL,
`belonging_building` int(11) NOT NULL,
`supporting_facilities` varchar(255) NOT NULL,
`responsible_manager` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `students` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`belonging_dormitory` int(11) NOT NULL,
`class` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `applications` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`type` varchar(255) NOT NULL,
`applicant_student_id` int(11) NOT NULL,
`application_content` varchar(255) NOT NULL,
`application_time` datetime NOT NULL,
`application_status` varchar(255) NOT NULL,
`administrator_opinion` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `type` (`type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_records` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`dormitory_id` int(11) NOT NULL,
`fault_description` varchar(255) NOT NULL,
`repair_time` datetime NOT NULL,
`maintenance_staff_id` int(11) NOT NULL,
`maintenance_situation` varchar(255) NOT NULL,
`completion_time` datetime NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `dormitory_id` (`dormitory_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcements` (
`id` int(11) NOT NULL AUTO_INCREMENT,
`title` varchar(255) NOT NULL,
`content` varchar(255) NOT NULL,
`release_time` datetime NOT NULL,
`administrator_id` int(11) NOT NULL,
PRIMARY KEY (`id`),
UNIQUE KEY `title` (`title`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
