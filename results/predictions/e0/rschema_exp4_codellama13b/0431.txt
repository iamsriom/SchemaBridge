-- Schema for database: rschema_0431

CREATE TABLE `departments` (
`department_id` int(11) NOT NULL AUTO_INCREMENT,
`department_name` varchar(255) NOT NULL,
`introduction` text NOT NULL,
`doctors` varchar(255) NOT NULL,
PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctors` (
`doctor_id` int(11) NOT NULL AUTO_INCREMENT,
`doctor_name` varchar(255) NOT NULL,
`department_id` int(11) NOT NULL,
`consultation_time` datetime NOT NULL,
`remaining_appointments` int(11) NOT NULL,
PRIMARY KEY (`doctor_id`),
FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `appointments` (
`appointment_id` int(11) NOT NULL AUTO_INCREMENT,
`doctor_id` int(11) NOT NULL,
`appointment_time` datetime NOT NULL,
`user_id` int(11) NOT NULL,
`status` varchar(255) NOT NULL,
PRIMARY KEY (`appointment_id`),
FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`contact_information` text NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrators` (
`administrator_id` int(11) NOT NULL AUTO_INCREMENT,
`username` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`contact_information` text NOT NULL,
PRIMARY KEY (`administrator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `statistics` (
`statistic_id` int(11) NOT NULL AUTO_INCREMENT,
`appointment_id` int(11) NOT NULL,
`statistic_type` varchar(255) NOT NULL,
`statistic_value` varchar(255) NOT NULL,
PRIMARY KEY (`statistic_id`),
FOREIGN KEY (`appointment_id`) REFERENCES `appointments` (`appointment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `data_security` (
`data_security_id` int(11) NOT NULL AUTO_INCREMENT,
`data_security_type` varchar(255) NOT NULL,
`data_security_value` varchar(255) NOT NULL,
PRIMARY KEY (`data_security_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `optimization` (
`optimization_id` int(11) NOT NULL AUTO_INCREMENT,
`optimization_type` varchar(255) NOT NULL,
`optimization_value` varchar(255) NOT NULL,
PRIMARY KEY (`optimization_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
