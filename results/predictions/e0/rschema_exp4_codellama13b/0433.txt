-- Schema for database: rschema_0433

CREATE TABLE business_entities (
  id INT(11) NOT NULL AUTO_INCREMENT,
  type VARCHAR(255) NOT NULL,
  module_code VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  creation_time DATETIME NOT NULL,
  update_time DATETIME NOT NULL,
  creator_user_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (creator_user_id) REFERENCES users(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE attachments (
  id INT(11) NOT NULL AUTO_INCREMENT,
  business_entity_id INT(11) NOT NULL,
  file_name VARCHAR(255) NOT NULL,
  file_size INT(11) NOT NULL,
  file_path VARCHAR(255) NOT NULL,
  creation_time DATETIME NOT NULL,
  file_status VARCHAR(255) NOT NULL,
  uploader_user_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (business_entity_id) REFERENCES business_entities(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (uploader_user_id) REFERENCES users(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  department VARCHAR(255) NOT NULL,
  role VARCHAR(255) NOT NULL,
  contact_details VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
