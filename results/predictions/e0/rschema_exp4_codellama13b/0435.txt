-- Schema for database: rschema_0435

CREATE TABLE `probes` (
  `probe_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `weight` int(11) NOT NULL,
  `launch_date` datetime NOT NULL,
  `service_life` int(11) NOT NULL,
  PRIMARY KEY (`probe_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `celestial_bodies` (
  `celestial_body_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `orbital_parameters` varchar(255) NOT NULL,
  `physical_characteristics` varchar(255) NOT NULL,
  PRIMARY KEY (`celestial_body_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `observational_data` (
  `observation_id` int(11) NOT NULL,
  `probe_id` int(11) NOT NULL,
  `celestial_body_id` int(11) NOT NULL,
  `observation_time` datetime NOT NULL,
  `observation_data` varchar(255) NOT NULL,
  `observation_results` varchar(255) NOT NULL,
  `observation_accuracy` varchar(255) NOT NULL,
  PRIMARY KEY (`observation_id`),
  FOREIGN KEY (`probe_id`) REFERENCES `probes` (`probe_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`celestial_body_id`) REFERENCES `celestial_bodies` (`celestial_body_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instruments` (
  `instrument_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `weight` int(11) NOT NULL,
  `power_consumption` int(11) NOT NULL,
  `accuracy` varchar(255) NOT NULL,
  PRIMARY KEY (`instrument_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instrument_operating_parameters` (
  `instrument_id` int(11) NOT NULL,
  `operating_parameter` varchar(255) NOT NULL,
  `value` varchar(255) NOT NULL,
  PRIMARY KEY (`instrument_id`, `operating_parameter`),
  FOREIGN KEY (`instrument_id`) REFERENCES `instruments` (`instrument_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instrument_calibration_data` (
  `instrument_id` int(11) NOT NULL,
  `calibration_data` varchar(255) NOT NULL,
  PRIMARY KEY (`instrument_id`),
  FOREIGN KEY (`instrument_id`) REFERENCES `instruments` (`instrument_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mission_planning` (
  `mission_id` int(11) NOT NULL,
  `probe_id` int(11) NOT NULL,
  `celestial_body_id` int(11) NOT NULL,
  `flight_trajectory` varchar(255) NOT NULL,
  `mission_execution_status` varchar(255) NOT NULL,
  `scientific_data_acquisition_status` varchar(255) NOT NULL,
  PRIMARY KEY (`mission_id`),
  FOREIGN KEY (`probe_id`) REFERENCES `probes` (`probe_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`celestial_body_id`) REFERENCES `celestial_bodies` (`celestial_body_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mission_priorities` (
  `mission_id` int(11) NOT NULL,
  `priority` int(11) NOT NULL,
  PRIMARY KEY (`mission_id`),
  FOREIGN KEY (`mission_id`) REFERENCES `mission_planning` (`mission_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_allocation` (
  `mission_id` int(11) NOT NULL,
  `resource_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`mission_id`, `resource_id`),
  FOREIGN KEY (`mission_id`) REFERENCES `mission_planning` (`mission_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`resource_id`) REFERENCES `instruments` (`instrument_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
