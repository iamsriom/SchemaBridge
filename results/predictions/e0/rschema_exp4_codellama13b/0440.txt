-- Schema for database: rschema_0440

CREATE TABLE `devices` (
`device_id` int(11) NOT NULL AUTO_INCREMENT,
`product_name` varchar(255) NOT NULL,
`device_model` varchar(255) NOT NULL,
`person_in_charge` varchar(255) NOT NULL,
`group` varchar(255) NOT NULL,
`type` varchar(255) NOT NULL,
`usage_location` varchar(255) NOT NULL,
`department` varchar(255) NOT NULL,
`ip_address` varchar(255) NOT NULL,
`device_images` varchar(255) NOT NULL,
`qr_code` varchar(255) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`device_id`),
UNIQUE KEY `qr_code` (`qr_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`role` varchar(255) NOT NULL,
`contact_information` varchar(255) NOT NULL,
`account_status` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_inventory` (
`device_id` int(11) NOT NULL,
`inventory_person_id` int(11) NOT NULL,
`inventory_time` datetime NOT NULL,
`device_location` varchar(255) NOT NULL,
PRIMARY KEY (`device_id`),
FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_requests` (
`device_id` int(11) NOT NULL,
`requester_id` int(11) NOT NULL,
`maintenance_time` datetime NOT NULL,
`maintenance_content` varchar(255) NOT NULL,
`handling_status` varchar(255) NOT NULL,
`handling_personnel` varchar(255) NOT NULL,
`handling_time` datetime NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`device_id`),
FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_details` (
`device_id` int(11) NOT NULL,
`maintenance_personnel_id` int(11) NOT NULL,
`maintenance_time` datetime NOT NULL,
`maintenance_content` varchar(255) NOT NULL,
`maintenance_results` varchar(255) NOT NULL,
`user_satisfaction` varchar(255) NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`device_id`),
FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_status` (
`device_id` int(11) NOT NULL,
`current_status` varchar(255) NOT NULL,
`last_update_time` datetime NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`device_id`),
FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
