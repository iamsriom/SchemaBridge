-- Schema for database: rschema_0441

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `budget` decimal(10,2) NOT NULL,
  `remaining_funds` decimal(10,2) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `record_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `transaction_amount` decimal(10,2) NOT NULL,
  `purpose_description` varchar(255) NOT NULL,
  `transaction_operator` varchar(255) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  `user_name` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_roles` (
  `user_id` int(11) NOT NULL,
  `role` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`, `role`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`role`) REFERENCES `roles` (`role`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `roles` (
  `role` varchar(255) NOT NULL,
  `permission` varchar(255) NOT NULL,
  PRIMARY KEY (`role`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role_permissions` (
  `role` varchar(255) NOT NULL,
  `permission` varchar(255) NOT NULL,
  PRIMARY KEY (`role`, `permission`),
  FOREIGN KEY (`role`) REFERENCES `roles` (`role`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`permission`) REFERENCES `permissions` (`permission`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `permissions` (
  `permission` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  PRIMARY KEY (`permission`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `analytics` (
  `project_id` int(11) NOT NULL,
  `analytic_type` varchar(255) NOT NULL,
  `analytic_value` decimal(10,2) NOT NULL,
  PRIMARY KEY (`project_id`, `analytic_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reports` (
  `report_id` int(11) NOT NULL,
  `report_name` varchar(255) NOT NULL,
  `report_type` varchar(255) NOT NULL,
  `report_data` text NOT NULL,
  PRIMARY KEY (`report_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `report_filters` (
  `report_id` int(11) NOT NULL,
  `filter_type` varchar(255) NOT NULL,
  `filter_value` varchar(255) NOT NULL,
  PRIMARY KEY (`report_id`, `filter_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `report_columns` (
  `report_id` int(11) NOT NULL,
  `column_name` varchar(255) NOT NULL,
  `column_type` varchar(255) NOT NULL,
  PRIMARY KEY (`report_id`, `column_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `report_data` (
  `report_id` int(11) NOT NULL,
  `data_row` int(11) NOT NULL,
  `data_column` int(11) NOT NULL,
  `data_value` varchar(255) NOT NULL,
  PRIMARY KEY (`report_id`, `data_row`, `data_column`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
