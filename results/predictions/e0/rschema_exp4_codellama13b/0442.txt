-- Schema for database: rschema_0442

CREATE TABLE medical_waste (
  id INT(11) NOT NULL AUTO_INCREMENT,
  waste_type VARCHAR(255) NOT NULL,
  weight DECIMAL(10,2) NOT NULL,
  generation_time DATETIME NOT NULL,
  collection_time DATETIME NOT NULL,
  current_status VARCHAR(255) NOT NULL,
  traceability_code VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (traceability_code) REFERENCES medical_waste_traceability (traceability_code) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE medical_institution (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE department (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (medical_institution_id) REFERENCES medical_institution (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE personnel (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  role VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (department_id) REFERENCES department (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE vehicle (
  id INT(11) NOT NULL AUTO_INCREMENT,
  license_plate_number VARCHAR(255) NOT NULL,
  model VARCHAR(255) NOT NULL,
  capacity DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (medical_institution_id) REFERENCES medical_institution (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE medical_waste_inbound (
  id INT(11) NOT NULL AUTO_INCREMENT,
  batch_number VARCHAR(255) NOT NULL,
  expected_quantity DECIMAL(10,2) NOT NULL,
  actual_quantity DECIMAL(10,2) NOT NULL,
  expected_weight DECIMAL(10,2) NOT NULL,
  actual_weight DECIMAL(10,2) NOT NULL,
  inbound_time DATETIME NOT NULL,
  collector_id INT(11) NOT NULL,
  confirmer_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (collector_id) REFERENCES personnel (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (confirmer_id) REFERENCES personnel (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE medical_waste_outbound (
  id INT(11) NOT NULL AUTO_INCREMENT,
  batch_number VARCHAR(255) NOT NULL,
  outbound_quantity DECIMAL(10,2) NOT NULL,
  outbound_weight DECIMAL(10,2) NOT NULL,
  outbound_time DATETIME NOT NULL,
  transporter_id INT(11) NOT NULL,
  confirmer_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (transporter_id) REFERENCES personnel (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (confirmer_id) REFERENCES personnel (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE transportation (
  id INT(11) NOT NULL AUTO_INCREMENT,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  recipient VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (medical_waste_outbound_id) REFERENCES medical_waste_outbound (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE medical_waste_traceability (
  traceability_code VARCHAR(255) NOT NULL,
  generation_time DATETIME NOT NULL,
  PRIMARY KEY (traceability_code)
);
