-- Schema for database: rschema_0443

CREATE TABLE `user_information` (
`user_id` int(11) NOT NULL AUTO_INCREMENT,
`user_name` varchar(255) NOT NULL,
`login_password` varchar(255) NOT NULL,
`registered_mobile_number` varchar(255) NOT NULL,
`role_permissions` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `asset_information` (
`asset_id` int(11) NOT NULL AUTO_INCREMENT,
`equipment_name` varchar(255) NOT NULL,
`equipment_id` varchar(255) NOT NULL,
`equipment_model` varchar(255) NOT NULL,
`equipment_status` varchar(255) NOT NULL,
`calibration_date` datetime NOT NULL,
`calibration_validity` varchar(255) NOT NULL,
`calibration_results` varchar(255) NOT NULL,
`equipment_location` varchar(255) NOT NULL,
`owning_department` varchar(255) NOT NULL,
`person_in_charge` varchar(255) NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`asset_id`),
FOREIGN KEY (`user_id`) REFERENCES `user_information` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status_log_information` (
`status_log_id` int(11) NOT NULL AUTO_INCREMENT,
`equipment_id` varchar(255) NOT NULL,
`borrower` varchar(255) NOT NULL,
`borrowing_time` datetime NOT NULL,
`borrowed_equipment_photo` varchar(255) NOT NULL,
`iou_photo` varchar(255) NOT NULL,
`recipient` varchar(255) NOT NULL,
`receiving_area` varchar(255) NOT NULL,
`receiving_time` datetime NOT NULL,
`received_equipment_photo` varchar(255) NOT NULL,
`returner` varchar(255) NOT NULL,
`return_time` datetime NOT NULL,
`returned_equipment_photo` varchar(255) NOT NULL,
`data_monitoring` varchar(255) NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`status_log_id`),
FOREIGN KEY (`asset_id`) REFERENCES `asset_information` (`asset_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account_activity_log` (
`account_activity_log_id` int(11) NOT NULL AUTO_INCREMENT,
`equipment_id` varchar(255) NOT NULL,
`borrowing_account` varchar(255) NOT NULL,
`receiving_account` varchar(255) NOT NULL,
`returning_account` varchar(255) NOT NULL,
`remarks` varchar(255) NOT NULL,
PRIMARY KEY (`account_activity_log_id`),
FOREIGN KEY (`asset_id`) REFERENCES `asset_information` (`asset_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
