-- Schema for database: rschema_0445

CREATE TABLE `personnel` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`personnel_type` varchar(255) NOT NULL,
`contact_phone_number` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `unit` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`unit_type` varchar(255) NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrator` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`management_role` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `health_info` (
`id` int(11) NOT NULL,
`personnel_id` int(11) NOT NULL,
`temperature` float NOT NULL,
`symptoms` varchar(255) NOT NULL,
`health_code_color` varchar(255) NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trajectory_info` (
`id` int(11) NOT NULL,
`personnel_id` int(11) NOT NULL,
`entry_time` datetime NOT NULL,
`exit_time` datetime NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `visitor_appointment` (
`id` int(11) NOT NULL,
`visitor_id` int(11) NOT NULL,
`visited_personnel_id` int(11) NOT NULL,
`scheduled_entry_time` datetime NOT NULL,
`scheduled_exit_time` datetime NOT NULL,
`appointment_status` varchar(255) NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`visitor_id`) REFERENCES `visitor` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`visited_personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `access_record` (
`id` int(11) NOT NULL,
`personnel_id` int(11) NOT NULL,
`entry_time` datetime NOT NULL,
`exit_time` datetime NOT NULL,
`access_control_device_id` int(11) NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`),
FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`access_control_device_id`) REFERENCES `access_control_device` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `access_control_device` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`device_status` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `visitor` (
`id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`other_info` varchar(255) NOT NULL,
PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
