-- Schema for database: rschema_0449

CREATE TABLE employees (
  employee_id INT(11) NOT NULL AUTO_INCREMENT,
  employee_name VARCHAR(255) NOT NULL,
  employee_gender VARCHAR(255) NOT NULL,
  employee_dob DATE NOT NULL,
  department_id INT(11) NOT NULL,
  entry_date DATE NOT NULL,
  resignation_date DATE,
  position_id INT(11) NOT NULL,
  contact_info VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  status VARCHAR(255) NOT NULL,
  PRIMARY KEY (employee_id),
  FOREIGN KEY (department_id) REFERENCES departments (department_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (position_id) REFERENCES positions (position_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE positions (
  position_id INT(11) NOT NULL AUTO_INCREMENT,
  position_name VARCHAR(255) NOT NULL,
  position_responsibilities VARCHAR(255) NOT NULL,
  position_requirements VARCHAR(255) NOT NULL,
  PRIMARY KEY (position_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trainings (
  training_id INT(11) NOT NULL AUTO_INCREMENT,
  training_time DATE NOT NULL,
  training_location VARCHAR(255) NOT NULL,
  training_content VARCHAR(255) NOT NULL,
  trainer VARCHAR(255) NOT NULL,
  PRIMARY KEY (training_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE departments (
  department_id INT(11) NOT NULL AUTO_INCREMENT,
  department_name VARCHAR(255) NOT NULL,
  department_function VARCHAR(255) NOT NULL,
  department_planning VARCHAR(255) NOT NULL,
  PRIMARY KEY (department_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE employee_position_changes (
  employee_id INT(11) NOT NULL,
  position_id INT(11) NOT NULL,
  change_date DATE NOT NULL,
  reason VARCHAR(255) NOT NULL,
  PRIMARY KEY (employee_id, position_id, change_date),
  FOREIGN KEY (employee_id) REFERENCES employees (employee_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (position_id) REFERENCES positions (position_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
