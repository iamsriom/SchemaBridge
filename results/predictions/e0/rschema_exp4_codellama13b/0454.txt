-- Schema for database: rschema_0454

CREATE TABLE `supermarkets` (
`supermarket_id` int(11) NOT NULL AUTO_INCREMENT,
`supermarket_name` varchar(255) NOT NULL,
`supermarket_address` varchar(255) NOT NULL,
`supermarket_contact` varchar(255) NOT NULL,
PRIMARY KEY (`supermarket_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `supply_requests` (
`supply_request_id` int(11) NOT NULL AUTO_INCREMENT,
`supermarket_id` int(11) NOT NULL,
`product_number` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`request_date` datetime NOT NULL,
PRIMARY KEY (`supply_request_id`),
FOREIGN KEY (`supermarket_id`) REFERENCES `supermarkets` (`supermarket_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
`product_id` int(11) NOT NULL AUTO_INCREMENT,
`product_name` varchar(255) NOT NULL,
`product_price` int(11) NOT NULL,
`inventory_quantity` int(11) NOT NULL,
PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `distribution_orders` (
`distribution_order_id` int(11) NOT NULL AUTO_INCREMENT,
`supply_request_id` int(11) NOT NULL,
`distribution_date` datetime NOT NULL,
`distribution_status` varchar(255) NOT NULL,
PRIMARY KEY (`distribution_order_id`),
FOREIGN KEY (`supply_request_id`) REFERENCES `supply_requests` (`supply_request_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_orders` (
`purchase_order_id` int(11) NOT NULL AUTO_INCREMENT,
`product_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`price` int(11) NOT NULL,
`purchase_date` datetime NOT NULL,
`status` varchar(255) NOT NULL,
PRIMARY KEY (`purchase_order_id`),
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory_updates` (
`inventory_update_id` int(11) NOT NULL AUTO_INCREMENT,
`product_id` int(11) NOT NULL,
`quantity` int(11) NOT NULL,
`update_date` datetime NOT NULL,
`status` varchar(255) NOT NULL,
PRIMARY KEY (`inventory_update_id`),
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
