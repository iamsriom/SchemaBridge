-- Schema for database: rschema_0456

CREATE TABLE supermarkets (
  supermarket_id INT(11) NOT NULL AUTO_INCREMENT,
  supermarket_name VARCHAR(255) NOT NULL,
  supermarket_address VARCHAR(255) NOT NULL,
  manager_phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (supermarket_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE products (
  product_id INT(11) NOT NULL AUTO_INCREMENT,
  product_name VARCHAR(255) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  storage_location VARCHAR(255) NOT NULL,
  PRIMARY KEY (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE distribution_departments (
  department_id INT(11) NOT NULL AUTO_INCREMENT,
  department_name VARCHAR(255) NOT NULL,
  department_address VARCHAR(255) NOT NULL,
  manager_phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (department_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE purchases (
  purchase_id INT(11) NOT NULL AUTO_INCREMENT,
  product_id INT(11) NOT NULL,
  purchase_date DATETIME NOT NULL,
  supplier VARCHAR(255) NOT NULL,
  purchase_quantity INT(11) NOT NULL,
  PRIMARY KEY (purchase_id),
  FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE payment_orders (
  order_id INT(11) NOT NULL AUTO_INCREMENT,
  purchase_id INT(11) NOT NULL,
  payment_amount DECIMAL(10,2) NOT NULL,
  payment_date DATETIME NOT NULL,
  payment_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (order_id),
  FOREIGN KEY (purchase_id) REFERENCES purchases(purchase_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE supply_requests (
  request_id INT(11) NOT NULL AUTO_INCREMENT,
  supermarket_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  request_time DATETIME NOT NULL,
  requested_quantity INT(11) NOT NULL,
  PRIMARY KEY (request_id),
  FOREIGN KEY (supermarket_id) REFERENCES supermarkets(supermarket_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE distribution_notifications (
  distribution_notice_number INT(11) NOT NULL AUTO_INCREMENT,
  product_id INT(11) NOT NULL,
  distribution_quantity INT(11) NOT NULL,
  distribution_time DATETIME NOT NULL,
  distribution_department_id INT(11) NOT NULL,
  PRIMARY KEY (distribution_notice_number),
  FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (distribution_department_id) REFERENCES distribution_departments(department_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE stockout_notifications (
  stockout_id INT(11) NOT NULL AUTO_INCREMENT,
  product_id INT(11) NOT NULL,
  stockout_time DATETIME NOT NULL,
  PRIMARY KEY (stockout_id),
  FOREIGN KEY (product_id) REFERENCES products(product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
