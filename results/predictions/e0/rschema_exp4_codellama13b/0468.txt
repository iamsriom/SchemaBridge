-- Schema for database: rschema_0468

CREATE TABLE `credit_card_application` (
`application_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_id` int(11) DEFAULT NULL,
`card_type` varchar(255) NOT NULL,
`applicant_name` varchar(255) NOT NULL,
`applicant_id` varchar(255) NOT NULL,
`application_time` datetime NOT NULL,
`application_result` varchar(255) NOT NULL DEFAULT 'pending review',
PRIMARY KEY (`application_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
`customer_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`card_type` varchar(255) NOT NULL,
`card_number` varchar(255) NOT NULL,
`validity_period` datetime NOT NULL,
`credit_limit` int(11) NOT NULL,
`creation_time` datetime NOT NULL,
`update_time` datetime NOT NULL,
PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction` (
`transaction_id` int(11) NOT NULL AUTO_INCREMENT,
`customer_id` int(11) NOT NULL,
`transaction_time` datetime NOT NULL,
`transaction_amount` int(11) NOT NULL,
`transaction_type` varchar(255) NOT NULL,
`transaction_description` varchar(255) NOT NULL,
PRIMARY KEY (`transaction_id`),
FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
