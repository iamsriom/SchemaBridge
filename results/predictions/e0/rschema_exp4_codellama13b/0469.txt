-- Schema for database: rschema_0469

CREATE TABLE `owners` (
`owner_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`workplace` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
PRIMARY KEY (`owner_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `housing_units` (
`house_number` int(11) NOT NULL,
`floor_area` int(11) NOT NULL,
`owner_id` int(11) NOT NULL,
PRIMARY KEY (`house_number`),
FOREIGN KEY (`owner_id`) REFERENCES `owners` (`owner_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
`department_id` int(11) NOT NULL,
`department_name` varchar(255) NOT NULL,
`department_phone_number` varchar(255) NOT NULL,
PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
`employee_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`date_of_birth` date NOT NULL,
`gender` varchar(255) NOT NULL,
`address` varchar(255) NOT NULL,
`contact_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL,
`password` varchar(255) NOT NULL,
`department_id` int(11) NOT NULL,
PRIMARY KEY (`employee_id`),
FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fees` (
`fee_id` int(11) NOT NULL,
`fee_type` varchar(255) NOT NULL,
`quantity` int(11) NOT NULL,
`fee_amount` int(11) NOT NULL,
`house_number` int(11) NOT NULL,
PRIMARY KEY (`fee_id`),
FOREIGN KEY (`house_number`) REFERENCES `housing_units` (`house_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fee_receipts` (
`fee_receipt_id` int(11) NOT NULL,
`fee_id` int(11) NOT NULL,
`receipt_date` date NOT NULL,
`receipt_number` varchar(255) NOT NULL,
`receipt_amount` int(11) NOT NULL,
PRIMARY KEY (`fee_receipt_id`),
FOREIGN KEY (`fee_id`) REFERENCES `fees` (`fee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `accounts_receivable` (
`accounts_receivable_id` int(11) NOT NULL,
`fee_receipt_id` int(11) NOT NULL,
`accounts_receivable_amount` int(11) NOT NULL,
PRIMARY KEY (`accounts_receivable_id`),
FOREIGN KEY (`fee_receipt_id`) REFERENCES `fee_receipts` (`fee_receipt_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
