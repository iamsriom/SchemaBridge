-- Schema for database: rschema_0471

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `application_status` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_cards` (
  `card_number` int(11) NOT NULL,
  `card_type` varchar(255) NOT NULL,
  `validity_period` datetime NOT NULL,
  `credit_limit` int(11) NOT NULL,
  `activation_status` varchar(255) NOT NULL,
  PRIMARY KEY (`card_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `transaction_id` int(11) NOT NULL,
  `transaction_date` datetime NOT NULL,
  `transaction_description` varchar(255) NOT NULL,
  `transaction_amount` int(11) NOT NULL,
  `transaction_status` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_credit_cards` (
  `customer_id` int(11) NOT NULL,
  `card_number` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`, `card_number`),
  CONSTRAINT `customer_credit_cards_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_credit_cards_ibfk_2` FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_customers` (
  `transaction_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`transaction_id`, `customer_id`),
  CONSTRAINT `transaction_customers_ibfk_1` FOREIGN KEY (`transaction_id`) REFERENCES `transactions` (`transaction_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_customers_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_credit_cards` (
  `transaction_id` int(11) NOT NULL,
  `card_number` int(11) NOT NULL,
  PRIMARY KEY (`transaction_id`, `card_number`),
  CONSTRAINT `transaction_credit_cards_ibfk_1` FOREIGN KEY (`transaction_id`) REFERENCES `transactions` (`transaction_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_credit_cards_ibfk_2` FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
