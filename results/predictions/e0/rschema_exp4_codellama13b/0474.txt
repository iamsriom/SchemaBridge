-- Schema for database: rschema_0474

CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `class` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instructors` (
  `instructor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `course` varchar(255) NOT NULL,
  PRIMARY KEY (`instructor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `assignments` (
  `assignment_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  `instructor_id` int(11) NOT NULL,
  `submitted_at` datetime NOT NULL,
  `graded_at` datetime DEFAULT NULL,
  `score` int(11) DEFAULT NULL,
  `evaluation` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`assignment_id`),
  FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`instructor_id`) REFERENCES `instructors` (`instructor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grades` (
  `grade_id` int(11) NOT NULL,
  `assignment_id` int(11) NOT NULL,
  `score` int(11) NOT NULL,
  `evaluation` varchar(255) NOT NULL,
  PRIMARY KEY (`grade_id`),
  FOREIGN KEY (`assignment_id`) REFERENCES `assignments` (`assignment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_assignments` (
  `student_id` int(11) NOT NULL,
  `assignment_id` int(11) NOT NULL,
  `submitted_at` datetime NOT NULL,
  `graded_at` datetime DEFAULT NULL,
  `score` int(11) DEFAULT NULL,
  `evaluation` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`, `assignment_id`),
  FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`assignment_id`) REFERENCES `assignments` (`assignment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instructor_assignments` (
  `instructor_id` int(11) NOT NULL,
  `assignment_id` int(11) NOT NULL,
  `submitted_at` datetime NOT NULL,
  `graded_at` datetime DEFAULT NULL,
  `score` int(11) DEFAULT NULL,
  `evaluation` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`instructor_id`, `assignment_id`),
  FOREIGN KEY (`instructor_id`) REFERENCES `instructors` (`instructor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`assignment_id`) REFERENCES `assignments` (`assignment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
