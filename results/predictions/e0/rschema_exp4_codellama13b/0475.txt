-- Schema for database: rschema_0475

CREATE TABLE supermarkets (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  manager_id INT(11) NOT NULL,
  contact_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (manager_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE departments (
  id INT(11) NOT NULL AUTO_INCREMENT,
  supermarket_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  department_manager_id INT(11) NOT NULL,
  contact_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (supermarket_id) REFERENCES supermarkets (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (department_manager_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE employees (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  salary DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE products (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  model VARCHAR(255) NOT NULL,
  unit_price DECIMAL(10,2) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE salespeople (
  id INT(11) NOT NULL AUTO_INCREMENT,
  employee_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (employee_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE product_distribution (
  id INT(11) NOT NULL AUTO_INCREMENT,
  salesperson_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (salesperson_id) REFERENCES salespeople (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
