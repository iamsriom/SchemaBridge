-- Schema for database: rschema_0477

CREATE TABLE `transaction` (
`transaction_id` int(11) NOT NULL AUTO_INCREMENT,
`input_amount` decimal(10,2) NOT NULL,
`billing_method_id` int(11) NOT NULL,
`actual_payment_amount` decimal(10,2) NOT NULL,
`transaction_time` datetime NOT NULL,
PRIMARY KEY (`transaction_id`),
FOREIGN KEY (`billing_method_id`) REFERENCES `billing_method` (`billing_method_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `billing_method` (
`billing_method_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`calculation_function` varchar(255) NOT NULL,
PRIMARY KEY (`billing_method_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_detail` (
`transaction_id` int(11) NOT NULL,
`input_amount` decimal(10,2) NOT NULL,
`billing_method_id` int(11) NOT NULL,
`actual_payment_amount` decimal(10,2) NOT NULL,
`transaction_time` datetime NOT NULL,
PRIMARY KEY (`transaction_id`, `input_amount`, `billing_method_id`, `actual_payment_amount`, `transaction_time`),
FOREIGN KEY (`transaction_id`) REFERENCES `transaction` (`transaction_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`billing_method_id`) REFERENCES `billing_method` (`billing_method_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
