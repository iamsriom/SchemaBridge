-- Schema for database: rschema_0478

CREATE TABLE `users` (
`user_id` int(11) NOT NULL,
`phone_number` varchar(255) NOT NULL,
`name` varchar(255) NOT NULL,
`credit` int(11) NOT NULL,
`status` varchar(255) NOT NULL,
PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bicycles` (
`bicycle_id` int(11) NOT NULL,
`location` varchar(255) NOT NULL,
`status` varchar(255) NOT NULL,
`unlock_code` varchar(255) NOT NULL,
PRIMARY KEY (`bicycle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trips` (
`trip_id` int(11) NOT NULL,
`user_id` int(11) NOT NULL,
`bicycle_id` int(11) NOT NULL,
`start_time` datetime NOT NULL,
`end_time` datetime NOT NULL,
`cost` int(11) NOT NULL,
`payment_status` varchar(255) NOT NULL,
PRIMARY KEY (`trip_id`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`bicycle_id`) REFERENCES `bicycles` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `faults` (
`fault_id` int(11) NOT NULL,
`user_id` int(11) NOT NULL,
`bicycle_id` int(11) NOT NULL,
`fault_description` varchar(255) NOT NULL,
`reporting_time` datetime NOT NULL,
PRIMARY KEY (`fault_id`),
FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`bicycle_id`) REFERENCES `bicycles` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `billing_rules` (
`billing_rule_id` int(11) NOT NULL,
`billing_method` varchar(255) NOT NULL,
`unit_price` int(11) NOT NULL,
`minimum_fee` int(11) NOT NULL,
PRIMARY KEY (`billing_rule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
