-- Schema for database: rschema_0482

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_number` varchar(255) NOT NULL,
  `client_name` varchar(255) NOT NULL,
  `client_type` varchar(255) NOT NULL,
  `discount_rate` decimal(10,2) NOT NULL,
  `contact_person` varchar(255) NOT NULL,
  `contact_phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate_number` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `color` varchar(255) NOT NULL,
  `vehicle_category` varchar(255) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`vehicle_id`),
  FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_number` varchar(255) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_number` varchar(255) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `supervisor` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_tasks` (
  `maintenance_task_id` int(11) NOT NULL AUTO_INCREMENT,
  `maintenance_order_number` varchar(255) NOT NULL,
  `license_plate_number` varchar(255) NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `man_hours` decimal(10,2) NOT NULL,
  `employee_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_task_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
