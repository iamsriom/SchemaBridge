-- Schema for database: rschema_0485

CREATE TABLE wetland_protection_areas (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  person_in_charge VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE flora_and_fauna (
  id INT(11) NOT NULL AUTO_INCREMENT,
  recording_time DATETIME NOT NULL,
  recording_personnel VARCHAR(255) NOT NULL,
  species VARCHAR(255) NOT NULL,
  quantity INT(11) NOT NULL,
  growth_status VARCHAR(255) NOT NULL,
  wetland_protection_area_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (wetland_protection_area_id) REFERENCES wetland_protection_areas(id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE environmental_changes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  time DATETIME NOT NULL,
  location VARCHAR(255) NOT NULL,
  temperature FLOAT NOT NULL,
  humidity FLOAT NOT NULL,
  rainfall FLOAT NOT NULL,
  wetland_protection_area_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (wetland_protection_area_id) REFERENCES wetland_protection_areas(id) ON DELETE CASCADE ON UPDATE CASCADE
);
