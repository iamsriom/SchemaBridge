-- Schema for database: rschema_0492

CREATE TABLE `departments` (
`department_id` int(11) NOT NULL AUTO_INCREMENT,
`department_name` varchar(255) NOT NULL,
PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `positions` (
`position_id` int(11) NOT NULL AUTO_INCREMENT,
`position_name` varchar(255) NOT NULL,
`salary` int(11) NOT NULL,
`department_id` int(11) NOT NULL,
PRIMARY KEY (`position_id`),
FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
`employee_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`position_id` int(11) NOT NULL,
`superior_id` int(11) NOT NULL,
PRIMARY KEY (`employee_id`),
FOREIGN KEY (`position_id`) REFERENCES `positions` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`superior_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `relationships` (
`relationship_id` int(11) NOT NULL AUTO_INCREMENT,
`superior_id` int(11) NOT NULL,
`subordinate_id` int(11) NOT NULL,
PRIMARY KEY (`relationship_id`),
FOREIGN KEY (`superior_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`subordinate_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reports` (
`report_id` int(11) NOT NULL AUTO_INCREMENT,
`report_type` varchar(255) NOT NULL,
`report_date` datetime NOT NULL,
`report_data` text NOT NULL,
PRIMARY KEY (`report_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
