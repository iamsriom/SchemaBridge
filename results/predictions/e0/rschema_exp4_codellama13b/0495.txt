-- Schema for database: rschema_0495

CREATE TABLE doctors (
  doctor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(10) NOT NULL,
  title VARCHAR(255) NOT NULL,
  years_of_service INT(11) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  department_id INT(11) NOT NULL,
  PRIMARY KEY (doctor_id),
  FOREIGN KEY (department_id) REFERENCES departments (department_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE departments (
  department_id INT(11) NOT NULL AUTO_INCREMENT,
  department_name VARCHAR(255) NOT NULL,
  person_in_charge VARCHAR(255) NOT NULL,
  total_personnel INT(11) NOT NULL,
  person_in_charge_id INT(11) NOT NULL,
  location VARCHAR(255) NOT NULL,
  department_phone_number VARCHAR(255) NOT NULL,
  PRIMARY KEY (department_id),
  FOREIGN KEY (person_in_charge_id) REFERENCES doctors (doctor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE wards (
  ward_id INT(11) NOT NULL AUTO_INCREMENT,
  number_of_beds INT(11) NOT NULL,
  ward_level VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  PRIMARY KEY (ward_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE patients (
  patient_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(10) NOT NULL,
  age INT(11) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  consultation_date DATETIME NOT NULL,
  diagnosis_description VARCHAR(255) NOT NULL,
  ward_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id),
  FOREIGN KEY (ward_id) REFERENCES wards (ward_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE medicines (
  medicine_id INT(11) NOT NULL AUTO_INCREMENT,
  medicine_name VARCHAR(255) NOT NULL,
  inventory_quantity INT(11) NOT NULL,
  sales_price DECIMAL(10,2) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (medicine_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE consultations (
  consultation_id INT(11) NOT NULL AUTO_INCREMENT,
  patient_id INT(11) NOT NULL,
  doctor_id INT(11) NOT NULL,
  consultation_time DATETIME NOT NULL,
  diagnosis_description VARCHAR(255) NOT NULL,
  treatment_results VARCHAR(255) NOT NULL,
  remarks VARCHAR(255) NOT NULL,
  PRIMARY KEY (consultation_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (doctor_id) REFERENCES doctors (doctor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE medicine_retrievals (
  medicine_retrieval_id INT(11) NOT NULL AUTO_INCREMENT,
  patient_id INT(11) NOT NULL,
  medicine_id INT(11) NOT NULL,
  retrieval_time DATETIME NOT NULL,
  retrieval_quantity INT(11) NOT NULL,
  PRIMARY KEY (medicine_retrieval_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (medicine_id) REFERENCES medicines (medicine_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
