-- Schema for database: rschema_0496

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `real_name` varchar(255) NOT NULL,
  `id_card_front` varchar(255) NOT NULL,
  `id_card_back` varchar(255) NOT NULL,
  `face_recognition` varchar(255) NOT NULL,
  `avatar` varchar(255) NOT NULL,
  `nickname` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `personal_introduction` varchar(255) NOT NULL,
  `account_security_settings` varchar(255) NOT NULL,
  `phone_binding` varchar(255) NOT NULL,
  `email_binding` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `condition_description` varchar(255) NOT NULL,
  `original_price` int(11) NOT NULL,
  `selling_price` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  `transaction_location` varchar(255) NOT NULL,
  `price_negotiation_range` varchar(255) NOT NULL,
  `validity_period` varchar(255) NOT NULL,
  `product_status` varchar(255) NOT NULL,
  `product_images` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `buyer_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `transaction_status` varchar(255) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_amount` int(11) NOT NULL,
  `transaction_location` varchar(255) NOT NULL,
  `transaction_method` varchar(255) NOT NULL,
  `transaction_evaluation` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`buyer_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`product_id`) REFERENCES `products` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
