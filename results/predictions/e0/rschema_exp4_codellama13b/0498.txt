-- Schema for database: rschema_0498

CREATE TABLE `students` (
`student_id` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`gender` varchar(255) NOT NULL,
`age` int(11) NOT NULL,
`dormitory_number` int(11) NOT NULL,
`building_number` int(11) NOT NULL,
`number_of_people` int(11) NOT NULL,
`administrator` varchar(255) NOT NULL,
PRIMARY KEY (`student_id`),
FOREIGN KEY (`dormitory_number`) REFERENCES `dormitories` (`dormitory_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classes` (
`class_number` int(11) NOT NULL,
`classroom_number` int(11) NOT NULL,
`class_name` varchar(255) NOT NULL,
PRIMARY KEY (`class_number`),
FOREIGN KEY (`classroom_number`) REFERENCES `classrooms` (`classroom_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sports_teams` (
`team_name` varchar(255) NOT NULL,
`activity_time` datetime NOT NULL,
`coach` varchar(255) NOT NULL,
`activity_venue` varchar(255) NOT NULL,
PRIMARY KEY (`team_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `schools` (
`school_name` varchar(255) NOT NULL,
`school_id` int(11) NOT NULL,
PRIMARY KEY (`school_id`),
FOREIGN KEY (`school_name`) REFERENCES `schools` (`school_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitories` (
`dormitory_number` int(11) NOT NULL,
`building_number` int(11) NOT NULL,
`number_of_people` int(11) NOT NULL,
`administrator` varchar(255) NOT NULL,
PRIMARY KEY (`dormitory_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classrooms` (
`classroom_number` int(11) NOT NULL,
`classroom_name` varchar(255) NOT NULL,
PRIMARY KEY (`classroom_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sports_team_members` (
`team_name` varchar(255) NOT NULL,
`student_id` int(11) NOT NULL,
PRIMARY KEY (`team_name`, `student_id`),
FOREIGN KEY (`team_name`) REFERENCES `sports_teams` (`team_name`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `school_sports_teams` (
`school_name` varchar(255) NOT NULL,
`team_name` varchar(255) NOT NULL,
PRIMARY KEY (`school_name`, `team_name`),
FOREIGN KEY (`school_name`) REFERENCES `schools` (`school_name`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`team_name`) REFERENCES `sports_teams` (`team_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
