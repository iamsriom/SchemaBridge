-- Schema for database: rschema_0501

CREATE TABLE aircraft (
  aircraft_id INT(11) NOT NULL AUTO_INCREMENT,
  aircraft_model VARCHAR(255) NOT NULL,
  manufacture_date DATE NOT NULL,
  maximum_passenger_capacity INT(11) NOT NULL,
  aircraft_status VARCHAR(255) NOT NULL,
  affiliated_airline VARCHAR(255) NOT NULL,
  PRIMARY KEY (aircraft_id),
  FOREIGN KEY (affiliated_airline) REFERENCES airlines (airline_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE crew_members (
  crew_member_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  position VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (crew_member_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE passengers (
  passenger_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  identification_number VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (passenger_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE flights (
  flight_id INT(11) NOT NULL AUTO_INCREMENT,
  aircraft_id INT(11) NOT NULL,
  responsible_captain_id INT(11) NOT NULL,
  departure_airport VARCHAR(255) NOT NULL,
  landing_airport VARCHAR(255) NOT NULL,
  departure_time TIMESTAMP NOT NULL,
  landing_time TIMESTAMP NOT NULL,
  PRIMARY KEY (flight_id),
  FOREIGN KEY (aircraft_id) REFERENCES aircraft (aircraft_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (responsible_captain_id) REFERENCES crew_members (crew_member_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE maintenance_history (
  maintenance_personnel_id INT(11) NOT NULL AUTO_INCREMENT,
  aircraft_id INT(11) NOT NULL,
  maintenance_date DATE NOT NULL,
  maintenance_description VARCHAR(255) NOT NULL,
  maintenance_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (maintenance_personnel_id),
  FOREIGN KEY (aircraft_id) REFERENCES aircraft (aircraft_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE ticket_purchases (
  passenger_id INT(11) NOT NULL,
  flight_id INT(11) NOT NULL,
  booking_time TIMESTAMP NOT NULL,
  seat_number INT(11) NOT NULL,
  ticket_price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (passenger_id, flight_id),
  FOREIGN KEY (passenger_id) REFERENCES passengers (passenger_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (flight_id) REFERENCES flights (flight_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
