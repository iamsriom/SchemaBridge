-- Schema for database: rschema_0502

CREATE TABLE venues (
  venue_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  area INT(11) NOT NULL,
  address VARCHAR(255) NOT NULL,
  status VARCHAR(255) NOT NULL,
  creation_time DATETIME NOT NULL,
  last_maintenance_time DATETIME NOT NULL,
  maximum_capacity INT(11) NOT NULL,
  PRIMARY KEY (venue_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE users (
  user_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  creation_time DATETIME NOT NULL,
  PRIMARY KEY (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE equipment (
  equipment_id INT(11) NOT NULL AUTO_INCREMENT,
  equipment_name VARCHAR(255) NOT NULL,
  start_usage_time DATETIME NOT NULL,
  last_maintenance_time DATETIME NOT NULL,
  venue_id INT(11) NOT NULL,
  PRIMARY KEY (equipment_id),
  FOREIGN KEY (venue_id) REFERENCES venues (venue_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reservations (
  user_id INT(11) NOT NULL,
  venue_id INT(11) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  reservation_status VARCHAR(255) NOT NULL,
  reservation_success_time DATETIME NOT NULL,
  reservation_cancellation_time DATETIME NOT NULL,
  PRIMARY KEY (user_id, venue_id, start_time),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (venue_id) REFERENCES venues (venue_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE evaluations (
  user_id INT(11) NOT NULL,
  venue_id INT(11) NOT NULL,
  evaluation_time DATETIME NOT NULL,
  evaluation_content VARCHAR(255) NOT NULL,
  rating INT(11) NOT NULL,
  PRIMARY KEY (user_id, venue_id, evaluation_time),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (venue_id) REFERENCES venues (venue_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
