-- Schema for database: rschema_0504

CREATE TABLE `cultural_relic` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `year` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  `excavation_site` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cultural_relic_exhibition` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `cultural_relic_id` int(11) NOT NULL,
  `exhibition_start_time` datetime NOT NULL,
  `exhibition_end_time` datetime NOT NULL,
  `exhibition_location` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`cultural_relic_id`) REFERENCES `cultural_relic` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cultural_relic_repair` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `staff_id` int(11) NOT NULL,
  `cultural_relic_id` int(11) NOT NULL,
  `repair_start_time` datetime NOT NULL,
  `repair_end_time` datetime NOT NULL,
  `repair_method` varchar(255) NOT NULL,
  `notes` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`staff_id`) REFERENCES `staff` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`cultural_relic_id`) REFERENCES `cultural_relic` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cultural_relic_monitoring_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `cultural_relic_id` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `person_in_charge_id` int(11) NOT NULL,
  `temperature` int(11) NOT NULL,
  `humidity` int(11) NOT NULL,
  `light` int(11) NOT NULL,
  `air_quality` int(11) NOT NULL,
  `notes` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`cultural_relic_id`) REFERENCES `cultural_relic` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`person_in_charge_id`) REFERENCES `staff` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
