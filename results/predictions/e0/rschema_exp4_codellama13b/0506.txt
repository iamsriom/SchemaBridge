-- Schema for database: rschema_0506

CREATE TABLE construction_management_unit (
  id INT(11) NOT NULL AUTO_INCREMENT,
  unit_name VARCHAR(255) NOT NULL,
  contact_number VARCHAR(255) NOT NULL,
  responsible_person VARCHAR(255) NOT NULL,
  associated_project VARCHAR(255) NOT NULL,
  annual_plan VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE business_project_department (
  id INT(11) NOT NULL AUTO_INCREMENT,
  department_name VARCHAR(255) NOT NULL,
  responsible_person VARCHAR(255) NOT NULL,
  project_progress VARCHAR(255) NOT NULL,
  issue_reporting VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE supervision_project_department (
  id INT(11) NOT NULL AUTO_INCREMENT,
  supervision_department_name VARCHAR(255) NOT NULL,
  responsible_person VARCHAR(255) NOT NULL,
  resource_allocation VARCHAR(255) NOT NULL,
  personnel_management VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE electrical_grid_infrastructure_project (
  id INT(11) NOT NULL AUTO_INCREMENT,
  project_id VARCHAR(255) NOT NULL,
  project_name VARCHAR(255) NOT NULL,
  construction_progress VARCHAR(255) NOT NULL,
  resource_allocation VARCHAR(255) NOT NULL,
  quality_control VARCHAR(255) NOT NULL,
  safety_management VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE resource (
  id INT(11) NOT NULL AUTO_INCREMENT,
  resource_type VARCHAR(255) NOT NULL,
  resource_quantity INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE task (
  id INT(11) NOT NULL AUTO_INCREMENT,
  task_name VARCHAR(255) NOT NULL,
  responsible_person VARCHAR(255) NOT NULL,
  task_progress VARCHAR(255) NOT NULL,
  task_priority VARCHAR(255) NOT NULL,
  completion_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE construction_management_unit_electrical_grid_infrastructure_project (
  id INT(11) NOT NULL AUTO_INCREMENT,
  construction_management_unit_id INT(11) NOT NULL,
  electrical_grid_infrastructure_project_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (construction_management_unit_id) REFERENCES construction_management_unit(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (electrical_grid_infrastructure_project_id) REFERENCES electrical_grid_infrastructure_project(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE business_project_department_electrical_grid_infrastructure_project (
  id INT(11) NOT NULL AUTO_INCREMENT,
  business_project_department_id INT(11) NOT NULL,
  electrical_grid_infrastructure_project_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (business_project_department_id) REFERENCES business_project_department(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (electrical_grid_infrastructure_project_id) REFERENCES electrical_grid_infrastructure_project(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE supervision_project_department_electrical_grid_infrastructure_project (
  id INT(11) NOT NULL AUTO_INCREMENT,
  supervision_project_department_id INT(11) NOT NULL,
  electrical_grid_infrastructure_project_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (supervision_project_department_id) REFERENCES supervision_project_department(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (electrical_grid_infrastructure_project_id) REFERENCES electrical_grid_infrastructure_project(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE resource_electrical_grid_infrastructure_project (
  id INT(11) NOT NULL AUTO_INCREMENT,
  resource_id INT(11) NOT NULL,
  electrical_grid_infrastructure_project_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (resource_id) REFERENCES resource(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (electrical_grid_infrastructure_project_id) REFERENCES electrical_grid_infrastructure_project(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE task_electrical_grid_infrastructure_project (
  id INT(11) NOT NULL AUTO_INCREMENT,
  task_id INT(11) NOT NULL,
  electrical_grid_infrastructure_project_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (task_id) REFERENCES task(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (electrical_grid_infrastructure_project_id) REFERENCES electrical_grid_infrastructure_project(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
