-- Schema for database: rschema_0510

CREATE TABLE parking_spaces (
  id INT(11) NOT NULL AUTO_INCREMENT,
  is_occupied BOOLEAN NOT NULL DEFAULT FALSE,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE car_owners (
  id INT(11) NOT NULL AUTO_INCREMENT,
  license_plate_number VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  contact_information VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (id) REFERENCES parking_spaces (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE parking_space_occupations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  parking_space_id INT(11) NOT NULL,
  car_owner_id INT(11) NOT NULL,
  start_time DATETIME NOT NULL,
  end_time DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (parking_space_id) REFERENCES parking_spaces (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (car_owner_id) REFERENCES car_owners (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
