-- Schema for database: rschema_0001

-- Table: Airplane
CREATE TABLE `Airplane` (
  `AirplaneNumber` int(11) NOT NULL,
  `Model` varchar(255) NOT NULL,
  `PassengerCapacity` int(11) NOT NULL,
  PRIMARY KEY (`AirplaneNumber`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Flight
CREATE TABLE `Flight` (
  `FlightNumber` int(11) NOT NULL,
  `DepartureTime` datetime NOT NULL,
  `ArrivalTime` datetime NOT NULL,
  `Origin` varchar(255) NOT NULL,
  `Destination` varchar(255) NOT NULL,
  `AirplaneNumber` int(11) NOT NULL,
  `CaptainID` int(11) NOT NULL,
  `FirstOfficerID` int(11) NOT NULL,
  `StartTime` datetime NOT NULL,
  `EndTime` datetime NOT NULL,
  PRIMARY KEY (`FlightNumber`),
  FOREIGN KEY (`AirplaneNumber`) REFERENCES `Airplane` (`AirplaneNumber`),
  FOREIGN KEY (`CaptainID`) REFERENCES `Captain` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`FirstOfficerID`) REFERENCES `FirstOfficer` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Captain
CREATE TABLE `Captain` (
  `ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: FirstOfficer
CREATE TABLE `FirstOfficer` (
  `ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
