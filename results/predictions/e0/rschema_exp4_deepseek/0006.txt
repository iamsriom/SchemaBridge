-- Schema for database: rschema_0006

-- Table: developers
CREATE TABLE `developers` (
    `developer_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`developer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: projects
CREATE TABLE `projects` (
    `project_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `start_date` DATETIME NOT NULL,
    `end_date` DATETIME NOT NULL,
    `budget` DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: project_updates
CREATE TABLE `project_updates` (
    `update_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_id` INT(11) NOT NULL,
    `update_time` DATETIME NOT NULL,
    `current_status` VARCHAR(255) NOT NULL,
    `performed_by` INT(11) NOT NULL,
    PRIMARY KEY (`update_id`),
    FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`performed_by`) REFERENCES `developers` (`developer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: developer_project
CREATE TABLE `developer_project` (
    `developer_id` INT(11) NOT NULL,
    `project_id` INT(11) NOT NULL,
    PRIMARY KEY (`developer_id`, `project_id`),
    FOREIGN KEY (`developer_id`) REFERENCES `developers` (`developer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
