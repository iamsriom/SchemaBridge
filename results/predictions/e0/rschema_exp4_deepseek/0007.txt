-- Schema for database: `rschema_0007`

-- Table: `advertising_clients`

CREATE TABLE `advertising_clients` (
    `client_id` INT NOT NULL AUTO_INCREMENT,
    `client_name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `advertisements`

CREATE TABLE `advertisements` (
    `ad_id` INT NOT NULL AUTO_INCREMENT,
    `client_id` INT NOT NULL,
    `ad_name` VARCHAR(255) NOT NULL,
    `broadcast_time` DATETIME NOT NULL,
    `duration` INT NOT NULL,
    `price` DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (`ad_id`),
    FOREIGN KEY (`client_id`) REFERENCES `advertising_clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `broadcasts`

CREATE TABLE `broadcasts` (
    `broadcast_id` INT NOT NULL AUTO_INCREMENT,
    `ad_id` INT NOT NULL,
    `broadcast_time` DATETIME NOT NULL,
    `broadcast_channel` VARCHAR(255) NOT NULL,
    `operating_personnel` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`broadcast_id`),
    FOREIGN KEY (`ad_id`) REFERENCES `advertisements` (`ad_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
