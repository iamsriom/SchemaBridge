-- Schema for database: rschema_0008

-- Table: equipment
CREATE TABLE `equipment` (
    `equipment_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `model` VARCHAR(255) NOT NULL,
    `usage_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: maintenance_personnel
CREATE TABLE `maintenance_personnel` (
    `maintenance_personnel_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `telephone` VARCHAR(255) NOT NULL,
    `email` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: equipment_maintenance
CREATE TABLE `equipment_maintenance` (
    `equipment_id` INT(11) NOT NULL,
    `maintenance_time` DATETIME NOT NULL,
    `maintenance_results` VARCHAR(255) NOT NULL,
    `maintenance_personnel_id` INT(11) NOT NULL,
    PRIMARY KEY (`equipment_id`, `maintenance_time`),
    FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `maintenance_personnel` (`maintenance_personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
