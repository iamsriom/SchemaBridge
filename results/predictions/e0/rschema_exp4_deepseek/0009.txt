-- Schema for database: rschema_0009

-- Table: candidates
CREATE TABLE `candidates` (
  `candidateID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `dateofbirth` datetime NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`candidateID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: voters
CREATE TABLE `voters` (
  `voterID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `dateofbirth` datetime NOT NULL,
  `contactinformation` varchar(255) NOT NULL,
  PRIMARY KEY (`voterID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: voting_records
CREATE TABLE `voting_records` (
  `voterID` int(11) NOT NULL,
  `candidateID` int(11) NOT NULL,
  `votingtime` datetime NOT NULL,
  `votinglocation` varchar(255) NOT NULL,
  PRIMARY KEY (`voterID`,`candidateID`),
  FOREIGN KEY (`voterID`) REFERENCES `voters` (`voterID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`candidateID`) REFERENCES `candidates` (`candidateID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
