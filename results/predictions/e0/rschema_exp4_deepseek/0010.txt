-- Schema for database: `rschema_001`

-- Table: `members`

CREATE TABLE `members` (
    `member_number` INT NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `date_of_birth` DATE NOT NULL,
    `contact_information` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`member_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `meetings`

CREATE TABLE `meetings` (
    `meeting_id` INT NOT NULL AUTO_INCREMENT,
    `type` VARCHAR(255) NOT NULL,
    `time` DATETIME NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    `content` TEXT NOT NULL,
    `theme` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`meeting_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `attendance`

CREATE TABLE `attendance` (
    `member_number` INT NOT NULL,
    `meeting_id` INT NOT NULL,
    `attendance_date` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `notes` TEXT,
    PRIMARY KEY (`member_number`, `meeting_id`),
    FOREIGN KEY (`member_number`) REFERENCES `members` (`member_number`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`meeting_id`) REFERENCES `meetings` (`meeting_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
