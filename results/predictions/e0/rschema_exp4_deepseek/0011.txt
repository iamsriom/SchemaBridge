-- Schema for database: `rschema_0011`

-- Table: `equipment`
CREATE TABLE `equipment` (
  `equipmentID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `purchasedate` datetime NOT NULL,
  `usagestatus` enum('normal','maintenance','disabled') NOT NULL,
  PRIMARY KEY (`equipmentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `farm`
CREATE TABLE `farm` (
  `farmID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contactinformation` varchar(255) NOT NULL,
  PRIMARY KEY (`farmID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance_record`
CREATE TABLE `maintenance_record` (
  `equipmentID` int(11) NOT NULL,
  `maintenancedate` datetime NOT NULL,
  `maintenancetype` enum('routineinspection','faultrepair') NOT NULL,
  `operatorID` int(11) NOT NULL,
  `maintenanceresult` enum('completed','uncompleted') NOT NULL,
  PRIMARY KEY (`equipmentID`, `maintenancedate`),
  FOREIGN KEY (`equipmentID`) REFERENCES `equipment` (`equipmentID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`operatorID`) REFERENCES `operator` (`operatorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `operator`
CREATE TABLE `operator` (
  `operatorID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phonenumber` varchar(255) NOT NULL,
  PRIMARY KEY (`operatorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
