-- Schema for database: `rschema_0012`

-- Table: `warehouses`

CREATE TABLE `warehouses` (
    `WarehouseID` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `address` varchar(255) NOT NULL,
    `capacity` int(11) NOT NULL,
    `current_inventory` int(11) NOT NULL,
    PRIMARY KEY (`WarehouseID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `products`

CREATE TABLE `products` (
    `ProductID` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `specification` varchar(255) NOT NULL,
    `date_of_storage` datetime NOT NULL,
    `date_of_dispatch` datetime NOT NULL,
    `quantity_stored` int(11) NOT NULL,
    `quantity_dispatched` int(11) NOT NULL,
    `current_inventory` int(11) NOT NULL,
    PRIMARY KEY (`ProductID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `storage_records`

CREATE TABLE `storage_records` (
    `WarehouseID` int(11) NOT NULL,
    `ProductID` int(11) NOT NULL,
    `storage_date` datetime NOT NULL,
    `quantity_stored` int(11) NOT NULL,
    `operator` varchar(255) NOT NULL,
    PRIMARY KEY (`WarehouseID`, `ProductID`),
    FOREIGN KEY (`WarehouseID`) REFERENCES `warehouses` (`WarehouseID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ProductID`) REFERENCES `products` (`ProductID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `dispatch_records`

CREATE TABLE `dispatch_records` (
    `WarehouseID` int(11) NOT NULL,
    `ProductID` int(11) NOT NULL,
    `dispatch_date` datetime NOT NULL,
    `quantity_dispatched` int(11) NOT NULL,
    `operator` varchar(255) NOT NULL,
    PRIMARY KEY (`WarehouseID`, `ProductID`),
    FOREIGN KEY (`WarehouseID`) REFERENCES `warehouses` (`WarehouseID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ProductID`) REFERENCES `products` (`ProductID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
