-- Schema for database: rschema_0013

-- Table: WeatherStation
CREATE TABLE `WeatherStation` (
  `WeatherStationID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`WeatherStationID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: WeatherData
CREATE TABLE `WeatherData` (
  `WeatherStationID` int(11) NOT NULL,
  `recording_date` datetime NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `precipitation` float NOT NULL,
  `wind_speed` float NOT NULL,
  `wind_direction` char(6) NOT NULL,
  PRIMARY KEY (`WeatherStationID`, `recording_date`),
  FOREIGN KEY (`WeatherStationID`) REFERENCES `WeatherStation` (`WeatherStationID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Farm
CREATE TABLE `Farm` (
  `FarmID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `subscription_date` datetime NOT NULL,
  `unsubscription_date` datetime,
  PRIMARY KEY (`FarmID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Farm_WeatherStation
CREATE TABLE `Farm_WeatherStation` (
  `FarmID` int(11) NOT NULL,
  `WeatherStationID` int(11) NOT NULL,
  `subscription_date` datetime NOT NULL,
  `unsubscription_date` datetime,
  PRIMARY KEY (`FarmID`, `WeatherStationID`),
  FOREIGN KEY (`FarmID`) REFERENCES `Farm` (`FarmID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`WeatherStationID`) REFERENCES `WeatherStation` (`WeatherStationID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
