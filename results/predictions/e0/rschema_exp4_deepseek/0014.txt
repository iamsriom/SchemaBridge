-- Schema for database: rschema_0014

-- Table: Plot
CREATE TABLE `Plot` (
  `PlotID` int(11) NOT NULL,
  `area` int(11) NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `current_irrigation_status` varchar(255) NOT NULL,
  PRIMARY KEY (`PlotID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Irrigation_System
CREATE TABLE `Irrigation_System` (
  `SystemID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `installation_date` date NOT NULL,
  PRIMARY KEY (`SystemID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Irrigation_Record
CREATE TABLE `Irrigation_Record` (
  `SystemID` int(11) NOT NULL,
  `PlotID` int(11) NOT NULL,
  `irrigation_date` date NOT NULL,
  `irrigation_duration` int(11) NOT NULL,
  `irrigation_amount` int(11) NOT NULL,
  PRIMARY KEY (`SystemID`,`PlotID`),
  FOREIGN KEY (`SystemID`) REFERENCES `Irrigation_System` (`SystemID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`PlotID`) REFERENCES `Plot` (`PlotID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
