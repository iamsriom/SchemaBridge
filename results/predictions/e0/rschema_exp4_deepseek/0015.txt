-- Schema for database: `rschema_0015`

-- For table: `plot`
CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL,
  `area` int(11) NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `geographical_location` varchar(255) NOT NULL,
  `current_pest_and_disease_status` varchar(255) NOT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `pest_and_disease`
CREATE TABLE `pest_and_disease` (
  `pest_and_disease_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `degree_of_damage` varchar(255) NOT NULL,
  `prevention_and_control_methods` varchar(255) NOT NULL,
  `season_of_occurrence` varchar(255) NOT NULL,
  PRIMARY KEY (`pest_and_disease_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `monitoring_record`
CREATE TABLE `monitoring_record` (
  `plot_id` int(11) NOT NULL,
  `pest_and_disease_id` int(11) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `monitoring_results` varchar(255) NOT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`plot_id`,`pest_and_disease_id`),
  FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`pest_and_disease_id`) REFERENCES `pest_and_disease` (`pest_and_disease_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`operator_id`) REFERENCES `operator` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `operator`
CREATE TABLE `operator` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
