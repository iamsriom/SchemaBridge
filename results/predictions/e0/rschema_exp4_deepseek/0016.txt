-- Schema for database: rschema_0016
-- Farm table: CREATE TABLE `farm` (
  `farm_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`farm_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
-- Plot table: CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL,
  `farm_id` int(11) NOT NULL,
  `area` float NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `current_status` varchar(255) NOT NULL,
  PRIMARY KEY (`plot_id`),
  FOREIGN KEY (`farm_id`) REFERENCES `farm` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
-- PlantingPlan table: CREATE TABLE `planting_plan` (
  `farm_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `crop_type` varchar(255) NOT NULL,
  `planting_time` datetime NOT NULL,
  `expected_harvest_time` datetime NOT NULL,
  `planting_quantity` int(11) NOT NULL,
  `operator` varchar(255) NOT NULL,
  FOREIGN KEY (`farm_id`) REFERENCES `farm` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
