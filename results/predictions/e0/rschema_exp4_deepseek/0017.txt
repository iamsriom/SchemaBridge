-- Schema for database: rschema_0017

-- Table: workshops
CREATE TABLE `workshops` (
  `workshop_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`workshop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: workers
CREATE TABLE `workers` (
  `worker_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `workshop_id` int(11) NOT NULL,
  PRIMARY KEY (`worker_id`),
  FOREIGN KEY (`workshop_id`) REFERENCES `workshops` (`workshop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: daily_work
CREATE TABLE `daily_work` (
  `work_id` int(11) NOT NULL AUTO_INCREMENT,
  `worker_id` int(11) NOT NULL,
  `raw_materials_used` varchar(255) NOT NULL,
  `processing_time` int(11) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `production_quantity` int(11) NOT NULL,
  `product_quality_rating` int(11) NOT NULL,
  PRIMARY KEY (`work_id`),
  FOREIGN KEY (`worker_id`) REFERENCES `workers` (`worker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: raw_materials
CREATE TABLE `raw_materials` (
  `material_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `workshop_id` int(11) NOT NULL,
  PRIMARY KEY (`material_id`),
  FOREIGN KEY (`workshop_id`) REFERENCES `workshops` (`workshop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: production_metrics
CREATE TABLE `production_metrics` (
  `metric_id` int(11) NOT NULL AUTO_INCREMENT,
  `workshop_id` int(11) NOT NULL,
  `monthly_processing_volume` int(11) NOT NULL,
  `product_qualification_rate` int(11) NOT NULL,
  `other_assessment_indicators` varchar(255) NOT NULL,
  PRIMARY KEY (`metric_id`),
  FOREIGN KEY (`workshop_id`) REFERENCES `workshops` (`workshop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
