-- Schema for database: rschema_0018

-- Table: equipment
CREATE TABLE `equipment` (
    `equipment_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `model` VARCHAR(255) NOT NULL,
    `usage_frequency` INT(11) NOT NULL,
    PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: usage
CREATE TABLE `usage` (
    `usage_id` INT(11) NOT NULL AUTO_INCREMENT,
    `equipment_id` INT(11) NOT NULL,
    `usage_time` DATETIME NOT NULL,
    `user` VARCHAR(255) NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`usage_id`),
    FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: maintenance
CREATE TABLE `maintenance` (
    `maintenance_id` INT(11) NOT NULL AUTO_INCREMENT,
    `equipment_id` INT(11) NOT NULL,
    `maintenance_time` DATETIME NOT NULL,
    `maintenance_content` TEXT NOT NULL,
    `maintenance_personnel` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`maintenance_id`),
    FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
