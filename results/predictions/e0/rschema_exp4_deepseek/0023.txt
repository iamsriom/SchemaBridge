-- Schema for database: `rschema_0023NaturalLanguage`

-- For table: `donors`

CREATE TABLE `donors` (
  `donor_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `contact_information` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `recipients`

CREATE TABLE `recipients` (
  `recipient_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `contact_information` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `donation_records`

CREATE TABLE `donation_records` (
  `donor_number` int(11) NOT NULL,
  `recipient_number` int(11) NOT NULL,
  `donation_time` datetime NOT NULL,
  `item_number` int(11) NOT NULL,
  PRIMARY KEY (`donor_number`,`recipient_number`,`donation_time`),
  FOREIGN KEY (`donor_number`) REFERENCES `donors` (`donor_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`recipient_number`) REFERENCES `recipients` (`recipient_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `donated_items`

CREATE TABLE `donated_items` (
  `item_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `description` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
