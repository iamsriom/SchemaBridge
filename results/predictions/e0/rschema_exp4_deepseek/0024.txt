-- Schema for database: `agricultural_processing_enterprise`

-- For table: `product`
CREATE TABLE `product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `origin` varchar(255) NOT NULL,
  `acquisition_price` decimal(10,2) NOT NULL,
  `processing_cost` decimal(10,2) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `department`
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `total_personnel` int(11) NOT NULL,
  `department_responsibilities` text NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `employee`
CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `telephone_number` varchar(20) NOT NULL,
  `position` varchar(255) NOT NULL,
  `entry_time` datetime NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `purchase`
CREATE TABLE `purchase` (
  `purchase_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `purchaser_information` varchar(255) NOT NULL,
  `consignor` varchar(255) NOT NULL,
  `purchase_time` datetime NOT NULL,
  `consignor_telephone_number` varchar(20) NOT NULL,
  `shipping_address` text NOT NULL,
  `shipping_time` datetime NOT NULL,
  PRIMARY KEY (`purchase_id`),
  FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `processing`
CREATE TABLE `processing` (
  `processing_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `processor` varchar(255) NOT NULL,
  `processing_time` datetime NOT NULL,
  PRIMARY KEY (`processing_id`),
  FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `sale`
CREATE TABLE `sale` (
  `sale_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `salesperson` varchar(255) NOT NULL,
  `customer` varchar(255) NOT NULL,
  `sale_time` datetime NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`sale_id`),
  FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
