-- Schema for database: `rschema_0031`

-- Table: `moving_teams`
CREATE TABLE `moving_teams` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_name` varchar(255) NOT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `tasks`
CREATE TABLE `tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone_number` varchar(255) NOT NULL,
  `moving_date` datetime NOT NULL,
  `starting_address` varchar(255) NOT NULL,
  `destination_address` varchar(255) NOT NULL,
  `quantity_of_items_moved` int(11) NOT NULL,
  `moving_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`team_id`) REFERENCES `moving_teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `feedback`
CREATE TABLE `feedback` (
  `task_id` int(11) NOT NULL,
  `customer_satisfaction` varchar(255) NOT NULL,
  `number_of_tasks_completed` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`task_id`) REFERENCES `tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
