-- Schema for database: `rschema_0032`

-- Table: `rental_tasks`

CREATE TABLE `rental_tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone_number` varchar(255) NOT NULL,
  `rental_date` datetime NOT NULL,
  `rental_duration` int(11) NOT NULL,
  `rental_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`task_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `rental_vehicles`

CREATE TABLE `rental_vehicles` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_condition` varchar(255) NOT NULL,
  `customer_feedback` text,
  `performance_assessment` int(11) NOT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `rental_allocations`

CREATE TABLE `rental_allocations` (
  `allocation_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  PRIMARY KEY (`allocation_id`),
  FOREIGN KEY (`task_id`) REFERENCES `rental_tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`vehicle_id`) REFERENCES `rental_vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
