-- Schema for database: `rschema_0037`

-- Table: `projects`
CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `project_manager` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `expected_completion_date` datetime NOT NULL,
  `actual_completion_date` datetime,
  `project_budget` decimal(10,2) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `inspectors`
CREATE TABLE `inspectors` (
  `inspector_id` int(11) NOT NULL AUTO_INCREMENT,
  `inspector_name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`inspector_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `inspection_records`
CREATE TABLE `inspection_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `inspector_id` int(11) NOT NULL,
  `inspection_date` datetime NOT NULL,
  `inspection_content` text NOT NULL,
  `inspection_results` text NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`inspector_id`) REFERENCES `inspectors` (`inspector_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
