-- Schema for database: `rschema_0038`

-- Table: `designers`

CREATE TABLE `designers` (
  `DesignerID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phonenumber` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  PRIMARY KEY (`DesignerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `clients`

CREATE TABLE `clients` (
  `ClientID` int(11) NOT NULL AUTO_INCREMENT,
  `phonenumber` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`ClientID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `projects`

CREATE TABLE `projects` (
  `ProjectID` int(11) NOT NULL AUTO_INCREMENT,
  `ProjectManager` varchar(255) NOT NULL,
  `projectname` varchar(255) NOT NULL,
  `startdate` datetime NOT NULL,
  `estimatedcompletiondate` datetime NOT NULL,
  `actualcompletiondate` datetime DEFAULT NULL,
  `projectbudget` decimal(10,2) NOT NULL,
  `ClientID` int(11) NOT NULL,
  PRIMARY KEY (`ProjectID`),
  FOREIGN KEY (`ClientID`) REFERENCES `clients` (`ClientID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `design_plans`

CREATE TABLE `design_plans` (
  `DesignPlanID` int(11) NOT NULL AUTO_INCREMENT,
  `DesignerID` int(11) NOT NULL,
  `designdate` datetime NOT NULL,
  `designplanname` varchar(255) NOT NULL,
  `designplancontent` text NOT NULL,
  `ProjectID` int(11) NOT NULL,
  PRIMARY KEY (`DesignPlanID`),
  FOREIGN KEY (`DesignerID`) REFERENCES `designers` (`DesignerID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ProjectID`) REFERENCES `projects` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
