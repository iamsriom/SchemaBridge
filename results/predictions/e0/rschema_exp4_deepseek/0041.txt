-- Schema for database: `real_estate_company`

-- Table: `community`

CREATE TABLE `community` (
  `community_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `property_manager_id` int(11) NOT NULL,
  PRIMARY KEY (`community_number`),
  KEY `property_manager_id` (`property_manager_id`),
  CONSTRAINT `community_ibfk_1` FOREIGN KEY (`property_manager_id`) REFERENCES `property_manager` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `property_manager`

CREATE TABLE `property_manager` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `resident`

CREATE TABLE `resident` (
  `resident_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  `building_number` varchar(255) NOT NULL,
  `community_number` int(11) NOT NULL,
  PRIMARY KEY (`resident_number`),
  KEY `community_number` (`community_number`),
  CONSTRAINT `resident_ibfk_1` FOREIGN KEY (`community_number`) REFERENCES `community` (`community_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance`

CREATE TABLE `maintenance` (
  `order_number` int(11) NOT NULL AUTO_INCREMENT,
  `community_number` int(11) NOT NULL,
  `resident_number` int(11) NOT NULL,
  `date` datetime NOT NULL,
  `content` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`order_number`),
  KEY `community_number` (`community_number`),
  KEY `resident_number` (`resident_number`),
  KEY `personnel_id` (`personnel_id`),
  CONSTRAINT `maintenance_ibfk_1` FOREIGN KEY (`community_number`) REFERENCES `community` (`community_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_ibfk_2` FOREIGN KEY (`resident_number`) REFERENCES `resident` (`resident_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_ibfk_3` FOREIGN KEY (`personnel_id`) REFERENCES `maintenance_personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance_personnel`

CREATE TABLE `maintenance_personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
