-- Schema for database: `property_management`

-- Table: `community`

CREATE TABLE `community` (
  `community_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_number` int(11) NOT NULL,
  `community_name` varchar(255) NOT NULL,
  `community_location` varchar(255) NOT NULL,
  `property_manager_id` int(11) NOT NULL,
  PRIMARY KEY (`community_id`),
  KEY `property_manager_id` (`property_manager_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `property_manager`

CREATE TABLE `property_manager` (
  `property_manager_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`property_manager_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `resident`

CREATE TABLE `resident` (
  `resident_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `resident_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  `building_number` int(11) NOT NULL,
  PRIMARY KEY (`resident_id`),
  KEY `community_id` (`community_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `building`

CREATE TABLE `building` (
  `building_number` int(11) NOT NULL,
  `building_name` varchar(255) NOT NULL,
  `building_location` varchar(255) NOT NULL,
  PRIMARY KEY (`building_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance_service`

CREATE TABLE `maintenance_service` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `resident_id` int(11) NOT NULL,
  `maintenance_date` date NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  KEY `community_id` (`community_id`),
  KEY `resident_id` (`resident_id`),
  KEY `maintenance_personnel_id` (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance_personnel`

CREATE TABLE `maintenance_personnel` (
  `maintenance_personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
