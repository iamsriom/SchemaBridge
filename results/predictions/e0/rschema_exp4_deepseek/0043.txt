-- Schema for database: `real_estate_loan_management`

-- Table: `loan_applications`

CREATE TABLE `loan_applications` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_number` varchar(255) NOT NULL,
  `loan_amount` decimal(10,2) NOT NULL,
  `loan_term` int(11) NOT NULL,
  `approval_status` varchar(255) NOT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `applicant_phone_number` varchar(255) NOT NULL,
  `applicant_ID_number` varchar(255) NOT NULL,
  `application_time` datetime NOT NULL,
  PRIMARY KEY (`application_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `approval_personnel`

CREATE TABLE `approval_personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `personnel_name` varchar(255) NOT NULL,
  `personnel_contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `loan_approval_tracking`

CREATE TABLE `loan_approval_tracking` (
  `track_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_result` varchar(255) NOT NULL,
  PRIMARY KEY (`track_id`),
  FOREIGN KEY (`application_id`) REFERENCES `loan_applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`personnel_id`) REFERENCES `approval_personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `approval_result`

CREATE TABLE `approval_result` (
  `result_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_id` int(11) NOT NULL,
  `approver_name` varchar(255) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_result` varchar(255) NOT NULL,
  PRIMARY KEY (`result_id`),
  FOREIGN KEY (`application_id`) REFERENCES `loan_applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
