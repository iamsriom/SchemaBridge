-- Schema for database: rschema_0044

-- Table: property_basic_information
CREATE TABLE `property_basic_information` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` int(11) NOT NULL,
  `sales_price` int(11) NOT NULL,
  `sales_date` datetime NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: analysis_report
CREATE TABLE `analysis_report` (
  `report_id` int(11) NOT NULL,
  `report_date` datetime NOT NULL,
  `report_type` varchar(255) NOT NULL,
  `analysis_results` text NOT NULL,
  PRIMARY KEY (`report_id`),
  FOREIGN KEY (`report_id`) REFERENCES `property_basic_information` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
