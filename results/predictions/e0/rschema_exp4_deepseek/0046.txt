CREATE TABLE `members`(
    `MemberID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    `CurrentPoints` int(11) NOT NULL DEFAULT 0,
    PRIMARY KEY (`MemberID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shopping_records`(
    `RecordID` int(11) NOT NULL AUTO_INCREMENT,
    `MemberID` int(11) NOT NULL,
    `ShoppingTime` datetime NOT NULL,
    `TotalPrice` decimal(10,2) NOT NULL,
    `PointsChange` int(11) NOT NULL,
    PRIMARY KEY (`RecordID`),
    FOREIGN KEY (`MemberID`) REFERENCES `members` (`MemberID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `point_redemption_records`(
    `RedemptionID` int(11) NOT NULL AUTO_INCREMENT,
    `MemberID` int(11) NOT NULL,
    `RedemptionTime` datetime NOT NULL,
    `RedeemedProductName` varchar(255) NOT NULL,
    `RedemptionQuantity` int(11) NOT NULL,
    `PointsConsumed` int(11) NOT NULL,
    PRIMARY KEY (`RedemptionID`),
    FOREIGN KEY (`MemberID`) REFERENCES `members` (`MemberID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
