-- Schema for database: rschema_0047

CREATE TABLE `reporters` (
    `reporter_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`reporter_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `editors` (
    `editor_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`editor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publishers` (
    `publisher_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`publisher_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `articles` (
    `article_id` INT(11) NOT NULL AUTO_INCREMENT,
    `title` VARCHAR(255) NOT NULL,
    `content` TEXT NOT NULL,
    `submission_time` DATETIME NOT NULL,
    `review_time` DATETIME,
    `publishing_time` DATETIME,
    `reporter_id` INT(11) NOT NULL,
    `editor_id` INT(11) NOT NULL,
    `publisher_id` INT(11) NOT NULL,
    PRIMARY KEY (`article_id`),
    FOREIGN KEY (`reporter_id`) REFERENCES `reporters` (`reporter_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`editor_id`) REFERENCES `editors` (`editor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`publisher_id`) REFERENCES `publishers` (`publisher_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
