-- Schema for database: rschema_0048

-- Table: reporters
CREATE TABLE `reporters` (
  `ReporterID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`ReporterID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: editors
CREATE TABLE `editors` (
  `EditorID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`EditorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: publishers
CREATE TABLE `publishers` (
  `PublisherID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`PublisherID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: manuscripts
CREATE TABLE `manuscripts` (
  `ManuscriptID` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `submission_time` datetime NOT NULL,
  `review_time` datetime NOT NULL,
  `publication_time` datetime NOT NULL,
  `status` enum('pending_review','reviewed','published') NOT NULL,
  PRIMARY KEY (`ManuscriptID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: manuscript_reporters
CREATE TABLE `manuscript_reporters` (
  `ManuscriptID` int(11) NOT NULL,
  `ReporterID` int(11) NOT NULL,
  PRIMARY KEY (`ManuscriptID`,`ReporterID`),
  FOREIGN KEY (`ManuscriptID`) REFERENCES `manuscripts` (`ManuscriptID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ReporterID`) REFERENCES `reporters` (`ReporterID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: manuscript_editors
CREATE TABLE `manuscript_editors` (
  `ManuscriptID` int(11) NOT NULL,
  `EditorID` int(11) NOT NULL,
  PRIMARY KEY (`ManuscriptID`,`EditorID`),
  FOREIGN KEY (`ManuscriptID`) REFERENCES `manuscripts` (`ManuscriptID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`EditorID`) REFERENCES `editors` (`EditorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: manuscript_publishers
CREATE TABLE `manuscript_publishers` (
  `ManuscriptID` int(11) NOT NULL,
  `PublisherID` int(11) NOT NULL,
  PRIMARY KEY (`ManuscriptID`,`PublisherID`),
  FOREIGN KEY (`ManuscriptID`) REFERENCES `manuscripts` (`ManuscriptID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`PublisherID`) REFERENCES `publishers` (`PublisherID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
