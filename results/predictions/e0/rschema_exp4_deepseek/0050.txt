CREATE TABLE `Movie`(
`MovieID` int(11) NOT NULL,
`Name` varchar(255) NOT NULL,
`Type` varchar(255) NOT NULL,
`ReleaseDate` datetime NOT NULL,
PRIMARY KEY (`MovieID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `CopyrightHolder`(
`HolderID` int(11) NOT NULL,
`Name` varchar(255) NOT NULL,
`Email` varchar(255) NOT NULL,
`Phone` varchar(255) NOT NULL,
PRIMARY KEY (`HolderID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `License`(
`LicenseID` int(11) NOT NULL,
`MovieID` int(11) NOT NULL,
`HolderID` int(11) NOT NULL,
`LicenseType` varchar(255) NOT NULL,
`StartTime` datetime NOT NULL,
`EndTime` datetime NOT NULL,
PRIMARY KEY (`LicenseID`),
FOREIGN KEY (`MovieID`) REFERENCES `Movie`(`MovieID`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`HolderID`) REFERENCES `CopyrightHolder`(`HolderID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
