-- Schema for database: `rschema_0051`

-- For table: `resource`
CREATE TABLE `resource` (
  `resource_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`resource_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `resource_monitoring`
CREATE TABLE `resource_monitoring` (
  `monitoring_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `monitoring_personnel` varchar(255) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `monitoring_point` varchar(255) NOT NULL,
  `water_quality_indicators` varchar(255) NOT NULL,
  `water_volume_data` varchar(255) NOT NULL,
  `pollution_level` varchar(255) NOT NULL,
  PRIMARY KEY (`monitoring_id`),
  FOREIGN KEY (`resource_id`) REFERENCES `resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `resource_maintenance`
CREATE TABLE `resource_maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_costs` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`resource_id`) REFERENCES `resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `personnel`
CREATE TABLE `personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `monitoring_maintenance`
CREATE TABLE `monitoring_maintenance` (
  `monitoring_id` int(11) NOT NULL,
  `maintenance_id` int(11) NOT NULL,
  PRIMARY KEY (`monitoring_id`, `maintenance_id`),
  FOREIGN KEY (`monitoring_id`) REFERENCES `resource_monitoring` (`monitoring_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`maintenance_id`) REFERENCES `resource_maintenance` (`maintenance_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
