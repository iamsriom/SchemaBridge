-- Schema for database: `land_management_system`

-- For table: `land_resources`
CREATE TABLE `land_resources` (
  `land_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `overall_area` int(11) NOT NULL,
  PRIMARY KEY (`land_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `projects`
CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `purpose` varchar(255) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `land_usage`
CREATE TABLE `land_usage` (
  `usage_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `land_id` int(11) NOT NULL,
  `area_used` int(11) NOT NULL,
  `years_of_usage` int(11) NOT NULL,
  `current_status` varchar(255) NOT NULL,
  PRIMARY KEY (`usage_id`),
  FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`land_id`) REFERENCES `land_resources` (`land_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `land_changes`
CREATE TABLE `land_changes` (
  `change_id` int(11) NOT NULL AUTO_INCREMENT,
  `land_id` int(11) NOT NULL,
  `content_of_change` varchar(255) NOT NULL,
  `reason_for_change` varchar(255) NOT NULL,
  `costs_associated` int(11) NOT NULL,
  `time_of_change` datetime NOT NULL,
  PRIMARY KEY (`change_id`),
  FOREIGN KEY (`land_id`) REFERENCES `land_resources` (`land_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `usage_tracking`
CREATE TABLE `usage_tracking` (
  `tracking_id` int(11) NOT NULL AUTO_INCREMENT,
  `usage_id` int(11) NOT NULL,
  `personnel` varchar(255) NOT NULL,
  `timing` datetime NOT NULL,
  PRIMARY KEY (`tracking_id`),
  FOREIGN KEY (`usage_id`) REFERENCES `land_usage` (`usage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
