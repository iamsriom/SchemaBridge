-- Schema for database: rschema_0053
-- For table: processing_pool
CREATE TABLE `processing_pool` (
  `processing_pool_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `responsible_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`processing_pool_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: staff
CREATE TABLE `staff` (
  `staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `responsibility` varchar(255) NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: operation_records
CREATE TABLE `operation_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_pool_id` int(11) NOT NULL,
  `capacity` int(11) NOT NULL,
  `efficiency` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `staff_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pool` (`processing_pool_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: maintenance_records
CREATE TABLE `maintenance_records` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `content` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `processing_pool_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pool` (`processing_pool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
