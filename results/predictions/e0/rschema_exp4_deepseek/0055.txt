-- Schema for database: rschema_0055

-- Table: users
CREATE TABLE `users` (
    `user_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `contact_information` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: electricity_usage
CREATE TABLE `electricity_usage` (
    `record_time` DATETIME NOT NULL,
    `electricity_consumption` DECIMAL(10,2) NOT NULL,
    `user_id` INT(11) NOT NULL,
    `operator_id` INT(11) NOT NULL,
    PRIMARY KEY (`record_time`),
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`operator_id`) REFERENCES `operators` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: operators
CREATE TABLE `operators` (
    `operator_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `department` VARCHAR(255) NOT NULL,
    `contact_information` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
