-- Schema for database: `heat_failure_management`

-- Table: `faults`
CREATE TABLE `faults` (
  `fault_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` varchar(255) NOT NULL,
  `occurrence_date` datetime NOT NULL,
  `resolution_date` datetime,
  `handler_id` int(11) NOT NULL,
  PRIMARY KEY (`fault_id`),
  KEY `handler_id` (`handler_id`),
  CONSTRAINT `faults_ibfk_1` FOREIGN KEY (`handler_id`) REFERENCES `handlers` (`handler_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `handlers`
CREATE TABLE `handlers` (
  `handler_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`handler_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `users`
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `fault_user`
CREATE TABLE `fault_user` (
  `fault_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`fault_id`,`user_id`),
  KEY `user_id` (`user_id`),
  CONSTRAINT `fault_user_ibfk_1` FOREIGN KEY (`fault_id`) REFERENCES `faults` (`fault_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fault_user_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
