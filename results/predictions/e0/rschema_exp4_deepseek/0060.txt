-- Schema for database: `research_institution`

-- Table: `project`

CREATE TABLE `project` (
    `project_id` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `start_date` datetime NOT NULL,
    `end_date` datetime NOT NULL,
    `project_leader` varchar(255) NOT NULL,
    `budget` int(11) NOT NULL,
    PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `researcher`

CREATE TABLE `researcher` (
    `employee_number` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `date_of_birth` datetime NOT NULL,
    `gender` char(6) NOT NULL,
    `phone_number` varchar(255) NOT NULL,
    `position` varchar(255) NOT NULL,
    PRIMARY KEY (`employee_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `project_funds`

CREATE TABLE `project_funds` (
    `fund_id` int(11) NOT NULL,
    `project_id` int(11) NOT NULL,
    `purpose` varchar(255) NOT NULL,
    `amount` int(11) NOT NULL,
    `usage_date` datetime NOT NULL,
    PRIMARY KEY (`fund_id`),
    FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `project_result`

CREATE TABLE `project_result` (
    `result_id` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `publication_date` datetime NOT NULL,
    `employee_number` int(11) NOT NULL,
    PRIMARY KEY (`result_id`),
    FOREIGN KEY (`employee_number`) REFERENCES `researcher` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
