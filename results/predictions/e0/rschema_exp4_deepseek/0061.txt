CREATE TABLE `patent`(
`patent_number` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`application_date` datetime NOT NULL,
`status` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventor`(
`work_number` int(11) NOT NULL,
`name` varchar(255) NOT NULL,
`date_of_birth` date NOT NULL,
`gender` char(6) NOT NULL,
`telephone_number` varchar(255) NOT NULL,
`position` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patent_application`(
`application_number` int(11) NOT NULL AUTO_INCREMENT,
`patent_number` int(11) NOT NULL,
`inventor` int(11) NOT NULL,
`application_date` datetime NOT NULL,
PRIMARY KEY (`application_number`),
KEY `patent_number` (`patent_number`),
KEY `inventor` (`inventor`),
CONSTRAINT `patent_application_ibfk_1` FOREIGN KEY (`patent_number`) REFERENCES `patent` (`patent_number`) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT `patent_application_ibfk_2` FOREIGN KEY (`inventor`) REFERENCES `inventor` (`work_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
