CREATE TABLE `documents` (
  `DocumentID` int(11) NOT NULL,
  `Title` varchar(255) NOT NULL,
  `CreationDate` datetime NOT NULL,
  `VersionNumber` int(11) NOT NULL,
  PRIMARY KEY (`DocumentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authors` (
  `ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `DateofBirth` date NOT NULL,
  `Gender` char(6) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `Position` varchar(255) NOT NULL,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `document_versions` (
  `ModificationID` int(11) NOT NULL,
  `DocumentID` int(11) NOT NULL,
  `VersionNumber` int(11) NOT NULL,
  `ModificationDate` datetime NOT NULL,
  `ModifyingAuthor` int(11) NOT NULL,
  `ModificationContent` text NOT NULL,
  PRIMARY KEY (`ModificationID`),
  FOREIGN KEY (`DocumentID`) REFERENCES `documents` (`DocumentID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ModifyingAuthor`) REFERENCES `authors` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
