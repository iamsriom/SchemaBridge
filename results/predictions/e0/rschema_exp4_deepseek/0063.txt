-- Schema for database: `rschema_0063`

-- Table: `achievements`

CREATE TABLE `achievements` (
    `achievement_number` INT NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `research_and_development_date` DATETIME NOT NULL,
    `technical_field` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`achievement_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `products`

CREATE TABLE `products` (
    `product_number` INT NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `model` VARCHAR(255) NOT NULL,
    `barcode` CHAR(6) NOT NULL,
    `price` DECIMAL(10, 2) NOT NULL,
    `transformation_date` DATETIME NOT NULL,
    PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `achievement_transformation`

CREATE TABLE `achievement_transformation` (
    `achievement_number` INT NOT NULL,
    `product_number` INT NOT NULL,
    `transformation_date` DATETIME NOT NULL,
    PRIMARY KEY (`achievement_number`, `product_number`),
    FOREIGN KEY (`achievement_number`) REFERENCES `achievements` (`achievement_number`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`product_number`) REFERENCES `products` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
