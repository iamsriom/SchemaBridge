-- Schema for database: `rschema_0064`

-- For table: `project`
CREATE TABLE `project` (
  `project_number` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `application_units` int(11) NOT NULL,
  PRIMARY KEY (`project_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `review_expert`
CREATE TABLE `review_expert` (
  `expert_number` int(11) NOT NULL,
  `expert_name` varchar(255) NOT NULL,
  `units` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`expert_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `project_review`
CREATE TABLE `project_review` (
  `project_number` int(11) NOT NULL,
  `expert_number` int(11) NOT NULL,
  `review_date` datetime NOT NULL,
  `review_opinion` varchar(255) NOT NULL,
  PRIMARY KEY (`project_number`,`expert_number`),
  FOREIGN KEY (`project_number`) REFERENCES `project` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`expert_number`) REFERENCES `review_expert` (`expert_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
