-- Schema for database: `rental_management_system`

-- For table: `customers`
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` char(6) NOT NULL,
  `address` varchar(255) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  `total_rental_times` int(11) NOT NULL DEFAULT 0,
  `total_rental_fees` decimal(10,2) NOT NULL DEFAULT 0.00,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `rental_items`
CREATE TABLE `rental_items` (
  `item_id` int(11) NOT NULL AUTO_INCREMENT,
  `item_name` varchar(255) NOT NULL,
  `category` varchar(255) NOT NULL,
  `rental_fee` decimal(10,2) NOT NULL,
  `total_inventory` int(11) NOT NULL,
  `remaining_quantity` int(11) NOT NULL,
  PRIMARY KEY (`item_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `rental_records`
CREATE TABLE `rental_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `item_id` int(11) NOT NULL,
  `rental_time_period` datetime NOT NULL,
  `number_of_rental_items` int(11) NOT NULL,
  `payment_fees` decimal(10,2) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`item_id`) REFERENCES `rental_items` (`item_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
