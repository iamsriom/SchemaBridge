-- Schema for database: `rschema_0070`

-- Table: `mine_management`
CREATE TABLE `mine_management` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `ore_management`
CREATE TABLE `ore_management` (
  `ore_id` int(11) NOT NULL AUTO_INCREMENT,
  `ore_name` varchar(255) NOT NULL,
  `characteristics` varchar(255) NOT NULL,
  PRIMARY KEY (`ore_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `processing_plant_management`
CREATE TABLE `processing_plant_management` (
  `plant_id` int(11) NOT NULL AUTO_INCREMENT,
  `plant_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`plant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `ore_output_management`
CREATE TABLE `ore_output_management` (
  `output_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_id` int(11) NOT NULL,
  `ore_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`output_id`),
  FOREIGN KEY (`mine_id`) REFERENCES `mine_management` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ore_id`) REFERENCES `ore_management` (`ore_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `ore_processing_management`
CREATE TABLE `ore_processing_management` (
  `processing_id` int(11) NOT NULL AUTO_INCREMENT,
  `plant_id` int(11) NOT NULL,
  `ore_id` int(11) NOT NULL,
  `processing_date` datetime NOT NULL,
  `processing_quantity` int(11) NOT NULL,
  `processing_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`processing_id`),
  FOREIGN KEY (`plant_id`) REFERENCES `processing_plant_management` (`plant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ore_id`) REFERENCES `ore_management` (`ore_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `employee_management`
CREATE TABLE `employee_management` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
