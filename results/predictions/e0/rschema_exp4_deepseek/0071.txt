CREATE TABLE `mines` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_officers` (
  `officer_id` int(11) NOT NULL AUTO_INCREMENT,
  `officer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `telephone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`officer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspections` (
  `inspection_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_id` int(11) NOT NULL,
  `officer_id` int(11) NOT NULL,
  `date` datetime NOT NULL,
  `content` text NOT NULL,
  `results` text NOT NULL,
  PRIMARY KEY (`inspection_id`),
  FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`officer_id`) REFERENCES `safety_officers` (`officer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
