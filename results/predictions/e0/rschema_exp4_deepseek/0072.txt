-- Schema for database: `rschema_0072`

-- Table: `oil_fields`

CREATE TABLE `oil_fields` (
  `field_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`field_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `pipelines`

CREATE TABLE `pipelines` (
  `pipeline_id` int(11) NOT NULL,
  `starting_point` varchar(255) NOT NULL,
  `ending_point` varchar_255 NOT NULL,
  `length` int(11) NOT NULL,
  `field_id` int(11) NOT NULL,
  PRIMARY KEY (`pipeline_id`),
  FOREIGN KEY (`field_id`) REFERENCES `oil_fields` (`field_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance_records`

CREATE TABLE `maintenance_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `maintenance_date` date NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `pipeline_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`pipeline_id`) REFERENCES `pipelines` (`pipeline_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
