-- Schema for database: `rschema_0076`

-- Table: `manufacturers`

CREATE TABLE `manufacturers` (
  `manufacturer_number` INT NOT NULL,
  `manufacturer_name` VARCHAR(255) NOT NULL,
  `manufacturer_country` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`manufacturer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `beer_types`

CREATE TABLE `beer_types` (
  `beer_number` INT NOT NULL,
  `beer_name` VARCHAR(255) NOT NULL,
  `manufacturer_number` INT NOT NULL,
  `beer_price` DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (`beer_number`),
  FOREIGN KEY (`manufacturer_number`) REFERENCES `manufacturers`(`manufacturer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `bars`

CREATE TABLE `bars` (
  `bar_number` INT NOT NULL,
  `bar_name` VARCHAR(255) NOT NULL,
  `bar_address` VARCHAR(255) NOT NULL,
  `bar_phone_number` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`bar_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `drinkers`

CREATE TABLE `drinkers` (
  `drinker_number` INT NOT NULL,
  `drinker_name` VARCHAR(255) NOT NULL,
  `drinker_gender` CHAR(6) NOT NULL,
  `drinker_age` INT NOT NULL,
  `drinker_phone_number` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`drinker_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sales`

CREATE TABLE `sales` (
  `sales_date` DATETIME NOT NULL,
  `sales_quantity` INT NOT NULL,
  `sales_price` DECIMAL(10,2) NOT NULL,
  `beer_number` INT NOT NULL,
  `bar_number` INT NOT NULL,
  `drinker_number` INT NOT NULL,
  PRIMARY KEY (`sales_date`, `beer_number`, `bar_number`, `drinker_number`),
  FOREIGN KEY (`beer_number`) REFERENCES `beer_types`(`beer_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`bar_number`) REFERENCES `bars`(`bar_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`drinker_number`) REFERENCES `drinkers`(`drinker_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
