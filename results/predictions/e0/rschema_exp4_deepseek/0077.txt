-- Schema for database: `rschema_0077`

-- Table: `users`
CREATE TABLE `users` (
  `membership_number` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `postal_code` varchar(255) NOT NULL,
  `address_details` varchar(255) NOT NULL,
  PRIMARY KEY (`membership_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `products`
CREATE TABLE `products` (
  `product_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `category_id` int(11) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `product_image_url` varchar(255) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `orders`
CREATE TABLE `orders` (
  `purchase_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `membership_number` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `delivery_status` varchar(255) NOT NULL,
  PRIMARY KEY (`purchase_id`),
  FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`),
  FOREIGN KEY (`membership_number`) REFERENCES `users` (`membership_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `announcements`
CREATE TABLE `announcements` (
  `announcement_number` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `registration_date` datetime NOT NULL,
  `publisher_id` int(11) NOT NULL,
  PRIMARY KEY (`announcement_number`),
  FOREIGN KEY (`publisher_id`) REFERENCES `users` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
