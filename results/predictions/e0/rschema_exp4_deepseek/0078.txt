CREATE TABLE `articles` (
  `article_id` int(11) NOT NULL,
  `article_title` varchar(255) NOT NULL,
  `article_content` text NOT NULL,
  `creation_time` datetime NOT NULL,
  `total_number_of_words` int(11) NOT NULL,
  `article_reading_volume` int(11) NOT NULL,
  `article_like_volume` int(11) NOT NULL,
  `article_collection_volume` int(11) NOT NULL,
  `article_comment_volume` int(11) NOT NULL,
  `last_modification_time` datetime NOT NULL,
  `last_modification_author_id` int(11) NOT NULL,
  PRIMARY KEY (`article_id`),
  FOREIGN KEY (`last_modification_author_id`) REFERENCES `authors` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authors` (
  `author_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  `introduction` text NOT NULL,
  PRIMARY KEY (`author_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `author_article` (
  `author_id` int(11) NOT NULL,
  `article_id` int(11) NOT NULL,
  `author_order` int(11) NOT NULL,
  `responsible_part` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  PRIMARY KEY (`author_id`,`article_id`),
  FOREIGN KEY (`author_id`) REFERENCES `authors` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`article_id`) REFERENCES `articles` (`article_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `modifications` (
  `modification_id` int(11) NOT NULL AUTO_INCREMENT,
  `article_id` int(11) NOT NULL,
  `modification_start_time` datetime NOT NULL,
  `modification_save_time` datetime NOT NULL,
  `modification_notes` text NOT NULL,
  PRIMARY KEY (`modification_id`),
  FOREIGN KEY (`article_id`) REFERENCES `articles` (`article_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
