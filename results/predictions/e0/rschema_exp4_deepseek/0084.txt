-- Schema for database: rschema_0084

-- Table: land_parcels

CREATE TABLE `land_parcels` (
  `LandID` int(11) NOT NULL,
  `LandLocation` varchar(255) NOT NULL,
  `TotalLandArea` int(11) NOT NULL,
  PRIMARY KEY (`LandID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: projects

CREATE TABLE `projects` (
  `ProjectID` int(11) NOT NULL,
  `ProjectName` varchar(255) NOT NULL,
  `ProjectLocation` varchar(255) NOT NULL,
  `Purpose` varchar(255) NOT NULL,
  `PersonInCharge` varchar(255) NOT NULL,
  PRIMARY KEY (`ProjectID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: land_usage_records

CREATE TABLE `land_usage_records` (
  `RecordID` int(11) NOT NULL AUTO_INCREMENT,
  `LandID` int(11) NOT NULL,
  `ProjectID` int(11) NOT NULL,
  `UsedLandArea` int(11) NOT NULL,
  `UsageDuration` varchar(255) NOT NULL,
  `CurrentStatus` varchar(255) NOT NULL,
  `RecordTime` datetime NOT NULL,
  `RecordingPersonnel` varchar(255) NOT NULL,
  PRIMARY KEY (`RecordID`),
  KEY `LandID` (`LandID`),
  KEY `ProjectID` (`ProjectID`),
  CONSTRAINT `land_usage_records_ibfk_1` FOREIGN KEY (`LandID`) REFERENCES `land_parcels` (`LandID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `land_usage_records_ibfk_2` FOREIGN KEY (`ProjectID`) REFERENCES `projects` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
