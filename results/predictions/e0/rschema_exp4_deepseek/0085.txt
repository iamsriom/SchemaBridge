-- Schema for database: rschema_0085

-- Table: User
CREATE TABLE `User` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Profile` text,
    `Avatar` varchar(255),
    PRIMARY KEY (`UserID`),
    UNIQUE KEY `Email` (`Email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Recipe
CREATE TABLE `Recipe` (
    `RecipeID` int(11) NOT NULL AUTO_INCREMENT,
    `RecipeName` varchar(255) NOT NULL,
    `Ingredients` text NOT NULL,
    `Steps` text NOT NULL,
    `TotalTime` int(11) NOT NULL,
    `Portion` int(11) NOT NULL,
    `SpicyLevel` int(11) NOT NULL,
    `BitterLevel` int(11) NOT NULL,
    `SweetLevel` int(11) NOT NULL,
    `MeatyLevel` int(11) NOT NULL,
    `SaltyLevel` int(11) NOT NULL,
    `SourLevel` int(11) NOT NULL,
    `Rating` decimal(3,1) NOT NULL,
    `NumberofRatings` int(11) NOT NULL,
    `LargeImageFilename` varchar(255),
    `Calories` int(11) NOT NULL,
    `CreatorID` int(11) NOT NULL,
    `SmallImageFilename` varchar(255),
    PRIMARY KEY (`RecipeID`),
    KEY `CreatorID` (`CreatorID`),
    CONSTRAINT `CreatorID` FOREIGN KEY (`CreatorID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Blog
CREATE TABLE `Blog` (
    `BlogID` int(11) NOT NULL AUTO_INCREMENT,
    `AuthorUserID` int(11) NOT NULL,
    `BlogTitle` varchar(255) NOT NULL,
    `PublicationTime` datetime NOT NULL,
    `BlogSummary` text NOT NULL,
    PRIMARY KEY (`BlogID`),
    KEY `AuthorUserID` (`AuthorUserID`),
    CONSTRAINT `AuthorUserID` FOREIGN KEY (`AuthorUserID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
