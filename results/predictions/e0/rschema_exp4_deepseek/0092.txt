-- Schema for database: `rschema_0092`

-- Table: `users`

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `identity_type` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `identity_status` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  `expiration_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_relationships`

CREATE TABLE `user_relationships` (
  `user_id` int(11) NOT NULL,
  `superior_id` int(11) DEFAULT NULL,
  `subordinate_id` int(11) DEFAULT NULL,
  `change_time` datetime NOT NULL,
  `operator_id` int(11) NOT NULL,
  `remarks` text,
  PRIMARY KEY (`user_id`),
  KEY `superior_id` (`superior_id`),
  KEY `subordinate_id` (`subordinate_id`),
  CONSTRAINT `user_relationships_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_relationships_ibfk_2` FOREIGN KEY (`superior_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `user_relationships_ibfk_3` FOREIGN KEY (`subordinate_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `operators`

CREATE TABLE `operators` (
  `operator_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `permission` varchar(255) NOT NULL,
  PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `change_audit`

CREATE TABLE `change_audit` (
  `audit_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `change_time` datetime NOT NULL,
  `before_superior_id` int(11) DEFAULT NULL,
  `after_superior_id` int(11) DEFAULT NULL,
  `before_subordinate_ids` text,
  `after_subordinate_ids` text,
  `operator_id` int(11) NOT NULL,
  `remarks` text,
  PRIMARY KEY (`audit_id`),
  KEY `user_id` (`user_id`),
  CONSTRAINT `change_audit_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `change_audit_ibfk_2` FOREIGN KEY (`operator_id`) REFERENCES `operators` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
