-- Schema for database: `rschema_0096Naturallanguagedescription`

-- Table: `customers`

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `hair_color` varchar(255) NOT NULL,
  `appointment_time` datetime NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `hairstylists`

CREATE TABLE `hairstylists` (
  `hairstylist_id` int(11) NOT NULL AUTO_INCREMENT,
  `hairstylist_name` varchar(255) NOT NULL,
  `expertise` varchar(255) NOT NULL,
  PRIMARY KEY (`hairstylist_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `services`

CREATE TABLE `services` (
  `service_id` int(11) NOT NULL AUTO_INCREMENT,
  `hairstylist_id` int(11) NOT NULL,
  `service_time` datetime NOT NULL,
  `service_content` varchar(255) NOT NULL,
  PRIMARY KEY (`service_id`),
  FOREIGN KEY (`hairstylist_id`) REFERENCES `hairstylists` (`hairstylist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `appointments`

CREATE TABLE `appointments` (
  `appointment_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `hairstylist_id` int(11) NOT NULL,
  `appointment_time` datetime NOT NULL,
  PRIMARY KEY (`appointment_id`),
  FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`hairstylist_id`) REFERENCES `hairstylists` (`hairstylist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
