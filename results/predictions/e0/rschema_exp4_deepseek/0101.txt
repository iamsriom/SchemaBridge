-- Schema for database: `rschema_0101`

-- Table: `employee_information`
CREATE TABLE `employee_information` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` date NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `ID_number` varchar(20) NOT NULL,
  `place_of_origin` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `department` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `level` varchar(255) NOT NULL,
  `contract_type` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department_information`
CREATE TABLE `department_information` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `responsibilities` text NOT NULL,
  `head_id` int(11) NOT NULL,
  `department_location` varchar(255) NOT NULL,
  `total_number_of_people` int(11) NOT NULL,
  PRIMARY KEY (`department_id`),
  FOREIGN KEY (`head_id`) REFERENCES `employee_information` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `attendance_records`
CREATE TABLE `attendance_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `clock_in_time` datetime NOT NULL,
  `clock_out_time` datetime NOT NULL,
  `clock_in_type` varchar(255) NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employee_information` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department_transfers`
CREATE TABLE `department_transfers` (
  `transfer_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `from_department` varchar(255) NOT NULL,
  `to_department` varchar(255) NOT NULL,
  `transfer_time` datetime NOT NULL,
  `position_after_transfer` varchar(255) NOT NULL,
  PRIMARY KEY (`transfer_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employee_information` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `employee_performance_evaluation`
CREATE TABLE `employee_performance_evaluation` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `evaluation_content` text NOT NULL,
  `evaluation_results` text NOT NULL,
  `evaluator` varchar(255) NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employee_information` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
