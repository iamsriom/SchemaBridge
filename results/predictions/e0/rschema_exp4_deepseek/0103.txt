-- Schema for database: `rschema_0103`

-- Table: `employee`

CREATE TABLE `employee` (
  `employeeID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `dateofbirth` datetime NOT NULL,
  `hiredate` datetime NOT NULL,
  PRIMARY KEY (`employeeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `position_history`

CREATE TABLE `position_history` (
  `employeeID` int(11) NOT NULL,
  `originaldepartment` varchar(255) NOT NULL,
  `originalposition` varchar(255) NOT NULL,
  `currentdepartment` varchar(255) NOT NULL,
  `currentposition` varchar(255) NOT NULL,
  `entrydate` datetime NOT NULL,
  `departuredate` datetime,
  PRIMARY KEY (`employeeID`, `entrydate`),
  FOREIGN KEY (`employeeID`) REFERENCES `employee` (`employeeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department`

CREATE TABLE `department` (
  `departmentcode` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `personincharge` varchar(255) NOT NULL,
  PRIMARY KEY (`departmentcode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department_affiliation`

CREATE TABLE `department_affiliation` (
  `employeeID` int(11) NOT NULL,
  `departmentcode` varchar(255) NOT NULL,
  `affiliationdate` datetime NOT NULL,
  PRIMARY KEY (`employeeID`, `departmentcode`),
  FOREIGN KEY (`employeeID`) REFERENCES `employee` (`employeeID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`departmentcode`) REFERENCES `department` (`departmentcode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `salary_standard`

CREATE TABLE `salary_standard` (
  `standarddescription` varchar(255) NOT NULL,
  `standardvalue` decimal(10, 2) NOT NULL,
  PRIMARY KEY (`standarddescription`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `salary_payment`

CREATE TABLE `salary_payment` (
  `employeeID` int(11) NOT NULL,
  `paymentdate` datetime NOT NULL,
  `paymentamount` decimal(10, 2) NOT NULL,
  PRIMARY KEY (`employeeID`, `paymentdate`),
  FOREIGN KEY (`employeeID`) REFERENCES `employee` (`employeeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
