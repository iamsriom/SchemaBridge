--Schemafordatabase:rschema_0105
--Foreachtable:

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `account` varchar(255) NOT NULL,
  `nickname` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `registration_time` datetime NOT NULL,
  `birthday` date NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_profiles` (
  `user_id` int(11) NOT NULL,
  `profile_picture` varchar(255),
  `bio` text,
  `privacy_settings` varchar(255),
  PRIMARY KEY (`user_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dynamics` (
  `dynamic_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `publishing_time` datetime NOT NULL,
  `number_of_likes` int(11) NOT NULL DEFAULT 0,
  `number_of_comments` int(11) NOT NULL DEFAULT 0,
  `number_of_favorites` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`dynamic_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `interactions` (
  `interaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `dynamic_id` int(11) NOT NULL,
  `interaction_type` varchar(255) NOT NULL,
  `interaction_time` datetime NOT NULL,
  PRIMARY KEY (`interaction_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`dynamic_id`) REFERENCES `dynamics` (`dynamic_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `contacts` (
  `user_id` int(11) NOT NULL,
  `contact_id` int(11) NOT NULL,
  `contact_type` varchar(255) NOT NULL,
  `contact_time` datetime NOT NULL,
  `notes` text,
  PRIMARY KEY (`user_id`,`contact_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`contact_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `social_activity` (
  `activity_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `dynamic_id` int(11) NOT NULL,
  `activity_type` varchar(255) NOT NULL,
  `activity_time` datetime NOT NULL,
  PRIMARY KEY (`activity_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`dynamic_id`) REFERENCES `dynamics` (`dynamic_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
