# -- Schema for database: rschema_0107

CREATE TABLE `game_record` (
  `game_id` int(11) NOT NULL AUTO_INCREMENT,
  `rounds` int(11) NOT NULL,
  `grid_size` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `game_result` varchar(255) NOT NULL,
  PRIMARY KEY (`game_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `space_ship` (
  `ship_id` int(11) NOT NULL AUTO_INCREMENT,
  `strength` int(11) NOT NULL,
  `position` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `game_id` int(11) NOT NULL,
  PRIMARY KEY (`ship_id`),
  FOREIGN KEY (`game_id`) REFERENCES `game_record` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fighter` (
  `fighter_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `strength` int(11) NOT NULL,
  `ship_id` int(11) NOT NULL,
  PRIMARY KEY (`fighter_id`),
  FOREIGN KEY (`ship_id`) REFERENCES `space_ship` (`ship_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
