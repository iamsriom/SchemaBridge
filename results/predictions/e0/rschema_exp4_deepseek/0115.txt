-- Schema for database: rschema_0115

-- Table: Unit
CREATE TABLE `Unit` (
    `UnitID` int(11) NOT NULL AUTO_INCREMENT,
    `UnitName` varchar(255) NOT NULL,
    PRIMARY KEY (`UnitID`),
    UNIQUE (`UnitName`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: BenchmarkTestingType
CREATE TABLE `BenchmarkTestingType` (
    `BenchmarkTestingTypeID` int(11) NOT NULL AUTO_INCREMENT,
    PRIMARY KEY (`BenchmarkTestingTypeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Unit_BenchmarkTestingType
CREATE TABLE `Unit_BenchmarkTestingType` (
    `UnitID` int(11) NOT NULL,
    `BenchmarkTestingTypeID` int(11) NOT NULL,
    PRIMARY KEY (`UnitID`, `BenchmarkTestingTypeID`),
    FOREIGN KEY (`UnitID`) REFERENCES `Unit` (`UnitID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`BenchmarkTestingTypeID`) REFERENCES `BenchmarkTestingType` (`BenchmarkTestingTypeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
