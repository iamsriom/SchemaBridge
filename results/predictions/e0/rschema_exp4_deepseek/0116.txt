-- Schema for database: rschema_0116

-- Table: customer_information
CREATE TABLE `customer_information` (
    `userID` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `contact_number` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`userID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: flight_route
CREATE TABLE `flight_route` (
    `flight_routeID` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `route` VARCHAR(255) NOT NULL,
    `time` DATETIME NOT NULL,
    PRIMARY KEY (`flight_routeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: customer_evaluation
CREATE TABLE `customer_evaluation` (
    `customerID` INT(11) NOT NULL,
    `evaluation_content` TEXT NOT NULL,
    `score_of_flight_route` INT(11) NOT NULL,
    `flight_routeID` INT(11) NOT NULL,
    PRIMARY KEY (`customerID`, `flight_routeID`),
    FOREIGN KEY (`customerID`) REFERENCES `customer_information` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`flight_routeID`) REFERENCES `flight_route` (`flight_routeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: customer_bank_account
CREATE TABLE `customer_bank_account` (
    `accountID` INT(11) NOT NULL,
    `account_number` VARCHAR(255) NOT NULL,
    `balance` DECIMAL(10, 2) NOT NULL,
    `customerID` INT(11) NOT NULL,
    PRIMARY KEY (`accountID`),
    UNIQUE (`account_number`),
    FOREIGN KEY (`customerID`) REFERENCES `customer_information` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
