--Schemafordatabase:rschema_0121

--Foreachtable:

CREATE TABLE `users` (
    `username` varchar(255) NOT NULL,
    `full_name` varchar(255) NOT NULL,
    `email` varchar(255) NOT NULL,
    `password_hash` varchar(255) NOT NULL,
    `registration_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `last_login_date` datetime DEFAULT NULL,
    PRIMARY KEY (`username`),
    UNIQUE KEY `email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authenticated_users` (
    `username` varchar(255) NOT NULL,
    `full_name` varchar(255) NOT NULL,
    `authentication_level` int(11) NOT NULL,
    `authentication_time` datetime NOT NULL,
    PRIMARY KEY (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_migrations` (
    `old_user_id` int(11) NOT NULL,
    `new_user_id` int(11) NOT NULL,
    `migration_time` datetime NOT NULL,
    PRIMARY KEY (`old_user_id`, `new_user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authenticated_consumers` (
    `name` varchar(255) NOT NULL,
    `description` text NOT NULL,
    `type` varchar(255) NOT NULL,
    `user_id` varchar(255) NOT NULL,
    `configuration` text NOT NULL,
    `permissions` text NOT NULL,
    `signature` text NOT NULL,
    PRIMARY KEY (`name`),
    FOREIGN KEY (`user_id`) REFERENCES `users` (`username`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sessions` (
    `session_id` varchar(255) NOT NULL,
    `consumer_id` varchar(255) NOT NULL,
    `expiration_time` datetime NOT NULL,
    `mfa_status` varchar(255) NOT NULL,
    `signature` text NOT NULL,
    PRIMARY KEY (`session_id`),
    FOREIGN KEY (`consumer_id`) REFERENCES `authenticated_consumers` (`name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `services` (
    `service_name` varchar(255) NOT NULL,
    `type` varchar(255) NOT NULL,
    `url` varchar(255) NOT NULL,
    `configuration` text NOT NULL,
    `last_heartbeat_time` datetime NOT NULL,
    `monitoring_status` varchar(255) NOT NULL,
    `signature` text NOT NULL,
    PRIMARY KEY (`service_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `worker_nodes` (
    `node_name` varchar(255) NOT NULL,
    `status` varchar(255) NOT NULL,
    `service_id` varchar(255) NOT NULL,
    `operating_system` varchar(255) NOT NULL,
    `architecture` varchar(255) NOT NULL,
    `last_heartbeat_time` datetime NOT NULL,
    `signature` text NOT NULL,
    PRIMARY KEY (`node_name`),
    FOREIGN KEY (`service_id`) REFERENCES `services` (`service_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
