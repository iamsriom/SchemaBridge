-- Schema for database: `rschema_0126`

-- Table: `deceased_info`
CREATE TABLE `deceased_info` (
  `deceased_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `unique_id` char(6) NOT NULL,
  `last_modified` datetime DEFAULT NULL,
  PRIMARY KEY (`deceased_id`),
  UNIQUE KEY `unique_id` (`unique_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `inspector_info`
CREATE TABLE `inspector_info` (
  `inspector_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `last_modified` datetime DEFAULT NULL,
  PRIMARY KEY (`inspector_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `death_record`
CREATE TABLE `death_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `deceased_id` int(11) NOT NULL,
  `inspector_id` int(11) NOT NULL,
  `death_date` datetime NOT NULL,
  `last_modified` datetime DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  KEY `deceased_id` (`deceased_id`),
  KEY `inspector_id` (`inspector_id`),
  CONSTRAINT `death_record_ibfk_1` FOREIGN KEY (`deceased_id`) REFERENCES `deceased_info` (`deceased_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `death_record_ibfk_2` FOREIGN KEY (`inspector_id`) REFERENCES `inspector_info` (`inspector_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
