-- Schema for database: rschema_0128

-- Table: task_execution_record
CREATE TABLE `task_execution_record` (
    `execution_id` int(11) NOT NULL AUTO_INCREMENT,
    `start_time` datetime NOT NULL,
    `end_time` datetime DEFAULT NULL,
    `task_name` varchar(255) NOT NULL,
    `exit_code` int(11) NOT NULL,
    `exit_message` varchar(255) DEFAULT NULL,
    `error_message` varchar(255) DEFAULT NULL,
    `last_update_time` datetime NOT NULL,
    PRIMARY KEY (`execution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_parameter
CREATE TABLE `task_parameter` (
    `execution_id` int(11) NOT NULL,
    `parameter_name` varchar(255) NOT NULL,
    `parameter_value` varchar(255) NOT NULL,
    PRIMARY KEY (`execution_id`,`parameter_name`),
    FOREIGN KEY (`execution_id`) REFERENCES `task_execution_record` (`execution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_batch_record
CREATE TABLE `task_batch_record` (
    `batch_id` int(11) NOT NULL AUTO_INCREMENT,
    `batch_name` varchar(255) NOT NULL,
    `start_time` datetime NOT NULL,
    `end_time` datetime DEFAULT NULL,
    PRIMARY KEY (`batch_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_execution_batch_record
CREATE TABLE `task_execution_batch_record` (
    `execution_id` int(11) NOT NULL,
    `batch_id` int(11) NOT NULL,
    PRIMARY KEY (`execution_id`,`batch_id`),
    FOREIGN KEY (`execution_id`) REFERENCES `task_execution_record` (`execution_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`batch_id`) REFERENCES `task_batch_record` (`batch_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
