-- Schema for database: rschema_0131

-- Table: employees
CREATE TABLE `employees` (
    `employee_id` INT(11) NOT NULL AUTO_INCREMENT,
    `first_name` VARCHAR(255) NOT NULL,
    `last_name` VARCHAR(255) NOT NULL,
    `position` VARCHAR(255) NOT NULL,
    `performance_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: positions
CREATE TABLE `positions` (
    `position_id` INT(11) NOT NULL AUTO_INCREMENT,
    `position_name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`position_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: performances
CREATE TABLE `performances` (
    `performance_id` INT(11) NOT NULL AUTO_INCREMENT,
    `employee_id` INT(11) NOT NULL,
    `performance_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`performance_id`),
    FOREIGN KEY (`employee_id`) REFERENCES `employees`(`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: attendance
CREATE TABLE `attendance` (
    `attendance_id` INT(11) NOT NULL AUTO_INCREMENT,
    `employee_id` INT(11) NOT NULL,
    `date` DATE NOT NULL,
    `status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`attendance_id`),
    FOREIGN KEY (`employee_id`) REFERENCES `employees`(`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: salaries
CREATE TABLE `salaries` (
    `salary_id` INT(11) NOT NULL AUTO_INCREMENT,
    `employee_id` INT(11) NOT NULL,
    `salary` DECIMAL(10,2) NOT NULL,
    `date` DATE NOT NULL,
    PRIMARY KEY (`salary_id`),
    FOREIGN KEY (`employee_id`) REFERENCES `employees`(`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: projects
CREATE TABLE `projects` (
    `project_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_name` VARCHAR(255) NOT NULL,
    `responsible_person` VARCHAR(255) NOT NULL,
    `supplier_id` INT(11) NOT NULL,
    PRIMARY KEY (`project_id`),
    FOREIGN KEY (`supplier_id`) REFERENCES `suppliers`(`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: suppliers
CREATE TABLE `suppliers` (
    `supplier_id` INT(11) NOT NULL AUTO_INCREMENT,
    `supplier_name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`supplier_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: project_progress
CREATE TABLE `project_progress` (
    `progress_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_id` INT(11) NOT NULL,
    `progress_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`progress_id`),
    FOREIGN KEY (`project_id`) REFERENCES `projects`(`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
