-- Schema for database: `rschema_0135`

-- Table: `products`

CREATE TABLE `products` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `brand` varchar(255) NOT NULL,
  `category` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  `shelf_location` varchar(255) NOT NULL,
  PRIMARY KEY (`product_id`),
  UNIQUE KEY `product_id` (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `shelves`

CREATE TABLE `shelves` (
  `shelf_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`shelf_id`),
  UNIQUE KEY `shelf_id` (`shelf_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `product_shelf`

CREATE TABLE `product_shelf` (
  `product_id` int(11) NOT NULL,
  `shelf_id` int(11) NOT NULL,
  PRIMARY KEY (`product_id`,`shelf_id`),
  KEY `shelf_id` (`shelf_id`),
  CONSTRAINT `product_shelf_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `product_shelf_ibfk_2` FOREIGN KEY (`shelf_id`) REFERENCES `shelves` (`shelf_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
