-- Schema for database: rschema_0138

-- Table: employee
CREATE TABLE `employee` (
    `employee_number` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `performance` INT(11) NOT NULL,
    PRIMARY KEY (`employee_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: store
CREATE TABLE `store` (
    `store_number` INT(11) NOT NULL,
    `store_name` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`store_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: product
CREATE TABLE `product` (
    `product_number` INT(11) NOT NULL,
    `product_name` VARCHAR(255) NOT NULL,
    `specifications` VARCHAR(255) NOT NULL,
    `unit_price` DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: employment
CREATE TABLE `employment` (
    `employee_number` INT(11) NOT NULL,
    `employment_period` INT(11) NOT NULL,
    `monthly_salary` DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (`employee_number`),
    FOREIGN KEY (`employee_number`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: sales
CREATE TABLE `sales` (
    `product_number` INT(11) NOT NULL,
    `monthly_sales_volume` INT(11) NOT NULL,
    PRIMARY KEY (`product_number`),
    FOREIGN KEY (`product_number`) REFERENCES `product` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
