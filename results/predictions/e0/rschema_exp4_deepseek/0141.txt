-- Schema for database: `rschema_0141`

-- Table: `student_information`
CREATE TABLE `student_information` (
    `student_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `school` VARCHAR(255) NOT NULL,
    `department` VARCHAR(255) NOT NULL,
    `major` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `identification_information`
CREATE TABLE `identification_information` (
    `identification_number` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `age` INT(11) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`identification_number`),
    FOREIGN KEY (`identification_number`) REFERENCES `student_information` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
