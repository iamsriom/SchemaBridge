CREATE TABLE `coaches` (
  `CoachID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Age` int(11) NOT NULL,
  PRIMARY KEY (`CoachID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teams` (
  `TeamName` varchar(255) NOT NULL,
  `RepresentativeRegion` varchar(255) NOT NULL,
  `EstablishmentDate` datetime NOT NULL,
  PRIMARY KEY (`TeamName`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `players` (
  `PlayerID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Weight` float NOT NULL,
  `Height` float NOT NULL,
  `JoiningDate` datetime NOT NULL,
  PRIMARY KEY (`PlayerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `coach_team_relationships` (
  `CoachID` int(11) NOT NULL,
  `TeamName` varchar(255) NOT NULL,
  PRIMARY KEY (`CoachID`,`TeamName`),
  FOREIGN KEY (`CoachID`) REFERENCES `coaches` (`CoachID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`TeamName`) REFERENCES `teams` (`TeamName`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `player_team_relationships` (
  `PlayerID` int(11) NOT NULL,
  `TeamName` varchar(255) NOT NULL,
  PRIMARY KEY (`PlayerID`,`TeamName`),
  FOREIGN KEY (`PlayerID`) REFERENCES `players` (`PlayerID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`TeamName`) REFERENCES `teams` (`TeamName`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
