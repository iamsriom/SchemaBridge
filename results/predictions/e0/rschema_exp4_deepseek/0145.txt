-- Schema for database: rschema_0145

-- Table: procurement
CREATE TABLE `procurement` (
  `procurement_id` int(11) NOT NULL AUTO_INCREMENT,
  `drive_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `date_procured` datetime NOT NULL,
  PRIMARY KEY (`procurement_id`),
  FOREIGN KEY (`drive_id`) REFERENCES `drives`(`drive_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: warehouse
CREATE TABLE `warehouse` (
  `warehouse_id` int(11) NOT NULL AUTO_INCREMENT,
  `drive_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `date_received` datetime NOT NULL,
  PRIMARY KEY (`warehouse_id`),
  FOREIGN KEY (`drive_id`) REFERENCES `drives`(`drive_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: sales
CREATE TABLE `sales` (
  `sale_id` int(11) NOT NULL AUTO_INCREMENT,
  `drive_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `date_sold` datetime NOT NULL,
  PRIMARY KEY (`sale_id`),
  FOREIGN KEY (`drive_id`) REFERENCES `drives`(`drive_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`customer_id`) REFERENCES `customers`(`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: customers
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: drives
CREATE TABLE `drives` (
  `drive_id` int(11) NOT NULL AUTO_INCREMENT,
  `drive_name` varchar(255) NOT NULL,
  PRIMARY KEY (`drive_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
