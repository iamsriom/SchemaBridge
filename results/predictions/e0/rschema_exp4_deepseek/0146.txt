-- Schema for database: `rschema_0146`

-- Table: `dormitory_information`

CREATE TABLE `dormitory_information` (
  `dormitory_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_number` varchar(255) NOT NULL,
  `number_of_dormitory_members` int(11) NOT NULL,
  `affiliated_college` varchar(255) NOT NULL,
  `dormitory_type` varchar(255) NOT NULL,
  PRIMARY KEY (`dormitory_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_information`

CREATE TABLE `user_information` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `personal_profile` text NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `dormitory_member_management`

CREATE TABLE `dormitory_member_management` (
  `dormitory_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`dormitory_id`,`user_id`),
  FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory_information` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `user_information` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `information_summary_management`

CREATE TABLE `information_summary_management` (
  `dormitory_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `summary_data` text NOT NULL,
  PRIMARY KEY (`dormitory_id`,`user_id`),
  FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory_information` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `user_information` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
