-- Schema for database: rschema_0147

-- Table: class
CREATE TABLE `class` (
    `class_id` int(11) NOT NULL AUTO_INCREMENT,
    `class_name` varchar(255) NOT NULL,
    `department_name` varchar(255) NOT NULL,
    `grade` varchar(255) NOT NULL,
    PRIMARY KEY (`class_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: member
CREATE TABLE `member` (
    `member_id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `password` varchar(255) NOT NULL,
    `email` varchar(255) NOT NULL,
    `age` int(11) NOT NULL,
    `workplace` varchar(255) NOT NULL,
    `contact_number` varchar(255) NOT NULL,
    `class_id` int(11) NOT NULL,
    PRIMARY KEY (`member_id`),
    FOREIGN KEY (`class_id`) REFERENCES `class` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: announcement
CREATE TABLE `announcement` (
    `announcement_id` int(11) NOT NULL AUTO_INCREMENT,
    `announcement_time` datetime NOT NULL,
    `announcement_content` text NOT NULL,
    `class_id` int(11) NOT NULL,
    PRIMARY KEY (`announcement_id`),
    FOREIGN KEY (`class_id`) REFERENCES `class` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: message
CREATE TABLE `message` (
    `message_id` int(11) NOT NULL AUTO_INCREMENT,
    `sender_id` int(11) NOT NULL,
    `receiver_id` int(11) NOT NULL,
    `message_content` text NOT NULL,
    `send_time` datetime NOT NULL,
    `read_status` varchar(255) NOT NULL,
    PRIMARY KEY (`message_id`),
    FOREIGN KEY (`sender_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`receiver_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: log
CREATE TABLE `log` (
    `log_id` int(11) NOT NULL AUTO_INCREMENT,
    `log_content` text NOT NULL,
    `time` datetime NOT NULL,
    `mood` varchar(255) NOT NULL,
    `member_id` int(11) NOT NULL,
    PRIMARY KEY (`log_id`),
    FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: leave
CREATE TABLE `leave` (
    `leave_id` int(11) NOT NULL AUTO_INCREMENT,
    `leave_time` datetime NOT NULL,
    `leave_content` text NOT NULL,
    `member_id` int(11) NOT NULL,
    PRIMARY KEY (`leave_id`),
    FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: application
CREATE TABLE `application` (
    `application_id` int(11) NOT NULL AUTO_INCREMENT,
    `application_time` datetime NOT NULL,
    `approval_status` varchar(255) NOT NULL,
    `application_content` text NOT NULL,
    `member_id` int(11) NOT NULL,
    PRIMARY KEY (`application_id`),
    FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
