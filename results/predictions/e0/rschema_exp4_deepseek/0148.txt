-- Schema for database: `university_management_system`

-- Table: `students`

CREATE TABLE `students` (
    `StudentID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Gender` char(6) NOT NULL,
    `Major_id` int(11) NOT NULL,
    `College_id` int(11) NOT NULL,
    PRIMARY KEY (`StudentID`),
    FOREIGN KEY (`Major_id`) REFERENCES `majors` (`MajorID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`College_id`) REFERENCES `colleges` (`CollegeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `majors`

CREATE TABLE `majors` (
    `MajorID` int(11) NOT NULL,
    `MajorName` varchar(255) NOT NULL,
    `CollegeID` int(11) NOT NULL,
    PRIMARY KEY (`MajorID`),
    FOREIGN KEY (`CollegeID`) REFERENCES `colleges` (`CollegeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `teachers`

CREATE TABLE `teachers` (
    `TeacherID` int(11) NOT NULL,
    `TeacherName` varchar(255) NOT NULL,
    `Gender` char(6) NOT NULL,
    `Position` varchar(255) NOT NULL,
    `College_id` int(11) NOT NULL,
    `HireDate` datetime NOT NULL,
    `ContactInformation` varchar(255) NOT NULL,
    PRIMARY KEY (`TeacherID`),
    FOREIGN KEY (`College_id`) REFERENCES `colleges` (`CollegeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `courses`

CREATE TABLE `courses` (
    `CourseID` int(11) NOT NULL,
    `CourseName` varchar(255) NOT NULL,
    `Credit` int(11) NOT NULL,
    `ClassHours` int(11) NOT NULL,
    `Major_id` int(11) NOT NULL,
    `Teacher_id` int(11) NOT NULL,
    PRIMARY KEY (`CourseID`),
    FOREIGN KEY (`Major_id`) REFERENCES `majors` (`MajorID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`Teacher_id`) REFERENCES `teachers` (`TeacherID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `colleges`

CREATE TABLE `colleges` (
    `CollegeID` int(11) NOT NULL,
    `CollegeName` varchar(255) NOT NULL,
    PRIMARY KEY (`CollegeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
