-- Schema for database: rschema_0151

-- Table: customers
CREATE TABLE `customers` (
    `customer_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    `ID_number` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: savings_banks
CREATE TABLE `savings_banks` (
    `bank_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`bank_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: deposits_withdrawals
CREATE TABLE `deposits_withdrawals` (
    `transaction_id` INT(11) NOT NULL AUTO_INCREMENT,
    `customer_id` INT(11) NOT NULL,
    `bank_id` INT(11) NOT NULL,
    `amount` DECIMAL(10,2) NOT NULL,
    `expiration_date` DATE NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `interest_rate` DECIMAL(4,2) NOT NULL,
    `type` ENUM('deposit','withdrawal') NOT NULL,
    PRIMARY KEY (`transaction_id`),
    FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`bank_id`) REFERENCES `savings_banks` (`bank_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
