--Foreachtable:

CREATE TABLE `courses` (
  `courseID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text,
  `category` varchar(255),
  `teacherID` int(11),
  PRIMARY KEY (`courseID`),
  KEY `teacherID` (`teacherID`),
  CONSTRAINT `courses_ibfk_1` FOREIGN KEY (`teacherID`) REFERENCES `teachers` (`teacherID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teachers` (
  `teacherID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text,
  `income` decimal(10,2),
  PRIMARY KEY (`teacherID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `students` (
  `studentID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11),
  `balance` decimal(10,2),
  PRIMARY KEY (`studentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `learning_relationships` (
  `studentID` int(11) NOT NULL,
  `courseID` int(11) NOT NULL,
  PRIMARY KEY (`studentID`,`courseID`),
  KEY `courseID` (`courseID`),
  CONSTRAINT `learning_relationships_ibfk_1` FOREIGN KEY (`studentID`) REFERENCES `students` (`studentID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `learning_relationships_ibfk_2` FOREIGN KEY (`courseID`) REFERENCES `courses` (`courseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
