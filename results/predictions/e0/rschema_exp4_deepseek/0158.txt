-- Schema for database: `online_examination_system`

-- Table: `users`

CREATE TABLE `users` (
    `user_id` int(11) NOT NULL AUTO_INCREMENT,
    `real_name` varchar(255) NOT NULL,
    `contact_information` varchar(255) NOT NULL,
    `other_details` varchar(255) DEFAULT NULL,
    PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `roles`

CREATE TABLE `roles` (
    `role_id` int(11) NOT NULL AUTO_INCREMENT,
    `role_name` varchar(255) NOT NULL,
    PRIMARY KEY (`role_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_roles`

CREATE TABLE `user_roles` (
    `user_id` int(11) NOT NULL,
    `role_id` int(11) NOT NULL,
    PRIMARY KEY (`user_id`,`role_id`),
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`role_id`) REFERENCES `roles` (`role_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `questions`

CREATE TABLE `questions` (
    `question_id` int(11) NOT NULL AUTO_INCREMENT,
    `difficulty_level` int(11) NOT NULL,
    `content` text NOT NULL,
    `options` text NOT NULL,
    `correct_answer` varchar(255) NOT NULL,
    `explanation` text NOT NULL,
    `category` varchar(255) NOT NULL,
    PRIMARY KEY (`question_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `exams`

CREATE TABLE `exams` (
    `exam_id` int(11) NOT NULL AUTO_INCREMENT,
    `exam_name` varchar(255) NOT NULL,
    `description` text NOT NULL,
    `time_range` int(11) NOT NULL,
    `status` varchar(255) NOT NULL,
    PRIMARY KEY (`exam_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `exam_questions`

CREATE TABLE `exam_questions` (
    `exam_id` int(11) NOT NULL,
    `question_id` int(11) NOT NULL,
    `order` int(11) NOT NULL,
    PRIMARY KEY (`exam_id`,`question_id`),
    FOREIGN KEY (`exam_id`) REFERENCES `exams` (`exam_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`question_id`) REFERENCES `questions` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `exam_scores`

CREATE TABLE `exam_scores` (
    `exam_id` int(11) NOT NULL,
    `user_id` int(11) NOT NULL,
    `score` int(11) NOT NULL,
    `submission_time` datetime NOT NULL,
    PRIMARY KEY (`exam_id`,`user_id`),
    FOREIGN KEY (`exam_id`) REFERENCES `exams` (`exam_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_answers`

CREATE TABLE `user_answers` (
    `exam_id` int(11) NOT NULL,
    `user_id` int(11) NOT NULL,
    `question_id` int(11) NOT NULL,
    `answer` text NOT NULL,
    `is_correct` tinyint(1) NOT NULL,
    PRIMARY KEY (`exam_id`,`user_id`,`question_id`),
    FOREIGN KEY (`exam_id`) REFERENCES `exams` (`exam_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`question_id`) REFERENCES `questions` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
