-- Schema for database: `rschema_0162`

-- For table: `sensor`

CREATE TABLE `sensor` (
  `sensor_id` int(11) NOT NULL,
  `sensor_type` varchar(255) NOT NULL,
  `installation_location` varchar(255) NOT NULL,
  `collection_frequency` int(11) NOT NULL,
  PRIMARY KEY (`sensor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `monitoring_data`

CREATE TABLE `monitoring_data` (
  `data_id` int(11) NOT NULL AUTO_INCREMENT,
  `sensor_id` int(11) NOT NULL,
  `displacement` float NOT NULL,
  `wind_speed` float NOT NULL,
  `acceleration` float NOT NULL,
  `tower_head_tilt` float NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`data_id`),
  FOREIGN KEY (`sensor_id`) REFERENCES `sensor` (`sensor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `environmental_data`

CREATE TABLE `environmental_data` (
  `data_id` int(11) NOT NULL AUTO_INCREMENT,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`data_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `alarm`

CREATE TABLE `alarm` (
  `alarm_id` int(11) NOT NULL AUTO_INCREMENT,
  `sensor_id` int(11) NOT NULL,
  `alarm_type` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`alarm_id`),
  FOREIGN KEY (`sensor_id`) REFERENCES `sensor` (`sensor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `maintenance`

CREATE TABLE `maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `responsible_area` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `employee`

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `responsible_area` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
