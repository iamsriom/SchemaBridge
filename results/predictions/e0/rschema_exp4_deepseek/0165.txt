-- For table: `meteorological_station_information`

CREATE TABLE `meteorological_station_information` (
  `StationID` int(11) NOT NULL,
  `StationType` varchar(255) NOT NULL,
  `StationName` varchar(255) NOT NULL,
  `Latitude` decimal(10,8) NOT NULL,
  `Longitude` decimal(11,8) NOT NULL,
  `Elevation` int(11) NOT NULL,
  `GeographicEnvironmentDescription` text NOT NULL,
  `ObservationElementsList` text NOT NULL,
  `ObservationInstrumentNameandModel` varchar(255) NOT NULL,
  `ObservationTime` datetime NOT NULL,
  `ObservationRecordMedium` varchar(255) NOT NULL,
  `ObservationStandards` varchar(255) NOT NULL,
  PRIMARY KEY (`StationID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `historical_evolution_information`

CREATE TABLE `historical_evolution_information` (
  `ChangeID` int(11) NOT NULL AUTO_INCREMENT,
  `StationID` int(11) NOT NULL,
  `ChangeType` varchar(255) NOT NULL,
  `OldValue` text NOT NULL,
  `NewValue` text NOT NULL,
  `StartTime` datetime NOT NULL,
  `EndTime` datetime NOT NULL,
  `ReasonforChange` text NOT NULL,
  `OperatorID` int(11) NOT NULL,
  `OperationTime` datetime NOT NULL,
  `OperationStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`ChangeID`),
  KEY `StationID` (`StationID`),
  CONSTRAINT `historical_evolution_information_ibfk_1` FOREIGN KEY (`StationID`) REFERENCES `meteorological_station_information` (`StationID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `operator_information`

  `OperatorID` int(11) NOT NULL,

  `OperatorLevel` varchar(255) NOT NULL,
  `OperatorRole` varchar(255) NOT NULL,
  `OperatorEmail` varchar(255) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `audit_records`

CREATE TABLE `audit_records` (
  `ChangeID` int(11) NOT NULL,
  `OperatorID` int(11) NOT NULL,
  `AuditTime` datetime NOT NULL,
  `AuditStatus` varchar(255) NOT NULL,
  `AuditComments` text NOT NULL,
  PRIMARY KEY (`ChangeID`,`OperatorID`),
