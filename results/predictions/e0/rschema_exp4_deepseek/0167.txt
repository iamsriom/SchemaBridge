-- Schema for database: `rschema_0167`

-- Table: `natural_disaster_data`
CREATE TABLE `natural_disaster_data` (
    `DataID` int(11) NOT NULL AUTO_INCREMENT,
    `DataType` varchar(255) NOT NULL,
    `DataSource` varchar(255) NOT NULL,
    `UploadTime` datetime NOT NULL,
    `DataContent` text NOT NULL,
    `DataFormat` varchar(255) NOT NULL,
    `LayerType` varchar(255) NOT NULL,
    PRIMARY KEY (`DataID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `region`
CREATE TABLE `region` (
    `RegionID` int(11) NOT NULL AUTO_INCREMENT,
    `RegionName` varchar(255) NOT NULL,
    `AdministrativeDivision` varchar(255) NOT NULL,
    `Latitude` decimal(10, 8) NOT NULL,
    `Longitude` decimal(11, 8) NOT NULL,
    `Area` decimal(10, 2) NOT NULL,
    `DisasterRiskLevel` varchar(255) NOT NULL,
    PRIMARY KEY (`RegionID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department`
CREATE TABLE `department` (
    `DepartmentID` int(11) NOT NULL AUTO_INCREMENT,
    `DepartmentName` varchar(255) NOT NULL,
    `PersonInCharge` varchar(255) NOT NULL,
    `ContactInformation` varchar(255) NOT NULL,
    `FunctionalScope` text NOT NULL,
    PRIMARY KEY (`DepartmentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user`
CREATE TABLE `user` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Role` varchar(255) NOT NULL,
    `ContactInformation` varchar(255) NOT NULL,
    PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `region_department`
CREATE TABLE `region_department` (
    `RegionID` int(11) NOT NULL,
    `DepartmentID` int(11) NOT NULL,
    PRIMARY KEY (`RegionID`, `DepartmentID`),
    FOREIGN KEY (`RegionID`) REFERENCES `region` (`RegionID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`DepartmentID`) REFERENCES `department` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department_user`
CREATE TABLE `department_user` (
    `DepartmentID` int(11) NOT NULL,
    `UserID` int(11) NOT NULL,
    `PermissionLevel` varchar(255) NOT NULL,
    PRIMARY KEY (`DepartmentID`, `UserID`),
    FOREIGN KEY (`DepartmentID`) REFERENCES `department` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
