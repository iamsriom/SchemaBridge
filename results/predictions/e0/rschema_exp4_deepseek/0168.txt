-- Schema for database: `rschema_0168`

-- For table: `equipment`
CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `equipment_type` varchar(255) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `installation_date` datetime NOT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `operational_parameters`
CREATE TABLE `operational_parameters` (
  `parameter_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `time_stamp` datetime NOT NULL,
  `temperature` float NOT NULL,
  `vibration` float NOT NULL,
  PRIMARY KEY (`parameter_id`),
  FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `maintenance`
CREATE TABLE `maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_type` varchar(255) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_cost` float NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `historical_data`
CREATE TABLE `historical_data` (
  `data_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `time_stamp` datetime NOT NULL,
  `temperature` float NOT NULL,
  `vibration` float NOT NULL,
  `maintenance_frequency` int(11) NOT NULL,
  PRIMARY KEY (`data_id`),
  FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
