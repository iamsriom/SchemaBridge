CREATE TABLE `parking_spaces` (
    `parking_space_number` INT NOT NULL,
    `status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`parking_space_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `entrances` (
    `entrance_number` INT NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`entrance_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exits` (
    `exit_number` INT NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`exit_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
    `vehicle_id` INT NOT NULL AUTO_INCREMENT,
    `ticket_number` INT NOT NULL,
    `vehicle_model` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_records` (
    `parking_record_id` INT NOT NULL AUTO_INCREMENT,
    `vehicle_id` INT NOT NULL,
    `entry_time` DATETIME NOT NULL,
    `exit_time` DATETIME NOT NULL,
    `entrance_number` INT NOT NULL,
    `exit_number` INT NOT NULL,
    `parking_space_number` INT NOT NULL,
    PRIMARY KEY (`parking_record_id`),
    FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`entrance_number`) REFERENCES `entrances` (`entrance_number`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`exit_number`) REFERENCES `exits` (`exit_number`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`parking_space_number`) REFERENCES `parking_spaces` (`parking_space_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
