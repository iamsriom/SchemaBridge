-- Schema for database: `rschema_0174`

-- Table: `patients`

CREATE TABLE `patients` (
  `medical_record_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `address` varchar(255) NOT NULL,
  `ID_card_number` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `admission_time` datetime NOT NULL,
  PRIMARY KEY (`medical_record_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `vitals_signs`

CREATE TABLE `vitals_signs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `body_temperature` float NOT NULL,
  `blood_pressure` varchar(255) NOT NULL,
  `heart_rate` int(11) NOT NULL,
  `time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `vitals_signs_limits`

CREATE TABLE `vitals_signs_limits` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `vitals_sign_id` int(11) NOT NULL,
  `lower_limit` float NOT NULL,
  `upper_limit` float NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`vitals_sign_id`) REFERENCES `vitals_signs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `medical_records`

CREATE TABLE `medical_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `diagnosis_time` datetime NOT NULL,
  `diagnosis_content` text NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `treatment_recommendations`

CREATE TABLE `treatment_recommendations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `recommendation` text NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
