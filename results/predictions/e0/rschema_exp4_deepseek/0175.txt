-- Schema for database: `rschema_0175`

-- Table: `alumni`

CREATE TABLE `alumni` (
    `AlumniID` int(11) NOT NULL AUTO_INCREMENT,
    `Name` varchar(255) NOT NULL,
    `Employer` varchar(255),
    `JobTitle` varchar(255),
    `Department` varchar(255),
    `Class` varchar(255),
    `EnrollmentYear` int(11) NOT NULL,
    `IDNumber` char(6) NOT NULL,
    PRIMARY KEY (`AlumniID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `donations`

CREATE TABLE `donations` (
    `DonationID` int(11) NOT NULL AUTO_INCREMENT,
    `DonatingAlumni` int(11) NOT NULL,
    `DonationTime` datetime NOT NULL,
    `DonationAmount` decimal(10,2) NOT NULL,
    `DonationType` varchar(255) NOT NULL,
    `DonationUsage` varchar(255) NOT NULL,
    PRIMARY KEY (`DonationID`),
    FOREIGN KEY (`DonatingAlumni`) REFERENCES `alumni` (`AlumniID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `donation_projects`

CREATE TABLE `donation_projects` (
    `ProjectID` int(11) NOT NULL AUTO_INCREMENT,
    `ProjectName` varchar(255) NOT NULL,
    `StartTime` datetime NOT NULL,
    `EndTime` datetime NOT NULL,
    PRIMARY KEY (`ProjectID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `beneficiary_units`

CREATE TABLE `beneficiary_units` (
    `BeneficiaryID` int(11) NOT NULL AUTO_INCREMENT,
    `ProjectID` int(11) NOT NULL,
    `BenefitPercentage` decimal(5,2) NOT NULL,
    `BenefitAmount` decimal(10,2) NOT NULL,
    PRIMARY KEY (`BeneficiaryID`),
    FOREIGN KEY (`ProjectID`) REFERENCES `donation_projects` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
