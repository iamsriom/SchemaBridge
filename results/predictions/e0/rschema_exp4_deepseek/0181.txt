--Foreachtable:

CREATE TABLE `trucks` (
  `license_plate_number` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `load_capacity` int(11) NOT NULL,
  PRIMARY KEY (`license_plate_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tasks` (
  `task_number` int(11) NOT NULL AUTO_INCREMENT,
  `origin` varchar(255) NOT NULL,
  `destination` varchar(255) NOT NULL,
  `cargo_name` varchar(255) NOT NULL,
  `weight` int(11) NOT NULL,
  `transportation_time` datetime NOT NULL,
  `start_time` datetime DEFAULT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`task_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_name` varchar(255) NOT NULL,
  `employee_phone_number` varchar(255) NOT NULL,
  `employee_role` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_assignment` (
  `task_number` int(11) NOT NULL,
  `license_plate_number` varchar(255) NOT NULL,
  `employee_id` int(11) NOT NULL,
  PRIMARY KEY (`task_number`,`license_plate_number`,`employee_id`),
  FOREIGN KEY (`task_number`) REFERENCES `tasks` (`task_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`license_plate_number`) REFERENCES `trucks` (`license_plate_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
