-- Schema for database: rschema_0182

-- Table: markets
CREATE TABLE `markets` (
    `market_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `contact_information` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`market_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: products
CREATE TABLE `products` (
    `product_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `specification` VARCHAR(255) NOT NULL,
    `price` DECIMAL(10,2) NOT NULL,
    `current_inventory` INT(11) NOT NULL,
    PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: sales
CREATE TABLE `sales` (
    `market_id` INT(11) NOT NULL,
    `product_id` INT(11) NOT NULL,
    `sales_date` DATETIME NOT NULL,
    `sales_quantity` INT(11) NOT NULL,
    `sales_unit_price` DECIMAL(10,2) NOT NULL,
    `operator` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`market_id`, `product_id`),
    FOREIGN KEY (`market_id`) REFERENCES `markets` (`market_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
