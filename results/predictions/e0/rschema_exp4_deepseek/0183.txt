-- Schema for database: `fishing_tool_management`

-- For table: `tool`

CREATE TABLE `tool` (
  `tool_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  PRIMARY KEY (`tool_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `usage`

CREATE TABLE `usage` (
  `usage_id` int(11) NOT NULL AUTO_INCREMENT,
  `tool_id` int(11) NOT NULL,
  `usage_time` datetime NOT NULL,
  `user` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`usage_id`),
  FOREIGN KEY (`tool_id`) REFERENCES `tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `maintenance`

CREATE TABLE `maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `tool_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `content` varchar(255) NOT NULL,
  `performed_by` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`tool_id`) REFERENCES `tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `performance`

CREATE TABLE `performance` (
  `tool_id` int(11) NOT NULL,
  `usage_frequency` int(11) NOT NULL,
  `maintenance_count` int(11) NOT NULL,
  PRIMARY KEY (`tool_id`),
  FOREIGN KEY (`tool_id`) REFERENCES `tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
