#--Schemafordatabase:rschema_0184

CREATE TABLE `members` (
  `membership_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `registration_date` datetime NOT NULL,
  `points` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`membership_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bookings` (
  `booking_id` int(11) NOT NULL AUTO_INCREMENT,
  `membership_number` int(11) NOT NULL,
  `room_information` varchar(255) NOT NULL,
  `booking_status` varchar(255) NOT NULL,
  `point_changes` int(11) NOT NULL,
  PRIMARY KEY (`booking_id`),
  KEY `membership_number` (`membership_number`),
  CONSTRAINT `bookings_ibfk_1` FOREIGN KEY (`membership_number`) REFERENCES `members` (`membership_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
