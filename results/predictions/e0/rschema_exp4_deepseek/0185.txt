CREATE TABLE `proposals` (
  `ProposalID` int(11) NOT NULL,
  `ProposalContent` varchar(255) NOT NULL,
  `ProposalTime` datetime NOT NULL,
  `Status` varchar(255) NOT NULL,
  PRIMARY KEY (`ProposalID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `members` (
  `MemberID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `DateofBirth` date NOT NULL,
  `PartyAffiliation` varchar(255) NOT NULL,
  `Position` varchar(255) NOT NULL,
  PRIMARY KEY (`MemberID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `handling` (
  `HandlingID` int(11) NOT NULL AUTO_INCREMENT,
  `ProposalID` int(11) NOT NULL,
  `HandlingTime` datetime NOT NULL,
  `Handler` int(11) NOT NULL,
  `HandlingOpinion` varchar(255) NOT NULL,
  PRIMARY KEY (`HandlingID`),
  FOREIGN KEY (`ProposalID`) REFERENCES `proposals` (`ProposalID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`Handler`) REFERENCES `members` (`MemberID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `proposer` (
  `ProposalID` int(11) NOT NULL,
  `MemberID` int(11) NOT NULL,
  PRIMARY KEY (`ProposalID`,`MemberID`),
  FOREIGN KEY (`ProposalID`) REFERENCES `proposals` (`ProposalID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`MemberID`) REFERENCES `members` (`MemberID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
