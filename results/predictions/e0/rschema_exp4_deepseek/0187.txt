-- Schema for database: `rschema_0187`

-- Table: `farms`

CREATE TABLE `farms` (
  `farm_id` int(11) NOT NULL,
  `farm_owner_name` varchar(255) NOT NULL,
  `owner_type` enum('company','individual') NOT NULL,
  `owner_contact_information` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`farm_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `plots`

CREATE TABLE `plots` (
  `plot_id` int(11) NOT NULL,
  `farm_id` int(11) NOT NULL,
  `area` float NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `crops` varchar(255) NOT NULL,
  `current_status` enum('sowing','growing','harvesting') NOT NULL,
  PRIMARY KEY (`plot_id`),
  FOREIGN KEY (`farm_id`) REFERENCES `farms` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `crops`

CREATE TABLE `crops` (
  `crop_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `growth_cycle` varchar(255) NOT NULL,
  `suitable_soil_type` varchar(255) NOT NULL,
  `suitable_climate_conditions` varchar(255) NOT NULL,
  PRIMARY KEY (`crop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `planting_operations`

CREATE TABLE `planting_operations` (
  `plot_id` int(11) NOT NULL,
  `crop` varchar(255) NOT NULL,
  `planting_time` datetime NOT NULL,
  `expected_harvest_time` datetime NOT NULL,
  `planting_quantity` int(11) NOT NULL,
  `operator` varchar(255) NOT NULL,
  FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `harvesting_operations`

CREATE TABLE `harvesting_operations` (
  `plot_id` int(11) NOT NULL,
  `crop` varchar(255) NOT NULL,
  `harvest_time` datetime NOT NULL,
  `harvest_quantity` int(11) NOT NULL,
  `operator` varchar(255) NOT NULL,
  FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
