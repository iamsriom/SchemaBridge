-- Schema for database: `motor_vehicle_repair_system`

-- For table: `repair_stations`

CREATE TABLE `repair_stations` (
  `station_id` int(11) NOT NULL,
  `station_name` varchar(255) NOT NULL,
  PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `repair_technicians`

CREATE TABLE `repair_technicians` (
  `technician_id` int(11) NOT NULL,
  `technician_name` varchar(255) NOT NULL,
  `technician_contact_number` varchar(255) NOT NULL,
  `technician_skill_level` varchar(255) NOT NULL,
  `station_id` int(11) NOT NULL,
  PRIMARY KEY (`technician_id`),
  FOREIGN KEY (`station_id`) REFERENCES `repair_stations` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `repair_tasks`

CREATE TABLE `repair_tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_license_plate_number` varchar(255) NOT NULL,
  `vehicle_model` varchar(255) NOT NULL,
  `vehicle_owner_name` varchar(255) NOT NULL,
  `owner_contact_number` varchar(255) NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(255) NOT NULL,
  `repair_duration` int(11) NOT NULL,
  `repair_cost` decimal(10,2) NOT NULL,
  `repair_effect` varchar(255) NOT NULL,
  `owner_evaluation` varchar(255) NOT NULL,
  `technician_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`technician_id`) REFERENCES `repair_technicians` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
