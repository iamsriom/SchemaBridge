-- Schema for database: rschema_0190

-- Table: `repair_tasks`

CREATE TABLE `repair_tasks` (
  `TaskID` int(11) NOT NULL AUTO_INCREMENT,
  `CustomerName` varchar(255) NOT NULL,
  `CustomerPhoneNumber` varchar(255) NOT NULL,
  `ProductName` varchar(255) NOT NULL,
  `ModelNumber` varchar(255) NOT NULL,
  `FaultDescription` varchar(255) NOT NULL,
  `RepairDate` datetime NOT NULL,
  `RepairType` varchar(255) NOT NULL,
  `RepairDuration` int(11) NOT NULL,
  PRIMARY KEY (`TaskID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `repair_results`

CREATE TABLE `repair_results` (
  `TaskID` int(11) NOT NULL,
  `RepairCost` decimal(10,2) NOT NULL,
  `RepairResults` varchar(255) NOT NULL,
  `CustomerFeedback` varchar(255) NOT NULL,
  PRIMARY KEY (`TaskID`),
  FOREIGN KEY (`TaskID`) REFERENCES `repair_tasks` (`TaskID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `technician_performance`

CREATE TABLE `technician_performance` (
  `TechnicianID` int(11) NOT NULL AUTO_INCREMENT,
  `CompletedTasks` int(11) NOT NULL,
  `CustomerSatisfactionRating` int(11) NOT NULL,
  PRIMARY KEY (`TechnicianID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
