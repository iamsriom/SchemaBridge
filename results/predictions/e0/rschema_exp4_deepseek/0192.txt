-- Schema for database: `rschema_0192`

-- Table: `projects`

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `estimated_completion_date` datetime NOT NULL,
  `actual_completion_date` datetime DEFAULT NULL,
  `project_budget` decimal(10,2) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `departments`

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `department_head` int(11) NOT NULL,
  PRIMARY KEY (`department_id`),
  FOREIGN KEY (`department_head`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `employees`

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `project_employee`

CREATE TABLE `project_employee` (
  `project_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `participation_time` int(11) NOT NULL,
  `role` varchar(255) NOT NULL,
  `work_hours_needed` int(11) NOT NULL,
  PRIMARY KEY (`project_id`,`employee_id`),
  FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
