-- Schema for database: rschema_0193

-- Table: materials
CREATE TABLE `materials` (
  `material_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `specification` varchar(255) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  `inventory` int(11) NOT NULL,
  PRIMARY KEY (`material_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: projects
CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `material_number` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `date_of_use` datetime NOT NULL,
  `department` varchar(255) NOT NULL,
  `employee` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`),
  KEY `material_number` (`material_number`),
  CONSTRAINT `material_number` FOREIGN KEY (`material_number`) REFERENCES `materials` (`material_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
