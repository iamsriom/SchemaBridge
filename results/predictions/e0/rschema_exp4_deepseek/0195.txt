-- Schema for database: `real_estate_management_system`

-- Table: `properties`

CREATE TABLE `properties` (
    `PropertyID` int(11) NOT NULL,
    `address` varchar(255) NOT NULL,
    `area` int(11) NOT NULL,
    `sale_price` int(11) NOT NULL,
    `sale_status` varchar(255) NOT NULL,
    PRIMARY KEY (`PropertyID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `salespersonnel`

CREATE TABLE `salespersonnel` (
    `EmployeeID` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `contact_phone` varchar(255) NOT NULL,
    `commission_rate` decimal(10,2) NOT NULL,
    PRIMARY KEY (`EmployeeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `clients`

CREATE TABLE `clients` (
    `ClientID` int(11) NOT NULL,
    `name` varchar(255) NOT NULL,
    `contact_phone` varchar(255) NOT NULL,
    `address` varchar(255) NOT NULL,
    PRIMARY KEY (`ClientID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sales_transactions`

CREATE TABLE `sales_transactions` (
    `PropertyID` int(11) NOT NULL,
    `sale_date` datetime NOT NULL,
    `sale_price` int(11) NOT NULL,
    `EmployeeID` int(11) NOT NULL,
    `ClientID` int(11) NOT NULL,
    PRIMARY KEY (`PropertyID`),
    FOREIGN KEY (`PropertyID`) REFERENCES `properties` (`PropertyID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`EmployeeID`) REFERENCES `salespersonnel` (`EmployeeID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ClientID`) REFERENCES `clients` (`ClientID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
