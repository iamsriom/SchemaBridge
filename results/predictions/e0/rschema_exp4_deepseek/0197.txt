-- Schema for database: rschema_0197

-- Table: Project
CREATE TABLE `project` (
    `ProjectID` int(11) NOT NULL,
    `ProjectName` varchar(255) NOT NULL,
    `ProjectLocation` varchar(255) NOT NULL,
    `ProjectManagerID` int(11) NOT NULL,
    PRIMARY KEY (`ProjectID`),
    KEY `ProjectManagerID` (`ProjectManagerID`),
    CONSTRAINT `ProjectManagerID` FOREIGN KEY (`ProjectManagerID`) REFERENCES `project_manager` (`ProjectManagerID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Project_Manager
CREATE TABLE `project_manager` (
    `ProjectManagerID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    PRIMARY KEY (`ProjectManagerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Construction_Cost
CREATE TABLE `construction_cost` (
    `RecordID` int(11) NOT NULL AUTO_INCREMENT,
    `ExpenseType` varchar(255) NOT NULL,
    `CostAmount` decimal(10,2) NOT NULL,
    `RecordDate` date NOT NULL,
    `ProjectID` int(11) NOT NULL,
    `StaffID` int(11) NOT NULL,
    PRIMARY KEY (`RecordID`),
    KEY `ProjectID` (`ProjectID`),
    KEY `StaffID` (`StaffID`),
    CONSTRAINT `ProjectID` FOREIGN KEY (`ProjectID`) REFERENCES `project` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `StaffID` FOREIGN KEY (`StaffID`) REFERENCES `staff` (`StaffID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Staff
CREATE TABLE `staff` (
    `StaffID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    PRIMARY KEY (`StaffID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
