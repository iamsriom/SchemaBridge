-- Schema for database: `retailer`

-- For table: `store`

CREATE TABLE `store` (
  `store_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `manager` varchar(255) NOT NULL,
  PRIMARY KEY (`store_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `product`

CREATE TABLE `product` (
  `product_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `sales`

CREATE TABLE `sales` (
  `sale_time` datetime NOT NULL,
  `store_number` int(11) NOT NULL,
  `product_number` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `salesperson` varchar(255) NOT NULL,
  PRIMARY KEY (`sale_time`,`store_number`,`product_number`),
  FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`product_number`) REFERENCES `product` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `appraisal`

CREATE TABLE `appraisal` (
  `store_number` int(11) NOT NULL,
  `sales` int(11) NOT NULL,
  `customer_satisfaction` int(11) NOT NULL,
  PRIMARY KEY (`store_number`),
  FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
