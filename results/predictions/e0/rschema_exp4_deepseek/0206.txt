#--Schemafordatabase:rschema_0206

CREATE TABLE `areas` (
  `area_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `geographic_location` varchar(255) NOT NULL,
  `administrator` varchar(255) NOT NULL,
  PRIMARY KEY (`area_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `records` (
  `record_number` int(11) NOT NULL AUTO_INCREMENT,
  `area_number` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `recorder` varchar(255) NOT NULL,
  `species` varchar(255) NOT NULL,
  `number` int(11) NOT NULL,
  PRIMARY KEY (`record_number`),
  KEY `area_number` (`area_number`),
  CONSTRAINT `records_ibfk_1` FOREIGN KEY (`area_number`) REFERENCES `areas` (`area_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `environmental_changes` (
  `change_number` int(11) NOT NULL AUTO_INCREMENT,
  `area_number` int(11) NOT NULL,
  `change_time` datetime NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `rainfall` float NOT NULL,
  PRIMARY KEY (`change_number`),
  KEY `area_number` (`area_number`),
  CONSTRAINT `environmental_changes_ibfk_1` FOREIGN KEY (`area_number`) REFERENCES `areas` (`area_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
