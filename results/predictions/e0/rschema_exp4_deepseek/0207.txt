CREATE TABLE `facilities` (
  `FacilityID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Location` varchar(255) NOT NULL,
  `ResponsiblePerson` varchar(255) NOT NULL,
  `CreationDate` datetime NOT NULL,
  PRIMARY KEY (`FacilityID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_details` (
  `MaintenanceID` int(11) NOT NULL AUTO_INCREMENT,
  `FacilityID` int(11) NOT NULL,
  `MaintenanceTime` datetime NOT NULL,
  `MaintenanceContent` varchar(255) NOT NULL,
  `MaintenancePersonnel` varchar(255) NOT NULL,
  `MaintenanceCosts` decimal(10,2) NOT NULL,
  PRIMARY KEY (`MaintenanceID`),
  KEY `FacilityID` (`FacilityID`),
  CONSTRAINT `maintenance_details_ibfk_1` FOREIGN KEY (`FacilityID`) REFERENCES `facilities` (`FacilityID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `usage_details` (
  `UsageID` int(11) NOT NULL AUTO_INCREMENT,
  `FacilityID` int(11) NOT NULL,
  `UsageTime` datetime NOT NULL,
  `UsageFrequency` int(11) NOT NULL,
  `UserFeedback` varchar(255) NOT NULL,
  PRIMARY KEY (`UsageID`),
  KEY `FacilityID` (`FacilityID`),
  CONSTRAINT `usage_details_ibfk_1` FOREIGN KEY (`FacilityID`) REFERENCES `facilities` (`FacilityID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `UserID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_interactions` (
  `InteractionID` int(11) NOT NULL AUTO_INCREMENT,
  `UserID` int(11) NOT NULL,
  `FacilityID` int(11) NOT NULL,
  PRIMARY KEY (`InteractionID`),
  KEY `UserID` (`UserID`),
  KEY `FacilityID` (`FacilityID`),
  CONSTRAINT `user_interactions_ibfk_1` FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_interactions_ibfk_2` FOREIGN KEY (`FacilityID`) REFERENCES `facilities` (`FacilityID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
