-- Schema for database: rschema_0208

-- Table: equipment
CREATE TABLE `equipment` (
    `EquipmentID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Model` varchar(255) NOT NULL,
    `InstallationLocation` varchar(255) NOT NULL,
    `InstallationDate` datetime NOT NULL,
    PRIMARY KEY (`EquipmentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: operation
CREATE TABLE `operation` (
    `OperationID` int(11) NOT NULL AUTO_INCREMENT,
    `EquipmentID` int(11) NOT NULL,
    `StartTime` datetime NOT NULL,
    `EndTime` datetime NOT NULL,
    `OperationalStatus` varchar(255) NOT NULL,
    `OperatorID` int(11) NOT NULL,
    PRIMARY KEY (`OperationID`),
    FOREIGN KEY (`EquipmentID`) REFERENCES `equipment` (`EquipmentID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`OperatorID`) REFERENCES `operator` (`OperatorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: operator
CREATE TABLE `operator` (
    `OperatorID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Department` varchar(255) NOT NULL,
    `ContactInformation` varchar(255) NOT NULL,
    PRIMARY KEY (`OperatorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
