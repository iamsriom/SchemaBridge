-- Schema for database: rschema_0209
-- For each table:

CREATE TABLE `laboratory` (
  `laboratory_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`laboratory_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `equipment_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`equipment_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_records` (
  `maintenance_number` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_number` int(11) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_content` text NOT NULL,
  PRIMARY KEY (`maintenance_number`),
  KEY `equipment_number` (`equipment_number`),
  CONSTRAINT `maintenance_records_ibfk_1` FOREIGN KEY (`equipment_number`) REFERENCES `equipment` (`equipment_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
