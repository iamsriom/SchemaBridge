-- Schema for database: rschema_0210

-- Table: attendees
CREATE TABLE `attendees` (
  `attendee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `organization` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`attendee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: attendance
CREATE TABLE `attendance` (
  `attendance_id` int(11) NOT NULL AUTO_INCREMENT,
  `attendee_id` int(11) NOT NULL,
  `conference` varchar(255) NOT NULL,
  PRIMARY KEY (`attendance_id`),
  FOREIGN KEY (`attendee_id`) REFERENCES `attendees` (`attendee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: presentations
CREATE TABLE `presentations` (
  `presentation_id` int(11) NOT NULL AUTO_INCREMENT,
  `speaker_id` int(11) NOT NULL,
  `topic` varchar(255) NOT NULL,
  `time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`presentation_id`),
  FOREIGN KEY (`speaker_id`) REFERENCES `attendees` (`attendee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
