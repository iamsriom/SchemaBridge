-- Schema for database: rschema_0211

-- ConsultingServiceSystem

CREATE TABLE `projects` (
  `ProjectID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `StartDate` datetime NOT NULL,
  `EndDate` datetime NOT NULL,
  `ClientName` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultants` (
  `ConsultantID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `DateofBirth` date NOT NULL,
  `Gender` char(6) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `ProfessionalField` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consulting_services` (
  `ServiceID` int(11) NOT NULL AUTO_INCREMENT,
  `ProjectID` int(11) NOT NULL,
  `ConsultantID` int(11) NOT NULL,
  `ServiceDate` datetime NOT NULL,
  `ServiceContent` text NOT NULL,
  PRIMARY KEY (`ServiceID`),
  KEY `ProjectID` (`ProjectID`),
  KEY `ConsultantID` (`ConsultantID`),
  CONSTRAINT `consulting_services_ibfk_1` FOREIGN KEY (`ProjectID`) REFERENCES `projects` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `consulting_services_ibfk_2` FOREIGN KEY (`ConsultantID`) REFERENCES `consultants` (`ConsultantID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
