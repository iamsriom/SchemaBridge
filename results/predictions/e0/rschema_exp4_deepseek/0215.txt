-- Schema for database: rschema_0215

-- Table: mine
CREATE TABLE `mine` (
  `mine_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` int(11) NOT NULL,
  PRIMARY KEY (`mine_id`),
  KEY `person_in_charge` (`person_in_charge`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: material_administrator
CREATE TABLE `material_administrator` (
  `administrator_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`administrator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: material_purchase
CREATE TABLE `material_purchase` (
  `purchase_id` int(11) NOT NULL AUTO_INCREMENT,
  `purchase_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `supplier` varchar(255) NOT NULL,
  `purchase_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`purchase_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: material_distribution
CREATE TABLE `material_distribution` (
  `distribution_id` int(11) NOT NULL AUTO_INCREMENT,
  `distribution_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `distributor` int(11) NOT NULL,
  `recipient` int(11) NOT NULL,
  PRIMARY KEY (`distribution_id`),
  KEY `distributor` (`distributor`),
  CONSTRAINT `material_distribution_ibfk_1` FOREIGN KEY (`distributor`) REFERENCES `material_administrator` (`administrator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: mine_material_administrator
CREATE TABLE `mine_material_administrator` (
  `mine_id` int(11) NOT NULL,
  `administrator_id` int(11) NOT NULL,
  PRIMARY KEY (`mine_id`,`administrator_id`),
  KEY `administrator_id` (`administrator_id`),
  CONSTRAINT `mine_material_administrator_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `mine_material_administrator_ibfk_2` FOREIGN KEY (`administrator_id`) REFERENCES `material_administrator` (`administrator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
