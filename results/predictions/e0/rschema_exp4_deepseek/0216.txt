CREATE TABLE `card_types` (
    `CardTypeCode` CHAR(3) NOT NULL,
    `CardTypeName` VARCHAR(255) NOT NULL,
    `AnnualFee` DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (`CardTypeCode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
    `CustomerID` INT(11) NOT NULL AUTO_INCREMENT,
    `Name` VARCHAR(255) NOT NULL,
    `IDNumber` CHAR(10) NOT NULL,
    `ContactNumber` CHAR(10) NOT NULL,
    PRIMARY KEY (`CustomerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_cards` (
    `CardNumber` CHAR(16) NOT NULL,
    `CardTypeCode` CHAR(3) NOT NULL,
    `AccountOpeningDate` DATETIME NOT NULL,
    `CreditLimit` DECIMAL(10,2) NOT NULL,
    `CustomerID` INT(11) NOT NULL,
    PRIMARY KEY (`CardNumber`),
    FOREIGN KEY (`CardTypeCode`) REFERENCES `card_types` (`CardTypeCode`),
    FOREIGN KEY (`CustomerID`) REFERENCES `customers` (`CustomerID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
    `TransactionID` INT(11) NOT NULL AUTO_INCREMENT,
    `Amount` DECIMAL(10,2) NOT NULL,
    `TransactionTime` DATETIME NOT NULL,
    `TransactionLocation` VARCHAR(255) NOT NULL,
    `TransactionType` ENUM('debit','credit') NOT NULL,
    `CardNumber` CHAR(16) NOT NULL,
    PRIMARY KEY (`TransactionID`),
    FOREIGN KEY (`CardNumber`) REFERENCES `credit_cards` (`CardNumber`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repayments` (
    `RepaymentID` INT(11) NOT NULL AUTO_INCREMENT,
    `RepaymentAmount` DECIMAL(10,2) NOT NULL,
    `RepaymentTime` DATETIME NOT NULL,
    `CardNumber` CHAR(16) NOT NULL,
    PRIMARY KEY (`RepaymentID`),
    FOREIGN KEY (`CardNumber`) REFERENCES `credit_cards` (`CardNumber`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
