-- Schema for database: rschema_0219

-- Table: Card
CREATE TABLE `Card` (
    `CardID` int(11) NOT NULL AUTO_INCREMENT,
    `CardName` varchar(255) NOT NULL,
    `CardType` varchar(255) NOT NULL,
    `CardArtist` varchar(255) NOT NULL,
    `CardEffect` varchar(255) NOT NULL,
    `CardPower` int(11) NOT NULL,
    `CardIntelligence` int(11) NOT NULL,
    `CardEndurance` int(11) NOT NULL,
    PRIMARY KEY (`CardID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: CardCollection
CREATE TABLE `CardCollection` (
    `CollectionID` int(11) NOT NULL AUTO_INCREMENT,
    `CollectionName` varchar(255) NOT NULL,
    PRIMARY KEY (`CollectionID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: User
CREATE TABLE `User` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: CardCollectionAssociation
CREATE TABLE `CardCollectionAssociation` (
    `CardID` int(11) NOT NULL,
    `CollectionID` int(11) NOT NULL,
    `CollectionCollectionNumber` int(11) NOT NULL,
    `CardRarity` varchar(255) NOT NULL,
    PRIMARY KEY (`CardID`,`CollectionID`),
    FOREIGN KEY (`CardID`) REFERENCES `Card` (`CardID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`CollectionID`) REFERENCES `CardCollection` (`CollectionID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: UserExchangeCard
CREATE TABLE `UserExchangeCard` (
    `User1ID` int(11) NOT NULL,
    `User2ID` int(11) NOT NULL,
    `ExchangeTime` datetime NOT NULL,
    `ExchangedCardID` int(11) NOT NULL,
    FOREIGN KEY (`User1ID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`User2ID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ExchangedCardID`) REFERENCES `Card` (`CardID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
