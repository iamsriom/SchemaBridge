-- Schema for database: `recipe_management_system`

-- Table: `recipes`
CREATE TABLE `recipes` (
  `recipe_id` int(11) NOT NULL AUTO_INCREMENT,
  `recipe_name` varchar(255) NOT NULL,
  `recipe_description` text,
  `recipe_image` varchar(255),
  `recipe_type_id` int(11),
  PRIMARY KEY (`recipe_id`),
  FOREIGN KEY (`recipe_type_id`) REFERENCES `recipe_types`(`recipe_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `recipe_types`
CREATE TABLE `recipe_types` (
  `recipe_type_id` int(11) NOT NULL AUTO_INCREMENT,
  `recipe_type_name` varchar(255) NOT NULL,
  PRIMARY KEY (`recipe_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `ingredients`
CREATE TABLE `ingredients` (
  `ingredient_id` int(11) NOT NULL AUTO_INCREMENT,
  `ingredient_name` varchar(255) NOT NULL,
  `ingredient_description` text,
  `ingredient_image` varchar(255),
  PRIMARY KEY (`ingredient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `recipe_ingredients`
CREATE TABLE `recipe_ingredients` (
  `recipe_id` int(11) NOT NULL,
  `ingredient_id` int(11) NOT NULL,
  `quantity` varchar(255) NOT NULL,
  PRIMARY KEY (`recipe_id`,`ingredient_id`),
  FOREIGN KEY (`recipe_id`) REFERENCES `recipes`(`recipe_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ingredient_id`) REFERENCES `ingredients`(`ingredient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `preparation_steps`
CREATE TABLE `preparation_steps` (
  `step_id` int(11) NOT NULL AUTO_INCREMENT,
  `recipe_id` int(11) NOT NULL,
  `step_description` text NOT NULL,
  PRIMARY KEY (`step_id`),
  FOREIGN KEY (`recipe_id`) REFERENCES `recipes`(`recipe_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
