-- Schema for database: `rschema_0228`

-- Table: `Customer`

CREATE TABLE `Customer` (
  `CustomerID` int(11) NOT NULL,
  `CustomerName` varchar(255) NOT NULL,
  `CustomerAddress` varchar(255) NOT NULL,
  PRIMARY KEY (`CustomerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `Magazine`

CREATE TABLE `Magazine` (
  `MagazineName` varchar(255) NOT NULL,
  `Publisher` varchar(255) NOT NULL,
  `Theme` varchar(255) NOT NULL,
  `PublicationFrequency` varchar(255) NOT NULL,
  `Website` varchar(255) NOT NULL,
  PRIMARY KEY (`MagazineName`, `Publisher`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `Pricing`

CREATE TABLE `Pricing` (
  `PricingID` int(11) NOT NULL,
  `MagazineName` varchar(255) NOT NULL,
  `Publisher` varchar(255) NOT NULL,
  `SubscriptionPackage` varchar(255) NOT NULL,
  `SubscriptionPeriod` varchar(255) NOT NULL,
  `Price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`PricingID`),
  FOREIGN KEY (`MagazineName`, `Publisher`) REFERENCES `Magazine` (`MagazineName`, `Publisher`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `Customer_Subscribes_to_Magazine`

CREATE TABLE `Customer_Subscribes_to_Magazine` (
  `CustomerID` int(11) NOT NULL,
  `PricingID` int(11) NOT NULL,
  `CreationTime` datetime NOT NULL,
  `SubscriptionStartDate` date NOT NULL,
  PRIMARY KEY (`CustomerID`, `PricingID`),
  FOREIGN KEY (`CustomerID`) REFERENCES `Customer` (`CustomerID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`PricingID`) REFERENCES `Pricing` (`PricingID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
