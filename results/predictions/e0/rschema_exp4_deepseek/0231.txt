-- Schema for database: `rschema_0231`

-- For table: `users`
CREATE TABLE `users` (
  `UserID` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `contact_email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `experience_records`
CREATE TABLE `experience_records` (
  `ExperienceID` int(11) NOT NULL AUTO_INCREMENT,
  `description` text NOT NULL,
  `code_content` text NOT NULL,
  `creation_time` datetime NOT NULL,
  `modification_time` datetime NOT NULL,
  `status` char(1) NOT NULL DEFAULT 'F',
  PRIMARY KEY (`ExperienceID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `user_activity_logs`
CREATE TABLE `user_activity_logs` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `UserID` int(11) NOT NULL,
  `ExperienceID` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `modification_time` datetime NOT NULL,
  PRIMARY KEY (`log_id`),
  FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ExperienceID`) REFERENCES `experience_records` (`ExperienceID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
