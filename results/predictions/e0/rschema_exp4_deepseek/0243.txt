-- Schema for database: `library_management_system`

-- For table: `readers`

CREATE TABLE `readers` (
  `reader_number` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `registration_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `last_update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`reader_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `books`

CREATE TABLE `books` (
  `book_number` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `author` varchar(255) NOT NULL,
  `ISBN_number` varchar(255) NOT NULL,
  `publication_date` datetime NOT NULL,
  `current_inventory` int(11) NOT NULL,
  `book_classification` varchar(255) NOT NULL,
  `last_update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`book_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `borrowing`

CREATE TABLE `borrowing` (
  `borrowing_id` int(11) NOT NULL AUTO_INCREMENT,
  `reader_number` int(11) NOT NULL,
  `book_number` int(11) NOT NULL,
  `borrowing_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `return_date` datetime DEFAULT NULL,
  `overdue_fine` decimal(10,2) DEFAULT NULL,
  `renewal_count` int(11) NOT NULL DEFAULT 0,
  `last_update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`borrowing_id`),
  KEY `reader_number` (`reader_number`),
  KEY `book_number` (`book_number`),
  CONSTRAINT `borrowing_ibfk_1` FOREIGN KEY (`reader_number`) REFERENCES `readers` (`reader_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `borrowing_ibfk_2` FOREIGN KEY (`book_number`) REFERENCES `books` (`book_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
