-- Schema for database: rschema_0244 

-- Table: User
CREATE TABLE `User` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `AccountNumber` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: BusRoute
CREATE TABLE `BusRoute` (
    `RouteID` int(11) NOT NULL AUTO_INCREMENT,
    `RouteName` varchar(255) NOT NULL,
    PRIMARY KEY (`RouteID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Station
CREATE TABLE `Station` (
    `StationID` int(11) NOT NULL AUTO_INCREMENT,
    `StationName` varchar(255) NOT NULL,
    PRIMARY KEY (`StationID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: RouteStation
CREATE TABLE `RouteStation` (
    `RouteID` int(11) NOT NULL,
    `StationID` int(11) NOT NULL,
    `Sequence` int(11) NOT NULL,
    PRIMARY KEY (`RouteID`,`StationID`),
    FOREIGN KEY (`RouteID`) REFERENCES `BusRoute` (`RouteID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`StationID`) REFERENCES `Station` (`StationID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: UserQuestion
CREATE TABLE `UserQuestion` (
    `QuestionID` int(11) NOT NULL AUTO_INCREMENT,
    `UserID` int(11) NOT NULL,
    `QuestionContent` text NOT NULL,
    PRIMARY KEY (`QuestionID`),
    FOREIGN KEY (`UserID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Announcement
CREATE TABLE `Announcement` (
    `AnnouncementID` int(11) NOT NULL AUTO_INCREMENT,
    `AdminID` int(11) NOT NULL,
    `AnnouncementContent` text NOT NULL,
    PRIMARY KEY (`AnnouncementID`),
    FOREIGN KEY (`AdminID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: UserAnswer
CREATE TABLE `UserAnswer` (
    `AnswerID` int(11) NOT NULL AUTO_INCREMENT,
    `UserID` int(11) NOT NULL,
    `QuestionID` int(11) NOT NULL,
    `AnswerContent` text NOT NULL,
    PRIMARY KEY (`AnswerID`),
    FOREIGN KEY (`UserID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`QuestionID`) REFERENCES `UserQuestion` (`QuestionID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
