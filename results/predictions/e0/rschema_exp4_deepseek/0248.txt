-- Schema for database: rschema_0248

-- Table: users
CREATE TABLE `users` (
  `UserID` int(11) NOT NULL AUTO_INCREMENT,
  `Username` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `RegistrationTime` datetime NOT NULL,
  `LastLoginTime` datetime DEFAULT NULL,
  `Role` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: user_interactions
CREATE TABLE `user_interactions` (
  `UserID` int(11) NOT NULL,
  `ActivityID` int(11) NOT NULL,
  `InteractionType` varchar(255) NOT NULL,
  `InteractionTime` datetime NOT NULL,
  PRIMARY KEY (`UserID`,`ActivityID`),
  FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ActivityID`) REFERENCES `activities` (`ActivityID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: books
CREATE TABLE `books` (
  `BookID` int(11) NOT NULL AUTO_INCREMENT,
  `Title` varchar(255) NOT NULL,
  `Author` varchar(255) NOT NULL,
  `Description` text NOT NULL,
  PRIMARY KEY (`BookID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: articles
CREATE TABLE `articles` (
  `ArticleID` int(11) NOT NULL AUTO_INCREMENT,
  `Title` varchar(255) NOT NULL,
  `Content` text NOT NULL,
  `Author` varchar(255) NOT NULL,
  `PublishTime` datetime NOT NULL,
  PRIMARY KEY (`ArticleID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: activities
CREATE TABLE `activities` (
  `ActivityID` int(11) NOT NULL AUTO_INCREMENT,
  `ActivityName` varchar(255) NOT NULL,
  `ActivityTime` datetime NOT NULL,
  `Description` text NOT NULL,
  PRIMARY KEY (`ActivityID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: video_interviews
CREATE TABLE `video_interviews` (
  `VideoID` int(11) NOT NULL AUTO_INCREMENT,
  `Title` varchar(255) NOT NULL,
  `VideoLink` varchar(255) NOT NULL,
  `PublishTime` datetime NOT NULL,
  PRIMARY KEY (`VideoID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: lottery_activities
CREATE TABLE `lottery_activities` (
  `ActivityID` int(11) NOT NULL,
  `Winner` int(11) NOT NULL,
  `Award` varchar(255) NOT NULL,
  PRIMARY KEY (`ActivityID`),
  FOREIGN KEY (`ActivityID`) REFERENCES `activities` (`ActivityID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`Winner`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
