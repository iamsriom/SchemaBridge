-- Schema for database: `rschema_0251`

-- Table: `groomers`

CREATE TABLE `groomers` (
  `groomer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specialization` varchar(255) NOT NULL,
  `years_of_experience` int(11) NOT NULL,
  `rating` int(11) NOT NULL,
  PRIMARY KEY (`groomer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `customers`

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `pets`

CREATE TABLE `pets` (
  `pet_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `breed` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `weight` float NOT NULL,
  `characteristics` varchar(255) NOT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`pet_id`),
  FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `services`

CREATE TABLE `services` (
  `service_id` int(11) NOT NULL AUTO_INCREMENT,
  `service_items` varchar(255) NOT NULL,
  `service_duration` int(11) NOT NULL,
  `pricing` float NOT NULL,
  `groomer_id` int(11) NOT NULL,
  PRIMARY KEY (`service_id`),
  FOREIGN KEY (`groomer_id`) REFERENCES `groomers` (`groomer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `ratings`

CREATE TABLE `ratings` (
  `rating_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_rating` int(11) NOT NULL,
  `groomer_id` int(11) NOT NULL,
  PRIMARY KEY (`rating_id`),
  FOREIGN KEY (`groomer_id`) REFERENCES `groomers` (`groomer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
