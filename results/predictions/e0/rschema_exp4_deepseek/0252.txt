CREATE TABLE `player_management` (
  `Player_ID` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `game` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`Player_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `game_record_management` (
  `Game_ID` int(11) NOT NULL AUTO_INCREMENT,
  `game_rules` text NOT NULL,
  `game_content` text NOT NULL,
  `max_players` int(11) NOT NULL,
  PRIMARY KEY (`Game_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `single_player_management` (
  `Player_ID` int(11) NOT NULL,
  `Game_ID` int(11) NOT NULL,
  `game_start_time` datetime NOT NULL,
  `game_end_time` datetime NOT NULL,
  `game_status` varchar(255) NOT NULL,
  PRIMARY KEY (`Player_ID`,`Game_ID`),
  FOREIGN KEY (`Player_ID`) REFERENCES `player_management` (`Player_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`Game_ID`) REFERENCES `game_record_management` (`Game_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `player_communication_management` (
  `first_Player_ID` int(11) NOT NULL,
  `second_Player_ID` int(11) NOT NULL,
  `communication_start_time` datetime NOT NULL,
  `communication_end_time` datetime NOT NULL,
  `communication_content` text NOT NULL,
  PRIMARY KEY (`first_Player_ID`,`second_Player_ID`),
  FOREIGN KEY (`first_Player_ID`) REFERENCES `player_management` (`Player_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`second_Player_ID`) REFERENCES `player_management` (`Player_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `two_player_battle_game_management` (
  `first_Player_ID` int(11) NOT NULL,
  `second_Player_ID` int(11) NOT NULL,
  `Game_ID` int(11) NOT NULL,
  `battle_start_time` datetime NOT NULL,
  `battle_end_time` datetime NOT NULL,
  `battle_results` text NOT NULL,
  PRIMARY KEY (`first_Player_ID`,`second_Player_ID`,`Game_ID`),
  FOREIGN KEY (`first_Player_ID`) REFERENCES `player_management` (`Player_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`second_Player_ID`) REFERENCES `player_management` (`Player_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`Game_ID`) REFERENCES `game_record_management` (`Game_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
