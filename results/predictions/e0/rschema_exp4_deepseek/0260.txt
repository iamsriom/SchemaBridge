-- Schema for database: `rschema_0260`

-- Table: `students`

CREATE TABLE `students` (
    `StudentID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `DateofBirth` date NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    `HomeAddress` varchar(255) NOT NULL,
    `RegistrationEligibility` varchar(255) NOT NULL,
    PRIMARY KEY (`StudentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `courses`

CREATE TABLE `courses` (
    `CourseID` int(11) NOT NULL,
    `CourseName` varchar(255) NOT NULL,
    `Credits` int(11) NOT NULL,
    `Instructor` varchar(255) NOT NULL,
    `ClassTime` datetime NOT NULL,
    `Location` varchar(255) NOT NULL,
    PRIMARY KEY (`CourseID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `degree_examination_results`

CREATE TABLE `degree_examination_results` (
    `ResultID` int(11) NOT NULL,
    `StudentID` int(11) NOT NULL,
    `ExaminationName` varchar(255) NOT NULL,
    `Grade` varchar(255) NOT NULL,
    `ExaminationDate` date NOT NULL,
    PRIMARY KEY (`ResultID`),
    FOREIGN KEY (`StudentID`) REFERENCES `students` (`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `registrations`

CREATE TABLE `registrations` (
    `StudentID` int(11) NOT NULL,
    `CourseID` int(11) NOT NULL,
    `RegistrationTime` datetime NOT NULL,
    `Status` varchar(255) NOT NULL,
    PRIMARY KEY (`StudentID`,`CourseID`),
    FOREIGN KEY (`StudentID`) REFERENCES `students` (`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`CourseID`) REFERENCES `courses` (`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
