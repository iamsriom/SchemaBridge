-- Schema for database: `rschema_0264`

-- Table: `department`
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `manager_id` int(11) NOT NULL,
  PRIMARY KEY (`department_id`),
  UNIQUE KEY `manager_id` (`manager_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `employee`
CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`),
  FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `room`
CREATE TABLE `room` (
  `room_number` int(11) NOT NULL AUTO_INCREMENT,
  `room_type` varchar(255) NOT NULL,
  `charging_standard` decimal(10,2) NOT NULL,
  `check_in_status` varchar(255) NOT NULL,
  PRIMARY KEY (`room_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `customer`
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `unit_name` varchar(255) NOT NULL,
  `contact_person` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `contact_address` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `reservation`
CREATE TABLE `reservation` (
  `application_number` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `check_in_time` datetime NOT NULL,
  `number_of_days` int(11) NOT NULL,
  `room_type` varchar(255) NOT NULL,
  `number_of_rooms` int(11) NOT NULL,
  PRIMARY KEY (`application_number`),
  FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `arrangement`
CREATE TABLE `arrangement` (
  `room_number` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `ID_number` varchar(255) NOT NULL,
  `check_in_time` datetime NOT NULL,
  `number_of_days` int(11) NOT NULL,
  `telephone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`room_number`,`ID_number`,`check_in_time`),
  FOREIGN KEY (`room_number`) REFERENCES `room` (`room_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
