-- Schema for database: `rschema_0270`

-- Table: `programs`

CREATE TABLE `programs` (
  `program_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`program_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `production_staff`

CREATE TABLE `production_staff` (
  `staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `program_production`

CREATE TABLE `program_production` (
  `program_id` int(11) NOT NULL,
  `staff_id` int(11) NOT NULL,
  `production_time` int(11) NOT NULL,
  PRIMARY KEY (`program_id`,`staff_id`),
  FOREIGN KEY (`program_id`) REFERENCES `programs` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`staff_id`) REFERENCES `production_staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `reviews`

CREATE TABLE `reviews` (
  `review_id` int(11) NOT NULL AUTO_INCREMENT,
  `program_id` int(11) NOT NULL,
  `reviewer_id` int(11) NOT NULL,
  `review_result` varchar(255) NOT NULL,
  PRIMARY KEY (`review_id`),
  FOREIGN KEY (`program_id`) REFERENCES `programs` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`reviewer_id`) REFERENCES `production_staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `reviewers`

CREATE TABLE `reviewers` (
  `staff_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`staff_id`),
  FOREIGN KEY (`staff_id`) REFERENCES `production_staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
