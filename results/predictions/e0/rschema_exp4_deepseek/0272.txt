CREATE TABLE `ranch` (
  `ranchID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`ranchID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `livestock_type` (
  `livestockID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `species` varchar(255) NOT NULL,
  `ranchID` int(11) NOT NULL,
  PRIMARY KEY (`livestockID`),
  KEY `ranchID` (`ranchID`),
  CONSTRAINT `livestock_type_ibfk_1` FOREIGN KEY (`ranchID`) REFERENCES `ranch` (`ranchID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `livestock_health` (
  `healthstatusID` int(11) NOT NULL,
  `healthstatus` varchar(255) NOT NULL,
  `recordingtime` datetime NOT NULL,
  `livestockID` int(11) NOT NULL,
  PRIMARY KEY (`healthstatusID`),
  KEY `livestockID` (`livestockID`),
  CONSTRAINT `livestock_health_ibfk_1` FOREIGN KEY (`livestockID`) REFERENCES `livestock_type` (`livestockID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `livestock_vaccination` (
  `vaccinationstatusID` int(11) NOT NULL,
  `vaccinationmeasures` varchar(255) NOT NULL,
  `vaccinationtime` datetime NOT NULL,
  `livestockID` int(11) NOT NULL,
  PRIMARY KEY (`vaccinationstatusID`),
  KEY `livestockID` (`livestockID`),
  CONSTRAINT `livestock_vaccination_ibfk_1` FOREIGN KEY (`livestockID`) REFERENCES `livestock_type` (`livestockID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `livestock_feed_consumption` (
  `feedconsumptionID` int(11) NOT NULL,
  `consumptiontime` datetime NOT NULL,
  `feedtype` varchar(255) NOT NULL,
  `consumptionamount` int(11) NOT NULL,
  `livestockID` int(11) NOT NULL,
  PRIMARY KEY (`feedconsumptionID`),
  KEY `livestockID` (`livestockID`),
  CONSTRAINT `livestock_feed_consumption_ibfk_1` FOREIGN KEY (`livestockID`) REFERENCES `livestock_type` (`livestockID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `livestock_entry_exit` (
  `entryexitrecordID` int(11) NOT NULL,
  `time` datetime NOT NULL,
  `entryexittype` varchar(255) NOT NULL,
  `livestockID` int(11) NOT NULL,
  PRIMARY KEY (`entryexitrecordID`),
  KEY `livestockID` (`livestockID`),
  CONSTRAINT `livestock_entry_exit_ibfk_1` FOREIGN KEY (`livestockID`) REFERENCES `livestock_type` (`livestockID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
