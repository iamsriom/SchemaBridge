-- Schema for database: `rschema_0275`

-- Table: `ecosystems`

CREATE TABLE `ecosystems` (
  `SystemID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Location` varchar(255) NOT NULL,
  `Type` varchar(255) NOT NULL,
  `CurrentStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`SystemID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `monitoring_records`

CREATE TABLE `monitoring_records` (
  `SystemID` int(11) NOT NULL,
  `MonitoringDate` datetime NOT NULL,
  `Temperature` float NOT NULL,
  `Humidity` float NOT NULL,
  `SoilMoisture` float NOT NULL,
  `PlantGrowth` float NOT NULL,
  PRIMARY KEY (`SystemID`, `MonitoringDate`),
  FOREIGN KEY (`SystemID`) REFERENCES `ecosystems` (`SystemID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `operators`

CREATE TABLE `operators` (
  `EmployeeID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `ContactInformation` varchar(255) NOT NULL,
  `Position` varchar(255) NOT NULL,
  PRIMARY KEY (`EmployeeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `operator_records`

CREATE TABLE `operator_records` (
  `EmployeeID` int(11) NOT NULL,
  `MonitoringDate` datetime NOT NULL,
  PRIMARY KEY (`EmployeeID`, `MonitoringDate`),
  FOREIGN KEY (`EmployeeID`) REFERENCES `operators` (`EmployeeID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`MonitoringDate`) REFERENCES `monitoring_records` (`MonitoringDate`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
