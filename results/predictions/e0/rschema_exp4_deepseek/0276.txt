CREATE TABLE `maintenance_technician` (
    `technician_id` INT(11) NOT NULL AUTO_INCREMENT,
    `technician_name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    `position` VARCHAR(255) NOT NULL,
    `employment_date` DATETIME NOT NULL,
    `team_id` INT(11) NOT NULL,
    PRIMARY KEY (`technician_id`),
    FOREIGN KEY (`team_id`) REFERENCES `management_maintenance_team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `management_maintenance_team` (
    `team_id` INT(11) NOT NULL AUTO_INCREMENT,
    `team_name` VARCHAR(255) NOT NULL,
    `team_leader_id` INT(11) NOT NULL,
    `team_responsibilities` TEXT NOT NULL,
    `total_members` INT(11) NOT NULL,
    PRIMARY KEY (`team_id`),
    FOREIGN KEY (`team_leader_id`) REFERENCES `maintenance_technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `management_customer` (
    `customer_id` INT(11) NOT NULL AUTO_INCREMENT,
    `customer_name` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    `address` TEXT NOT NULL,
    PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `management_maintenance_task` (
    `task_id` INT(11) NOT NULL AUTO_INCREMENT,
    `maintenance_time` DATETIME NOT NULL,
    `maintenance_duration` INT(11) NOT NULL,
    `costs` DECIMAL(10,2) NOT NULL,
    `results` TEXT NOT NULL,
    `customer_feedback` TEXT NOT NULL,
    `product_model` VARCHAR(255) NOT NULL,
    `product_name` VARCHAR(255) NOT NULL,
    `customer_id` INT(11) NOT NULL,
    `technician_id` INT(11) NOT NULL,
    PRIMARY KEY (`task_id`),
    FOREIGN KEY (`customer_id`) REFERENCES `management_customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`technician_id`) REFERENCES `maintenance_technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
