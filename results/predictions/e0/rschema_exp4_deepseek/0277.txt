-- Schema for database: rschema_0277

-- Table: staff
CREATE TABLE `staff` (
  `staff_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `service_type` varchar(255) NOT NULL,
  `years_of_experience` int(11) NOT NULL,
  `professional_skills` varchar(255) NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: tasks
CREATE TABLE `tasks` (
  `task_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone` varchar(255) NOT NULL,
  `service_date` datetime NOT NULL,
  `service_address` varchar(255) NOT NULL,
  `service_type` varchar(255) NOT NULL,
  `service_duration` int(11) NOT NULL,
  PRIMARY KEY (`task_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: service_fee
CREATE TABLE `service_fee` (
  `task_id` int(11) NOT NULL,
  `service_fee` int(11) NOT NULL,
  `service_effectiveness` varchar(255) NOT NULL,
  `customer_evaluation` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`),
  FOREIGN KEY (`task_id`) REFERENCES `tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: performance
CREATE TABLE `performance` (
  `staff_id` int(11) NOT NULL,
  `completed_tasks` int(11) NOT NULL,
  `customer_satisfaction` varchar(255) NOT NULL,
  PRIMARY KEY (`staff_id`),
  FOREIGN KEY (`staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
