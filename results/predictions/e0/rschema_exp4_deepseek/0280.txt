-- Schema for database: `rschema_0280`

-- For table: `projects`
CREATE TABLE `projects` (
    `project_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_name` VARCHAR(255) NOT NULL,
    `project_type` VARCHAR(255) NOT NULL,
    `budget` DECIMAL(10,2) NOT NULL,
    `filming_time` DATETIME NOT NULL,
    PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `directors`
CREATE TABLE `directors` (
    `director_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `email` VARCHAR(255) NOT NULL,
    `phone_number` CHAR(15) NOT NULL,
    PRIMARY KEY (`director_id`),
    FOREIGN KEY (`director_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `actors`
CREATE TABLE `actors` (
    `actor_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `email` VARCHAR(255) NOT NULL,
    `phone_number` CHAR(15) NOT NULL,
    PRIMARY KEY (`actor_id`),
    FOREIGN KEY (`actor_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `producers`
CREATE TABLE `producers` (
    `producer_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `email` VARCHAR(255) NOT NULL,
    `phone_number` CHAR(15) NOT NULL,
    PRIMARY KEY (`producer_id`),
    FOREIGN KEY (`producer_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
