CREATE TABLE `exhibitions` (
  `exhibition_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`exhibition_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `curators` (
  `curator_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`curator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exhibits` (
  `exhibit_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `curator_id` int(11) NOT NULL,
  `exhibition_id` int(11) NOT NULL,
  PRIMARY KEY (`exhibit_id`),
  KEY `curator_id` (`curator_id`),
  KEY `exhibition_id` (`exhibition_id`),
  CONSTRAINT `exhibits_ibfk_1` FOREIGN KEY (`curator_id`) REFERENCES `curators` (`curator_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `exhibits_ibfk_2` FOREIGN KEY (`exhibition_id`) REFERENCES `exhibitions` (`exhibition_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
