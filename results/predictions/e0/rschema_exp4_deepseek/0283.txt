-- Schema for database: `book_management_system`

-- For table: `books`

CREATE TABLE `books` (
    `BookID` int(11) NOT NULL,
    `Title` varchar(255) NOT NULL,
    `ISBN` varchar(255) NOT NULL,
    `PublicationDate` datetime NOT NULL,
    `Category` varchar(255) NOT NULL,
    PRIMARY KEY (`BookID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `authors`

CREATE TABLE `authors` (
    `AuthorID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `Phone` varchar(255) NOT NULL,
    PRIMARY KEY (`AuthorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `editors`

CREATE TABLE `editors` (
    `EditorID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `Phone` varchar(255) NOT NULL,
    PRIMARY KEY (`EditorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `book_authors`

CREATE TABLE `book_authors` (
    `BookID` int(11) NOT NULL,
    `AuthorID` int(11) NOT NULL,
    PRIMARY KEY (`BookID`,`AuthorID`),
    FOREIGN KEY (`BookID`) REFERENCES `books` (`BookID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`AuthorID`) REFERENCES `authors` (`AuthorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `book_editors`

CREATE TABLE `book_editors` (
    `BookID` int(11) NOT NULL,
    `EditorID` int(11) NOT NULL,
    PRIMARY KEY (`BookID`,`EditorID`),
    FOREIGN KEY (`BookID`) REFERENCES `books` (`BookID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`EditorID`) REFERENCES `editors` (`EditorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
