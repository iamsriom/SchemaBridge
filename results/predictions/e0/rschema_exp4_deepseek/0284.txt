-- Schema for database: radio_station

-- For table: advertisements
CREATE TABLE `advertisements` (
  `advertisement_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `content` varchar(255) NOT NULL,
  PRIMARY KEY (`advertisement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: clients
CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: broadcast_time_slots
CREATE TABLE `broadcast_time_slots` (
  `time_slot_id` int(11) NOT NULL AUTO_INCREMENT,
  `broadcast_time` datetime NOT NULL,
  PRIMARY KEY (`time_slot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: advertisement_clients
CREATE TABLE `advertisement_clients` (
  `advertisement_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`advertisement_id`,`client_id`),
  FOREIGN KEY (`advertisement_id`) REFERENCES `advertisements` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: advertisement_time_slots
CREATE TABLE `advertisement_time_slots` (
  `advertisement_id` int(11) NOT NULL,
  `time_slot_id` int(11) NOT NULL,
  PRIMARY KEY (`advertisement_id`,`time_slot_id`),
  FOREIGN KEY (`advertisement_id`) REFERENCES `advertisements` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`time_slot_id`) REFERENCES `broadcast_time_slots` (`time_slot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
