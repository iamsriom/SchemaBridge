-- Schema for database: `rschema_0287`

-- Table: `parks`

CREATE TABLE `parks` (
  `park_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `area` int(11) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  PRIMARY KEY (`park_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `plants`

CREATE TABLE `plants` (
  `plant_id` int(11) NOT NULL AUTO_INCREMENT,
  `park_id` int(11) NOT NULL,
  `species` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `growth_status` varchar(255) NOT NULL,
  `recorded_time` datetime NOT NULL,
  `recorder` varchar(255) NOT NULL,
  PRIMARY KEY (`plant_id`),
  FOREIGN KEY (`park_id`) REFERENCES `parks` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maintenance`

CREATE TABLE `maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `park_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_costs` decimal(10,2) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  FOREIGN KEY (`park_id`) REFERENCES `parks` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
