CREATE TABLE `monitoring_sites` (
  `site_id` int(11) NOT NULL,
  `site_name` varchar(255) NOT NULL,
  `site_location` varchar(255) NOT NULL,
  `responsible_person_id` int(11) NOT NULL,
  PRIMARY KEY (`site_id`),
  KEY `responsible_person_id` (`responsible_person_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_personnel` (
  `person_id` int(11) NOT NULL,
  `person_name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `scope_of_monitoring` varchar(255) NOT NULL,
  PRIMARY KEY (`person_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `person_id` int(11) NOT NULL,
  `person_name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `scope_of_maintenance` varchar(255) NOT NULL,
  PRIMARY KEY (`person_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_sessions` (
  `session_id` int(11) NOT NULL AUTO_INCREMENT,
  `air_quality` varchar(255) NOT NULL,
  `water_quality` varchar(255) NOT NULL,
  `noise_level` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  `person_id` int(11) NOT NULL,
  `monitoring_result` varchar(255) NOT NULL,
  `site_id` int(11) NOT NULL,
  PRIMARY KEY (`session_id`),
  KEY `person_id` (`person_id`),
  KEY `site_id` (`site_id`),
  CONSTRAINT `monitoring_sessions_ibfk_1` FOREIGN KEY (`person_id`) REFERENCES `monitoring_personnel` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_sessions_ibfk_2` FOREIGN KEY (`site_id`) REFERENCES `monitoring_sites` (`site_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_activities` (
  `activity_id` int(11) NOT NULL AUTO_INCREMENT,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `person_id` int(11) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `site_id` int(11) NOT NULL,
  PRIMARY KEY (`activity_id`),
  KEY `person_id` (`person_id`),
  KEY `site_id` (`site_id`),
  CONSTRAINT `maintenance_activities_ibfk_1` FOREIGN KEY (`person_id`) REFERENCES `maintenance_personnel` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_activities_ibfk_2` FOREIGN KEY (`site_id`) REFERENCES `monitoring_sites` (`site_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
