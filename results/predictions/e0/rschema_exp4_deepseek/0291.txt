-- Schema for database: `rschema_0291`

-- Table: `reagents`

CREATE TABLE `reagents` (
    `ReagentID` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `specifications` varchar(255) NOT NULL,
    `inventory_quantity` int(11) NOT NULL,
    `expiration_date` datetime NOT NULL,
    PRIMARY KEY (`ReagentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `laboratories`

CREATE TABLE `laboratories` (
    `LaboratoryID` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `person_in_charge` varchar(255) NOT NULL,
    `location` varchar(255) NOT NULL,
    PRIMARY KEY (`LaboratoryID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `reagent_usage`

CREATE TABLE `reagent_usage` (
    `UsageID` int(11) NOT NULL AUTO_INCREMENT,
    `ReagentID` int(11) NOT NULL,
    `LaboratoryID` int(11) NOT NULL,
    `usage_date` datetime NOT NULL,
    `user` varchar(255) NOT NULL,
    `usage_quantity` int(11) NOT NULL,
    PRIMARY KEY (`UsageID`),
    FOREIGN KEY (`ReagentID`) REFERENCES `reagents` (`ReagentID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`LaboratoryID`) REFERENCES `laboratories` (`LaboratoryID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
