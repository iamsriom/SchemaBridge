-- Schema for database: rschema_0292

-- Table: facilities
CREATE TABLE `facilities` (
  `facility_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`facility_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: maintenance_events
CREATE TABLE `maintenance_events` (
  `event_id` int(11) NOT NULL,
  `facility_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `content` text NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`event_id`),
  FOREIGN KEY (`facility_id`) REFERENCES `facilities` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: personnel
CREATE TABLE `personnel` (
  `personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: event_personnel
CREATE TABLE `event_personnel` (
  `event_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`event_id`,`personnel_id`),
  FOREIGN KEY (`event_id`) REFERENCES `maintenance_events` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
