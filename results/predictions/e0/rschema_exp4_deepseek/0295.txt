CREATE TABLE `traders` (
  `trader_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `hiredate` datetime NOT NULL,
  `contact_number` char(15) NOT NULL,
  PRIMARY KEY (`trader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `ID_number` varchar(255) NOT NULL,
  `contact_number` char(15) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `futures_code` varchar(255) NOT NULL,
  `transaction_type` enum('buy','sell') NOT NULL,
  `transaction_quantity` int(11) NOT NULL,
  `transaction_price` decimal(10,2) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `trader_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  KEY `trader_id` (`trader_id`),
  KEY `client_id` (`client_id`),
  CONSTRAINT `transactions_ibfk_1` FOREIGN KEY (`trader_id`) REFERENCES `traders` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transactions_ibfk_2` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance` (
  `client_id` int(11) NOT NULL,
  `transaction_count` int(11) NOT NULL,
  `total_quantity` int(11) NOT NULL,
  `average_price` decimal(10,2) NOT NULL,
  `risk_level` enum('low','medium','high') NOT NULL,
  PRIMARY KEY (`client_id`),
  CONSTRAINT `performance_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
