CREATE TABLE `Financial_Advisor` (
  `AdvisorID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `HireDate` datetime NOT NULL,
  `ContactNumber` char(15) NOT NULL,
  PRIMARY KEY (`AdvisorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Client` (
  `ClientID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `IDNumber` char(15) NOT NULL,
  `ContactNumber` char(15) NOT NULL,
  `RiskPreference` varchar(255) NOT NULL,
  `TotalAssets` decimal(15,2) NOT NULL,
  PRIMARY KEY (`ClientID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Financial_Product` (
  `ProductCode` char(10) NOT NULL,
  `ProductName` varchar(255) NOT NULL,
  `ProductType` varchar(255) NOT NULL,
  `ProductValue` decimal(15,2) NOT NULL,
  PRIMARY KEY (`ProductCode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Investment_Portfolio` (
  `PortfolioID` int(11) NOT NULL,
  `ClientID` int(11) NOT NULL,
  `ProductCode` char(10) NOT NULL,
  `QuantityHeld` int(11) NOT NULL,
  PRIMARY KEY (`PortfolioID`),
  FOREIGN KEY (`ClientID`) REFERENCES `Client` (`ClientID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ProductCode`) REFERENCES `Financial_Product` (`ProductCode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Transaction_Record` (
  `RecordID` int(11) NOT NULL,
  `PortfolioID` int(11) NOT NULL,
  `ProductCode` char(10) NOT NULL,
  `TransactionType` varchar(255) NOT NULL,
  `TransactionQuantity` int(11) NOT NULL,
  `TransactionPrice` decimal(15,2) NOT NULL,
  `TransactionTime` datetime NOT NULL,
  PRIMARY KEY (`RecordID`),
  FOREIGN KEY (`PortfolioID`) REFERENCES `Investment_Portfolio` (`PortfolioID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ProductCode`) REFERENCES `Financial_Product` (`ProductCode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
