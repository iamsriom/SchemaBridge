-- Schema for database: `university_document_management`

-- For table: `document_type`

CREATE TABLE `document_type` (
  `document_type_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_type_name` varchar(255) NOT NULL,
  `document_type_version` int(11) NOT NULL,
  PRIMARY KEY (`document_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `status_category`

CREATE TABLE `status_category` (
  `status_category_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_type_id` int(11) NOT NULL,
  `status_category_name` varchar(255) NOT NULL,
  `status_category_version` int(11) NOT NULL,
  PRIMARY KEY (`status_category_id`),
  FOREIGN KEY (`document_type_id`) REFERENCES `document_type` (`document_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `status`

CREATE TABLE `status` (
  `status_id` int(11) NOT NULL AUTO_INCREMENT,
  `status_category_id` int(11) NOT NULL,
  `status_name` varchar(255) NOT NULL,
  `status_sequence` int(11) NOT NULL,
  PRIMARY KEY (`status_id`),
  FOREIGN KEY (`status_category_id`) REFERENCES `status_category` (`status_category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
