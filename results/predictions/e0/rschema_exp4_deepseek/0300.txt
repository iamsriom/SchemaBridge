-- Schema for database: rschema_0300

-- Table: material_supply_project

CREATE TABLE `material_supply_project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_number` varchar(255) NOT NULL,
  `material_code` varchar(255) NOT NULL,
  `description` text NOT NULL,
  `quantity` int(11) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `charge_amount` decimal(10,2) NOT NULL,
  `creation_date` datetime NOT NULL,
  `closing_date` datetime DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: bill

CREATE TABLE `bill` (
  `bill_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `material_quantity` int(11) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `charge_amount` decimal(10,2) NOT NULL,
  `creation_date` datetime NOT NULL,
  `modification_date` datetime DEFAULT NULL,
  `verification_status` varchar(255) NOT NULL DEFAULT 'pending',
  PRIMARY KEY (`bill_id`),
  KEY `project_id` (`project_id`),
  CONSTRAINT `bill_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `material_supply_project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
