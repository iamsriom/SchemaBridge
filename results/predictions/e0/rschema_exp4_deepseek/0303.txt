-- Schema for database: `rschema_0303`

-- Table: `projects`
CREATE TABLE `projects` (
    `project_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_name` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `stories`
CREATE TABLE `stories` (
    `story_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_id` INT(11) NOT NULL,
    `story_title` VARCHAR(255) NOT NULL,
    `story_description` TEXT,
    `story_status` VARCHAR(255) NOT NULL,
    `story_points` INT(11),
    PRIMARY KEY (`story_id`),
    FOREIGN KEY (`project_id`) REFERENCES `projects`(`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `tasks`
CREATE TABLE `tasks` (
    `task_id` INT(11) NOT NULL AUTO_INCREMENT,
    `story_id` INT(11) NOT NULL,
    `task_title` VARCHAR(255) NOT NULL,
    `task_description` TEXT,
    `task_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`task_id`),
    FOREIGN KEY (`story_id`) REFERENCES `stories`(`story_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sprints`
CREATE TABLE `sprints` (
    `sprint_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_id` INT(11) NOT NULL,
    `sprint_name` VARCHAR(255) NOT NULL,
    `sprint_start_time` DATETIME NOT NULL,
    `sprint_end_time` DATETIME NOT NULL,
    `sprint_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`sprint_id`),
    FOREIGN KEY (`project_id`) REFERENCES `projects`(`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `acceptance_criteria`
CREATE TABLE `acceptance_criteria` (
    `criteria_id` INT(11) NOT NULL AUTO_INCREMENT,
    `story_id` INT(11) NOT NULL,
    `criteria_description` TEXT NOT NULL,
    `criteria_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`criteria_id`),
    FOREIGN KEY (`story_id`) REFERENCES `stories`(`story_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `message_events`
CREATE TABLE `message_events` (
    `event_id` INT(11) NOT NULL AUTO_INCREMENT,
    `subject` VARCHAR(255) NOT NULL,
    `message_reference` VARCHAR(255),
    `version` VARCHAR(255),
    `creation_time` DATETIME NOT NULL,
    `update_time` DATETIME NOT NULL,
    PRIMARY KEY (`event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
