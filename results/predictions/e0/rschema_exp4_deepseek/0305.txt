CREATE TABLE `users` (
  `AccountID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `VerificationStatus` char(6) NOT NULL DEFAULT 'UNVERIFIED',
  `AccountStatus` char(6) NOT NULL DEFAULT 'ACTIVE',
  PRIMARY KEY (`AccountID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `follow_relationships` (
  `FollowerAccountID` int(11) NOT NULL,
  `FolloweeAccountID` int(11) NOT NULL,
  PRIMARY KEY (`FollowerAccountID`,`FolloweeAccountID`),
  FOREIGN KEY (`FollowerAccountID`) REFERENCES `users` (`AccountID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`FolloweeAccountID`) REFERENCES `users` (`AccountID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `posts` (
  `PostID` int(11) NOT NULL AUTO_INCREMENT,
  `PublisherAccountID` int(11) NOT NULL,
  `Content` text NOT NULL,
  `PublishTime` datetime NOT NULL,
  `AttachmentLink` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`PostID`),
  FOREIGN KEY (`PublisherAccountID`) REFERENCES `users` (`AccountID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notifications` (
  `NotificationID` int(11) NOT NULL AUTO_INCREMENT,
  `NotificationType` varchar(255) NOT NULL,
  `RecipientAccountID` int(11) NOT NULL,
  `RelatedPostorUser` varchar(255) NOT NULL,
  `NotificationTime` datetime NOT NULL,
  PRIMARY KEY (`NotificationID`),
  FOREIGN KEY (`RecipientAccountID`) REFERENCES `users` (`AccountID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
