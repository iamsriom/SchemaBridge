-- Schema for database: `rschema_0309`

-- Table: `customers`

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `marital_status` varchar(255) NOT NULL,
  `relevant_region` varchar(255) NOT NULL,
  `income_group` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `regions`

CREATE TABLE `regions` (
  `region_id` int(11) NOT NULL AUTO_INCREMENT,
  `region_name` varchar(255) NOT NULL,
  PRIMARY KEY (`region_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `income_groups`

CREATE TABLE `income_groups` (
  `income_group_id` int(11) NOT NULL AUTO_INCREMENT,
  `income_group_name` varchar(255) NOT NULL,
  PRIMARY KEY (`income_group_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `vehicle_segments`

CREATE TABLE `vehicle_segments` (
  `vehicle_segment_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_segment_name` varchar(255) NOT NULL,
  PRIMARY KEY (`vehicle_segment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `policies`

CREATE TABLE `policies` (
  `policy_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `vehicle_segment_id` int(11) NOT NULL,
  `insurance_type` varchar(255) NOT NULL,
  `premium` decimal(10,2) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `expiration_date` datetime NOT NULL,
  PRIMARY KEY (`policy_id`),
  FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`vehicle_segment_id`) REFERENCES `vehicle_segments` (`vehicle_segment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
