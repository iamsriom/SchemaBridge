-- Schema for database: rschema_0310

-- For table: Session
CREATE TABLE `Session` (
  `SessionID` int(11) NOT NULL,
  `CreationTime` datetime NOT NULL,
  `LastAccessTime` datetime NOT NULL,
  `MaximumInactivityInterval` int(11) NOT NULL,
  `ExpirationTime` datetime NOT NULL,
  `PrincipalName` varchar(255) NOT NULL,
  `AttributesValue` varchar(255) NOT NULL,
  `UserID` int(11) NOT NULL,
  PRIMARY KEY (`SessionID`),
  KEY `UserID` (`UserID`),
  CONSTRAINT `Session_ibfk_1` FOREIGN KEY (`UserID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: User
CREATE TABLE `User` (
  `UserID` int(11) NOT NULL,
  `Username` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `CreationTime` datetime NOT NULL,
  `LastAccessTime` datetime NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
