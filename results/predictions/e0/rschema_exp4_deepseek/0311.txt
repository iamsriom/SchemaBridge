CREATE TABLE `organizations` (
  `OrganizationID` int(11) NOT NULL,
  `OrganizationName` varchar(255) NOT NULL,
  PRIMARY KEY (`OrganizationID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processes` (
  `ProcessID` int(11) NOT NULL,
  `ParentProcessID` int(11) DEFAULT NULL,
  `OrganizationID` int(11) NOT NULL,
  `ProcessType` varchar(255) NOT NULL,
  `LogInformation` text,
  `ResourceID` int(11) NOT NULL,
  `Status` varchar(255) NOT NULL,
  `StartTime` datetime NOT NULL,
  `EndTime` datetime NOT NULL,
  PRIMARY KEY (`ProcessID`),
  KEY `OrganizationID` (`OrganizationID`),
  CONSTRAINT `processes_ibfk_1` FOREIGN KEY (`OrganizationID`) REFERENCES `organizations` (`OrganizationID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `events` (
  `EventID` int(11) NOT NULL,
  `ProcessID` int(11) NOT NULL,
  `EventType` varchar(255) NOT NULL,
  `LogInformation` text,
  `Status` varchar(255) NOT NULL,
  `Timestamp` datetime NOT NULL,
  PRIMARY KEY (`EventID`),
  KEY `ProcessID` (`ProcessID`),
  CONSTRAINT `events_ibfk_1` FOREIGN KEY (`ProcessID`) REFERENCES `processes` (`ProcessID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
