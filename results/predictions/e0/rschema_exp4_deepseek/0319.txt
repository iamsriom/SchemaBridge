-- Schema for database: `rschema_0319`

-- Table: `regions`

CREATE TABLE `regions` (
  `uniqueRegionalCode` int(11) NOT NULL,
  `CityCode` int(11) NOT NULL,
  `CenterCoordinates` varchar(255) NOT NULL,
  `RegionalLevel` int(11) NOT NULL,
  `MergedName` varchar(255) NOT NULL,
  `RegionalName` varchar(255) NOT NULL,
  `ParentRegionalID` int(11) DEFAULT NULL,
  `Pinyin` varchar(255) NOT NULL,
  `Abbreviation` varchar(255) NOT NULL,
  `PostalCode` varchar(255) NOT NULL,
  PRIMARY KEY (`uniqueRegionalCode`),
  KEY `ParentRegionalID` (`ParentRegionalID`),
  CONSTRAINT `regions_ibfk_1` FOREIGN KEY (`ParentRegionalID`) REFERENCES `regions` (`uniqueRegionalCode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
