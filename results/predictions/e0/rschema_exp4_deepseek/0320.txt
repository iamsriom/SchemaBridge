-- Schema for database: rschema_0320

-- Table: product
CREATE TABLE `product` (
  `ProductID` int(11) NOT NULL,
  `ProductName` varchar(255) NOT NULL,
  `ProductSlogan` varchar(255),
  `ProductDescription` text,
  `ProductCategory` varchar(255) NOT NULL,
  `DefaultPrice` decimal(10,2) NOT NULL,
  PRIMARY KEY (`ProductID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: style
CREATE TABLE `style` (
  `StyleID` int(11) NOT NULL,
  `AssociatedProductID` int(11) NOT NULL,
  `StyleName` varchar(255) NOT NULL,
  `SalePrice` decimal(10,2) NOT NULL,
  `OriginalPrice` decimal(10,2) NOT NULL,
  `IsDefaultStyle` char(6) NOT NULL,
  PRIMARY KEY (`StyleID`),
  FOREIGN KEY (`AssociatedProductID`) REFERENCES `product` (`ProductID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: feature
CREATE TABLE `feature` (
  `FeatureID` int(11) NOT NULL,
  `AssociatedProductID` int(11) NOT NULL,
  `FeatureName` varchar(255) NOT NULL,
  `FeatureValue` varchar(255) NOT NULL,
  PRIMARY KEY (`FeatureID`),
  FOREIGN KEY (`AssociatedProductID`) REFERENCES `product` (`ProductID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: sku
CREATE TABLE `sku` (
  `SKUID` int(11) NOT NULL,
  `AssociatedStyleID` int(11) NOT NULL,
  `Size` varchar(255) NOT NULL,
  `StockQuantity` int(11) NOT NULL,
  PRIMARY KEY (`SKUID`),
  FOREIGN KEY (`AssociatedStyleID`) REFERENCES `style` (`StyleID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: photo
CREATE TABLE `photo` (
  `PhotoID` int(11) NOT NULL,
  `AssociatedStyleID` int(11) NOT NULL,
  `PhotoURL` varchar(255) NOT NULL,
  `ThumbnailURL` varchar(255) NOT NULL,
  PRIMARY KEY (`PhotoID`),
  FOREIGN KEY (`AssociatedStyleID`) REFERENCES `style` (`StyleID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
