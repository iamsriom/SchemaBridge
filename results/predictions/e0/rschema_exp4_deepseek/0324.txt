-- Schema for database: `rschema_0324`

-- Table: `ip_addresses`

CREATE TABLE `ip_addresses` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `ip_address` varchar(255) NOT NULL,
    `is_anonymous_proxy` tinyint(1) NOT NULL DEFAULT 0,
    `is_satellite_provider` tinyint(1) NOT NULL DEFAULT 0,
    PRIMARY KEY (`id`),
    UNIQUE KEY `ip_address` (`ip_address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `geographic_locations`

CREATE TABLE `geographic_locations` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `ip_address_id` int(11) NOT NULL,
    `continent` varchar(255) NOT NULL,
    `country` varchar(255) NOT NULL,
    `iso_code` char(2) NOT NULL,
    `is_in_european_union` tinyint(1) NOT NULL DEFAULT 0,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`ip_address_id`) REFERENCES `ip_addresses` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
