-- Schema for database: `rschema_0326`

-- Table: `users`

CREATE TABLE `users` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `UserName` varchar(255) NOT NULL,
    `FullName` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `DateofBirth` date NOT NULL,
    `AvatarFileID` int(11),
    `AvatarFileName` varchar(255),
    PRIMARY KEY (`UserID`),
    UNIQUE KEY `UserName` (`UserName`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `tasks`

CREATE TABLE `tasks` (
    `TaskID` int(11) NOT NULL AUTO_INCREMENT,
    `TaskName` varchar(255) NOT NULL,
    `TaskStatus` varchar(255) NOT NULL,
    `TaskPriority` varchar(255) NOT NULL,
    `CreatorUserID` int(11) NOT NULL,
    `CreationDate` datetime NOT NULL,
    `AssigneeUserID` int(11),
    `CompletionDate` datetime,
    `TaskRemarks` text,
    PRIMARY KEY (`TaskID`),
    KEY `CreatorUserID` (`CreatorUserID`),
    CONSTRAINT `tasks_ibfk_1` FOREIGN KEY (`CreatorUserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `task_files`

CREATE TABLE `task_files` (
    `TaskFileID` int(11) NOT NULL AUTO_INCREMENT,
    `FileName` varchar(255) NOT NULL,
    `TaskID` int(11) NOT NULL,
    PRIMARY KEY (`TaskFileID`),
    KEY `TaskID` (`TaskID`),
    CONSTRAINT `task_files_ibfk_1` FOREIGN KEY (`TaskID`) REFERENCES `tasks` (`TaskID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
