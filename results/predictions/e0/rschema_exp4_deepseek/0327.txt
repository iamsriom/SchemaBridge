-- Schema for database: rschema_0327

-- Table: customers
CREATE TABLE `customers` (
    `phone_number` char(11) NOT NULL,
    `last_name` varchar(255) NOT NULL,
    `first_name` varchar(255) NOT NULL,
    `resource_ID` int(11) NOT NULL,
    `address_ID` int(11) NOT NULL,
    PRIMARY KEY (`phone_number`),
    KEY `resource_ID` (`resource_ID`),
    CONSTRAINT `resource_ID` FOREIGN KEY (`resource_ID`) REFERENCES `resources` (`resource_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: addresses
CREATE TABLE `addresses` (
    `address_ID` int(11) NOT NULL AUTO_INCREMENT,
    `country` varchar(255) NOT NULL,
    `city` varchar(255) NOT NULL,
    `postal_code` char(6) NOT NULL,
    `district` varchar(255) NOT NULL,
    `street_address` varchar(255) NOT NULL,
    `customer_phone_number` char(11) NOT NULL,
    PRIMARY KEY (`address_ID`),
    KEY `customer_phone_number` (`customer_phone_number`),
    CONSTRAINT `customer_phone_number` FOREIGN KEY (`customer_phone_number`) REFERENCES `customers` (`phone_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: resources
CREATE TABLE `resources` (
    `resource_ID` int(11) NOT NULL AUTO_INCREMENT,
    `resource_name` varchar(255) NOT NULL,
    `resource_type` varchar(255) NOT NULL,
    PRIMARY KEY (`resource_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: orders
CREATE TABLE `orders` (
    `order_ID` int(11) NOT NULL AUTO_INCREMENT,
    `amount` decimal(10,2) NOT NULL,
    `currency` varchar(255) NOT NULL,
    `resource_ID` int(11) NOT NULL,
    `start_time` datetime NOT NULL,
    `type` varchar(255) NOT NULL,
    `customer_phone_number` char(11) NOT NULL,
    PRIMARY KEY (`order_ID`),
    KEY `resource_ID` (`resource_ID`),
    KEY `customer_phone_number` (`customer_phone_number`),
    CONSTRAINT `resource_ID` FOREIGN KEY (`resource_ID`) REFERENCES `resources` (`resource_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `customer_phone_number` FOREIGN KEY (`customer_phone_number`) REFERENCES `customers` (`phone_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
