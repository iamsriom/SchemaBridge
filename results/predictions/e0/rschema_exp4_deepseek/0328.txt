-- Schema for database: `rschema_0328`

-- Table: `administrators`

CREATE TABLE `administrators` (
  `admin_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `account` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`admin_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `surveys`

CREATE TABLE `surveys` (
  `survey_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `description` text,
  `start_time` datetime,
  `end_time` datetime,
  `access_rule` varchar(255) NOT NULL,
  `status` varchar(255) NOT NULL,
  `logo_url` varchar(255),
  `background_image_url` varchar(255),
  `anonymity_setting` varchar(255),
  `admin_id` int(11) NOT NULL,
  PRIMARY KEY (`survey_id`),
  FOREIGN KEY (`admin_id`) REFERENCES `administrators` (`admin_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `questions`

CREATE TABLE `questions` (
  `question_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `description` text,
  `required_status` varchar(255) NOT NULL,
  `validation_type` varchar(255) NOT NULL,
  `display_order` int(11) NOT NULL,
  `display_style` varchar(255) NOT NULL,
  `evaluation_status` varchar(255) NOT NULL,
  `evaluation_score` int(11) NOT NULL,
  `survey_id` int(11) NOT NULL,
  PRIMARY KEY (`question_id`),
  FOREIGN KEY (`survey_id`) REFERENCES `surveys` (`survey_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `options`

CREATE TABLE `options` (
  `option_id` int(11) NOT NULL AUTO_INCREMENT,
  `content` text NOT NULL,
  `display_order` int(11) NOT NULL,
  `correct_answer_status` varchar(255) NOT NULL,
  `question_id` int(11) NOT NULL,
  PRIMARY KEY (`option_id`),
  FOREIGN KEY (`question_id`) REFERENCES `questions` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `answers`

CREATE TABLE `answers` (
  `answer_id` int(11) NOT NULL AUTO_INCREMENT,
  `answer_type` varchar(255) NOT NULL,
  `submission_time` datetime NOT NULL,
  `respondent_information` text,
  `option_id` int(11) NOT NULL,
  `question_id` int(11) NOT NULL,
  PRIMARY KEY (`answer_id`),
  FOREIGN KEY (`option_id`) REFERENCES `options` (`option_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`question_id`) REFERENCES `questions` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
