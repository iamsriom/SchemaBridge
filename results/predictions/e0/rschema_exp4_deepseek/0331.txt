-- Schema for database: rschema_0331

-- Table: users
CREATE TABLE `users` (
    `user_id` int(11) NOT NULL AUTO_INCREMENT,
    `username` varchar(255) NOT NULL,
    `contact_info` varchar(255) NOT NULL,
    `account_status` varchar(255) NOT NULL,
    PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: matches
CREATE TABLE `matches` (
    `match_id` int(11) NOT NULL AUTO_INCREMENT,
    `match_type` varchar(255) NOT NULL,
    `start_time` datetime NOT NULL,
    `end_time` datetime NOT NULL,
    `winner_id` int(11),
    PRIMARY KEY (`match_id`),
    FOREIGN KEY (`winner_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: match_participation
CREATE TABLE `match_participation` (
    `user_id` int(11) NOT NULL,
    `match_id` int(11) NOT NULL,
    PRIMARY KEY (`user_id`, `match_id`),
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`match_id`) REFERENCES `matches` (`match_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: match_results
CREATE TABLE `match_results` (
    `match_id` int(11) NOT NULL,
    `user_id` int(11) NOT NULL,
    `score` int(11) NOT NULL,
    `win_loss_status` varchar(255) NOT NULL,
    `rank` int(11) NOT NULL,
    PRIMARY KEY (`match_id`, `user_id`),
    FOREIGN KEY (`match_id`) REFERENCES `matches` (`match_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: battle_records
CREATE TABLE `battle_records` (
    `record_id` int(11) NOT NULL AUTO_INCREMENT,
    `match_id` int(11) NOT NULL,
    `side_a_id` int(11) NOT NULL,
    `side_b_id` int(11) NOT NULL,
    `start_time` datetime NOT NULL,
    `end_time` datetime NOT NULL,
    `win_loss_status` varchar(255) NOT NULL,
    PRIMARY KEY (`record_id`),
    FOREIGN KEY (`match_id`) REFERENCES `matches` (`match_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`side_a_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`side_b_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
