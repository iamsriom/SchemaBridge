-- Schema for database: `rschema_0332`

-- Table: `album`

CREATE TABLE `album` (
    `AlbumID` int(11) NOT NULL,
    `AlbumName` varchar(255) NOT NULL,
    `ReleaseDate` datetime NOT NULL,
    PRIMARY KEY (`AlbumID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `artist`

CREATE TABLE `artist` (
    `ArtistID` int(11) NOT NULL,
    `ArtistName` varchar(255) NOT NULL,
    PRIMARY KEY (`ArtistID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `song`

CREATE TABLE `song` (
    `SongID` int(11) NOT NULL,
    `SongName` varchar(255) NOT NULL,
    `VideoDuration` int(11) NOT NULL,
    `Popularity` int(11) NOT NULL,
    `VideoID` varchar(255) NOT NULL,
    `AlbumID` int(11) NOT NULL,
    PRIMARY KEY (`SongID`),
    FOREIGN KEY (`AlbumID`) REFERENCES `album` (`AlbumID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `playlist`

CREATE TABLE `playlist` (
    `PlaylistID` int(11) NOT NULL,
    `PlaylistName` varchar(255) NOT NULL,
    `PlayCount` int(11) NOT NULL,
    `CreatorUserID` int(11) NOT NULL,
    PRIMARY KEY (`PlaylistID`),
    FOREIGN KEY (`CreatorUserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user`

CREATE TABLE `user` (
    `UserID` int(11) NOT NULL,
    PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
