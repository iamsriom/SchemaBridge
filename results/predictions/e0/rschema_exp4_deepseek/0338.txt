-- Schema for database: `rschema_0338`

-- Table: `college`

CREATE TABLE `college` (
  `college_id` int(11) NOT NULL AUTO_INCREMENT,
  `college_name` varchar(255) NOT NULL,
  `college_location` varchar(255) NOT NULL,
  `total_staff` int(11) NOT NULL,
  `total_students` int(11) NOT NULL,
  PRIMARY KEY (`college_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `student`

CREATE TABLE `student` (
  `student_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `age` int(11) NOT NULL,
  `telephone` varchar(255) NOT NULL,
  `registration_time` datetime NOT NULL,
  `college_id` int(11) NOT NULL,
  PRIMARY KEY (`student_id`),
  FOREIGN KEY (`college_id`) REFERENCES `college` (`college_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `teacher`

CREATE TABLE `teacher` (
  `teacher_id` int(11) NOT NULL AUTO_INCREMENT,
  `gender` char(6) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  `college_id` int(11) NOT NULL,
  PRIMARY KEY (`teacher_id`),
  FOREIGN KEY (`college_id`) REFERENCES `college` (`college_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `course`

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `credit_number` int(11) NOT NULL,
  `teaching_teacher_id` int(11) NOT NULL,
  `course_summary` text NOT NULL,
  `class_hours` int(11) NOT NULL,
  `classroom` varchar(255) NOT NULL,
  PRIMARY KEY (`course_id`),
  FOREIGN KEY (`teaching_teacher_id`) REFERENCES `teacher` (`teacher_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `enrollment`

CREATE TABLE `enrollment` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `enrollment_time` datetime NOT NULL,
  `course_grade` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`, `course_id`),
  FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
