-- Schema for database: `rschema_0339`

-- Table: `projects`
CREATE TABLE `projects` (
    `project_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_name` VARCHAR(255) NOT NULL,
    `avatar_info` VARCHAR(255),
    PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `apis`
CREATE TABLE `apis` (
    `api_id` INT(11) NOT NULL AUTO_INCREMENT,
    `project_id` INT(11) NOT NULL,
    `api_name` VARCHAR(255) NOT NULL,
    `api_path` VARCHAR(255) NOT NULL,
    `creation_time` DATETIME NOT NULL,
    `update_time` DATETIME NOT NULL,
    PRIMARY KEY (`api_id`),
    FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `api_stats`
CREATE TABLE `api_stats` (
    `api_id` INT(11) NOT NULL,
    `call_frequency` INT(11) NOT NULL,
    `average_response_time` DECIMAL(10, 2) NOT NULL,
    `p90_response_time` DECIMAL(10, 2) NOT NULL,
    `stat_time` DATETIME NOT NULL,
    PRIMARY KEY (`api_id`, `stat_time`),
    FOREIGN KEY (`api_id`) REFERENCES `apis` (`api_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
