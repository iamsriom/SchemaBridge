-- Schema for database: rschema_0340

-- Table: applications
CREATE TABLE `applications` (
    `application_id` int(11) NOT NULL AUTO_INCREMENT,
    `application_name` varchar(255) NOT NULL,
    PRIMARY KEY (`application_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: languages
CREATE TABLE `languages` (
    `language_id` int(11) NOT NULL AUTO_INCREMENT,
    `iso_code` char(2) NOT NULL,
    `language_name` varchar(255) NOT NULL,
    PRIMARY KEY (`language_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: regions
CREATE TABLE `regions` (
    `region_id` int(11) NOT NULL AUTO_INCREMENT,
    `iso_code` char(2) NOT NULL,
    `region_name` varchar(255) NOT NULL,
    PRIMARY KEY (`region_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: resources
CREATE TABLE `resources` (
    `resource_id` int(11) NOT NULL AUTO_INCREMENT,
    `path` varchar(255) NOT NULL,
    `basename` varchar(255) NOT NULL,
    `language_id` int(11) NOT NULL,
    `region_id` int(11) NOT NULL,
    `version` varchar(255) NOT NULL,
    PRIMARY KEY (`resource_id`),
    FOREIGN KEY (`language_id`) REFERENCES `languages` (`language_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: resource_keys
CREATE TABLE `resource_keys` (
    `key_id` int(11) NOT NULL AUTO_INCREMENT,
    `key_name` varchar(255) NOT NULL,
    `key_version` varchar(255) NOT NULL,
    PRIMARY KEY (`key_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: resource_values
CREATE TABLE `resource_values` (
    `value_id` int(11) NOT NULL AUTO_INCREMENT,
    `value_content` text NOT NULL,
    `version` varchar(255) NOT NULL,
    PRIMARY KEY (`value_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: resource_key_value
CREATE TABLE `resource_key_value` (
    `resource_id` int(11) NOT NULL,
    `key_id` int(11) NOT NULL,
    `value_id` int(11) NOT NULL,
    PRIMARY KEY (`resource_id`, `key_id`, `value_id`),
    FOREIGN KEY (`resource_id`) REFERENCES `resources` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`key_id`) REFERENCES `resource_keys` (`key_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`value_id`) REFERENCES `resource_values` (`value_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: countries
CREATE TABLE `countries` (
    `country_id` int(11) NOT NULL AUTO_INCREMENT,
    `iso_code` char(2) NOT NULL,
    `country_name` varchar(255) NOT NULL,
    PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: country_region
CREATE TABLE `country_region` (
    `country_id` int(11) NOT NULL,
    `region_id` int(11) NOT NULL,
    PRIMARY KEY (`country_id`, `region_id`),
    FOREIGN KEY (`country_id`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: application_country
CREATE TABLE `application_country` (
    `application_id` int(11) NOT NULL,
    `country_id` int(11) NOT NULL,
    PRIMARY KEY (`application_id`, `country_id`),
    FOREIGN KEY (`application_id`) REFERENCES `applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`country_id`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
