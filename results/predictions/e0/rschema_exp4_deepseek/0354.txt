-- Schema for database: `ImageSending_Management_System`

-- For table: `images`
CREATE TABLE `images` (
  `Image_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Image_URL` varchar(255) NOT NULL,
  `image_description` varchar(255) NOT NULL,
  `compressed_signature` varchar(255) NOT NULL,
  PRIMARY KEY (`Image_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `users`
CREATE TABLE `users` (
  `User_ID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `image_sending_records`
CREATE TABLE `image_sending_records` (
  `sender` int(11) NOT NULL,
  `sending_time` datetime NOT NULL,
  `Image_ID` int(11) NOT NULL,
  PRIMARY KEY (`sender`,`sending_time`,`Image_ID`),
  FOREIGN KEY (`sender`) REFERENCES `users` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`Image_ID`) REFERENCES `images` (`Image_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `sending_link_management`
CREATE TABLE `sending_link_management` (
  `sending_link` varchar(255) NOT NULL,
  `sender` int(11) NOT NULL,
  `sending_time` datetime NOT NULL,
  `Image_ID` int(11) NOT NULL,
  PRIMARY KEY (`sending_link`),
  FOREIGN KEY (`sender`) REFERENCES `users` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`Image_ID`) REFERENCES `images` (`Image_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
