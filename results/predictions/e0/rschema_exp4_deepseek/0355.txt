-- Schema for database: `horse_racing_betting_system`

-- Table: `race`

CREATE TABLE `race` (
  `race_id` int(11) NOT NULL AUTO_INCREMENT,
  `race_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `result` varchar(255) NOT NULL,
  PRIMARY KEY (`race_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `horse`

CREATE TABLE `horse` (
  `horse_id` int(11) NOT NULL AUTO_INCREMENT,
  `horse_name` varchar(255) NOT NULL,
  PRIMARY KEY (`horse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `bettor`

CREATE TABLE `bettor` (
  `bettor_id` int(11) NOT NULL AUTO_INCREMENT,
  `phone_number` varchar(255) NOT NULL,
  `identity_info` varchar(255) NOT NULL,
  PRIMARY KEY (`bettor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `bet`

CREATE TABLE `bet` (
  `bet_id` int(11) NOT NULL AUTO_INCREMENT,
  `bettor_id` int(11) NOT NULL,
  `horse_id` int(11) NOT NULL,
  `race_id` int(11) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`bet_id`),
  FOREIGN KEY (`bettor_id`) REFERENCES `bettor` (`bettor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`horse_id`) REFERENCES `horse` (`horse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`race_id`) REFERENCES `race` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `bet_result`

CREATE TABLE `bet_result` (
  `bet_id` int(11) NOT NULL,
  `horse_number` int(11) NOT NULL,
  `rank` int(11) NOT NULL,
  PRIMARY KEY (`bet_id`),
  FOREIGN KEY (`bet_id`) REFERENCES `bet` (`bet_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
