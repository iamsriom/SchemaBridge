-- Schema for database: rschema_0360

-- Table: operators
CREATE TABLE `operators` (
  `operator_id` int(11) NOT NULL AUTO_INCREMENT,
  `account` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `permissions` varchar(255) NOT NULL,
  `user_role` varchar(255) NOT NULL,
  PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: logs
CREATE TABLE `logs` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `log_level` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `category` varchar(255) NOT NULL,
  `creation_timestamp` datetime NOT NULL,
  `modification_timestamp` datetime DEFAULT NULL,
  `deletion_timestamp` datetime DEFAULT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`log_id`),
  FOREIGN KEY (`operator_id`) REFERENCES `operators` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: log_operations
CREATE TABLE `log_operations` (
  `log_operation_id` int(11) NOT NULL AUTO_INCREMENT,
  `operation_time` datetime NOT NULL,
  `operation_type` varchar(255) NOT NULL,
  `execution_content` text NOT NULL,
  `log_id` int(11) NOT NULL,
  PRIMARY KEY (`log_operation_id`),
  FOREIGN KEY (`log_id`) REFERENCES `logs` (`log_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
