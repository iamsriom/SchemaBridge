-- Schema for database: `restaurant_recommendation_system`

-- For table: `restaurants`

CREATE TABLE `restaurants` (
  `restaurant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `business_hours` varchar(255) NOT NULL,
  `average_cost_per_person` int(11) NOT NULL,
  `latitude` decimal(10, 8) NOT NULL,
  `longitude` decimal(10, 8) NOT NULL,
  PRIMARY KEY (`restaurant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `restaurant_features`

CREATE TABLE `restaurant_features` (
  `restaurant_id` int(11) NOT NULL,
  `feature` varchar(255) NOT NULL,
  PRIMARY KEY (`restaurant_id`, `feature`),
  FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `reviews`

CREATE TABLE `reviews` (
  `review_id` int(11) NOT NULL AUTO_INCREMENT,
  `restaurant_id` int(11) NOT NULL,
  `review_content` text NOT NULL,
  `rating` int(11) NOT NULL,
  `dining_time` datetime NOT NULL,
  `review_image` varchar(255),
  PRIMARY KEY (`review_id`),
  FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `reply`

CREATE TABLE `reply` (
  `reply_id` int(11) NOT NULL AUTO_INCREMENT,
  `review_id` int(11) NOT NULL,
  `reply_content` text NOT NULL,
  `reply_time` datetime NOT NULL,
  PRIMARY KEY (`reply_id`),
  FOREIGN KEY (`review_id`) REFERENCES `reviews` (`review_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `dishes`

CREATE TABLE `dishes` (
  `dish_id` int(11) NOT NULL AUTO_INCREMENT,
  `restaurant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10, 2) NOT NULL,
  `description` text NOT NULL,
  `dish_image` varchar(255),
  `satisfaction_rate` int(11) NOT NULL,
  `monthly_sales` int(11) NOT NULL,
  PRIMARY KEY (`dish_id`),
  FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `dish_categories`

CREATE TABLE `dish_categories` (
  `dish_id` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  PRIMARY KEY (`dish_id`, `category`),
  FOREIGN KEY (`dish_id`) REFERENCES `dishes` (`dish_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
