--Foreachtable:

CREATE TABLE `user` (
    `UserName` varchar(255) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `PersonalBio` text,
    `RelationshipStatus` varchar(255),
    `PhoneNumber` varchar(255),
    `CreationTime` datetime NOT NULL,
    `LastEditTime` datetime NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `post` (
    `PostID` int(11) NOT NULL AUTO_INCREMENT,
    `PostContent` text NOT NULL,
    `PostTitle` varchar(255) NOT NULL,
    `UserName` varchar(255) NOT NULL,
    `CreationTime` datetime NOT NULL,
    `LastEditTime` datetime NOT NULL,
    PRIMARY KEY (`PostID`),
    KEY `UserName` (`UserName`),
    CONSTRAINT `post_ibfk_1` FOREIGN KEY (`UserName`) REFERENCES `user` (`UserName`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
    `CommentID` int(11) NOT NULL AUTO_INCREMENT,
    `PostID` int(11) NOT NULL,
    `UserName` varchar(255) NOT NULL,
    `CommentContent` text NOT NULL,
    `CreationTime` datetime NOT NULL,
    `LastEditTime` datetime NOT NULL,
    PRIMARY KEY (`CommentID`),
    KEY `PostID` (`PostID`),
    KEY `UserName` (`UserName`),
    CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`PostID`) REFERENCES `post` (`PostID`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`UserName`) REFERENCES `user` (`UserName`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
