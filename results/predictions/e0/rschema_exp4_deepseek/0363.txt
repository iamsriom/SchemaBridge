-- Schema for database: `rschema_0363`

-- Table: `books`

CREATE TABLE `books` (
  `ISBN` char(13) NOT NULL,
  `title` varchar(255) NOT NULL,
  `author` varchar(255) NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `publication_date` datetime NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `number_of_copies_available` int(11) NOT NULL,
  `number_of_copies_in_circulation` int(11) NOT NULL,
  `library_location` varchar(255) NOT NULL,
  `shelf_number` varchar(255) NOT NULL,
  PRIMARY KEY (`ISBN`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `readers`

CREATE TABLE `readers` (
  `reader_id` int(11) NOT NULL AUTO_INCREMENT,
  `library_card_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `phone` varchar(20) NOT NULL,
  `email` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `reader_type` varchar(255) NOT NULL,
  `maximum_borrowing_limit` int(11) NOT NULL,
  `loan_period` int(11) NOT NULL,
  PRIMARY KEY (`reader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `book_borrowing`

CREATE TABLE `book_borrowing` (
  `reader_id` int(11) NOT NULL,
  `ISBN` char(13) NOT NULL,
  `borrowing_time` datetime NOT NULL,
  `due_date` datetime NOT NULL,
  `actual_return_date` datetime,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`reader_id`, `ISBN`),
  FOREIGN KEY (`reader_id`) REFERENCES `readers` (`reader_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ISBN`) REFERENCES `books` (`ISBN`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `book_reservation`

CREATE TABLE `book_reservation` (
  `reader_id` int(11) NOT NULL,
  `ISBN` char(13) NOT NULL,
  `reservation_time` datetime NOT NULL,
  `reservation_expiration_time` datetime NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`reader_id`, `ISBN`),
  FOREIGN KEY (`reader_id`) REFERENCES `readers` (`reader_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ISBN`) REFERENCES `books` (`ISBN`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
