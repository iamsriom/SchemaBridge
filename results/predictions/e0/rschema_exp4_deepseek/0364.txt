-- Schema for database: rschema_0364

-- Table: companies
CREATE TABLE `companies` (
    `CompanyID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    PRIMARY KEY (`CompanyID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: users
CREATE TABLE `users` (
    `UserID` int(11) NOT NULL,
    `CompanyID` int(11) NOT NULL,
    `Username` varchar(255) NOT NULL,
    `PhoneNumber` varchar(255) NOT NULL,
    `JobTitle` varchar(255) NOT NULL,
    PRIMARY KEY (`UserID`),
    FOREIGN KEY (`CompanyID`) REFERENCES `companies` (`CompanyID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: login_response_types
CREATE TABLE `login_response_types` (
    `TypeID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    PRIMARY KEY (`TypeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: login_responses
CREATE TABLE `login_responses` (
    `UserID` int(11) NOT NULL,
    `LoginTime` datetime NOT NULL,
    `ResponseTypeID` int(11) NOT NULL,
    `ResponseLabel` varchar(255) NOT NULL,
    FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ResponseTypeID`) REFERENCES `login_response_types` (`TypeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
