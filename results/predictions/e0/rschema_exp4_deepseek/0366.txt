-- Schema for database: `rschema_0366`

-- Table: `engineers`

CREATE TABLE `engineers` (
    `engineer_id` INT NOT NULL AUTO_INCREMENT,
    `engineer_name` VARCHAR(255) NOT NULL,
    `status` VARCHAR(255) NOT NULL,
    `join_date` DATETIME NOT NULL,
    PRIMARY KEY (`engineer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `machines`

CREATE TABLE `machines` (
    `machine_id` INT NOT NULL AUTO_INCREMENT,
    `machine_name` VARCHAR(255) NOT NULL,
    `cost` DECIMAL(10,2) NOT NULL,
    `receiving_date` DATETIME NOT NULL,
    PRIMARY KEY (`machine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `engineer_machine`

CREATE TABLE `engineer_machine` (
    `engineer_id` INT NOT NULL,
    `machine_id` INT NOT NULL,
    PRIMARY KEY (`engineer_id`, `machine_id`),
    FOREIGN KEY (`engineer_id`) REFERENCES `engineers` (`engineer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`machine_id`) REFERENCES `machines` (`machine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `migration_history`

CREATE TABLE `migration_history` (
    `migration_id` INT NOT NULL AUTO_INCREMENT,
    `migration_date` DATETIME NOT NULL,
    `source_version` VARCHAR(255) NOT NULL,
    `target_version` VARCHAR(255) NOT NULL,
    `description` TEXT NOT NULL,
    PRIMARY KEY (`migration_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
