-- Schema for database: `rschema_0367`

-- Table: `users`

CREATE TABLE `users` (
    `user_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `email` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    `birth_date` DATE NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `insurance_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `appointments`

CREATE TABLE `appointments` (
    `appointment_id` INT(11) NOT NULL AUTO_INCREMENT,
    `user_id` INT(11) NOT NULL,
    `examination_description` TEXT NOT NULL,
    `examination_document` VARCHAR(255) NOT NULL,
    `medical_institution_id` INT(11) NOT NULL,
    `payment_amount` DECIMAL(10, 2) NOT NULL,
    `creation_date` DATETIME NOT NULL,
    `appointment_date` DATETIME NOT NULL,
    `status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`appointment_id`),
    FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`medical_institution_id`) REFERENCES `medical_institutions` (`medical_institution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `medical_institutions`

CREATE TABLE `medical_institutions` (
    `medical_institution_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    `opening_time` TIME NOT NULL,
    `closing_time` TIME NOT NULL,
    PRIMARY KEY (`medical_institution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `operators`

CREATE TABLE `operators` (
    `operator_id` INT(11) NOT NULL AUTO_INCREMENT,
    `username` VARCHAR(255) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `medical_institution_id` INT(11) NOT NULL,
    PRIMARY KEY (`operator_id`),
    FOREIGN KEY (`medical_institution_id`) REFERENCES `medical_institutions` (`medical_institution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
