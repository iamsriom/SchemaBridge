-- Schema for database: `user_management_system`

-- Table: `user`

CREATE TABLE `user` (
  `UserID` int(11) NOT NULL,
  `Username` varchar(255) NOT NULL,
  `UserEmail` varchar(255) NOT NULL,
  `EmailVerificationTime` datetime DEFAULT NULL,
  `UserAvatarURL` varchar(255) DEFAULT NULL,
  `UserCreationTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `UserUpdateTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `account`

CREATE TABLE `account` (
  `AccountID` int(11) NOT NULL AUTO_INCREMENT,
  `UserID` int(11) NOT NULL,
  `ThirdPartyServiceProviderType` varchar(255) NOT NULL,
  `ThirdPartyServiceProviderID` varchar(255) NOT NULL,
  `ThirdPartyServiceProviderAccountID` varchar(255) NOT NULL,
  `RefreshToken` varchar(255) NOT NULL,
  `AccessToken` varchar(255) NOT NULL,
  `AccessTokenExpirationTime` datetime NOT NULL,
  `AccountCreationTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `AccountUpdateTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`AccountID`),
  FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `session`

CREATE TABLE `session` (
  `SessionID` int(11) NOT NULL AUTO_INCREMENT,
  `UserID` int(11) NOT NULL,
  `SessionExpirationTime` datetime NOT NULL,
  `SessionToken` varchar(255) NOT NULL,
  `AccessToken` varchar(255) NOT NULL,
  `SessionCreationTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `SessionUpdateTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`SessionID`),
  FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `verification_request`

CREATE TABLE `verification_request` (
  `VerificationRequestID` int(11) NOT NULL AUTO_INCREMENT,
  `UserID` int(11) NOT NULL,
  `VerificationRequestIdentifier` varchar(255) NOT NULL,
  `VerificationToken` varchar(255) NOT NULL,
  `VerificationRequestExpirationTime` datetime NOT NULL,
  `VerificationRequestCreationTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `VerificationRequestUpdateTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`VerificationRequestID`),
  FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
