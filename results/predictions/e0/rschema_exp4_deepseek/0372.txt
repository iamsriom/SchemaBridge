-- Schema for database: `rschema_0372`

-- Table: `league`
CREATE TABLE `league` (
  `league_id` int(11) NOT NULL AUTO_INCREMENT,
  `league_name` varchar(255) NOT NULL,
  `region` varchar(255) NOT NULL,
  `parent_league` varchar(255),
  `person_in_charge` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`league_id`),
  KEY `region` (`region`),
  CONSTRAINT `FK_league_region` FOREIGN KEY (`region`) REFERENCES `region` (`region_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `region`
CREATE TABLE `region` (
  `region_name` varchar(255) NOT NULL,
  PRIMARY KEY (`region_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `league_hierarchy`
CREATE TABLE `league_hierarchy` (
  `league_id` int(11) NOT NULL,
  `parent_league_id` int(11),
  PRIMARY KEY (`league_id`),
  KEY `parent_league_id` (`parent_league_id`),
  CONSTRAINT `FK_league_hierarchy_league` FOREIGN KEY (`league_id`) REFERENCES `league` (`league_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `FK_league_hierarchy_parent_league` FOREIGN KEY (`parent_league_id`) REFERENCES `league` (`league_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user`
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `league_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`),
  KEY `league_id` (`league_id`),
  CONSTRAINT `FK_user_league` FOREIGN KEY (`league_id`) REFERENCES `league` (`league_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `league_version`
CREATE TABLE `league_version` (
  `league_id` int(11) NOT NULL,
  `version_number` int(11) NOT NULL,
  `league_name` varchar(255) NOT NULL,
  `region` varchar(255) NOT NULL,
  `parent_league` varchar(255),
  `person_in_charge` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`league_id`,`version_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
