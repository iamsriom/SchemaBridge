CREATE TABLE `device_management` (
  `DeviceID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Type` varchar(255) NOT NULL,
  `IPAddress` varchar(255) NOT NULL,
  `InstallationLocation` varchar(255) NOT NULL,
  `InstallationTime` datetime NOT NULL,
  `DeviceStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`DeviceID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_management` (
  `UserID` int(11) NOT NULL,
  `UserAccount` varchar(255) NOT NULL,
  `UserPassword` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `network_traffic_management` (
  `DeviceID` int(11) NOT NULL,
  `Timestamp` datetime NOT NULL,
  `UploadTraffic` int(11) NOT NULL,
  `DownloadTraffic` int(11) NOT NULL,
  `TotalTraffic` int(11) NOT NULL,
  `TrafficStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`DeviceID`, `Timestamp`),
  FOREIGN KEY (`DeviceID`) REFERENCES `device_management` (`DeviceID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_failure_record_management` (
  `DeviceID` int(11) NOT NULL,
  `FailureOccurrenceTime` datetime NOT NULL,
  `FailureType` varchar(255) NOT NULL,
  `FailureDescription` varchar(255) NOT NULL,
  `HandlingStatus` varchar(255) NOT NULL,
  `Remarks` varchar(255) NOT NULL,
  PRIMARY KEY (`DeviceID`, `FailureOccurrenceTime`),
  FOREIGN KEY (`DeviceID`) REFERENCES `device_management` (`DeviceID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff_management` (
  `StaffID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `ContactInformation` varchar(255) NOT NULL,
  PRIMARY KEY (`StaffID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_connection_device_management` (
  `UserID` int(11) NOT NULL,
  `DeviceID` int(11) NOT NULL,
  `ConnectionTime` datetime NOT NULL,
  `DisconnectionTime` datetime NOT NULL,
  `BandwidthUsed` int(11) NOT NULL,
  `UserIP` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`, `DeviceID`),
  FOREIGN KEY (`UserID`) REFERENCES `user_management` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`DeviceID`) REFERENCES `device_management` (`DeviceID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_maintenance_record_management` (
  `DeviceID` int(11) NOT NULL,
  `StaffID` int(11) NOT NULL,
  `MaintenanceTime` datetime NOT NULL,
  `MaintenanceContent` varchar(255) NOT NULL,
  `MaintenanceResult` varchar(255) NOT NULL,
  `Remarks` varchar(255) NOT NULL,
  PRIMARY KEY (`DeviceID`, `MaintenanceTime`),
  FOREIGN KEY (`DeviceID`) REFERENCES `device_management` (`DeviceID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`StaffID`) REFERENCES `staff_management` (`StaffID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
