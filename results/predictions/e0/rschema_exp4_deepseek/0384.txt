-- Schema for database: rschema_0384

-- Table: task
CREATE TABLE `task` (
    `TaskID` int(11) NOT NULL AUTO_INCREMENT,
    `TaskTitle` varchar(255) NOT NULL,
    `CreationDate` datetime NOT NULL,
    `TaskStatus` varchar(255) NOT NULL,
    `EndDate` datetime DEFAULT NULL,
    `StatusReason` varchar(255) DEFAULT NULL,
    `QueryStatement` text NOT NULL,
    `Database` varchar(255) NOT NULL,
    `ExecutableFile` varchar(255) NOT NULL,
    `Algorithm` varchar(255) NOT NULL,
    `Options` varchar(255) NOT NULL,
    `TaskOwner` varchar(255) NOT NULL,
    PRIMARY KEY (`TaskID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_parameter
CREATE TABLE `task_parameter` (
    `ParameterID` int(11) NOT NULL AUTO_INCREMENT,
    `TaskID` int(11) NOT NULL,
    `ParameterName` varchar(255) NOT NULL,
    `ParameterValue` text NOT NULL,
    PRIMARY KEY (`ParameterID`),
    FOREIGN KEY (`TaskID`) REFERENCES `task` (`TaskID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_related_organism
CREATE TABLE `task_related_organism` (
    `OrganismID` int(11) NOT NULL,
    `TaskID` int(11) NOT NULL,
    `OrganismName` varchar(255) NOT NULL,
    PRIMARY KEY (`OrganismID`,`TaskID`),
    FOREIGN KEY (`TaskID`) REFERENCES `task` (`TaskID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_excluded_organism
CREATE TABLE `task_excluded_organism` (
    `OrganismID` int(11) NOT NULL,
    `TaskID` int(11) NOT NULL,
    `OrganismName` varchar(255) NOT NULL,
    PRIMARY KEY (`OrganismID`,`TaskID`),
    FOREIGN KEY (`TaskID`) REFERENCES `task` (`TaskID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
