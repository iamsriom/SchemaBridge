-- Schema for database: rschema_0386

-- Table: User
CREATE TABLE `User` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `PublicKey` char(6) NOT NULL,
    `Address` varchar(255) NOT NULL,
    `CreationTime` datetime NOT NULL,
    PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Block
CREATE TABLE `Block` (
    `BlockID` int(11) NOT NULL AUTO_INCREMENT,
    `BlockHash` char(6) NOT NULL,
    `CreationTime` datetime NOT NULL,
    `BlockSize` int(11) NOT NULL,
    `PreviousBlockID` int(11) NOT NULL,
    PRIMARY KEY (`BlockID`),
    FOREIGN KEY (`PreviousBlockID`) REFERENCES `Block` (`BlockID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: TransactionRecord
CREATE TABLE `TransactionRecord` (
    `SenderID` int(11) NOT NULL,
    `ReceiverID` int(11) NOT NULL,
    `TransactionAmount` decimal(10, 2) NOT NULL,
    `TransactionTime` datetime NOT NULL,
    `BlockID` int(11) NOT NULL,
    `OperationType` varchar(255) NOT NULL,
    `OperationContent` text NOT NULL,
    FOREIGN KEY (`SenderID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ReceiverID`) REFERENCES `User` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`BlockID`) REFERENCES `Block` (`BlockID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
