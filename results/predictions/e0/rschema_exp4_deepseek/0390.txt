-- Schema for database: rschema_0390

-- For table: Consumer
CREATE TABLE `Consumer` (
    `consumerID` INT(11) NOT NULL AUTO_INCREMENT,
    `address` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`consumerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: CourierCompany
CREATE TABLE `CourierCompany` (
    `company_name` VARCHAR(255) NOT NULL,
    `arrival_time` DATETIME NOT NULL,
    `charging_price` DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (`company_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: CourierTracking
CREATE TABLE `CourierTracking` (
    `courier_number` INT(11) NOT NULL,
    `recipient_ID` INT(11) NOT NULL,
    `recipient_name` VARCHAR(255) NOT NULL,
    `recipient_phone_number` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`courier_number`),
    FOREIGN KEY (`recipient_ID`) REFERENCES `Consumer` (`consumerID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: Administrator
CREATE TABLE `Administrator` (
    `administratorID` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `login_password` VARCHAR(255) NOT NULL,
    `courier_username` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`administratorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: Partnership
CREATE TABLE `Partnership` (
    `company_name` VARCHAR(255) NOT NULL,
    `administratorID` INT(11) NOT NULL,
    PRIMARY KEY (`company_name`, `administratorID`),
    FOREIGN KEY (`company_name`) REFERENCES `CourierCompany` (`company_name`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`administratorID`) REFERENCES `Administrator` (`administratorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
