-- Schema for database: `hospital_management_system`

-- For table: `patient_information`
CREATE TABLE `patient_information` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `diagnosis_and_treatment`
CREATE TABLE `diagnosis_and_treatment` (
  `patient_id` int(11) NOT NULL,
  `diagnosis` varchar(255) NOT NULL,
  `treatment` varchar(255) NOT NULL,
  PRIMARY KEY (`patient_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patient_information` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `ward_information`
CREATE TABLE `ward_information` (
  `ward_id` int(11) NOT NULL AUTO_INCREMENT,
  `ward_type` varchar(255) NOT NULL,
  `ward_status` varchar(255) NOT NULL,
  PRIMARY KEY (`ward_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `registration_category`
CREATE TABLE `registration_category` (
  `registration_id` int(11) NOT NULL AUTO_INCREMENT,
  `registration_type` varchar(255) NOT NULL,
  PRIMARY KEY (`registration_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `drug_information`
CREATE TABLE `drug_information` (
  `drug_id` int(11) NOT NULL AUTO_INCREMENT,
  `drug_name` varchar(255) NOT NULL,
  `drug_description` varchar(255) NOT NULL,
  PRIMARY KEY (`drug_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `drug_list`
CREATE TABLE `drug_list` (
  `patient_id` int(11) NOT NULL,
  `drug_id` int(11) NOT NULL,
  PRIMARY KEY (`patient_id`, `drug_id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patient_information` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`drug_id`) REFERENCES `drug_information` (`drug_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
