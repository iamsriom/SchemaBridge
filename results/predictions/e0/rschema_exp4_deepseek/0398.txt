-- Schema for database: rschema_0398

-- Table: MenuManagement
CREATE TABLE `menu_management` (
  `DishID` int(11) NOT NULL,
  `DishName` varchar(255) NOT NULL,
  `DishCategory` varchar(255) NOT NULL,
  `DishPrice` decimal(10,2) NOT NULL,
  PRIMARY KEY (`DishID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: TableManagement
CREATE TABLE `table_management` (
  `TableID` int(11) NOT NULL,
  `SeatNumber` int(11) NOT NULL,
  PRIMARY KEY (`TableID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: CustomerManagement
CREATE TABLE `customer_management` (
  `CustomerID` int(11) NOT NULL,
  `CustomerName` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  PRIMARY KEY (`CustomerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: EmployeeManagement
CREATE TABLE `employee_management` (
  `EmployeeID` int(11) NOT NULL,
  `EmployeeName` varchar(255) NOT NULL,
  `Age` int(11) NOT NULL,
  `Gender` char(6) NOT NULL,
  `Salary` decimal(10,2) NOT NULL,
  PRIMARY KEY (`EmployeeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: OrderManagement
CREATE TABLE `order_management` (
  `OrderID` int(11) NOT NULL,
  `TableID` int(11) NOT NULL,
  `DiningTime` datetime NOT NULL,
  PRIMARY KEY (`OrderID`),
  FOREIGN KEY (`TableID`) REFERENCES `table_management` (`TableID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: ServiceManagement
CREATE TABLE `service_management` (
  `ServiceEmployeeID` int(11) NOT NULL,
  `OrderID` int(11) NOT NULL,
  PRIMARY KEY (`ServiceEmployeeID`,`OrderID`),
  FOREIGN KEY (`ServiceEmployeeID`) REFERENCES `employee_management` (`EmployeeID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`OrderID`) REFERENCES `order_management` (`OrderID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
