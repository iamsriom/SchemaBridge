-- Schema for database: `rschema`

-- Table: `users`

CREATE TABLE `users` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `RegistrationTime` datetime NOT NULL,
    `PersonalInformation` text,
    PRIMARY KEY (`UserID`),
    UNIQUE KEY `Username` (`Username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `codes`

CREATE TABLE `codes` (
    `CodeID` int(11) NOT NULL AUTO_INCREMENT,
    `CodeContent` text NOT NULL,
    `UploadTime` datetime NOT NULL,
    `ProblemID` int(11) NOT NULL,
    PRIMARY KEY (`CodeID`),
    FOREIGN KEY (`ProblemID`) REFERENCES `problems`(`ProblemID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `problems`

CREATE TABLE `problems` (
    `ProblemID` int(11) NOT NULL AUTO_INCREMENT,
    `ProblemName` varchar(255) NOT NULL,
    `ProblemDescription` text NOT NULL,
    `DifficultyLevel` int(11) NOT NULL,
    `CreationTime` datetime NOT NULL,
    PRIMARY KEY (`ProblemID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `projects`

CREATE TABLE `projects` (
    `ProjectID` int(11) NOT NULL AUTO_INCREMENT,
    `ProjectName` varchar(255) NOT NULL,
    `ProjectDescription` text NOT NULL,
    `CreationTime` datetime NOT NULL,
    PRIMARY KEY (`ProjectID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sign_in`

CREATE TABLE `sign_in` (
    `Sign-InID` int(11) NOT NULL AUTO_INCREMENT,
    `UserID` int(11) NOT NULL,
    `Sign-InTime` datetime NOT NULL,
    `Sign-InDate` date NOT NULL,
    PRIMARY KEY (`Sign-InID`),
    FOREIGN KEY (`UserID`) REFERENCES `users`(`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_codes`

CREATE TABLE `user_codes` (
    `UserID` int(11) NOT NULL,
    `CodeID` int(11) NOT NULL,
    PRIMARY KEY (`UserID`,`CodeID`),
    FOREIGN KEY (`UserID`) REFERENCES `users`(`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`CodeID`) REFERENCES `codes`(`CodeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_problems`

CREATE TABLE `user_problems` (
    `UserID` int(11) NOT NULL,
    `ProblemID` int(11) NOT NULL,
    PRIMARY KEY (`UserID`,`ProblemID`),
    FOREIGN KEY (`UserID`) REFERENCES `users`(`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ProblemID`) REFERENCES `problems`(`ProblemID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `user_projects`

CREATE TABLE `user_projects` (
    `UserID` int(11) NOT NULL,
    `ProjectID` int(11) NOT NULL,
    PRIMARY KEY (`UserID`,`ProjectID`),
    FOREIGN KEY (`UserID`) REFERENCES `users`(`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ProjectID`) REFERENCES `projects`(`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
