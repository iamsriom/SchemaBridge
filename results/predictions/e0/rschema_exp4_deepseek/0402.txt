-- Schema for database: rschema_0402

-- Table: doctors
CREATE TABLE `doctors` (
  `doctor_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `title` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `years_of_employment` int(11) NOT NULL,
  `work_permit` varchar(255) NOT NULL,
  PRIMARY KEY (`doctor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: departments
CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `personnel` int(11) NOT NULL,
  `department_number` varchar(255) NOT NULL,
  `department_phone_number` varchar(255) NOT NULL,
  `department_address` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: patients
CREATE TABLE `patients` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `diagnosis_description` text NOT NULL,
  `date_of_consultation` datetime NOT NULL,
  PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: diagnosis_and_treatment
CREATE TABLE `diagnosis_and_treatment` (
  `case_id` int(11) NOT NULL AUTO_INCREMENT,
  `diagnosis_and_treatment_content` text NOT NULL,
  PRIMARY KEY (`case_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: wards
CREATE TABLE `wards` (
  `ward_id` int(11) NOT NULL AUTO_INCREMENT,
  `ward_number` varchar(255) NOT NULL,
  `number_of_beds` int(11) NOT NULL,
  `ward_level` varchar(255) NOT NULL,
  PRIMARY KEY (`ward_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: registration_categories
CREATE TABLE `registration_categories` (
  `category_id` int(11) NOT NULL AUTO_INCREMENT,
  `category_number` varchar(255) NOT NULL,
  `category_name` varchar(255) NOT NULL,
  PRIMARY KEY (`category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: drugs
CREATE TABLE `drugs` (
  `drug_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  PRIMARY KEY (`drug_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: drug_lists
CREATE TABLE `drug_lists` (
  `drug_list_id` int(11) NOT NULL AUTO_INCREMENT,
  `drug_list_number` varchar(255) NOT NULL,
  `drug_information` text NOT NULL,
  PRIMARY KEY (`drug_list_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
