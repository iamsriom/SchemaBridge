-- Schema for database: rschema_0408

-- For Course table:
CREATE TABLE `courses` (
  `CourseID` int(11) NOT NULL,
  `CourseName` varchar(255) NOT NULL,
  PRIMARY KEY (`CourseID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For Chapter table:
CREATE TABLE `chapters` (
  `ChapterID` int(11) NOT NULL,
  `ChapterName` varchar(255) NOT NULL,
  `ChapterContent` text,
  `AssociatedCourseID` int(11) NOT NULL,
  PRIMARY KEY (`ChapterID`),
  FOREIGN KEY (`AssociatedCourseID`) REFERENCES `courses` (`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For QuestionType table:
CREATE TABLE `question_types` (
  `QuestionTypeID` int(11) NOT NULL,
  `QuestionTypeName` varchar(255) NOT NULL,
  PRIMARY KEY (`QuestionTypeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For Question table:
CREATE TABLE `questions` (
  `QuestionID` int(11) NOT NULL,
  `QuestionContent` text NOT NULL,
  `QuestionTypeID` int(11) NOT NULL,
  `ChapterID` int(11) NOT NULL,
  `CourseID` int(11) NOT NULL,
  `CorrectAnswer` text NOT NULL,
  `AlternativeAnswers` text,
  PRIMARY KEY (`QuestionID`),
  FOREIGN KEY (`QuestionTypeID`) REFERENCES `question_types` (`QuestionTypeID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ChapterID`) REFERENCES `chapters` (`ChapterID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`CourseID`) REFERENCES `courses` (`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
