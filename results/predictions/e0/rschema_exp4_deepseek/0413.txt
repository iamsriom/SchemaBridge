-- Schema for database: `rschema_0413`

-- Table: `employee_information`

CREATE TABLE `employee_information` (
  `EmployeeID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `Position` varchar(255) NOT NULL,
  `Department` int(11) NOT NULL,
  `ContactNumber` varchar(255) NOT NULL,
  `HireDate` datetime NOT NULL,
  PRIMARY KEY (`EmployeeID`),
  FOREIGN KEY (`Department`) REFERENCES `department_management` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `department_management`

CREATE TABLE `department_management` (
  `DepartmentID` int(11) NOT NULL,
  `DepartmentName` varchar(255) NOT NULL,
  `DepartmentManager` varchar(255) NOT NULL,
  `DepartmentPhone` varchar(255) NOT NULL,
  PRIMARY KEY (`DepartmentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `training_course_management`

CREATE TABLE `training_course_management` (
  `CourseID` int(11) NOT NULL,
  `CourseName` varchar(255) NOT NULL,
  `CourseDuration` int(11) NOT NULL,
  `Instructor` varchar(255) NOT NULL,
  `CourseDescription` text NOT NULL,
  PRIMARY KEY (`CourseID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `training_record_management`

CREATE TABLE `training_record_management` (
  `EmployeeID` int(11) NOT NULL,
  `CourseID` int(11) NOT NULL,
  `TrainingDate` datetime NOT NULL,
  `TrainingScore` int(11) NOT NULL,
  `PassStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`EmployeeID`,`CourseID`),
  FOREIGN KEY (`EmployeeID`) REFERENCES `employee_information` (`EmployeeID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`CourseID`) REFERENCES `training_course_management` (`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
