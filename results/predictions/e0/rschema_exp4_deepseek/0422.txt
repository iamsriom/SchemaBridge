-- Schema for database: rschema_0422

-- Table: UserTable
CREATE TABLE `UserTable` (
  `UserID` int(11) NOT NULL AUTO_INCREMENT,
  `UserName` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `Phone` char(15) NOT NULL,
  `Role` varchar(255) NOT NULL,
  `RegistrationTime` datetime NOT NULL,
  `LastLoginTime` datetime DEFAULT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: ProjectTable
CREATE TABLE `ProjectTable` (
  `ProjectID` int(11) NOT NULL AUTO_INCREMENT,
  `ProjectName` varchar(255) NOT NULL,
  `ProjectIntroduction` text NOT NULL,
  `ResponsiblePersonID` int(11) NOT NULL,
  `CreationTime` datetime NOT NULL,
  `ProjectStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`ProjectID`),
  FOREIGN KEY (`ResponsiblePersonID`) REFERENCES `UserTable` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: ProjectVersionTable
CREATE TABLE `ProjectVersionTable` (
  `VersionID` int(11) NOT NULL AUTO_INCREMENT,
  `ProjectID` int(11) NOT NULL,
  `VersionName` varchar(255) NOT NULL,
  `ReleaseDate` datetime NOT NULL,
  `VersionDescription` text NOT NULL,
  `IsStableVersion` tinyint(1) NOT NULL,
  PRIMARY KEY (`VersionID`),
  FOREIGN KEY (`ProjectID`) REFERENCES `ProjectTable` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: TaskTable
CREATE TABLE `TaskTable` (
  `ProjectID` int(11) NOT NULL,
  `AssignedToUserID` int(11) NOT NULL,
  `TaskName` varchar(255) NOT NULL,
  `TaskStatus` varchar(255) NOT NULL,
  `StartTime` datetime NOT NULL,
  `EndTime` datetime DEFAULT NULL,
  `TaskDescription` text NOT NULL,
  PRIMARY KEY (`ProjectID`,`AssignedToUserID`),
  FOREIGN KEY (`ProjectID`) REFERENCES `ProjectTable` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`AssignedToUserID`) REFERENCES `UserTable` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: CodeSubmissionTable
CREATE TABLE `CodeSubmissionTable` (
  `UserID` int(11) NOT NULL,
  `ProjectID` int(11) NOT NULL,
  `SubmissionTime` datetime NOT NULL,
  `SubmissionDescription` text NOT NULL,
  `CodePath` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`,`ProjectID`),
  FOREIGN KEY (`UserID`) REFERENCES `UserTable` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ProjectID`) REFERENCES `ProjectTable` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
