-- Schema for database: rschema_0424

-- Table: user
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `account_status` varchar(255) NOT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `login_ip` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: system
CREATE TABLE `system` (
  `system_id` int(11) NOT NULL AUTO_INCREMENT,
  `system_name` varchar(255) NOT NULL,
  `system_domain` varchar(255) NOT NULL,
  `access_time` datetime NOT NULL,
  `callback_url` varchar(255) NOT NULL,
  `secret_key` varchar(255) NOT NULL,
  `token_validity_period` int(11) NOT NULL,
  `system_status` varchar(255) NOT NULL,
  `approval_required` varchar(255) NOT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: login_record
CREATE TABLE `login_record` (
  `user_id` int(11) NOT NULL,
  `system_id` int(11) NOT NULL,
  `login_time` datetime NOT NULL,
  `login_ip` varchar(255) NOT NULL,
  `token` varchar(255) NOT NULL,
  `expiration_time` datetime NOT NULL,
  `login_status` varchar(255) NOT NULL,
  `logout_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`,`system_id`),
  FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`system_id`) REFERENCES `system` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: system_configuration
CREATE TABLE `system_configuration` (
  `system_id` int(11) NOT NULL,
  `callback_url` varchar(255) NOT NULL,
  `secret_key` varchar(255) NOT NULL,
  `token_validity_period` int(11) NOT NULL,
  PRIMARY KEY (`system_id`),
  FOREIGN KEY (`system_id`) REFERENCES `system` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: authorization
CREATE TABLE `authorization` (
  `user_id` int(11) NOT NULL,
  `system_id` int(11) NOT NULL,
  `authorization_time` datetime NOT NULL,
  `expiration_time` datetime NOT NULL,
  `authorization_status` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`,`system_id`),
  FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`system_id`) REFERENCES `system` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
