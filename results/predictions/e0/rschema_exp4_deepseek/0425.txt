-- Schema for database: rschema_0425

-- Table: users
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: buildings
CREATE TABLE `buildings` (
  `building_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `description` text,
  `dormitory_manager_id` int(11),
  PRIMARY KEY (`building_id`),
  FOREIGN KEY (`dormitory_manager_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: dormitories
CREATE TABLE `dormitories` (
  `dormitory_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `num_beds` int(11) NOT NULL,
  `building_id` int(11) NOT NULL,
  `supporting_facilities` text,
  `dormitory_manager_id` int(11),
  PRIMARY KEY (`dormitory_id`),
  FOREIGN KEY (`building_id`) REFERENCES `buildings` (`building_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`dormitory_manager_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: students
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `dormitory_id` int(11),
  `class` varchar(255),
  PRIMARY KEY (`student_id`),
  FOREIGN KEY (`dormitory_id`) REFERENCES `dormitories` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: applications
CREATE TABLE `applications` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `type` varchar(255) NOT NULL,
  `student_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `time` datetime NOT NULL,
  `status` varchar(255) NOT NULL,
  `admin_opinion` text,
  PRIMARY KEY (`application_id`),
  FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: maintenance
CREATE TABLE `maintenance` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_id` int(11) NOT NULL,
  `fault_description` text NOT NULL,
  `repair_time` datetime NOT NULL,
  `staff_id` int(11) NOT NULL,
  `situation` text NOT NULL,
  `completion_time` datetime,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`dormitory_id`) REFERENCES `dormitories` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`staff_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: announcements
CREATE TABLE `announcements` (
  `announcement_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `release_time` datetime NOT NULL,
  `admin_id` int(11) NOT NULL,
  PRIMARY KEY (`announcement_id`),
  FOREIGN KEY (`admin_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
