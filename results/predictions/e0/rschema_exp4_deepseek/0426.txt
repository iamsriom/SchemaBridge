-- Schema for database: `rschema_0426`

-- Table: `users`
CREATE TABLE `users` (
  `account_number` int(11) NOT NULL AUTO_INCREMENT,
  `password` varchar(255) NOT NULL,
  `nickname` varchar(255) NOT NULL,
  `bio` text,
  PRIMARY KEY (`account_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `attractions`
CREATE TABLE `attractions` (
  `attraction_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `picture` varchar(255) NOT NULL,
  `reference_price` decimal(10,2) NOT NULL,
  `description` text NOT NULL,
  PRIMARY KEY (`attraction_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `bookmarks`
CREATE TABLE `bookmarks` (
  `user_id` int(11) NOT NULL,
  `attraction_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`,`attraction_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`account_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`attraction_id`) REFERENCES `attractions` (`attraction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `reservations`
CREATE TABLE `reservations` (
  `reservation_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `attraction_id` int(11) NOT NULL,
  `reservation_time` datetime NOT NULL,
  `ticket_purchase_time` datetime NOT NULL,
  `number_of_tickets` int(11) NOT NULL,
  PRIMARY KEY (`reservation_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`account_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`attraction_id`) REFERENCES `attractions` (`attraction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `comments`
CREATE TABLE `comments` (
  `comment_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `attraction_id` int(11) NOT NULL,
  `comment_time` datetime NOT NULL,
  `comment_content` text NOT NULL,
  `comment_status` varchar(255) NOT NULL,
  PRIMARY KEY (`comment_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`account_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`attraction_id`) REFERENCES `attractions` (`attraction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
