-- Schema for database: rschema_0427

-- Table: students
CREATE TABLE `students` (
    `student_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `department` VARCHAR(255) NOT NULL,
    `age` INT(11) NOT NULL,
    `enrollment_date` DATETIME NOT NULL,
    `major` VARCHAR(255) NOT NULL,
    `login_password` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: teachers
CREATE TABLE `teachers` (
    `teacher_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `gender` CHAR(6) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    `title` VARCHAR(255) NOT NULL,
    `department` VARCHAR(255) NOT NULL,
    `course_id` INT(11) NOT NULL,
    PRIMARY KEY (`teacher_id`),
    FOREIGN KEY (`course_id`) REFERENCES `courses`(`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: courses
CREATE TABLE `courses` (
    `course_id` INT(11) NOT NULL AUTO_INCREMENT,
    `course_name` VARCHAR(255) NOT NULL,
    `prerequisite_course_id` INT(11) NOT NULL,
    `credits` INT(11) NOT NULL,
    PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: course_selections
CREATE TABLE `course_selections` (
    `student_id` INT(11) NOT NULL,
    `course_id` INT(11) NOT NULL,
    `grade` INT(11),
    PRIMARY KEY (`student_id`, `course_id`),
    FOREIGN KEY (`student_id`) REFERENCES `students`(`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`course_id`) REFERENCES `courses`(`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: teaching_information
CREATE TABLE `teaching_information` (
    `teacher_id` INT(11) NOT NULL,
    `course_id` INT(11) NOT NULL,
    `teaching_location` VARCHAR(255) NOT NULL,
    `class_time` DATETIME NOT NULL,
    PRIMARY KEY (`teacher_id`, `course_id`),
    FOREIGN KEY (`teacher_id`) REFERENCES `teachers`(`teacher_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`course_id`) REFERENCES `courses`(`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: student_timetable
CREATE TABLE `student_timetable` (
    `student_id` INT(11) NOT NULL,
    `course_id` INT(11) NOT NULL,
    `class_time` DATETIME NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`student_id`, `course_id`),
    FOREIGN KEY (`student_id`) REFERENCES `students`(`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`course_id`) REFERENCES `courses`(`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
