-- Schema for database: `rschema_0432`

-- Table: `department`

CREATE TABLE `department` (
    `DepartmentID` int(11) NOT NULL,
    `DepartmentName` varchar(255) NOT NULL,
    PRIMARY KEY (`DepartmentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `student`

CREATE TABLE `student` (
    `StudentID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Address` varchar(255) NOT NULL,
    `EnrollmentYear` int(11) NOT NULL,
    `DateofBirth` datetime NOT NULL,
    `DepartmentID` int(11) NOT NULL,
    PRIMARY KEY (`StudentID`),
    FOREIGN KEY (`DepartmentID`) REFERENCES `department` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `professor`

CREATE TABLE `professor` (
    `ProfessorID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `OfficeNumber` int(11) NOT NULL,
    `OfficeStartTime` datetime NOT NULL,
    PRIMARY KEY (`ProfessorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `course`

CREATE TABLE `course` (
    `CourseID` int(11) NOT NULL,
    `CourseName` varchar(255) NOT NULL,
    `Semester` varchar(255) NOT NULL,
    `Year` int(11) NOT NULL,
    `TeachingProfessor` int(11) NOT NULL,
    `ClassDay` varchar(255) NOT NULL,
    `CourseDuration` varchar(255) NOT NULL,
    PRIMARY KEY (`CourseID`),
    FOREIGN KEY (`TeachingProfessor`) REFERENCES `professor` (`ProfessorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `registration`

CREATE TABLE `registration` (
    `RegistrationTime` datetime NOT NULL,
    `CourseID` int(11) NOT NULL,
    `StudentID` int(11) NOT NULL,
    FOREIGN KEY (`CourseID`) REFERENCES `course` (`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`StudentID`) REFERENCES `student` (`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
