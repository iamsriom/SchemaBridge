-- Schema for database: `rschema_0433`

-- Table: `business_entities`

CREATE TABLE `business_entities` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `type` varchar(255) NOT NULL,
    `module_code` varchar(255) NOT NULL,
    `name` varchar(255) NOT NULL,
    `creation_time` datetime NOT NULL,
    `update_time` datetime NOT NULL,
    `creator_user_id` int(11) NOT NULL,
    PRIMARY KEY (`id`),
    KEY `creator_user_id` (`creator_user_id`),
    CONSTRAINT `business_entities_ibfk_1` FOREIGN KEY (`creator_user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `attachments`

CREATE TABLE `attachments` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `attachment_id` varchar(255) NOT NULL,
    `filename` varchar(255) NOT NULL,
    `filesize` int(11) NOT NULL,
    `filepath` varchar(255) NOT NULL,
    `creation_time` datetime NOT NULL,
    `file_status` varchar(255) NOT NULL,
    `uploader_user_id` int(11) NOT NULL,
    `business_entity_id` int(11) NOT NULL,
    PRIMARY KEY (`id`),
    KEY `uploader_user_id` (`uploader_user_id`),
    KEY `business_entity_id` (`business_entity_id`),
    CONSTRAINT `attachments_ibfk_1` FOREIGN KEY (`uploader_user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `attachments_ibfk_2` FOREIGN KEY (`business_entity_id`) REFERENCES `business_entities` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `users`

CREATE TABLE `users` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `department` varchar(255) NOT NULL,
    `role` varchar(255) NOT NULL,
    `contact_details` varchar(255) NOT NULL,
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
