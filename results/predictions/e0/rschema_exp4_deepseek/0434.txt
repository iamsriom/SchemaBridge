-- Schema for database: `rschema_0434`

-- Table: `user`

CREATE TABLE `user` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `RegistrationDate` datetime NOT NULL,
    PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `maze`

CREATE TABLE `maze` (
    `MazeID` int(11) NOT NULL AUTO_INCREMENT,
    `Name` varchar(255) NOT NULL,
    `Description` text,
    `CreationDate` datetime NOT NULL,
    `LastModifiedDate` datetime NOT NULL,
    PRIMARY KEY (`MazeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `map`

CREATE TABLE `map` (
    `MapID` int(11) NOT NULL AUTO_INCREMENT,
    `AssociatedMazeID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `MapData` json NOT NULL,
    `CreationDate` datetime NOT NULL,
    `LastModifiedDate` datetime NOT NULL,
    PRIMARY KEY (`MapID`),
    FOREIGN KEY (`AssociatedMazeID`) REFERENCES `maze` (`MazeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `skybox`

CREATE TABLE `skybox` (
    `SkyboxID` int(11) NOT NULL AUTO_INCREMENT,
    `Name` varchar(255) NOT NULL,
    `FilePath` varchar(255) NOT NULL,
    `Description` text,
    `CreationDate` datetime NOT NULL,
    `LastModifiedDate` datetime NOT NULL,
    PRIMARY KEY (`SkyboxID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `game`

CREATE TABLE `game` (
    `UserID` int(11) NOT NULL,
    `MazeID` int(11) NOT NULL,
    `UserScore` int(11) NOT NULL,
    `GameStartTime` datetime NOT NULL,
    `GameEndTime` datetime,
    `GameStatus` enum('InProgress','Completed','Failed') NOT NULL,
    `GameCreationDate` datetime NOT NULL,
    `GameLastModifiedDate` datetime NOT NULL,
    PRIMARY KEY (`UserID`,`MazeID`),
    FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`MazeID`) REFERENCES `maze` (`MazeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
