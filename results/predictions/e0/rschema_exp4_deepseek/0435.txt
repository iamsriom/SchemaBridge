-- Schema for database: rschema_0435
-- For table: probe_management
CREATE TABLE `probe_management` (
  `probeID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `weight` int(11) NOT NULL,
  `launch_date` datetime NOT NULL,
  `service_life` int(11) NOT NULL,
  `scientific_instrument_configuration` varchar(255) NOT NULL,
  `operational_status` varchar(255) NOT NULL,
  `remaining_fuel` int(11) NOT NULL,
  PRIMARY KEY (`probeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: celestial_target_management
CREATE TABLE `celestial_target_management` (
  `celestial_bodyID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `orbital_parameters` varchar(255) NOT NULL,
  `physical_characteristics` varchar(255) NOT NULL,
  `observation_data` varchar(255) NOT NULL,
  `observation_results` varchar(255) NOT NULL,
  `observation_accuracy` varchar(255) NOT NULL,
  `research_value` varchar(255) NOT NULL,
  PRIMARY KEY (`celestial_bodyID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: scientific_instrument_management
CREATE TABLE `scientific_instrument_management` (
  `instrumentID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `weight` int(11) NOT NULL,
  `power_consumption` int(11) NOT NULL,
  `accuracy` varchar(255) NOT NULL,
  `operating_parameters` varchar(255) NOT NULL,
  `calibration_data` varchar(255) NOT NULL,
  PRIMARY KEY (`instrumentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: exploration_mission_management
CREATE TABLE `exploration_mission_management` (
  `probeID` int(11) NOT NULL,
  `celestial_bodyID` int(11) NOT NULL,
  `flight_trajectory` varchar(255) NOT NULL,
  `mission_execution_status` varchar(255) NOT NULL,
  `scientific_data_acquisition_status` varchar(255) NOT NULL,
  `mission_priority` varchar(255) NOT NULL,
  `resource_allocation` varchar(255) NOT NULL,
  PRIMARY KEY (`probeID`, `celestial_bodyID`),
  FOREIGN KEY (`probeID`) REFERENCES `probe_management` (`probeID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`celestial_bodyID`) REFERENCES `celestial_target_management` (`celestial_bodyID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
