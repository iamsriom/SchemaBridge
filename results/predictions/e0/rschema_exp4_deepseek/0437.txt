-- Schema for database: rschema_0437
-- For table: student_information
CREATE TABLE `student_information` (
  `StudentID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Class` varchar(255) NOT NULL,
  `Major` varchar(255) NOT NULL,
  PRIMARY KEY (`StudentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: task_information
CREATE TABLE `task_information` (
  `TaskID` int(11) NOT NULL,
  `TaskTitle` varchar(255) NOT NULL,
  `TaskDescription` text,
  `TaskStartDate` datetime NOT NULL,
  `TaskEndDate` datetime NOT NULL,
  `TaskLocation` varchar(255) NOT NULL,
  PRIMARY KEY (`TaskID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: simulation_module_information
CREATE TABLE `simulation_module_information` (
  `ModuleID` int(11) NOT NULL,
  `ModuleName` varchar(255) NOT NULL,
  `ModuleType` char(6) NOT NULL,
  `ModuleDescription` text,
  `ModuleCode` varchar(255) NOT NULL,
  PRIMARY KEY (`ModuleID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: simulation_result_information
CREATE TABLE `simulation_result_information` (
  `StudentID` int(11) NOT NULL,
  `ModuleID` int(11) NOT NULL,
  `SimulationResultData` text,
  `ResultAnalysis` text,
  `SimulationDate` datetime NOT NULL,
  PRIMARY KEY (`StudentID`,`ModuleID`),
  FOREIGN KEY (`StudentID`) REFERENCES `student_information` (`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ModuleID`) REFERENCES `simulation_module_information` (`ModuleID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: thesis_information
CREATE TABLE `thesis_information` (
  `ThesisID` int(11) NOT NULL,
  `StudentID` int(11) NOT NULL,
  `ThesisTitle` varchar(255) NOT NULL,
  `ThesisContent` text,
  `SubmissionDate` datetime NOT NULL,
  PRIMARY KEY (`ThesisID`),
  FOREIGN KEY (`StudentID`) REFERENCES `student_information` (`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
