-- Schema for database: rschema_0440NaturalLanguageDescription
-- For table: Device
CREATE TABLE `Device` (
  `deviceID` int(11) NOT NULL AUTO_INCREMENT,
  `productName` varchar(255) NOT NULL,
  `deviceModel` varchar(255) NOT NULL,
  `personInCharge` varchar(255) NOT NULL,
  `group` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `usageLocation` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `IPaddress` varchar(255) NOT NULL,
  `deviceImage` varchar(255) NOT NULL,
  `creationTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updateTime` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`deviceID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: User
CREATE TABLE `User` (
  `userID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  `contactInformation` varchar(255) NOT NULL,
  `accountStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`userID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: DeviceInventory
CREATE TABLE `DeviceInventory` (
  `deviceID` int(11) NOT NULL,
  `inventoryPersonID` int(11) NOT NULL,
  `inventoryTime` datetime NOT NULL,
  `deviceLocation` varchar(255) NOT NULL,
  PRIMARY KEY (`deviceID`),
  FOREIGN KEY (`deviceID`) REFERENCES `Device` (`deviceID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: DeviceMaintenance
CREATE TABLE `DeviceMaintenance` (
  `maintenanceID` int(11) NOT NULL AUTO_INCREMENT,
  `deviceID` int(11) NOT NULL,
  `requesterID` int(11) NOT NULL,
  `maintenanceTime` datetime NOT NULL,
  `maintenanceContent` varchar(255) NOT NULL,
  `handlingStatus` varchar(255) NOT NULL,
  `handlingPersonnel` varchar(255) NOT NULL,
  `handlingTime` datetime NOT NULL,
  `remarks` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenanceID`),
  FOREIGN KEY (`deviceID`) REFERENCES `Device` (`deviceID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: DeviceStatus
CREATE TABLE `DeviceStatus` (
  `deviceID` int(11) NOT NULL,
  `currentStatus` varchar(255) NOT NULL,
  `lastUpdateTime` datetime NOT NULL,
  `remarks` varchar(255) NOT NULL,
  PRIMARY KEY (`deviceID`),
  FOREIGN KEY (`deviceID`) REFERENCES `Device` (`deviceID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
