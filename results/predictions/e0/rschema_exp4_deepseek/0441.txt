-- Schema for database: ResearchGrantManagementSystem
-- For table: Projects
CREATE TABLE `projects` (
  `ProjectID` int(11) NOT NULL AUTO_INCREMENT,
  `ProjectName` varchar(255) NOT NULL,
  `Budget` decimal(10,2) NOT NULL,
  `RemainingFunds` decimal(10,2) NOT NULL,
  PRIMARY KEY (`ProjectID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: Funds
CREATE TABLE `funds` (
  `RecordID` int(11) NOT NULL AUTO_INCREMENT,
  `RecordTime` datetime NOT NULL,
  `TransactionAmount` decimal(10,2) NOT NULL,
  `PurposeDescription` varchar(255) NOT NULL,
  `TransactionOperator` varchar(255) NOT NULL,
  `ProjectID` int(11) NOT NULL,
  PRIMARY KEY (`RecordID`),
  FOREIGN KEY (`ProjectID`) REFERENCES `projects` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: Users
CREATE TABLE `users` (
  `UserID` int(11) NOT NULL AUTO_INCREMENT,
  `UserName` varchar(255) NOT NULL,
  `Role` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: User_Project
CREATE TABLE `user_project` (
  `UserID` int(11) NOT NULL,
  `ProjectID` int(11) NOT NULL,
  PRIMARY KEY (`UserID`,`ProjectID`),
  FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ProjectID`) REFERENCES `projects` (`ProjectID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
