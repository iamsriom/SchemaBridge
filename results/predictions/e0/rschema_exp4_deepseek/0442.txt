-- Schema for database: rschema_0442

-- Table: medical_waste
CREATE TABLE `medical_waste` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `waste_type` varchar(255) NOT NULL,
  `weight` decimal(5,2) NOT NULL,
  `generation_time` datetime NOT NULL,
  `collection_time` datetime NOT NULL,
  `current_status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: department
CREATE TABLE `department` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `institution_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`institution_id`) REFERENCES `institution`(`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: institution
CREATE TABLE `institution` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: personnel
CREATE TABLE `personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: vehicle
CREATE TABLE `vehicle` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate_number` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `capacity` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: inbound
CREATE TABLE `inbound` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `batch_number` varchar(255) NOT NULL,
  `expected_quantity` int(11) NOT NULL,
  `actual_quantity` int(11) NOT NULL,
  `expected_weight` decimal(5,2) NOT NULL,
  `actual_weight` decimal(5,2) NOT NULL,
  `inbound_time` datetime NOT NULL,
  `collector_id` int(11) NOT NULL,
  `confirmer_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`collector_id`) REFERENCES `personnel`(`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`confirmer_id`) REFERENCES `personnel`(`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: outbound
CREATE TABLE `outbound` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `batch_number` varchar(255) NOT NULL,
  `outbound_quantity` int(11) NOT NULL,
  `outbound_weight` decimal(5,2) NOT NULL,
  `outbound_time` datetime NOT NULL,
  `transporter_id` int(11) NOT NULL,
  `confirmer_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`transporter_id`) REFERENCES `personnel`(`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`confirmer_id`) REFERENCES `personnel`(`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
