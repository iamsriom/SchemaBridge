-- Schema for database: rschema_0444

-- Table: students
CREATE TABLE `students` (
  `StudentID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `phonenumber` varchar(255) NOT NULL,
  `gamepreference` varchar(255),
  `gender` char(6) NOT NULL,
  PRIMARY KEY (`StudentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: courses
CREATE TABLE `courses` (
  `CourseID` int(11) NOT NULL,
  `coursename` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `credits` int(11) NOT NULL,
  `instructorID` int(11) NOT NULL,
  PRIMARY KEY (`CourseID`),
  FOREIGN KEY (`instructorID`) REFERENCES `instructors`(`InstructorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: instructors
CREATE TABLE `instructors` (
  `InstructorID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(6) NOT NULL,
  `title` varchar(255) NOT NULL,
  `workduration` int(11) NOT NULL,
  PRIMARY KEY (`InstructorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: attendance
CREATE TABLE `attendance` (
  `StudentID` int(11) NOT NULL,
  `CourseID` int(11) NOT NULL,
  `check-in_time` datetime NOT NULL,
  `attendance_status` varchar(255) NOT NULL,
  `class_time` datetime NOT NULL,
  `remarks` varchar(255),
  PRIMARY KEY (`StudentID`,`CourseID`),
  FOREIGN KEY (`StudentID`) REFERENCES `students`(`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`CourseID`) REFERENCES `courses`(`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: leaves
CREATE TABLE `leaves` (
  `StudentID` int(11) NOT NULL,
  `CourseID` int(11) NOT NULL,
  `leave_time` datetime NOT NULL,
  `class_time` datetime NOT NULL,
  `leave_reason` varchar(255) NOT NULL,
  `leave_status` varchar(255) NOT NULL,
  PRIMARY KEY (`StudentID`,`CourseID`),
  FOREIGN KEY (`StudentID`) REFERENCES `students`(`StudentID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`CourseID`) REFERENCES `courses`(`CourseID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
