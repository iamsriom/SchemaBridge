-- Schema for database: rschema_0446

-- Table: UserInformation
CREATE TABLE `UserInformation` (
    `username` varchar(255) NOT NULL,
    `password` varchar(255) NOT NULL,
    `phonenumber` varchar(255) NOT NULL,
    `registration_time` datetime NOT NULL,
    `last_login_time` datetime NOT NULL,
    `role` varchar(255) NOT NULL,
    PRIMARY KEY (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: UserActions
CREATE TABLE `UserActions` (
    `action_id` int(11) NOT NULL AUTO_INCREMENT,
    `username` varchar(255) NOT NULL,
    `action_time` datetime NOT NULL,
    `location` varchar(255) NOT NULL,
    `action_type` varchar(255) NOT NULL,
    `action_detail` varchar(255) NOT NULL,
    PRIMARY KEY (`action_id`),
    FOREIGN KEY (`username`) REFERENCES `UserInformation` (`username`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: SensorDeployment
CREATE TABLE `SensorDeployment` (
    `sensor_id` int(11) NOT NULL AUTO_INCREMENT,
    `data_collection_time` datetime NOT NULL,
    `soil_moisture` float NOT NULL,
    `temperature` float NOT NULL,
    `rainfall` float NOT NULL,
    `light_intensity` float NOT NULL,
    `sensor_location` varchar(255) NOT NULL,
    `burial_depth` float NOT NULL,
    PRIMARY KEY (`sensor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Meteorology
CREATE TABLE `Meteorology` (
    `weather_id` int(11) NOT NULL AUTO_INCREMENT,
    `timestamp` datetime NOT NULL,
    `temperature` float NOT NULL,
    `humidity` float NOT NULL,
    `rainfall` float NOT NULL,
    `wind_speed` float NOT NULL,
    `light_intensity` float NOT NULL,
    `location` varchar(255) NOT NULL,
    PRIMARY KEY (`weather_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Crops
CREATE TABLE `Crops` (
    `crop_id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `type` varchar(255) NOT NULL,
    PRIMARY KEY (`crop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: GrowthModel
CREATE TABLE `GrowthModel` (
    `model_id` int(11) NOT NULL AUTO_INCREMENT,
    `model_name` varchar(255) NOT NULL,
    `parameters` varchar(255) NOT NULL,
    `creation_time` datetime NOT NULL,
    PRIMARY KEY (`model_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Evaluation
CREATE TABLE `Evaluation` (
    `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
    `crop_id` int(11) NOT NULL,
    `metric_name` varchar(255) NOT NULL,
    `value` float NOT NULL,
    `location` varchar(255) NOT NULL,
    PRIMARY KEY (`evaluation_id`),
    FOREIGN KEY (`crop_id`) REFERENCES `Crops` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
