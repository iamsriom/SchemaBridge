-- Schema for database: rschema_0447

-- Table: UserInformation
CREATE TABLE `UserInformation` (
    `userID` int(11) NOT NULL AUTO_INCREMENT,
    `username` varchar(255) NOT NULL,
    `password` varchar(255) NOT NULL,
    `phonenumber` varchar(255) NOT NULL,
    `associatedunit` varchar(255) NOT NULL,
    `role` varchar(255) NOT NULL,
    PRIMARY KEY (`userID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: FinancialDocument
CREATE TABLE `FinancialDocument` (
    `documentID` int(11) NOT NULL AUTO_INCREMENT,
    `documentTitle` varchar(255) NOT NULL,
    `documentNumber` varchar(255) NOT NULL,
    `issuingUnit` varchar(255) NOT NULL,
    `issueDate` datetime NOT NULL,
    `documentCategory` varchar(255) NOT NULL,
    `documentLevel` varchar(255) NOT NULL,
    `validityStatus` varchar(255) NOT NULL,
    `remarks` text,
    `documentContent` text NOT NULL,
    `attachmentPath` varchar(255) NOT NULL,
    PRIMARY KEY (`documentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Notification
CREATE TABLE `Notification` (
    `notificationID` int(11) NOT NULL AUTO_INCREMENT,
    `notificationContent` text NOT NULL,
    `issueDate` datetime NOT NULL,
    `notificationStatus` varchar(255) NOT NULL,
    PRIMARY KEY (`notificationID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: WebSystemConfiguration
CREATE TABLE `WebSystemConfiguration` (
    `configID` int(11) NOT NULL AUTO_INCREMENT,
    `systemName` varchar(255) NOT NULL,
    `databaseDirectory` varchar(255) NOT NULL,
    `dataType` varchar(255) NOT NULL,
    `fileUploadDirectory` varchar(255) NOT NULL,
    PRIMARY KEY (`configID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
