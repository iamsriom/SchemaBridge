-- Schema for database: rschema_0448

-- Table: user
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `role` enum('Student','Teacher','LaboratoryAdministrator','ExperimentalCenterDirectorAdministrator','SystemAdministrator') NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `unit` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: equipment
CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `category` varchar(255) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `status` enum('inactive','inuse','active') NOT NULL,
  `acquisition_time` datetime NOT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: usage_records
CREATE TABLE `usage_records` (
  `usage_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `usage_time` datetime NOT NULL,
  `user_id` int(11) NOT NULL,
  `purpose` text NOT NULL,
  PRIMARY KEY (`usage_id`),
  FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: course
CREATE TABLE `course` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_name` varchar(255) NOT NULL,
  `class_time` datetime NOT NULL,
  `class_location` varchar(255) NOT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: course_association
CREATE TABLE `course_association` (
  `course_id` int(11) NOT NULL,
  `equipment_id` int(11) NOT NULL,
  PRIMARY KEY (`course_id`,`equipment_id`),
  FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: consumables
CREATE TABLE `consumables` (
  `consumable_id` int(11) NOT NULL AUTO_INCREMENT,
  `consumable_name` varchar(255) NOT NULL,
  `category` varchar(255) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  `status` enum('archived','inuse','available') NOT NULL,
  PRIMARY KEY (`consumable_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: hazardous_chemicals
CREATE TABLE `hazardous_chemicals` (
  `chemical_id` int(11) NOT NULL AUTO_INCREMENT,
  `chemical_name` varchar(255) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  `issuance_time` datetime NOT NULL,
  `status` enum('archived','inuse','available') NOT NULL,
  PRIMARY KEY (`chemical_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: purchase_records
CREATE TABLE `purchase_records` (
  `purchase_id` int(11) NOT NULL AUTO_INCREMENT,
  `consumable_id` int(11) NOT NULL,
  `purchase_time` datetime NOT NULL,
  PRIMARY KEY (`purchase_id`),
  FOREIGN KEY (`consumable_id`) REFERENCES `consumables` (`consumable_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: issue_records
CREATE TABLE `issue_records` (
  `issue_id` int(11) NOT NULL AUTO_INCREMENT,
  `chemical_id` int(11) NOT NULL,
  `issue_time` datetime NOT NULL,
  PRIMARY KEY (`issue_id`),
  FOREIGN KEY (`chemical_id`) REFERENCES `hazardous_chemicals` (`chemical_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: return_records
CREATE TABLE `return_records` (
  `return_id` int(11) NOT NULL AUTO_INCREMENT,
  `consumable_id` int(11) NOT NULL,
  `return_time` datetime NOT NULL,
  PRIMARY KEY (`return_id`),
  FOREIGN KEY (`consumable_id`) REFERENCES `consumables` (`consumable_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: student_registration
CREATE TABLE `student_registration` (
  `course_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`course_id`,`user_id`),
  FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: teacher_assignment
CREATE TABLE `teacher_assignment` (
  `course_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`course_id`,`user_id`),
  FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
