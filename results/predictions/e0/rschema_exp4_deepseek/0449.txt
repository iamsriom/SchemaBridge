-- Schema for database: rschema_0449

-- Table: Employee
CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_name` varchar(255) NOT NULL,
  `employee_gender` char(6) NOT NULL,
  `employee_dob` datetime NOT NULL,
  `department` int(11) NOT NULL,
  `employee_entry_date` datetime NOT NULL,
  `employee_resignation_date` datetime,
  `employee_position` int(11) NOT NULL,
  `employee_contact_info` varchar(255) NOT NULL,
  `employee_email` varchar(255) NOT NULL,
  `employee_status` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`),
  FOREIGN KEY (`department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`employee_position`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Position
CREATE TABLE `position` (
  `position_id` int(11) NOT NULL AUTO_INCREMENT,
  `position_name` varchar(255) NOT NULL,
  `position_responsibilities` text NOT NULL,
  `position_requirements` text NOT NULL,
  PRIMARY KEY (`position_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Training
CREATE TABLE `training` (
  `training_id` int(11) NOT NULL AUTO_INCREMENT,
  `training_time` datetime NOT NULL,
  `training_location` varchar(255) NOT NULL,
  `training_content` text NOT NULL,
  `trainer` int(11) NOT NULL,
  PRIMARY KEY (`training_id`),
  FOREIGN KEY (`trainer`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Department
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `department_function` text NOT NULL,
  `department_planning` text NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Employee_Position_Change
CREATE TABLE `employee_position_change` (
  `employee_id` int(11) NOT NULL,
  `position_id` int(11) NOT NULL,
  `change_date` datetime NOT NULL,
  `change_reason` text NOT NULL,
  PRIMARY KEY (`employee_id`, `position_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
