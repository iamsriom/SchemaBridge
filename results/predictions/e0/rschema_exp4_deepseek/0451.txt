CREATE TABLE `intrusion_detection` (
  `record_id` int(11) NOT NULL,
  `detection_time` datetime NOT NULL,
  `source_ip_address` varchar(255) NOT NULL,
  `destination_ip_address` varchar(255) NOT NULL,
  `source_port_number` int(11) NOT NULL,
  `destination_port_number` int(11) NOT NULL,
  `attack_type` varchar(255) NOT NULL,
  `threat_level` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `response_plan` (
  `plan_id` int(11) NOT NULL,
  `plan_generation_time` datetime NOT NULL,
  `plan_type` varchar(255) NOT NULL,
  `plan_content` text NOT NULL,
  `execution_status` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `knowledge_base` (
  `knowledge_id` int(11) NOT NULL,
  `knowledge_category` varchar(255) NOT NULL,
  `knowledge_description` text NOT NULL,
  `knowledge_source` varchar(255) NOT NULL,
  `knowledge_reliability` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alarm` (
  `alarm_id` int(11) NOT NULL,
  `alarm_time` datetime NOT NULL,
  `alarm_level` varchar(255) NOT NULL,
  `alarm_content` text NOT NULL,
  `processing_status` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
