-- Schema for database: rschema_0453

-- Table: stores
CREATE TABLE `stores` (
  `StoreID` int(11) NOT NULL,
  `StoreName` varchar(255) NOT NULL,
  `Address` varchar(255) NOT NULL,
  `ContactPhoneNumber` varchar(255) NOT NULL,
  PRIMARY KEY (`StoreID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: departments
CREATE TABLE `departments` (
  `DepartmentID` int(11) NOT NULL,
  `StoreID` int(11) NOT NULL,
  `DepartmentName` varchar(255) NOT NULL,
  `LocationDistribution` varchar(255) NOT NULL,
  `ContactPhoneNumber` varchar(255) NOT NULL,
  PRIMARY KEY (`DepartmentID`),
  FOREIGN KEY (`StoreID`) REFERENCES `stores` (`StoreID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: employees
CREATE TABLE `employees` (
  `EmployeeID` int(11) NOT NULL,
  `DepartmentID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Position` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `Salary` decimal(10,2) NOT NULL,
  PRIMARY KEY (`EmployeeID`),
  FOREIGN KEY (`DepartmentID`) REFERENCES `departments` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: managers
CREATE TABLE `managers` (
  `ManagerID` int(11) NOT NULL,
  `DepartmentID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Position` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `Salary` decimal(10,2) NOT NULL,
  PRIMARY KEY (`ManagerID`),
  FOREIGN KEY (`DepartmentID`) REFERENCES `departments` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
