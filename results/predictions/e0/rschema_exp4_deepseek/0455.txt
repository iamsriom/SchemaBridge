-- Schema for database: `parking_management_system`

-- For table: `parking_space`

CREATE TABLE `parking_space` (
    `parking_space_id` int(11) NOT NULL AUTO_INCREMENT,
    `status` varchar(255) NOT NULL,
    PRIMARY KEY (`parking_space_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `entry`

CREATE TABLE `entry` (
    `entry_id` int(11) NOT NULL AUTO_INCREMENT,
    `vehicle_id` int(11) NOT NULL,
    `parking_space_id` int(11) NOT NULL,
    `entry_time` datetime NOT NULL,
    PRIMARY KEY (`entry_id`),
    FOREIGN KEY (`parking_space_id`) REFERENCES `parking_space` (`parking_space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `exit`

CREATE TABLE `exit` (
    `exit_id` int(11) NOT NULL AUTO_INCREMENT,
    `vehicle_id` int(11) NOT NULL,
    `parking_space_id` int(11) NOT NULL,
    `exit_time` datetime NOT NULL,
    PRIMARY KEY (`exit_id`),
    FOREIGN KEY (`parking_space_id`) REFERENCES `parking_space` (`parking_space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
