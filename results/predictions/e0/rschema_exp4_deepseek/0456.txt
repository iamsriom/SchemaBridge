CREATE TABLE `Supermarket` (

  `supermarket_name` varchar(255) NOT NULL,

  `manager_phone_number` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Product
CREATE TABLE `Product` (
  `product_ID` int(11) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  `storage_location` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Distribution_Department
CREATE TABLE `Distribution_Department` (
  `department_ID` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_address` varchar(255) NOT NULL,
  `manager_phone_number` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Purchase_Order
CREATE TABLE `Purchase_Order` (
  `order_ID` int(11) NOT NULL,
  `product_ID` int(11) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `supplier` varchar(255) NOT NULL,
  `purchase_quantity` int(11) NOT NULL,
  `other_information` varchar(255)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Payment_Order
CREATE TABLE `Payment_Order` (
  `order_ID` int(11) NOT NULL,

  `payment_amount` decimal(10,2) NOT NULL,

  `payment_status` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
