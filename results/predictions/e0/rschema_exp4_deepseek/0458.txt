-- Schema for database: `rschema_0458`

-- Table: `members`

CREATE TABLE `members` (
    `MemberID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Address` varchar(255) NOT NULL,
    `PostalCode` char(6) NOT NULL,
    `Telephone` varchar(255) NOT NULL,
    `Expenditure` decimal(10,2) NOT NULL DEFAULT 0.00,
    `Points` int(11) NOT NULL DEFAULT 0,
    PRIMARY KEY (`MemberID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `books`

CREATE TABLE `books` (
    `BookID` int(11) NOT NULL AUTO_INCREMENT,
    `TypeName` varchar(255) NOT NULL,
    `BookTitle` varchar(255) NOT NULL,
    `Author` varchar(255) NOT NULL,
    `Publisher` varchar(255) NOT NULL,
    `PublicationDate` date NOT NULL,
    `ISBN` varchar(255) NOT NULL,
    `Price` decimal(10,2) NOT NULL,
    PRIMARY KEY (`BookID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `orders`

CREATE TABLE `orders` (
    `OrderID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Sales` decimal(10,2) NOT NULL,
    `OrderDate` date NOT NULL,
    `ShippingDate` date NOT NULL,
    PRIMARY KEY (`OrderID`),
    FOREIGN KEY (`Username`) REFERENCES `members` (`Username`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `order_details`

CREATE TABLE `order_details` (
    `OrderDetailID` int(11) NOT NULL AUTO_INCREMENT,
    `OrderID` int(11) NOT NULL,
    `BookID` int(11) NOT NULL,
    `Quantity` int(11) NOT NULL,
    PRIMARY KEY (`OrderDetailID`),
    FOREIGN KEY (`OrderID`) REFERENCES `orders` (`OrderID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`BookID`) REFERENCES `books` (`BookID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
