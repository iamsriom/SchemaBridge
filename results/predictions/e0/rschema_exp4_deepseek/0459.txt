-- Schema for database: `rschema_0459`

-- Table: `department`

CREATE TABLE `department` (
  `department_number` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `manager` int(11) NOT NULL,
  PRIMARY KEY (`department_number`),
  CONSTRAINT `fk_department_manager` FOREIGN KEY (`manager`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `employee`

CREATE TABLE `employee` (
  `employee_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `department` int(11) NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `position` varchar(255) NOT NULL,
  `education_level` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_number`),
  CONSTRAINT `fk_employee_department` FOREIGN KEY (`department`) REFERENCES `department` (`department_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `project`

CREATE TABLE `project` (
  `project_number` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `associated_department` int(11) NOT NULL,
  `responsible_person` int(11) NOT NULL,
  PRIMARY KEY (`project_number`),
  CONSTRAINT `fk_project_department` FOREIGN KEY (`associated_department`) REFERENCES `department` (`department_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_project_employee` FOREIGN KEY (`responsible_person`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `employee_project`

CREATE TABLE `employee_project` (
  `employee_number` int(11) NOT NULL,
  `project_number` int(11) NOT NULL,
  PRIMARY KEY (`employee_number`, `project_number`),
  CONSTRAINT `fk_employee_project_employee` FOREIGN KEY (`employee_number`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_employee_project_project` FOREIGN KEY (`project_number`) REFERENCES `project` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
