-- Schema for database: rschema_0460

-- For table: stores
CREATE TABLE `stores` (
    `StoreID` INT(11) NOT NULL,
    `StoreName` VARCHAR(255) NOT NULL,
    `Address` VARCHAR(255) NOT NULL,
    `ContactNumber` CHAR(10) NOT NULL,
    PRIMARY KEY (`StoreID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: departments
CREATE TABLE `departments` (
    `DepartmentID` INT(11) NOT NULL,
    `DepartmentName` VARCHAR(255) NOT NULL,
    `Location` VARCHAR(255) NOT NULL,
    `ContactNumber` CHAR(10) NOT NULL,
    PRIMARY KEY (`DepartmentID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: employees
CREATE TABLE `employees` (
    `EmployeeID` INT(11) NOT NULL,
    `Name` VARCHAR(255) NOT NULL,
    `Position` VARCHAR(255) NOT NULL,
    `PhoneNumber` CHAR(10) NOT NULL,
    `Salary` DECIMAL(10,2) NOT NULL,
    `DepartmentID` INT(11) NOT NULL,
    PRIMARY KEY (`EmployeeID`),
    FOREIGN KEY (`DepartmentID`) REFERENCES `departments` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: procurement
CREATE TABLE `procurement` (
    `ProcurementID` INT(11) NOT NULL,
    `ItemName` VARCHAR(255) NOT NULL,
    `Quantity` INT(11) NOT NULL,
    `Price` DECIMAL(10,2) NOT NULL,
    `StoreID` INT(11) NOT NULL,
    PRIMARY KEY (`ProcurementID`),
    FOREIGN KEY (`StoreID`) REFERENCES `stores` (`StoreID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: suppliers
CREATE TABLE `suppliers` (
    `SupplierID` INT(11) NOT NULL,
    `SupplierName` VARCHAR(255) NOT NULL,
    `ContactNumber` CHAR(10) NOT NULL,
    `Address` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`SupplierID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: supplier_procurement
CREATE TABLE `supplier_procurement` (
    `SupplierID` INT(11) NOT NULL,
    `ProcurementID` INT(11) NOT NULL,
    PRIMARY KEY (`SupplierID`, `ProcurementID`),
    FOREIGN KEY (`SupplierID`) REFERENCES `suppliers` (`SupplierID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ProcurementID`) REFERENCES `procurement` (`ProcurementID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
