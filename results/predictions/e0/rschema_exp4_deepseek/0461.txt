-- Schema for database: rschema_0461

-- For table: Product
CREATE TABLE `Product` (
    `ProductID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `UnitPrice` decimal(10,2) NOT NULL,
    `StockQuantity` int(11) NOT NULL,
    `UnitofMeasurement` varchar(255) NOT NULL,
    `Category` varchar(255) NOT NULL,
    `Description` text NOT NULL,
    `StorageLocation` varchar(255) NOT NULL,
    PRIMARY KEY (`ProductID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: DistributionPersonnel
CREATE TABLE `DistributionPersonnel` (
    `DeliveryPersonnelID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Phone` varchar(255) NOT NULL,
    `Gender` char(1) NOT NULL,
    `Position` varchar(255) NOT NULL,
    PRIMARY KEY (`DeliveryPersonnelID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: Customer
CREATE TABLE `Customer` (
    `CustomerID` int(11) NOT NULL,
    `Name` varchar(255) NOT NULL,
    `Gender` char(1) NOT NULL,
    `Age` int(11) NOT NULL,
    `Phone` varchar(255) NOT NULL,
    `Address` text NOT NULL,
    PRIMARY KEY (`CustomerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: Order
CREATE TABLE `Order` (
    `CustomerID` int(11) NOT NULL,
    `ProductID` int(11) NOT NULL,
    `OrderCreationTime` datetime NOT NULL,
    `ProductQuantity` int(11) NOT NULL,
    `TotalOrderPrice` decimal(10,2) NOT NULL,
    `DeliveryAddress` text NOT NULL,
    `OrderStatus` varchar(255) NOT NULL,
    `EstimatedDeliveryTime` datetime NOT NULL,
    `ResponsibleDeliveryPersonnelID` int(11) NOT NULL,
    PRIMARY KEY (`CustomerID`, `ProductID`),
    FOREIGN KEY (`CustomerID`) REFERENCES `Customer` (`CustomerID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ProductID`) REFERENCES `Product` (`ProductID`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`ResponsibleDeliveryPersonnelID`) REFERENCES `DistributionPersonnel` (`DeliveryPersonnelID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
