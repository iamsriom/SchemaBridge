-- Schema for database: `rschema_0467`

-- Table: `customers`

CREATE TABLE `customers` (
    `customer_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `contact_information` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `products`

CREATE TABLE `products` (
    `product_id` INT(11) NOT NULL,
    `product_name` VARCHAR(255) NOT NULL,
    `price` DECIMAL(10,2) NOT NULL,
    `inventory_quantity` INT(11) NOT NULL,
    PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `orders`

CREATE TABLE `orders` (
    `order_id` INT(11) NOT NULL,
    `product_id` INT(11) NOT NULL,
    `customer_id` INT(11) NOT NULL,
    `order_quantity` INT(11) NOT NULL,
    `order_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`order_id`),
    FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `picking_list`

CREATE TABLE `picking_list` (
    `picking_list_id` INT(11) NOT NULL,
    `order_id` INT(11) NOT NULL,
    `product_id` INT(11) NOT NULL,
    `picking_quantity` INT(11) NOT NULL,
    `picking_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`picking_list_id`),
    FOREIGN KEY (`order_id`) REFERENCES `orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `delivery_order`

CREATE TABLE `delivery_order` (
    `delivery_order_id` INT(11) NOT NULL,
    `order_id` INT(11) NOT NULL,
    `delivery_time` DATETIME NOT NULL,
    `delivery_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`delivery_order_id`),
    FOREIGN KEY (`order_id`) REFERENCES `orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `customer_invoice`

CREATE TABLE `customer_invoice` (
    `invoice_id` INT(11) NOT NULL,
    `order_id` INT(11) NOT NULL,
    `invoice_amount` DECIMAL(10,2) NOT NULL,
    `invoice_date` DATE NOT NULL,
    PRIMARY KEY (`invoice_id`),
    FOREIGN KEY (`order_id`) REFERENCES `orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `accounts_receivable_report`

CREATE TABLE `accounts_receivable_report` (
    `report_id` INT(11) NOT NULL,
    `order_id` INT(11) NOT NULL,
    `receivable_amount` DECIMAL(10,2) NOT NULL,
    `report_date` DATE NOT NULL,
    PRIMARY KEY (`report_id`),
    FOREIGN KEY (`order_id`) REFERENCES `orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
