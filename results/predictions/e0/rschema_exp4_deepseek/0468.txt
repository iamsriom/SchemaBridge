-- Schema for database: `credit_card_management`

-- Table: `CreditCardApplication`

CREATE TABLE `credit_card_application` (
  `application_number` int(11) NOT NULL AUTO_INCREMENT,
  `customer_number` int(11) DEFAULT NULL,
  `type_of_card_applied` varchar(255) NOT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `applicant_id` varchar(255) NOT NULL,
  `application_submission_time` datetime NOT NULL,
  `application_result` varchar(255) NOT NULL DEFAULT 'pending review',
  PRIMARY KEY (`application_number`),
  UNIQUE KEY `customer_number` (`customer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `TransactionInformation`

CREATE TABLE `transaction_information` (
  `transaction_number` int(11) NOT NULL AUTO_INCREMENT,
  `customer_number` int(11) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_amount` decimal(10,2) NOT NULL,
  `transaction_type` varchar(255) NOT NULL,
  `transaction_description` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_number`),
  KEY `customer_number` (`customer_number`),
  CONSTRAINT `transaction_information_ibfk_1` FOREIGN KEY (`customer_number`) REFERENCES `credit_card_application` (`customer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
