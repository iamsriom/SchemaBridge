-- Schema for database: `credit_card_management`

-- For table: `customer`

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `application_status` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `credit_card`

CREATE TABLE `credit_card` (
  `card_number` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `validity_period` varchar(255) NOT NULL,
  `credit_limit` varchar(255) NOT NULL,
  `activation_status` varchar(255) NOT NULL,
  PRIMARY KEY (`card_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `transaction`

CREATE TABLE `transaction` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `transaction_date` datetime NOT NULL,
  `transaction_description` varchar(255) NOT NULL,
  `transaction_amount` varchar(255) NOT NULL,
  `transaction_status` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `customer_credit_card`

CREATE TABLE `customer_credit_card` (
  `customer_id` int(11) NOT NULL,
  `card_number` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`,`card_number`),
  FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`card_number`) REFERENCES `credit_card` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `customer_transaction`

CREATE TABLE `customer_transaction` (
  `customer_id` int(11) NOT NULL,
  `transaction_id` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`,`transaction_id`),
  FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`transaction_id`) REFERENCES `transaction` (`transaction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
