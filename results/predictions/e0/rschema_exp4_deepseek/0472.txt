CREATE TABLE `student_management` (
  `unique_student_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `class` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`unique_student_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instructor_management` (
  `unique_instructor_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `courses_taught` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`unique_instructor_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homework_management` (
  `unique_homework_ID` int(11) NOT NULL,
  `homework_content` text NOT NULL,
  `submission_time` datetime NOT NULL,
  `grading_status` varchar(255) NOT NULL,
  PRIMARY KEY (`unique_homework_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grading_management` (
  `unique_homework_ID` int(11) NOT NULL,
  `score` int(11) NOT NULL,
  `feedback` text NOT NULL,
  `grading_time` datetime NOT NULL,
  PRIMARY KEY (`unique_homework_ID`),
  FOREIGN KEY (`unique_homework_ID`) REFERENCES `homework_management` (`unique_homework_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
