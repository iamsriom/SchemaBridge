CREATE TABLE `publication`(
`PublicationID` int(11) NOT NULL AUTO_INCREMENT,
`Title` varchar(255) NOT NULL,
`Author` varchar(255) NOT NULL,
`Price` decimal(10,2) NOT NULL,
`Type` varchar(255) NOT NULL,
PRIMARY KEY (`PublicationID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer`(
`CustomerID` int(11) NOT NULL AUTO_INCREMENT,
`Name` varchar(255) NOT NULL,
`Phone` varchar(255) NOT NULL,
`Email` varchar(255) NOT NULL,
`Password` varchar(255) NOT NULL,
`RegistrationStatus` varchar(255) NOT NULL,
PRIMARY KEY (`CustomerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order`(
`OrderID` int(11) NOT NULL AUTO_INCREMENT,
`PublicationID` int(11) NOT NULL,
`CustomerID` int(11) NOT NULL,
`ShippingAddress` varchar(255) NOT NULL,
`PaymentMethod` varchar(255) NOT NULL,
`OrderStatus` varchar(255) NOT NULL,
`OrderTime` datetime NOT NULL,
PRIMARY KEY (`OrderID`),
FOREIGN KEY (`PublicationID`) REFERENCES `publication` (`PublicationID`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`CustomerID`) REFERENCES `customer` (`CustomerID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrator`(
`AdministratorID` int(11) NOT NULL AUTO_INCREMENT,
`Name` varchar(255) NOT NULL,
`Phone` varchar(255) NOT NULL,
`Email` varchar(255) NOT NULL,
`Password` varchar(255) NOT NULL,
PRIMARY KEY (`AdministratorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
