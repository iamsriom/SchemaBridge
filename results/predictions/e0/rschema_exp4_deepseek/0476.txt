-- Schema for database: `rschema_0476`

-- Table: `customers`

CREATE TABLE `customers` (
  `CustomerID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `ContactNumber` varchar(255) NOT NULL,
  `Address` varchar(255) NOT NULL,
  PRIMARY KEY (`CustomerID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sales_orders`

CREATE TABLE `sales_orders` (
  `OrderID` int(11) NOT NULL,
  `CustomerID` int(11) NOT NULL,
  `ExpectedDeliveryDate` datetime NOT NULL,
  `OrderStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`OrderID`),
  FOREIGN KEY (`CustomerID`) REFERENCES `customers` (`CustomerID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `production_planning`

CREATE TABLE `production_planning` (
  `OrderID` int(11) NOT NULL,
  `ProductionStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`OrderID`),
  FOREIGN KEY (`OrderID`) REFERENCES `sales_orders` (`OrderID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `inventory_management`

CREATE TABLE `inventory_management` (
  `ItemID` int(11) NOT NULL,
  `ItemName` varchar(255) NOT NULL,
  `Quantity` int(11) NOT NULL,
  `Status` varchar(255) NOT NULL,
  PRIMARY KEY (`ItemID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `purchase_orders`

CREATE TABLE `purchase_orders` (
  `OrderID` int(11) NOT NULL,
  `SupplierID` int(11) NOT NULL,
  `RawMaterialID` int(11) NOT NULL,
  `OrderQuantity` int(11) NOT NULL,
  `SupplyQuantity` int(11) NOT NULL,
  `OrderStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`OrderID`),
  FOREIGN KEY (`SupplierID`) REFERENCES `suppliers` (`SupplierID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`RawMaterialID`) REFERENCES `inventory_management` (`ItemID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `delivery_records`

CREATE TABLE `delivery_records` (
  `OrderID` int(11) NOT NULL,
  `DeliveryTime` datetime NOT NULL,
  `Location` varchar(255) NOT NULL,
  `Status` varchar(255) NOT NULL,
  PRIMARY KEY (`OrderID`),
  FOREIGN KEY (`OrderID`) REFERENCES `sales_orders` (`OrderID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `financial_records`

CREATE TABLE `financial_records` (
  `TransactionID` int(11) NOT NULL,
  `TransactionType` varchar(255) NOT NULL,
  `Amount` decimal(10,2) NOT NULL,
  `Date` date NOT NULL,
  `Status` varchar(255) NOT NULL,
  PRIMARY KEY (`TransactionID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `suppliers`

CREATE TABLE `suppliers` (
  `SupplierID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `ContactNumber` varchar(255) NOT NULL,
  `Address` varchar(255) NOT NULL,
  PRIMARY KEY (`SupplierID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `pizza_recipes`

CREATE TABLE `pizza_recipes` (
  `PizzaID` int(11) NOT NULL,
  `IngredientID` int(11) NOT NULL,
  `Quantity` int(11) NOT NULL,
  PRIMARY KEY (`PizzaID`,`IngredientID`),
  FOREIGN KEY (`PizzaID`) REFERENCES `inventory_management` (`ItemID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`IngredientID`) REFERENCES `inventory_management` (`ItemID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `raw_material_details`

CREATE TABLE `raw_material_details` (
  `RawMaterialID` int(11) NOT NULL,
  `Detail` varchar(255) NOT NULL,
  PRIMARY KEY (`RawMaterialID`),
  FOREIGN KEY (`RawMaterialID`) REFERENCES `inventory_management` (`ItemID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
