-- Schema for database: `real_estate_management_system`

-- For table: `property`

CREATE TABLE `property` (
  `property_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_certificate` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `construction_year` int(11) NOT NULL,
  `built_area` float NOT NULL,
  `sale_price` float NOT NULL,
  `property_photo` varchar(255) NOT NULL,
  `sales_status` enum('for_sale','sold','off_market') NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `apartment`

CREATE TABLE `apartment` (
  `apartment_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_id` int(11) NOT NULL,
  `apartment_type` varchar(255) NOT NULL,
  PRIMARY KEY (`apartment_id`),
  FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `agent`

CREATE TABLE `agent` (
  `agent_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  PRIMARY KEY (`agent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `responsibility`

CREATE TABLE `responsibility` (
  `agent_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`agent_id`,`property_id`),
  FOREIGN KEY (`agent_id`) REFERENCES `agent` (`agent_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `user`

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role` enum('real_estate_agent','system_administrator') NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `property_info`

CREATE TABLE `property_info` (
  `user_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `info` text NOT NULL,
  PRIMARY KEY (`user_id`,`property_id`),
  FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
