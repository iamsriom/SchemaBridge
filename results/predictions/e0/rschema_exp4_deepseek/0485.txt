-- Schema for database: `wetland_protection`

-- For table: `protection_areas`

CREATE TABLE `protection_areas` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `species`

CREATE TABLE `species` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `flora_and_fauna`

CREATE TABLE `flora_and_fauna` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `species_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `growth_status` varchar(255) NOT NULL,
  `recording_time` datetime NOT NULL,
  `recording_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`species_id`) REFERENCES `species` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `environmental_changes`

CREATE TABLE `environmental_changes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `protection_area_id` int(11) NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `rainfall` float NOT NULL,
  `recording_time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`protection_area_id`) REFERENCES `protection_areas` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
