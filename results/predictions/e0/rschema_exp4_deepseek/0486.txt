-- Schema for database: `rschema_0486`

-- For table: `mineral_resources`

CREATE TABLE `mineral_resources` (
  `resource_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `reserves` int(11) NOT NULL,
  `discovery_date` datetime NOT NULL,
  PRIMARY KEY (`resource_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `mines`

CREATE TABLE `mines` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `extraction`

CREATE TABLE `extraction` (
  `extraction_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `mine_id` int(11) NOT NULL,
  `extraction_date` datetime NOT NULL,
  `extraction_volume` int(11) NOT NULL,
  `extraction_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`extraction_id`),
  FOREIGN KEY (`resource_id`) REFERENCES `mineral_resources` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
