-- Table: `KODELISTE`

CREATE TABLE `KODELISTE` (
  `CODE_ID` int(11) NOT NULL AUTO_INCREMENT,
  `CODE_SYSTEM` varchar(255) NOT NULL,
  `CODE_VALUE` varchar(255) NOT NULL,
  `START_DATE` datetime NOT NULL,
  `END_DATE` datetime NOT NULL,
  `CREATED_BY` varchar(255) NOT NULL,
  `CREATION_TIME` datetime NOT NULL,
  `MODIFIED_BY` varchar(255) NOT NULL,
  `MODIFICATION_TIME` datetime NOT NULL,
  PRIMARY KEY (`CODE_ID`),
  UNIQUE (`CODE_SYSTEM`, `CODE_VALUE`),
  CONSTRAINT `FK_KODELISTE_KODELISTE` FOREIGN KEY (`CODE_SYSTEM`) REFERENCES `KODELISTE` (`CODE_SYSTEM`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
