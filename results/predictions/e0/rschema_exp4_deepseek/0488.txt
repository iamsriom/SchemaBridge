-- Schema for database: rschema_0488

-- Table: departments
CREATE TABLE `departments` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `description` text,
    `email_address` varchar(255) NOT NULL,
    `sorting_order` int(11) NOT NULL DEFAULT 0,
    `allow_customers_to_submit_work_orders` tinyint(1) NOT NULL DEFAULT 1,
    `respond_to_work_orders_via_email` tinyint(1) NOT NULL DEFAULT 1,
    `disable_auto_replies` tinyint(1) NOT NULL DEFAULT 0,
    `hide_from_users` tinyint(1) NOT NULL DEFAULT 0,
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: email_configurations
CREATE TABLE `email_configurations` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `department_id` int(11) NOT NULL,
    `host_address` varchar(255) NOT NULL,
    `port_number` int(11) NOT NULL,
    `login_username` varchar(255) NOT NULL,
    `login_password` varchar(255) NOT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: employee_email_server_maintenance
CREATE TABLE `employee_email_server_maintenance` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `employee_id` int(11) NOT NULL,
    `email_server_id` int(11) NOT NULL,
    `action` varchar(255) NOT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`email_server_id`) REFERENCES `email_configurations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
