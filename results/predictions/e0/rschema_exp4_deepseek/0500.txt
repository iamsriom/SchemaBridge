-- Schema for database: rschema_0500

-- Table: users
CREATE TABLE `users` (
    `UserID` int(11) NOT NULL AUTO_INCREMENT,
    `Username` varchar(255) NOT NULL,
    `Password` varchar(255) NOT NULL,
    `Email` varchar(255) NOT NULL,
    `RoleID` int(11) NOT NULL,
    PRIMARY KEY (`UserID`),
    KEY `RoleID` (`RoleID`),
    CONSTRAINT `users_ibfk_1` FOREIGN KEY (`RoleID`) REFERENCES `roles` (`RoleID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: roles
CREATE TABLE `roles` (
    `RoleID` int(11) NOT NULL AUTO_INCREMENT,
    `RoleName` varchar(255) NOT NULL,
    `RoleDescription` varchar(255) NOT NULL,
    PRIMARY KEY (`RoleID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: tasks
CREATE TABLE `tasks` (
    `TaskID` int(11) NOT NULL AUTO_INCREMENT,
    `TaskName` varchar(255) NOT NULL,
    `TaskDescription` text NOT NULL,
    `TaskDeadline` datetime NOT NULL,
    `TaskProgress` int(11) NOT NULL,
    `TaskTypeID` int(11) NOT NULL,
    `TaskStatusID` int(11) NOT NULL,
    `TaskPriorityID` int(11) NOT NULL,
    `UserID` int(11) NOT NULL,
    PRIMARY KEY (`TaskID`),
    KEY `TaskTypeID` (`TaskTypeID`),
    KEY `TaskStatusID` (`TaskStatusID`),
    KEY `TaskPriorityID` (`TaskPriorityID`),
    KEY `UserID` (`UserID`),
    CONSTRAINT `tasks_ibfk_1` FOREIGN KEY (`TaskTypeID`) REFERENCES `task_types` (`TaskTypeID`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `tasks_ibfk_2` FOREIGN KEY (`TaskStatusID`) REFERENCES `task_statuses` (`TaskStatusID`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `tasks_ibfk_3` FOREIGN KEY (`TaskPriorityID`) REFERENCES `task_priorities` (`TaskPriorityID`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `tasks_ibfk_4` FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_types
CREATE TABLE `task_types` (
    `TaskTypeID` int(11) NOT NULL AUTO_INCREMENT,
    `TaskTypeName` varchar(255) NOT NULL,
    `TaskTypeDescription` text NOT NULL,
    PRIMARY KEY (`TaskTypeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_statuses
CREATE TABLE `task_statuses` (
    `TaskStatusID` int(11) NOT NULL AUTO_INCREMENT,
    `StatusName` varchar(255) NOT NULL,
    `StatusDescription` text NOT NULL,
    PRIMARY KEY (`TaskStatusID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: task_priorities
CREATE TABLE `task_priorities` (
    `TaskPriorityID` int(11) NOT NULL AUTO_INCREMENT,
    `TaskPriorityName` varchar(255) NOT NULL,
    `TaskPriorityDescription` text NOT NULL,
    PRIMARY KEY (`TaskPriorityID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
