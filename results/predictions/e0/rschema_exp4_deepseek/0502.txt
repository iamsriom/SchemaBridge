CREATE TABLE `venues` (
  `VenueID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Type` varchar(255) NOT NULL,
  `AreaSize` float NOT NULL,
  `Address` varchar(255) NOT NULL,
  `Status` varchar(255) NOT NULL DEFAULT 'Active',
  `CreationTime` datetime NOT NULL,
  `LastMaintenanceTime` datetime DEFAULT NULL,
  `MaximumCapacity` int(11) NOT NULL,
  PRIMARY KEY (`VenueID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `UserID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `CreationTime` datetime NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `EquipmentID` int(11) NOT NULL,
  `EquipmentName` varchar(255) NOT NULL,
  `StartUsageTime` datetime NOT NULL,
  `LastMaintenanceTime` datetime DEFAULT NULL,
  `VenueID` int(11) NOT NULL,
  PRIMARY KEY (`EquipmentID`),
  FOREIGN KEY (`VenueID`) REFERENCES `venues` (`VenueID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reservations` (
  `UserID` int(11) NOT NULL,
  `VenueID` int(11) NOT NULL,
  `StartTime` datetime NOT NULL,
  `EndTime` datetime NOT NULL,
  `ReservationStatus` varchar(255) NOT NULL,
  `ReservationSuccessTime` datetime DEFAULT NULL,
  `ReservationCancellationTime` datetime DEFAULT NULL,
  PRIMARY KEY (`UserID`,`VenueID`),
  FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`VenueID`) REFERENCES `venues` (`VenueID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `venue_evaluations` (
  `UserID` int(11) NOT NULL,
  `VenueID` int(11) NOT NULL,
  `EvaluationTime` datetime NOT NULL,
  `EvaluationContent` text NOT NULL,
  `Rating` int(11) NOT NULL,
  PRIMARY KEY (`UserID`,`VenueID`),
  FOREIGN KEY (`UserID`) REFERENCES `users` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`VenueID`) REFERENCES `venues` (`VenueID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
