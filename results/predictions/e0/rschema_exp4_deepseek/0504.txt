-- Schema for database: `rschema_0504`

-- Table: `cultural_relic`

CREATE TABLE `cultural_relic` (
  `CulturalRelicID` int(11) NOT NULL AUTO_INCREMENT,
  `Name` varchar(255) NOT NULL,
  `Year` int(11) NOT NULL,
  `Category` varchar(255) NOT NULL,
  PRIMARY KEY (`CulturalRelicID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `staff`

CREATE TABLE `staff` (
  `StaffID` int(11) NOT NULL AUTO_INCREMENT,
  `Name` varchar(255) NOT NULL,
  `PhoneNumber` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  PRIMARY KEY (`StaffID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `cultural_relic_exhibition`

CREATE TABLE `cultural_relic_exhibition` (
  `CulturalRelicID` int(11) NOT NULL,
  `ExhibitionStartTime` datetime NOT NULL,
  `ExhibitionEndTime` datetime NOT NULL,
  `ExhibitionLocation` varchar(255) NOT NULL,
  PRIMARY KEY (`CulturalRelicID`),
  FOREIGN KEY (`CulturalRelicID`) REFERENCES `cultural_relic` (`CulturalRelicID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `cultural_relic_repair`

CREATE TABLE `cultural_relic_repair` (
  `StaffID` int(11) NOT NULL,
  `CulturalRelicID` int(11) NOT NULL,
  `RepairStartTime` datetime NOT NULL,
  `RepairEndTime` datetime NOT NULL,
  `RepairMethod` varchar(255) NOT NULL,
  `Notes` text,
  PRIMARY KEY (`StaffID`,`CulturalRelicID`),
  FOREIGN KEY (`StaffID`) REFERENCES `staff` (`StaffID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`CulturalRelicID`) REFERENCES `cultural_relic` (`CulturalRelicID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `cultural_relic_monitoring_record`

CREATE TABLE `cultural_relic_monitoring_record` (
  `CulturalRelicID` int(11) NOT NULL,
  `MonitoringTime` datetime NOT NULL,
  `PersoninChargeID` int(11) NOT NULL,
  `Temperature` float NOT NULL,
  `Humidity` float NOT NULL,
  `Light` float NOT NULL,
  `AirQuality` varchar(255) NOT NULL,
  `Notes` text,
  PRIMARY KEY (`CulturalRelicID`,`MonitoringTime`),
  FOREIGN KEY (`CulturalRelicID`) REFERENCES `cultural_relic` (`CulturalRelicID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`PersoninChargeID`) REFERENCES `staff` (`StaffID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
