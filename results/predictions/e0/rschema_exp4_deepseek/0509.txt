-- Schema for database: rschema_0509

-- Table: Patient
CREATE TABLE `Patient` (
  `PatientID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `Age` int(11) NOT NULL,
  `HospitalizationTime` datetime NOT NULL,
  `Ward` int(11) NOT NULL,
  `Phone` varchar(255) NOT NULL,
  `IdentityCardNumber` varchar(255) NOT NULL,
  `MedicalRecordNumber` varchar(255) NOT NULL,
  `EmergencyContactPhone` varchar(255) NOT NULL,
  PRIMARY KEY (`PatientID`),
  KEY `Ward` (`Ward`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Department
CREATE TABLE `Department` (
  `DepartmentID` int(11) NOT NULL,
  `DepartmentName` varchar(255) NOT NULL,
  `DepartmentHeadID` int(11) NOT NULL,
  `ContactPhone` varchar(255) NOT NULL,
  `Location` varchar(255) NOT NULL,
  PRIMARY KEY (`DepartmentID`),
  KEY `DepartmentHeadID` (`DepartmentHeadID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Doctor
CREATE TABLE `Doctor` (
  `DoctorID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `ContactPhone` varchar(255) NOT NULL,
  `Title` varchar(255) NOT NULL,
  `DepartmentID` int(11) NOT NULL,
  PRIMARY KEY (`DoctorID`),
  KEY `DepartmentID` (`DepartmentID`),
  CONSTRAINT `Doctor_ibfk_1` FOREIGN KEY (`DepartmentID`) REFERENCES `Department` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Ward
CREATE TABLE `Ward` (
  `WardID` int(11) NOT NULL,
  `Location` varchar(255) NOT NULL,
  `HeadID` int(11) NOT NULL,
  `BedCount` int(11) NOT NULL,
  `RemainingBedCount` int(11) NOT NULL,
  PRIMARY KEY (`WardID`),
  KEY `HeadID` (`HeadID`),
  CONSTRAINT `Ward_ibfk_1` FOREIGN KEY (`HeadID`) REFERENCES `Doctor` (`DoctorID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Nurse
CREATE TABLE `Nurse` (
  `NurseID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` char(6) NOT NULL,
  `Phone` varchar(255) NOT NULL,
  `DepartmentID` int(11) NOT NULL,
  `Duties` varchar(255) NOT NULL,
  `Title` varchar(255) NOT NULL,
  PRIMARY KEY (`NurseID`),
  KEY `DepartmentID` (`DepartmentID`),
  CONSTRAINT `Nurse_ibfk_1` FOREIGN KEY (`DepartmentID`) REFERENCES `Department` (`DepartmentID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
