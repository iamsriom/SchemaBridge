-- Schema for database: `parking_lot_management_system`

-- For table: `parking_spaces`

CREATE TABLE `parking_spaces` (
  `space_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_status` varchar(255) NOT NULL,
  `vehicle_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`space_id`),
  KEY `vehicle_id` (`vehicle_id`),
  CONSTRAINT `parking_spaces_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `parking_space_owners`

CREATE TABLE `parking_space_owners` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate_number` varchar(255) NOT NULL,
  `owner_name` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `space_id` int(11) NOT NULL,
  PRIMARY KEY (`owner_id`),
  KEY `space_id` (`space_id`),
  CONSTRAINT `parking_space_owners_ibfk_1` FOREIGN KEY (`space_id`) REFERENCES `parking_spaces` (`space_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `vehicles`

CREATE TABLE `vehicles` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate_number` varchar(255) NOT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
