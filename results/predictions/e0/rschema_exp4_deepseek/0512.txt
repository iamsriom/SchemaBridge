-- Schema for database: rschema_0512

-- Table: Customer
CREATE TABLE `customer` (
    `customer_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `id_number` CHAR(18) NOT NULL,
    `contact_phone_number` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Account
CREATE TABLE `account` (
    `account_number` INT(11) NOT NULL,
    `customer_id` INT(11) NOT NULL,
    `account_balance` DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    PRIMARY KEY (`account_number`),
    FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Transaction
CREATE TABLE `transaction` (
    `transaction_id` INT(11) NOT NULL AUTO_INCREMENT,
    `account_number` INT(11) NOT NULL,
    `transaction_type` ENUM('deposit','withdrawal') NOT NULL,
    `transaction_amount` DECIMAL(10,2) NOT NULL,
    `transaction_time` DATETIME NOT NULL,
    `transaction_status` ENUM('pending','completed','failed') NOT NULL DEFAULT 'pending',
    PRIMARY KEY (`transaction_id`),
    FOREIGN KEY (`account_number`) REFERENCES `account` (`account_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
