-- Schema for database: rschema_0518

-- Table: Maze
CREATE TABLE `maze` (
  `MazeID` int(11) NOT NULL,
  `MazeName` varchar(255) NOT NULL,
  `MazeFilePath` varchar(255) NOT NULL,
  PRIMARY KEY (`MazeID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Robot
CREATE TABLE `robot` (
  `RobotID` int(11) NOT NULL,
  `RobotType` varchar(255) NOT NULL,
  `CurrentPosition` varchar(255) NOT NULL,
  `TimerStatus` varchar(255) NOT NULL,
  `MazeID` int(11) NOT NULL,
  PRIMARY KEY (`RobotID`),
  FOREIGN KEY (`MazeID`) REFERENCES `maze` (`MazeID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: ForwardSensor
CREATE TABLE `forward_sensor` (
  `SensorID` int(11) NOT NULL,
  `RobotID` int(11) NOT NULL,
  `LeftStatus` varchar(255) NOT NULL,
  `RightStatus` varchar(255) NOT NULL,
  `FrontStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`SensorID`),
  FOREIGN KEY (`RobotID`) REFERENCES `robot` (`RobotID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: ProximitySensor
CREATE TABLE `proximity_sensor` (
  `SensorID` int(11) NOT NULL,
  `RobotID` int(11) NOT NULL,
  `CoinStatus` varchar(255) NOT NULL,
  `GhostStatus` varchar(255) NOT NULL,
  PRIMARY KEY (`SensorID`),
  FOREIGN KEY (`RobotID`) REFERENCES `robot` (`RobotID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: Timer
CREATE TABLE `timer` (
  `TimerID` int(11) NOT NULL,
  `RobotID` int(11) NOT NULL,
  `TimerEventDescription` varchar(255) NOT NULL,
  `TimerEventTime` datetime NOT NULL,
  PRIMARY KEY (`TimerID`),
  FOREIGN KEY (`RobotID`) REFERENCES `robot` (`RobotID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: User
CREATE TABLE `user` (
  `UserID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `ContactInformation` varchar(255) NOT NULL,
  PRIMARY KEY (`UserID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: UserRobotOperation
CREATE TABLE `user_robot_operation` (
  `UserID` int(11) NOT NULL,
  `OperationType` varchar(255) NOT NULL,
  `OperationTime` datetime NOT NULL,
  `RobotID` int(11) NOT NULL,
  FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`RobotID`) REFERENCES `robot` (`RobotID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
