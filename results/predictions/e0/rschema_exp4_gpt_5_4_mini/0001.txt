-- Schema for database: rschema_0001
CREATE TABLE `airplanes` (
  `airplane_number` int(11) NOT NULL,
  `model` varchar(255) NOT NULL,
  `passenger_capacity` int(11) NOT NULL,
  PRIMARY KEY (`airplane_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crew_members` (
  `crew_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`crew_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flights` (
  `flight_number` int(11) NOT NULL,
  `departure_time` datetime NOT NULL,
  `arrival_time` datetime NOT NULL,
  `origin` varchar(255) NOT NULL,
  `destination` varchar(255) NOT NULL,
  `airplane_number` int(11) NOT NULL,
  `captain_id` int(11) NOT NULL,
  `first_officer_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`flight_number`),
  KEY `flights_ibfk_1` (`airplane_number`),
  KEY `flights_ibfk_2` (`captain_id`),
  KEY `flights_ibfk_3` (`first_officer_id`),
  CONSTRAINT `flights_ibfk_1` FOREIGN KEY (`airplane_number`) REFERENCES `airplanes` (`airplane_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flights_ibfk_2` FOREIGN KEY (`captain_id`) REFERENCES `crew_members` (`crew_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flights_ibfk_3` FOREIGN KEY (`first_officer_id`) REFERENCES `crew_members` (`crew_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
