-- Schema for database: rschema_0002
CREATE TABLE `transport_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_number` varchar(50) NOT NULL,
  `origin` varchar(255) NOT NULL,
  `destination` varchar(255) NOT NULL,
  `cargo_name` varchar(255) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `transportation_time` int(11) NOT NULL,
  `carrier_name` varchar(255) NOT NULL,
  `carrier_phone` varchar(20) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`),
  UNIQUE KEY `uk_transport_task_task_number` (`task_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transport_method` (
  `method_id` int(11) NOT NULL AUTO_INCREMENT,
  `method_name` varchar(50) NOT NULL,
  PRIMARY KEY (`method_id`),
  UNIQUE KEY `uk_transport_method_method_name` (`method_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_transport_method` (
  `task_id` int(11) NOT NULL,
  `method_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`,`method_id`),
  KEY `task_transport_method_ibfk_2` (`method_id`),
  CONSTRAINT `task_transport_method_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `transport_task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_transport_method_ibfk_2` FOREIGN KEY (`method_id`) REFERENCES `transport_method` (`method_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
