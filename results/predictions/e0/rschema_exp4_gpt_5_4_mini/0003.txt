-- Schema for database: rschema_0003
CREATE TABLE `warehouses` (
  `warehouse_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_number` varchar(50) NOT NULL,
  `address` varchar(255) NOT NULL,
  `capacity` int(11) NOT NULL,
  PRIMARY KEY (`warehouse_id`),
  UNIQUE KEY `warehouse_number` (`warehouse_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `workers` (
  `worker_id` int(11) NOT NULL AUTO_INCREMENT,
  `worker_number` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `worker_type` varchar(20) NOT NULL,
  PRIMARY KEY (`worker_id`),
  UNIQUE KEY `worker_number` (`worker_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_id` int(11) NOT NULL,
  `loader_worker_id` int(11) DEFAULT NULL,
  `unloader_worker_id` int(11) DEFAULT NULL,
  `task_type` varchar(20) NOT NULL,
  `goods_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `task_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`),
  KEY `warehouse_id` (`warehouse_id`),
  KEY `loader_worker_id` (`loader_worker_id`),
  KEY `unloader_worker_id` (`unloader_worker_id`),
  CONSTRAINT `tasks_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouses` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `tasks_ibfk_2` FOREIGN KEY (`loader_worker_id`) REFERENCES `workers` (`worker_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `tasks_ibfk_3` FOREIGN KEY (`unloader_worker_id`) REFERENCES `workers` (`worker_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
