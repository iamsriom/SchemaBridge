-- Schema for database: rschema_0007
CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_name` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisements` (
  `advertisement_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`advertisement_id`),
  KEY `advertisements_ibfk_1` (`client_id`),
  CONSTRAINT `advertisements_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `broadcast_records` (
  `broadcast_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `advertisement_id` int(11) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `broadcast_channel` varchar(255) NOT NULL,
  `operating_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`broadcast_record_id`),
  KEY `broadcast_records_ibfk_1` (`advertisement_id`),
  CONSTRAINT `broadcast_records_ibfk_1` FOREIGN KEY (`advertisement_id`) REFERENCES `advertisements` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
