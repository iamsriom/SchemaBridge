-- Schema for database: rschema_0010
CREATE TABLE `members` (
  `member_no` varchar(50) NOT NULL,
  `member_name` varchar(100) NOT NULL,
  `gender` char(1) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`member_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meetings` (
  `meeting_id` varchar(50) NOT NULL,
  `meeting_type` varchar(100) DEFAULT NULL,
  `meeting_time` datetime NOT NULL,
  `meeting_location` varchar(255) DEFAULT NULL,
  `meeting_content` text DEFAULT NULL,
  `meeting_theme` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`meeting_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendance_records` (
  `member_no` varchar(50) NOT NULL,
  `meeting_id` varchar(50) NOT NULL,
  `attendance_time` datetime NOT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`member_no`,`meeting_id`),
  CONSTRAINT `attendance_records_ibfk_1` FOREIGN KEY (`member_no`) REFERENCES `members` (`member_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `attendance_records_ibfk_2` FOREIGN KEY (`meeting_id`) REFERENCES `meetings` (`meeting_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
