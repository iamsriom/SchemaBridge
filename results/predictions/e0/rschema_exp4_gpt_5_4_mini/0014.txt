-- Schema for database: rschema_0014
CREATE TABLE `irrigation_systems` (
  `system_id` int(11) NOT NULL,
  `system_name` varchar(255) NOT NULL,
  `system_type` varchar(100) NOT NULL,
  `installation_date` date DEFAULT NULL,
  `maintenance_records` text DEFAULT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plots` (
  `plot_id` int(11) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  `soil_type` varchar(100) NOT NULL,
  `current_irrigation_status` varchar(50) NOT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `irrigation_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `system_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `irrigation_date` date NOT NULL,
  `irrigation_duration` int(11) NOT NULL,
  `irrigation_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`record_id`),
  KEY `system_id` (`system_id`),
  KEY `plot_id` (`plot_id`),
  CONSTRAINT `irrigation_records_ibfk_1` FOREIGN KEY (`system_id`) REFERENCES `irrigation_systems` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `irrigation_records_ibfk_2` FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
