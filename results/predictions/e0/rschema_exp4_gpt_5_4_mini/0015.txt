-- Schema for database: rschema_0015
CREATE TABLE `plots` (
  `plot_id` int(11) NOT NULL AUTO_INCREMENT,
  `area` decimal(10,2) NOT NULL,
  `soil_type` varchar(100) NOT NULL,
  `geographical_location` varchar(255) NOT NULL,
  `current_pest_disease_status` varchar(50) NOT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pest_diseases` (
  `pest_disease_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `damage_degree` varchar(50) NOT NULL,
  `prevention_control_methods` text NOT NULL,
  `occurrence_season` varchar(50) NOT NULL,
  PRIMARY KEY (`pest_disease_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operators` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_records` (
  `monitoring_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `plot_id` int(11) NOT NULL,
  `pest_disease_id` int(11) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `monitoring_results` text NOT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`monitoring_record_id`),
  KEY `monitoring_records_ibfk_1` (`plot_id`),
  KEY `monitoring_records_ibfk_2` (`pest_disease_id`),
  KEY `monitoring_records_ibfk_3` (`operator_id`),
  CONSTRAINT `monitoring_records_ibfk_1` FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_records_ibfk_2` FOREIGN KEY (`pest_disease_id`) REFERENCES `pest_diseases` (`pest_disease_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_records_ibfk_3` FOREIGN KEY (`operator_id`) REFERENCES `operators` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
