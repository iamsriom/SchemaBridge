-- Schema for database: rschema_0016
CREATE TABLE `farms` (
  `farm_id` int(11) NOT NULL,
  `farm_name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`farm_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plots` (
  `plot_id` int(11) NOT NULL,
  `area` decimal(10,2) DEFAULT NULL,
  `soil_type` varchar(100) DEFAULT NULL,
  `current_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `planting_plans` (
  `plan_id` int(11) NOT NULL,
  `farm_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `crop_type` varchar(100) NOT NULL,
  `planting_time` datetime NOT NULL,
  `expected_harvest_time` datetime DEFAULT NULL,
  `planting_quantity` int(11) DEFAULT NULL,
  `operator` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`plan_id`),
  KEY `planting_plans_ibfk_1` (`farm_id`),
  KEY `planting_plans_ibfk_2` (`plot_id`),
  CONSTRAINT `planting_plans_ibfk_1` FOREIGN KEY (`farm_id`) REFERENCES `farms` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `planting_plans_ibfk_2` FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
