-- Schema for database: rschema_0017
CREATE TABLE `workshop` (
  `workshop_id` int(11) NOT NULL AUTO_INCREMENT,
  `workshop_name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`workshop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `worker` (
  `worker_id` int(11) NOT NULL AUTO_INCREMENT,
  `workshop_id` int(11) NOT NULL,
  `worker_name` varchar(255) NOT NULL,
  `contact_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`worker_id`),
  KEY `worker_ibfk_1` (`workshop_id`),
  CONSTRAINT `worker_ibfk_1` FOREIGN KEY (`workshop_id`) REFERENCES `workshop` (`workshop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `worker_id` int(11) NOT NULL,
  `raw_material_name` varchar(255) NOT NULL,
  `processing_time` datetime NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `production_quantity` int(11) NOT NULL,
  `product_quality_rating` varchar(50) NOT NULL,
  PRIMARY KEY (`record_id`),
  KEY `processing_record_ibfk_1` (`worker_id`),
  CONSTRAINT `processing_record_ibfk_1` FOREIGN KEY (`worker_id`) REFERENCES `worker` (`worker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `raw_material_inventory` (
  `inventory_id` int(11) NOT NULL AUTO_INCREMENT,
  `workshop_id` int(11) NOT NULL,
  `raw_material_name` varchar(255) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  PRIMARY KEY (`inventory_id`),
  KEY `raw_material_inventory_ibfk_1` (`workshop_id`),
  CONSTRAINT `raw_material_inventory_ibfk_1` FOREIGN KEY (`workshop_id`) REFERENCES `workshop` (`workshop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `workshop_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `workshop_id` int(11) NOT NULL,
  `performance_month` char(7) NOT NULL,
  `monthly_total_processing_volume` int(11) NOT NULL,
  `product_qualification_rate` decimal(5,2) NOT NULL,
  `assessment_indicators` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`performance_id`),
  KEY `workshop_performance_ibfk_1` (`workshop_id`),
  CONSTRAINT `workshop_performance_ibfk_1` FOREIGN KEY (`workshop_id`) REFERENCES `workshop` (`workshop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
