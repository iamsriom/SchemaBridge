-- Schema for database: rschema_0018
CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL,
  `equipment_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `usage_frequency` int(11) NOT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_usage_record` (
  `usage_record_id` int(11) NOT NULL,
  `equipment_id` int(11) NOT NULL,
  `usage_time` datetime NOT NULL,
  `user_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`usage_record_id`),
  KEY `equipment_usage_record_ibfk_1` (`equipment_id`),
  CONSTRAINT `equipment_usage_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_maintenance_record` (
  `maintenance_record_id` int(11) NOT NULL,
  `equipment_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(500) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_record_id`),
  KEY `equipment_maintenance_record_ibfk_1` (`equipment_id`),
  CONSTRAINT `equipment_maintenance_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
