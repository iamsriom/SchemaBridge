-- Schema for database: rschema_0023
CREATE TABLE `donors` (
  `donor_no` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`donor_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipients` (
  `recipient_no` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`recipient_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donated_items` (
  `item_no` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`item_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donation_records` (
  `donation_no` int(11) NOT NULL,
  `donor_no` int(11) NOT NULL,
  `recipient_no` int(11) NOT NULL,
  `donation_time` datetime NOT NULL,
  `item_no` int(11) NOT NULL,
  PRIMARY KEY (`donation_no`),
  KEY `donation_records_ibfk_1` (`donor_no`),
  KEY `donation_records_ibfk_2` (`recipient_no`),
  KEY `donation_records_ibfk_3` (`item_no`),
  CONSTRAINT `donation_records_ibfk_1` FOREIGN KEY (`donor_no`) REFERENCES `donors` (`donor_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `donation_records_ibfk_2` FOREIGN KEY (`recipient_no`) REFERENCES `recipients` (`recipient_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `donation_records_ibfk_3` FOREIGN KEY (`item_no`) REFERENCES `donated_items` (`item_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
