-- Schema for database: rschema_0027
CREATE TABLE `clothing` (
  `clothing_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `style` varchar(255) NOT NULL,
  `production_date` date NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`clothing_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_code` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `position` varchar(255) NOT NULL,
  `department_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  UNIQUE KEY `employee_code` (`employee_code`),
  KEY `department_id` (`department_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_record` (
  `production_id` int(11) NOT NULL AUTO_INCREMENT,
  `clothing_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `producer_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  PRIMARY KEY (`production_id`),
  KEY `clothing_id` (`clothing_id`),
  KEY `producer_id` (`producer_id`),
  CONSTRAINT `production_record_ibfk_1` FOREIGN KEY (`clothing_id`) REFERENCES `clothing` (`clothing_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_record_ibfk_2` FOREIGN KEY (`producer_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory_record` (
  `inventory_id` int(11) NOT NULL AUTO_INCREMENT,
  `clothing_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `storage_time` datetime NOT NULL,
  PRIMARY KEY (`inventory_id`),
  KEY `clothing_id` (`clothing_id`),
  CONSTRAINT `inventory_record_ibfk_1` FOREIGN KEY (`clothing_id`) REFERENCES `clothing` (`clothing_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `sales_id` int(11) NOT NULL AUTO_INCREMENT,
  `clothing_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `salesperson_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `sales_time` datetime NOT NULL,
  PRIMARY KEY (`sales_id`),
  KEY `clothing_id` (`clothing_id`),
  KEY `salesperson_id` (`salesperson_id`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`clothing_id`) REFERENCES `clothing` (`clothing_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`salesperson_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department_appraisal` (
  `appraisal_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `appraisal_period` varchar(50) NOT NULL,
  `score` decimal(5,2) NOT NULL,
  `remarks` varchar(255) DEFAULT NULL,
  `appraisal_time` datetime NOT NULL,
  PRIMARY KEY (`appraisal_id`),
  KEY `department_id` (`department_id`),
  CONSTRAINT `department_appraisal_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
