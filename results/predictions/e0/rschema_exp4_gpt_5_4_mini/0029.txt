-- Schema for database: rschema_0029
CREATE TABLE `furniture_product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `product_type` varchar(255) NOT NULL,
  `production_date` date DEFAULT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `supervisor` varchar(255) DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_number` varchar(50) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `position` varchar(255) NOT NULL,
  `department_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  UNIQUE KEY `uk_employee_number` (`employee_number`),
  KEY `employee_ibfk_1` (`department_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_event` (
  `production_event_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `producer_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  PRIMARY KEY (`production_event_id`),
  KEY `production_event_ibfk_1` (`product_id`),
  KEY `production_event_ibfk_2` (`producer_id`),
  CONSTRAINT `production_event_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `furniture_product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_event_ibfk_2` FOREIGN KEY (`producer_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory_record` (
  `inventory_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `storage_time` datetime NOT NULL,
  PRIMARY KEY (`inventory_record_id`),
  KEY `inventory_record_ibfk_1` (`product_id`),
  CONSTRAINT `inventory_record_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `furniture_product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_transaction` (
  `sales_transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `sales_rep_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `sales_time` datetime NOT NULL,
  PRIMARY KEY (`sales_transaction_id`),
  KEY `sales_transaction_ibfk_1` (`product_id`),
  KEY `sales_transaction_ibfk_2` (`sales_rep_id`),
  CONSTRAINT `sales_transaction_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `furniture_product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_transaction_ibfk_2` FOREIGN KEY (`sales_rep_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `evaluation_period` varchar(50) NOT NULL,
  `performance_score` decimal(5,2) NOT NULL,
  `evaluation_time` datetime NOT NULL,
  `remarks` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`performance_id`),
  KEY `department_performance_ibfk_1` (`department_id`),
  CONSTRAINT `department_performance_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
