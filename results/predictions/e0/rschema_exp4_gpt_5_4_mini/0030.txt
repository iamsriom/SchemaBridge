-- Schema for database: rschema_0030
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position` (
  `position_id` int(11) NOT NULL AUTO_INCREMENT,
  `position_name` varchar(255) NOT NULL,
  PRIMARY KEY (`position_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` varchar(50) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `department_id` int(11) NOT NULL,
  `position_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`),
  KEY `employee_ibfk_1` (`department_id`),
  KEY `employee_ibfk_2` (`position_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`position_id`) REFERENCES `position` (`position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chemical_raw_material` (
  `raw_material_id` int(11) NOT NULL AUTO_INCREMENT,
  `raw_material_name` varchar(255) NOT NULL,
  `production_date` date NOT NULL,
  `shelf_life` int(11) NOT NULL,
  PRIMARY KEY (`raw_material_id`) 
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `composition` varchar(255) NOT NULL,
  `production_date` date NOT NULL,
  `shelf_life` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_record` (
  `production_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `producer_id` varchar(50) NOT NULL,
  `production_time` datetime NOT NULL,
  PRIMARY KEY (`production_record_id`),
  KEY `production_record_ibfk_1` (`product_id`),
  KEY `production_record_ibfk_2` (`producer_id`),
  CONSTRAINT `production_record_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_record_ibfk_2` FOREIGN KEY (`producer_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory_record` (
  `inventory_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `stocking_person_id` varchar(50) NOT NULL,
  `stocking_time` datetime NOT NULL,
  PRIMARY KEY (`inventory_record_id`),
  KEY `inventory_record_ibfk_1` (`product_id`),
  KEY `inventory_record_ibfk_2` (`stocking_person_id`),
  CONSTRAINT `inventory_record_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `inventory_record_ibfk_2` FOREIGN KEY (`stocking_person_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `sales_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `salesperson_id` varchar(50) NOT NULL,
  `sale_time` datetime NOT NULL,
  PRIMARY KEY (`sales_record_id`),
  KEY `sales_record_ibfk_1` (`product_id`),
  KEY `sales_record_ibfk_2` (`salesperson_id`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`salesperson_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department_performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `evaluation_time` datetime NOT NULL,
  `performance_score` decimal(10,2) NOT NULL,
  `evaluation_comment` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`performance_id`),
  KEY `department_performance_ibfk_1` (`department_id`),
  CONSTRAINT `department_performance_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
