-- Schema for database: rschema_0032
CREATE TABLE `rental_location` (
  `location_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_name` varchar(255) NOT NULL,
  `location_address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`location_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_vehicle` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_id` int(11) NOT NULL,
  `vehicle_plate_number` varchar(50) NOT NULL,
  `vehicle_model` varchar(255) DEFAULT NULL,
  `vehicle_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`),
  KEY `rental_vehicle_ibfk_1` (`location_id`),
  CONSTRAINT `rental_vehicle_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `rental_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone_number` varchar(50) NOT NULL,
  `rental_date` datetime NOT NULL,
  `rental_duration` int(11) NOT NULL,
  `rental_cost` decimal(10,2) NOT NULL,
  `vehicle_condition` varchar(255) DEFAULT NULL,
  `customer_feedback` text DEFAULT NULL,
  `customer_satisfaction` int(11) DEFAULT NULL,
  PRIMARY KEY (`task_id`),
  KEY `rental_task_ibfk_1` (`vehicle_id`),
  CONSTRAINT `rental_task_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `rental_vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
