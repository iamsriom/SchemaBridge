-- Schema for database: rschema_0037
CREATE TABLE `civil_engineering_project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `project_manager` varchar(255) NOT NULL,
  `start_date` date NOT NULL,
  `expected_completion_date` date NOT NULL,
  `actual_completion_date` date DEFAULT NULL,
  `project_budget` decimal(12,2) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspector` (
  `inspector_id` int(11) NOT NULL AUTO_INCREMENT,
  `inspector_name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`inspector_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `quality_inspection` (
  `inspection_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `inspector_id` int(11) NOT NULL,
  `inspection_date` date NOT NULL,
  `inspection_content` text NOT NULL,
  `inspection_result` varchar(255) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  KEY `quality_inspection_ibfk_1` (`project_id`),
  KEY `quality_inspection_ibfk_2` (`inspector_id`),
  CONSTRAINT `quality_inspection_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `civil_engineering_project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `quality_inspection_ibfk_2` FOREIGN KEY (`inspector_id`) REFERENCES `inspector` (`inspector_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
