-- Schema for database: rschema_0038
CREATE TABLE `designers` (
  `designer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  PRIMARY KEY (`designer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL,
  `project_manager` varchar(255) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `start_date` date NOT NULL,
  `estimated_completion_date` date NOT NULL,
  `actual_completion_date` date DEFAULT NULL,
  `project_budget` decimal(12,2) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`project_id`),
  KEY `projects_ibfk_1` (`client_id`),
  CONSTRAINT `projects_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `design_plans` (
  `design_plan_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `designer_id` int(11) NOT NULL,
  `design_date` date NOT NULL,
  `design_plan_name` varchar(255) NOT NULL,
  `design_plan_content` text NOT NULL,
  `is_archived` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`design_plan_id`),
  KEY `design_plans_ibfk_1` (`project_id`),
  KEY `design_plans_ibfk_2` (`designer_id`),
  CONSTRAINT `design_plans_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `design_plans_ibfk_2` FOREIGN KEY (`designer_id`) REFERENCES `designers` (`designer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
