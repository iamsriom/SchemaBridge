-- Schema for database: rschema_0043
CREATE TABLE `loan_application` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `property_number` varchar(100) NOT NULL,
  `loan_amount` decimal(12,2) NOT NULL,
  `loan_term_months` int(11) NOT NULL,
  `approval_status` varchar(50) NOT NULL,
  `applicant_name` varchar(100) NOT NULL,
  `applicant_phone` varchar(20) NOT NULL,
  `applicant_id_number` varchar(30) NOT NULL,
  `application_time` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `approval_personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `loan_approval_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `loan_application_id` int(11) NOT NULL,
  `approval_personnel_id` int(11) NOT NULL,
  `approval_time` datetime NOT NULL,
  `approval_result` varchar(50) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `loan_approval_record_ibfk_1` (`loan_application_id`),
  KEY `loan_approval_record_ibfk_2` (`approval_personnel_id`),
  CONSTRAINT `loan_approval_record_ibfk_1` FOREIGN KEY (`loan_application_id`) REFERENCES `loan_application` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_record_ibfk_2` FOREIGN KEY (`approval_personnel_id`) REFERENCES `approval_personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
