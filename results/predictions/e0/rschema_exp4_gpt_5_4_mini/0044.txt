-- Schema for database: rschema_0044
CREATE TABLE `property` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  `sales_price` decimal(12,2) NOT NULL,
  `sales_date` date NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `analysis_report` (
  `report_id` int(11) NOT NULL,
  `report_date` date NOT NULL,
  `report_type` varchar(100) NOT NULL,
  `analysis_results` text NOT NULL,
  PRIMARY KEY (`report_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `report_generation` (
  `report_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `report_date` date NOT NULL,
  `report_type` varchar(100) NOT NULL,
  `analysis_results` text NOT NULL,
  PRIMARY KEY (`report_id`, `property_id`, `report_date`),
  CONSTRAINT `report_generation_ibfk_1` FOREIGN KEY (`report_id`) REFERENCES `analysis_report` (`report_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `report_generation_ibfk_2` FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
