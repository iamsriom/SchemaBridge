-- Schema for database: rschema_0046
CREATE TABLE `member` (
  `member_id` int(11) NOT NULL,
  `name` varchar(100) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `current_points` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shopping_record` (
  `shopping_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `shopping_time` datetime NOT NULL,
  `total_price` decimal(10,2) NOT NULL,
  `points_change` int(11) NOT NULL,
  PRIMARY KEY (`shopping_record_id`),
  KEY `shopping_record_ibfk_1` (`member_id`),
  CONSTRAINT `shopping_record_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `redemption_record` (
  `redemption_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `redemption_time` datetime NOT NULL,
  `redeemed_product_name` varchar(255) NOT NULL,
  `redemption_quantity` int(11) NOT NULL,
  `points_consumed` int(11) NOT NULL,
  PRIMARY KEY (`redemption_record_id`),
  KEY `redemption_record_ibfk_1` (`member_id`),
  CONSTRAINT `redemption_record_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
