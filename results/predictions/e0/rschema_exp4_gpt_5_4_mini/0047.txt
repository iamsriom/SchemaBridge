-- Schema for database: rschema_0047
CREATE TABLE `reporters` (
  `reporter_id` int(11) NOT NULL AUTO_INCREMENT,
  `reporter_name` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`reporter_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `editors` (
  `editor_id` int(11) NOT NULL AUTO_INCREMENT,
  `editor_name` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`editor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publishers` (
  `publisher_id` int(11) NOT NULL AUTO_INCREMENT,
  `publisher_name` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`publisher_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `articles` (
  `article_id` int(11) NOT NULL AUTO_INCREMENT,
  `reporter_id` int(11) NOT NULL,
  `editor_id` int(11) NOT NULL,
  `publisher_id` int(11) DEFAULT NULL,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `submission_time` datetime NOT NULL,
  `review_time` datetime DEFAULT NULL,
  `publishing_time` datetime DEFAULT NULL,
  `review_status` varchar(50) NOT NULL,
  PRIMARY KEY (`article_id`),
  KEY `articles_ibfk_1` (`reporter_id`),
  KEY `articles_ibfk_2` (`editor_id`),
  KEY `articles_ibfk_3` (`publisher_id`),
  CONSTRAINT `articles_ibfk_1` FOREIGN KEY (`reporter_id`) REFERENCES `reporters` (`reporter_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `articles_ibfk_2` FOREIGN KEY (`editor_id`) REFERENCES `editors` (`editor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `articles_ibfk_3` FOREIGN KEY (`publisher_id`) REFERENCES `publishers` (`publisher_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
