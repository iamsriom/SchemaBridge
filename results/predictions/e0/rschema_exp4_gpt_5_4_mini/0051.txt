-- Schema for database: rschema_0051
CREATE TABLE `personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `personnel_name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `water_resource` (
  `resource_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `resource_type` varchar(50) NOT NULL,
  PRIMARY KEY (`resource_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_monitoring` (
  `monitoring_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `monitoring_point` varchar(255) NOT NULL,
  `water_quality_indicators` varchar(500) DEFAULT NULL,
  `water_volume_data` varchar(500) DEFAULT NULL,
  `pollution_level` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`monitoring_id`),
  KEY `resource_monitoring_ibfk_1` (`resource_id`),
  KEY `resource_monitoring_ibfk_2` (`personnel_id`),
  CONSTRAINT `resource_monitoring_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `water_resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resource_monitoring_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `resource_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(1000) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL DEFAULT '0.00',
  PRIMARY KEY (`maintenance_id`),
  KEY `resource_maintenance_ibfk_1` (`resource_id`),
  KEY `resource_maintenance_ibfk_2` (`personnel_id`),
  CONSTRAINT `resource_maintenance_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `water_resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resource_maintenance_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
