-- Schema for database: rschema_0052
CREATE TABLE `land_resource` (
  `land_resource_id` int(11) NOT NULL AUTO_INCREMENT,
  `land_location` varchar(255) NOT NULL,
  `total_area` decimal(12,2) NOT NULL,
  PRIMARY KEY (`land_resource_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  `project_purpose` varchar(255) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_usage_record` (
  `land_usage_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `land_resource_id` int(11) NOT NULL,
  `used_area` decimal(12,2) NOT NULL,
  `usage_years` int(11) NOT NULL,
  `current_status` varchar(100) NOT NULL,
  `record_personnel` varchar(255) NOT NULL,
  `record_time` datetime NOT NULL,
  PRIMARY KEY (`land_usage_record_id`),
  KEY `land_usage_record_ibfk_1` (`project_id`),
  KEY `land_usage_record_ibfk_2` (`land_resource_id`),
  CONSTRAINT `land_usage_record_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `land_usage_record_ibfk_2` FOREIGN KEY (`land_resource_id`) REFERENCES `land_resource` (`land_resource_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_change_record` (
  `land_change_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `land_resource_id` int(11) NOT NULL,
  `change_content` varchar(500) NOT NULL,
  `change_reason` varchar(500) NOT NULL,
  `change_cost` decimal(12,2) NOT NULL,
  `change_time` datetime NOT NULL,
  PRIMARY KEY (`land_change_record_id`),
  KEY `land_change_record_ibfk_1` (`land_resource_id`),
  CONSTRAINT `land_change_record_ibfk_1` FOREIGN KEY (`land_resource_id`) REFERENCES `land_resource` (`land_resource_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
