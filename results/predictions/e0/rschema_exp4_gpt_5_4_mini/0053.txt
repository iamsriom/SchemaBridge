CREATE TABLE `processing_pool` (
  `processing_pool_id` int(11) NOT NULL AUTO_INCREMENT,
  `pool_name` varchar(255) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `responsible_staff_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`processing_pool_id`),
  KEY `processing_pool_ibfk_1` (`responsible_staff_id`),
  CONSTRAINT `processing_pool_ibfk_1` FOREIGN KEY (`responsible_staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff` (
  `staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `staff_name` varchar(255) NOT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `responsibility` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`staff_id`),
  UNIQUE KEY `staff_email_unique` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_pool_operation_record` (
  `operation_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_pool_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `processing_capacity` decimal(10,2) DEFAULT NULL,
  `processing_efficiency` decimal(10,2) DEFAULT NULL,
  `recorded_by_staff_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`operation_record_id`),
  KEY `processing_pool_operation_record_ibfk_1` (`processing_pool_id`),
  KEY `processing_pool_operation_record_ibfk_2` (`recorded_by_staff_id`),
  CONSTRAINT `processing_pool_operation_record_ibfk_1` FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pool` (`processing_pool_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `processing_pool_operation_record_ibfk_2` FOREIGN KEY (`recorded_by_staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_pool_maintenance_record` (
  `maintenance_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_pool_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintenance_cost` decimal(10,2) DEFAULT NULL,
  `maintenance_staff_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`maintenance_record_id`),
  KEY `processing_pool_maintenance_record_ibfk_1` (`processing_pool_id`),
  KEY `processing_pool_maintenance_record_ibfk_2` (`maintenance_staff_id`),
  CONSTRAINT `processing_pool_maintenance_record_ibfk_1` FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pool` (`processing_pool_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `processing_pool_maintenance_record_ibfk_2` FOREIGN KEY (`maintenance_staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
