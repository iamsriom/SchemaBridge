-- Schema for database: rschema_0057
CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `installation_location` varchar(255) NOT NULL,
  `installation_date` date NOT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintainer` (
  `maintainer_id` int(11) NOT NULL AUTO_INCREMENT,
  `maintainer_name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`maintainer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `maintenance_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `maintenance_date` date NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintainer_id` int(11) NOT NULL,
  `equipment_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_record_id`),
  KEY `maintenance_record_ibfk_1` (`maintainer_id`),
  KEY `maintenance_record_ibfk_2` (`equipment_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`maintainer_id`) REFERENCES `maintainer` (`maintainer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_2` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
