-- Schema for database: rschema_0059
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `handlers` (
  `handler_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`handler_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `faults` (
  `fault_id` int(11) NOT NULL,
  `description` varchar(255) NOT NULL,
  `occurrence_date` datetime NOT NULL,
  `resolution_date` datetime DEFAULT NULL,
  `handler_id` int(11) DEFAULT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`fault_id`),
  KEY `faults_ibfk_1` (`handler_id`),
  KEY `faults_ibfk_2` (`user_id`),
  CONSTRAINT `faults_ibfk_1` FOREIGN KEY (`handler_id`) REFERENCES `handlers` (`handler_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `faults_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
