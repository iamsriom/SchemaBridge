-- Schema for database: rschema_0061
CREATE TABLE `patent` (
  `patent_number` varchar(50) NOT NULL,
  `patent_name` varchar(255) NOT NULL,
  `application_date` date NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`patent_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventor` (
  `work_number` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `telephone_number` varchar(20) DEFAULT NULL,
  `position` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`work_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patent_application_record` (
  `application_number` varchar(50) NOT NULL,
  `patent_number` varchar(50) NOT NULL,
  `work_number` varchar(50) NOT NULL,
  `application_date` date NOT NULL,
  PRIMARY KEY (`application_number`),
  KEY `patent_number` (`patent_number`),
  KEY `work_number` (`work_number`),
  CONSTRAINT `patent_application_record_ibfk_1` FOREIGN KEY (`patent_number`) REFERENCES `patent` (`patent_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patent_application_record_ibfk_2` FOREIGN KEY (`work_number`) REFERENCES `inventor` (`work_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
