-- Schema for database: rschema_0063
CREATE TABLE `achievements` (
  `achievement_id` int(11) NOT NULL AUTO_INCREMENT,
  `achievement_number` varchar(100) NOT NULL,
  `achievement_name` varchar(255) NOT NULL,
  `research_and_development_date` date NOT NULL,
  `technical_field` varchar(255) NOT NULL,
  PRIMARY KEY (`achievement_id`),
  UNIQUE KEY `achievement_number` (`achievement_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_number` varchar(100) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `transformation_date` date NOT NULL,
  PRIMARY KEY (`product_id`),
  UNIQUE KEY `product_number` (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `achievement_transformations` (
  `transformation_id` int(11) NOT NULL AUTO_INCREMENT,
  `achievement_number` varchar(100) NOT NULL,
  `product_number` varchar(100) NOT NULL,
  `transformation_date` date NOT NULL,
  PRIMARY KEY (`transformation_id`),
  KEY `achievement_number` (`achievement_number`),
  KEY `product_number` (`product_number`),
  CONSTRAINT `achievement_transformations_ibfk_1` FOREIGN KEY (`achievement_number`) REFERENCES `achievements` (`achievement_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `achievement_transformations_ibfk_2` FOREIGN KEY (`product_number`) REFERENCES `products` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
