-- Schema for database: rschema_0064
CREATE TABLE `project` (
  `project_no` varchar(50) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `application_date` date NOT NULL,
  `application_unit` varchar(255) NOT NULL,
  PRIMARY KEY (`project_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review_expert` (
  `expert_no` varchar(50) NOT NULL,
  `expert_name` varchar(255) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`expert_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_review_record` (
  `project_no` varchar(50) NOT NULL,
  `expert_no` varchar(50) NOT NULL,
  `review_date` date NOT NULL,
  `review_opinion` varchar(1000) NOT NULL,
  PRIMARY KEY (`project_no`,`expert_no`),
  CONSTRAINT `project_review_record_ibfk_1` FOREIGN KEY (`project_no`) REFERENCES `project` (`project_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_review_record_ibfk_2` FOREIGN KEY (`expert_no`) REFERENCES `review_expert` (`expert_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
