-- Schema for database: rschema_0066
CREATE TABLE `community` (
  `community_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_name` varchar(255) NOT NULL,
  PRIMARY KEY (`community_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `owner` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `owner_name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `house_number` varchar(50) NOT NULL,
  PRIMARY KEY (`owner_id`),
  KEY `owner_ibfk_1` (`community_id`),
  CONSTRAINT `owner_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_application` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `owner_id` int(11) NOT NULL,
  `application_time` datetime NOT NULL,
  `maintenance_content` varchar(500) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`application_id`),
  KEY `maintenance_application_ibfk_1` (`owner_id`),
  CONSTRAINT `maintenance_application_ibfk_1` FOREIGN KEY (`owner_id`) REFERENCES `owner` (`owner_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
