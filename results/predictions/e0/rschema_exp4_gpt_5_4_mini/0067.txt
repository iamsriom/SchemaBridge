-- Schema for database: rschema_0067
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `credit_score` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rentals` (
  `rental_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `rental_time` datetime NOT NULL,
  `return_time` datetime DEFAULT NULL,
  `rental_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`rental_id`),
  KEY `rentals_ibfk_1` (`customer_id`),
  CONSTRAINT `rentals_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_credit_changes` (
  `change_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `change_time` datetime NOT NULL,
  `change_reason` varchar(255) NOT NULL,
  `old_credit_value` int(11) NOT NULL,
  `new_credit_value` int(11) NOT NULL,
  PRIMARY KEY (`change_id`),
  KEY `customer_credit_changes_ibfk_1` (`customer_id`),
  CONSTRAINT `customer_credit_changes_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
