-- Schema for database: rschema_0068
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `services` (
  `service_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `service_date` datetime NOT NULL,
  `description` varchar(255) NOT NULL,
  `fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`service_id`),
  KEY `services_ibfk_1` (`customer_id`),
  CONSTRAINT `services_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_feedback` (
  `feedback_id` int(11) NOT NULL,
  `service_id` int(11) NOT NULL,
  `rating` int(11) NOT NULL,
  `feedback_comments` varchar(1000) DEFAULT NULL,
  PRIMARY KEY (`feedback_id`),
  KEY `service_feedback_ibfk_1` (`service_id`),
  CONSTRAINT `service_feedback_ibfk_1` FOREIGN KEY (`service_id`) REFERENCES `services` (`service_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
