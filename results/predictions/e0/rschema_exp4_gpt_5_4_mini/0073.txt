-- Schema for database: rschema_0073
CREATE TABLE `agents` (
  `agent_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `hire_date` date NOT NULL,
  `contact_phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`agent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `id_card_number` varchar(50) NOT NULL,
  `contact_phone_number` varchar(20) NOT NULL,
  `agent_id` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`),
  UNIQUE KEY `uk_customers_id_card_number` (`id_card_number`),
  KEY `customers_ibfk_1` (`agent_id`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`agent_id`) REFERENCES `agents` (`agent_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `insurance_policies` (
  `policy_number` varchar(50) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `insurance_type` varchar(20) NOT NULL,
  `insured_amount` decimal(15,2) NOT NULL,
  `insurance_term` int(11) NOT NULL,
  `premium` decimal(15,2) NOT NULL,
  PRIMARY KEY (`policy_number`),
  KEY `insurance_policies_ibfk_1` (`customer_id`),
  CONSTRAINT `insurance_policies_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `premium_payments` (
  `payment_id` int(11) NOT NULL,
  `policy_number` varchar(50) NOT NULL,
  `payment_amount` decimal(15,2) NOT NULL,
  `payment_time` datetime NOT NULL,
  `payment_method` varchar(50) NOT NULL,
  PRIMARY KEY (`payment_id`),
  KEY `premium_payments_ibfk_1` (`policy_number`),
  CONSTRAINT `premium_payments_ibfk_1` FOREIGN KEY (`policy_number`) REFERENCES `insurance_policies` (`policy_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `claim_records` (
  `claim_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `claim_time` datetime NOT NULL,
  `claim_amount` decimal(15,2) NOT NULL,
  `claim_description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`claim_id`),
  KEY `claim_records_ibfk_1` (`customer_id`),
  CONSTRAINT `claim_records_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
