-- Schema for database: rschema_0075
CREATE TABLE `analysts` (
  `analyst_id` int(11) NOT NULL,
  `analyst_name` varchar(255) NOT NULL,
  `date_of_hire` date NOT NULL,
  `contact_number` varchar(20) NOT NULL,
  PRIMARY KEY (`analyst_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stocks` (
  `stock_code` varchar(20) NOT NULL,
  `stock_name` varchar(255) NOT NULL,
  `current_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `analyses` (
  `analysis_id` int(11) NOT NULL AUTO_INCREMENT,
  `analyst_id` int(11) NOT NULL,
  `stock_code` varchar(20) NOT NULL,
  `analysis_date` date NOT NULL,
  `results` text NOT NULL,
  `reasoning` text NOT NULL,
  PRIMARY KEY (`analysis_id`),
  KEY `analyst_id` (`analyst_id`),
  KEY `stock_code` (`stock_code`),
  CONSTRAINT `analyses_ibfk_1` FOREIGN KEY (`analyst_id`) REFERENCES `analysts` (`analyst_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `analyses_ibfk_2` FOREIGN KEY (`stock_code`) REFERENCES `stocks` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stock_price_history` (
  `price_history_id` int(11) NOT NULL AUTO_INCREMENT,
  `stock_code` varchar(20) NOT NULL,
  `price_date` date NOT NULL,
  `closing_price` decimal(10,2) NOT NULL,
  `opening_price` decimal(10,2) NOT NULL,
  `highest_price` decimal(10,2) NOT NULL,
  `lowest_price` decimal(10,2) NOT NULL,
  `trading_volume` bigint(20) NOT NULL,
  PRIMARY KEY (`price_history_id`),
  KEY `stock_code` (`stock_code`),
  CONSTRAINT `stock_price_history_ibfk_1` FOREIGN KEY (`stock_code`) REFERENCES `stocks` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
