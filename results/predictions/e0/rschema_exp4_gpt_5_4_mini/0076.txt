-- Schema for database: rschema_0076
CREATE TABLE `beer_manufacturer` (
  `manufacturer_no` int(11) NOT NULL,
  `manufacturer_name` varchar(255) NOT NULL,
  `manufacturer_country` varchar(255) NOT NULL,
  PRIMARY KEY (`manufacturer_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beer_type` (
  `beer_no` int(11) NOT NULL,
  `beer_name` varchar(255) NOT NULL,
  `manufacturer_no` int(11) NOT NULL,
  `beer_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`beer_no`),
  KEY `beer_type_ibfk_1` (`manufacturer_no`),
  CONSTRAINT `beer_type_ibfk_1` FOREIGN KEY (`manufacturer_no`) REFERENCES `beer_manufacturer` (`manufacturer_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bar` (
  `bar_no` int(11) NOT NULL,
  `bar_name` varchar(255) NOT NULL,
  `bar_address` varchar(255) NOT NULL,
  `bar_phone` varchar(20) NOT NULL,
  PRIMARY KEY (`bar_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drinker` (
  `drinker_no` int(11) NOT NULL,
  `drinker_name` varchar(255) NOT NULL,
  `drinker_gender` char(1) NOT NULL,
  `drinker_age` int(11) NOT NULL,
  `drinker_phone` varchar(20) NOT NULL,
  PRIMARY KEY (`drinker_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `sales_id` int(11) NOT NULL,
  `beer_no` int(11) NOT NULL,
  `bar_no` int(11) NOT NULL,
  `sales_date` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`sales_id`),
  KEY `sales_record_ibfk_1` (`beer_no`),
  KEY `sales_record_ibfk_2` (`bar_no`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`beer_no`) REFERENCES `beer_type` (`beer_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`bar_no`) REFERENCES `bar` (`bar_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
