-- Schema for database: rschema_0079
CREATE TABLE `investment_asset` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `code` varchar(32) NOT NULL,
  `asset_type` varchar(20) NOT NULL,
  `market` varchar(50) DEFAULT NULL,
  `owner` varchar(100) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `category` varchar(50) DEFAULT NULL,
  `subcategory` varchar(50) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_investment_asset_code_type` (`code`,`asset_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `investment_holding` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `asset_id` int(11) NOT NULL,
  `total_investment_amount` decimal(18,2) NOT NULL DEFAULT '0.00',
  `holding_shares` decimal(18,6) NOT NULL DEFAULT '0.000000',
  `last_trading_date` date DEFAULT NULL,
  `last_settled_at` datetime DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_investment_holding_asset_id` (`asset_id`),
  CONSTRAINT `investment_holding_ibfk_1` FOREIGN KEY (`asset_id`) REFERENCES `investment_asset` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `investment_trading_config` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `asset_id` int(11) NOT NULL,
  `buy_fee_rate` decimal(10,6) DEFAULT NULL,
  `sell_fee_rate` decimal(10,6) DEFAULT NULL,
  `compensation_fee_rate` decimal(10,6) DEFAULT NULL,
  `min_fee` decimal(18,2) DEFAULT NULL,
  `max_fee` decimal(18,2) DEFAULT NULL,
  `fee_note` varchar(255) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_investment_trading_config_asset_id` (`asset_id`),
  CONSTRAINT `investment_trading_config_ibfk_1` FOREIGN KEY (`asset_id`) REFERENCES `investment_asset` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `investment_order` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `asset_id` int(11) NOT NULL,
  `out_id` varchar(64) NOT NULL,
  `order_type` varchar(20) NOT NULL,
  `status` varchar(20) NOT NULL,
  `is_settled` tinyint(1) NOT NULL DEFAULT '0',
  `created_at` datetime NOT NULL,
  `trading_date` date DEFAULT NULL,
  `settled_at` datetime DEFAULT NULL,
  `transaction_amount` decimal(18,2) NOT NULL DEFAULT '0.00',
  `transaction_quantity` decimal(18,6) NOT NULL DEFAULT '0.000000',
  `transaction_service_fee` decimal(18,2) NOT NULL DEFAULT '0.00',
  `currency` char(6) DEFAULT NULL,
  `remark` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_investment_order_out_id` (`out_id`),
  KEY `investment_order_ibfk_1` (`asset_id`),
  CONSTRAINT `investment_order_ibfk_1` FOREIGN KEY (`asset_id`) REFERENCES `investment_asset` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
