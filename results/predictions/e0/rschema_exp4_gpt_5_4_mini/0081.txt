-- Schema for database: rschema_0081
CREATE TABLE `user_groups` (
  `user_group_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_group_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `one_time_password` varchar(255) DEFAULT NULL,
  `password_reset_key` varchar(255) DEFAULT NULL,
  `user_group_id` int(11) DEFAULT NULL,
  `activation_status` varchar(50) NOT NULL,
  `created_at` datetime NOT NULL,
  `created_by` int(11) DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  `updated_by` int(11) DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  KEY `users_ibfk_1` (`user_group_id`),
  KEY `users_ibfk_2` (`created_by`),
  KEY `users_ibfk_3` (`updated_by`),
  CONSTRAINT `users_ibfk_1` FOREIGN KEY (`user_group_id`) REFERENCES `user_groups` (`user_group_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `users_ibfk_2` FOREIGN KEY (`created_by`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `users_ibfk_3` FOREIGN KEY (`updated_by`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_keys` (
  `key_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `key_content` varchar(255) NOT NULL,
  `permission_level` varchar(50) NOT NULL,
  `ignore_call_limits` tinyint(1) NOT NULL DEFAULT '0',
  `is_private_key` tinyint(1) NOT NULL DEFAULT '0',
  `allowed_ip_address_list` varchar(1000) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`key_id`),
  KEY `api_keys_ibfk_1` (`user_id`),
  CONSTRAINT `api_keys_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_call_logs` (
  `log_id` int(11) NOT NULL,
  `api_resource_path` varchar(255) NOT NULL,
  `http_request_method` varchar(20) NOT NULL,
  `request_parameters` text DEFAULT NULL,
  `key_id` int(11) DEFAULT NULL,
  `client_ip_address` varchar(45) NOT NULL,
  `call_time` datetime NOT NULL,
  `response_time` datetime DEFAULT NULL,
  `authorization_status` varchar(50) NOT NULL,
  `api_response_status_code` int(11) DEFAULT NULL,
  PRIMARY KEY (`log_id`),
  KEY `api_call_logs_ibfk_1` (`key_id`),
  CONSTRAINT `api_call_logs_ibfk_1` FOREIGN KEY (`key_id`) REFERENCES `api_keys` (`key_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
