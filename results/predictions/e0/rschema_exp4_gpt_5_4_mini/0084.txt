-- Schema for database: rschema_0084
CREATE TABLE `land_parcel` (
  `land_id` int(11) NOT NULL,
  `land_location` varchar(255) NOT NULL,
  `total_land_area` decimal(10,2) NOT NULL,
  PRIMARY KEY (`land_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  `purpose` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_usage_record` (
  `usage_record_id` int(11) NOT NULL,
  `land_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `used_land_area` decimal(10,2) NOT NULL,
  `usage_duration` varchar(100) NOT NULL,
  `current_status` varchar(50) NOT NULL,
  `record_time` datetime NOT NULL,
  `recording_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`usage_record_id`),
  KEY `land_usage_record_ibfk_1` (`land_id`),
  KEY `land_usage_record_ibfk_2` (`project_id`),
  CONSTRAINT `land_usage_record_ibfk_1` FOREIGN KEY (`land_id`) REFERENCES `land_parcel` (`land_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `land_usage_record_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
