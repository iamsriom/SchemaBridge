-- Schema for database: rschema_0085
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `profile` text DEFAULT NULL,
  `avatar` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `username` (`username`),
  UNIQUE KEY `email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipe` (
  `recipe_id` int(11) NOT NULL AUTO_INCREMENT,
  `recipe_name` varchar(255) NOT NULL,
  `ingredients` text NOT NULL,
  `steps` text NOT NULL,
  `total_time` int(11) DEFAULT NULL,
  `portion` int(11) DEFAULT NULL,
  `spicy_level` int(11) DEFAULT NULL,
  `bitter_level` int(11) DEFAULT NULL,
  `sweet_level` int(11) DEFAULT NULL,
  `meaty_level` int(11) DEFAULT NULL,
  `salty_level` int(11) DEFAULT NULL,
  `sour_level` int(11) DEFAULT NULL,
  `rating` decimal(3,2) DEFAULT NULL,
  `number_of_ratings` int(11) DEFAULT NULL,
  `large_image_filename` varchar(255) DEFAULT NULL,
  `calories` int(11) DEFAULT NULL,
  `creator_id` int(11) NOT NULL,
  `small_image_filename` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`recipe_id`),
  KEY `creator_id` (`creator_id`),
  CONSTRAINT `recipe_ibfk_1` FOREIGN KEY (`creator_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `blog` (
  `blog_id` int(11) NOT NULL AUTO_INCREMENT,
  `author_user_id` int(11) NOT NULL,
  `blog_title` varchar(255) NOT NULL,
  `publication_time` datetime NOT NULL,
  `blog_summary` text DEFAULT NULL,
  PRIMARY KEY (`blog_id`),
  KEY `author_user_id` (`author_user_id`),
  CONSTRAINT `blog_ibfk_1` FOREIGN KEY (`author_user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipe_collection` (
  `user_id` int(11) NOT NULL,
  `recipe_id` int(11) NOT NULL,
  `collected_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`,`recipe_id`),
  KEY `recipe_id` (`recipe_id`),
  CONSTRAINT `recipe_collection_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `recipe_collection_ibfk_2` FOREIGN KEY (`recipe_id`) REFERENCES `recipe` (`recipe_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipe_rating` (
  `user_id` int(11) NOT NULL,
  `recipe_id` int(11) NOT NULL,
  `rating_value` decimal(3,2) NOT NULL,
  `rated_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`,`recipe_id`),
  KEY `recipe_id` (`recipe_id`),
  CONSTRAINT `recipe_rating_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `recipe_rating_ibfk_2` FOREIGN KEY (`recipe_id`) REFERENCES `recipe` (`recipe_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
