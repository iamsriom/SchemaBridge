-- Schema for database: rschema_0096
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `hair_color` varchar(100) DEFAULT NULL,
  `appointment_time` datetime NOT NULL,
  `hairstylist_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`customer_id`),
  KEY `customers_ibfk_1` (`hairstylist_id`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`hairstylist_id`) REFERENCES `hairstylists` (`hairstylist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `hairstylists` (
  `hairstylist_id` int(11) NOT NULL AUTO_INCREMENT,
  `hairstylist_name` varchar(255) NOT NULL,
  `expertise` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`hairstylist_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `services` (
  `service_id` int(11) NOT NULL AUTO_INCREMENT,
  `hairstylist_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `service_time` datetime NOT NULL,
  `service_content` text NOT NULL,
  PRIMARY KEY (`service_id`),
  KEY `services_ibfk_1` (`hairstylist_id`),
  KEY `services_ibfk_2` (`customer_id`),
  CONSTRAINT `services_ibfk_1` FOREIGN KEY (`hairstylist_id`) REFERENCES `hairstylists` (`hairstylist_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `services_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
