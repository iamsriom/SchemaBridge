-- Schema for database: rschema_0098
CREATE TABLE `technology_solutions` (
  `solution_id` int(11) NOT NULL AUTO_INCREMENT,
  `solution_name` varchar(255) NOT NULL,
  `solution_type` varchar(100) NOT NULL,
  `vendor_name` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `selection_date` date DEFAULT NULL,
  `implementation_start_date` date DEFAULT NULL,
  `implementation_end_date` date DEFAULT NULL,
  `maintenance_status` varchar(50) DEFAULT NULL,
  `current_status` varchar(50) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`solution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `solution_evaluations` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `solution_id` int(11) NOT NULL,
  `evaluation_date` date NOT NULL,
  `evaluator_name` varchar(255) DEFAULT NULL,
  `cost_effectiveness_score` decimal(5,2) DEFAULT NULL,
  `implementation_progress_score` decimal(5,2) DEFAULT NULL,
  `user_feedback_score` decimal(5,2) DEFAULT NULL,
  `overall_score` decimal(5,2) DEFAULT NULL,
  `evaluation_notes` text DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  KEY `solution_evaluations_ibfk_1` (`solution_id`),
  CONSTRAINT `solution_evaluations_ibfk_1` FOREIGN KEY (`solution_id`) REFERENCES `technology_solutions` (`solution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
