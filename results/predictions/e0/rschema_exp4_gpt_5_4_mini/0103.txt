-- Schema for database: rschema_0103
CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `employee_name` varchar(100) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `hire_date` date NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_code` varchar(20) NOT NULL,
  `department_name` varchar(100) NOT NULL,
  `person_in_charge_employee_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`department_code`),
  KEY `department_ibfk_1` (`person_in_charge_employee_id`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`person_in_charge_employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_department` (
  `employee_department_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `department_code` varchar(20) NOT NULL,
  `entry_date` date NOT NULL,
  `departure_date` date DEFAULT NULL,
  PRIMARY KEY (`employee_department_id`),
  KEY `employee_department_ibfk_1` (`employee_id`),
  KEY `employee_department_ibfk_2` (`department_code`),
  CONSTRAINT `employee_department_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_department_ibfk_2` FOREIGN KEY (`department_code`) REFERENCES `department` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position_change_history` (
  `position_change_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `original_department_code` varchar(20) DEFAULT NULL,
  `original_position_name` varchar(100) DEFAULT NULL,
  `current_department_code` varchar(20) DEFAULT NULL,
  `current_position_name` varchar(100) DEFAULT NULL,
  `entry_date` date NOT NULL,
  `departure_date` date DEFAULT NULL,
  PRIMARY KEY (`position_change_id`),
  KEY `position_change_history_ibfk_1` (`employee_id`),
  KEY `position_change_history_ibfk_2` (`original_department_code`),
  KEY `position_change_history_ibfk_3` (`current_department_code`),
  CONSTRAINT `position_change_history_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `position_change_history_ibfk_2` FOREIGN KEY (`original_department_code`) REFERENCES `department` (`department_code`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `position_change_history_ibfk_3` FOREIGN KEY (`current_department_code`) REFERENCES `department` (`department_code`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_standard` (
  `salary_standard_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `standard_description` varchar(255) NOT NULL,
  `standard_value` decimal(10,2) NOT NULL,
  `effective_date` date DEFAULT NULL,
  `end_date` date DEFAULT NULL,
  PRIMARY KEY (`salary_standard_id`),
  KEY `salary_standard_ibfk_1` (`employee_id`),
  CONSTRAINT `salary_standard_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_payment` (
  `salary_payment_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `salary_standard_id` int(11) DEFAULT NULL,
  `pay_period` char(7) NOT NULL,
  `pay_date` date NOT NULL,
  `amount_paid` decimal(10,2) NOT NULL,
  PRIMARY KEY (`salary_payment_id`),
  KEY `salary_payment_ibfk_1` (`employee_id`),
  KEY `salary_payment_ibfk_2` (`salary_standard_id`),
  CONSTRAINT `salary_payment_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `salary_payment_ibfk_2` FOREIGN KEY (`salary_standard_id`) REFERENCES `salary_standard` (`salary_standard_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
