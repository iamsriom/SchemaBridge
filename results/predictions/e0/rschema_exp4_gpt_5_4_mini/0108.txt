-- Schema for database: rschema_0108
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(100) NOT NULL,
  `department_responsibilities` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `gender` enum('female','male') NOT NULL DEFAULT 'female',
  `age` int(11) NOT NULL,
  `social_security_number` varchar(20) NOT NULL,
  `department_id` int(11) NOT NULL,
  `position` enum('staff','assistant','supervisor','manager','director','president') NOT NULL DEFAULT 'staff',
  `email_address` varchar(255) NOT NULL,
  `phone_number` varchar(30) NOT NULL,
  `base_salary` decimal(10,2) NOT NULL,
  PRIMARY KEY (`employee_id`),
  UNIQUE KEY `social_security_number` (`social_security_number`),
  KEY `department_id` (`department_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_chk_1` CHECK (`age` >= 18 AND `age` <= 100)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
