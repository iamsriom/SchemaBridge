-- Schema for database: rschema_0109
CREATE TABLE `members` (
  `member_id` int(11) NOT NULL AUTO_INCREMENT,
  `country_code` char(2) NOT NULL,
  `name` varchar(255) NOT NULL,
  `name_pronunciation` varchar(255) DEFAULT NULL,
  `company_name` varchar(255) DEFAULT NULL,
  `is_public_country_code` tinyint(1) NOT NULL DEFAULT 0,
  `is_public_name` tinyint(1) NOT NULL DEFAULT 0,
  `is_public_name_pronunciation` tinyint(1) NOT NULL DEFAULT 0,
  `is_public_company_name` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_data_cache` (
  `member_id` int(11) NOT NULL,
  `country_code` char(2) NOT NULL,
  `name` varchar(255) NOT NULL,
  `name_pronunciation` varchar(255) DEFAULT NULL,
  `company_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`member_id`),
  CONSTRAINT `member_data_cache_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_addresses` (
  `address_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `country` varchar(100) NOT NULL,
  `postal_code` varchar(20) NOT NULL,
  `address_line` varchar(255) NOT NULL,
  `telephone_number` varchar(30) DEFAULT NULL,
  PRIMARY KEY (`address_id`),
  KEY `member_id` (`member_id`),
  CONSTRAINT `member_addresses_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `members` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
