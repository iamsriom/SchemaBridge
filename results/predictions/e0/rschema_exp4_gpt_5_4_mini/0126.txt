-- Schema for database: rschema_0126
CREATE TABLE `death_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `deceased_name` varchar(100) NOT NULL,
  `deceased_id_number` varchar(32) NOT NULL,
  `inspector_id_number` varchar(32) NOT NULL,
  `death_time` datetime NOT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_deceased_id_number` (`deceased_id_number`),
  KEY `idx_inspector_id_number` (`inspector_id_number`),
  KEY `idx_death_time` (`death_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
