-- Schema for database: rschema_0132
CREATE TABLE `student_info` (
  `student_no` varchar(20) NOT NULL,
  `student_name` varchar(100) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `class_name` varchar(100) DEFAULT NULL,
  `enrollment_time` datetime DEFAULT NULL,
  `residence` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `course_info` (
  `course_id` varchar(20) NOT NULL,
  `course_name` varchar(255) NOT NULL,
  `course_description` text DEFAULT NULL,
  `course_status` varchar(20) NOT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `teacher_info` (
  `teacher_no` varchar(20) NOT NULL,
  `teacher_name` varchar(100) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `title` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`teacher_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `student_course_selection` (
  `selection_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_no` varchar(20) NOT NULL,
  `course_id` varchar(20) NOT NULL,
  `selection_time` datetime DEFAULT NULL,
  PRIMARY KEY (`selection_id`),
  UNIQUE KEY `uk_student_course` (`student_no`,`course_id`),
  KEY `student_course_selection_ibfk_2` (`course_id`),
  CONSTRAINT `student_course_selection_ibfk_1` FOREIGN KEY (`student_no`) REFERENCES `student_info` (`student_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_course_selection_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course_info` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `teacher_teaching` (
  `teaching_id` int(11) NOT NULL AUTO_INCREMENT,
  `teacher_no` varchar(20) NOT NULL,
  `course_id` varchar(20) NOT NULL,
  `class_start_time` datetime DEFAULT NULL,
  PRIMARY KEY (`teaching_id`),
  UNIQUE KEY `uk_teacher_course` (`teacher_no`,`course_id`),
  KEY `teacher_teaching_ibfk_2` (`course_id`),
  CONSTRAINT `teacher_teaching_ibfk_1` FOREIGN KEY (`teacher_no`) REFERENCES `teacher_info` (`teacher_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `teacher_teaching_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course_info` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
