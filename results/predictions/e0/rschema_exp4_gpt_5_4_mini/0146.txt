-- Schema for database: rschema_0146
CREATE TABLE `dormitory_info` (
  `dormitory_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_number` varchar(50) NOT NULL,
  `member_count` int(11) NOT NULL,
  `affiliated_college` varchar(255) DEFAULT NULL,
  `dormitory_type` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`dormitory_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_info` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `personal_profile` text DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitory_member` (
  `dormitory_member_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`dormitory_member_id`),
  KEY `dormitory_member_ibfk_1` (`dormitory_id`),
  KEY `dormitory_member_ibfk_2` (`user_id`),
  CONSTRAINT `dormitory_member_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory_info` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dormitory_member_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user_info` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `information_summary` (
  `summary_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `summary_content` text DEFAULT NULL,
  `summary_time` datetime DEFAULT NULL,
  PRIMARY KEY (`summary_id`),
  KEY `information_summary_ibfk_1` (`dormitory_id`),
  KEY `information_summary_ibfk_2` (`user_id`),
  CONSTRAINT `information_summary_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory_info` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `information_summary_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user_info` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
