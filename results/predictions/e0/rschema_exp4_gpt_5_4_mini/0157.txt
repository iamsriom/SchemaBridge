CREATE TABLE `school_info` (
  `school_id` int(11) NOT NULL AUTO_INCREMENT,
  `school_number` varchar(50) NOT NULL,
  `school_name` varchar(255) NOT NULL,
  `school_introduction` text DEFAULT NULL,
  `school_facilities` text DEFAULT NULL,
  PRIMARY KEY (`school_id`),
  UNIQUE KEY `school_number` (`school_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `branch_info` (
  `branch_id` int(11) NOT NULL AUTO_INCREMENT,

  PRIMARY KEY (`branch_id`),
  UNIQUE KEY `branch_number` (`branch_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `class_info` (
  `class_id` int(11) NOT NULL AUTO_INCREMENT,
  `class_number` varchar(50) NOT NULL,
  `class_name` varchar(255) NOT NULL,
  `class_introduction` text DEFAULT NULL,
  `branch_id` int(11) NOT NULL,
  `teacher_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`class_id`),
  UNIQUE KEY `class_number` (`class_number`),
  KEY `branch_id` (`branch_id`),
  KEY `teacher_id` (`teacher_id`),
  CONSTRAINT `class_info_ibfk_1` FOREIGN KEY (`branch_id`) REFERENCES `branch_info` (`branch_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `class_info_ibfk_2` FOREIGN KEY (`teacher_id`) REFERENCES `teacher_info` (`teacher_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
