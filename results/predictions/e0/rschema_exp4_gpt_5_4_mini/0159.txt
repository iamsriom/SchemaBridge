-- Schema for database: rschema_0159
CREATE TABLE `route` (
  `route_id` int(11) NOT NULL AUTO_INCREMENT,
  `route_number` varchar(50) NOT NULL,
  `start_point` varchar(255) NOT NULL,
  `end_point` varchar(255) NOT NULL,
  `passing_stations` text DEFAULT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`route_id`),
  UNIQUE KEY `uk_route_number` (`route_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_number` varchar(50) NOT NULL,
  `vehicle_type` varchar(100) NOT NULL,
  `vehicle_status` varchar(50) NOT NULL,
  `capacity` int(11) DEFAULT NULL,
  `license_plate` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`),
  UNIQUE KEY `uk_vehicle_number` (`vehicle_number`),
  UNIQUE KEY `uk_license_plate` (`license_plate`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_number` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `id_number` varchar(18) NOT NULL,
  `position` varchar(100) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `birth_date` date DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `id_card_recognition_source` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  UNIQUE KEY `uk_employee_number` (`employee_number`),
  UNIQUE KEY `uk_employee_id_number` (`id_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `passenger_user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_number` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `username` varchar(100) NOT NULL,
  `password_hash` varchar(255) NOT NULL,
  `registration_time` datetime NOT NULL,
  `registration_info` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_user_number` (`user_number`),
  UNIQUE KEY `uk_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcement` (
  `announcement_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `release_time` datetime NOT NULL,
  `publisher_employee_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`announcement_id`),
  KEY `idx_announcement_publisher_employee_id` (`publisher_employee_id`),
  CONSTRAINT `announcement_ibfk_1` FOREIGN KEY (`publisher_employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dispatch` (
  `dispatch_id` int(11) NOT NULL AUTO_INCREMENT,
  `route_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `driver_employee_id` int(11) NOT NULL,
  `dispatch_date` date NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `status` varchar(50) NOT NULL,
  `passenger_flow` int(11) DEFAULT NULL,
  `notes` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`dispatch_id`),
  KEY `idx_dispatch_route_id` (`route_id`),
  KEY `idx_dispatch_vehicle_id` (`vehicle_id`),
  KEY `idx_dispatch_driver_employee_id` (`driver_employee_id`),
  CONSTRAINT `dispatch_ibfk_1` FOREIGN KEY (`route_id`) REFERENCES `route` (`route_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dispatch_ibfk_2` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dispatch_ibfk_3` FOREIGN KEY (`driver_employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ride_registration` (
  `registration_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `route_id` int(11) NOT NULL,
  `dispatch_id` int(11) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  `ride_date` date DEFAULT NULL,
  `status` varchar(50) NOT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`registration_id`),
  KEY `idx_ride_registration_user_id` (`user_id`),
  KEY `idx_ride_registration_route_id` (`route_id`),
  KEY `idx_ride_registration_dispatch_id` (`dispatch_id`),
  CONSTRAINT `ride_registration_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `passenger_user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ride_registration_ibfk_2` FOREIGN KEY (`route_id`) REFERENCES `route` (`route_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ride_registration_ibfk_3` FOREIGN KEY (`dispatch_id`) REFERENCES `dispatch` (`dispatch_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trip_statistics` (
  `statistics_id` int(11) NOT NULL AUTO_INCREMENT,
  `stat_date` date NOT NULL,
  `time_slot` varchar(50) NOT NULL,
  `route_id` int(11) DEFAULT NULL,
  `trip_count` int(11) NOT NULL DEFAULT '0',
  `passenger_flow` int(11) NOT NULL DEFAULT '0',
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`statistics_id`),
  KEY `idx_trip_statistics_route_id` (`route_id`),
  CONSTRAINT `trip_statistics_ibfk_1` FOREIGN KEY (`route_id`) REFERENCES `route` (`route_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
