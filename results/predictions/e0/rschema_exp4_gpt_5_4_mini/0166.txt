-- Schema for database: rschema_0166
CREATE TABLE `patient` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) DEFAULT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `medical_records` text DEFAULT NULL,
  PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor` (
  `doctor_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`doctor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_image` (
  `image_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `image_type` varchar(255) DEFAULT NULL,
  `film_time` datetime DEFAULT NULL,
  `image_data` longblob DEFAULT NULL,
  `image_resolution` varchar(100) DEFAULT NULL,
  `image_enhancement_information` text DEFAULT NULL,
  PRIMARY KEY (`image_id`),
  KEY `medical_image_ibfk_1` (`patient_id`),
  CONSTRAINT `medical_image_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor_image_operation` (
  `operation_id` int(11) NOT NULL AUTO_INCREMENT,
  `doctor_id` int(11) NOT NULL,
  `image_id` int(11) NOT NULL,
  `operation_content` text DEFAULT NULL,
  `operation_time` datetime DEFAULT NULL,
  PRIMARY KEY (`operation_id`),
  KEY `doctor_image_operation_ibfk_1` (`doctor_id`),
  KEY `doctor_image_operation_ibfk_2` (`image_id`),
  CONSTRAINT `doctor_image_operation_ibfk_1` FOREIGN KEY (`doctor_id`) REFERENCES `doctor` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `doctor_image_operation_ibfk_2` FOREIGN KEY (`image_id`) REFERENCES `medical_image` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `image_upload` (
  `upload_id` int(11) NOT NULL AUTO_INCREMENT,
  `image_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `upload_time` datetime DEFAULT NULL,
  `server_path` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`upload_id`),
  KEY `image_upload_ibfk_1` (`image_id`),
  KEY `image_upload_ibfk_2` (`doctor_id`),
  CONSTRAINT `image_upload_ibfk_1` FOREIGN KEY (`image_id`) REFERENCES `medical_image` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `image_upload_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctor` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `image_sharing` (
  `sharing_id` int(11) NOT NULL AUTO_INCREMENT,
  `image_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `sharing_time` datetime DEFAULT NULL,
  `target_institution` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`sharing_id`),
  KEY `image_sharing_ibfk_1` (`image_id`),
  KEY `image_sharing_ibfk_2` (`doctor_id`),
  CONSTRAINT `image_sharing_ibfk_1` FOREIGN KEY (`image_id`) REFERENCES `medical_image` (`image_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `image_sharing_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctor` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
