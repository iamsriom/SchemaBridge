-- Schema for database: rschema_0175
CREATE TABLE `alumni` (
  `alumni_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `employer` varchar(255) DEFAULT NULL,
  `job_title` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `class` varchar(50) DEFAULT NULL,
  `enrollment_year` int(11) DEFAULT NULL,
  `id_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`alumni_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donation_project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `start_time` datetime DEFAULT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donation` (
  `donation_id` int(11) NOT NULL AUTO_INCREMENT,
  `alumni_id` int(11) NOT NULL,
  `donation_time` datetime NOT NULL,
  `donation_amount` decimal(12,2) NOT NULL,
  `donation_type` varchar(100) NOT NULL,
  `donation_usage` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`donation_id`),
  KEY `donation_ibfk_1` (`alumni_id`),
  CONSTRAINT `donation_ibfk_1` FOREIGN KEY (`alumni_id`) REFERENCES `alumni` (`alumni_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_beneficiary_unit` (
  `project_beneficiary_unit_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `beneficiary_unit_name` varchar(255) NOT NULL,
  `benefit_percentage` decimal(5,2) NOT NULL,
  `benefit_amount` decimal(12,2) NOT NULL,
  PRIMARY KEY (`project_beneficiary_unit_id`),
  KEY `project_beneficiary_unit_ibfk_1` (`project_id`),
  CONSTRAINT `project_beneficiary_unit_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `donation_project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
