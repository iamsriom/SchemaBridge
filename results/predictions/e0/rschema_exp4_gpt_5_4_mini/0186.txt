-- Schema for database: rschema_0186
CREATE TABLE `insured` (
  `social_security_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `identity_card_number` varchar(20) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`social_security_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `social_security_type` (
  `type_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `payment_detail` (
  `payment_id` int(11) NOT NULL,
  `social_security_id` int(11) NOT NULL,
  `payment_time` datetime NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `payer` varchar(255) NOT NULL,
  PRIMARY KEY (`payment_id`),
  KEY `payment_detail_ibfk_1` (`social_security_id`),
  CONSTRAINT `payment_detail_ibfk_1` FOREIGN KEY (`social_security_id`) REFERENCES `insured` (`social_security_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `receipt_detail` (
  `receipt_id` int(11) NOT NULL,
  `social_security_id` int(11) NOT NULL,
  `receipt_time` datetime NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `recipient` varchar(255) NOT NULL,
  PRIMARY KEY (`receipt_id`),
  KEY `receipt_detail_ibfk_1` (`social_security_id`),
  CONSTRAINT `receipt_detail_ibfk_1` FOREIGN KEY (`social_security_id`) REFERENCES `insured` (`social_security_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
