-- Schema for database: rschema_0189
CREATE TABLE `repair_station` (
  `station_id` int(11) NOT NULL AUTO_INCREMENT,
  `station_name` varchar(255) NOT NULL,
  PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_technician` (
  `technician_id` int(11) NOT NULL AUTO_INCREMENT,
  `station_id` int(11) NOT NULL,
  `technician_name` varchar(255) NOT NULL,
  `technician_contact_number` varchar(20) NOT NULL,
  `technician_skill_level` varchar(50) NOT NULL,
  PRIMARY KEY (`technician_id`),
  KEY `repair_technician_ibfk_1` (`station_id`),
  CONSTRAINT `repair_technician_ibfk_1` FOREIGN KEY (`station_id`) REFERENCES `repair_station` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `technician_id` int(11) NOT NULL,
  `license_plate_number` varchar(20) NOT NULL,
  `vehicle_model` varchar(255) NOT NULL,
  `vehicle_owner_name` varchar(255) NOT NULL,
  `owner_contact_number` varchar(20) NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(100) NOT NULL,
  `repair_duration` int(11) NOT NULL,
  `repair_cost` decimal(10,2) DEFAULT NULL,
  `repair_effect` varchar(255) DEFAULT NULL,
  `owner_evaluation` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`task_id`),
  KEY `repair_task_ibfk_1` (`technician_id`),
  CONSTRAINT `repair_task_ibfk_1` FOREIGN KEY (`technician_id`) REFERENCES `repair_technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
