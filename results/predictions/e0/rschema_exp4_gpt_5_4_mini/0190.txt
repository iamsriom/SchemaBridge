-- Schema for database: rschema_0190
CREATE TABLE `workstation` (
  `workstation_id` int(11) NOT NULL AUTO_INCREMENT,
  `workstation_name` varchar(255) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`workstation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technician` (
  `technician_id` int(11) NOT NULL AUTO_INCREMENT,
  `workstation_id` int(11) NOT NULL,
  `technician_name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `hire_date` datetime DEFAULT NULL,
  PRIMARY KEY (`technician_id`),
  KEY `technician_ibfk_1` (`workstation_id`),
  CONSTRAINT `technician_ibfk_1` FOREIGN KEY (`workstation_id`) REFERENCES `workstation` (`workstation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `technician_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone_number` varchar(20) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `model_number` varchar(255) NOT NULL,
  `fault_description` text NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(100) NOT NULL,
  `repair_duration` int(11) NOT NULL,
  `repair_cost` decimal(10,2) DEFAULT NULL,
  `repair_results` text DEFAULT NULL,
  `customer_feedback` text DEFAULT NULL,
  `customer_satisfaction_rating` decimal(3,2) DEFAULT NULL,
  `task_status` varchar(50) NOT NULL DEFAULT 'completed',
  PRIMARY KEY (`task_id`),
  KEY `repair_task_ibfk_1` (`technician_id`),
  CONSTRAINT `repair_task_ibfk_1` FOREIGN KEY (`technician_id`) REFERENCES `technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
