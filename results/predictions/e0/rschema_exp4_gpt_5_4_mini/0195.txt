-- Schema for database: rschema_0195
CREATE TABLE `properties` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  `sale_price` decimal(12,2) NOT NULL,
  `sale_status` varchar(50) NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_personnel` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_phone` varchar(20) NOT NULL,
  `commission_rate` decimal(5,2) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_phone` varchar(20) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sale_transactions` (
  `sale_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `sale_date` datetime NOT NULL,
  `sale_price` decimal(12,2) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`sale_id`),
  KEY `sale_transactions_ibfk_1` (`property_id`),
  KEY `sale_transactions_ibfk_2` (`employee_id`),
  KEY `sale_transactions_ibfk_3` (`client_id`),
  CONSTRAINT `sale_transactions_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `properties` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sale_transactions_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `sales_personnel` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sale_transactions_ibfk_3` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
