-- Schema for database: rschema_0196
CREATE TABLE `applicant` (
  `applicant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `identification_number` varchar(50) NOT NULL,
  PRIMARY KEY (`applicant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approver` (
  `approver_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`approver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_application` (
  `application_number` varchar(50) NOT NULL,
  `applicant_id` int(11) NOT NULL,
  `requested_loan_amount` decimal(15,2) NOT NULL,
  `loan_term` int(11) NOT NULL,
  `approval_status` varchar(50) NOT NULL,
  PRIMARY KEY (`application_number`),
  KEY `loan_application_ibfk_1` (`applicant_id`),
  CONSTRAINT `loan_application_ibfk_1` FOREIGN KEY (`applicant_id`) REFERENCES `applicant` (`applicant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval_form` (
  `approval_form_id` int(11) NOT NULL,
  `application_number` varchar(50) NOT NULL,
  `approval_date` datetime NOT NULL,
  `approval_result` varchar(50) NOT NULL,
  `approver_id` int(11) NOT NULL,
  `approval_stage` int(11) NOT NULL,
  PRIMARY KEY (`approval_form_id`),
  KEY `loan_approval_form_ibfk_1` (`application_number`),
  KEY `loan_approval_form_ibfk_2` (`approver_id`),
  CONSTRAINT `loan_approval_form_ibfk_1` FOREIGN KEY (`application_number`) REFERENCES `loan_application` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_form_ibfk_2` FOREIGN KEY (`approver_id`) REFERENCES `approver` (`approver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
