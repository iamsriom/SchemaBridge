-- Schema for database: rschema_0198
CREATE TABLE `community` (
  `community_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `property_manager` varchar(255) NOT NULL,
  PRIMARY KEY (`community_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property` (
  `property_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  `maintenance_status` varchar(50) NOT NULL,
  PRIMARY KEY (`property_id`),
  KEY `community_id` (`community_id`),
  CONSTRAINT `property_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `personnel_name` varchar(255) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `skill_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_task` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_id` int(11) NOT NULL,
  `personnel_id` int(11) DEFAULT NULL,
  `task_content` varchar(500) NOT NULL,
  `cost` decimal(10,2) DEFAULT NULL,
  `duration_hours` decimal(10,2) DEFAULT NULL,
  `status` varchar(50) NOT NULL,
  `start_time` datetime DEFAULT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`task_id`),
  KEY `property_id` (`property_id`),
  KEY `personnel_id` (`personnel_id`),
  CONSTRAINT `maintenance_task_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_task_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `maintenance_personnel` (`personnel_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
