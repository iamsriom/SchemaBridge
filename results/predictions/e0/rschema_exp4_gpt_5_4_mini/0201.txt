-- Schema for database: rschema_0201
CREATE TABLE `member_level` (
  `level_id` int(11) NOT NULL AUTO_INCREMENT,
  `level_name` varchar(100) NOT NULL,
  `points_min` int(11) NOT NULL,
  `points_max` int(11) NOT NULL,
  `shopping_frequency_min` int(11) NOT NULL,
  `shopping_frequency_max` int(11) NOT NULL,
  PRIMARY KEY (`level_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member` (
  `member_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `points` int(11) NOT NULL DEFAULT '0',
  `level_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`member_id`),
  KEY `member_ibfk_1` (`level_id`),
  CONSTRAINT `member_ibfk_1` FOREIGN KEY (`level_id`) REFERENCES `member_level` (`level_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shopping_record` (
  `shopping_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `shopping_time` datetime NOT NULL,
  `store_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `total_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`shopping_record_id`),
  KEY `shopping_record_ibfk_1` (`member_id`),
  KEY `shopping_record_ibfk_2` (`product_id`),
  CONSTRAINT `shopping_record_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `shopping_record_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
