-- Schema for database: rschema_0205
CREATE TABLE `performance` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `performance_name` varchar(255) NOT NULL,
  `performance_time` datetime NOT NULL,
  `location_id` int(11) NOT NULL,
  PRIMARY KEY (`performance_id`),
  KEY `location_id` (`location_id`),
  CONSTRAINT `performance_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `performance_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `actor` (
  `actor_id` int(11) NOT NULL AUTO_INCREMENT,
  `actor_name` varchar(255) NOT NULL,
  PRIMARY KEY (`actor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_location` (
  `location_id` int(11) NOT NULL AUTO_INCREMENT,
  `location_name` varchar(255) NOT NULL,
  `location_address` varchar(255) DEFAULT NULL,
  `location_detail` text DEFAULT NULL,
  PRIMARY KEY (`location_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_actor` (
  `performance_id` int(11) NOT NULL,
  `actor_id` int(11) NOT NULL,
  PRIMARY KEY (`performance_id`,`actor_id`),
  KEY `actor_id` (`actor_id`),
  CONSTRAINT `performance_actor_ibfk_1` FOREIGN KEY (`performance_id`) REFERENCES `performance` (`performance_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `performance_actor_ibfk_2` FOREIGN KEY (`actor_id`) REFERENCES `actor` (`actor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
