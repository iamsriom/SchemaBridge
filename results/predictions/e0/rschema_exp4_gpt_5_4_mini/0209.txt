-- Schema for database: rschema_0209
CREATE TABLE `laboratories` (
  `laboratory_id` int(11) NOT NULL AUTO_INCREMENT,
  `laboratory_number` varchar(50) NOT NULL,
  `laboratory_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`laboratory_id`),
  UNIQUE KEY `laboratory_number` (`laboratory_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL AUTO_INCREMENT,
  `laboratory_id` int(11) NOT NULL,
  `equipment_number` varchar(50) NOT NULL,
  `equipment_name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `purchase_date` date NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`equipment_id`),
  UNIQUE KEY `equipment_number` (`equipment_number`),
  KEY `laboratory_id` (`laboratory_id`),
  CONSTRAINT `equipment_ibfk_1` FOREIGN KEY (`laboratory_id`) REFERENCES `laboratories` (`laboratory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_maintenance_records` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `maintenance_date` date NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_content` text NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  KEY `equipment_id` (`equipment_id`),
  CONSTRAINT `equipment_maintenance_records_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
