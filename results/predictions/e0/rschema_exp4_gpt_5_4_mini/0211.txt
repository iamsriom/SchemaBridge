-- Schema for database: rschema_0211
CREATE TABLE `consulting_projects` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  `client_name` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultants` (
  `consultant_id` int(11) NOT NULL,
  `consultant_name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `professional_field` varchar(255) NOT NULL,
  PRIMARY KEY (`consultant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consulting_service_records` (
  `service_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `consultant_id` int(11) NOT NULL,
  `service_date` date NOT NULL,
  `service_content` text NOT NULL,
  PRIMARY KEY (`service_id`),
  KEY `consulting_service_records_ibfk_1` (`project_id`),
  KEY `consulting_service_records_ibfk_2` (`consultant_id`),
  CONSTRAINT `consulting_service_records_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `consulting_projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `consulting_service_records_ibfk_2` FOREIGN KEY (`consultant_id`) REFERENCES `consultants` (`consultant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
