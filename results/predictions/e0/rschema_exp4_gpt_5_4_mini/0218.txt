-- Schema for database: rschema_0218
CREATE TABLE `printer_manufacturers` (
  `manufacturer_id` int(11) NOT NULL AUTO_INCREMENT,
  `manufacturer_name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`manufacturer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `printer_models` (
  `model_id` int(11) NOT NULL AUTO_INCREMENT,
  `manufacturer_id` int(11) NOT NULL,
  `model_name` varchar(255) NOT NULL,
  `standard_driver_config` text DEFAULT NULL,
  PRIMARY KEY (`model_id`),
  KEY `printer_models_ibfk_1` (`manufacturer_id`),
  CONSTRAINT `printer_models_ibfk_1` FOREIGN KEY (`manufacturer_id`) REFERENCES `printer_manufacturers` (`manufacturer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `printer_devices` (
  `device_id` int(11) NOT NULL AUTO_INCREMENT,
  `model_id` int(11) NOT NULL,
  `installation_location` varchar(255) NOT NULL,
  `network_hostname` varchar(255) NOT NULL,
  `personalized_driver_settings` text DEFAULT NULL,
  PRIMARY KEY (`device_id`),
  KEY `printer_devices_ibfk_1` (`model_id`),
  CONSTRAINT `printer_devices_ibfk_1` FOREIGN KEY (`model_id`) REFERENCES `printer_models` (`model_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
