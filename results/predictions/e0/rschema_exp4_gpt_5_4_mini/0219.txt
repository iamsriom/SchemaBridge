-- Schema for database: rschema_0219
CREATE TABLE `card` (
  `card_id` int(11) NOT NULL AUTO_INCREMENT,
  `card_name` varchar(255) NOT NULL,
  `card_type` varchar(255) NOT NULL,
  `card_artist` varchar(255) DEFAULT NULL,
  `card_effect` text DEFAULT NULL,
  `card_power` int(11) DEFAULT NULL,
  `card_intelligence` int(11) DEFAULT NULL,
  `card_endurance` int(11) DEFAULT NULL,
  PRIMARY KEY (`card_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `card_collection` (
  `collection_id` int(11) NOT NULL AUTO_INCREMENT,
  `collection_name` varchar(255) NOT NULL,
  PRIMARY KEY (`collection_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `card_collection_association` (
  `card_id` int(11) NOT NULL,
  `collection_id` int(11) NOT NULL,
  `collection_number` int(11) DEFAULT NULL,
  `card_rarity` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`card_id`,`collection_id`),
  KEY `card_collection_association_ibfk_2` (`collection_id`),
  CONSTRAINT `card_collection_association_ibfk_1` FOREIGN KEY (`card_id`) REFERENCES `card` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `card_collection_association_ibfk_2` FOREIGN KEY (`collection_id`) REFERENCES `card_collection` (`collection_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `user_exchange_card` (
  `user_1_id` int(11) NOT NULL,
  `user_2_id` int(11) NOT NULL,
  `exchange_time` datetime NOT NULL,
  `exchanged_card_id` int(11) NOT NULL,
  PRIMARY KEY (`user_1_id`,`user_2_id`,`exchange_time`,`exchanged_card_id`),
  KEY `user_exchange_card_ibfk_2` (`user_2_id`),
  KEY `user_exchange_card_ibfk_3` (`exchanged_card_id`),
  CONSTRAINT `user_exchange_card_ibfk_1` FOREIGN KEY (`user_1_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_exchange_card_ibfk_2` FOREIGN KEY (`user_2_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_exchange_card_ibfk_3` FOREIGN KEY (`exchanged_card_id`) REFERENCES `card` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
