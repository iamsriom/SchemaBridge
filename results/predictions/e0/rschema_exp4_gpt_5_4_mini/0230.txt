-- Schema for database: rschema_0230
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(50) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mechanics` (
  `mechanic_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`mechanic_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_models` (
  `vehicle_model_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`vehicle_model_id`),
  UNIQUE KEY `uk_vehicle_models_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_tasks` (
  `maintenance_task_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `vehicle_model_id` int(11) NOT NULL,
  `mechanic_id` int(11) DEFAULT NULL,
  `status` varchar(50) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  PRIMARY KEY (`maintenance_task_id`),
  KEY `maintenance_tasks_ibfk_1` (`customer_id`),
  KEY `maintenance_tasks_ibfk_2` (`vehicle_model_id`),
  KEY `maintenance_tasks_ibfk_3` (`mechanic_id`),
  CONSTRAINT `maintenance_tasks_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_tasks_ibfk_2` FOREIGN KEY (`vehicle_model_id`) REFERENCES `vehicle_models` (`vehicle_model_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_tasks_ibfk_3` FOREIGN KEY (`mechanic_id`) REFERENCES `mechanics` (`mechanic_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parts` (
  `part_id` int(11) NOT NULL AUTO_INCREMENT,
  `serial_number` varchar(100) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `stock_quantity` int(11) NOT NULL,
  `supplier` varchar(255) NOT NULL,
  PRIMARY KEY (`part_id`),
  UNIQUE KEY `uk_parts_serial_number` (`serial_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `part_orders` (
  `part_order_id` int(11) NOT NULL AUTO_INCREMENT,
  `maintenance_task_id` int(11) NOT NULL,
  `part_id` int(11) NOT NULL,
  `dispatch_date` datetime DEFAULT NULL,
  `delivered` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`part_order_id`),
  KEY `part_orders_ibfk_1` (`maintenance_task_id`),
  KEY `part_orders_ibfk_2` (`part_id`),
  CONSTRAINT `part_orders_ibfk_1` FOREIGN KEY (`maintenance_task_id`) REFERENCES `maintenance_tasks` (`maintenance_task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `part_orders_ibfk_2` FOREIGN KEY (`part_id`) REFERENCES `parts` (`part_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
