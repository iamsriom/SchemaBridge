-- Schema for database: rschema_0231
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `contact_email` varchar(255) NOT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_users_username` (`username`),
  UNIQUE KEY `uk_users_contact_email` (`contact_email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `experience_records` (
  `experience_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` text NOT NULL,
  `code_content` longtext NOT NULL,
  `creator_user_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `modification_time` datetime NOT NULL,
  `status` char(1) NOT NULL DEFAULT 'F',
  PRIMARY KEY (`experience_id`),
  KEY `experience_records_ibfk_1` (`creator_user_id`),
  CONSTRAINT `experience_records_ibfk_1` FOREIGN KEY (`creator_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
