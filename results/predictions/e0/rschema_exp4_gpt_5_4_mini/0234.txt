CREATE TABLE `tag_info` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `primary_category` varchar(255) NOT NULL,
  `primary_code` int(11) NOT NULL,
  `secondary_category` varchar(255) NOT NULL,
  `secondary_code` int(11) NOT NULL,
  `status` tinyint(4) NOT NULL DEFAULT '1',
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_primary_category_primary_code` (`primary_category`,`primary_code`),
  KEY `idx_secondary_category_secondary_code` (`secondary_category`,`secondary_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
