-- Schema for database: rschema_0251
CREATE TABLE `groomers` (
  `groomer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specialization` varchar(255) DEFAULT NULL,
  `years_of_experience` int(11) DEFAULT NULL,
  `rating` decimal(3,2) DEFAULT NULL,
  PRIMARY KEY (`groomer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone` varchar(50) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pets` (
  `pet_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `breed` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `weight` decimal(10,2) DEFAULT NULL,
  `fur_characteristics` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pet_id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `pets_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grooming_services` (
  `service_id` int(11) NOT NULL AUTO_INCREMENT,
  `pet_id` int(11) NOT NULL,
  `groomer_id` int(11) NOT NULL,
  `service_items` varchar(255) NOT NULL,
  `service_duration` int(11) NOT NULL,
  `pricing` decimal(10,2) NOT NULL,
  `customer_rating` decimal(3,2) DEFAULT NULL,
  PRIMARY KEY (`service_id`),
  KEY `pet_id` (`pet_id`),
  KEY `groomer_id` (`groomer_id`),
  CONSTRAINT `grooming_services_ibfk_1` FOREIGN KEY (`pet_id`) REFERENCES `pets` (`pet_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grooming_services_ibfk_2` FOREIGN KEY (`groomer_id`) REFERENCES `groomers` (`groomer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
