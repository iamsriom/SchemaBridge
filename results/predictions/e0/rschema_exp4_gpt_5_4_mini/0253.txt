-- Schema for database: rschema_0253
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `group_chats` (
  `group_chat_id` int(11) NOT NULL AUTO_INCREMENT,
  `group_chat_name` varchar(255) NOT NULL,
  `creator_user_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `member_count` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`group_chat_id`),
  KEY `group_chats_ibfk_1` (`creator_user_id`),
  CONSTRAINT `group_chats_ibfk_1` FOREIGN KEY (`creator_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chat_records` (
  `chat_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `sender_user_id` int(11) NOT NULL,
  `receiver_user_id` int(11) NOT NULL,
  `sent_time` datetime NOT NULL,
  `message_type` varchar(50) NOT NULL,
  `content` text NOT NULL,
  `read_time` datetime DEFAULT NULL,
  PRIMARY KEY (`chat_record_id`),
  KEY `chat_records_ibfk_1` (`sender_user_id`),
  KEY `chat_records_ibfk_2` (`receiver_user_id`),
  CONSTRAINT `chat_records_ibfk_1` FOREIGN KEY (`sender_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `chat_records_ibfk_2` FOREIGN KEY (`receiver_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `group_chat_joins` (
  `join_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `group_chat_id` int(11) NOT NULL,
  `join_time` datetime NOT NULL,
  `application_notes` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`join_id`),
  KEY `group_chat_joins_ibfk_1` (`user_id`),
  KEY `group_chat_joins_ibfk_2` (`group_chat_id`),
  CONSTRAINT `group_chat_joins_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `group_chat_joins_ibfk_2` FOREIGN KEY (`group_chat_id`) REFERENCES `group_chats` (`group_chat_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `group_chat_leaves` (
  `leave_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `group_chat_id` int(11) NOT NULL,
  `leave_time` datetime NOT NULL,
  PRIMARY KEY (`leave_id`),
  KEY `group_chat_leaves_ibfk_1` (`user_id`),
  KEY `group_chat_leaves_ibfk_2` (`group_chat_id`),
  CONSTRAINT `group_chat_leaves_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `group_chat_leaves_ibfk_2` FOREIGN KEY (`group_chat_id`) REFERENCES `group_chats` (`group_chat_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
