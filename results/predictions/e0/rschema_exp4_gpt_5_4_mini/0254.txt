-- Schema for database: rschema_0254
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(100) NOT NULL,
  `password` varchar(255) NOT NULL,
  `name` varchar(100) NOT NULL,
  `station_name` varchar(255) DEFAULT NULL,
  `role` varchar(50) NOT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `station_name` varchar(255) DEFAULT NULL,
  `position` varchar(100) DEFAULT NULL,
  `phone` varchar(50) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `seismometer` (
  `seismometer_id` int(11) NOT NULL AUTO_INCREMENT,
  `station_name` varchar(255) NOT NULL,
  `seismometer_name` varchar(255) NOT NULL,
  `model` varchar(100) DEFAULT NULL,
  `serial_number` varchar(100) DEFAULT NULL,
  `voltage_sensitivity` decimal(10,4) DEFAULT NULL,
  `coil_sensitivity` decimal(10,4) DEFAULT NULL,
  `moment_of_inertia` decimal(18,6) DEFAULT NULL,
  `equivalent_pendulum_length` decimal(18,6) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`seismometer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `calibration_record` (
  `calibration_id` int(11) NOT NULL AUTO_INCREMENT,
  `seismometer_id` int(11) NOT NULL,
  `user_id` int(11) DEFAULT NULL,
  `calibration_time` datetime NOT NULL,
  `amplitude` decimal(10,4) DEFAULT NULL,
  `width` decimal(10,4) DEFAULT NULL,
  `mechanical_period` decimal(10,4) DEFAULT NULL,
  `equivalent_pendulum_length` decimal(18,6) DEFAULT NULL,
  `moment_of_inertia` decimal(18,6) DEFAULT NULL,
  `coil_sensitivity` decimal(10,4) DEFAULT NULL,
  `voltage_sensitivity` decimal(10,4) DEFAULT NULL,
  `comparison_result` text DEFAULT NULL,
  `vibration_table_test_result` text DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`calibration_id`),
  KEY `seismometer_id` (`seismometer_id`),
  KEY `user_id` (`user_id`),
  CONSTRAINT `calibration_record_ibfk_1` FOREIGN KEY (`seismometer_id`) REFERENCES `seismometer` (`seismometer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `calibration_record_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
