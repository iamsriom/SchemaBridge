-- Schema for database: rschema_0260
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  `registration_eligibility` varchar(20) NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL,
  `course_name` varchar(255) NOT NULL,
  `credits` int(11) NOT NULL,
  `instructor` varchar(255) DEFAULT NULL,
  `class_time` varchar(100) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `degree_examination_result` (
  `result_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  `examination_name` varchar(255) NOT NULL,
  `grade` varchar(20) NOT NULL,
  `examination_date` date DEFAULT NULL,
  PRIMARY KEY (`result_id`),
  KEY `degree_examination_result_ibfk_1` (`student_id`),
  CONSTRAINT `degree_examination_result_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `registration` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `registration_time` datetime NOT NULL,
  `status` varchar(20) NOT NULL,
  PRIMARY KEY (`student_id`,`course_id`),
  KEY `registration_ibfk_2` (`course_id`),
  CONSTRAINT `registration_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `registration_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
