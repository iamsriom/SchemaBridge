-- Schema for database: rschema_0261
CREATE TABLE `district` (
  `district_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(20) NOT NULL,
  PRIMARY KEY (`district_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `address` (
  `address_id` int(11) NOT NULL,
  `detailed_address` varchar(255) NOT NULL,
  `type` varchar(20) NOT NULL,
  `district_id` int(11) NOT NULL,
  PRIMARY KEY (`address_id`),
  KEY `address_ibfk_1` (`district_id`),
  CONSTRAINT `address_ibfk_1` FOREIGN KEY (`district_id`) REFERENCES `district` (`district_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voter` (
  `voter_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address_id` int(11) NOT NULL,
  `registration_status` varchar(20) NOT NULL,
  `district_id` int(11) NOT NULL,
  PRIMARY KEY (`voter_id`),
  UNIQUE KEY `voter_address_id_unique` (`address_id`),
  KEY `voter_ibfk_1` (`address_id`),
  KEY `voter_ibfk_2` (`district_id`),
  CONSTRAINT `voter_ibfk_1` FOREIGN KEY (`address_id`) REFERENCES `address` (`address_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `voter_ibfk_2` FOREIGN KEY (`district_id`) REFERENCES `district` (`district_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
