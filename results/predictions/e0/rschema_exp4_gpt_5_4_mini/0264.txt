-- Schema for database: rschema_0264
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `manager_employee_no` int(11) NOT NULL,
  PRIMARY KEY (`department_id`),
  UNIQUE KEY `uk_department_manager_employee_no` (`manager_employee_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_no` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `position` varchar(50) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `salary` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`employee_no`),
  KEY `employee_ibfk_1` (`department_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `room` (
  `room_no` varchar(20) NOT NULL,
  `room_type` varchar(50) NOT NULL,
  `charging_standard` decimal(10,2) NOT NULL,
  `check_in_status` varchar(20) NOT NULL,
  PRIMARY KEY (`room_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_no` int(11) NOT NULL AUTO_INCREMENT,
  `unit_name` varchar(255) NOT NULL,
  `contact_person` varchar(255) NOT NULL,
  `contact_number` varchar(20) NOT NULL,
  `contact_address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reservation_application` (
  `application_no` int(11) NOT NULL AUTO_INCREMENT,
  `customer_no` int(11) NOT NULL,
  `check_in_time` datetime NOT NULL,
  `number_of_days` int(11) NOT NULL,
  `room_type` varchar(50) NOT NULL,
  `number_of_rooms` int(11) NOT NULL,
  PRIMARY KEY (`application_no`),
  KEY `reservation_application_ibfk_1` (`customer_no`),
  CONSTRAINT `reservation_application_ibfk_1` FOREIGN KEY (`customer_no`) REFERENCES `customer` (`customer_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `room_arrangement` (
  `room_no` varchar(20) NOT NULL,
  `application_no` int(11) NOT NULL,
  `salesperson_employee_no` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `id_number` varchar(30) NOT NULL,
  `check_in_time` datetime NOT NULL,
  `number_of_days` int(11) NOT NULL,
  `telephone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`room_no`,`id_number`,`check_in_time`),
  KEY `room_arrangement_ibfk_1` (`room_no`),
  KEY `room_arrangement_ibfk_2` (`application_no`),
  KEY `room_arrangement_ibfk_3` (`salesperson_employee_no`),
  CONSTRAINT `room_arrangement_ibfk_1` FOREIGN KEY (`room_no`) REFERENCES `room` (`room_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `room_arrangement_ibfk_2` FOREIGN KEY (`application_no`) REFERENCES `reservation_application` (`application_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `room_arrangement_ibfk_3` FOREIGN KEY (`salesperson_employee_no`) REFERENCES `employee` (`employee_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
