-- Schema for database: rschema_0270
CREATE TABLE `program` (
  `program_id` int(11) NOT NULL,
  `program_name` varchar(255) NOT NULL,
  `program_type` varchar(100) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `duration_minutes` int(11) NOT NULL,
  PRIMARY KEY (`program_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_staff` (
  `staff_id` int(11) NOT NULL,
  `staff_name` varchar(255) NOT NULL,
  `staff_role` varchar(100) NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviewer` (
  `reviewer_id` int(11) NOT NULL,
  `reviewer_name` varchar(255) NOT NULL,
  PRIMARY KEY (`reviewer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_production` (
  `production_id` int(11) NOT NULL,
  `program_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  `review_result` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`production_id`),
  KEY `program_id` (`program_id`),
  CONSTRAINT `program_production_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `program` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_production_staff` (
  `production_id` int(11) NOT NULL,
  `staff_id` int(11) NOT NULL,
  PRIMARY KEY (`production_id`,`staff_id`),
  KEY `staff_id` (`staff_id`),
  CONSTRAINT `program_production_staff_ibfk_1` FOREIGN KEY (`production_id`) REFERENCES `program_production` (`production_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_production_staff_ibfk_2` FOREIGN KEY (`staff_id`) REFERENCES `production_staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_review` (
  `review_id` int(11) NOT NULL,
  `production_id` int(11) NOT NULL,
  `reviewer_id` int(11) NOT NULL,
  `review_time` datetime NOT NULL,
  `review_result` varchar(100) NOT NULL,
  PRIMARY KEY (`review_id`),
  KEY `production_id` (`production_id`),
  KEY `reviewer_id` (`reviewer_id`),
  CONSTRAINT `program_review_ibfk_1` FOREIGN KEY (`production_id`) REFERENCES `program_production` (`production_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_review_ibfk_2` FOREIGN KEY (`reviewer_id`) REFERENCES `reviewer` (`reviewer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
