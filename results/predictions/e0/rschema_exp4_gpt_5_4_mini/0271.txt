-- Schema for database: rschema_0271
CREATE TABLE `services` (
  `service_id` int(11) NOT NULL,
  `service_name` varchar(255) NOT NULL,
  `transmission_type` varchar(100) NOT NULL,
  `transmission_rate` varchar(100) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`service_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL,
  `client_name` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_subscriptions` (
  `subscription_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `service_id` int(11) NOT NULL,
  `subscription_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `operator` varchar(255) NOT NULL,
  PRIMARY KEY (`subscription_id`),
  KEY `service_subscriptions_ibfk_1` (`client_id`),
  KEY `service_subscriptions_ibfk_2` (`service_id`),
  CONSTRAINT `service_subscriptions_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_subscriptions_ibfk_2` FOREIGN KEY (`service_id`) REFERENCES `services` (`service_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
