-- Schema for database: rschema_0275
CREATE TABLE `ecosystems` (
  `system_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `current_status` varchar(100) NOT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operators` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(20) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `system_id` int(11) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `temperature` decimal(5,2) NOT NULL,
  `humidity` decimal(5,2) NOT NULL,
  `soil_moisture` decimal(5,2) NOT NULL,
  `plant_growth` varchar(255) NOT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  KEY `monitoring_records_ibfk_1` (`system_id`),
  KEY `monitoring_records_ibfk_2` (`operator_id`),
  CONSTRAINT `monitoring_records_ibfk_1` FOREIGN KEY (`system_id`) REFERENCES `ecosystems` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_records_ibfk_2` FOREIGN KEY (`operator_id`) REFERENCES `operators` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
