-- Schema for database: rschema_0281
CREATE TABLE `exhibitions` (
  `exhibition_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`exhibition_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `curators` (
  `curator_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `information` text DEFAULT NULL,
  PRIMARY KEY (`curator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exhibits` (
  `exhibit_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `information` text DEFAULT NULL,
  PRIMARY KEY (`exhibit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exhibition_curators` (
  `exhibition_id` int(11) NOT NULL,
  `curator_id` int(11) NOT NULL,
  PRIMARY KEY (`exhibition_id`,`curator_id`),
  KEY `exhibition_curators_ibfk_2` (`curator_id`),
  CONSTRAINT `exhibition_curators_ibfk_1` FOREIGN KEY (`exhibition_id`) REFERENCES `exhibitions` (`exhibition_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `exhibition_curators_ibfk_2` FOREIGN KEY (`curator_id`) REFERENCES `curators` (`curator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exhibition_exhibits` (
  `exhibition_id` int(11) NOT NULL,
  `exhibit_id` int(11) NOT NULL,
  PRIMARY KEY (`exhibition_id`,`exhibit_id`),
  KEY `exhibition_exhibits_ibfk_2` (`exhibit_id`),
  CONSTRAINT `exhibition_exhibits_ibfk_1` FOREIGN KEY (`exhibition_id`) REFERENCES `exhibitions` (`exhibition_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `exhibition_exhibits_ibfk_2` FOREIGN KEY (`exhibit_id`) REFERENCES `exhibits` (`exhibit_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
