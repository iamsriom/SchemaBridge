-- Schema for database: rschema_0284
CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_name` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisements` (
  `advertisement_id` int(11) NOT NULL AUTO_INCREMENT,
  `advertisement_name` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `broadcast_time` datetime NOT NULL,
  PRIMARY KEY (`advertisement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `broadcast_time_slots` (
  `time_slot_id` int(11) NOT NULL AUTO_INCREMENT,
  `slot_start` datetime NOT NULL,
  `slot_end` datetime NOT NULL,
  `arrangement` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`time_slot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisement_clients` (
  `advertisement_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`advertisement_id`,`client_id`),
  KEY `advertisement_clients_ibfk_2` (`client_id`),
  CONSTRAINT `advertisement_clients_ibfk_1` FOREIGN KEY (`advertisement_id`) REFERENCES `advertisements` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `advertisement_clients_ibfk_2` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisement_time_slots` (
  `advertisement_id` int(11) NOT NULL,
  `time_slot_id` int(11) NOT NULL,
  PRIMARY KEY (`advertisement_id`,`time_slot_id`),
  KEY `advertisement_time_slots_ibfk_2` (`time_slot_id`),
  CONSTRAINT `advertisement_time_slots_ibfk_1` FOREIGN KEY (`advertisement_id`) REFERENCES `advertisements` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `advertisement_time_slots_ibfk_2` FOREIGN KEY (`time_slot_id`) REFERENCES `broadcast_time_slots` (`time_slot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
