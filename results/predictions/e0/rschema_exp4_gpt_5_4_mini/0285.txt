-- Schema for database: rschema_0285
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `contact_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisements` (
  `advertisement_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `advertisement_name` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `duration_seconds` int(11) NOT NULL,
  `broadcast_schedule` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`advertisement_id`),
  KEY `advertisements_ibfk_1` (`customer_id`),
  CONSTRAINT `advertisements_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisement_broadcasts` (
  `broadcast_id` int(11) NOT NULL AUTO_INCREMENT,
  `advertisement_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `broadcast_duration_seconds` int(11) NOT NULL,
  `ratings` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`broadcast_id`),
  KEY `advertisement_broadcasts_ibfk_1` (`advertisement_id`),
  CONSTRAINT `advertisement_broadcasts_ibfk_1` FOREIGN KEY (`advertisement_id`) REFERENCES `advertisements` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
