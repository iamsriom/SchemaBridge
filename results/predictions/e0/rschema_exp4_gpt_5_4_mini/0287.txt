-- Schema for database: rschema_0287
CREATE TABLE `park` (
  `park_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `area` decimal(10,2) DEFAULT NULL,
  `responsible_person` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`park_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `park_plant_record` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `park_id` int(11) NOT NULL,
  `recording_time` datetime NOT NULL,
  `recorder` varchar(255) NOT NULL,
  `plant_species` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `growth_status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  KEY `park_plant_record_ibfk_1` (`park_id`),
  CONSTRAINT `park_plant_record_ibfk_1` FOREIGN KEY (`park_id`) REFERENCES `park` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `park_maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `park_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(500) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_cost` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`maintenance_id`),
  KEY `park_maintenance_ibfk_1` (`park_id`),
  CONSTRAINT `park_maintenance_ibfk_1` FOREIGN KEY (`park_id`) REFERENCES `park` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
