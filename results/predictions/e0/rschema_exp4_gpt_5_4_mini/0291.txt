-- Schema for database: rschema_0291
CREATE TABLE `reagent` (
  `reagent_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specifications` varchar(255) DEFAULT NULL,
  `inventory_quantity` int(11) NOT NULL DEFAULT 0,
  `expiration_date` date NOT NULL,
  PRIMARY KEY (`reagent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `laboratory` (
  `laboratory_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`laboratory_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reagent_usage` (
  `usage_id` int(11) NOT NULL AUTO_INCREMENT,
  `reagent_id` int(11) NOT NULL,
  `laboratory_id` int(11) NOT NULL,
  `usage_date` datetime NOT NULL,
  `user` varchar(255) NOT NULL,
  `usage_quantity` int(11) NOT NULL,
  PRIMARY KEY (`usage_id`),
  KEY `reagent_usage_ibfk_1` (`reagent_id`),
  KEY `reagent_usage_ibfk_2` (`laboratory_id`),
  CONSTRAINT `reagent_usage_ibfk_1` FOREIGN KEY (`reagent_id`) REFERENCES `reagent` (`reagent_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reagent_usage_ibfk_2` FOREIGN KEY (`laboratory_id`) REFERENCES `laboratory` (`laboratory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
