-- Schema for database: rschema_0292
CREATE TABLE `facilities` (
  `facility_id` int(11) NOT NULL,
  `facility_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`facility_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `personnel_id` int(11) NOT NULL,
  `personnel_name` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_events` (
  `maintenance_event_id` int(11) NOT NULL,
  `facility_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `content` varchar(500) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_event_id`),
  KEY `maintenance_events_ibfk_1` (`facility_id`),
  KEY `maintenance_events_ibfk_2` (`personnel_id`),
  CONSTRAINT `maintenance_events_ibfk_1` FOREIGN KEY (`facility_id`) REFERENCES `facilities` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_events_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `maintenance_personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
