-- Schema for database: rschema_0295
CREATE TABLE `traders` (
  `trader_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `hire_date` date NOT NULL,
  `contact_number` varchar(50) NOT NULL,
  PRIMARY KEY (`trader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `id_number` varchar(50) NOT NULL,
  `contact_number` varchar(50) NOT NULL,
  `risk_level` varchar(50) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `futures_transactions` (
  `transaction_id` int(11) NOT NULL,
  `trader_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `futures_code` varchar(50) NOT NULL,
  `transaction_type` varchar(10) NOT NULL,
  `transaction_quantity` int(11) NOT NULL,
  `transaction_price` decimal(10,2) NOT NULL,
  `transaction_time` datetime NOT NULL,
  PRIMARY KEY (`transaction_id`),
  KEY `trader_id` (`trader_id`),
  KEY `client_id` (`client_id`),
  CONSTRAINT `futures_transactions_ibfk_1` FOREIGN KEY (`trader_id`) REFERENCES `traders` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `futures_transactions_ibfk_2` FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trader_performance` (
  `performance_id` int(11) NOT NULL,
  `trader_id` int(11) NOT NULL,
  `total_transactions` int(11) NOT NULL,
  `total_volume` int(11) NOT NULL,
  `total_profit` decimal(12,2) NOT NULL,
  `performance_date` date NOT NULL,
  PRIMARY KEY (`performance_id`),
  KEY `trader_id` (`trader_id`),
  CONSTRAINT `trader_performance_ibfk_1` FOREIGN KEY (`trader_id`) REFERENCES `traders` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
