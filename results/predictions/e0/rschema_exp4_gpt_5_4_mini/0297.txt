CREATE TABLE `document_types` (
  `document_type_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `version_number` int(11) NOT NULL,
  PRIMARY KEY (`document_type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status_categories` (
  `status_category_id` int(11) NOT NULL AUTO_INCREMENT,
  `document_type_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `version_number` int(11) NOT NULL,
  PRIMARY KEY (`status_category_id`),
  KEY `status_categories_ibfk_1` (`document_type_id`),
  CONSTRAINT `status_categories_ibfk_1` FOREIGN KEY (`document_type_id`) REFERENCES `document_types` (`document_type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `statuses` (
  `status_id` int(11) NOT NULL AUTO_INCREMENT,
  `status_category_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `sequence_number` int(11) NOT NULL,
  PRIMARY KEY (`status_id`),
  KEY `statuses_ibfk_1` (`status_category_id`),
  CONSTRAINT `statuses_ibfk_1` FOREIGN KEY (`status_category_id`) REFERENCES `status_categories` (`status_category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
