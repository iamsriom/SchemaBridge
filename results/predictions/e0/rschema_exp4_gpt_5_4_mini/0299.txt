-- Schema for database: rschema_0299
CREATE TABLE `city` (
  `city_id` int(11) NOT NULL AUTO_INCREMENT,
  `city_name` varchar(255) NOT NULL,
  `longitude` decimal(10,6) NOT NULL,
  `latitude` decimal(10,6) NOT NULL,
  `country` varchar(100) NOT NULL,
  `population` int(11) DEFAULT NULL,
  `timezone` varchar(100) NOT NULL,
  PRIMARY KEY (`city_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather_forecast` (
  `forecast_id` int(11) NOT NULL AUTO_INCREMENT,
  `city_id` int(11) NOT NULL,
  `forecast_date` date NOT NULL,
  `sunrise_time` time DEFAULT NULL,
  `sunset_time` time DEFAULT NULL,
  `temperature_morning` decimal(5,2) DEFAULT NULL,
  `temperature_day` decimal(5,2) DEFAULT NULL,
  `temperature_evening` decimal(5,2) DEFAULT NULL,
  `temperature_night` decimal(5,2) DEFAULT NULL,
  `feels_like_morning` decimal(5,2) DEFAULT NULL,
  `feels_like_day` decimal(5,2) DEFAULT NULL,
  `feels_like_evening` decimal(5,2) DEFAULT NULL,
  `feels_like_night` decimal(5,2) DEFAULT NULL,
  `pressure` int(11) DEFAULT NULL,
  `humidity` int(11) DEFAULT NULL,
  `wind_speed` decimal(5,2) DEFAULT NULL,
  `wind_direction` int(11) DEFAULT NULL,
  `cloud_cover` int(11) DEFAULT NULL,
  `precipitation_probability` decimal(5,2) DEFAULT NULL,
  `weather_main` varchar(100) DEFAULT NULL,
  `weather_description` varchar(255) DEFAULT NULL,
  `weather_icon` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`forecast_id`),
  KEY `city_id` (`city_id`),
  CONSTRAINT `weather_forecast_ibfk_1` FOREIGN KEY (`city_id`) REFERENCES `city` (`city_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
