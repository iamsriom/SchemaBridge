-- Schema for database: rschema_0301
CREATE TABLE `seat_layout_template` (
  `template_id` int(11) NOT NULL AUTO_INCREMENT,
  `template_name` varchar(255) NOT NULL,
  `template_description` varchar(500) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`template_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `seat` (
  `seat_id` int(11) NOT NULL AUTO_INCREMENT,
  `template_id` int(11) NOT NULL,
  `seat_x` int(11) NOT NULL,
  `seat_y` int(11) NOT NULL,
  `seat_row` varchar(50) NOT NULL,
  `seat_column` varchar(50) NOT NULL,
  `seat_type` varchar(100) NOT NULL,
  `seat_price` decimal(10,2) NOT NULL,
  `is_valid` char(1) NOT NULL DEFAULT '1',
  `seat_status` varchar(20) NOT NULL DEFAULT 'available',
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`seat_id`),
  KEY `seat_ibfk_1` (`template_id`),
  CONSTRAINT `seat_ibfk_1` FOREIGN KEY (`template_id`) REFERENCES `seat_layout_template` (`template_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_name` varchar(255) NOT NULL,
  `contact_number` varchar(30) NOT NULL,
  `email` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `user_email_uk` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `seat_reservation` (
  `reservation_id` int(11) NOT NULL AUTO_INCREMENT,
  `seat_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `reservation_type` varchar(20) NOT NULL,
  `reservation_status` varchar(20) NOT NULL DEFAULT 'pending',
  `reserved_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`reservation_id`),
  KEY `seat_reservation_ibfk_1` (`seat_id`),
  KEY `seat_reservation_ibfk_2` (`user_id`),
  CONSTRAINT `seat_reservation_ibfk_1` FOREIGN KEY (`seat_id`) REFERENCES `seat` (`seat_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `seat_reservation_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
