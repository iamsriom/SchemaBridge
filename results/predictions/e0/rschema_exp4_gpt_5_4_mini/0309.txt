-- Schema for database: rschema_0309
CREATE TABLE `regions` (
  `region_id` int(11) NOT NULL AUTO_INCREMENT,
  `region_name` varchar(255) NOT NULL,
  PRIMARY KEY (`region_id`),
  UNIQUE KEY `region_name` (`region_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `income_groups` (
  `income_group_id` int(11) NOT NULL AUTO_INCREMENT,
  `income_group_name` varchar(255) NOT NULL,
  `min_income` decimal(12,2) DEFAULT NULL,
  `max_income` decimal(12,2) DEFAULT NULL,
  PRIMARY KEY (`income_group_id`),
  UNIQUE KEY `income_group_name` (`income_group_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_segments` (
  `vehicle_segment_id` int(11) NOT NULL AUTO_INCREMENT,
  `segment_name` varchar(255) NOT NULL,
  `segment_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`vehicle_segment_id`),
  UNIQUE KEY `segment_name` (`segment_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `marital_status` varchar(50) DEFAULT NULL,
  `region_id` int(11) NOT NULL,
  `income_group_id` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`),
  UNIQUE KEY `email` (`email`),
  KEY `region_id` (`region_id`),
  KEY `income_group_id` (`income_group_id`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customers_ibfk_2` FOREIGN KEY (`income_group_id`) REFERENCES `income_groups` (`income_group_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `policy_attributes` (
  `policy_attribute_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_segment_id` int(11) NOT NULL,
  `attribute_name` varchar(255) NOT NULL,
  `premium_amount` decimal(12,2) NOT NULL,
  `attribute_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`policy_attribute_id`),
  KEY `vehicle_segment_id` (`vehicle_segment_id`),
  CONSTRAINT `policy_attributes_ibfk_1` FOREIGN KEY (`vehicle_segment_id`) REFERENCES `vehicle_segments` (`vehicle_segment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `policies` (
  `policy_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `vehicle_segment_id` int(11) NOT NULL,
  `policy_number` varchar(100) NOT NULL,
  `purchase_date` date NOT NULL,
  `expiration_date` date NOT NULL,
  PRIMARY KEY (`policy_id`),
  UNIQUE KEY `policy_number` (`policy_number`),
  KEY `customer_id` (`customer_id`),
  KEY `vehicle_segment_id` (`vehicle_segment_id`),
  CONSTRAINT `policies_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `policies_ibfk_2` FOREIGN KEY (`vehicle_segment_id`) REFERENCES `vehicle_segments` (`vehicle_segment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `policy_attribute_purchases` (
  `policy_attribute_purchase_id` int(11) NOT NULL AUTO_INCREMENT,
  `policy_id` int(11) NOT NULL,
  `policy_attribute_id` int(11) NOT NULL,
  `premium_amount` decimal(12,2) NOT NULL,
  PRIMARY KEY (`policy_attribute_purchase_id`),
  KEY `policy_id` (`policy_id`),
  KEY `policy_attribute_id` (`policy_attribute_id`),
  CONSTRAINT `policy_attribute_purchases_ibfk_1` FOREIGN KEY (`policy_id`) REFERENCES `policies` (`policy_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `policy_attribute_purchases_ibfk_2` FOREIGN KEY (`policy_attribute_id`) REFERENCES `policy_attributes` (`policy_attribute_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
