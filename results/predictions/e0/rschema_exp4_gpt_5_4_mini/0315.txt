-- Schema for database: rschema_0315
CREATE TABLE `personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `personnel_type` varchar(50) NOT NULL,
  `name` varchar(100) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `department` varchar(100) DEFAULT NULL,
  `title` varchar(100) DEFAULT NULL,
  `student_number` varchar(50) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `book` (
  `book_id` int(11) NOT NULL AUTO_INCREMENT,
  `isbn` varchar(20) NOT NULL,
  `title` varchar(255) NOT NULL,
  `author` varchar(255) DEFAULT NULL,
  `publisher` varchar(255) DEFAULT NULL,
  `publication_date` date DEFAULT NULL,
  `category` varchar(100) DEFAULT NULL,
  `edition` varchar(50) DEFAULT NULL,
  `total_copies` int(11) NOT NULL DEFAULT '0',
  `available_copies` int(11) NOT NULL DEFAULT '0',
  `location` varchar(100) DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`book_id`),
  UNIQUE KEY `uk_book_isbn` (`isbn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `book_copy` (
  `copy_id` int(11) NOT NULL AUTO_INCREMENT,
  `book_id` int(11) NOT NULL,
  `copy_code` varchar(50) NOT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'available',
  `shelf_location` varchar(100) DEFAULT NULL,
  `acquired_at` date DEFAULT NULL,
  PRIMARY KEY (`copy_id`),
  UNIQUE KEY `uk_book_copy_code` (`copy_code`),
  KEY `book_copy_ibfk_1` (`book_id`),
  CONSTRAINT `book_copy_ibfk_1` FOREIGN KEY (`book_id`) REFERENCES `book` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `borrowing` (
  `borrowing_id` int(11) NOT NULL AUTO_INCREMENT,
  `personnel_id` int(11) NOT NULL,
  `copy_id` int(11) NOT NULL,
  `borrow_time` datetime NOT NULL,
  `due_time` datetime NOT NULL,
  `return_time` datetime DEFAULT NULL,
  `return_status` varchar(20) NOT NULL DEFAULT 'borrowed',
  `remarks` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`borrowing_id`),
  KEY `borrowing_ibfk_1` (`personnel_id`),
  KEY `borrowing_ibfk_2` (`copy_id`),
  CONSTRAINT `borrowing_ibfk_1` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `borrowing_ibfk_2` FOREIGN KEY (`copy_id`) REFERENCES `book_copy` (`copy_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `return_record` (
  `return_id` int(11) NOT NULL AUTO_INCREMENT,
  `borrowing_id` int(11) NOT NULL,
  `copy_id` int(11) NOT NULL,
  `return_date` datetime NOT NULL,
  `condition_note` varchar(255) DEFAULT NULL,
  `processed_by` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`return_id`),
  UNIQUE KEY `uk_return_borrowing` (`borrowing_id`),
  KEY `return_record_ibfk_2` (`copy_id`),
  CONSTRAINT `return_record_ibfk_1` FOREIGN KEY (`borrowing_id`) REFERENCES `borrowing` (`borrowing_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `return_record_ibfk_2` FOREIGN KEY (`copy_id`) REFERENCES `book_copy` (`copy_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
