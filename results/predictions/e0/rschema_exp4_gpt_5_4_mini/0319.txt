-- Schema for database: rschema_0319
CREATE TABLE `regions` (
  `regional_id` int(11) NOT NULL AUTO_INCREMENT,
  `regional_code` varchar(50) NOT NULL,
  `city_code` varchar(50) DEFAULT NULL,
  `center_coordinates` varchar(100) DEFAULT NULL,
  `regional_level` int(11) DEFAULT NULL,
  `merged_name` varchar(255) DEFAULT NULL,
  `regional_name` varchar(255) NOT NULL,
  `parent_regional_id` int(11) DEFAULT NULL,
  `pinyin` varchar(255) DEFAULT NULL,
  `abbreviation` varchar(100) DEFAULT NULL,
  `postal_code` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`regional_id`),
  UNIQUE KEY `uk_regions_regional_code` (`regional_code`),
  KEY `regions_ibfk_1` (`parent_regional_id`),
  CONSTRAINT `regions_ibfk_1` FOREIGN KEY (`parent_regional_id`) REFERENCES `regions` (`regional_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
