-- Schema for database: rschema_0320
CREATE TABLE `product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `product_slogan` varchar(255) DEFAULT NULL,
  `product_description` text DEFAULT NULL,
  `product_category` varchar(255) DEFAULT NULL,
  `default_price` varchar(50) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `style` (
  `style_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `style_name` varchar(255) NOT NULL,
  `sale_price` varchar(50) DEFAULT NULL,
  `original_price` varchar(50) DEFAULT NULL,
  `is_default` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`style_id`),
  KEY `style_ibfk_1` (`product_id`),
  CONSTRAINT `style_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `feature` (
  `feature_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_id` int(11) NOT NULL,
  `feature_name` varchar(255) NOT NULL,
  `feature_value` varchar(255) NOT NULL,
  PRIMARY KEY (`feature_id`),
  KEY `feature_ibfk_1` (`product_id`),
  CONSTRAINT `feature_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sku` (
  `sku_id` int(11) NOT NULL AUTO_INCREMENT,
  `style_id` int(11) NOT NULL,
  `size` varchar(50) NOT NULL,
  `stock_quantity` int(11) NOT NULL,
  PRIMARY KEY (`sku_id`),
  KEY `sku_ibfk_1` (`style_id`),
  CONSTRAINT `sku_ibfk_1` FOREIGN KEY (`style_id`) REFERENCES `style` (`style_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `photo` (
  `photo_id` int(11) NOT NULL AUTO_INCREMENT,
  `style_id` int(11) NOT NULL,
  `photo_url` varchar(2083) NOT NULL,
  `thumbnail_url` varchar(2083) DEFAULT NULL,
  PRIMARY KEY (`photo_id`),
  KEY `photo_ibfk_1` (`style_id`),
  CONSTRAINT `photo_ibfk_1` FOREIGN KEY (`style_id`) REFERENCES `style` (`style_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
