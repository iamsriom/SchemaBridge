-- Schema for database: rschema_0323
CREATE TABLE `salespeople` (
  `salesperson_id` int(11) NOT NULL AUTO_INCREMENT,
  `salesperson_name` varchar(255) NOT NULL,
  PRIMARY KEY (`salesperson_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_records` (
  `sales_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `salesperson_id` int(11) NOT NULL,
  `customer_id` int(11) DEFAULT NULL,
  `sales_amount` decimal(10,2) NOT NULL,
  `sales_date` datetime NOT NULL,
  `transaction_quantity` int(11) NOT NULL,
  `visit_count` int(11) NOT NULL,
  PRIMARY KEY (`sales_record_id`),
  KEY `sales_records_ibfk_1` (`salesperson_id`),
  KEY `sales_records_ibfk_2` (`customer_id`),
  CONSTRAINT `sales_records_ibfk_1` FOREIGN KEY (`salesperson_id`) REFERENCES `salespeople` (`salesperson_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_records_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
