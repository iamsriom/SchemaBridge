-- Schema for database: rschema_0333
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `user_type` varchar(50) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone` varchar(50) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `birthday` date DEFAULT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `relationship_status` varchar(50) DEFAULT NULL,
  `interests` varchar(255) DEFAULT NULL,
  `profile` text DEFAULT NULL,
  `motto` varchar(255) DEFAULT NULL,
  `hometown` varchar(255) DEFAULT NULL,
  `current_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `connections` (
  `connection_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id_1` int(11) NOT NULL,
  `user_id_2` int(11) NOT NULL,
  `connected_at` datetime NOT NULL,
  PRIMARY KEY (`connection_id`),
  KEY `user_id_1` (`user_id_1`),
  KEY `user_id_2` (`user_id_2`),
  CONSTRAINT `connections_ibfk_1` FOREIGN KEY (`user_id_1`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `connections_ibfk_2` FOREIGN KEY (`user_id_2`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `connection_requests` (
  `request_id` int(11) NOT NULL AUTO_INCREMENT,
  `sender_user_id` int(11) NOT NULL,
  `receiver_user_id` int(11) NOT NULL,
  `sent_at` datetime NOT NULL,
  `request_status` varchar(20) NOT NULL,
  PRIMARY KEY (`request_id`),
  KEY `sender_user_id` (`sender_user_id`),
  KEY `receiver_user_id` (`receiver_user_id`),
  CONSTRAINT `connection_requests_ibfk_1` FOREIGN KEY (`sender_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `connection_requests_ibfk_2` FOREIGN KEY (`receiver_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homepages` (
  `homepage_id` int(11) NOT NULL AUTO_INCREMENT,
  `owner_user_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `introduction` text DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`homepage_id`),
  KEY `owner_user_id` (`owner_user_id`),
  CONSTRAINT `homepages_ibfk_1` FOREIGN KEY (`owner_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homepage_members` (
  `homepage_member_id` int(11) NOT NULL AUTO_INCREMENT,
  `homepage_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `joined_at` datetime NOT NULL,
  PRIMARY KEY (`homepage_member_id`),
  KEY `homepage_id` (`homepage_id`),
  KEY `user_id` (`user_id`),
  CONSTRAINT `homepage_members_ibfk_1` FOREIGN KEY (`homepage_id`) REFERENCES `homepages` (`homepage_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `homepage_members_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homepage_posts` (
  `post_id` int(11) NOT NULL AUTO_INCREMENT,
  `homepage_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `posted_at` datetime NOT NULL,
  PRIMARY KEY (`post_id`),
  KEY `homepage_id` (`homepage_id`),
  KEY `user_id` (`user_id`),
  CONSTRAINT `homepage_posts_ibfk_1` FOREIGN KEY (`homepage_id`) REFERENCES `homepages` (`homepage_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `homepage_posts_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
