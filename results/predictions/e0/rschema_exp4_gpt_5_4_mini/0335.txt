-- Schema for database: rschema_0335
CREATE TABLE `oss_bucket` (
  `id` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `oss_access` (
  `id` int(11) NOT NULL,
  `bucket_id` int(11) NOT NULL,
  `type` varchar(50) NOT NULL,
  `access_key` varchar(255) NOT NULL,
  `secret_key` varchar(255) NOT NULL,
  `status` varchar(20) NOT NULL,
  `create_time` datetime NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  `user_id` int(11) DEFAULT NULL,
  `tenant_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `oss_access_ibfk_1` (`bucket_id`),
  CONSTRAINT `oss_access_ibfk_1` FOREIGN KEY (`bucket_id`) REFERENCES `oss_bucket` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
