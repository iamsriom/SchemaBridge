-- Schema for database: rschema_0342
CREATE TABLE `cors_network` (
  `network_id` int(11) NOT NULL,
  `network_name` varchar(255) NOT NULL,
  `network_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`network_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cors_station` (
  `station_id` int(11) NOT NULL,
  PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cors_station_network_relationship` (
  `relationship_id` int(11) NOT NULL,
  `cors_station_id` int(11) NOT NULL,
  `cors_network_id` int(11) NOT NULL,
  `effective_start_time` datetime NOT NULL,
  `effective_end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`relationship_id`),
  KEY `cors_station_network_relationship_ibfk_1` (`cors_station_id`),
  KEY `cors_station_network_relationship_ibfk_2` (`cors_network_id`),
  CONSTRAINT `cors_station_network_relationship_ibfk_1` FOREIGN KEY (`cors_station_id`) REFERENCES `cors_station` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `cors_station_network_relationship_ibfk_2` FOREIGN KEY (`cors_network_id`) REFERENCES `cors_network` (`network_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
