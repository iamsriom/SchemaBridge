-- Schema for database: rschema_0343
CREATE TABLE `personnel` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `street` varchar(255) DEFAULT NULL,
  `city` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel_phone` (
  `id` int(11) NOT NULL,
  `number` varchar(50) NOT NULL,
  PRIMARY KEY (`id`,`number`),
  CONSTRAINT `personnel_phone_ibfk_1` FOREIGN KEY (`id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vin` char(17) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `year` int(11) DEFAULT NULL,
  `driver_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`vin`),
  KEY `driver_id` (`driver_id`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`driver_id`) REFERENCES `personnel` (`id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_insurance` (
  `insurance_id` int(11) NOT NULL AUTO_INCREMENT,
  `vin` char(17) NOT NULL,
  `insurance_type_id` int(11) NOT NULL,
  `insurance_company_name` varchar(255) NOT NULL,
  `insurance_cost` decimal(10,2) NOT NULL,
  `insurance_content` text DEFAULT NULL,
  `purchase_time` datetime NOT NULL,
  `valid_from` date NOT NULL,
  `valid_to` date NOT NULL,
  PRIMARY KEY (`insurance_id`),
  KEY `vin` (`vin`),
  CONSTRAINT `vehicle_insurance_ibfk_1` FOREIGN KEY (`vin`) REFERENCES `vehicle` (`vin`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
