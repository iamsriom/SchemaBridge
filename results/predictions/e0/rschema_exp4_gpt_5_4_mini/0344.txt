-- Schema for database: rschema_0344
CREATE TABLE `applications` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `code` varchar(100) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  PRIMARY KEY (`application_id`),
  UNIQUE KEY `uk_applications_code` (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `roles` (
  `role_id` int(11) NOT NULL AUTO_INCREMENT,
  `role_name` varchar(100) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`role_id`),
  UNIQUE KEY `uk_roles_role_name` (`role_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `permissions` (
  `permission_id` int(11) NOT NULL AUTO_INCREMENT,
  `permission_name` varchar(100) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`permission_id`),
  UNIQUE KEY `uk_permissions_permission_name` (`permission_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(100) NOT NULL,
  `email` varchar(255) NOT NULL,
  `role_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_users_username` (`username`),
  UNIQUE KEY `uk_users_email` (`email`),
  KEY `users_ibfk_1` (`role_id`),
  CONSTRAINT `users_ibfk_1` FOREIGN KEY (`role_id`) REFERENCES `roles` (`role_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role_permissions` (
  `role_id` int(11) NOT NULL,
  `permission_id` int(11) NOT NULL,
  PRIMARY KEY (`role_id`,`permission_id`),
  KEY `role_permissions_ibfk_2` (`permission_id`),
  CONSTRAINT `role_permissions_ibfk_1` FOREIGN KEY (`role_id`) REFERENCES `roles` (`role_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `role_permissions_ibfk_2` FOREIGN KEY (`permission_id`) REFERENCES `permissions` (`permission_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_application_access` (
  `user_id` int(11) NOT NULL,
  `application_id` int(11) NOT NULL,
  `authorized_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`,`application_id`),
  KEY `user_application_access_ibfk_2` (`application_id`),
  CONSTRAINT `user_application_access_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_application_access_ibfk_2` FOREIGN KEY (`application_id`) REFERENCES `applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `application_user_permissions` (
  `application_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `permission_id` int(11) NOT NULL,
  PRIMARY KEY (`application_id`,`user_id`,`permission_id`),
  KEY `application_user_permissions_ibfk_2` (`user_id`),
  KEY `application_user_permissions_ibfk_3` (`permission_id`),
  CONSTRAINT `application_user_permissions_ibfk_1` FOREIGN KEY (`application_id`) REFERENCES `applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `application_user_permissions_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `application_user_permissions_ibfk_3` FOREIGN KEY (`permission_id`) REFERENCES `permissions` (`permission_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `web_services` (
  `web_service_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  `creation_date` datetime NOT NULL,
  PRIMARY KEY (`web_service_id`),
  KEY `web_services_ibfk_1` (`application_id`),
  CONSTRAINT `web_services_ibfk_1` FOREIGN KEY (`application_id`) REFERENCES `applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `web_service_errors` (
  `error_id` int(11) NOT NULL AUTO_INCREMENT,
  `web_service_id` int(11) NOT NULL,
  `error_code` varchar(50) NOT NULL,
  `error_description` varchar(500) DEFAULT NULL,
  `error_timestamp` datetime NOT NULL,
  `return_status` varchar(50) NOT NULL,
  PRIMARY KEY (`error_id`),
  KEY `web_service_errors_ibfk_1` (`web_service_id`),
  CONSTRAINT `web_service_errors_ibfk_1` FOREIGN KEY (`web_service_id`) REFERENCES `web_services` (`web_service_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `web_service_monitoring` (
  `monitor_id` int(11) NOT NULL AUTO_INCREMENT,
  `web_service_id` int(11) NOT NULL,
  `invocation_time` datetime NOT NULL,
  `return_status` varchar(50) NOT NULL,
  `delay_time_ms` int(11) DEFAULT NULL,
  `success_rate` decimal(5,2) DEFAULT NULL,
  `call_count` int(11) DEFAULT NULL,
  `time_period_start` datetime DEFAULT NULL,
  `time_period_end` datetime DEFAULT NULL,
  PRIMARY KEY (`monitor_id`),
  KEY `web_service_monitoring_ibfk_1` (`web_service_id`),
  CONSTRAINT `web_service_monitoring_ibfk_1` FOREIGN KEY (`web_service_id`) REFERENCES `web_services` (`web_service_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_error_links` (
  `monitor_id` int(11) NOT NULL,
  `error_id` int(11) NOT NULL,
  PRIMARY KEY (`monitor_id`,`error_id`),
  KEY `monitoring_error_links_ibfk_2` (`error_id`),
  CONSTRAINT `monitoring_error_links_ibfk_1` FOREIGN KEY (`monitor_id`) REFERENCES `web_service_monitoring` (`monitor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_error_links_ibfk_2` FOREIGN KEY (`error_id`) REFERENCES `web_service_errors` (`error_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
