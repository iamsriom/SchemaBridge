-- Schema for database: rschema_0348
CREATE TABLE `plugin_execution_log` (
  `execution_id` int(11) NOT NULL AUTO_INCREMENT,
  `plugin_version` varchar(100) NOT NULL,
  `plugin_type` varchar(100) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `sql_statement` text NOT NULL,
  PRIMARY KEY (`execution_id`),
  KEY `idx_plugin_execution_log_time` (`start_time`,`end_time`),
  KEY `idx_plugin_execution_log_type` (`plugin_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
