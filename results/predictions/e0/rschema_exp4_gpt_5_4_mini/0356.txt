-- Schema for database: rschema_0356
CREATE TABLE `work_items` (
  `work_item_id` int(11) NOT NULL AUTO_INCREMENT,
  `work_item_name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `creation_date` datetime NOT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`work_item_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `link_statuses` (
  `link_status_id` int(11) NOT NULL AUTO_INCREMENT,
  `status_name` varchar(255) NOT NULL,
  `status_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`link_status_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `work_item_attributes` (
  `attribute_id` int(11) NOT NULL AUTO_INCREMENT,
  `work_item_id` int(11) NOT NULL,
  `attribute_name` varchar(255) NOT NULL,
  `attribute_value` varchar(255) DEFAULT NULL,
  `link_status_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`attribute_id`),
  KEY `work_item_attributes_ibfk_1` (`work_item_id`),
  KEY `work_item_attributes_ibfk_2` (`link_status_id`),
  CONSTRAINT `work_item_attributes_ibfk_1` FOREIGN KEY (`work_item_id`) REFERENCES `work_items` (`work_item_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `work_item_attributes_ibfk_2` FOREIGN KEY (`link_status_id`) REFERENCES `link_statuses` (`link_status_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
