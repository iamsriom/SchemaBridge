-- Schema for database: rschema_0361
CREATE TABLE `restaurant` (
  `restaurant_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `business_hours` varchar(255) DEFAULT NULL,
  `average_cost_per_person` decimal(10,2) DEFAULT NULL,
  `latitude` decimal(10,6) DEFAULT NULL,
  `longitude` decimal(10,6) DEFAULT NULL,
  `rating` decimal(3,2) DEFAULT NULL,
  `review_count` int(11) NOT NULL DEFAULT '0',
  PRIMARY KEY (`restaurant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `restaurant_tag` (
  `tag_id` int(11) NOT NULL AUTO_INCREMENT,
  `tag_name` varchar(100) NOT NULL,
  PRIMARY KEY (`tag_id`),
  UNIQUE KEY `uk_restaurant_tag_name` (`tag_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `restaurant_tag_map` (
  `restaurant_id` int(11) NOT NULL,
  `tag_id` int(11) NOT NULL,
  PRIMARY KEY (`restaurant_id`,`tag_id`),
  KEY `restaurant_tag_map_ibfk_2` (`tag_id`),
  CONSTRAINT `restaurant_tag_map_ibfk_1` FOREIGN KEY (`restaurant_id`) REFERENCES `restaurant` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `restaurant_tag_map_ibfk_2` FOREIGN KEY (`tag_id`) REFERENCES `restaurant_tag` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish` (
  `dish_id` int(11) NOT NULL AUTO_INCREMENT,
  `restaurant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `description` text DEFAULT NULL,
  `satisfaction_rate` decimal(5,2) DEFAULT NULL,
  `monthly_sales` int(11) NOT NULL DEFAULT '0',
  PRIMARY KEY (`dish_id`),
  KEY `dish_ibfk_1` (`restaurant_id`),
  CONSTRAINT `dish_ibfk_1` FOREIGN KEY (`restaurant_id`) REFERENCES `restaurant` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish_tag` (
  `tag_id` int(11) NOT NULL AUTO_INCREMENT,
  `tag_name` varchar(100) NOT NULL,
  PRIMARY KEY (`tag_id`),
  UNIQUE KEY `uk_dish_tag_name` (`tag_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish_tag_map` (
  `dish_id` int(11) NOT NULL,
  `tag_id` int(11) NOT NULL,
  PRIMARY KEY (`dish_id`,`tag_id`),
  KEY `dish_tag_map_ibfk_2` (`tag_id`),
  CONSTRAINT `dish_tag_map_ibfk_1` FOREIGN KEY (`dish_id`) REFERENCES `dish` (`dish_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dish_tag_map_ibfk_2` FOREIGN KEY (`tag_id`) REFERENCES `dish_tag` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish_image` (
  `image_id` int(11) NOT NULL AUTO_INCREMENT,
  `dish_id` int(11) NOT NULL,
  `image_url` varchar(500) NOT NULL,
  `upload_time` datetime NOT NULL,
  PRIMARY KEY (`image_id`),
  KEY `dish_image_ibfk_1` (`dish_id`),
  CONSTRAINT `dish_image_ibfk_1` FOREIGN KEY (`dish_id`) REFERENCES `dish` (`dish_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_review` (
  `review_id` int(11) NOT NULL AUTO_INCREMENT,
  `restaurant_id` int(11) NOT NULL,
  `user_name` varchar(100) DEFAULT NULL,
  `rating` decimal(3,2) NOT NULL,
  `content` text DEFAULT NULL,
  `dining_time` datetime DEFAULT NULL,
  `reply_content` text DEFAULT NULL,
  `reply_time` datetime DEFAULT NULL,
  PRIMARY KEY (`review_id`),
  KEY `user_review_ibfk_1` (`restaurant_id`),
  CONSTRAINT `user_review_ibfk_1` FOREIGN KEY (`restaurant_id`) REFERENCES `restaurant` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review_photo` (
  `photo_id` int(11) NOT NULL AUTO_INCREMENT,
  `review_id` int(11) NOT NULL,
  `photo_url` varchar(500) NOT NULL,
  `upload_time` datetime NOT NULL,
  PRIMARY KEY (`photo_id`),
  KEY `review_photo_ibfk_1` (`review_id`),
  CONSTRAINT `review_photo_ibfk_1` FOREIGN KEY (`review_id`) REFERENCES `user_review` (`review_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
