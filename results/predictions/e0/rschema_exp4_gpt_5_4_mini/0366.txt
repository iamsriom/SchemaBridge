-- Schema for database: rschema_0366
CREATE TABLE `engineer` (
  `engineer_id` int(11) NOT NULL AUTO_INCREMENT,
  `engineer_name` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `join_date` date NOT NULL,
  PRIMARY KEY (`engineer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `machine` (
  `machine_id` int(11) NOT NULL AUTO_INCREMENT,
  `machine_name` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `receiving_date` date NOT NULL,
  PRIMARY KEY (`machine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `engineer_machine` (
  `engineer_id` int(11) NOT NULL,
  `machine_id` int(11) NOT NULL,
  PRIMARY KEY (`engineer_id`,`machine_id`),
  KEY `engineer_machine_ibfk_2` (`machine_id`),
  CONSTRAINT `engineer_machine_ibfk_1` FOREIGN KEY (`engineer_id`) REFERENCES `engineer` (`engineer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `engineer_machine_ibfk_2` FOREIGN KEY (`machine_id`) REFERENCES `machine` (`machine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `migration_history` (
  `migration_id` int(11) NOT NULL AUTO_INCREMENT,
  `migration_date` datetime NOT NULL,
  `source_version` varchar(100) NOT NULL,
  `target_version` varchar(100) NOT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`migration_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
