-- Schema for database: rschema_0390
CREATE TABLE `consumer` (
  `consumer_id` int(11) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`consumer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courier_company` (
  `company_id` int(11) NOT NULL,
  `company_name` varchar(255) NOT NULL,
  `arrival_time` datetime DEFAULT NULL,
  `charging_price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrator` (
  `administrator_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `courier_user_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`administrator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courier_tracking` (
  `courier_number` varchar(50) NOT NULL,
  `recipient_id` int(11) NOT NULL,
  `recipient_name` varchar(255) NOT NULL,
  `recipient_phone_number` varchar(20) DEFAULT NULL,
  `company_id` int(11) DEFAULT NULL,
  `consumer_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`courier_number`),
  KEY `courier_tracking_ibfk_1` (`recipient_id`),
  KEY `courier_tracking_ibfk_2` (`company_id`),
  KEY `courier_tracking_ibfk_3` (`consumer_id`),
  CONSTRAINT `courier_tracking_ibfk_1` FOREIGN KEY (`recipient_id`) REFERENCES `consumer` (`consumer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `courier_tracking_ibfk_2` FOREIGN KEY (`company_id`) REFERENCES `courier_company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `courier_tracking_ibfk_3` FOREIGN KEY (`consumer_id`) REFERENCES `consumer` (`consumer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `partnership` (
  `partnership_id` int(11) NOT NULL,
  `company_id` int(11) NOT NULL,
  `administrator_id` int(11) NOT NULL,
  PRIMARY KEY (`partnership_id`),
  KEY `partnership_ibfk_1` (`company_id`),
  KEY `partnership_ibfk_2` (`administrator_id`),
  CONSTRAINT `partnership_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `courier_company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `partnership_ibfk_2` FOREIGN KEY (`administrator_id`) REFERENCES `administrator` (`administrator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
