-- Schema for database: rschema_0395
CREATE TABLE `students` (
  `student_id` varchar(20) NOT NULL,
  `student_name` varchar(100) NOT NULL,
  `department` varchar(100) NOT NULL,
  `major` varchar(100) DEFAULT NULL,
  `grade` varchar(20) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `meal_cards` (
  `card_id` varchar(20) NOT NULL,
  `student_id` varchar(20) NOT NULL,
  `card_number` varchar(30) NOT NULL,
  `balance` decimal(10,2) NOT NULL DEFAULT '0.00',
  `status` varchar(20) NOT NULL DEFAULT 'active',
  `issued_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`card_id`),
  UNIQUE KEY `uk_meal_cards_card_number` (`card_number`),
  KEY `meal_cards_ibfk_1` (`student_id`),
  CONSTRAINT `meal_cards_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `card_recharges` (
  `recharge_id` int(11) NOT NULL AUTO_INCREMENT,
  `card_id` varchar(20) NOT NULL,
  `recharge_time` datetime NOT NULL,
  `recharge_amount` decimal(10,2) NOT NULL,
  `recharge_location` varchar(255) NOT NULL,
  `operator_name` varchar(100) DEFAULT NULL,
  `remark` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`recharge_id`),
  KEY `card_recharges_ibfk_1` (`card_id`),
  CONSTRAINT `card_recharges_ibfk_1` FOREIGN KEY (`card_id`) REFERENCES `meal_cards` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `consumption_records` (
  `consumption_id` int(11) NOT NULL AUTO_INCREMENT,
  `card_id` varchar(20) NOT NULL,
  `student_id` varchar(20) NOT NULL,
  `consumption_time` datetime NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `location` varchar(255) NOT NULL,
  `merchant_type` varchar(50) NOT NULL,
  `balance_after` decimal(10,2) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`consumption_id`),
  KEY `consumption_records_ibfk_1` (`card_id`),
  KEY `consumption_records_ibfk_2` (`student_id`),
  CONSTRAINT `consumption_records_ibfk_1` FOREIGN KEY (`card_id`) REFERENCES `meal_cards` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `consumption_records_ibfk_2` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
