-- Schema for database: rschema_0401
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_size` int(11) DEFAULT NULL,
  `subordinate_majors` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teacher` (
  `teacher_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `grade` varchar(50) DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `entry_time` datetime DEFAULT NULL,
  PRIMARY KEY (`teacher_id`),
  KEY `teacher_ibfk_1` (`department_id`),
  CONSTRAINT `teacher_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `class` (
  `class_id` int(11) NOT NULL,
  `class_name` varchar(255) NOT NULL,
  `grade` varchar(50) DEFAULT NULL,
  `major` varchar(255) DEFAULT NULL,
  `head_teacher_id` int(11) DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`class_id`),
  KEY `class_ibfk_1` (`head_teacher_id`),
  KEY `class_ibfk_2` (`department_id`),
  CONSTRAINT `class_ibfk_1` FOREIGN KEY (`head_teacher_id`) REFERENCES `teacher` (`teacher_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `class_ibfk_2` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `grade` varchar(50) DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  `class_id` int(11) DEFAULT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `enrollment_time` datetime DEFAULT NULL,
  PRIMARY KEY (`student_id`),
  KEY `student_ibfk_1` (`department_id`),
  KEY `student_ibfk_2` (`class_id`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_ibfk_2` FOREIGN KEY (`class_id`) REFERENCES `class` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
