-- Schema for database: rschema_0413
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_manager` varchar(255) DEFAULT NULL,
  `department_phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `department_id` int(11) NOT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  `hire_date` date DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  KEY `employee_ibfk_1` (`department_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `training_course` (
  `course_id` int(11) NOT NULL,
  `course_name` varchar(255) NOT NULL,
  `course_duration` int(11) DEFAULT NULL,
  `instructor` varchar(255) DEFAULT NULL,
  `course_description` text DEFAULT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `training_record` (
  `employee_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `training_date` date NOT NULL,
  `training_score` decimal(5,2) DEFAULT NULL,
  `pass_status` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`employee_id`,`course_id`,`training_date`),
  KEY `training_record_ibfk_2` (`course_id`),
  CONSTRAINT `training_record_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `training_record_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `training_course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
