-- Schema for database: rschema_0415
CREATE TABLE `user_info` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `user_ip_address` varchar(45) NOT NULL,
  `user_port_number` int(11) NOT NULL,
  `last_login_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chat_record` (
  `message_id` int(11) NOT NULL,
  `sender_id` int(11) NOT NULL,
  `receiver_id` int(11) NOT NULL,
  `message_content` text NOT NULL,
  `sending_time` datetime NOT NULL,
  PRIMARY KEY (`message_id`),
  KEY `chat_record_ibfk_1` (`sender_id`),
  KEY `chat_record_ibfk_2` (`receiver_id`),
  CONSTRAINT `chat_record_ibfk_1` FOREIGN KEY (`sender_id`) REFERENCES `user_info` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `chat_record_ibfk_2` FOREIGN KEY (`receiver_id`) REFERENCES `user_info` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `server_config` (
  `server_id` int(11) NOT NULL,
  `server_ip_address` varchar(45) NOT NULL,
  `server_port_number` int(11) NOT NULL,
  `max_client_connections` int(11) NOT NULL,
  `receive_buffer_size` int(11) NOT NULL,
  `protocol` varchar(50) NOT NULL,
  PRIMARY KEY (`server_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
