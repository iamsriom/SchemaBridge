CREATE TABLE `dish_basic_info` (
  `dish_id` int(11) NOT NULL AUTO_INCREMENT,
  `dish_name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `dish_image` varchar(255) DEFAULT NULL,
  `dish_description` varchar(500) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`dish_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish_detail_info` (
  `dish_id` int(11) NOT NULL,
  `ingredients` varchar(500) DEFAULT NULL,
  `taste` varchar(100) DEFAULT NULL,
  `recommendation_index` int(11) DEFAULT NULL,
  `spicy_level` varchar(50) DEFAULT NULL,
  `cooking_method` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`dish_id`),
  CONSTRAINT `dish_detail_info_ibfk_1` FOREIGN KEY (`dish_id`) REFERENCES `dish_basic_info` (`dish_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish_classification_info` (
  `classification_id` int(11) NOT NULL AUTO_INCREMENT,
  `dish_id` int(11) NOT NULL,
  `cuisine_type` varchar(100) DEFAULT NULL,
  `special_features` varchar(255) DEFAULT NULL,
  `recommendation_status` tinyint(1) NOT NULL DEFAULT '0',
  `is_vegetarian` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`classification_id`),
  KEY `dish_id` (`dish_id`),
  CONSTRAINT `dish_classification_info_ibfk_1` FOREIGN KEY (`dish_id`) REFERENCES `dish_basic_info` (`dish_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_basic_info` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(100) NOT NULL,
  `password` varchar(255) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `account_id` int(11) NOT NULL AUTO_INCREMENT,
