CREATE TABLE `user_info` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `wechat_openid` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `qq_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `registration_time` datetime DEFAULT NULL,
  `account_status` tinyint(4) NOT NULL DEFAULT '1',
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_user_info_username` (`username`),
  UNIQUE KEY `uk_user_info_phone_number` (`phone_number`),
  UNIQUE KEY `uk_user_info_wechat_openid` (`wechat_openid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `category_id` int(11) NOT NULL AUTO_INCREMENT,

CREATE TABLE `news_info` (
  `news_id` int(11) NOT NULL AUTO_INCREMENT,
  `category_id` int(11) DEFAULT NULL,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `publication_time` datetime DEFAULT NULL,
  `read_count` int(11) NOT NULL DEFAULT '0',
  `news_status` tinyint(4) NOT NULL DEFAULT '0',
  PRIMARY KEY (`news_id`),
  KEY `news_info_ibfk_1` (`category_id`),
  CONSTRAINT `news_info_ibfk_1` FOREIGN KEY (`category_id`) REFERENCES `news_category` (`category_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
