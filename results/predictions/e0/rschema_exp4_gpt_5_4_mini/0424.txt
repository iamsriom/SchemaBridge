-- Schema for database: rschema_0424
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `account_status` varchar(50) DEFAULT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `login_ip` varchar(45) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `systems` (
  `system_id` int(11) NOT NULL AUTO_INCREMENT,
  `system_name` varchar(255) NOT NULL,
  `system_domain` varchar(255) NOT NULL,
  `access_time` datetime DEFAULT NULL,
  `callback_url` varchar(500) DEFAULT NULL,
  `secret_key` varchar(255) DEFAULT NULL,
  `token_validity_period` int(11) DEFAULT NULL,
  `is_enabled` tinyint(1) DEFAULT NULL,
  `requires_approval` tinyint(1) DEFAULT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `login_records` (
  `login_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `system_id` int(11) NOT NULL,
  `login_time` datetime NOT NULL,
  `login_ip` varchar(45) DEFAULT NULL,
  `token` varchar(500) DEFAULT NULL,
  `expiration_time` datetime DEFAULT NULL,
  `is_valid` tinyint(1) DEFAULT NULL,
  `logout_time` datetime DEFAULT NULL,
  PRIMARY KEY (`login_record_id`),
  KEY `login_records_ibfk_1` (`user_id`),
  KEY `login_records_ibfk_2` (`system_id`),
  CONSTRAINT `login_records_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `login_records_ibfk_2` FOREIGN KEY (`system_id`) REFERENCES `systems` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `system_authorizations` (
  `authorization_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `system_id` int(11) NOT NULL,
  `authorization_time` datetime NOT NULL,
  `is_effective` tinyint(1) DEFAULT NULL,
  `expiration_time` datetime DEFAULT NULL,
  PRIMARY KEY (`authorization_id`),
  KEY `system_authorizations_ibfk_1` (`user_id`),
  KEY `system_authorizations_ibfk_2` (`system_id`),
  CONSTRAINT `system_authorizations_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `system_authorizations_ibfk_2` FOREIGN KEY (`system_id`) REFERENCES `systems` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
