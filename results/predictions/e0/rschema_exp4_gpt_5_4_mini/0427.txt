-- Schema for database: rschema_0427
CREATE TABLE `student_info` (
  `student_id` varchar(20) NOT NULL,
  `student_name` varchar(100) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `department` varchar(100) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `enrollment_date` date DEFAULT NULL,
  `major` varchar(100) DEFAULT NULL,
  `login_password` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teacher_info` (
  `employee_id` varchar(20) NOT NULL,
  `teacher_name` varchar(100) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `login_password` varchar(255) NOT NULL,
  `title` varchar(100) DEFAULT NULL,
  `department` varchar(100) DEFAULT NULL,
  `course_id` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_info` (
  `course_id` varchar(20) NOT NULL,
  `course_name` varchar(200) NOT NULL,
  `prerequisite_course_id` varchar(20) DEFAULT NULL,
  `credits` decimal(4,1) NOT NULL,
  PRIMARY KEY (`course_id`),
  KEY `course_info_ibfk_1` (`prerequisite_course_id`),
  CONSTRAINT `course_info_ibfk_1` FOREIGN KEY (`prerequisite_course_id`) REFERENCES `course_info` (`course_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_selection` (
  `selection_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_id` varchar(20) NOT NULL,
  `course_id` varchar(20) NOT NULL,
  `grade` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`selection_id`),
  UNIQUE KEY `course_selection_uni_1` (`student_id`,`course_id`),
  KEY `course_selection_ibfk_1` (`student_id`),
  KEY `course_selection_ibfk_2` (`course_id`),
  CONSTRAINT `course_selection_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student_info` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_selection_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course_info` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teaching_info` (
  `teaching_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` varchar(20) NOT NULL,
  `course_id` varchar(20) NOT NULL,
  `teaching_location` varchar(255) NOT NULL,
  `class_time` varchar(100) NOT NULL,
  PRIMARY KEY (`teaching_id`),
  KEY `teaching_info_ibfk_1` (`employee_id`),
  KEY `teaching_info_ibfk_2` (`course_id`),
  CONSTRAINT `teaching_info_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `teacher_info` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `teaching_info_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course_info` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_timetable` (
  `timetable_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_id` varchar(20) NOT NULL,
  `course_id` varchar(20) NOT NULL,
  `class_time` varchar(100) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`timetable_id`),
  UNIQUE KEY `student_timetable_uni_1` (`student_id`,`course_id`,`class_time`,`location`),
  KEY `student_timetable_ibfk_1` (`student_id`),
  KEY `student_timetable_ibfk_2` (`course_id`),
  CONSTRAINT `student_timetable_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student_info` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_timetable_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course_info` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
