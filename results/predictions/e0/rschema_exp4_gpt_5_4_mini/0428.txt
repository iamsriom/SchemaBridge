-- Schema for database: rschema_0428
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(100) NOT NULL,
  `password` varchar(255) NOT NULL,
  `real_name` varchar(100) NOT NULL,
  `identity_type` varchar(20) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_users_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitories` (
  `dormitory_id` int(11) NOT NULL AUTO_INCREMENT,
  `dormitory_name` varchar(100) NOT NULL,
  `building_name` varchar(100) DEFAULT NULL,
  `room_number` varchar(50) NOT NULL,
  `capacity` int(11) NOT NULL,
  `current_occupancy` int(11) NOT NULL DEFAULT 0,
  `manager_user_id` int(11) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`dormitory_id`),
  KEY `dormitories_ibfk_1` (`manager_user_id`),
  CONSTRAINT `dormitories_ibfk_1` FOREIGN KEY (`manager_user_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courses` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_code` varchar(50) NOT NULL,
  `course_name` varchar(100) NOT NULL,
  `course_description` varchar(500) DEFAULT NULL,
  `teacher_user_id` int(11) DEFAULT NULL,
  `credit` decimal(4,1) NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`course_id`),
  UNIQUE KEY `uk_courses_course_code` (`course_code`),
  KEY `courses_ibfk_1` (`teacher_user_id`),
  CONSTRAINT `courses_ibfk_1` FOREIGN KEY (`teacher_user_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grades` (
  `grade_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_user_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `score` decimal(5,2) NOT NULL,
  `grade_level` varchar(20) DEFAULT NULL,
  `remark` varchar(255) DEFAULT NULL,
  `recorded_by` int(11) DEFAULT NULL,
  `recorded_at` datetime NOT NULL,
  PRIMARY KEY (`grade_id`),
  KEY `grades_ibfk_1` (`student_user_id`),
  KEY `grades_ibfk_2` (`course_id`),
  KEY `grades_ibfk_3` (`recorded_by`),
  CONSTRAINT `grades_ibfk_1` FOREIGN KEY (`student_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grades_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grades_ibfk_3` FOREIGN KEY (`recorded_by`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitory_repair_requests` (
  `request_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_user_id` int(11) NOT NULL,
  `dormitory_id` int(11) NOT NULL,
  `request_title` varchar(100) NOT NULL,
  `request_content` varchar(1000) NOT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'pending',
  `submitted_at` datetime NOT NULL,
  `handled_by` int(11) DEFAULT NULL,
  `handled_at` datetime DEFAULT NULL,
  PRIMARY KEY (`request_id`),
  KEY `dormitory_repair_requests_ibfk_1` (`student_user_id`),
  KEY `dormitory_repair_requests_ibfk_2` (`dormitory_id`),
  KEY `dormitory_repair_requests_ibfk_3` (`handled_by`),
  CONSTRAINT `dormitory_repair_requests_ibfk_1` FOREIGN KEY (`student_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dormitory_repair_requests_ibfk_2` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitories` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dormitory_repair_requests_ibfk_3` FOREIGN KEY (`handled_by`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
