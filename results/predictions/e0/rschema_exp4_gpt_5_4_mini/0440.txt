-- Schema for database: rschema_0440
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `role` varchar(50) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `account_status` varchar(20) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `devices` (
  `device_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `device_model` varchar(255) DEFAULT NULL,
  `person_in_charge_id` int(11) DEFAULT NULL,
  `group_name` varchar(100) DEFAULT NULL,
  `device_type` varchar(100) DEFAULT NULL,
  `usage_location` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `ip_address` varchar(45) DEFAULT NULL,
  `device_image` varchar(255) DEFAULT NULL,
  `qr_code` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`device_id`),
  UNIQUE KEY `uk_devices_qr_code` (`qr_code`),
  KEY `devices_ibfk_1` (`person_in_charge_id`),
  CONSTRAINT `devices_ibfk_1` FOREIGN KEY (`person_in_charge_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_qr_labels` (
  `label_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `printed_at` datetime DEFAULT NULL,
  `pasted_at` datetime DEFAULT NULL,
  `label_status` varchar(30) NOT NULL,
  PRIMARY KEY (`label_id`),
  KEY `device_qr_labels_ibfk_1` (`device_id`),
  CONSTRAINT `device_qr_labels_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_inventory_records` (
  `inventory_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `inventory_person_id` int(11) NOT NULL,
  `inventory_time` datetime NOT NULL,
  `device_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`inventory_id`),
  KEY `device_inventory_records_ibfk_1` (`device_id`),
  KEY `device_inventory_records_ibfk_2` (`inventory_person_id`),
  CONSTRAINT `device_inventory_records_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `device_inventory_records_ibfk_2` FOREIGN KEY (`inventory_person_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_maintenance_requests` (
  `request_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `requester_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text NOT NULL,
  `handling_status` varchar(30) NOT NULL,
  `handling_personnel_id` int(11) DEFAULT NULL,
  `handling_time` datetime DEFAULT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`request_id`),
  KEY `device_maintenance_requests_ibfk_1` (`device_id`),
  KEY `device_maintenance_requests_ibfk_2` (`requester_id`),
  KEY `device_maintenance_requests_ibfk_3` (`handling_personnel_id`),
  CONSTRAINT `device_maintenance_requests_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `device_maintenance_requests_ibfk_2` FOREIGN KEY (`requester_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `device_maintenance_requests_ibfk_3` FOREIGN KEY (`handling_personnel_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_maintenance_records` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` text NOT NULL,
  `maintenance_results` text DEFAULT NULL,
  `user_satisfaction` tinyint(4) DEFAULT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  KEY `device_maintenance_records_ibfk_1` (`device_id`),
  KEY `device_maintenance_records_ibfk_2` (`maintenance_personnel_id`),
  CONSTRAINT `device_maintenance_records_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `device_maintenance_records_ibfk_2` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_status_records` (
  `status_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_id` int(11) NOT NULL,
  `current_status` varchar(50) NOT NULL,
  `last_update_time` datetime NOT NULL,
  `remarks` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`status_id`),
  KEY `device_status_records_ibfk_1` (`device_id`),
  CONSTRAINT `device_status_records_ibfk_1` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
