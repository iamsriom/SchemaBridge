-- Schema for database: rschema_0441
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_name` varchar(255) NOT NULL,
  `role` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `budget` decimal(12,2) NOT NULL,
  `remaining_funds` decimal(12,2) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fund_transactions` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `record_time` datetime NOT NULL,
  `transaction_amount` decimal(12,2) NOT NULL,
  `purpose_description` varchar(500) DEFAULT NULL,
  `transaction_operator` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  KEY `fund_transactions_ibfk_1` (`transaction_operator`),
  KEY `fund_transactions_ibfk_2` (`project_id`),
  CONSTRAINT `fund_transactions_ibfk_1` FOREIGN KEY (`transaction_operator`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fund_transactions_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
