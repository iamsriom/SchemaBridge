-- Schema for database: rschema_0443
CREATE TABLE `user_information` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_name` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `registered_mobile_number` varchar(20) DEFAULT NULL,
  `role_permissions` varchar(255) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `asset_information` (
  `asset_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `equipment_id` varchar(100) NOT NULL,
  `equipment_model` varchar(255) DEFAULT NULL,
  `equipment_status` varchar(100) DEFAULT NULL,
  `calibration_date` date DEFAULT NULL,
  `calibration_validity` varchar(100) DEFAULT NULL,
  `calibration_results` varchar(255) DEFAULT NULL,
  `equipment_location` varchar(255) DEFAULT NULL,
  `owning_department` varchar(255) DEFAULT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`asset_id`),
  UNIQUE KEY `uk_asset_information_equipment_id` (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status_log_information` (
  `status_log_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` varchar(100) NOT NULL,
  `borrower` varchar(255) DEFAULT NULL,
  `borrowing_time` datetime DEFAULT NULL,
  `borrowed_equipment_photo` varchar(255) DEFAULT NULL,
  `iou_photo` varchar(255) DEFAULT NULL,
  `recipient` varchar(255) DEFAULT NULL,
  `receiving_area` varchar(255) DEFAULT NULL,
  `receiving_time` datetime DEFAULT NULL,
  `received_equipment_photo` varchar(255) DEFAULT NULL,
  `returner` varchar(255) DEFAULT NULL,
  `return_time` datetime DEFAULT NULL,
  `returned_equipment_photo` varchar(255) DEFAULT NULL,
  `data_monitoring` varchar(255) DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`status_log_id`),
  KEY `status_log_information_ibfk_1` (`equipment_id`),
  CONSTRAINT `status_log_information_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `asset_information` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account_activity_log` (
  `activity_log_id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` varchar(100) NOT NULL,
  `borrowing_account` varchar(255) DEFAULT NULL,
  `receiving_account` varchar(255) DEFAULT NULL,
  `returning_account` varchar(255) DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`activity_log_id`),
  KEY `account_activity_log_ibfk_1` (`equipment_id`),
  CONSTRAINT `account_activity_log_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `asset_information` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
