-- Schema for database: rschema_0445
CREATE TABLE `personnel` (
  `person_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `personnel_type` varchar(50) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `unit_id` int(11) DEFAULT NULL,
  `other_info` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`person_id`),
  KEY `personnel_ibfk_1` (`unit_id`),
  CONSTRAINT `personnel_ibfk_1` FOREIGN KEY (`unit_id`) REFERENCES `unit` (`unit_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `unit` (
  `unit_id` int(11) NOT NULL AUTO_INCREMENT,
  `unit_name` varchar(255) NOT NULL,
  `unit_type` varchar(50) NOT NULL,
  `other_info` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`unit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `administrator` (
  `admin_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `management_role` varchar(100) NOT NULL,
  PRIMARY KEY (`admin_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `health_record` (
  `health_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `person_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `temperature` decimal(4,1) DEFAULT NULL,
  `symptoms` varchar(500) DEFAULT NULL,
  `health_code_color` varchar(20) DEFAULT NULL,
  `travel_trajectory` varchar(1000) DEFAULT NULL,
  PRIMARY KEY (`health_record_id`),
  KEY `health_record_ibfk_1` (`person_id`),
  CONSTRAINT `health_record_ibfk_1` FOREIGN KEY (`person_id`) REFERENCES `personnel` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `visitor_appointment` (
  `appointment_id` int(11) NOT NULL AUTO_INCREMENT,
  `visitor_id` int(11) NOT NULL,
  `visited_person_id` int(11) NOT NULL,
  `scheduled_entry_time` datetime NOT NULL,
  `scheduled_exit_time` datetime DEFAULT NULL,
  `appointment_status` varchar(20) NOT NULL,
  `appointment_info` varchar(500) DEFAULT NULL,
  `created_time` datetime DEFAULT NULL,
  PRIMARY KEY (`appointment_id`),
  KEY `visitor_appointment_ibfk_1` (`visitor_id`),
  KEY `visitor_appointment_ibfk_2` (`visited_person_id`),
  CONSTRAINT `visitor_appointment_ibfk_1` FOREIGN KEY (`visitor_id`) REFERENCES `personnel` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `visitor_appointment_ibfk_2` FOREIGN KEY (`visited_person_id`) REFERENCES `personnel` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `access_control_device` (
  `device_id` int(11) NOT NULL AUTO_INCREMENT,
  `device_name` varchar(255) NOT NULL,
  `device_location` varchar(255) NOT NULL,
  `device_status` varchar(20) NOT NULL,
  `other_info` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`device_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `access_record` (
  `access_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `person_id` int(11) NOT NULL,
  `device_id` int(11) NOT NULL,
  `entry_time` datetime NOT NULL,
  `exit_time` datetime DEFAULT NULL,
  `verification_method` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`access_record_id`),
  KEY `access_record_ibfk_1` (`person_id`),
  KEY `access_record_ibfk_2` (`device_id`),
  CONSTRAINT `access_record_ibfk_1` FOREIGN KEY (`person_id`) REFERENCES `personnel` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `access_record_ibfk_2` FOREIGN KEY (`device_id`) REFERENCES `access_control_device` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
