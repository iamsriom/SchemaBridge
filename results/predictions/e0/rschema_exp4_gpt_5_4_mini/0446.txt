CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `registration_time` datetime NOT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `role` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_users_username` (`username`),
  UNIQUE KEY `uk_users_phone_number` (`phone_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crops` (
  `crop_id` int(11) NOT NULL AUTO_INCREMENT,
  `crop_name` varchar(255) NOT NULL,
  `crop_type` varchar(255) NOT NULL,
  `growth_model_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`crop_id`),
  KEY `idx_crops_growth_model_id` (`growth_model_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `growth_models` (
  `growth_model_id` int(11) NOT NULL AUTO_INCREMENT,
  `crop_id` int(11) NOT NULL,
  `model_name` varchar(255) NOT NULL,
  `parameters` text NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`growth_model_id`),
  UNIQUE KEY `uk_growth_models_crop_id` (`crop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sensors` (
  `sensor_id` int(11) NOT NULL AUTO_INCREMENT,
  `sensor_location` varchar(255) NOT NULL,
  `burial_depth` decimal(10,2) NOT NULL,
  PRIMARY KEY (`sensor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meteorological_records` (
  `meteorological_record_id` int(11) NOT NULL AUTO_INCREMENT,

  `record_time` datetime NOT NULL,

  `temperature` decimal(10,2) DEFAULT NULL,
  `humidity` decimal(10,2) DEFAULT NULL,
  `rainfall` decimal(10,2) DEFAULT NULL,
  `wind_speed` decimal(10,2) DEFAULT NULL,
  `light_intensity` decimal(10,2) DEFAULT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`meteorological_record_id`),
  KEY `idx_meteorological_records_sensor_id` (`sensor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  PRIMARY KEY (`supply_record_id`),

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
