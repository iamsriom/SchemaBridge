-- Schema for database: rschema_0449
CREATE TABLE `department` (
  `department_no` varchar(20) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_function` varchar(255) DEFAULT NULL,
  `department_planning` varchar(255) DEFAULT NULL,
  `other_content` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position` (
  `position_no` varchar(20) NOT NULL,
  `position_name` varchar(255) NOT NULL,
  `position_responsibilities` varchar(255) DEFAULT NULL,
  `position_requirements` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`position_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_no` varchar(20) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `employee_gender` varchar(10) DEFAULT NULL,
  `employee_birth_date` date DEFAULT NULL,
  `department_no` varchar(20) DEFAULT NULL,
  `employee_entry_date` date DEFAULT NULL,
  `employee_resignation_date` date DEFAULT NULL,
  `position_no` varchar(20) DEFAULT NULL,
  `employee_contact_info` varchar(255) DEFAULT NULL,
  `employee_email` varchar(255) DEFAULT NULL,
  `employee_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`employee_no`),
  KEY `employee_ibfk_1` (`department_no`),
  KEY `employee_ibfk_2` (`position_no`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_no`) REFERENCES `department` (`department_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`position_no`) REFERENCES `position` (`position_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `training` (
  `training_no` varchar(20) NOT NULL,
  `training_time` datetime DEFAULT NULL,
  `training_location` varchar(255) DEFAULT NULL,
  `training_content` varchar(255) DEFAULT NULL,
  `trainer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`training_no`) 
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_position_change` (
  `employee_no` varchar(20) NOT NULL,
  `position_no` varchar(20) NOT NULL,
  `change_date` date NOT NULL,
  `change_reason` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_no`,`position_no`,`change_date`),
  KEY `employee_position_change_ibfk_1` (`employee_no`),
  KEY `employee_position_change_ibfk_2` (`position_no`),
  CONSTRAINT `employee_position_change_ibfk_1` FOREIGN KEY (`employee_no`) REFERENCES `employee` (`employee_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_position_change_ibfk_2` FOREIGN KEY (`position_no`) REFERENCES `position` (`position_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
