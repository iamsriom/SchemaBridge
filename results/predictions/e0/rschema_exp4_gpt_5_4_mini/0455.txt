CREATE TABLE `parking_space` (
  `parking_space_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_code` varchar(50) NOT NULL,
  `is_occupied` tinyint(1) NOT NULL DEFAULT 0,
  `occupied_vehicle_plate` varchar(20) DEFAULT NULL,
  `occupied_at` datetime DEFAULT NULL,
  PRIMARY KEY (`parking_space_id`),
  UNIQUE KEY `uk_parking_space_space_code` (`space_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `entry_point` (
  `entry_point_id` int(11) NOT NULL AUTO_INCREMENT,
  `entry_code` varchar(50) NOT NULL,
  `has_vehicle_entered` tinyint(1) NOT NULL DEFAULT 0,
  `last_vehicle_plate` varchar(20) DEFAULT NULL,
  `last_entry_time` datetime DEFAULT NULL,
  PRIMARY KEY (`entry_point_id`),
  UNIQUE KEY `uk_entry_point_entry_code` (`entry_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exit_point` (
  `exit_point_id` int(11) NOT NULL AUTO_INCREMENT,
  `exit_code` varchar(50) NOT NULL,
  `has_vehicle_left` tinyint(1) NOT NULL DEFAULT 0,
  `last_vehicle_plate` varchar(20) DEFAULT NULL,
  `last_exit_time` datetime DEFAULT NULL,
  PRIMARY KEY (`exit_point_id`),
  UNIQUE KEY `uk_exit_point_exit_code` (`exit_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
