-- Schema for database: rschema_0459
CREATE TABLE `department` (
  `department_no` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `manager_employee_no` int(11) DEFAULT NULL,
  PRIMARY KEY (`department_no`),
  KEY `department_ibfk_1` (`manager_employee_no`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`manager_employee_no`) REFERENCES `employee` (`employee_no`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_no` int(11) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `department_no` int(11) NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `position` varchar(255) NOT NULL,
  `education_level` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_no`),
  KEY `employee_ibfk_1` (`department_no`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_no`) REFERENCES `department` (`department_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_no` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `department_no` int(11) NOT NULL,
  `responsible_employee_no` int(11) NOT NULL,
  PRIMARY KEY (`project_no`),
  KEY `project_ibfk_1` (`department_no`),
  KEY `project_ibfk_2` (`responsible_employee_no`),
  CONSTRAINT `project_ibfk_1` FOREIGN KEY (`department_no`) REFERENCES `department` (`department_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_ibfk_2` FOREIGN KEY (`responsible_employee_no`) REFERENCES `employee` (`employee_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_project` (
  `employee_no` int(11) NOT NULL,
  `project_no` int(11) NOT NULL,
  PRIMARY KEY (`employee_no`,`project_no`),
  KEY `employee_project_ibfk_2` (`project_no`),
  CONSTRAINT `employee_project_ibfk_1` FOREIGN KEY (`employee_no`) REFERENCES `employee` (`employee_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_project_ibfk_2` FOREIGN KEY (`project_no`) REFERENCES `project` (`project_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
