-- Schema for database: rschema_0471
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_number` varchar(50) DEFAULT NULL,
  `application_status` varchar(50) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_card_applications` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) DEFAULT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `applicant_address` varchar(255) DEFAULT NULL,
  `applicant_contact_number` varchar(50) DEFAULT NULL,
  `card_type` varchar(100) NOT NULL,
  `application_status` varchar(50) NOT NULL,
  `submitted_at` datetime DEFAULT NULL,
  PRIMARY KEY (`application_id`),
  KEY `credit_card_applications_ibfk_1` (`customer_id`),
  CONSTRAINT `credit_card_applications_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_cards` (
  `card_number` varchar(32) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `card_type` varchar(100) NOT NULL,
  `valid_from` date DEFAULT NULL,
  `valid_to` date DEFAULT NULL,
  `credit_limit` decimal(12,2) NOT NULL,
  `activation_status` varchar(50) NOT NULL,
  `password_hash` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`card_number`),
  KEY `credit_cards_ibfk_1` (`customer_id`),
  CONSTRAINT `credit_cards_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `card_notifications` (
  `notification_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_id` int(11) DEFAULT NULL,
  `card_number` varchar(32) DEFAULT NULL,
  `notification_type` varchar(50) NOT NULL,
  `message` varchar(500) DEFAULT NULL,
  `sent_at` datetime DEFAULT NULL,
  PRIMARY KEY (`notification_id`),
  KEY `card_notifications_ibfk_1` (`application_id`),
  KEY `card_notifications_ibfk_2` (`card_number`),
  CONSTRAINT `card_notifications_ibfk_1` FOREIGN KEY (`application_id`) REFERENCES `credit_card_applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `card_notifications_ibfk_2` FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `card_activations` (
  `activation_id` int(11) NOT NULL AUTO_INCREMENT,
  `card_number` varchar(32) NOT NULL,
  `activation_password` varchar(255) NOT NULL,
  `activation_status` varchar(50) NOT NULL,
  `activated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`activation_id`),
  KEY `card_activations_ibfk_1` (`card_number`),
  CONSTRAINT `card_activations_ibfk_1` FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `card_number` varchar(32) NOT NULL,
  `transaction_date` datetime NOT NULL,
  `transaction_description` varchar(500) DEFAULT NULL,
  `transaction_amount` decimal(12,2) NOT NULL,
  `transaction_status` varchar(50) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  KEY `transactions_ibfk_1` (`card_number`),
  CONSTRAINT `transactions_ibfk_1` FOREIGN KEY (`card_number`) REFERENCES `credit_cards` (`card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
