-- Schema for database: rschema_0472
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `student_name` varchar(255) NOT NULL,
  `class_name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instructors` (
  `instructor_id` int(11) NOT NULL,
  `instructor_name` varchar(255) NOT NULL,
  `courses_taught` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`instructor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homework_submissions` (
  `homework_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  `instructor_id` int(11) DEFAULT NULL,
  `homework_content` text NOT NULL,
  `submission_time` datetime NOT NULL,
  `grading_status` varchar(20) NOT NULL,
  `graded_time` datetime DEFAULT NULL,
  `overall_feedback` text DEFAULT NULL,
  PRIMARY KEY (`homework_id`),
  KEY `homework_submissions_ibfk_1` (`student_id`),
  KEY `homework_submissions_ibfk_2` (`instructor_id`),
  CONSTRAINT `homework_submissions_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `homework_submissions_ibfk_2` FOREIGN KEY (`instructor_id`) REFERENCES `instructors` (`instructor_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homework_grades` (
  `grade_id` int(11) NOT NULL,
  `homework_id` int(11) NOT NULL,
  `question_number` int(11) NOT NULL,
  `score` decimal(5,2) NOT NULL,
  `feedback` text DEFAULT NULL,
  `grading_time` datetime NOT NULL,
  PRIMARY KEY (`grade_id`),
  KEY `homework_grades_ibfk_1` (`homework_id`),
  CONSTRAINT `homework_grades_ibfk_1` FOREIGN KEY (`homework_id`) REFERENCES `homework_submissions` (`homework_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
