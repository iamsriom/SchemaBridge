-- Schema for database: rschema_0474
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `class` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instructors` (
  `instructor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `course` varchar(255) NOT NULL,
  PRIMARY KEY (`instructor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `assignments` (
  `assignment_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  `submitted_assignment` text NOT NULL,
  `submission_time` datetime NOT NULL,
  `graded_assignment` text DEFAULT NULL,
  `question_scores` text DEFAULT NULL,
  `overall_evaluation` text DEFAULT NULL,
  `graded_time` datetime DEFAULT NULL,
  `is_graded` tinyint(1) NOT NULL DEFAULT '0',
  `graded_by_instructor_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`assignment_id`),
  KEY `assignments_ibfk_1` (`student_id`),
  KEY `assignments_ibfk_2` (`graded_by_instructor_id`),
  CONSTRAINT `assignments_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `assignments_ibfk_2` FOREIGN KEY (`graded_by_instructor_id`) REFERENCES `instructors` (`instructor_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
