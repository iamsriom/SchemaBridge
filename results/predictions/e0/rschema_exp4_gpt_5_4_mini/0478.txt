-- Schema for database: rschema_0478
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `name` varchar(100) NOT NULL,
  `credit` int(11) NOT NULL DEFAULT 0,
  `status` varchar(20) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bicycles` (
  `bicycle_id` int(11) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `status` varchar(20) NOT NULL,
  `unlock_code` varchar(50) NOT NULL,
  PRIMARY KEY (`bicycle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `billing_rules` (
  `rule_id` int(11) NOT NULL,
  `billing_method` varchar(100) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  `minimum_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`rule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trips` (
  `trip_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `bicycle_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `cost` decimal(10,2) NOT NULL DEFAULT 0.00,
  `payment_status` varchar(20) NOT NULL,
  PRIMARY KEY (`trip_id`),
  KEY `trips_ibfk_1` (`user_id`),
  KEY `trips_ibfk_2` (`bicycle_id`),
  CONSTRAINT `trips_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trips_ibfk_2` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycles` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fault_records` (
  `fault_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `bicycle_id` int(11) NOT NULL,
  `fault_description` varchar(500) NOT NULL,
  `reporting_time` datetime NOT NULL,
  PRIMARY KEY (`fault_id`),
  KEY `fault_records_ibfk_1` (`user_id`),
  KEY `fault_records_ibfk_2` (`bicycle_id`),
  CONSTRAINT `fault_records_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fault_records_ibfk_2` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycles` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
