-- Schema for database: rschema_0480
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(100) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uk_users_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `agents` (
  `agent_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  `contact_number` varchar(30) DEFAULT NULL,
  `user_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`agent_id`),
  UNIQUE KEY `uk_agents_user_id` (`user_id`),
  CONSTRAINT `agents_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `properties` (
  `property_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_certificate` varchar(100) NOT NULL,
  `address` varchar(255) NOT NULL,
  `construction_year` int(11) DEFAULT NULL,
  `built_area` decimal(10,2) DEFAULT NULL,
  `sale_price` decimal(12,2) DEFAULT NULL,
  `property_photos` text DEFAULT NULL,
  `sales_status` varchar(20) NOT NULL,
  `property_type` varchar(20) NOT NULL,
  `agent_id` int(11) DEFAULT NULL,
  `responsibility_start_time` datetime DEFAULT NULL,
  `responsibility_end_time` datetime DEFAULT NULL,
  `approval_status` varchar(20) NOT NULL DEFAULT 'pending',
  `approved_by` int(11) DEFAULT NULL,
  `approved_at` datetime DEFAULT NULL,
  PRIMARY KEY (`property_id`),
  UNIQUE KEY `uk_properties_certificate` (`property_certificate`),
  KEY `properties_ibfk_1` (`agent_id`),
  KEY `properties_ibfk_2` (`approved_by`),
  CONSTRAINT `properties_ibfk_1` FOREIGN KEY (`agent_id`) REFERENCES `agents` (`agent_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `properties_ibfk_2` FOREIGN KEY (`approved_by`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `apartment_properties` (
  `property_id` int(11) NOT NULL,
  `unit_number` varchar(50) NOT NULL,
  `floor_number` int(11) DEFAULT NULL,
  `building_name` varchar(100) DEFAULT NULL,
  `total_floors` int(11) DEFAULT NULL,
  `apartment_area` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`property_id`),
  CONSTRAINT `apartment_properties_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `properties` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property_status_history` (
  `history_id` int(11) NOT NULL AUTO_INCREMENT,
  `property_id` int(11) NOT NULL,
  `status` varchar(20) NOT NULL,
  `changed_at` datetime NOT NULL,
  `changed_by` int(11) DEFAULT NULL,
  `remark` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`history_id`),
  KEY `property_status_history_ibfk_1` (`property_id`),
  KEY `property_status_history_ibfk_2` (`changed_by`),
  CONSTRAINT `property_status_history_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `properties` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `property_status_history_ibfk_2` FOREIGN KEY (`changed_by`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
