-- Schema for database: rschema_0482
CREATE TABLE `clients` (
  `client_no` int(11) NOT NULL,
  `client_name` varchar(255) NOT NULL,
  `client_type` varchar(100) NOT NULL,
  `discount_rate` decimal(5,2) NOT NULL,
  `contact_person` varchar(255) NOT NULL,
  `contact_phone` varchar(50) NOT NULL,
  PRIMARY KEY (`client_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_no` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `supervisor_employee_no` int(11) DEFAULT NULL,
  `phone_number` varchar(50) NOT NULL,
  PRIMARY KEY (`department_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_no` int(11) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `position` varchar(100) NOT NULL,
  `phone_number` varchar(50) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  `department_no` int(11) NOT NULL,
  PRIMARY KEY (`employee_no`),
  KEY `employees_ibfk_1` (`department_no`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`department_no`) REFERENCES `departments` (`department_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
  `license_plate_no` varchar(20) NOT NULL,
  `model` varchar(255) NOT NULL,
  `color` varchar(50) NOT NULL,
  `vehicle_category` varchar(100) NOT NULL,
  `client_no` int(11) NOT NULL,
  PRIMARY KEY (`license_plate_no`),
  KEY `vehicles_ibfk_1` (`client_no`),
  CONSTRAINT `vehicles_ibfk_1` FOREIGN KEY (`client_no`) REFERENCES `clients` (`client_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_tasks` (
  `maintenance_order_no` int(11) NOT NULL,
  `license_plate_no` varchar(20) NOT NULL,
  `maintenance_content` varchar(500) NOT NULL,
  `man_hours` decimal(8,2) NOT NULL,
  `maintenance_worker_employee_no` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_order_no`),
  KEY `maintenance_tasks_ibfk_1` (`license_plate_no`),
  KEY `maintenance_tasks_ibfk_2` (`maintenance_worker_employee_no`),
  CONSTRAINT `maintenance_tasks_ibfk_1` FOREIGN KEY (`license_plate_no`) REFERENCES `vehicles` (`license_plate_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_tasks_ibfk_2` FOREIGN KEY (`maintenance_worker_employee_no`) REFERENCES `employees` (`employee_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
