-- Schema for database: rschema_0485
CREATE TABLE `protection_area` (
  `area_id` int(11) NOT NULL,
  `area_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`area_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flora_fauna_record` (
  `record_id` int(11) NOT NULL,
  `area_id` int(11) NOT NULL,
  `recording_time` datetime NOT NULL,
  `recording_personnel` varchar(255) NOT NULL,
  `species` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `growth_status` varchar(255) NOT NULL,
  PRIMARY KEY (`record_id`),
  KEY `flora_fauna_record_ibfk_1` (`area_id`),
  CONSTRAINT `flora_fauna_record_ibfk_1` FOREIGN KEY (`area_id`) REFERENCES `protection_area` (`area_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `environmental_change_record` (
  `change_id` int(11) NOT NULL,
  `area_id` int(11) NOT NULL,
  `change_time` datetime NOT NULL,
  `change_location` varchar(255) NOT NULL,
  `temperature` decimal(10,2) DEFAULT NULL,
  `humidity` decimal(10,2) DEFAULT NULL,
  `rainfall` decimal(10,2) DEFAULT NULL,
  `other_changes` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`change_id`),
  KEY `environmental_change_record_ibfk_1` (`area_id`),
  CONSTRAINT `environmental_change_record_ibfk_1` FOREIGN KEY (`area_id`) REFERENCES `protection_area` (`area_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
