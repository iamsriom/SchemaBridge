-- Schema for database: rschema_0488
CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `sort_order` int(11) NOT NULL DEFAULT '0',
  `customer_only` tinyint(1) NOT NULL DEFAULT '0',
  `email_pipeline_only` tinyint(1) NOT NULL DEFAULT '0',
  `disable_auto_reply` tinyint(1) NOT NULL DEFAULT '0',
  `hidden_from_users` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department_email_config` (
  `config_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `host_address` varchar(255) NOT NULL,
  `port_number` int(11) NOT NULL,
  `login_username` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  PRIMARY KEY (`config_id`),
  KEY `department_email_config_ibfk_1` (`department_id`),
  CONSTRAINT `department_email_config_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_email_servers` (
  `server_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_name` varchar(255) NOT NULL,
  `server_host` varchar(255) NOT NULL,
  `server_port` int(11) NOT NULL,
  `server_username` varchar(255) NOT NULL,
  `server_password` varchar(255) NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`server_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
