-- Schema for database: rschema_0503
CREATE TABLE `material` (
  `material_id` int(11) NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `material_type` varchar(100) DEFAULT NULL,
  `specification` varchar(255) DEFAULT NULL,
  `weight` decimal(10,2) DEFAULT NULL,
  `volume` decimal(10,2) DEFAULT NULL,
  `source_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`material_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouse` (
  `warehouse_id` int(11) NOT NULL,
  `warehouse_name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `warehouse_type` varchar(50) DEFAULT NULL,
  `max_capacity` int(11) DEFAULT NULL,
  `current_inventory_level` int(11) DEFAULT NULL,
  `remaining_capacity` int(11) DEFAULT NULL,
  PRIMARY KEY (`warehouse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `driver` (
  `driver_id` int(11) NOT NULL,
  `name` varchar(100) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `driving_license_number` varchar(50) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  PRIMARY KEY (`driver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL,
  `vehicle_type` varchar(50) DEFAULT NULL,
  `license_plate_number` varchar(20) NOT NULL,
  `max_load` decimal(10,2) DEFAULT NULL,
  `current_load` decimal(10,2) DEFAULT NULL,
  `assigned_driver_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`),
  UNIQUE KEY `uk_vehicle_license_plate_number` (`license_plate_number`),
  KEY `vehicle_ibfk_1` (`assigned_driver_id`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`assigned_driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_storage` (
  `storage_id` int(11) NOT NULL,
  `material_id` int(11) NOT NULL,
  `warehouse_id` int(11) NOT NULL,
  `storage_time` datetime NOT NULL,
  `storage_quantity` int(11) NOT NULL,
  `notes` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`storage_id`),
  KEY `material_storage_ibfk_1` (`material_id`),
  KEY `material_storage_ibfk_2` (`warehouse_id`),
  CONSTRAINT `material_storage_ibfk_1` FOREIGN KEY (`material_id`) REFERENCES `material` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_storage_ibfk_2` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_dispatch` (
  `dispatch_id` int(11) NOT NULL,
  `material_id` int(11) NOT NULL,
  `warehouse_id` int(11) NOT NULL,
  `dispatch_time` datetime NOT NULL,
  `dispatch_quantity` int(11) NOT NULL,
  `notes` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`dispatch_id`),
  KEY `material_dispatch_ibfk_1` (`material_id`),
  KEY `material_dispatch_ibfk_2` (`warehouse_id`),
  CONSTRAINT `material_dispatch_ibfk_1` FOREIGN KEY (`material_id`) REFERENCES `material` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_dispatch_ibfk_2` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_transport` (
  `transport_id` int(11) NOT NULL,
  `material_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `driver_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `destination` varchar(255) NOT NULL,
  PRIMARY KEY (`transport_id`),
  KEY `material_transport_ibfk_1` (`material_id`),
  KEY `material_transport_ibfk_2` (`vehicle_id`),
  KEY `material_transport_ibfk_3` (`driver_id`),
  CONSTRAINT `material_transport_ibfk_1` FOREIGN KEY (`material_id`) REFERENCES `material` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_transport_ibfk_2` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_transport_ibfk_3` FOREIGN KEY (`driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
