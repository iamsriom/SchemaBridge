-- Schema for database: rschema_0510
CREATE TABLE `parking_space` (
  `space_id` int(11) NOT NULL AUTO_INCREMENT,
  `space_number` varchar(50) NOT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'empty',
  PRIMARY KEY (`space_id`),
  UNIQUE KEY `uk_parking_space_space_number` (`space_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_space_owner` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate_number` varchar(20) NOT NULL,
  `owner_name` varchar(100) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `space_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`owner_id`),
  UNIQUE KEY `uk_parking_space_owner_license_plate_number` (`license_plate_number`),
  KEY `parking_space_owner_ibfk_1` (`space_id`),
  CONSTRAINT `parking_space_owner_ibfk_1` FOREIGN KEY (`space_id`) REFERENCES `parking_space` (`space_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
