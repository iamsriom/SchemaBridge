-- Schema for database: rschema_0511
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_name` varchar(100) NOT NULL,
  `department_head` varchar(100) DEFAULT NULL,
  `department_phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_no` varchar(50) NOT NULL,
  `employee_name` varchar(100) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  `department_id` int(11) NOT NULL,
  `position` varchar(50) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`),
  UNIQUE KEY `uk_employee_no` (`employee_no`),
  KEY `employee_ibfk_1` (`department_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `owner` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `owner_no` varchar(50) NOT NULL,
  `owner_name` varchar(100) NOT NULL,
  `room_no` varchar(50) NOT NULL,
  `house_area` decimal(10,2) DEFAULT NULL,
  `work_unit` varchar(255) DEFAULT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`owner_id`),
  UNIQUE KEY `uk_owner_no` (`owner_no`),
  UNIQUE KEY `uk_owner_room_no` (`room_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `charge_record` (
  `charge_id` int(11) NOT NULL AUTO_INCREMENT,
  `room_no` varchar(50) NOT NULL,
  `owner_no` varchar(50) NOT NULL,
  `charge_date` datetime NOT NULL,
  `charge_type` varchar(100) NOT NULL,
  `quantity` decimal(10,2) DEFAULT NULL,
  `charge_amount` decimal(10,2) NOT NULL,
  `employee_no` varchar(50) NOT NULL,
  `charge_status` varchar(20) NOT NULL DEFAULT 'unpaid',
  PRIMARY KEY (`charge_id`),
  KEY `charge_record_ibfk_1` (`owner_no`),
  KEY `charge_record_ibfk_2` (`employee_no`),
  CONSTRAINT `charge_record_ibfk_1` FOREIGN KEY (`owner_no`) REFERENCES `owner` (`owner_no`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `charge_record_ibfk_2` FOREIGN KEY (`employee_no`) REFERENCES `employee` (`employee_no`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
