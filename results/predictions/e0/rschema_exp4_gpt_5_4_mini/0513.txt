-- Schema for database: rschema_0513
CREATE TABLE `district` (
  `district_id` int(11) NOT NULL AUTO_INCREMENT,
  `district_name` varchar(255) NOT NULL,
  `district_type` varchar(20) NOT NULL,
  `parent_address_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`district_id`),
  KEY `district_ibfk_1` (`parent_address_id`),
  CONSTRAINT `district_ibfk_1` FOREIGN KEY (`parent_address_id`) REFERENCES `address` (`address_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `address` (
  `address_id` int(11) NOT NULL AUTO_INCREMENT,
  `address_name` varchar(255) NOT NULL,
  `address_type` varchar(20) NOT NULL,
  `district_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`address_id`),
  KEY `address_ibfk_1` (`district_id`),
  CONSTRAINT `address_ibfk_1` FOREIGN KEY (`district_id`) REFERENCES `district` (`district_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voter` (
  `voter_id` int(11) NOT NULL AUTO_INCREMENT,
  `full_name` varchar(255) NOT NULL,
  `is_legitimate` tinyint(1) NOT NULL DEFAULT '1',
  `residential_address_id` int(11) NOT NULL,
  `registered_district_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`voter_id`),
  UNIQUE KEY `voter_ibfk_1` (`registered_district_id`),
  KEY `voter_ibfk_2` (`residential_address_id`),
  CONSTRAINT `voter_ibfk_1` FOREIGN KEY (`registered_district_id`) REFERENCES `district` (`district_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `voter_ibfk_2` FOREIGN KEY (`residential_address_id`) REFERENCES `address` (`address_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
