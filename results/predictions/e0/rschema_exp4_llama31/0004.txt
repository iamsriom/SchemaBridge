-- Schema for database: rschema_0004
CREATE TABLE `customers` (
  `customer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `address` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_plans` (
  `plan_number` char(6) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `included_services` varchar(255) NOT NULL,
  PRIMARY KEY (`plan_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subscriptions` (
  `subscription_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_number` int(11) NOT NULL,
  `plan_number` char(6) NOT NULL,
  `subscription_time` datetime NOT NULL,
  `cancellation_time` datetime DEFAULT NULL,
  `operating_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`subscription_id`),
  KEY `customer_number` (`customer_number`),
  KEY `plan_number` (`plan_number`),
  CONSTRAINT `subscriptions_ibfk_1` FOREIGN KEY (`customer_number`) REFERENCES `customers` (`customer_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `subscriptions_ibfk_2` FOREIGN KEY (`plan_number`) REFERENCES `service_plans` (`plan_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
