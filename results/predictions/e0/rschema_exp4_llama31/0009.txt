-- Schema for database: rschema_0009
CREATE TABLE `candidates` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voters` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voting_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `voter_id` int(11) NOT NULL,
  `candidate_id` int(11) NOT NULL,
  `voting_time` datetime NOT NULL,
  `voting_location` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `voting_records_ibfk_1` (`voter_id`),
  KEY `voting_records_ibfk_2` (`candidate_id`),
  CONSTRAINT `voting_records_ibfk_1` FOREIGN KEY (`voter_id`) REFERENCES `voters` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `voting_records_ibfk_2` FOREIGN KEY (`candidate_id`) REFERENCES `candidates` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `candidate_votes` (
  `candidate_id` int(11) NOT NULL,
  `vote_count` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`candidate_id`),
  KEY `candidate_votes_ibfk_1` (`candidate_id`),
  CONSTRAINT `candidate_votes_ibfk_1` FOREIGN KEY (`candidate_id`) REFERENCES `candidates` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
