-- Schema for database: rschema_0012
CREATE TABLE `warehouses` (
  `warehouse_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `capacity` int(11) NOT NULL,
  `current_inventory` int(11) NOT NULL,
  PRIMARY KEY (`warehouse_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specification` varchar(255) NOT NULL,
  `storage_date` datetime NOT NULL,
  `dispatch_date` datetime DEFAULT NULL,
  `quantity_stored` int(11) NOT NULL,
  `quantity_dispatched` int(11) NOT NULL,
  `current_inventory` int(11) NOT NULL,
  PRIMARY KEY (`product_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouses_products` (
  `warehouse_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `storage_date` datetime NOT NULL,
  `quantity_stored` int(11) NOT NULL,
  `operator` varchar(255) NOT NULL,
  PRIMARY KEY (`warehouse_id`, `product_id`, `storage_date`),
  KEY `warehouse_id` (`warehouse_id`),
  KEY `product_id` (`product_id`),
  CONSTRAINT `warehouses_products_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouses` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `warehouses_products_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dispatches` (
  `dispatch_id` int(11) NOT NULL AUTO_INCREMENT,
  `warehouse_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `dispatch_date` datetime NOT NULL,
  `quantity_dispatched` int(11) NOT NULL,
  `operator` varchar(255) NOT NULL,
  PRIMARY KEY (`dispatch_id`),
  KEY `warehouse_id` (`warehouse_id`),
  KEY `product_id` (`product_id`),
  CONSTRAINT `dispatches_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouses` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dispatches_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
