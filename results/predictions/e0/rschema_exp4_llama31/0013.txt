-- Schema for database: rschema_0013
CREATE TABLE `weather_stations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `weather_station_id` int(11) NOT NULL,
  `recording_date` datetime NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `precipitation` float NOT NULL,
  `wind_speed` float NOT NULL,
  `wind_direction` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `weather_station_id` (`weather_station_id`),
  CONSTRAINT `weather_records_ibfk_1` FOREIGN KEY (`weather_station_id`) REFERENCES `weather_stations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `farm_subscriptions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `farm_id` int(11) NOT NULL,
  `weather_station_id` int(11) NOT NULL,
  `subscription_date` datetime NOT NULL,
  `unsubscription_date` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `farm_id` (`farm_id`),
  KEY `weather_station_id` (`weather_station_id`),
  CONSTRAINT `farm_subscriptions_ibfk_1` FOREIGN KEY (`farm_id`),
  CONSTRAINT `farm_subscriptions_ibfk_2` FOREIGN KEY (`weather_station_id`) REFERENCES `weather_stations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
