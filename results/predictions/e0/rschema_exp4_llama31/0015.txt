-- Schema for database: rschema_0015
CREATE TABLE `plots` (
  `plot_id` int(11) NOT NULL AUTO_INCREMENT,
  `area` varchar(255) NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `pest_disease_status` varchar(255) NOT NULL,
  PRIMARY KEY (`plot_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pests_diseases` (
  `pest_disease_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `degree_of_damage` varchar(255) NOT NULL,
  `prevention_control_methods` varchar(255) NOT NULL,
  `season_of_occurrence` varchar(255) NOT NULL,
  PRIMARY KEY (`pest_disease_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_records` (
  `monitoring_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `plot_id` int(11) NOT NULL,
  `pest_disease_id` int(11) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `monitoring_results` varchar(255) NOT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`monitoring_record_id`),
  KEY `plot_id` (`plot_id`),
  KEY `pest_disease_id` (`pest_disease_id`),
  KEY `operator_id` (`operator_id`),
  CONSTRAINT `monitoring_records_ibfk_1` FOREIGN KEY (`plot_id`) REFERENCES `plots` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_records_ibfk_2` FOREIGN KEY (`pest_disease_id`) REFERENCES `pests_diseases` (`pest_disease_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_records_ibfk_3` FOREIGN KEY (`operator_id`) REFERENCES `operators` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operators` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
