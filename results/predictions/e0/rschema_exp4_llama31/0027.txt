-- Schema for database: rschema_0027
CREATE TABLE `clothing` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `style` varchar(255) NOT NULL,
  `production_date` datetime NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` enum('male','female') NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `clothing_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `producer_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `production_ibfk_1` (`clothing_id`),
  KEY `production_ibfk_2` (`producer_id`),
  CONSTRAINT `production_ibfk_1` FOREIGN KEY (`clothing_id`) REFERENCES `clothing` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_ibfk_2` FOREIGN KEY (`producer_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `clothing_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `storage_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `inventory_ibfk_1` (`clothing_id`),
  CONSTRAINT `inventory_ibfk_1` FOREIGN KEY (`clothing_id`) REFERENCES `clothing` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `clothing_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `salesperson_id` int(11) NOT NULL,
  `customer` varchar(255) NOT NULL,
  `sales_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `sales_ibfk_1` (`clothing_id`),
  KEY `sales_ibfk_2` (`salesperson_id`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`clothing_id`) REFERENCES `clothing` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`salesperson_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_appraisal` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `appraisal_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `performance_appraisal_ibfk_1` (`department_id`),
  CONSTRAINT `performance_appraisal_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
