-- Schema for database: rschema_0032
CREATE TABLE `customers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` char(20) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `locations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_tasks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `location_id` int(11) NOT NULL,
  `rental_date` datetime NOT NULL,
  `rental_duration` int(11) NOT NULL,
  `rental_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `rental_tasks_ibfk_1` (`customer_id`),
  KEY `rental_tasks_ibfk_2` (`location_id`),
  CONSTRAINT `rental_tasks_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_tasks_ibfk_2` FOREIGN KEY (`location_id`) REFERENCES `locations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_conditions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `rental_task_id` int(11) NOT NULL,
  `condition` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `vehicle_conditions_ibfk_1` (`rental_task_id`),
  CONSTRAINT `vehicle_conditions_ibfk_1` FOREIGN KEY (`rental_task_id`) REFERENCES `rental_tasks` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_feedbacks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `rental_task_id` int(11) NOT NULL,
  `feedback` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `customer_feedbacks_ibfk_1` (`rental_task_id`),
  CONSTRAINT `customer_feedbacks_ibfk_1` FOREIGN KEY (`rental_task_id`) REFERENCES `rental_tasks` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_assessments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `location_id` int(11) NOT NULL,
  `num_rentals` int(11) NOT NULL,
  `customer_satisfaction` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `performance_assessments_ibfk_1` (`location_id`),
  CONSTRAINT `performance_assessments_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `locations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
