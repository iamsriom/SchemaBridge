-- Schema for database: rschema_0040
CREATE TABLE `properties` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `address` varchar(255) NOT NULL,
  `area` varchar(255) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  `sales_status` enum('unsold','sold') NOT NULL DEFAULT 'unsold',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salespeople` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_contracts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `property_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `contract_date` datetime NOT NULL,
  `contract_price` decimal(10,2) NOT NULL,
  `signing_time` datetime NOT NULL,
  `contract_content` text NOT NULL,
  PRIMARY KEY (`id`),
  KEY `property_id` (`property_id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `sales_contracts_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `properties` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_contracts_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_tracking` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `salesperson_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `sales_contract_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `salesperson_id` (`salesperson_id`),
  KEY `property_id` (`property_id`),
  KEY `sales_contract_id` (`sales_contract_id`),
  CONSTRAINT `sales_tracking_ibfk_1` FOREIGN KEY (`salesperson_id`) REFERENCES `salespeople` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_tracking_ibfk_2` FOREIGN KEY (`property_id`) REFERENCES `properties` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_tracking_ibfk_3` FOREIGN KEY (`sales_contract_id`) REFERENCES `sales_contracts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
