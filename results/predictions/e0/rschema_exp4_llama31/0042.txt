-- Schema for database: rschema_0042
CREATE TABLE `community` (
  `community_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_number` varchar(255) NOT NULL,
  `community_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `property_manager_id` int(11) NOT NULL,
  PRIMARY KEY (`community_id`),
  KEY `community_ibfk_1` (`property_manager_id`),
  CONSTRAINT `community_ibfk_1` FOREIGN KEY (`property_manager_id`) REFERENCES `property_manager` (`property_manager_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property_manager` (
  `property_manager_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`property_manager_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resident` (
  `resident_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `resident_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `building_number` varchar(255) NOT NULL,
  PRIMARY KEY (`resident_id`),
  KEY `resident_ibfk_1` (`community_id`),
  CONSTRAINT `resident_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `building` (
  `building_id` int(11) NOT NULL AUTO_INCREMENT,
  `building_number` varchar(255) NOT NULL,
  `building_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`building_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_order` (
  `maintenance_order_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `resident_id` int(11) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_order_id`),
  KEY `maintenance_order_ibfk_1` (`community_id`),
  KEY `maintenance_order_ibfk_2` (`resident_id`),
  KEY `maintenance_order_ibfk_3` (`maintenance_personnel_id`),
  CONSTRAINT `maintenance_order_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_order_ibfk_2` FOREIGN KEY (`resident_id`) REFERENCES `resident` (`resident_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_order_ibfk_3` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `property_manager` (`property_manager_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `maintenance_personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_personnel_id`),
  UNIQUE KEY `maintenance_personnel_ibfk_1` (`maintenance_personnel_id`),
  CONSTRAINT `maintenance_personnel_ibfk_1` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `property_manager` (`property_manager_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
