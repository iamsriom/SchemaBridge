-- Schema for database: rschema_0052
CREATE TABLE `land` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `area` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `purpose` varchar(255) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_usage` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `land_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `area_used` decimal(10,2) NOT NULL,
  `years_of_usage` int(11) NOT NULL,
  `status` varchar(255) NOT NULL,
  `personnel` varchar(255) NOT NULL,
  `timing` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `land_usage_ibfk_1` (`land_id`),
  KEY `land_usage_ibfk_2` (`project_id`),
  CONSTRAINT `land_usage_ibfk_1` FOREIGN KEY (`land_id`) REFERENCES `land` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `land_usage_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_change` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `land_id` int(11) NOT NULL,
  `content` varchar(255) NOT NULL,
  `reason` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `land_change_ibfk_1` (`land_id`),
  CONSTRAINT `land_change_ibfk_1` FOREIGN KEY (`land_id`) REFERENCES `land` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
