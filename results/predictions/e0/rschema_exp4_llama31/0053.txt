-- Schema for database: rschema_0053
CREATE TABLE `processing_pools` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `responsible_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staffs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `responsibilities` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_pool_operation_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_pool_id` int(11) NOT NULL,
  `processing_capacity` int(11) NOT NULL,
  `efficiency` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `record_staff` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `processing_pool_id` (`processing_pool_id`),
  CONSTRAINT `processing_pool_operation_records_ibfk_1` FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pools` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_tasks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_pool_id` int(11) NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `processing_pool_id` (`processing_pool_id`),
  CONSTRAINT `maintenance_tasks_ibfk_1` FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pools` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
