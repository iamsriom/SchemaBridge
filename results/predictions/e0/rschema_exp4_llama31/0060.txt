-- Schema for database: rschema_0060
CREATE TABLE `projects` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  `leader_id` int(11) NOT NULL,
  `budget` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `projects_ibfk_1` (`leader_id`),
  CONSTRAINT `projects_ibfk_1` FOREIGN KEY (`leader_id`) REFERENCES `researchers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `researchers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_results` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `publication_date` datetime NOT NULL,
  `researcher_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `project_results_ibfk_1` (`project_id`),
  KEY `project_results_ibfk_2` (`researcher_id`),
  CONSTRAINT `project_results_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_results_ibfk_2` FOREIGN KEY (`researcher_id`) REFERENCES `researchers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fund_usage` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `purpose` varchar(255) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `usage_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `fund_usage_ibfk_1` (`project_id`),
  CONSTRAINT `fund_usage_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
