-- Schema for database: rschema_0061
CREATE TABLE `patents` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `patent_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventors` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `work_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `gender` varchar(255) NOT NULL,
  `telephone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patent_applications` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `application_number` varchar(255) NOT NULL,
  `patent_id` int(11) NOT NULL,
  `inventor_id` int(11) NOT NULL,
  `application_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `patent_id` (`patent_id`),
  KEY `inventor_id` (`inventor_id`),
  CONSTRAINT `patent_applications_ibfk_1` FOREIGN KEY (`patent_id`) REFERENCES `patents` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patent_applications_ibfk_2` FOREIGN KEY (`inventor_id`) REFERENCES `inventors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
