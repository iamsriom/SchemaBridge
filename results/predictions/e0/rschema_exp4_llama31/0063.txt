-- Schema for database: rschema_0063
CREATE TABLE `achievements` (
  `achievement_id` int(11) NOT NULL AUTO_INCREMENT,
  `achievement_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `research_and_development_date` datetime NOT NULL,
  `technical_field` varchar(255) NOT NULL,
  PRIMARY KEY (`achievement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `products` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `transformation_date` datetime NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `achievement_transformations` (
  `achievement_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `transformation_date` datetime NOT NULL,
  PRIMARY KEY (`achievement_id`, `product_id`),
  KEY `achievement_id` (`achievement_id`),
  KEY `product_id` (`product_id`),
  CONSTRAINT `achievement_transformations_ibfk_1` FOREIGN KEY (`achievement_id`) REFERENCES `achievements` (`achievement_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `achievement_transformations_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
