-- Schema for database: rschema_0064
CREATE TABLE `projects` (
  `project_number` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `application_unit` varchar(255) NOT NULL,
  PRIMARY KEY (`project_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review_experts` (
  `expert_number` int(11) NOT NULL,
  `expert_name` varchar(255) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`expert_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_reviews` (
  `project_number` int(11) NOT NULL,
  `expert_number` int(11) NOT NULL,
  `review_date` datetime NOT NULL,
  `review_opinion` varchar(255) NOT NULL,
  PRIMARY KEY (`project_number`, `expert_number`),
  KEY `project_number` (`project_number`),
  KEY `expert_number` (`expert_number`),
  CONSTRAINT `project_reviews_ibfk_1` FOREIGN KEY (`project_number`) REFERENCES `projects` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_reviews_ibfk_2` FOREIGN KEY (`expert_number`) REFERENCES `review_experts` (`expert_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
