-- Schema for database: rschema_0070
CREATE TABLE `mines` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `person_in_charge_id` (`person_in_charge_id`),
  CONSTRAINT `mines_ibfk_1` FOREIGN KEY (`person_in_charge_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ores` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `characteristics` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_plants` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `person_in_charge_id` (`person_in_charge_id`),
  CONSTRAINT `processing_plants_ibfk_1` FOREIGN KEY (`person_in_charge_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ore_outputs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_id` int(11) NOT NULL,
  `ore_id` int(11) NOT NULL,
  `production_date` datetime NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `mine_id` (`mine_id`),
  KEY `ore_id` (`ore_id`),
  CONSTRAINT `ore_outputs_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mines` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ore_outputs_ibfk_2` FOREIGN KEY (`ore_id`) REFERENCES `ores` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ore_processings` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `processing_plant_id` int(11) NOT NULL,
  `ore_id` int(11) NOT NULL,
  `processing_date` datetime NOT NULL,
  `quantity` int(11) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `processing_plant_id` (`processing_plant_id`),
  KEY `ore_id` (`ore_id`),
  CONSTRAINT `ore_processings_ibfk_1` FOREIGN KEY (`processing_plant_id`) REFERENCES `processing_plants` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ore_processings_ibfk_2` FOREIGN KEY (`ore_id`) REFERENCES `ores` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
