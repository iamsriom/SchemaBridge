-- Schema for database: rschema_0071
CREATE TABLE `mines` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_officers` (
  `officer_id` int(11) NOT NULL AUTO_INCREMENT,
  `officer_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `telephone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`officer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_officer_mine` (
  `officer_id` int(11) NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`officer_id`, `mine_id`),
  KEY `mine_id` (`mine_id`),
  CONSTRAINT `safety_officer_mine_ibfk_1` FOREIGN KEY (`officer_id`) REFERENCES `safety_officers` (`officer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `safety_officer_mine_ibfk_2` FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspection` (
  `inspection_id` int(11) NOT NULL AUTO_INCREMENT,
  `mine_id` int(11) NOT NULL,
  `inspection_date` datetime NOT NULL,
  `inspection_content` varchar(255) NOT NULL,
  `inspection_results` varchar(255) NOT NULL,
  `inspector_id` int(11) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  KEY `mine_id` (`mine_id`),
  KEY `inspector_id` (`inspector_id`),
  CONSTRAINT `safety_inspection_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `safety_inspection_ibfk_2` FOREIGN KEY (`inspector_id`) REFERENCES `safety_officers` (`officer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
