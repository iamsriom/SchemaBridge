-- Schema for database: rschema_0075
CREATE TABLE `analysts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `date_of_hire` date NOT NULL,
  `contact_number` char(10) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `analyses` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `analyst_id` int(11) NOT NULL,
  `stock_code` varchar(255) NOT NULL,
  `analysis_date` date NOT NULL,
  `results` varchar(255) NOT NULL,
  `reasoning` text NOT NULL,
  PRIMARY KEY (`id`),
  KEY `analyses_ibfk_1` (`analyst_id`),
  CONSTRAINT `analyses_ibfk_1` FOREIGN KEY (`analyst_id`) REFERENCES `analysts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `historical_prices` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `stock_code` varchar(255) NOT NULL,
  `date` date NOT NULL,
  `closing_price` decimal(10,2) NOT NULL,
  `opening_price` decimal(10,2) NOT NULL,
  `highest_price` decimal(10,2) NOT NULL,
  `lowest_price` decimal(10,2) NOT NULL,
  `trading_volume` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `historical_prices_ibfk_1` (`stock_code`),
  CONSTRAINT `historical_prices_ibfk_1` FOREIGN KEY (`stock_code`) REFERENCES `stocks` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stocks` (
  `code` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `current_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
