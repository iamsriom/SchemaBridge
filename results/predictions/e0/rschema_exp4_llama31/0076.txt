-- Schema for database: rschema_0076
CREATE TABLE `beer_manufacturers` (
  `manufacturer_id` int(11) NOT NULL AUTO_INCREMENT,
  `manufacturer_name` varchar(255) NOT NULL,
  `manufacturer_country` varchar(255) NOT NULL,
  PRIMARY KEY (`manufacturer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beer_types` (
  `beer_id` int(11) NOT NULL AUTO_INCREMENT,
  `beer_name` varchar(255) NOT NULL,
  `manufacturer_id` int(11) NOT NULL,
  `beer_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`beer_id`),
  KEY `beer_types_ibfk_1` (`manufacturer_id`),
  CONSTRAINT `beer_types_ibfk_1` FOREIGN KEY (`manufacturer_id`) REFERENCES `beer_manufacturers` (`manufacturer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bars` (
  `bar_id` int(11) NOT NULL AUTO_INCREMENT,
  `bar_name` varchar(255) NOT NULL,
  `bar_address` varchar(255) NOT NULL,
  `bar_phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`bar_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drinkers` (
  `drinker_id` int(11) NOT NULL AUTO_INCREMENT,
  `drinker_name` varchar(255) NOT NULL,
  `drinker_gender` enum('male','female') NOT NULL,
  `drinker_age` int(3) NOT NULL,
  `drinker_phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`drinker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `sale_id` int(11) NOT NULL AUTO_INCREMENT,
  `sale_date` datetime NOT NULL,
  `beer_id` int(11) NOT NULL,
  `bar_id` int(11) NOT NULL,
  `sale_quantity` int(11) NOT NULL,
  `sale_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`sale_id`),
  KEY `sales_ibfk_1` (`beer_id`),
  KEY `sales_ibfk_2` (`bar_id`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`beer_id`) REFERENCES `beer_types` (`beer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`bar_id`) REFERENCES `bars` (`bar_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
