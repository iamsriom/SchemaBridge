-- Schema for database: rschema_0081
CREATE TABLE `user_groups` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` char(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `one_time_password` char(255) DEFAULT NULL,
  `password_reset_key` char(255) DEFAULT NULL,
  `associated_user_group_id` int(11) NOT NULL,
  `activation_status` tinyint(1) NOT NULL DEFAULT 0,
  `user_creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `creator` int(11) NOT NULL,
  `last_updated_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  `updater` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `FK_users_user_groups` (`associated_user_group_id`),
  CONSTRAINT `FK_users_user_groups` FOREIGN KEY (`associated_user_group_id`) REFERENCES `user_groups` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_keys` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `associated_user_id` int(11) NOT NULL,
  `key_content` char(255) NOT NULL,
  `permission_level` tinyint(1) NOT NULL,
  `ignore_call_limits` tinyint(1) NOT NULL DEFAULT 0,
  `is_private_key` tinyint(1) NOT NULL DEFAULT 0,
  `allowed_ip_address_list` varchar(255) DEFAULT NULL,
  `key_creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `FK_api_keys_users` (`associated_user_id`),
  CONSTRAINT `FK_api_keys_users` FOREIGN KEY (`associated_user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_call_logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `called_api_resource_path` varchar(255) NOT NULL,
  `http_request_method` varchar(255) NOT NULL,
  `request_parameters` text,
  `used_api_key_id` int(11) NOT NULL,
  `client_ip_address` varchar(255) NOT NULL,
  `call_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `response_time` float NOT NULL DEFAULT 0,
  `authorization_status` tinyint(1) NOT NULL DEFAULT 0,
  `api_response_status_code` tinyint(3) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `FK_api_call_logs_api_keys` (`used_api_key_id`),
  CONSTRAINT `FK_api_call_logs_api_keys` FOREIGN KEY (`used_api_key_id`) REFERENCES `api_keys` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
