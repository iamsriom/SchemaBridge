-- Schema for database: rschema_0085
CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` char(60) NOT NULL,
  `profile` text,
  `avatar` varchar(255),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `ingredients` text,
  `steps` text,
  `total_time` int(11),
  `portion` varchar(255),
  `spicy_level` int(11),
  `bitter_level` int(11),
  `sweet_level` int(11),
  `meaty_level` int(11),
  `salty_level` int(11),
  `sour_level` int(11),
  `rating` decimal(10,2),
  `num_ratings` int(11) NOT NULL DEFAULT 0,
  `large_image_filename` varchar(255),
  `calories` int(11),
  `creator_id` int(11) NOT NULL,
  `small_image_filename` varchar(255),
  PRIMARY KEY (`id`),
  KEY `recipes_ibfk_1` (`creator_id`),
  CONSTRAINT `recipes_ibfk_1` FOREIGN KEY (`creator_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `blogs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `author_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `publication_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `summary` text,
  PRIMARY KEY (`id`),
  KEY `blogs_ibfk_1` (`author_id`),
  CONSTRAINT `blogs_ibfk_1` FOREIGN KEY (`author_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipe_collections` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `recipe_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `recipe_collections_ibfk_1` (`user_id`),
  KEY `recipe_collections_ibfk_2` (`recipe_id`),
  CONSTRAINT `recipe_collections_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `recipe_collections_ibfk_2` FOREIGN KEY (`recipe_id`) REFERENCES `recipes` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
