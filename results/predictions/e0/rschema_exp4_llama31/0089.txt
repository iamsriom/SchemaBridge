-- Schema for database: rschema_0089
CREATE TABLE `accounts` (
  `address` varchar(255) NOT NULL,
  `balance` decimal(65,30) NOT NULL DEFAULT 0.000000000000000000000000000000,
  PRIMARY KEY (`address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `blocks` (
  `hash` char(64) NOT NULL,
  `height` int(11) NOT NULL,
  `timestamp` datetime NOT NULL,
  `previous_hash` char(64) NOT NULL,
  `merkle_root` char(64) NOT NULL,
  PRIMARY KEY (`hash`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `hash` char(64) NOT NULL,
  `timestamp` datetime NOT NULL,
  `fee` decimal(65,30) NOT NULL DEFAULT 0.000000000000000000000000000000,
  `input_address` varchar(255) NOT NULL,
  `output_address` varchar(255) NOT NULL,
  PRIMARY KEY (`hash`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_blocks` (
  `transaction_hash` char(64) NOT NULL,
  `block_hash` char(64) NOT NULL,
  PRIMARY KEY (`transaction_hash`),
  KEY `transaction_blocks_ibfk_1` (`block_hash`),
  CONSTRAINT `transaction_blocks_ibfk_1` FOREIGN KEY (`block_hash`) REFERENCES `blocks` (`hash`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_accounts` (
  `transaction_hash` char(64) NOT NULL,
  `account_address` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_hash`),
  KEY `transaction_accounts_ibfk_1` (`account_address`),
  CONSTRAINT `transaction_accounts_ibfk_1` FOREIGN KEY (`account_address`) REFERENCES `accounts` (`address`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
