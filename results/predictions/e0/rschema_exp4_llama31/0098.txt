-- Schema for database: rschema_0098
CREATE TABLE `technology_solutions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text,
  `implementation_status` enum('selected','implemented','maintained') NOT NULL DEFAULT 'selected',
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `evaluation_criteria` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `solution_evaluation` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `technology_solution_id` int(11) NOT NULL,
  `evaluation_criteria_id` int(11) NOT NULL,
  `effectiveness` decimal(10,2) NOT NULL,
  `cost_effectiveness` decimal(10,2) NOT NULL,
  `implementation_progress` decimal(10,2) NOT NULL,
  `user_feedback` text,
  PRIMARY KEY (`id`),
  KEY `solution_evaluation_ibfk_1` (`technology_solution_id`),
  KEY `solution_evaluation_ibfk_2` (`evaluation_criteria_id`),
  CONSTRAINT `solution_evaluation_ibfk_1` FOREIGN KEY (`technology_solution_id`) REFERENCES `technology_solutions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `solution_evaluation_ibfk_2` FOREIGN KEY (`evaluation_criteria_id`) REFERENCES `evaluation_criteria` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_feedback` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `solution_evaluation_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `feedback` text,
  PRIMARY KEY (`id`),
  KEY `user_feedback_ibfk_1` (`solution_evaluation_id`),
  KEY `user_feedback_ibfk_2` (`user_id`),
  CONSTRAINT `user_feedback_ibfk_1` FOREIGN KEY (`solution_evaluation_id`) REFERENCES `solution_evaluation` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_feedback_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `email` (`email`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
