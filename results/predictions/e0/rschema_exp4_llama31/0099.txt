-- Schema for database: rschema_0099
CREATE TABLE `speakers` (
  `speaker_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `surname` varchar(255) NOT NULL,
  PRIMARY KEY (`speaker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conferences` (
  `conference_id` int(11) NOT NULL AUTO_INCREMENT,
  `conference_time` datetime NOT NULL,
  `speaker_id` int(11) NOT NULL,
  PRIMARY KEY (`conference_id`),
  KEY `speaker_id` (`speaker_id`),
  CONSTRAINT `conferences_ibfk_1` FOREIGN KEY (`speaker_id`) REFERENCES `speakers` (`speaker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
