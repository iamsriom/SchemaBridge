-- Schema for database: rschema_0103
CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `hire_date` datetime NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_positions` (
  `employee_id` int(11) NOT NULL,
  `original_department` varchar(255) NOT NULL,
  `original_position` varchar(255) NOT NULL,
  `current_department` varchar(255) NOT NULL,
  `current_position` varchar(255) NOT NULL,
  `entry_date` datetime NOT NULL,
  `departure_date` datetime DEFAULT NULL,
  PRIMARY KEY (`employee_id`, `entry_date`),
  KEY `employee_id` (`employee_id`),
  CONSTRAINT `employee_positions_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_code` char(6) NOT NULL,
  `name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`department_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_departments` (
  `employee_id` int(11) NOT NULL,
  `department_code` char(6) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  PRIMARY KEY (`employee_id`, `department_code`),
  KEY `employee_id` (`employee_id`),
  KEY `department_code` (`department_code`),
  CONSTRAINT `employee_departments_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_departments_ibfk_2` FOREIGN KEY (`department_code`) REFERENCES `departments` (`department_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_standards` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `description` varchar(255) NOT NULL,
  `value` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salaries` (
  `employee_id` int(11) NOT NULL,
  `salary_standard_id` int(11) NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `payment_date` datetime NOT NULL,
  PRIMARY KEY (`employee_id`, `salary_standard_id`, `payment_date`),
  KEY `employee_id` (`employee_id`),
  KEY `salary_standard_id` (`salary_standard_id`),
  CONSTRAINT `salaries_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `salaries_ibfk_2` FOREIGN KEY (`salary_standard_id`) REFERENCES `salary_standards` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
