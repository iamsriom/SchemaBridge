-- Schema for database: rschema_0109
CREATE TABLE `members` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `country_code` char(3) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `pronunciation` varchar(255) DEFAULT NULL,
  `company_name` varchar(255) DEFAULT NULL,
  `publicly_accessible` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `name_unique` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_addresses` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `country` char(3) NOT NULL,
  `postal_code` char(10) NOT NULL,
  `address` varchar(255) NOT NULL,
  `telephone_number` char(20) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `member_id_foreign` (`member_id`),
  CONSTRAINT `member_id_foreign` FOREIGN KEY (`member_id`) REFERENCES `members` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `data_cache` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `country_code` char(3) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `member_id_foreign_1` (`member_id`),
  CONSTRAINT `member_id_foreign_1` FOREIGN KEY (`member_id`) REFERENCES `members` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
