-- Schema for database: rschema_0118
CREATE TABLE `departments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `head` varchar(255) NOT NULL,
  `responsibilities` text,
  `location` varchar(255) NOT NULL,
  `total_personnel` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `date_of_birth` date NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `entry_time` date NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `department_id` (`department_id`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendance_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `attendance_time` datetime NOT NULL,
  `attendance_type` varchar(255) NOT NULL,
  `leave_time` datetime DEFAULT NULL,
  `leave_type` varchar(255) DEFAULT NULL,
  `overtime_time` datetime DEFAULT NULL,
  `overtime_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `employee_id` (`employee_id`),
  CONSTRAINT `attendance_records_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
