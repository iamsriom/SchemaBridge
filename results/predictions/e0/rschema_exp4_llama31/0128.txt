-- Schema for database: rschema_0128
CREATE TABLE `task_execution` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `execution_id` int(11) NOT NULL,
  `parent_execution_id` int(11) DEFAULT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `task_name` varchar(255) NOT NULL,
  `exit_code` int(11) NOT NULL,
  `exit_message` varchar(255) DEFAULT NULL,
  `error_message` varchar(255) DEFAULT NULL,
  `last_update_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `task_execution_ibfk_1` (`execution_id`),
  KEY `task_execution_ibfk_2` (`parent_execution_id`),
  CONSTRAINT `task_execution_ibfk_1` FOREIGN KEY (`execution_id`) REFERENCES `task_execution` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_execution_ibfk_2` FOREIGN KEY (`parent_execution_id`) REFERENCES `task_execution` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_parameter` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `task_execution_id` int(11) NOT NULL,
  `parameter_name` varchar(255) NOT NULL,
  `parameter_value` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `task_parameter_ibfk_1` (`task_execution_id`),
  CONSTRAINT `task_parameter_ibfk_1` FOREIGN KEY (`task_execution_id`) REFERENCES `task_execution` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_batch` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `batch_id` int(11) NOT NULL,
  `batch_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_batch_execution` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `task_execution_id` int(11) NOT NULL,
  `task_batch_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `task_batch_execution_ibfk_1` (`task_execution_id`),
  KEY `task_batch_execution_ibfk_2` (`task_batch_id`),
  CONSTRAINT `task_batch_execution_ibfk_1` FOREIGN KEY (`task_execution_id`) REFERENCES `task_execution` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_batch_execution_ibfk_2` FOREIGN KEY (`task_batch_id`) REFERENCES `task_batch` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
