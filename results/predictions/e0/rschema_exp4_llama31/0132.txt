-- Schema for database: rschema_0132
CREATE TABLE `students` (
  `student_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `class` varchar(255) NOT NULL,
  `enrollment_time` datetime NOT NULL,
  `place_of_residence` varchar(255) NOT NULL,
  PRIMARY KEY (`student_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courses` (
  `course_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `status` char(1) NOT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teachers` (
  `teacher_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`teacher_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_courses` (
  `student_number` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  PRIMARY KEY (`student_number`, `course_id`),
  KEY `student_courses_ibfk_1` (`student_number`),
  KEY `student_courses_ibfk_2` (`course_id`),
  CONSTRAINT `student_courses_ibfk_1` FOREIGN KEY (`student_number`) REFERENCES `students` (`student_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_courses_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teaching_situations` (
  `course_id` int(11) NOT NULL,
  `teacher_number` int(11) NOT NULL,
  `class_start_time` datetime NOT NULL,
  PRIMARY KEY (`course_id`, `teacher_number`),
  KEY `teaching_situations_ibfk_1` (`course_id`),
  KEY `teaching_situations_ibfk_2` (`teacher_number`),
  CONSTRAINT `teaching_situations_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `teaching_situations_ibfk_2` FOREIGN KEY (`teacher_number`) REFERENCES `teachers` (`teacher_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
