-- Schema for database: rschema_0133
CREATE TABLE `book_categories` (
  `book_type_id` int(11) NOT NULL,
  `type_name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`book_type_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `books` (
  `book_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `author` varchar(255) NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `publication_date` datetime NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `book_type_id` int(11) NOT NULL,
  `content_description` text NOT NULL,
  PRIMARY KEY (`book_id`),
  KEY `book_type_id` (`book_type_id`),
  CONSTRAINT `books_ibfk_1` FOREIGN KEY (`book_type_id`) REFERENCES `book_categories` (`book_type_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reader_categories` (
  `reader_type_id` int(11) NOT NULL,
  `type_name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`reader_type_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `readers` (
  `reader_id` int(11) NOT NULL,
  `reader_type_id` int(11) NOT NULL,
  `contact_number` varchar(255) NOT NULL,
  `credit_score` decimal(10,2) NOT NULL,
  `description` varchar(255) NOT NULL,
  PRIMARY KEY (`reader_id`),
  KEY `reader_type_id` (`reader_type_id`),
  CONSTRAINT `readers_ibfk_1` FOREIGN KEY (`reader_type_id`) REFERENCES `reader_categories` (`reader_type_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `borrowings` (
  `borrowing_id` int(11) NOT NULL AUTO_INCREMENT,
  `book_id` int(11) NOT NULL,
  `reader_id` int(11) NOT NULL,
  `borrow_date` datetime NOT NULL,
  `return_date` datetime DEFAULT NULL,
  `renewal_count` int(11) NOT NULL,
  `fine` decimal(10,2) NOT NULL,
  PRIMARY KEY (`borrowing_id`),
  KEY `book_id` (`book_id`),
  KEY `reader_id` (`reader_id`),
  CONSTRAINT `borrowings_ibfk_1` FOREIGN KEY (`book_id`) REFERENCES `books` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `borrowings_ibfk_2` FOREIGN KEY (`reader_id`) REFERENCES `readers` (`reader_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
