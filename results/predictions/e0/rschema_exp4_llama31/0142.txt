-- Schema for database: rschema_0142
CREATE TABLE `coaches` (
  `coach_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  PRIMARY KEY (`coach_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teams` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `representative_region` varchar(255) NOT NULL,
  `establishment_date` date NOT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `players` (
  `player_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `height` decimal(5,2) NOT NULL,
  `joining_date` date NOT NULL,
  PRIMARY KEY (`player_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `coach_team` (
  `coach_id` int(11) NOT NULL,
  `team_name` varchar(255) NOT NULL,
  PRIMARY KEY (`coach_id`, `team_name`),
  KEY `coach_team_ibfk_1` (`coach_id`),
  KEY `coach_team_ibfk_2` (`team_name`),
  CONSTRAINT `coach_team_ibfk_1` FOREIGN KEY (`coach_id`) REFERENCES `coaches` (`coach_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `coach_team_ibfk_2` FOREIGN KEY (`team_name`) REFERENCES `teams` (`name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `player_team` (
  `player_id` int(11) NOT NULL,
  `team_name` varchar(255) NOT NULL,
  PRIMARY KEY (`player_id`, `team_name`),
  KEY `player_team_ibfk_1` (`player_id`),
  KEY `player_team_ibfk_2` (`team_name`),
  CONSTRAINT `player_team_ibfk_1` FOREIGN KEY (`player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `player_team_ibfk_2` FOREIGN KEY (`team_name`) REFERENCES `teams` (`name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
