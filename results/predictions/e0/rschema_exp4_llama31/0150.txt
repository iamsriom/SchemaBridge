-- Schema for database: rschema_0150
CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `job_title` varchar(255) NOT NULL,
  `work_location` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`),
  KEY `employees_ibfk_1` (`work_location`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`work_location`) REFERENCES `warehouses` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL,
  `budget` decimal(10,2) NOT NULL,
  `start_date` datetime NOT NULL,
  PRIMARY KEY (`project_id`),
  KEY `projects_ibfk_1` (`project_id`),
  KEY `projects_ibfk_2` (`project_id`),
  KEY `projects_ibfk_3` (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_requires_parts` (
  `project_id` int(11) NOT NULL,
  `part_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`project_id`, `part_id`),
  KEY `project_requires_parts_ibfk_1` (`project_id`),
  KEY `project_requires_parts_ibfk_2` (`part_id`),
  CONSTRAINT `project_requires_parts_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_requires_parts_ibfk_2` FOREIGN KEY (`part_id`) REFERENCES `parts` (`part_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `suppliers` (
  `supplier_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`supplier_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `supplier_supplies_project` (
  `supplier_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `supply_quantity` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`supplier_id`, `project_id`),
  KEY `supplier_supplies_project_ibfk_1` (`supplier_id`),
  KEY `supplier_supplies_project_ibfk_2` (`project_id`),
  CONSTRAINT `supplier_supplies_project_ibfk_1` FOREIGN KEY (`supplier_id`) REFERENCES `suppliers` (`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `supplier_supplies_project_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouses` (
  `warehouse_id` int(11) NOT NULL,
  `area` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`warehouse_id`),
  KEY `warehouses_ibfk_1` (`warehouse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouse_inventory_parts` (
  `warehouse_id` int(11) NOT NULL,
  `part_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`warehouse_id`, `part_id`),
  KEY `warehouse_inventory_parts_ibfk_1` (`warehouse_id`),
  KEY `warehouse_inventory_parts_ibfk_2` (`part_id`),
  CONSTRAINT `warehouse_inventory_parts_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouses` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `warehouse_inventory_parts_ibfk_2` FOREIGN KEY (`part_id`) REFERENCES `parts` (`part_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parts` (
  `part_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `specification` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`part_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
