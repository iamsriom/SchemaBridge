-- Schema for database: rschema_0166
CREATE TABLE `patients` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `medical_records` text,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctors` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `images` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_id` int(11) NOT NULL,
  `image_type` varchar(255) NOT NULL,
  `film_time` datetime NOT NULL,
  `image_data` blob NOT NULL,
  `image_resolution` varchar(255) NOT NULL,
  `image_enhancement_info` text,
  `upload_time` datetime NOT NULL,
  `server_path` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `images_ibfk_1` (`patient_id`),
  CONSTRAINT `images_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patients` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `doctor_id` int(11) NOT NULL,
  `image_id` int(11) NOT NULL,
  `operation_content` text NOT NULL,
  `operation_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `operations_ibfk_1` (`doctor_id`),
  KEY `operations_ibfk_2` (`image_id`),
  CONSTRAINT `operations_ibfk_1` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `operations_ibfk_2` FOREIGN KEY (`image_id`) REFERENCES `images` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `image_shares` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `image_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `sharing_time` datetime NOT NULL,
  `target_institution` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `image_shares_ibfk_1` (`image_id`),
  KEY `image_shares_ibfk_2` (`doctor_id`),
  CONSTRAINT `image_shares_ibfk_1` FOREIGN KEY (`image_id`) REFERENCES `images` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `image_shares_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
