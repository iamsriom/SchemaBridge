-- Schema for database: rschema_0171
CREATE TABLE `parking_spaces` (
  `parking_space_id` int(11) NOT NULL AUTO_INCREMENT,
  `parking_space_number` varchar(255) NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`parking_space_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `entrances` (
  `entrance_id` int(11) NOT NULL AUTO_INCREMENT,
  `entrance_number` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`entrance_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exits` (
  `exit_id` int(11) NOT NULL AUTO_INCREMENT,
  `exit_number` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`exit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `ticket_number` varchar(255) NOT NULL,
  `vehicle_model` varchar(255) NOT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_records` (
  `parking_record_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `parking_space_id` int(11) NOT NULL,
  `entrance_id` int(11) NOT NULL,
  `exit_id` int(11) DEFAULT NULL,
  `entry_time` datetime NOT NULL,
  `exit_time` datetime DEFAULT NULL,
  PRIMARY KEY (`parking_record_id`),
  KEY `vehicle_id` (`vehicle_id`),
  KEY `parking_space_id` (`parking_space_id`),
  KEY `entrance_id` (`entrance_id`),
  KEY `exit_id` (`exit_id`),
  CONSTRAINT `parking_records_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_records_ibfk_2` FOREIGN KEY (`parking_space_id`) REFERENCES `parking_spaces` (`parking_space_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_records_ibfk_3` FOREIGN KEY (`entrance_id`) REFERENCES `entrances` (`entrance_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_records_ibfk_4` FOREIGN KEY (`exit_id`) REFERENCES `exits` (`exit_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
