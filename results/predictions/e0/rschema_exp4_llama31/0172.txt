-- Schema for database: rschema_0172
CREATE TABLE `vital_signs` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `record_time` datetime NOT NULL,
  `temperature` float NOT NULL,
  `blood_pressure` float NOT NULL,
  `heart_rate` int(11) NOT NULL,
  PRIMARY KEY (`log_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `normal_ranges` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `temperature_min` float NOT NULL,
  `temperature_max` float NOT NULL,
  `blood_pressure_min` float NOT NULL,
  `blood_pressure_max` float NOT NULL,
  `heart_rate_min` int(11) NOT NULL,
  `heart_rate_max` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alerts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `alert_id` int(11) NOT NULL,
  `patient_id` int(11) NOT NULL,
  `abnormal_vital_signs` varchar(255) NOT NULL,
  `alert_time` datetime NOT NULL,
  `alert_content` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patients` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alert_normal_ranges` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `alert_id` int(11) NOT NULL,
  `normal_range_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`alert_id`) REFERENCES `alerts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`normal_range_id`) REFERENCES `normal_ranges` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
