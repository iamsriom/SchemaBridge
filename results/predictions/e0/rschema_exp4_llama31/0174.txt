-- Schema for database: rschema_0174
CREATE TABLE `patients` (
  `medical_record_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `address` varchar(255) NOT NULL,
  `id_card_number` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `admission_time` datetime NOT NULL,
  PRIMARY KEY (`medical_record_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_signs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_record_number` int(11) NOT NULL,
  `body_temperature` float NOT NULL,
  `blood_pressure` varchar(255) NOT NULL,
  `heart_rate` int(11) NOT NULL,
  `vital_sign_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `patients_ibfk_1` (`medical_record_number`),
  CONSTRAINT `patients_ibfk_1` FOREIGN KEY (`medical_record_number`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_sign_limits` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `vital_sign_name` varchar(255) NOT NULL,
  `upper_limit` float NOT NULL,
  `lower_limit` float NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_record_number` int(11) NOT NULL,
  `diagnosis_time` datetime NOT NULL,
  `diagnosis_content` text NOT NULL,
  PRIMARY KEY (`id`),
  KEY `patients_ibfk_2` (`medical_record_number`),
  CONSTRAINT `patients_ibfk_2` FOREIGN KEY (`medical_record_number`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `treatment_recommendations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_record_number` int(11) NOT NULL,
  `recommendation_time` datetime NOT NULL,
  `recommendation_content` text NOT NULL,
  PRIMARY KEY (`id`),
  KEY `patients_ibfk_3` (`medical_record_number`),
  CONSTRAINT `patients_ibfk_3` FOREIGN KEY (`medical_record_number`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_record_number` int(11) NOT NULL,
  `vital_sign_name` varchar(255) NOT NULL,
  `vital_sign_value` float NOT NULL,
  `log_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `patients_ibfk_4` (`medical_record_number`),
  KEY `vital_sign_limits_ibfk_1` (`vital_sign_name`),
  CONSTRAINT `patients_ibfk_4` FOREIGN KEY (`medical_record_number`) REFERENCES `patients` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `vital_sign_limits_ibfk_1` FOREIGN KEY (`vital_sign_name`) REFERENCES `vital_sign_limits` (`vital_sign_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
