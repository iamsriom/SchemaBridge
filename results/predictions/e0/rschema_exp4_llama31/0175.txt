-- Schema for database: rschema_0175
CREATE TABLE `alumni` (
  `alumni_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `employer` varchar(255) DEFAULT NULL,
  `job_title` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `class` varchar(255) DEFAULT NULL,
  `enrollment_year` int(11) DEFAULT NULL,
  `id_number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`alumni_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donations` (
  `donation_id` int(11) NOT NULL AUTO_INCREMENT,
  `donating_alumni_id` int(11) NOT NULL,
  `donation_time` datetime NOT NULL,
  `donation_amount` decimal(10,2) NOT NULL,
  `donation_type` varchar(255) NOT NULL,
  `donation_usage` varchar(255) NOT NULL,
  PRIMARY KEY (`donation_id`),
  KEY `donating_alumni_id` (`donating_alumni_id`),
  CONSTRAINT `donations_ibfk_1` FOREIGN KEY (`donating_alumni_id`) REFERENCES `alumni` (`alumni_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `project_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_benefits` (
  `project_id` int(11) NOT NULL,
  `beneficiary_unit` varchar(255) NOT NULL,
  `benefit_percentage` decimal(5,2) NOT NULL,
  `benefit_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`project_id`, `beneficiary_unit`),
  KEY `project_id` (`project_id`),
  CONSTRAINT `project_benefits_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donation_project` (
  `donation_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`donation_id`, `project_id`),
  KEY `donation_id` (`donation_id`),
  KEY `project_id` (`project_id`),
  CONSTRAINT `donation_project_ibfk_1` FOREIGN KEY (`donation_id`) REFERENCES `donations` (`donation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `donation_project_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
