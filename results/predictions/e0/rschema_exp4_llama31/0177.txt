-- Schema for database: rschema_0177
CREATE TABLE `abstract_file` (
  `file_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` enum('file','directory') NOT NULL,
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`file_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `folder` (
  `folder_id` int(11) NOT NULL AUTO_INCREMENT,
  `file_id` int(11) NOT NULL,
  `child_list` text,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`folder_id`),
  KEY `folder_ibfk_1` (`file_id`),
  CONSTRAINT `folder_ibfk_1` FOREIGN KEY (`file_id`) REFERENCES `abstract_file` (`file_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `file` (
  `file_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` enum('file') NOT NULL,
  `file_size` int(11) NOT NULL,
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`file_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
