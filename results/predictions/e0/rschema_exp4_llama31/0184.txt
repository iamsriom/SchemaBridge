-- Schema for database: rschema_0184
CREATE TABLE `members` (
  `membership_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `registration_date` datetime NOT NULL,
  `points` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`membership_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rooms` (
  `room_id` int(11) NOT NULL,
  `room_number` varchar(10) NOT NULL,
  `room_type` varchar(50) NOT NULL,
  PRIMARY KEY (`room_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bookings` (
  `booking_id` int(11) NOT NULL,
  `membership_number` int(11) NOT NULL,
  `room_id` int(11) NOT NULL,
  `booking_status` enum('pending','confirmed','cancelled') NOT NULL DEFAULT 'pending',
  `points_change` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`booking_id`),
  KEY `members_ibfk_1` (`membership_number`),
  KEY `rooms_ibfk_1` (`room_id`),
  CONSTRAINT `members_ibfk_1` FOREIGN KEY (`membership_number`) REFERENCES `members` (`membership_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rooms_ibfk_1` FOREIGN KEY (`room_id`) REFERENCES `rooms` (`room_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
