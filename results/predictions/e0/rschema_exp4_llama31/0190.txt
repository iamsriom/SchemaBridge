-- Schema for database: rschema_0190
CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone_number` char(20) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `devices` (
  `device_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `model_number` varchar(255) NOT NULL,
  `fault_description` text,
  PRIMARY KEY (`device_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `device_id` int(11) NOT NULL,
  `repair_date` datetime NOT NULL,
  `repair_type` varchar(255) NOT NULL,
  `repair_duration` int(11) NOT NULL,
  `repair_cost` decimal(10,2) NOT NULL,
  `repair_results` text,
  `customer_feedback` text,
  PRIMARY KEY (`task_id`),
  KEY `customer_id` (`customer_id`),
  KEY `device_id` (`device_id`),
  CONSTRAINT `repair_tasks_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_tasks_ibfk_2` FOREIGN KEY (`device_id`) REFERENCES `devices` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_workstations` (
  `workstation_id` int(11) NOT NULL AUTO_INCREMENT,
  `workstation_name` varchar(255) NOT NULL,
  PRIMARY KEY (`workstation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technicians` (
  `technician_id` int(11) NOT NULL AUTO_INCREMENT,
  `workstation_id` int(11) NOT NULL,
  `technician_name` varchar(255) NOT NULL,
  PRIMARY KEY (`technician_id`),
  KEY `workstation_id` (`workstation_id`),
  CONSTRAINT `technicians_ibfk_1` FOREIGN KEY (`workstation_id`) REFERENCES `repair_workstations` (`workstation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_assignments` (
  `task_id` int(11) NOT NULL,
  `technician_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`, `technician_id`),
  KEY `task_id` (`task_id`),
  KEY `technician_id` (`technician_id`),
  CONSTRAINT `task_assignments_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `repair_tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_assignments_ibfk_2` FOREIGN KEY (`technician_id`) REFERENCES `technicians` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
