-- Schema for database: rschema_0196
CREATE TABLE `applicants` (
  `applicant_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  `identification_number` varchar(20) NOT NULL,
  PRIMARY KEY (`applicant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_applications` (
  `application_number` int(11) NOT NULL AUTO_INCREMENT,
  `applicant_id` int(11) NOT NULL,
  `requested_loan_amount` decimal(10,2) NOT NULL,
  `loan_term` int(11) NOT NULL,
  `approval_status` enum('pending','approved','rejected') NOT NULL DEFAULT 'pending',
  PRIMARY KEY (`application_number`),
  KEY `applicant_id` (`applicant_id`),
  CONSTRAINT `loan_applications_ibfk_1` FOREIGN KEY (`applicant_id`) REFERENCES `applicants` (`applicant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval_forms` (
  `approval_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_number` int(11) NOT NULL,
  `approval_date` datetime NOT NULL,
  `approval_result` enum('approved','rejected') NOT NULL,
  `approver_id` int(11) NOT NULL,
  PRIMARY KEY (`approval_id`),
  KEY `application_number` (`application_number`),
  KEY `approver_id` (`approver_id`),
  CONSTRAINT `loan_approval_forms_ibfk_1` FOREIGN KEY (`application_number`) REFERENCES `loan_applications` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_forms_ibfk_2` FOREIGN KEY (`approver_id`) REFERENCES `approvers` (`approver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approvers` (
  `approver_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `email_address` varchar(255) NOT NULL,
  PRIMARY KEY (`approver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_actions` (
  `action_id` int(11) NOT NULL AUTO_INCREMENT,
  `application_number` int(11) NOT NULL,
  `approval_id` int(11) NOT NULL,
  `approver_id` int(11) NOT NULL,
  `action_date` datetime NOT NULL,
  `action_result` enum('approved','rejected') NOT NULL,
  PRIMARY KEY (`action_id`),
  KEY `application_number` (`application_number`),
  KEY `approval_id` (`approval_id`),
  KEY `approver_id` (`approver_id`),
  CONSTRAINT `approval_actions_ibfk_1` FOREIGN KEY (`application_number`) REFERENCES `loan_applications` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `approval_actions_ibfk_2` FOREIGN KEY (`approval_id`) REFERENCES `loan_approval_forms` (`approval_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `approval_actions_ibfk_3` FOREIGN KEY (`approver_id`) REFERENCES `approvers` (`approver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
