-- Schema for database: rschema_0200
CREATE TABLE `customers` (
  `customer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_person` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`customer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
  `order_number` int(11) NOT NULL,
  `customer_number` int(11) NOT NULL,
  `order_time` datetime NOT NULL,
  `product_number` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `total_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`order_number`),
  KEY `customer_number` (`customer_number`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`customer_number`) REFERENCES `customers` (`customer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_ratings` (
  `customer_number` int(11) NOT NULL,
  `credit_rating` int(11) NOT NULL,
  `historical_orders` int(11) NOT NULL,
  `order_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`customer_number`),
  KEY `customer_number` (`customer_number`),
  CONSTRAINT `credit_ratings_ibfk_1` FOREIGN KEY (`customer_number`) REFERENCES `customers` (`customer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
