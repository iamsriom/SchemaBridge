-- Schema for database: rschema_0214
CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `assessments` (
  `assessment_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `assessment_date` datetime NOT NULL,
  `assessment_content` text NOT NULL,
  `assessment_score` int(11) NOT NULL,
  PRIMARY KEY (`assessment_id`),
  KEY `employees_ibfk_1` (`employee_id`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monthly_scores` (
  `month` char(6) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `total_score` int(11) NOT NULL,
  PRIMARY KEY (`month`, `employee_id`),
  KEY `employees_ibfk_2` (`employee_id`),
  CONSTRAINT `employees_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
