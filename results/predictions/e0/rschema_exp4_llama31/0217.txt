-- Schema for database: rschema_0217
CREATE TABLE `courses` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `teacher` varchar(255) DEFAULT NULL,
  `class_hours` int(11) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `schedule` datetime NOT NULL,
  PRIMARY KEY (`course_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_class` (
  `student_id` int(11) NOT NULL,
  `class` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`, `class`),
  FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `students` (
  `student_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `grade` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_enrollments` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  PRIMARY KEY (`student_id`, `course_id`),
  FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `system_settings` (
  `system_name` varchar(255) NOT NULL,
  `theme` varchar(255) NOT NULL,
  `page_configuration` varchar(255) NOT NULL,
  `version` varchar(255) NOT NULL,
  PRIMARY KEY (`system_name`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tool_assets` (
  `asset_number` varchar(255) NOT NULL,
  `holder` varchar(255) NOT NULL,
  PRIMARY KEY (`asset_number`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tool_loans` (
  `record_id` int(11) NOT NULL AUTO_INCREMENT,
  `borrower` varchar(255) NOT NULL,
  `asset_number` varchar(255) NOT NULL,
  `time` datetime NOT NULL,
  PRIMARY KEY (`record_id`),
  FOREIGN KEY (`asset_number`) REFERENCES `tool_assets` (`asset_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `username` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
