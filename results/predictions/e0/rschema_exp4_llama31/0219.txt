-- Schema for database: rschema_0219
CREATE TABLE `card` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `artist` varchar(255) DEFAULT NULL,
  `effect` varchar(255) DEFAULT NULL,
  `power` int(11) DEFAULT NULL,
  `intelligence` int(11) DEFAULT NULL,
  `endurance` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `card_collection` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `card_collection_association` (
  `card_id` int(11) NOT NULL,
  `collection_id` int(11) NOT NULL,
  `collection_number` int(11) NOT NULL,
  `card_rarity` varchar(255) NOT NULL,
  PRIMARY KEY (`card_id`, `collection_id`),
  KEY `card_id` (`card_id`),
  KEY `collection_id` (`collection_id`),
  CONSTRAINT `card_collection_association_ibfk_1` FOREIGN KEY (`card_id`) REFERENCES `card` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `card_collection_association_ibfk_2` FOREIGN KEY (`collection_id`) REFERENCES `card_collection` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_exchange_card` (
  `user1_id` int(11) NOT NULL,
  `user2_id` int(11) NOT NULL,
  `exchange_time` datetime NOT NULL,
  `exchanged_card_id` int(11) NOT NULL,
  PRIMARY KEY (`user1_id`, `user2_id`),
  KEY `user1_id` (`user1_id`),
  KEY `user2_id` (`user2_id`),
  KEY `exchanged_card_id` (`exchanged_card_id`),
  CONSTRAINT `user_exchange_card_ibfk_1` FOREIGN KEY (`user1_id`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_exchange_card_ibfk_2` FOREIGN KEY (`user2_id`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_exchange_card_ibfk_3` FOREIGN KEY (`exchanged_card_id`) REFERENCES `card` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
