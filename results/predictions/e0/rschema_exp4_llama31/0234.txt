-- Schema for database: rschema_0234
CREATE TABLE `tags` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `primary_category` varchar(255) NOT NULL,
  `primary_code` int(11) NOT NULL,
  `secondary_category` varchar(255) NOT NULL,
  `secondary_code` int(11) NOT NULL,
  `status` tinyint(1) NOT NULL DEFAULT 1,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `category_code_idx` (`primary_category`, `primary_code`, `secondary_category`, `secondary_code`),
  KEY `status_idx` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
