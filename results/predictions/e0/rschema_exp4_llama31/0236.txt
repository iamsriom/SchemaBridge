-- Schema for database: rschema_0236
CREATE TABLE `camera` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `rtsp_stream_address` varchar(255) NOT NULL,
  `real_time_video_stream_path` varchar(255) NOT NULL,
  `analysis_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `video_recording` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `camera_id` int(11) NOT NULL,
  `video_file_path` varchar(255) NOT NULL,
  `cover_path` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `video_recording_ibfk_1` (`camera_id`),
  CONSTRAINT `video_recording_ibfk_1` FOREIGN KEY (`camera_id`) REFERENCES `camera` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alarm` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `video_id` int(11) NOT NULL,
  `alarm_type` varchar(255) NOT NULL,
  `level` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `alarm_ibfk_1` (`video_id`),
  CONSTRAINT `alarm_ibfk_1` FOREIGN KEY (`video_id`) REFERENCES `video_recording` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `event` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `video_id` int(11) NOT NULL,
  `coordinate_x` int(11) NOT NULL,
  `coordinate_y` int(11) NOT NULL,
  `dimension_x` int(11) NOT NULL,
  `dimension_y` int(11) NOT NULL,
  `event_type` varchar(255) NOT NULL,
  `occurrence_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `event_ibfk_1` (`video_id`),
  CONSTRAINT `event_ibfk_1` FOREIGN KEY (`video_id`) REFERENCES `video_recording` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
