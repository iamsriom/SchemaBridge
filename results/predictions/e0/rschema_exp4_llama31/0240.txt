-- Schema for database: rschema_0240
CREATE TABLE `ship` (
  `ship_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `fleet_id` int(11) NOT NULL,
  `construction_date` datetime NOT NULL,
  `service_status` varchar(255) NOT NULL,
  PRIMARY KEY (`ship_id`),
  KEY `ship_ibfk_1` (`fleet_id`),
  CONSTRAINT `ship_ibfk_1` FOREIGN KEY (`fleet_id`) REFERENCES `fleet` (`fleet_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crew_member` (
  `crew_member_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `fleet_id` int(11) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `length_of_service` datetime NOT NULL,
  PRIMARY KEY (`crew_member_id`),
  KEY `crew_member_ibfk_1` (`fleet_id`),
  CONSTRAINT `crew_member_ibfk_1` FOREIGN KEY (`fleet_id`) REFERENCES `fleet` (`fleet_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fleet` (
  `fleet_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `commander_id` int(11),
  `base` varchar(255) NOT NULL,
  PRIMARY KEY (`fleet_id`),
  KEY `fleet_ibfk_1` (`commander_id`),
  CONSTRAINT `fleet_ibfk_1` FOREIGN KEY (`commander_id`) REFERENCES `crew_member` (`crew_member_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mission_execution` (
  `crew_member_id` int(11) NOT NULL,
  `fleet_id` int(11) NOT NULL,
  `execution_time` datetime NOT NULL,
  `remarks` varchar(255) NOT NULL,
  `is_crew_member_primary_responsible` tinyint(1) NOT NULL,
  PRIMARY KEY (`crew_member_id`, `fleet_id`),
  KEY `mission_execution_ibfk_1` (`crew_member_id`),
  KEY `mission_execution_ibfk_2` (`fleet_id`),
  CONSTRAINT `mission_execution_ibfk_1` FOREIGN KEY (`crew_member_id`) REFERENCES `crew_member` (`crew_member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `mission_execution_ibfk_2` FOREIGN KEY (`fleet_id`) REFERENCES `fleet` (`fleet_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ship_supply` (
  `responsible_supply_crew_member_id` int(11) NOT NULL,
  `ship_id` int(11) NOT NULL,
  `supply_time` datetime NOT NULL,
  `supply_items` varchar(255) NOT NULL,
  `supply_quantity` int(11) NOT NULL,
  PRIMARY KEY (`responsible_supply_crew_member_id`, `ship_id`),
  KEY `ship_supply_ibfk_1` (`responsible_supply_crew_member_id`),
  KEY `ship_supply_ibfk_2` (`ship_id`),
  CONSTRAINT `ship_supply_ibfk_1` FOREIGN KEY (`responsible_supply_crew_member_id`) REFERENCES `crew_member` (`crew_member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ship_supply_ibfk_2` FOREIGN KEY (`ship_id`) REFERENCES `ship` (`ship_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
