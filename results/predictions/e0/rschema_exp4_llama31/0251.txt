-- Schema for database: rschema_0251
CREATE TABLE `customers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone` varchar(20) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `groomers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specialization` varchar(255) NOT NULL,
  `years_of_experience` int(11) NOT NULL,
  `rating` decimal(3,2) NOT NULL DEFAULT 0.00,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grooming_services` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `groomer_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `service_items` varchar(255) NOT NULL,
  `service_duration` int(11) NOT NULL,
  `pricing` decimal(10,2) NOT NULL,
  `customer_rating` decimal(3,2) NOT NULL DEFAULT 0.00,
  PRIMARY KEY (`id`),
  KEY `groomer_id` (`groomer_id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `grooming_services_ibfk_1` FOREIGN KEY (`groomer_id`) REFERENCES `groomers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grooming_services_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pets` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `breed` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `fur_characteristics` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `pets_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
