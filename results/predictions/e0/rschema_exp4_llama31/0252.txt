-- Schema for database: rschema_0252
CREATE TABLE `players` (
  `player_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `game` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `registration_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`player_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `games` (
  `game_id` int(11) NOT NULL AUTO_INCREMENT,
  `game_rules` varchar(255) NOT NULL,
  `game_content` varchar(255) NOT NULL,
  `max_players` int(11) NOT NULL,
  PRIMARY KEY (`game_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `single_player_games` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `player_id` int(11) NOT NULL,
  `game_id` int(11) NOT NULL,
  `game_start_time` datetime NOT NULL,
  `game_end_time` datetime DEFAULT NULL,
  `game_status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `player_id` (`player_id`),
  KEY `game_id` (`game_id`),
  CONSTRAINT `single_player_games_ibfk_1` FOREIGN KEY (`player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `single_player_games_ibfk_2` FOREIGN KEY (`game_id`) REFERENCES `games` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `player_communications` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `first_player_id` int(11) NOT NULL,
  `second_player_id` int(11) NOT NULL,
  `communication_start_time` datetime NOT NULL,
  `communication_end_time` datetime DEFAULT NULL,
  `communication_content` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `first_player_id` (`first_player_id`),
  KEY `second_player_id` (`second_player_id`),
  CONSTRAINT `player_communications_ibfk_1` FOREIGN KEY (`first_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `player_communications_ibfk_2` FOREIGN KEY (`second_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `two_player_battles` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `first_player_id` int(11) NOT NULL,
  `second_player_id` int(11) NOT NULL,
  `game_id` int(11) NOT NULL,
  `battle_start_time` datetime NOT NULL,
  `battle_end_time` datetime DEFAULT NULL,
  `battle_results` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `first_player_id` (`first_player_id`),
  KEY `second_player_id` (`second_player_id`),
  KEY `game_id` (`game_id`),
  CONSTRAINT `two_player_battles_ibfk_1` FOREIGN KEY (`first_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `two_player_battles_ibfk_2` FOREIGN KEY (`second_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `two_player_battles_ibfk_3` FOREIGN KEY (`game_id`) REFERENCES `games` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
