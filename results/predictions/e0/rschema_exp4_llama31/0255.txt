  `id` int(11) NOT NULL AUTO_INCREMENT,

  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_failures` (
  `id` int(11) NOT NULL AUTO_INCREMENT,

  `failure_type` varchar(255) NOT NULL,

  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
