-- Schema for database: rschema_0260
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  `registration_eligibility` tinyint(1) NOT NULL DEFAULT 0
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courses` (
  `course_id` int(11) NOT NULL,
  `course_name` varchar(255) NOT NULL,
  `credits` tinyint(3) NOT NULL,
  `instructor` varchar(255) NOT NULL,
  `class_time` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `degree_examination_results` (
  `result_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  `examination_name` varchar(255) NOT NULL,
  `grade` char(1) NOT NULL,
  `examination_date` datetime NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `registrations` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `registration_time` datetime NOT NULL,
  `status` tinyint(1) NOT NULL DEFAULT 0
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

ALTER TABLE `registrations` ADD CONSTRAINT `registrations_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE `registrations` ADD CONSTRAINT `registrations_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE `degree_examination_results` ADD CONSTRAINT `degree_examination_results_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE;
