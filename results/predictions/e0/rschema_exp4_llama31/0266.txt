-- Schema for database: rschema_0266
CREATE TABLE `pipelines` (
  `pipeline_id` int(11) NOT NULL,
  `start_point` varchar(255) NOT NULL,
  `end_point` varchar(255) NOT NULL,
  `length` decimal(10,2) NOT NULL,
  `diameter` decimal(5,2) NOT NULL,
  PRIMARY KEY (`pipeline_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`personnel_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `pipeline_id` int(11) NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `maintenance_details` text NOT NULL,
  `maintenance_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`),
  KEY `pipeline_id` (`pipeline_id`),
  KEY `personnel_id` (`personnel_id`),
  CONSTRAINT `maintenance_tasks_ibfk_1` FOREIGN KEY (`pipeline_id`) REFERENCES `pipelines` (`pipeline_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_tasks_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `maintenance_personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
