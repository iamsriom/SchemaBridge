-- Schema for database: rschema_0269
CREATE TABLE `dining_tables` (
  `table_number` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `seating_capacity` int(11) NOT NULL,
  PRIMARY KEY (`table_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reservations` (
  `reservation_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `table_number` int(11) NOT NULL,
  `reservation_time` datetime NOT NULL,
  `num_diners` int(11) NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`reservation_id`),
  KEY `reservations_ibfk_1` (`customer_id`),
  KEY `reservations_ibfk_2` (`table_number`),
  CONSTRAINT `reservations_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reservations_ibfk_2` FOREIGN KEY (`table_number`) REFERENCES `dining_tables` (`table_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
