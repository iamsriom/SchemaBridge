-- Schema for database: rschema_0270
CREATE TABLE `programs` (
  `program_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`program_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_staff` (
  `staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  PRIMARY KEY (`staff_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_production` (
  `program_id` int(11) NOT NULL,
  `staff_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  PRIMARY KEY (`program_id`, `staff_id`),
  KEY `program_id` (`program_id`),
  KEY `staff_id` (`staff_id`),
  CONSTRAINT `program_production_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `programs` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_production_ibfk_2` FOREIGN KEY (`staff_id`) REFERENCES `production_staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviewers` (
  `reviewer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`reviewer_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_reviews` (
  `program_id` int(11) NOT NULL,
  `reviewer_id` int(11) NOT NULL,
  `review_time` datetime NOT NULL,
  `review_result` varchar(255) NOT NULL,
  PRIMARY KEY (`program_id`, `reviewer_id`),
  KEY `program_id` (`program_id`),
  KEY `reviewer_id` (`reviewer_id`),
  CONSTRAINT `program_reviews_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `programs` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_reviews_ibfk_2` FOREIGN KEY (`reviewer_id`) REFERENCES `reviewers` (`reviewer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
