CREATE TABLE `ranches` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `livestock_types` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `species` varchar(255) NOT NULL,
  `ranch_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `livestock_types_ibfk_1` (`ranch_id`),
  CONSTRAINT `livestock_types_ibfk_1` FOREIGN KEY (`ranch_id`) REFERENCES `ranches` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `id` int(11) NOT NULL AUTO_INCREMENT,

  `status_id` int(11) NOT NULL,
  `livestock_id` int(11) NOT NULL,

  KEY `health_records_ibfk_1` (`livestock_id`),
  KEY `health_records_ibfk_2` (`status_id`),
  CONSTRAINT `health_records_ibfk_1` FOREIGN KEY (`livestock_id`) REFERENCES `livestock` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `health_records_ibfk_2` FOREIGN KEY (`status_id`) REFERENCES `health_status` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
