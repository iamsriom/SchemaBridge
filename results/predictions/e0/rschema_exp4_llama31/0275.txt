-- Schema for database: rschema_0275
CREATE TABLE `ecosystems` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ecosystem_id` int(11) NOT NULL,
  `monitoring_date` datetime NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `soil_moisture` float NOT NULL,
  `plant_growth` float NOT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `monitoring_records_ibfk_1` (`ecosystem_id`),
  KEY `monitoring_records_ibfk_2` (`operator_id`),
  CONSTRAINT `monitoring_records_ibfk_1` FOREIGN KEY (`ecosystem_id`) REFERENCES `ecosystems` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_records_ibfk_2` FOREIGN KEY (`operator_id`) REFERENCES `operators` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operators` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
