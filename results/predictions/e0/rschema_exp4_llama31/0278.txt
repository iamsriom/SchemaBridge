-- Schema for database: rschema_0278
CREATE TABLE `projects` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `project_manager` varchar(255) NOT NULL,
  `construction_start_date` datetime NOT NULL,
  `expected_completion_date` datetime NOT NULL,
  `actual_completion_date` datetime DEFAULT NULL,
  `project_budget` decimal(10,2) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspection_records` (
  `inspection_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `inspection_date` datetime NOT NULL,
  `inspection_content` text NOT NULL,
  `inspection_results` varchar(255) NOT NULL,
  `inspector` varchar(255) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  KEY `project_id` (`project_id`),
  CONSTRAINT `safety_inspection_records_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tracking_records` (
  `tracking_record_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `tracking_date` datetime NOT NULL,
  `tracking_content` text NOT NULL,
  `tracking_results` varchar(255) NOT NULL,
  PRIMARY KEY (`tracking_record_id`),
  KEY `project_id` (`project_id`),
  CONSTRAINT `tracking_records_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
