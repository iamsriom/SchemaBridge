-- Schema for database: rschema_0288
CREATE TABLE `monitoring_sites` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `responsible_personnel_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `FK_monitoring_sites_responsible_personnel_id` (`responsible_personnel_id`),
  CONSTRAINT `FK_monitoring_sites_responsible_personnel_id` FOREIGN KEY (`responsible_personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `scope` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `scope` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_sessions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `air_quality` varchar(255) NOT NULL,
  `water_quality` varchar(255) NOT NULL,
  `noise_level` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  `monitoring_site_id` int(11) NOT NULL,
  `monitoring_personnel_id` int(11) NOT NULL,
  `result` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `FK_monitoring_sessions_monitoring_site_id` (`monitoring_site_id`),
  KEY `FK_monitoring_sessions_monitoring_personnel_id` (`monitoring_personnel_id`),
  CONSTRAINT `FK_monitoring_sessions_monitoring_site_id` FOREIGN KEY (`monitoring_site_id`) REFERENCES `monitoring_sites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `FK_monitoring_sessions_monitoring_personnel_id` FOREIGN KEY (`monitoring_personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_activities` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `time` datetime NOT NULL,
  `content` varchar(255) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  `monitoring_site_id` int(11) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `FK_maintenance_activities_maintenance_personnel_id` (`maintenance_personnel_id`),
  KEY `FK_maintenance_activities_monitoring_site_id` (`monitoring_site_id`),
  CONSTRAINT `FK_maintenance_activities_maintenance_personnel_id` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `maintenance_personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `FK_maintenance_activities_monitoring_site_id` FOREIGN KEY (`monitoring_site_id`) REFERENCES `monitoring_sites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
