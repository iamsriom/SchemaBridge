-- Schema for database: rschema_0291
CREATE TABLE `reagents` (
  `reagent_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `specifications` varchar(255) DEFAULT NULL,
  `inventory_quantity` int(11) NOT NULL,
  `expiration_date` datetime NOT NULL,
  PRIMARY KEY (`reagent_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `laboratories` (
  `laboratory_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`laboratory_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `usage_records` (
  `usage_id` int(11) NOT NULL AUTO_INCREMENT,
  `reagent_id` int(11) NOT NULL,
  `laboratory_id` int(11) NOT NULL,
  `usage_date` datetime NOT NULL,
  `user` varchar(255) NOT NULL,
  `usage_quantity` int(11) NOT NULL,
  PRIMARY KEY (`usage_id`),
  KEY `reagent_id` (`reagent_id`),
  KEY `laboratory_id` (`laboratory_id`),
  CONSTRAINT `usage_records_ibfk_1` FOREIGN KEY (`reagent_id`) REFERENCES `reagents` (`reagent_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `usage_records_ibfk_2` FOREIGN KEY (`laboratory_id`) REFERENCES `laboratories` (`laboratory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
