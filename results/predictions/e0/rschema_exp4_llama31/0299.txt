-- Schema for database: rschema_0299
CREATE TABLE `cities` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `longitude` decimal(10,7) NOT NULL,
  `latitude` decimal(10,7) NOT NULL,
  `country` varchar(255) NOT NULL,
  `population` int(11) NOT NULL,
  `timezone` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather_forecasts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `city_id` int(11) NOT NULL,
  `date` date NOT NULL,
  `sunrise` time NOT NULL,
  `sunset` time NOT NULL,
  `temperature_morning` decimal(5,2) NOT NULL,
  `temperature_afternoon` decimal(5,2) NOT NULL,
  `temperature_evening` decimal(5,2) NOT NULL,
  `feels_like` decimal(5,2) NOT NULL,
  `atmospheric_pressure` decimal(5,2) NOT NULL,
  `humidity` int(11) NOT NULL,
  `wind_speed` decimal(5,2) NOT NULL,
  `wind_direction` varchar(255) NOT NULL,
  `cloud_cover` int(11) NOT NULL,
  `precipitation_probability` int(11) NOT NULL,
  `weather_type_main` varchar(255) NOT NULL,
  `weather_type_description` varchar(255) NOT NULL,
  `weather_type_icon` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `city_id` (`city_id`),
  CONSTRAINT `weather_forecasts_ibfk_1` FOREIGN KEY (`city_id`) REFERENCES `cities` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
