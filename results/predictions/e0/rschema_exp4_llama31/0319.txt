-- Schema for database: rschema_0319
CREATE TABLE `regions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `regional_code` varchar(255) NOT NULL,
  `city_code` varchar(255) NOT NULL,
  `center_coordinates` varchar(255) DEFAULT NULL,
  `regional_level` int(11) NOT NULL,
  `merged_name` varchar(255) NOT NULL,
  `regional_name` varchar(255) NOT NULL,
  `parent_regional_id` int(11) DEFAULT NULL,
  `pinyin` varchar(255) NOT NULL,
  `abbreviation` varchar(255) NOT NULL,
  `postal_code` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `parent_regional_id` (`parent_regional_id`),
  CONSTRAINT `regions_ibfk_1` FOREIGN KEY (`parent_regional_id`) REFERENCES `regions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
