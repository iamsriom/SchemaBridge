-- Schema for database: rschema_0323
CREATE TABLE `salespeople` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `sales_amount` decimal(10,2) NOT NULL,
  `sales_date` datetime NOT NULL,
  `transaction_quantity` int(11) NOT NULL,
  `visits` int(11) NOT NULL,
  `salesperson_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `salesperson_id` (`salesperson_id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `sales_records_ibfk_1` FOREIGN KEY (`salesperson_id`) REFERENCES `salespeople` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_records_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
