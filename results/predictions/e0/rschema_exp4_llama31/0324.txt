-- Schema for database: rschema_0324
CREATE TABLE `ip_address` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ip_address` varchar(45) NOT NULL,
  `continent` varchar(255) DEFAULT NULL,
  `country` varchar(255) DEFAULT NULL,
  `is_anonymous_proxy` tinyint(1) NOT NULL DEFAULT 0,
  `is_satellite_provider` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `ip_address_unique` (`ip_address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `geographic_location` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `iso_code` char(2) NOT NULL,
  `name` varchar(255) NOT NULL,
  `is_in_eu` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `iso_code_unique` (`iso_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ip_network` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ip_address_id` int(11) NOT NULL,
  `geographic_location_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `ip_address_id` (`ip_address_id`),
  KEY `geographic_location_id` (`geographic_location_id`),
  CONSTRAINT `ip_network_ibfk_1` FOREIGN KEY (`ip_address_id`) REFERENCES `ip_address` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ip_network_ibfk_2` FOREIGN KEY (`geographic_location_id`) REFERENCES `geographic_location` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `import_log` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ip_address_id` int(11) DEFAULT NULL,
  `geographic_location_id` int(11) DEFAULT NULL,
  `import_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `ip_address_id` (`ip_address_id`),
  KEY `geographic_location_id` (`geographic_location_id`),
  CONSTRAINT `import_log_ibfk_1` FOREIGN KEY (`ip_address_id`) REFERENCES `ip_address` (`id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `import_log_ibfk_2` FOREIGN KEY (`geographic_location_id`) REFERENCES `geographic_location` (`id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
