-- Schema for database: rschema_0325
CREATE TABLE `characters` (
  `character_id` int(11) NOT NULL,
  `class_id` int(11) NOT NULL,
  `character_name` varchar(100) NOT NULL,
  `star_level` smallint(6) NOT NULL,
  `rarity_id` smallint(6) DEFAULT NULL,
  PRIMARY KEY (`character_id`),
  KEY `class_id` (`class_id`),
  CONSTRAINT `characters_ibfk_1` FOREIGN KEY (`class_id`) REFERENCES `classes` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classes` (
  `class_id` int(11) NOT NULL,
  `class_name` varchar(100) NOT NULL,
  PRIMARY KEY (`class_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
