CREATE TABLE `object_classes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `objects` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `class_id` int(11) NOT NULL,
  `parent_id` int(11) DEFAULT NULL,
  `owner_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `objects_ibfk_1` (`class_id`),
  KEY `objects_ibfk_2` (`parent_id`),
  KEY `objects_ibfk_3` (`owner_id`),
  CONSTRAINT `objects_ibfk_1` FOREIGN KEY (`class_id`) REFERENCES `object_classes` (`id`),
  CONSTRAINT `objects_ibfk_2` FOREIGN KEY (`parent_id`) REFERENCES `objects` (`id`),
  CONSTRAINT `objects_ibfk_3` FOREIGN KEY (`owner_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `roles` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role_users` (
  `role_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`role_id`, `user_id`),
  KEY `role_users_ibfk_1` (`role_id`),
  KEY `role_users_ibfk_2` (`user_id`),
  CONSTRAINT `role_users_ibfk_1` FOREIGN KEY (`role_id`) REFERENCES `roles` (`id`),
  CONSTRAINT `role_users_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `object_permissions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `object_id` int(11) NOT NULL,
  `role_id` int(11) NOT NULL,
  `record_audit` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  KEY `object_permissions_ibfk_1` (`object_id`),
  KEY `object_permissions_ibfk_2` (`role_id`),
  CONSTRAINT `object_permissions_ibfk_1` FOREIGN KEY (`object_id`) REFERENCES `objects` (`id`),
  CONSTRAINT `object_permissions_ibfk_2` FOREIGN KEY (`role_id`) REFERENCES `roles` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `access_attempts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `object_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `success` tinyint(1) NOT NULL DEFAULT 1,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `access_attempts_ibfk_1` (`object_id`),
  KEY `access_attempts_ibfk_2` (`user_id`),
  CONSTRAINT `access_attempts_ibfk_1` FOREIGN KEY (`object_id`) REFERENCES `objects` (`id`),
  CONSTRAINT `access_attempts_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
