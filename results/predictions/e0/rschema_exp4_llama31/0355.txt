-- Schema for database: rschema_0355
CREATE TABLE `bet` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `amount` decimal(10,2) NOT NULL,
  `race_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `bet_ibfk_1` (`race_id`),
  CONSTRAINT `bet_ibfk_1` FOREIGN KEY (`race_id`) REFERENCES `race` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bet_result` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bet_id` int(11) NOT NULL,
  `horse_number` int(11) NOT NULL,
  `ranking` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `bet_result_ibfk_1` (`bet_id`),
  KEY `bet_result_ibfk_2` (`horse_number`),
  CONSTRAINT `bet_result_ibfk_1` FOREIGN KEY (`bet_id`) REFERENCES `bet` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bet_result_ibfk_2` FOREIGN KEY (`horse_number`) REFERENCES `racehorse` (`number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bettor` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `phone` char(20) NOT NULL,
  `other_info` text,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `horse` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `race` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `racehorse` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `horse_id` int(11) NOT NULL,
  `number` int(11) NOT NULL,
  `race_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `racehorse_ibfk_1` (`horse_id`),
  KEY `racehorse_ibfk_2` (`race_id`),
  CONSTRAINT `racehorse_ibfk_1` FOREIGN KEY (`horse_id`) REFERENCES `horse` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `racehorse_ibfk_2` FOREIGN KEY (`race_id`) REFERENCES `race` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
