-- Schema for database: rschema_0369
CREATE TABLE `content_item` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `created_by` int(11) NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  `updated_by` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `content_ibfk_1` (`created_by`),
  KEY `content_ibfk_2` (`updated_by`),
  CONSTRAINT `content_ibfk_1` FOREIGN KEY (`created_by`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `content_ibfk_2` FOREIGN KEY (`updated_by`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `content_version` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `content_item_id` int(11) NOT NULL,
  `version` int(11) NOT NULL,
  `state` enum('draft','published','archived') NOT NULL DEFAULT 'draft',
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `created_by` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `content_version_ibfk_1` (`content_item_id`),
  KEY `content_version_ibfk_2` (`created_by`),
  CONSTRAINT `content_version_ibfk_1` FOREIGN KEY (`content_item_id`) REFERENCES `content_item` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `content_version_ibfk_2` FOREIGN KEY (`created_by`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operation` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `content_version_id` int(11) NOT NULL,
  `operation_type` enum('create','update','delete') NOT NULL,
  `operation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `operating_user` int(11) NOT NULL,
  `comments` text,
  PRIMARY KEY (`id`),
  KEY `operation_ibfk_1` (`content_version_id`),
  KEY `operation_ibfk_2` (`operating_user`),
  CONSTRAINT `operation_ibfk_1` FOREIGN KEY (`content_version_id`) REFERENCES `content_version` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `operation_ibfk_2` FOREIGN KEY (`operating_user`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `role` enum('admin','moderator','user') NOT NULL DEFAULT 'user',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
