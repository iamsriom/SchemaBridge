-- Schema for database: rschema_0384
CREATE TABLE `tasks` (
  `task_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_title` varchar(255) NOT NULL,
  `creation_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `task_status` varchar(255) NOT NULL,
  `end_date` datetime DEFAULT NULL,
  `status_reason` varchar(255) DEFAULT NULL,
  `query_statement` text,
  `database` varchar(255) NOT NULL,
  `executable_file` varchar(255) NOT NULL,
  `algorithm` varchar(255) NOT NULL,
  `options` text,
  `task_owner` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_parameters` (
  `parameter_id` int(11) NOT NULL AUTO_INCREMENT,
  `task_id` int(11) NOT NULL,
  `parameter_name` varchar(255) NOT NULL,
  `parameter_value` text NOT NULL,
  PRIMARY KEY (`parameter_id`),
  KEY `task_id` (`task_id`),
  CONSTRAINT `task_parameters_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `tasks` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_related_organisms` (
  `organism_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `organism_name` varchar(255) NOT NULL,
  PRIMARY KEY (`organism_id`),
  KEY `task_id` (`task_id`),
  CONSTRAINT `task_related_organisms_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `tasks` (`task_id`),
  CONSTRAINT `task_related_organisms_ibfk_2` FOREIGN KEY (`organism_id`) REFERENCES `organisms` (`organism_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_excluded_organisms` (
  `organism_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `organism_name` varchar(255) NOT NULL,
  PRIMARY KEY (`organism_id`),
  KEY `task_id` (`task_id`),
  CONSTRAINT `task_excluded_organisms_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `tasks` (`task_id`),
  CONSTRAINT `task_excluded_organisms_ibfk_2` FOREIGN KEY (`organism_id`) REFERENCES `organisms` (`organism_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organisms` (
  `organism_id` int(11) NOT NULL AUTO_INCREMENT,
  `organism_name` varchar(255) NOT NULL,
  PRIMARY KEY (`organism_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
