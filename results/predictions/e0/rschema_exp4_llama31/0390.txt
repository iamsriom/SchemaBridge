-- Schema for database: rschema_0390
CREATE TABLE `consumers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `address` text NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courier_companies` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `arrival_time` datetime DEFAULT NULL,
  `charging_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courier_trackings` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `courier_number` varchar(20) NOT NULL,
  `recipient_id` int(11) NOT NULL,
  `recipient_name` varchar(255) NOT NULL,
  `recipient_phone_number` varchar(20) NOT NULL,
  `courier_company_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `courier_company_id` (`courier_company_id`),
  CONSTRAINT `courier_trackings_ibfk_1` FOREIGN KEY (`courier_company_id`) REFERENCES `courier_companies` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `administrators` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `courier_user_name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `partnerships` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `courier_company_id` int(11) NOT NULL,
  `administrator_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `courier_company_id` (`courier_company_id`),
  KEY `administrator_id` (`administrator_id`),
  CONSTRAINT `partnerships_ibfk_1` FOREIGN KEY (`courier_company_id`) REFERENCES `courier_companies` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `partnerships_ibfk_2` FOREIGN KEY (`administrator_id`) REFERENCES `administrators` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
