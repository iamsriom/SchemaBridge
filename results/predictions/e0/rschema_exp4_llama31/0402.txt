-- Schema for database: rschema_0402
CREATE TABLE `doctors` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `title` varchar(255) DEFAULT NULL,
  `date_of_birth` datetime NOT NULL,
  `years_of_employment` int(11) NOT NULL,
  `work_permit` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `personnel` int(11) NOT NULL,
  `department_number` varchar(255) NOT NULL,
  `department_phone_number` varchar(255) NOT NULL,
  `department_address` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patients` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `diagnosis_description` varchar(255) NOT NULL,
  `date_of_consultation` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `diagnoses` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `case_number` varchar(255) NOT NULL,
  `diagnosis_content` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `treatments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `case_number` varchar(255) NOT NULL,
  `treatment_content` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `wards` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ward_number` varchar(255) NOT NULL,
  `number_of_beds` int(11) NOT NULL,
  `ward_level` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `registration_categories` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `category_number` varchar(255) NOT NULL,
  `category_name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drugs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drug_lists` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `drug_list_number` varchar(255) NOT NULL,
  `drug_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `drug_ibfk_1` (`drug_id`),
  CONSTRAINT `drug_ibfk_1` FOREIGN KEY (`drug_id`) REFERENCES `drugs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `diagnosis_treatment` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `case_number` varchar(255) NOT NULL,
  `diagnosis_id` int(11) NOT NULL,
  `treatment_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `diagnosis_ibfk_1` (`diagnosis_id`),
  KEY `treatment_ibfk_1` (`treatment_id`),
  CONSTRAINT `diagnosis_ibfk_1` FOREIGN KEY (`diagnosis_id`) REFERENCES `diagnoses` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `treatment_ibfk_1` FOREIGN KEY (`treatment_id`) REFERENCES `treatments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ward_patient` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ward_id` int(11) NOT NULL,
  `patient_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `ward_ibfk_1` (`ward_id`),
  KEY `patient_ibfk_1` (`patient_id`),
  CONSTRAINT `ward_ibfk_1` FOREIGN KEY (`ward_id`) REFERENCES `wards` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patient_ibfk_1` FOREIGN KEY (`patient_id`) REFERENCES `patients` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor_department` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `doctor_id` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `doctor_ibfk_1` (`doctor_id`),
  KEY `department_ibfk_1` (`department_id`),
  CONSTRAINT `doctor_ibfk_1` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor_patient` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `doctor_id` int(11) NOT NULL,
  `patient_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `doctor_ibfk_2` (`doctor_id`),
  KEY `patient_ibfk_2` (`patient_id`),
  CONSTRAINT `doctor_ibfk_2` FOREIGN KEY (`doctor_id`) REFERENCES `doctors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patient_ibfk_2` FOREIGN KEY (`patient_id`) REFERENCES `patients` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `registration_category_drug` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `registration_category_id` int(11) NOT NULL,
  `drug_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `registration_category_ibfk_1` (`registration_category_id`),
  KEY `drug_ibfk_2` (`drug_id`),
  CONSTRAINT `registration_category_ibfk_1` FOREIGN KEY (`registration_category_id`) REFERENCES `registration_categories` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `drug_ibfk_2` FOREIGN KEY (`drug_id`) REFERENCES `drugs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
