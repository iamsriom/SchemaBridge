-- Schema for database: rschema_0427
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `department` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `enrollment_date` datetime NOT NULL,
  `major` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teachers` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `password` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `course_id` int(11),
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courses` (
  `course_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `prerequisite_course_id` int(11),
  `credits` int(11) NOT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_selections` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `grades` varchar(255),
  PRIMARY KEY (`student_id`, `course_id`),
  KEY `course_id` (`course_id`),
  CONSTRAINT `course_selections_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_selections_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teaching_info` (
  `employee_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `teaching_location` varchar(255) NOT NULL,
  `class_time` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`, `course_id`),
  KEY `course_id` (`course_id`),
  CONSTRAINT `teaching_info_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `teachers` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `teaching_info_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timetables` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `class_time` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`, `course_id`),
  KEY `course_id` (`course_id`),
  CONSTRAINT `timetables_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `timetables_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
