-- Schema for database: rschema_0432
CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `professors` (
  `professor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `office_number` varchar(255) DEFAULT NULL,
  `office_start_time` datetime DEFAULT NULL,
  PRIMARY KEY (`professor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `enrollment_year` int(11) NOT NULL,
  `date_of_birth` datetime NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `courses` (
  `course_id` int(11) NOT NULL,
  `course_name` varchar(255) NOT NULL,
  `semester` varchar(255) NOT NULL,
  `year` int(11) NOT NULL,
  `teaching_professor_id` int(11) NOT NULL,
  `class_day` varchar(255) NOT NULL,
  `course_duration` varchar(255) NOT NULL,
  PRIMARY KEY (`course_id`),
  KEY `courses_ibfk_1` (`teaching_professor_id`),
  CONSTRAINT `courses_ibfk_1` FOREIGN KEY (`teaching_professor_id`) REFERENCES `professors` (`professor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `registrations` (
  `registration_id` int(11) NOT NULL AUTO_INCREMENT,
  `registration_time` datetime NOT NULL,
  `course_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  PRIMARY KEY (`registration_id`),
  KEY `registrations_ibfk_1` (`course_id`),
  KEY `registrations_ibfk_2` (`student_id`),
  CONSTRAINT `registrations_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `registrations_ibfk_2` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
