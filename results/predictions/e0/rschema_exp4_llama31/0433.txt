-- Schema for database: rschema_0433
CREATE TABLE `business_modules` (
  `module_code` char(6) NOT NULL,
  `module_name` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  `creator_user_id` int(11) NOT NULL,
  PRIMARY KEY (`module_code`),
  KEY `module_code` (`module_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `business_entities` (
  `entity_id` int(11) NOT NULL AUTO_INCREMENT,
  `module_code` char(6) NOT NULL,
  `entity_type` varchar(255) NOT NULL,
  `entity_name` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  `creator_user_id` int(11) NOT NULL,
  PRIMARY KEY (`entity_id`),
  KEY `module_code` (`module_code`),
  KEY `entity_id` (`entity_id`),
  CONSTRAINT `business_entities_ibfk_1` FOREIGN KEY (`module_code`) REFERENCES `business_modules` (`module_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attachments` (
  `attachment_id` int(11) NOT NULL AUTO_INCREMENT,
  `entity_id` int(11) NOT NULL,
  `file_name` varchar(255) NOT NULL,
  `file_size` int(11) NOT NULL,
  `file_path` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `file_status` varchar(255) NOT NULL,
  `uploader_user_id` int(11) NOT NULL,
  PRIMARY KEY (`attachment_id`),
  KEY `entity_id` (`entity_id`),
  KEY `attachment_id` (`attachment_id`),
  CONSTRAINT `attachments_ibfk_1` FOREIGN KEY (`entity_id`) REFERENCES `business_entities` (`entity_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
