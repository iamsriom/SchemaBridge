-- Schema for database: rschema_0435
CREATE TABLE `probes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `launch_date` datetime NOT NULL,
  `service_life` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `probe_instruments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `probe_id` int(11) NOT NULL,
  `instrument_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `probe_id` (`probe_id`),
  KEY `instrument_id` (`instrument_id`),
  CONSTRAINT `probe_instruments_ibfk_1` FOREIGN KEY (`probe_id`) REFERENCES `probes` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `probe_instruments_ibfk_2` FOREIGN KEY (`instrument_id`) REFERENCES `instruments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instruments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `power_consumption` decimal(10,2) NOT NULL,
  `accuracy` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `celestial_bodies` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orbital_parameters` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `celestial_body_id` int(11) NOT NULL,
  `semi_major_axis` decimal(10,2) NOT NULL,
  `eccentricity` decimal(10,2) NOT NULL,
  `inclination` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `celestial_body_id` (`celestial_body_id`),
  CONSTRAINT `orbital_parameters_ibfk_1` FOREIGN KEY (`celestial_body_id`) REFERENCES `celestial_bodies` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `physical_characteristics` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `celestial_body_id` int(11) NOT NULL,
  `radius` decimal(10,2) NOT NULL,
  `mass` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `celestial_body_id` (`celestial_body_id`),
  CONSTRAINT `physical_characteristics_ibfk_1` FOREIGN KEY (`celestial_body_id`) REFERENCES `celestial_bodies` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `observations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `celestial_body_id` int(11) NOT NULL,
  `observation_time` datetime NOT NULL,
  `data` text NOT NULL,
  `results` text NOT NULL,
  `accuracy` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `celestial_body_id` (`celestial_body_id`),
  CONSTRAINT `observations_ibfk_1` FOREIGN KEY (`celestial_body_id`) REFERENCES `celestial_bodies` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `missions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `probe_id` int(11) NOT NULL,
  `target_celestial_body_id` int(11) NOT NULL,
  `mission_priority` int(11) NOT NULL,
  `resource_allocation` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `probe_id` (`probe_id`),
  KEY `target_celestial_body_id` (`target_celestial_body_id`),
  CONSTRAINT `missions_ibfk_1` FOREIGN KEY (`probe_id`) REFERENCES `probes` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `missions_ibfk_2` FOREIGN KEY (`target_celestial_body_id`) REFERENCES `celestial_bodies` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `id` int(11) NOT NULL AUTO_INCREMENT,
  `mission_id` int(11) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `mission_id` (`mission_id`),
  CONSTRAINT `flight_trajectories_ibfk_1` FOREIGN KEY (`mission_id`) REFERENCES `missions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `scientific_data` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `mission_id` int(11) NOT NULL,
  `data` text NOT NULL,
  PRIMARY KEY (`id`),
  KEY `mission_id` (`mission_id`),
  CONSTRAINT `scientific_data_ibfk_1` FOREIGN KEY (`mission_id`) REFERENCES `missions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
