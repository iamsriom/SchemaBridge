-- Schema for database: rschema_0442
CREATE TABLE `medical_institutions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_institution_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `departments_ibfk_1` (`medical_institution_id`),
  CONSTRAINT `departments_ibfk_1` FOREIGN KEY (`medical_institution_id`) REFERENCES `medical_institutions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `management_personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_institution_id` int(11) NOT NULL,
  `license_plate_number` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `capacity` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `vehicles_ibfk_1` (`medical_institution_id`),
  CONSTRAINT `vehicles_ibfk_1` FOREIGN KEY (`medical_institution_id`) REFERENCES `medical_institutions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_waste` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `medical_institution_id` int(11) NOT NULL,
  `traceability_code_number` varchar(255) NOT NULL,
  `generation_time` datetime NOT NULL,
  `waste_type` varchar(255) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `medical_waste_ibfk_1` (`department_id`),
  KEY `medical_waste_ibfk_2` (`medical_institution_id`),
  CONSTRAINT `medical_waste_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `medical_waste_ibfk_2` FOREIGN KEY (`medical_institution_id`) REFERENCES `medical_institutions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inbound` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_waste_id` int(11) NOT NULL,
  `batch_number` varchar(255) NOT NULL,
  `expected_quantity` int(11) NOT NULL,
  `actual_quantity` int(11) NOT NULL,
  `expected_weight` decimal(10,2) NOT NULL,
  `actual_weight` decimal(10,2) NOT NULL,
  `inbound_time` datetime NOT NULL,
  `collector_id` int(11) NOT NULL,
  `confirmer_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `inbound_ibfk_1` (`medical_waste_id`),
  CONSTRAINT `inbound_ibfk_1` FOREIGN KEY (`medical_waste_id`) REFERENCES `medical_waste` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `outbound` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medical_waste_id` int(11) NOT NULL,
  `batch_number` varchar(255) NOT NULL,
  `outbound_quantity` int(11) NOT NULL,
  `outbound_weight` decimal(10,2) NOT NULL,
  `outbound_time` datetime NOT NULL,
  `transporter_id` int(11) NOT NULL,
  `confirmer_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `outbound_ibfk_1` (`medical_waste_id`),
  CONSTRAINT `outbound_ibfk_1` FOREIGN KEY (`medical_waste_id`) REFERENCES `medical_waste` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transportation` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `outbound_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `recipient` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `transportation_ibfk_1` (`outbound_id`),
  CONSTRAINT `transportation_ibfk_1` FOREIGN KEY (`outbound_id`) REFERENCES `outbound` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
