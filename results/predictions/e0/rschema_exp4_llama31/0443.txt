-- Schema for database: rschema_0443
CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `mobile_number` varchar(20) NOT NULL,
  `role` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `assets` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_name` varchar(255) NOT NULL,
  `equipment_id` varchar(255) NOT NULL,
  `equipment_model` varchar(255) NOT NULL,
  `equipment_status` varchar(255) NOT NULL,
  `calibration_date` datetime DEFAULT NULL,
  `calibration_validity` datetime DEFAULT NULL,
  `calibration_results` text,
  `equipment_location` varchar(255) NOT NULL,
  `owning_department` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `remarks` text,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`id`) REFERENCES `status_logs` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status_logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `borrower` varchar(255) NOT NULL,
  `borrowing_time` datetime NOT NULL,
  `borrowed_equipment_photo` blob,
  `iou_photo` blob,
  `recipient` varchar(255) NOT NULL,
  `receiving_area` varchar(255) NOT NULL,
  `receiving_time` datetime NOT NULL,
  `received_equipment_photo` blob,
  `returner` varchar(255) NOT NULL,
  `return_time` datetime NOT NULL,
  `returned_equipment_photo` blob,
  `data_monitoring` text,
  `remarks` text,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`equipment_id`) REFERENCES `assets` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account_activity_logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `equipment_id` int(11) NOT NULL,
  `borrowing_account` varchar(255) NOT NULL,
  `receiving_account` varchar(255) NOT NULL,
  `returning_account` varchar(255) NOT NULL,
  `remarks` text,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`equipment_id`) REFERENCES `assets` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
