-- Schema for database: rschema_0449
CREATE TABLE `departments` (
  `department_number` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_function` varchar(255) NOT NULL,
  `department_planning` varchar(255) NOT NULL,
  PRIMARY KEY (`department_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_number` int(11) NOT NULL,
  `employee_name` varchar(255) NOT NULL,
  `employee_gender` varchar(255) NOT NULL,
  `employee_date_of_birth` datetime NOT NULL,
  `department_number` int(11) NOT NULL,
  `employee_entry_date` datetime NOT NULL,
  `employee_resignation_date` datetime DEFAULT NULL,
  `employee_position` varchar(255) NOT NULL,
  `employee_contact_information` varchar(255) NOT NULL,
  `employee_email_address` varchar(255) NOT NULL,
  `employee_status` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_number`),
  KEY `department_number` (`department_number`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`department_number`) REFERENCES `departments` (`department_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `positions` (
  `position_number` int(11) NOT NULL,
  `position_name` varchar(255) NOT NULL,
  `position_responsibilities` varchar(255) NOT NULL,
  `position_requirements` varchar(255) NOT NULL,
  PRIMARY KEY (`position_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trainings` (
  `training_number` int(11) NOT NULL,
  `training_time` datetime NOT NULL,
  `training_location` varchar(255) NOT NULL,
  `training_content` varchar(255) NOT NULL,
  `trainer` varchar(255) NOT NULL,
  PRIMARY KEY (`training_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_position_changes` (
  `employee_number` int(11) NOT NULL,
  `position_number` int(11) NOT NULL,
  `change_date` datetime NOT NULL,
  `reason_for_change` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_number`, `position_number`),
  KEY `employee_number` (`employee_number`),
  KEY `position_number` (`position_number`),
  CONSTRAINT `employee_position_changes_ibfk_1` FOREIGN KEY (`employee_number`) REFERENCES `employees` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_position_changes_ibfk_2` FOREIGN KEY (`position_number`) REFERENCES `positions` (`position_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
