-- Schema for database: rschema_0451
CREATE TABLE `intrusion_detection_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `detection_time` datetime NOT NULL,
  `source_ip` varchar(255) NOT NULL,
  `destination_ip` varchar(255) NOT NULL,
  `source_port` int(11) NOT NULL,
  `destination_port` int(11) NOT NULL,
  `attack_type` varchar(255) NOT NULL,
  `threat_level` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `response_plans` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `generation_time` datetime NOT NULL,
  `plan_type` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `execution_status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `knowledge_base` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `category` varchar(255) NOT NULL,
  `description` text NOT NULL,
  `source` varchar(255) NOT NULL,
  `reliability` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alarms` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `alarm_time` datetime NOT NULL,
  `alarm_level` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `processing_status` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `intrusion_behavior_knowledge` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `intrusion_detection_record_id` int(11) NOT NULL,
  `knowledge_base_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `intrusion_behavior_knowledge_ibfk_1` (`intrusion_detection_record_id`),
  KEY `intrusion_behavior_knowledge_ibfk_2` (`knowledge_base_id`),
  CONSTRAINT `intrusion_behavior_knowledge_ibfk_1` FOREIGN KEY (`intrusion_detection_record_id`) REFERENCES `intrusion_detection_records` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `intrusion_behavior_knowledge_ibfk_2` FOREIGN KEY (`knowledge_base_id`) REFERENCES `knowledge_base` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `response_plan_execution` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `response_plan_id` int(11) NOT NULL,
  `intrusion_detection_record_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `response_plan_execution_ibfk_1` (`response_plan_id`),
  KEY `response_plan_execution_ibfk_2` (`intrusion_detection_record_id`),
  CONSTRAINT `response_plan_execution_ibfk_1` FOREIGN KEY (`response_plan_id`) REFERENCES `response_plans` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `response_plan_execution_ibfk_2` FOREIGN KEY (`intrusion_detection_record_id`) REFERENCES `intrusion_detection_records` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alarm_processing` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `alarm_id` int(11) NOT NULL,
  `response_plan_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `alarm_processing_ibfk_1` (`alarm_id`),
  KEY `alarm_processing_ibfk_2` (`response_plan_id`),
  CONSTRAINT `alarm_processing_ibfk_1` FOREIGN KEY (`alarm_id`) REFERENCES `alarms` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `alarm_processing_ibfk_2` FOREIGN KEY (`response_plan_id`) REFERENCES `response_plans` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
