-- Schema for database: rschema_0468
CREATE TABLE `credit_card_applications` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `application_number` varchar(255) NOT NULL,
  `customer_number` varchar(255) DEFAULT NULL,
  `credit_card_type` varchar(255) NOT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `applicant_id` varchar(255) NOT NULL,
  `application_time` datetime NOT NULL,
  `application_result` varchar(255) NOT NULL DEFAULT 'pending review',
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `credit_card_type` varchar(255) NOT NULL,
  `credit_card_number` varchar(255) NOT NULL,
  `validity_period` varchar(255) NOT NULL,
  `credit_limit` decimal(10,2) NOT NULL,
  `creation_time` datetime NOT NULL,
  `update_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_number` varchar(255) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_amount` decimal(10,2) NOT NULL,
  `transaction_type` varchar(255) NOT NULL,
  `transaction_description` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `credit_card_application_customer` (
  `application_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`application_id`, `customer_id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `credit_card_application_customer_ibfk_1` FOREIGN KEY (`application_id`) REFERENCES `credit_card_applications` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `credit_card_application_customer_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
