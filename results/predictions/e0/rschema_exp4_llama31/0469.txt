-- Schema for database: rschema_0469
CREATE TABLE `owners` (
  `owner_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `workplace` varchar(255) DEFAULT NULL,
  `contact_number` char(11) NOT NULL,
  PRIMARY KEY (`owner_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `housing_units` (
  `house_number` char(6) NOT NULL,
  `floor_area` int(11) NOT NULL,
  `owner_id` int(11) NOT NULL,
  PRIMARY KEY (`house_number`),
  KEY `owner_id` (`owner_id`),
  CONSTRAINT `housing_units_ibfk_1` FOREIGN KEY (`owner_id`) REFERENCES `owners` (`owner_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` char(11) NOT NULL,
  `supervisor_id` int(11) NOT NULL,
  PRIMARY KEY (`department_id`),
  KEY `supervisor_id` (`supervisor_id`),
  CONSTRAINT `departments_ibfk_1` FOREIGN KEY (`supervisor_id`) REFERENCES `employees` (`employee_id`) ON DELETE RESTRICT ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` char(1) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_number` char(11) NOT NULL,
  `position` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`),
  KEY `department_id` (`department_id`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fee_standards` (
  `fee_type` enum('property_management_fee','sanitation_fee','water_fee','electricity_fee') NOT NULL,
  `quantity` int(11) NOT NULL,
  `fee_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`fee_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fee_collections` (
  `collection_id` int(11) NOT NULL AUTO_INCREMENT,
  `fee_type` enum('property_management_fee','sanitation_fee','water_fee','electricity_fee') NOT NULL,
  `quantity` int(11) NOT NULL,
  `fee_amount` decimal(10,2) NOT NULL,
  `housing_unit_id` char(6) NOT NULL,
  PRIMARY KEY (`collection_id`),
  KEY `housing_unit_id` (`housing_unit_id`),
  CONSTRAINT `fee_collections_ibfk_1` FOREIGN KEY (`housing_unit_id`) REFERENCES `housing_units` (`house_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
