-- Schema for database: rschema_0477
CREATE TABLE `billing_methods` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `calculation_function` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transactions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `input_amount` decimal(10,2) NOT NULL,
  `billing_method_id` int(11) NOT NULL,
  `actual_payment_amount` decimal(10,2) NOT NULL,
  `transaction_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `billing_method_id` (`billing_method_id`),
  CONSTRAINT `transactions_ibfk_1` FOREIGN KEY (`billing_method_id`) REFERENCES `billing_methods` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
