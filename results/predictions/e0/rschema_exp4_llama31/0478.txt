-- Schema for database: rschema_0478
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `phone_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `credit` decimal(10,2) NOT NULL DEFAULT 0.00,
  `status` enum('active','inactive') NOT NULL DEFAULT 'active',
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bicycles` (
  `bicycle_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `status` enum('available','unavailable') NOT NULL DEFAULT 'available',
  `unlock_code` varchar(255) NOT NULL,
  PRIMARY KEY (`bicycle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trips` (
  `trip_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `bicycle_id` int(11) NOT NULL,
  `bicycle_number` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `payment_status` enum('paid','unpaid') NOT NULL DEFAULT 'unpaid',
  PRIMARY KEY (`trip_id`),
  KEY `trips_ibfk_1` (`user_id`),
  KEY `trips_ibfk_2` (`bicycle_id`),
  CONSTRAINT `trips_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trips_ibfk_2` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycles` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `faults` (
  `fault_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `bicycle_number` varchar(255) NOT NULL,
  `fault_description` varchar(255) NOT NULL,
  `reporting_time` datetime NOT NULL,
  PRIMARY KEY (`fault_id`),
  KEY `faults_ibfk_1` (`user_id`),
  KEY `faults_ibfk_2` (`bicycle_number`),
  CONSTRAINT `faults_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `billing_rules` (
  `rule_id` int(11) NOT NULL AUTO_INCREMENT,
  `billing_method` enum('hourly','daily','monthly') NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  `min_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`rule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
