-- Schema for database: rschema_0483
CREATE TABLE `platforms` (
  `platform_id` int(11) NOT NULL AUTO_INCREMENT,
  `platform_name` varchar(255) NOT NULL,
  `platform_address` varchar(255) NOT NULL,
  PRIMARY KEY (`platform_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `registration_method` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `intermediaries` (
  `intermediary_id` int(11) NOT NULL AUTO_INCREMENT,
  `intermediary_name` varchar(255) NOT NULL,
  `intermediary_status` varchar(255) NOT NULL,
  `operation_id` int(11) NOT NULL,
  PRIMARY KEY (`intermediary_id`),
  KEY `intermediary_ibfk_1` (`operation_id`),
  CONSTRAINT `intermediary_ibfk_1` FOREIGN KEY (`operation_id`) REFERENCES `operations` (`operation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operations` (
  `operation_id` int(11) NOT NULL AUTO_INCREMENT,
  `operation_type` varchar(255) NOT NULL,
  `operation_amount` decimal(10,2) NOT NULL,
  `operation_time` datetime NOT NULL,
  `platform_id` int(11) NOT NULL,
  PRIMARY KEY (`operation_id`),
  KEY `operation_ibfk_1` (`platform_id`),
  CONSTRAINT `operation_ibfk_1` FOREIGN KEY (`platform_id`) REFERENCES `platforms` (`platform_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_payment_operations` (
  `user_id` int(11) NOT NULL,
  `operation_id` int(11) NOT NULL,
  `payment_status` varchar(255) NOT NULL,
  `payment_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`, `operation_id`),
  KEY `user_payment_operation_ibfk_1` (`user_id`),
  KEY `user_payment_operation_ibfk_2` (`operation_id`),
  CONSTRAINT `user_payment_operation_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_payment_operation_ibfk_2` FOREIGN KEY (`operation_id`) REFERENCES `operations` (`operation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
