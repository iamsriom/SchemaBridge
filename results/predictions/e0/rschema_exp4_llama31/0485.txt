-- Schema for database: rschema_0485
CREATE TABLE `protection_areas` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flora_fauna_records` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `protection_area_id` int(11) NOT NULL,
  `recording_time` datetime NOT NULL,
  `recording_personnel` varchar(255) NOT NULL,
  `species` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `flora_fauna_records_ibfk_1` (`protection_area_id`),
  CONSTRAINT `flora_fauna_records_ibfk_1` FOREIGN KEY (`protection_area_id`) REFERENCES `protection_areas` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `environmental_changes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `protection_area_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `rainfall` float NOT NULL,
  `recording_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `environmental_changes_ibfk_1` (`protection_area_id`),
  CONSTRAINT `environmental_changes_ibfk_1` FOREIGN KEY (`protection_area_id`) REFERENCES `protection_areas` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
