-- Schema for database: rschema_0486
CREATE TABLE `mines` (
  `mine_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mineral_resources` (
  `resource_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `reserves` decimal(10,2) NOT NULL,
  `discovery_date` datetime NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`resource_id`),
  KEY `mine_id` (`mine_id`),
  CONSTRAINT `mineral_resources_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mines` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `extraction_data` (
  `extraction_id` int(11) NOT NULL AUTO_INCREMENT,
  `extraction_date` datetime NOT NULL,
  `extraction_volume` decimal(10,2) NOT NULL,
  `extraction_cost` decimal(10,2) NOT NULL,
  `resource_id` int(11) NOT NULL,
  PRIMARY KEY (`extraction_id`),
  KEY `resource_id` (`resource_id`),
  CONSTRAINT `extraction_data_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `mineral_resources` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
