-- Schema for database: rschema_0487
CREATE TABLE `code_systems` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `code_relationships` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `code_system_id` int(11) NOT NULL,
  `effective_start_date` datetime NOT NULL,
  `effective_end_date` datetime DEFAULT NULL,
  `creator` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  `modifier` varchar(255) DEFAULT NULL,
  `modification_time` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `code_system_id` (`code_system_id`),
  CONSTRAINT `code_relationships_ibfk_1` FOREIGN KEY (`code_system_id`) REFERENCES `code_systems` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `kode_liste` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `code` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `code_relationships_kode_liste` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `code_relationship_id` int(11) NOT NULL,
  `kode_liste_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `code_relationship_id` (`code_relationship_id`),
  KEY `kode_liste_id` (`kode_liste_id`),
  CONSTRAINT `code_relationships_kode_liste_ibfk_1` FOREIGN KEY (`code_relationship_id`) REFERENCES `code_relationships` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `code_relationships_kode_liste_ibfk_2` FOREIGN KEY (`kode_liste_id`) REFERENCES `kode_liste` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
