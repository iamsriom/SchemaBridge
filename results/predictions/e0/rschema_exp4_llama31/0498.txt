-- Schema for database: rschema_0498
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `dormitory_number` int(11) DEFAULT NULL,
  `building_number` int(11) DEFAULT NULL,
  `number_of_people` int(11) DEFAULT NULL,
  `administrator` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classes` (
  `class_id` int(11) NOT NULL,
  `class_number` varchar(255) NOT NULL,
  `classroom_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`class_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sports_teams` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) NOT NULL,
  `activity_time` datetime NOT NULL,
  `coach` varchar(255) NOT NULL,
  `activity_venue` varchar(255) NOT NULL,
  PRIMARY KEY (`team_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `schools` (
  `school_id` int(11) NOT NULL,
  `school_name` varchar(255) NOT NULL,
  PRIMARY KEY (`school_id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `school_sports_teams` (
  `school_id` int(11) NOT NULL,
  `team_id` int(11) NOT NULL,
  PRIMARY KEY (`school_id`, `team_id`),
  KEY `school_id` (`school_id`),
  KEY `team_id` (`team_id`),
  CONSTRAINT `school_ibfk_1` FOREIGN KEY (`school_id`) REFERENCES `schools` (`school_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `team_ibfk_1` FOREIGN KEY (`team_id`) REFERENCES `sports_teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_classes` (
  `student_id` int(11) NOT NULL,
  `class_id` int(11) NOT NULL,
  PRIMARY KEY (`student_id`, `class_id`),
  KEY `student_id` (`student_id`),
  KEY `class_id` (`class_id`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `class_ibfk_1` FOREIGN KEY (`class_id`) REFERENCES `classes` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
