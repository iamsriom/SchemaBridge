-- Schema for database: rschema_0509
CREATE TABLE `patients` (
  `patient_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `hospitalization_time` datetime NOT NULL,
  `ward_id` int(11) NOT NULL,
  `phone` varchar(20) NOT NULL,
  `identity_card_number` varchar(255) NOT NULL,
  `medical_record_number` varchar(255) NOT NULL,
  `emergency_contact_phone` varchar(20) NOT NULL,
  PRIMARY KEY (`patient_id`),
  KEY `patients_ibfk_1` (`ward_id`),
  CONSTRAINT `patients_ibfk_1` FOREIGN KEY (`ward_id`) REFERENCES `wards` (`ward_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department_head_id` int(11) NOT NULL,
  `contact_phone` varchar(20) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`),
  KEY `departments_ibfk_1` (`department_head_id`),
  CONSTRAINT `departments_ibfk_1` FOREIGN KEY (`department_head_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctors` (
  `doctor_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_phone` varchar(20) NOT NULL,
  `title` varchar(255) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`doctor_id`),
  KEY `doctors_ibfk_1` (`department_id`),
  CONSTRAINT `doctors_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `wards` (
  `ward_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `head_id` int(11) NOT NULL,
  `bed_count` int(11) NOT NULL,
  `remaining_bed_count` int(11) NOT NULL,
  PRIMARY KEY (`ward_id`),
  KEY `wards_ibfk_1` (`head_id`),
  CONSTRAINT `wards_ibfk_1` FOREIGN KEY (`head_id`) REFERENCES `doctors` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `nurses` (
  `nurse_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone` varchar(20) NOT NULL,
  `department_id` int(11) NOT NULL,
  `duties` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`nurse_id`),
  KEY `nurses_ibfk_1` (`department_id`),
  CONSTRAINT `nurses_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
