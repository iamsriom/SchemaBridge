-- Schema for database: rschema_0510
CREATE TABLE `parking_spaces` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `space_number` varchar(255) NOT NULL,
  `is_occupied` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `space_number` (`space_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_owners` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `license_plate` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_space_owners` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `parking_space_id` int(11) NOT NULL,
  `owner_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `parking_space_id` (`parking_space_id`),
  KEY `owner_id` (`owner_id`),
  CONSTRAINT `parking_space_owners_ibfk_1` FOREIGN KEY (`parking_space_id`) REFERENCES `parking_spaces` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_space_owners_ibfk_2` FOREIGN KEY (`owner_id`) REFERENCES `parking_owners` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
