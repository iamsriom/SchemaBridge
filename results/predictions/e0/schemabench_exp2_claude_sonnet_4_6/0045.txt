CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `github_id` int(11) NOT NULL,
  `github_login` varchar(255) NOT NULL,
  `github_token` varchar(255) DEFAULT NULL,
  `personal_access_token` varchar(255) DEFAULT NULL,
  `avatar_url` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `last_synced_at` datetime DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `admin` tinyint(1) NOT NULL DEFAULT '0',
  `refresh_interval` int(11) DEFAULT NULL,
  `minimum_refresh_interval` int(11) DEFAULT NULL,
  `sync_notifications` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`id`),
  UNIQUE KEY `index_users_on_github_id` (`github_id`),
  UNIQUE KEY `index_users_on_github_login` (`github_login`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `id` int(11) NOT NULL AUTO_INCREMENT,

  `github_id` int(11) DEFAULT NULL,
  `title` varchar(255) DEFAULT NULL,
  `author` varchar(255) DEFAULT NULL,
  `state` varchar(50) DEFAULT NULL,
  `body` text DEFAULT NULL,
  `comment_count` int(11) DEFAULT NULL,
  `assignees` varchar(255) DEFAULT NULL,
  `labels` text DEFAULT NULL,
  `locked` tinyint(1) DEFAULT NULL,
  `sha` varchar(255) DEFAULT NULL,
  `draft` tinyint(1) DEFAULT NULL,
  `merged` tinyint(1) DEFAULT NULL,
  `ci_status` varchar(50) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `index_subjects_on_url` (`url`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notifications` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `repository_id` int(11) DEFAULT NULL,
  `subject_id` int(11) DEFAULT NULL,
  `github_id` int(11) NOT NULL,
  `reason` varchar(100) DEFAULT NULL,
  `unread` tinyint(1) NOT NULL DEFAULT '1',
  `archived` tinyint(1) NOT NULL DEFAULT '0',
  `starred` tinyint(1) NOT NULL DEFAULT '0',
  `last_read_at` datetime DEFAULT NULL,
  `updated_at` datetime NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `index_notifications_on_github_id_and_user_id` (`github_id`, `user_id`),
  KEY `index_notifications_on_user_id` (`user_id`),
  KEY `index_notifications_on_repository_id` (`repository_id`),
  KEY `index_notifications_on_subject_id` (`subject_id`),
  CONSTRAINT `notifications_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `notifications_ibfk_2` FOREIGN KEY (`repository_id`) REFERENCES `repositories` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `notifications_ibfk_3` FOREIGN KEY (`subject_id`) REFERENCES `subjects` (`id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
