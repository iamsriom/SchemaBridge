CREATE TABLE `participant` (
  `participant_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `date_of_birth` date NOT NULL,
  `age_at_entry` int(11) NOT NULL,
  `gender` varchar(10) NOT NULL DEFAULT 'Male',
  `education_level` varchar(100) DEFAULT NULL,
  `occupation` varchar(100) DEFAULT NULL,
  `enrollment_date` date NOT NULL,
  `study_status` varchar(50) NOT NULL DEFAULT 'Active',
  PRIMARY KEY (`participant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `family_history_diabetes` tinyint(1) NOT NULL DEFAULT 0,
  `family_history_hypertension` tinyint(1) NOT NULL DEFAULT 0,
  `previous_cardiovascular_event` tinyint(1) NOT NULL DEFAULT 0,
  `diabetes_status` varchar(50) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`medical_history_id`),
  CONSTRAINT `medical_history_ibfk_1` FOREIGN KEY (`participant_id`) REFERENCES `participant` (`participant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
