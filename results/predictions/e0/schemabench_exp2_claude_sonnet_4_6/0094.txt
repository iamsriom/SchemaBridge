-- Schema for database: schemabench_0094

CREATE TABLE `molecule` (
  `molecule_id` varchar(20) NOT NULL,
  `mutagenic` varchar(10) DEFAULT NULL,
  `half_life` float DEFAULT NULL,
  `biodegradability_label` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`molecule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `atom` (
  `atom_id` varchar(20) NOT NULL,
  `molecule_id` varchar(20) NOT NULL,
  `element` varchar(10) DEFAULT NULL,
  `atom_type` varchar(50) DEFAULT NULL,
  `charge` float DEFAULT NULL,
  PRIMARY KEY (`atom_id`),
  CONSTRAINT `atom_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bond` (
  `bond_id` varchar(20) NOT NULL,
  `molecule_id` varchar(20) NOT NULL,
  `atom1_id` varchar(20) NOT NULL,
  `atom2_id` varchar(20) NOT NULL,
  `bond_type` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`bond_id`),
  CONSTRAINT `bond_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bond_ibfk_2` FOREIGN KEY (`atom1_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bond_ibfk_3` FOREIGN KEY (`atom2_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `molecular_group` (
  `group_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_id` varchar(20) NOT NULL,
  `group_name` varchar(100) DEFAULT NULL,
  `group_type` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`group_id`),
  CONSTRAINT `molecular_group_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `group_atom` (
  `group_id` int(11) NOT NULL,
  `atom_id` varchar(20) NOT NULL,
  PRIMARY KEY (`group_id`, `atom_id`),
  CONSTRAINT `group_atom_ibfk_1` FOREIGN KEY (`group_id`) REFERENCES `molecular_group` (`group_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `group_atom_ibfk_2` FOREIGN KEY (`atom_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `biodegradability_stat` (
  `stat_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_id` varchar(20) NOT NULL,
  `measured_half_life` float DEFAULT NULL,
  `predicted_half_life` float DEFAULT NULL,
  `regression_score` float DEFAULT NULL,
  `measurement_date` datetime DEFAULT NULL,
  `notes` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`stat_id`),
  CONSTRAINT `biodegradability_stat_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
