-- Schema for database: schemabench_0099

CREATE TABLE `publication` (
  `publication_id` int(11) NOT NULL,
  `class_label` varchar(64) NOT NULL,
  PRIMARY KEY (`publication_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `citation` (
  `citation_id` int(11) NOT NULL AUTO_INCREMENT,
  `citing_publication_id` int(11) NOT NULL,
  `cited_publication_id` int(11) NOT NULL,
  PRIMARY KEY (`citation_id`),
  CONSTRAINT `citation_ibfk_1` FOREIGN KEY (`citing_publication_id`) REFERENCES `publication` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `citation_ibfk_2` FOREIGN KEY (`cited_publication_id`) REFERENCES `publication` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `word` (
  `word_id` int(11) NOT NULL AUTO_INCREMENT,
  `word_token` varchar(128) NOT NULL,
  PRIMARY KEY (`word_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publication_word` (
  `publication_word_id` int(11) NOT NULL AUTO_INCREMENT,
  `publication_id` int(11) NOT NULL,
  `word_id` int(11) NOT NULL,
  `is_present` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`publication_word_id`),
  CONSTRAINT `publication_word_ibfk_1` FOREIGN KEY (`publication_id`) REFERENCES `publication` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `publication_word_ibfk_2` FOREIGN KEY (`word_id`) REFERENCES `word` (`word_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `class` (
  `class_id` int(11) NOT NULL AUTO_INCREMENT,
  `class_name` varchar(64) NOT NULL,
  `class_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`class_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
