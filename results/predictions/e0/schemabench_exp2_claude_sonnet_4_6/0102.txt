-- Schema for database: schemabench_0102

CREATE TABLE `states` (
  `state_id` int(11) NOT NULL AUTO_INCREMENT,
  `state_code` char(2) NOT NULL,
  `state_name` varchar(100) NOT NULL,
  PRIMARY KEY (`state_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `breweries` (
  `brewery_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `city` varchar(100) NOT NULL,
  `state_id` int(11) NOT NULL,
  PRIMARY KEY (`brewery_id`),
  CONSTRAINT `breweries_ibfk_1` FOREIGN KEY (`state_id`) REFERENCES `states` (`state_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beer_styles` (
  `style_id` int(11) NOT NULL AUTO_INCREMENT,
  `style_name` varchar(150) NOT NULL,
  PRIMARY KEY (`style_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beers` (
  `beer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `style_id` int(11) DEFAULT NULL,
  `brewery_id` int(11) NOT NULL,
  `abv` decimal(5,2) DEFAULT NULL,
  `ibu` decimal(6,1) DEFAULT NULL,
  `ounces` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`beer_id`),
  CONSTRAINT `beers_ibfk_1` FOREIGN KEY (`style_id`) REFERENCES `beer_styles` (`style_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `beers_ibfk_2` FOREIGN KEY (`brewery_id`) REFERENCES `breweries` (`brewery_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
