  `product_brand` varchar(255) DEFAULT NULL,
  `product_color` varchar(255) DEFAULT NULL,
  `unit_cost` decimal(10,2) DEFAULT NULL,
  `unit_price` decimal(10,2) DEFAULT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_daily_sales` (
  `retailer_key` int(11) NOT NULL,
  `product_key` int(11) NOT NULL,
  `order_method_key` int(11) NOT NULL,
  `date` datetime NOT NULL,
  `quantity` int(11) DEFAULT NULL,
  `unit_price` decimal(10,2) DEFAULT NULL,
  `unit_sale_price` decimal(10,2) DEFAULT NULL,
  `revenue` decimal(15,2) DEFAULT NULL,
  `product_cost` decimal(15,2) DEFAULT NULL,
  `gross_profit` decimal(15,2) DEFAULT NULL,
  PRIMARY KEY (`retailer_key`, `product_key`, `order_method_key`, `date`),
  CONSTRAINT `go_daily_sales_ibfk_1` FOREIGN KEY (`retailer_key`) REFERENCES `go_retailers` (`retailer_key`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_daily_sales_ibfk_2` FOREIGN KEY (`product_key`) REFERENCES `go_products` (`product_key`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_daily_sales_ibfk_3` FOREIGN KEY (`order_method_key`) REFERENCES `go_methods` (`order_method_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
