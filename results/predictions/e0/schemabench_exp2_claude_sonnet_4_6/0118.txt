-- Schema for database: schemabench_0118

CREATE TABLE `country` (
  `country_id` int(11) NOT NULL AUTO_INCREMENT,
  `country_name` varchar(255) NOT NULL,
  PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country_relation` (
  `relation_id` int(11) NOT NULL AUTO_INCREMENT,
  `country_id_1` int(11) NOT NULL,
  `country_id_2` int(11) NOT NULL,
  `economic_tie` decimal(15,4) DEFAULT NULL,
  `political_tie` decimal(15,4) DEFAULT NULL,
  `military_tie` decimal(15,4) DEFAULT NULL,
  PRIMARY KEY (`relation_id`),
  CONSTRAINT `country_relation_ibfk_1` FOREIGN KEY (`country_id_1`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `country_relation_ibfk_2` FOREIGN KEY (`country_id_2`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country_statistics` (
  `stat_id` int(11) NOT NULL AUTO_INCREMENT,
  `country_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `population` bigint(20) DEFAULT NULL,
  `gnp` decimal(20,4) DEFAULT NULL,
  `exports` decimal(20,4) DEFAULT NULL,
  `imports` decimal(20,4) DEFAULT NULL,
  `literacy_rate` decimal(5,2) DEFAULT NULL,
  `unemployment_rate` decimal(5,2) DEFAULT NULL,
  `gdp_per_capita` decimal(15,4) DEFAULT NULL,
  `inflation_rate` decimal(5,2) DEFAULT NULL,
  `government_type` varchar(255) DEFAULT NULL,
  `political_stability_index` decimal(10,4) DEFAULT NULL,
  PRIMARY KEY (`stat_id`),
  CONSTRAINT `country_statistics_ibfk_1` FOREIGN KEY (`country_id`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
