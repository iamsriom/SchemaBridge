-- Schema for database: schemabench_0120

CREATE TABLE `teams` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) NOT NULL,
  `team_abbrev` varchar(10) DEFAULT NULL,
  `region` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `seasons` (
  `season_id` int(11) NOT NULL AUTO_INCREMENT,
  `season_year` int(11) NOT NULL,
  `region_w` varchar(50) DEFAULT NULL,
  `region_x` varchar(50) DEFAULT NULL,
  `region_y` varchar(50) DEFAULT NULL,
  `region_z` varchar(50) DEFAULT NULL,
  `dayzero` datetime DEFAULT NULL,
  PRIMARY KEY (`season_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `seeds` (
  `seed_id` int(11) NOT NULL AUTO_INCREMENT,
  `season_id` int(11) NOT NULL,
  `team_id` int(11) NOT NULL,
  `seed` varchar(10) NOT NULL,
  `region` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`seed_id`),
  CONSTRAINT `seeds_ibfk_1` FOREIGN KEY (`season_id`) REFERENCES `seasons` (`season_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `seeds_ibfk_2` FOREIGN KEY (`team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `slots` (
  `slot_id` int(11) NOT NULL AUTO_INCREMENT,
  `season_id` int(11) NOT NULL,
  `slot` varchar(20) NOT NULL,
  `strong_seed` varchar(10) DEFAULT NULL,
  `weak_seed` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`slot_id`),
  CONSTRAINT `slots_ibfk_1` FOREIGN KEY (`season_id`) REFERENCES `seasons` (`season_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regular_season_results` (
  `result_id` int(11) NOT NULL AUTO_INCREMENT,
  `season_id` int(11) NOT NULL,
  `day_num` int(11) NOT NULL,
  `winning_team_id` int(11) NOT NULL,
  `losing_team_id` int(11) NOT NULL,
  `winning_score` int(11) NOT NULL,
  `losing_score` int(11) NOT NULL,
  `location` char(1) DEFAULT NULL,
  `num_ot` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`result_id`),
  CONSTRAINT `regular_season_results_ibfk_1` FOREIGN KEY (`season_id`) REFERENCES `seasons` (`season_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `regular_season_results_ibfk_2` FOREIGN KEY (`winning_team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `regular_season_results_ibfk_3` FOREIGN KEY (`losing_team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tournament_results` (
  `result_id` int(11) NOT NULL AUTO_INCREMENT,
  `season_id` int(11) NOT NULL,
  `day_num` int(11) NOT NULL,
  `winning_team_id` int(11) NOT NULL,
  `losing_team_id` int(11) NOT NULL,
  `winning_score` int(11) NOT NULL,
  `losing_score` int(11) NOT NULL,
  `location` char(1) DEFAULT NULL,
  `num_ot` int(11) NOT NULL DEFAULT 0,
  `slot` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`result_id`),
  CONSTRAINT `tournament_results_ibfk_1` FOREIGN KEY (`season_id`) REFERENCES `seasons` (`season_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `tournament_results_ibfk_2` FOREIGN KEY (`winning_team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `tournament_results_ibfk_3` FOREIGN KEY (`losing_team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sample_submissions` (
  `submission_id` int(11) NOT NULL AUTO_INCREMENT,
  `match_id` varchar(50) NOT NULL,
  `season_id` int(11) NOT NULL,
  `team1_id` int(11) NOT NULL,
  `team2_id` int(11) NOT NULL,
  `predicted_probability` float NOT NULL DEFAULT 0.5,
  PRIMARY KEY (`submission_id`),

  CONSTRAINT `sample_submissions_ibfk_3` FOREIGN KEY (`team2_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `prediction_models` (
  `model_id` int(11) NOT NULL AUTO_INCREMENT,
  `model_name` varchar(255) NOT NULL,
  `model_description` varchar(1000) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`model_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `prediction_results` (
  `prediction_id` int(11) NOT NULL AUTO_INCREMENT,

  CONSTRAINT `prediction_results_ibfk_4` FOREIGN KEY (`team2_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
