-- Schema for database: schemabench_0135

CREATE TABLE `trains` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `direction` varchar(10) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cars` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `train_id` int(11) NOT NULL,
  `position` int(11) NOT NULL,
  `shape` varchar(50) DEFAULT NULL,
  `length` int(11) DEFAULT NULL,
  `sides` varchar(50) DEFAULT NULL,
  `roof` varchar(50) DEFAULT NULL,
  `wheels` int(11) DEFAULT NULL,
  `load_shape` varchar(50) DEFAULT NULL,
  `load_num` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `cars_ibfk_1` FOREIGN KEY (`train_id`) REFERENCES `trains` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
