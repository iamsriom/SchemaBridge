-- Schema for database: schemabench_0143

CREATE TABLE `country` (
  `code` char(3) NOT NULL,
  `name` varchar(255) NOT NULL,
  `continent` varchar(100) NOT NULL,
  `region` varchar(100) NOT NULL,
  `surface_area` decimal(10,2) NOT NULL DEFAULT '0.00',
  `indep_year` smallint(6) DEFAULT NULL,
  `population` int(11) NOT NULL DEFAULT '0',
  `life_expectancy` decimal(4,1) DEFAULT NULL,
  `gnp` decimal(10,2) DEFAULT NULL,
  `gnp_old` decimal(10,2) DEFAULT NULL,
  `local_name` varchar(255) NOT NULL,
  `government_form` varchar(100) NOT NULL,
  `head_of_state` varchar(100) DEFAULT NULL,
  `capital` int(11) DEFAULT NULL,
  `code2` char(2) NOT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `city` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `country_code` char(3) NOT NULL,
  `district` varchar(100) NOT NULL,
  `population` int(11) NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`),
  CONSTRAINT `city_ibfk_1` FOREIGN KEY (`country_code`) REFERENCES `country` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country_language` (
  `country_code` char(3) NOT NULL,
  `language` varchar(100) NOT NULL,
  `is_official` char(1) NOT NULL DEFAULT 'F',
  `percentage` decimal(4,1) NOT NULL DEFAULT '0.0',
  PRIMARY KEY (`country_code`, `language`),
  CONSTRAINT `country_language_ibfk_1` FOREIGN KEY (`country_code`) REFERENCES `country` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
