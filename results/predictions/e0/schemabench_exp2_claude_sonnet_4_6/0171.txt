-- Schema for database: schemabench_0171

CREATE TABLE `portfolio` (
  `portfolio_id` int(11) NOT NULL AUTO_INCREMENT,
  `portfolio_name` varchar(255) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`portfolio_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `currency` (
  `currency_code` char(3) NOT NULL,
  `currency_name` varchar(100) NOT NULL,
  `symbol` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`currency_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instrument` (
  `instrument_id` int(11) NOT NULL AUTO_INCREMENT,
  `instrument_name` varchar(255) NOT NULL,
  `instrument_type` varchar(100) NOT NULL,
  `maturity_date` datetime DEFAULT NULL,
  `settlement_type` varchar(50) DEFAULT NULL,
  `currency_code` char(3) NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`instrument_id`),
  CONSTRAINT `instrument_ibfk_1` FOREIGN KEY (`currency_code`) REFERENCES `currency` (`currency_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trader` (
  `trader_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone` varchar(50) DEFAULT NULL,
  `desk` varchar(100) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`trader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `counterparty` (
  `counterparty_id` int(11) NOT NULL AUTO_INCREMENT,
  `counterparty_name` varchar(255) NOT NULL,
  `contact_name` varchar(255) DEFAULT NULL,
  `contact_email` varchar(255) DEFAULT NULL,
  `contact_phone` varchar(50) DEFAULT NULL,
  `exposure_limit` decimal(18,2) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`counterparty_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trade_status` (
  `status_id` int(11) NOT NULL AUTO_INCREMENT,
  `status_name` varchar(100) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`status_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trade` (
  `trade_id` int(11) NOT NULL AUTO_INCREMENT,
  `portfolio_id` int(11) NOT NULL,
  `trader_id` int(11) NOT NULL,
  `instrument_id` int(11) NOT NULL,
  `counterparty_id` int(11) NOT NULL,
  `status_id` int(11) NOT NULL,
  `trade_date` datetime NOT NULL,
  `settlement_date` datetime DEFAULT NULL,
  `quantity` decimal(18,6) NOT NULL,
  `price` decimal(18,6) NOT NULL,
  `total_value` decimal(18,2) NOT NULL,
  `trade_type` varchar(50) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`trade_id`),
  CONSTRAINT `trade_ibfk_1` FOREIGN KEY (`portfolio_id`) REFERENCES `portfolio` (`portfolio_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_ibfk_2` FOREIGN KEY (`trader_id`) REFERENCES `trader` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_ibfk_3` FOREIGN KEY (`instrument_id`) REFERENCES `instrument` (`instrument_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_ibfk_4` FOREIGN KEY (`counterparty_id`) REFERENCES `counterparty` (`counterparty_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_ibfk_5` FOREIGN KEY (`status_id`) REFERENCES `trade_status` (`status_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trade_audit_log` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `trade_id` int(11) NOT NULL,
  `action` varchar(100) NOT NULL,
  `action_description` varchar(500) DEFAULT NULL,
  `performed_by` int(11) DEFAULT NULL,
  `performed_at` datetime NOT NULL,
  PRIMARY KEY (`log_id`),
  CONSTRAINT `trade_audit_log_ibfk_1` FOREIGN KEY (`trade_id`) REFERENCES `trade` (`trade_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_audit_log_ibfk_2` FOREIGN KEY (`performed_by`) REFERENCES `trader` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trade_status_history` (
  `history_id` int(11) NOT NULL AUTO_INCREMENT,
  `trade_id` int(11) NOT NULL,
  `status_id` int(11) NOT NULL,
  `changed_at` datetime NOT NULL,
  `changed_by` int(11) DEFAULT NULL,
  `notes` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`history_id`),
  CONSTRAINT `trade_status_history_ibfk_1` FOREIGN KEY (`trade_id`) REFERENCES `trade` (`trade_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_status_history_ibfk_2` FOREIGN KEY (`status_id`) REFERENCES `trade_status` (`status_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trade_status_history_ibfk_3` FOREIGN KEY (`changed_by`) REFERENCES `trader` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
