CREATE TABLE `circuit` (
  `circuit_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  `length_km` decimal(6,3) DEFAULT NULL,
  `configuration` varchar(255) DEFAULT NULL,
  `configuration_year` year(4) DEFAULT NULL,
  PRIMARY KEY (`circuit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `race` (
  `race_id` int(11) NOT NULL AUTO_INCREMENT,
  `year` year(4) NOT NULL,
  `date` date DEFAULT NULL,
  `start_time` time DEFAULT NULL,
  `circuit_id` int(11) NOT NULL,
  `is_cancelled` tinyint(1) NOT NULL DEFAULT 0,
  `cancellation_reason` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`race_id`),
  CONSTRAINT `race_ibfk_1` FOREIGN KEY (`circuit_id`) REFERENCES `circuit` (`circuit_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country` (
  `country_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `code` char(3) DEFAULT NULL,
  PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `driver` (
  `driver_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `nationality` varchar(100) DEFAULT NULL,
  `country_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`driver_id`),
  CONSTRAINT `driver_ibfk_1` FOREIGN KEY (`country_id`) REFERENCES `country` (`country_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `team` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,

  PRIMARY KEY (`team_id`),
  CONSTRAINT `team_ibfk_1` FOREIGN KEY (`country_id`) REFERENCES `country` (`country_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tyre` (
  `tyre_id` int(11) NOT NULL AUTO_INCREMENT,

  `engine` varchar(100) DEFAULT NULL,
  `tyre_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`car_id`),
  CONSTRAINT `car_ibfk_1` FOREIGN KEY (`tyre_id`) REFERENCES `tyre` (`tyre_id`) ON DELETE SET NULL ON UPDATE CASCADE
