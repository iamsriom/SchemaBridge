-- Schema for database: schemabench_0195

CREATE TABLE `runners` (
  `runner_id` int(11) NOT NULL,
  `registration_date` datetime NOT NULL,
  PRIMARY KEY (`runner_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_orders` (
  `order_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `pizza_id` int(11) NOT NULL,
  `exclusions` varchar(255) DEFAULT NULL,
  `extras` varchar(255) DEFAULT NULL,
  `order_time` datetime NOT NULL,
  PRIMARY KEY (`order_id`, `pizza_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `runner_orders` (
  `order_id` int(11) NOT NULL,
  `runner_id` int(11) NOT NULL,
  `pickup_time` datetime DEFAULT NULL,
  `distance` varchar(50) DEFAULT NULL,
  `duration` varchar(50) DEFAULT NULL,
  `cancellation` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `runner_orders_ibfk_1` FOREIGN KEY (`runner_id`) REFERENCES `runners` (`runner_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pizza_names` (
  `pizza_id` int(11) NOT NULL,
  `pizza_name` varchar(255) NOT NULL,
  PRIMARY KEY (`pizza_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pizza_recipes` (
  `pizza_id` int(11) NOT NULL,
  `toppings` varchar(255) NOT NULL,
  PRIMARY KEY (`pizza_id`),
  CONSTRAINT `pizza_recipes_ibfk_1` FOREIGN KEY (`pizza_id`) REFERENCES `pizza_names` (`pizza_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pizza_toppings` (
  `topping_id` int(11) NOT NULL,
  `topping_name` varchar(255) NOT NULL,
  PRIMARY KEY (`topping_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
