  PRIMARY KEY (`id`),
  CONSTRAINT `product_hierarchy_ibfk_1` FOREIGN KEY (`parent_id`) REFERENCES `product_hierarchy` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product_details` (
  `product_id` char(6) NOT NULL,
  `price` int(11) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `category_id` int(11) NOT NULL,
  `segment_id` int(11) NOT NULL,
  `style_id` int(11) NOT NULL,
  `category_name` varchar(100) NOT NULL,
  `segment_name` varchar(100) NOT NULL,
  `style_name` varchar(100) NOT NULL,
  PRIMARY KEY (`product_id`),
  CONSTRAINT `product_details_ibfk_1` FOREIGN KEY (`category_id`) REFERENCES `product_hierarchy` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `product_details_ibfk_2` FOREIGN KEY (`segment_id`) REFERENCES `product_hierarchy` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `product_details_ibfk_3` FOREIGN KEY (`style_id`) REFERENCES `product_hierarchy` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `prod_id` char(6) NOT NULL,
  `qty` int(11) NOT NULL,
  `price` int(11) NOT NULL,
  `discount` int(11) NOT NULL,
  `member` char(1) NOT NULL,
  `txn_id` char(6) NOT NULL,
  `start_txn_time` datetime NOT NULL,
  PRIMARY KEY (`prod_id`, `txn_id`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`prod_id`) REFERENCES `product_details` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
