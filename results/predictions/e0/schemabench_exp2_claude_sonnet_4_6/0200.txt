-- Schema for database: schemabench_0200

CREATE TABLE `interest_map` (
  `id` int(11) NOT NULL,
  `interest_name` varchar(255) NOT NULL,
  `interest_summary` varchar(1000) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `last_modified` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `interest_metrics` (
  `_month` int(11) DEFAULT NULL,
  `_year` int(11) DEFAULT NULL,
  `month_year` varchar(10) DEFAULT NULL,
  `interest_id` int(11) DEFAULT NULL,
  `composition` decimal(10,2) DEFAULT NULL,
  `index_value` decimal(10,2) DEFAULT NULL,
  `ranking` int(11) DEFAULT NULL,
  `percentile_ranking` decimal(10,2) DEFAULT NULL,
  CONSTRAINT `interest_metrics_ibfk_1` FOREIGN KEY (`interest_id`) REFERENCES `interest_map` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
