-- Schema for database: schemabench_0211

CREATE TABLE `companies` (
  `company_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `industry` varchar(255) DEFAULT NULL,
  `headquarters` varchar(255) DEFAULT NULL,
  `company_size` varchar(100) DEFAULT NULL,
  `website` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `job_postings` (
  `job_id` int(11) NOT NULL AUTO_INCREMENT,
  `company_id` int(11) NOT NULL,
  `job_title` varchar(255) NOT NULL,
  `job_description` text DEFAULT NULL,
  `salary_year_avg` decimal(10,2) DEFAULT NULL,
  `salary_hour_avg` decimal(10,2) DEFAULT NULL,
  `job_location` varchar(255) DEFAULT NULL,
  `job_country` varchar(100) DEFAULT NULL,
  `job_state` varchar(100) DEFAULT NULL,
  `job_city` varchar(100) DEFAULT NULL,
  `job_schedule_type` varchar(100) DEFAULT NULL,
  `job_work_from_home` tinyint(1) DEFAULT 0,
  `job_posted_date` datetime DEFAULT NULL,
  `job_no_degree_mention` tinyint(1) DEFAULT 0,
  `job_health_insurance` tinyint(1) DEFAULT 0,
  `search_location` varchar(255) DEFAULT NULL,
  `job_via` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`job_id`),
  CONSTRAINT `job_postings_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `companies` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `skills` (
  `skill_id` int(11) NOT NULL AUTO_INCREMENT,
  `skill_name` varchar(255) NOT NULL,
  `skill_category` varchar(100) DEFAULT NULL,
  `skill_type` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`skill_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `job_skills` (
  `job_id` int(11) NOT NULL,
  `skill_id` int(11) NOT NULL,
  PRIMARY KEY (`job_id`, `skill_id`),
  CONSTRAINT `job_skills_ibfk_1` FOREIGN KEY (`job_id`) REFERENCES `job_postings` (`job_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `job_skills_ibfk_2` FOREIGN KEY (`skill_id`) REFERENCES `skills` (`skill_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
