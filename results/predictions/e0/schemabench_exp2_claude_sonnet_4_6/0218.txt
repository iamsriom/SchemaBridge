-- Schema for database: schemabench_0218

CREATE TABLE `grampanchayat` (
  `gp_id` int(11) NOT NULL AUTO_INCREMENT,
  `gp_name` varchar(255) NOT NULL,
  `taluka_name` varchar(255) NOT NULL,
  `total_wards` int(11) NOT NULL DEFAULT 0,
  `total_villages` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`gp_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `village` (
  `village_id` int(11) NOT NULL AUTO_INCREMENT,
  `village_name` varchar(255) NOT NULL,
  `gp_id` int(11) NOT NULL,
  PRIMARY KEY (`village_id`),
  CONSTRAINT `village_ibfk_1` FOREIGN KEY (`gp_id`) REFERENCES `grampanchayat` (`gp_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ward` (
  `ward_id` int(11) NOT NULL AUTO_INCREMENT,
  `ward_no` int(11) NOT NULL,
  `village_id` int(11) NOT NULL,
  `gp_id` int(11) NOT NULL,
  PRIMARY KEY (`ward_id`),
  CONSTRAINT `ward_ibfk_1` FOREIGN KEY (`village_id`) REFERENCES `village` (`village_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ward_ibfk_2` FOREIGN KEY (`gp_id`) REFERENCES `grampanchayat` (`gp_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ward_member` (
  `member_id` int(11) NOT NULL AUTO_INCREMENT,
  `member_name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_no` varchar(15) DEFAULT NULL,
  `ward_id` int(11) NOT NULL,
  PRIMARY KEY (`member_id`),
  CONSTRAINT `ward_member_ibfk_1` FOREIGN KEY (`ward_id`) REFERENCES `ward` (`ward_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `farmer` (
  `farmer_id` int(11) NOT NULL AUTO_INCREMENT,
  `aadh_no` char(12) NOT NULL,
  `farmer_name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_no` varchar(15) DEFAULT NULL,
  `village_id` int(11) NOT NULL,
  `ward_id` int(11) NOT NULL,
  PRIMARY KEY (`farmer_id`),
  UNIQUE KEY `aadh_no` (`aadh_no`),
  CONSTRAINT `farmer_ibfk_1` FOREIGN KEY (`village_id`) REFERENCES `village` (`village_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `farmer_ibfk_2` FOREIGN KEY (`ward_id`) REFERENCES `ward` (`ward_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `machine` (
  `mach_id` int(11) NOT NULL AUTO_INCREMENT,
  `mach_name` varchar(255) NOT NULL,
  `mach_type` varchar(255) NOT NULL,
  `rent_per_hour` decimal(10,2) NOT NULL,
  `availability_status` varchar(50) NOT NULL DEFAULT 'available',
  `gp_id` int(11) NOT NULL,
  PRIMARY KEY (`mach_id`),
  CONSTRAINT `machine_ibfk_1` FOREIGN KEY (`gp_id`) REFERENCES `grampanchayat` (`gp_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental` (
  `rental_id` int(11) NOT NULL AUTO_INCREMENT,
  `farmer_id` int(11) NOT NULL,
  `mach_id` int(11) NOT NULL,
  `ward_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  `rental_start_date` datetime NOT NULL,
  `rental_end_date` datetime DEFAULT NULL,
  `duration_hours` decimal(10,2) DEFAULT NULL,
  `rental_status` varchar(50) NOT NULL DEFAULT 'active',
  PRIMARY KEY (`rental_id`),
  CONSTRAINT `rental_ibfk_1` FOREIGN KEY (`farmer_id`) REFERENCES `farmer` (`farmer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_ibfk_2` FOREIGN KEY (`mach_id`) REFERENCES `machine` (`mach_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_ibfk_3` FOREIGN KEY (`ward_id`) REFERENCES `ward` (`ward_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_ibfk_4` FOREIGN KEY (`member_id`) REFERENCES `ward_member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bill` (
  `bill_id` int(11) NOT NULL AUTO_INCREMENT,
  `rental_id` int(11) NOT NULL,
  `farmer_id` int(11) NOT NULL,
  `bill_date` datetime NOT NULL,
  `total_amount` decimal(10,2) NOT NULL,
  `payment_method` varchar(100) DEFAULT NULL,
  `payment_date` datetime DEFAULT NULL,
  `payment_status` varchar(50) NOT NULL DEFAULT 'pending',
  PRIMARY KEY (`bill_id`),
  CONSTRAINT `bill_ibfk_1` FOREIGN KEY (`rental_id`) REFERENCES `rental` (`rental_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bill_ibfk_2` FOREIGN KEY (`farmer_id`) REFERENCES `farmer` (`farmer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
