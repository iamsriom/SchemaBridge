  `name` varchar(255) NOT NULL,

  `capacity` int(11) DEFAULT NULL,
  `condition` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`stadium_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `team` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,

  `captain_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `player_id` int(11) NOT NULL AUTO_INCREMENT,

  `team_id` int(11) DEFAULT NULL,
  `team_number` int(11) DEFAULT NULL,
  `injury_status` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`player_id`),
  CONSTRAINT `player_ibfk_1` FOREIGN KEY (`team_id`) REFERENCES `team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
