-- Schema for database: schemabench_0024
CREATE TABLE `taggit_tag` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `slug` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `taggit_tag_name_unique` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `taggit_taggeditem` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `tag_id` int(11) NOT NULL,
  `object_id` int(11) NOT NULL,
  `content_type_id` int(11) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `taggit_taggeditem_tag_id_object_id_content_type_id_unique` (`tag_id`, `object_id`, `content_type_id`),
  CONSTRAINT `taggit_taggeditem_tag_id_foreign` FOREIGN KEY (`tag_id`) REFERENCES `taggit_tag` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `taggit_taggeditem_object_id_foreign` FOREIGN KEY (`object_id`) REFERENCES `taggit_taggeditem` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
