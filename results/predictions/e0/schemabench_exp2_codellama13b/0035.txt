-- Schema for database: schemabench_0035

CREATE TABLE easy_thumbnails_source (
  id INT(11) NOT NULL AUTO_INCREMENT,
  image_path VARCHAR(255) NOT NULL,
  modified_date DATETIME NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE easy_thumbnails_thumbnail (
  id INT(11) NOT NULL AUTO_INCREMENT,
  source_id INT(11) NOT NULL,
  thumbnail_path VARCHAR(255) NOT NULL,
  modified_date DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (source_id) REFERENCES easy_thumbnails_source (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE easy_thumbnails_thumbnaildimensions (
  id INT(11) NOT NULL AUTO_INCREMENT,
  thumbnail_id INT(11) NOT NULL,
  width INT(11) NOT NULL,
  height INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (thumbnail_id) REFERENCES easy_thumbnails_thumbnail (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
