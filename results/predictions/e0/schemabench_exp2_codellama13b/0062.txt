CREATE TABLE `reaction_types` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reactable_models` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reactions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `reaction_type_id` int(11) NOT NULL,
  `reactable_model_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `rate` float NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`reaction_type_id`) REFERENCES `reaction_types` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`reactable_model_id`) REFERENCES `reactable_models` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reaction_statistics` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `reaction_type_id` int(11) NOT NULL,
  `reactable_model_id` int(11) NOT NULL,
  `total_count` int(11) NOT NULL,
  `total_weight` float NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`reaction_type_id`) REFERENCES `reaction_types` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`reactable_model_id`) REFERENCES `reactable_models` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
