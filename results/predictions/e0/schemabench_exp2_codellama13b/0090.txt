CREATE TABLE airlines (
  airline_id INT(11) NOT NULL AUTO_INCREMENT,
  airline_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (airline_id)
);

CREATE TABLE airports (
  airport_id INT(11) NOT NULL AUTO_INCREMENT,
  airport_code VARCHAR(255) NOT NULL,
  airport_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (airport_id)
);

CREATE TABLE flights (
  flight_id INT(11) NOT NULL AUTO_INCREMENT,
  flight_number VARCHAR(255) NOT NULL,
  departure_airport_id INT(11) NOT NULL,
  arrival_airport_id INT(11) NOT NULL,
  departure_time DATETIME NOT NULL,
  arrival_time DATETIME NOT NULL,
  airline_id INT(11) NOT NULL,
  PRIMARY KEY (flight_id),
  FOREIGN KEY (departure_airport_id) REFERENCES airports (airport_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (arrival_airport_id) REFERENCES airports (airport_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (airline_id) REFERENCES airlines (airline_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE flight_delays (
  flight_delay_id INT(11) NOT NULL AUTO_INCREMENT,
  flight_id INT(11) NOT NULL,
  delay_reason VARCHAR(255) NOT NULL,
  delay_duration INT(11) NOT NULL,
  PRIMARY KEY (flight_delay_id),
  FOREIGN KEY (flight_id) REFERENCES flights (flight_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE flight_cancellations (
  flight_cancellation_id INT(11) NOT NULL AUTO_INCREMENT,
  flight_id INT(11) NOT NULL,
  cancellation_reason VARCHAR(255) NOT NULL,
  PRIMARY KEY (flight_cancellation_id),
  FOREIGN KEY (flight_id) REFERENCES flights (flight_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE flight_diversions (
  flight_diversion_id INT(11) NOT NULL AUTO_INCREMENT,
  flight_id INT(11) NOT NULL,
  diversion_airport_id INT(11) NOT NULL,
  PRIMARY KEY (flight_diversion_id),
  FOREIGN KEY (flight_id) REFERENCES flights (flight_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (diversion_airport_id) REFERENCES airports (airport_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE flight_performance (
  flight_performance_id INT(11) NOT NULL AUTO_INCREMENT,
  flight_id INT(11) NOT NULL,
  on_time_performance INT(11) NOT NULL,
  PRIMARY KEY (flight_performance_id),
  FOREIGN KEY (flight_id) REFERENCES flights (flight_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE flight_schedules (
  flight_schedule_id INT(11) NOT NULL AUTO_INCREMENT,
  flight_id INT(11) NOT NULL,
  departure_time DATETIME NOT NULL,
  arrival_time DATETIME NOT NULL,
  PRIMARY KEY (flight_schedule_id),

CREATE TABLE flight_status (

  PRIMARY KEY (flight_status_id),

CREATE TABLE flight_tracks (
  flight_track_id INT(11) NOT NULL AUTO_INCREMENT,

  track_id INT(11) NOT NULL,
