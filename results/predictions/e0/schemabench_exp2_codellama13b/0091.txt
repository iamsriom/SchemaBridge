-- Schema for database: schemabench_0091

CREATE TABLE participants (
  participant_id INT(11) NOT NULL AUTO_INCREMENT,
  first_name VARCHAR(255) NOT NULL,
  last_name VARCHAR(255) NOT NULL,
  gender VARCHAR(10) NOT NULL,
  birth_date DATE NOT NULL,
  PRIMARY KEY (participant_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE medical_history (
  medical_history_id INT(11) NOT NULL AUTO_INCREMENT,
  participant_id INT(11) NOT NULL,
  cholesterol_level DECIMAL(10,2) NOT NULL,
  blood_pressure DECIMAL(10,2) NOT NULL,
  body_mass_index DECIMAL(10,2) NOT NULL,
  smoking_habits VARCHAR(255) NOT NULL,
  alcohol_consumption VARCHAR(255) NOT NULL,
  PRIMARY KEY (medical_history_id),
  FOREIGN KEY (participant_id) REFERENCES participants(participant_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE lifestyle (
  lifestyle_id INT(11) NOT NULL AUTO_INCREMENT,
  participant_id INT(11) NOT NULL,
  exercise_frequency VARCHAR(255) NOT NULL,
  diet_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (lifestyle_id),
  FOREIGN KEY (participant_id) REFERENCES participants(participant_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE health_outcomes (
  health_outcome_id INT(11) NOT NULL AUTO_INCREMENT,
  participant_id INT(11) NOT NULL,
  atherosclerosis_risk_factor VARCHAR(255) NOT NULL,
  PRIMARY KEY (health_outcome_id),
  FOREIGN KEY (participant_id) REFERENCES participants(participant_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reports (
  report_id INT(11) NOT NULL AUTO_INCREMENT,
  participant_id INT(11) NOT NULL,
  report_type VARCHAR(255) NOT NULL,
  report_date DATE NOT NULL,
  PRIMARY KEY (report_id),
  FOREIGN KEY (participant_id) REFERENCES participants(participant_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_sources (
  data_source_id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (data_source_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_integrations (
  data_integration_id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_id INT(11) NOT NULL,
  data_integration_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (data_integration_id),
  FOREIGN KEY (data_source_id) REFERENCES data_sources(data_source_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_security (
  data_security_id INT(11) NOT NULL AUTO_INCREMENT,
  data_security_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (data_security_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE system_scalability (
  system_scalability_id INT(11) NOT NULL AUTO_INCREMENT,
  system_scalability_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (system_scalability_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_accessibility (
  user_accessibility_id INT(11) NOT NULL AUTO_INCREMENT,
  user_accessibility_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (user_accessibility_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
