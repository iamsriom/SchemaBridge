-- Schema for database: schemabench_0094
CREATE TABLE chemical_compounds (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  half_life_aerobic_aqueous_biodegradation VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE atoms (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chemical_compound_id INT(11) NOT NULL,
  element VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chemical_compound_id) REFERENCES chemical_compounds (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE bonds (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chemical_compound_id INT(11) NOT NULL,
  atom_1_id INT(11) NOT NULL,
  atom_2_id INT(11) NOT NULL,
  bond_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chemical_compound_id) REFERENCES chemical_compounds (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom_1_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom_2_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE molecular_groups (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chemical_compound_id INT(11) NOT NULL,
  atom_ids INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chemical_compound_id) REFERENCES chemical_compounds (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom_ids) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE molecules (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chemical_compound_id INT(11) NOT NULL,
  molecular_group_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chemical_compound_id) REFERENCES chemical_compounds (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (molecular_group_id) REFERENCES molecular_groups (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE chemical_compound_updates (
  id INT(11) NOT NULL AUTO_INCREMENT,
  chemical_compound_id INT(11) NOT NULL,
  update_type VARCHAR(255) NOT NULL,
  update_date DATETIME NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (chemical_compound_id) REFERENCES chemical_compounds (id) ON DELETE CASCADE ON UPDATE CASCADE

CREATE TABLE chemical_compound_deletions (
  id INT(11) NOT NULL AUTO_INCREMENT,

CREATE TABLE chemical_compound_analyses (
