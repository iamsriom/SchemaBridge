-- Schema for database: schemabench_0095
CREATE TABLE atoms (
  id INT(11) NOT NULL AUTO_INCREMENT,
  type VARCHAR(255) NOT NULL,
  charge INT(11) NOT NULL,
  designation VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE molecules (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE sbond_1 (
  id INT(11) NOT NULL AUTO_INCREMENT,
  atom_id INT(11) NOT NULL,
  molecule_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (atom_id) REFERENCES atoms(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (molecule_id) REFERENCES molecules(id) ON DELETE CASCADE ON UPDATE CASCADE
);
