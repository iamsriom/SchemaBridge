-- Schema for database: schemabench_0097
CREATE TABLE schools (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  district VARCHAR(255) NOT NULL,
  cdsccode VARCHAR(255) NOT NULL,
  location POINT NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (cdsccode)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sat_scores (
  id INT(11) NOT NULL AUTO_INCREMENT,
  school_id INT(11) NOT NULL,
  year INT(11) NOT NULL,
  average_score INT(11) NOT NULL,
  percent_above_1500 INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (school_id) REFERENCES schools(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE frpm_eligibility (
  id INT(11) NOT NULL AUTO_INCREMENT,
  school_id INT(11) NOT NULL,
  year INT(11) NOT NULL,
  age_group VARCHAR(255) NOT NULL,
  percent_eligible INT(11) NOT NULL,
  count_eligible INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (school_id) REFERENCES schools(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
