-- Schema for database: schemabench_0100
CREATE TABLE `household` (
  `household_id` int(11) NOT NULL AUTO_INCREMENT,
  `household_name` varchar(255) NOT NULL,
  `household_size` int(11) NOT NULL,
  `income` decimal(10,2) NOT NULL,
  `demographics` varchar(255) NOT NULL,
  PRIMARY KEY (`household_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expenditure` (
  `expenditure_id` int(11) NOT NULL AUTO_INCREMENT,
  `household_id` int(11) NOT NULL,
  `expenditure_type` varchar(255) NOT NULL,
  `expenditure_amount` decimal(10,2) NOT NULL,
  `expenditure_date` datetime NOT NULL,
  PRIMARY KEY (`expenditure_id`),
  FOREIGN KEY (`household_id`) REFERENCES `household` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `anonymized_data` (
  `anonymized_data_id` int(11) NOT NULL AUTO_INCREMENT,
  `household_id` int(11) NOT NULL,
  `anonymized_data` varchar(255) NOT NULL,
  PRIMARY KEY (`anonymized_data_id`),
  FOREIGN KEY (`household_id`) REFERENCES `household` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
