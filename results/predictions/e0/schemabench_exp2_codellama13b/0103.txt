CREATE TABLE `category` (
  `category_id` int(11) NOT NULL AUTO_INCREMENT,
  `category_name` varchar(255) NOT NULL,
  PRIMARY KEY (`category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `charge` (
  `charge_id` int(11) NOT NULL AUTO_INCREMENT,
  `charge_amount` decimal(10,2) NOT NULL,
  `charge_date` datetime NOT NULL,
  `charge_category_id` int(11) NOT NULL,
  `charge_member_id` int(11) NOT NULL,
  `charge_provider_id` int(11) NOT NULL,
  `charge_corporation_id` int(11) NOT NULL,
  `charge_region_id` int(11) NOT NULL,
  `charge_statement_id` int(11) NOT NULL,
  PRIMARY KEY (`charge_id`),
  FOREIGN KEY (`charge_category_id`) REFERENCES `category` (`category_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`charge_member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`charge_provider_id`) REFERENCES `provider` (`provider_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`charge_corporation_id`) REFERENCES `corporation` (`corporation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`charge_region_id`) REFERENCES `region` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`charge_statement_id`) REFERENCES `statement` (`statement_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
