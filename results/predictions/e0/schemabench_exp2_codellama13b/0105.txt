-- Schema for database: schemabench_0105

CREATE TABLE sentences (
  id INT(11) NOT NULL AUTO_INCREMENT,
  text VARCHAR(255) NOT NULL,
  classification VARCHAR(10) NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT classification_check CHECK (classification IN ('positive', 'negative'))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE terms (
  id INT(11) NOT NULL AUTO_INCREMENT,
  text VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sentence_terms (
  sentence_id INT(11) NOT NULL,
  term_id INT(11) NOT NULL,
  PRIMARY KEY (sentence_id, term_id),
  CONSTRAINT sentence_terms_ibfk_1 FOREIGN KEY (sentence_id) REFERENCES sentences (id) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT sentence_terms_ibfk_2 FOREIGN KEY (term_id) REFERENCES terms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE dcg_rules (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE dcg_rule_terms (
  dcg_rule_id INT(11) NOT NULL,
  term_id INT(11) NOT NULL,
  PRIMARY KEY (dcg_rule_id, term_id),
  CONSTRAINT dcg_rule_terms_ibfk_1 FOREIGN KEY (dcg_rule_id) REFERENCES dcg_rules (id) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT dcg_rule_terms_ibfk_2 FOREIGN KEY (term_id) REFERENCES terms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE dcg_rule_sentences (
  dcg_rule_id INT(11) NOT NULL,
  sentence_id INT(11) NOT NULL,
  PRIMARY KEY (dcg_rule_id, sentence_id),
  CONSTRAINT dcg_rule_sentences_ibfk_1 FOREIGN KEY (dcg_rule_id) REFERENCES dcg_rules (id) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT dcg_rule_sentences_ibfk_2 FOREIGN KEY (sentence_id) REFERENCES sentences (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
