-- Schema for database: schemabench_0106
CREATE TABLE `race` (
  `race_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `round` int(11) NOT NULL,
  `circuit_id` int(11) NOT NULL,
  `date` datetime NOT NULL,
  `time` time NOT NULL,
  `results` text NOT NULL,
  PRIMARY KEY (`race_id`),
  FOREIGN KEY (`circuit_id`) REFERENCES `circuit` (`circuit_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `circuit` (
  `circuit_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  PRIMARY KEY (`circuit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `driver` (
  `driver_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `nationality` varchar(255) NOT NULL,
  `birth_date` datetime NOT NULL,
  `birth_place` varchar(255) NOT NULL,
  `team_id` int(11) NOT NULL,
  PRIMARY KEY (`driver_id`),
  FOREIGN KEY (`team_id`) REFERENCES `team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `team` (
  `team_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `nationality` varchar(255) NOT NULL,
  `founded` int(11) NOT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `lap_time` (
  `lap_time_id` int(11) NOT NULL,
  `race_id` int(11) NOT NULL,
  `driver_id` int(11) NOT NULL,
  `lap` int(11) NOT NULL,
  `time` time NOT NULL,
  PRIMARY KEY (`lap_time_id`),
  FOREIGN KEY (`race_id`) REFERENCES `race` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pit_stop` (
  `pit_stop_id` int(11) NOT NULL,
  `race_id` int(11) NOT NULL,
  `driver_id` int(11) NOT NULL,
  `lap` int(11) NOT NULL,
  `stop_time` time NOT NULL,
  PRIMARY KEY (`pit_stop_id`),
  FOREIGN KEY (`race_id`) REFERENCES `race` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `qualifying` (
  `qualifying_id` int(11) NOT NULL,
  `race_id` int(11) NOT NULL,
  `driver_id` int(11) NOT NULL,
  `position` int(11) NOT NULL,
  `time` time NOT NULL,
  PRIMARY KEY (`qualifying_id`),
  FOREIGN KEY (`race_id`) REFERENCES `race` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `standings` (
  `standings_id` int(11) NOT NULL,
  `race_id` int(11) NOT NULL,
  `driver_id` int(11) NOT NULL,
  `points` int(11) NOT NULL,
  `position` int(11) NOT NULL,
  PRIMARY KEY (`standings_id`),
  FOREIGN KEY (`race_id`) REFERENCES `race` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
