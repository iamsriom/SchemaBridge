-- Schema for database: schemabench_0108
CREATE TABLE genes (
  gene_id INT(11) NOT NULL AUTO_INCREMENT,
  gene_name VARCHAR(255) NOT NULL,
  gene_type VARCHAR(255) NOT NULL,
  gene_function VARCHAR(255) NOT NULL,
  gene_localization VARCHAR(255) NOT NULL,
  PRIMARY KEY (gene_id),
  UNIQUE KEY gene_name (gene_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE interactions (
  interaction_id INT(11) NOT NULL AUTO_INCREMENT,
  gene_id INT(11) NOT NULL,
  interaction_type VARCHAR(255) NOT NULL,
  correlation_score DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (interaction_id),
  FOREIGN KEY (gene_id) REFERENCES genes (gene_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE gene_classification (
  gene_id INT(11) NOT NULL,
  classification_type VARCHAR(255) NOT NULL,
  classification_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (gene_id, classification_type),
  FOREIGN KEY (gene_id) REFERENCES genes (gene_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE gene_localization (
  gene_id INT(11) NOT NULL,
  localization_type VARCHAR(255) NOT NULL,
  localization_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (gene_id, localization_type),
  FOREIGN KEY (gene_id) REFERENCES genes (gene_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
