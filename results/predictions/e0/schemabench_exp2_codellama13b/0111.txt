-- Schema for database: schemabench_0111

CREATE TABLE patients (
  patient_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  age INT(11) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  PRIMARY KEY (patient_id)
);

CREATE TABLE medical_parameters (
  patient_id INT(11) NOT NULL,
  parameter_id INT(11) NOT NULL,
  value VARCHAR(255) NOT NULL,
  PRIMARY KEY (patient_id, parameter_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE fibrosis_activity (
  patient_id INT(11) NOT NULL,
  activity_id INT(11) NOT NULL,
  value VARCHAR(255) NOT NULL,
  PRIMARY KEY (patient_id, activity_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE disease_type (
  patient_id INT(11) NOT NULL,
  type_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, type_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE medical_history (
  patient_id INT(11) NOT NULL,
  history_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, history_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE relationships (
  patient_id INT(11) NOT NULL,
  relationship_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, relationship_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE data_integrity (
  patient_id INT(11) NOT NULL,
  integrity_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, integrity_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE performance (
  patient_id INT(11) NOT NULL,
  performance_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, performance_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE scalability (
  patient_id INT(11) NOT NULL,
  scalability_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, scalability_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE security (
  patient_id INT(11) NOT NULL,
  security_id INT(11) NOT NULL,
  PRIMARY KEY (patient_id, security_id),
  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE reliability (
  patient_id INT(11) NOT NULL,

  FOREIGN KEY (patient_id) REFERENCES patients (patient_id) ON DELETE CASCADE ON UPDATE CASCADE
