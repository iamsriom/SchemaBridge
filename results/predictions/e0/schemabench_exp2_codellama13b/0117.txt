-- Schema for database: schemabench_0117

CREATE TABLE atoms (
  id INT(11) NOT NULL AUTO_INCREMENT,
  element VARCHAR(255) NOT NULL,
  charge INT(11) NOT NULL,
  type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE bonds (
  id INT(11) NOT NULL AUTO_INCREMENT,
  bond_type VARCHAR(255) NOT NULL,
  atom1_id INT(11) NOT NULL,
  atom2_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (atom1_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom2_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE compounds (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  activity INT(11) NOT NULL,
  logP INT(11) NOT NULL,
  lumo INT(11) NOT NULL,
  active BOOLEAN NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE ring_structures (
  id INT(11) NOT NULL AUTO_INCREMENT,
  compound_id INT(11) NOT NULL,
  ring_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (compound_id) REFERENCES compounds (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (ring_id) REFERENCES rings (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE rings (
  id INT(11) NOT NULL AUTO_INCREMENT,
  atom1_id INT(11) NOT NULL,
  atom2_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (atom1_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom2_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE compound_atoms (
  compound_id INT(11) NOT NULL,
  atom_id INT(11) NOT NULL,
  PRIMARY KEY (compound_id, atom_id),
  FOREIGN KEY (compound_id) REFERENCES compounds (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE bond_atoms (
  bond_id INT(11) NOT NULL,
  atom1_id INT(11) NOT NULL,
  atom2_id INT(11) NOT NULL,
  PRIMARY KEY (bond_id, atom1_id, atom2_id),
  FOREIGN KEY (bond_id) REFERENCES bonds (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom1_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom2_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE ring_atoms (
  ring_id INT(11) NOT NULL,
  atom_id INT(11) NOT NULL,
  PRIMARY KEY (ring_id, atom_id),
  FOREIGN KEY (ring_id) REFERENCES rings (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (atom_id) REFERENCES atoms (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
