-- Schema for database: schemabench_0123
CREATE TABLE molecules (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  smiles VARCHAR(255) NOT NULL,
  inchi VARCHAR(255) NOT NULL,
  inchikey VARCHAR(255) NOT NULL,
  molecular_weight DECIMAL(10,2) NOT NULL,
  logp DECIMAL(10,2) NOT NULL,
  flexibility DECIMAL(10,2) NOT NULL,
  polarizability DECIMAL(10,2) NOT NULL,
  sigma DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (name, smiles, inchi, inchikey)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inhibitors (
  id INT(11) NOT NULL AUTO_INCREMENT,
  molecule_id INT(11) NOT NULL,
  inhibition_activity DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (molecule_id) REFERENCES molecules (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE predictions (
  id INT(11) NOT NULL AUTO_INCREMENT,
  molecule_id INT(11) NOT NULL,
  prediction DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (molecule_id) REFERENCES molecules (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reports (
  id INT(11) NOT NULL AUTO_INCREMENT,
  molecule_id INT(11) NOT NULL,
  report_type VARCHAR(255) NOT NULL,
  report_data TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (molecule_id) REFERENCES molecules (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE users (
  id INT(11) NOT NULL AUTO_INCREMENT,
  username VARCHAR(255) NOT NULL,
  password VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE user_roles (
  id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  role VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE permissions (
  id INT(11) NOT NULL AUTO_INCREMENT,
  role VARCHAR(255) NOT NULL,
  permission VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (role) REFERENCES user_roles (role) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_sources (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description TEXT NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_entries (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_id INT(11) NOT NULL,
  molecule_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_id) REFERENCES data_sources (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (molecule_id) REFERENCES molecules (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_metadata (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_id INT(11) NOT NULL,
  key VARCHAR(255) NOT NULL,
  value TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_id) REFERENCES data_sources (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_files (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_id INT(11) NOT NULL,
  file_name VARCHAR(255) NOT NULL,
  file_type VARCHAR(255) NOT NULL,
  file_size INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_id) REFERENCES data_sources (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_file_metadata (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_file_id INT(11) NOT NULL,
  key VARCHAR(255) NOT NULL,
  value TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_file_id) REFERENCES data_source_files (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_file_data (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_file_id INT(11) NOT NULL,
  data TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_file_id) REFERENCES data_source_files (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_file_data_metadata (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_file_data_id INT(11) NOT NULL,
  key VARCHAR(255) NOT NULL,
  value TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_file_data_id) REFERENCES data_source_file_data (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_file_data_data (
  id INT(11) NOT NULL AUTO_INCREMENT,
  data_source_file_data_id INT(11) NOT NULL,
  data TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_file_data_id) REFERENCES data_source_file_data (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE data_source_file_data_data_metadata (
  id INT(11) NOT NULL AUTO_INCREMENT,

  key VARCHAR(255) NOT NULL,
  value TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (data_source_file_data_data_id) REFERENCES data_source_file_data_data (id)
