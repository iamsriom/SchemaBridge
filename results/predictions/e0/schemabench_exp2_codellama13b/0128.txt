-- Schema for database: schemabench_0128

CREATE TABLE businesses (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  owner VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inspections (
  id INT(11) NOT NULL AUTO_INCREMENT,
  business_id INT(11) NOT NULL,
  score INT(11) NOT NULL,
  date DATE NOT NULL,
  type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (business_id) REFERENCES businesses (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE violations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  business_id INT(11) NOT NULL,
  inspection_id INT(11) NOT NULL,
  type VARCHAR(255) NOT NULL,
  risk_category VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (business_id) REFERENCES businesses (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (inspection_id) REFERENCES inspections (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inspection_scores (
  id INT(11) NOT NULL AUTO_INCREMENT,
  inspection_id INT(11) NOT NULL,
  score INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (inspection_id) REFERENCES inspections (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE business_violations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  business_id INT(11) NOT NULL,
  violation_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (business_id) REFERENCES businesses (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (violation_id) REFERENCES violations (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE inspection_violations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  inspection_id INT(11) NOT NULL,
  violation_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (inspection_id) REFERENCES inspections (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (violation_id) REFERENCES violations (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
