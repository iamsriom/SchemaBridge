-- Schema for database: schemabench_0129
CREATE TABLE `shakespeare_works` (
  `work_id` int(11) NOT NULL AUTO_INCREMENT,
  `work_title` varchar(255) NOT NULL,
  `work_author` varchar(255) NOT NULL,
  `work_date` datetime NOT NULL,
  `work_text` text NOT NULL,
  PRIMARY KEY (`work_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shakespeare_characters` (
  `character_id` int(11) NOT NULL AUTO_INCREMENT,
  `character_name` varchar(255) NOT NULL,
  `character_description` text NOT NULL,
  `work_id` int(11) NOT NULL,
  PRIMARY KEY (`character_id`),
  FOREIGN KEY (`work_id`) REFERENCES `shakespeare_works` (`work_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `shakespeare_lines` (
  `line_id` int(11) NOT NULL AUTO_INCREMENT,
  `line_text` text NOT NULL,
  `line_number` int(11) NOT NULL,
  `character_id` int(11) NOT NULL,
  `work_id` int(11) NOT NULL,
  PRIMARY KEY (`line_id`),
  FOREIGN KEY (`character_id`) REFERENCES `shakespeare_characters` (`character_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`work_id`) REFERENCES `shakespeare_works` (`work_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
