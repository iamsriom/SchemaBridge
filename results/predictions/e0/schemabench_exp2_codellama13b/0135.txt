CREATE TABLE trains (
  train_id INT(11) NOT NULL AUTO_INCREMENT,
  train_name VARCHAR(255) NOT NULL,
  direction VARCHAR(10) NOT NULL,
  PRIMARY KEY (train_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE cars (
  car_id INT(11) NOT NULL AUTO_INCREMENT,
  train_id INT(11) NOT NULL,
  position INT(11) NOT NULL,
  shape VARCHAR(255) NOT NULL,
  length INT(11) NOT NULL,
  sides INT(11) NOT NULL,
  roof_type VARCHAR(255) NOT NULL,
  num_wheels INT(11) NOT NULL,
  load_shape VARCHAR(255) NOT NULL,
  load_quantity INT(11) NOT NULL,
  PRIMARY KEY (car_id),
  FOREIGN KEY (train_id) REFERENCES trains (train_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE train_loads (
  train_id INT(11) NOT NULL,
  load_id INT(11) NOT NULL,
  PRIMARY KEY (train_id, load_id),
  FOREIGN KEY (train_id) REFERENCES trains (train_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (load_id) REFERENCES loads (load_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE loads (

CREATE TABLE train_stops (
  train_id INT(11) NOT NULL,

  PRIMARY KEY (train_id, stop_id),
