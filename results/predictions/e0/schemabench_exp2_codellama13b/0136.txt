-- Schema for database: schemabench_0136
CREATE TABLE students (
  student_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  PRIMARY KEY (student_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE professors (
  professor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  department VARCHAR(255) NOT NULL,
  PRIMARY KEY (professor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE courses (
  course_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  difficulty VARCHAR(255) NOT NULL,
  PRIMARY KEY (course_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE registrations (
  registration_id INT(11) NOT NULL AUTO_INCREMENT,
  student_id INT(11) NOT NULL,
  course_id INT(11) NOT NULL,
  grade VARCHAR(255) NOT NULL,
  PRIMARY KEY (registration_id),
  FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE course_evaluations (
  evaluation_id INT(11) NOT NULL AUTO_INCREMENT,
  student_id INT(11) NOT NULL,
  course_id INT(11) NOT NULL,
  rating INT(11) NOT NULL,
  PRIMARY KEY (evaluation_id),
  FOREIGN KEY (student_id) REFERENCES students(student_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (course_id) REFERENCES courses(course_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE research_assistants (
  assistant_id INT(11) NOT NULL AUTO_INCREMENT,
  professor_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (assistant_id),
  FOREIGN KEY (professor_id) REFERENCES professors(professor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE capability_assessments (
  assessment_id INT(11) NOT NULL AUTO_INCREMENT,
  assistant_id INT(11) NOT NULL,
  capability VARCHAR(255) NOT NULL,
  score INT(11) NOT NULL,
  PRIMARY KEY (assessment_id),
  FOREIGN KEY (assistant_id) REFERENCES research_assistants(assistant_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE salary_details (
  salary_id INT(11) NOT NULL AUTO_INCREMENT,
  assistant_id INT(11) NOT NULL,
  salary INT(11) NOT NULL,
  PRIMARY KEY (salary_id),
  FOREIGN KEY (assistant_id) REFERENCES research_assistants(assistant_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
