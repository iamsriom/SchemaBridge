-- Schema for database: schemabench_0138

CREATE TABLE `department` (
`department_id` int(11) NOT NULL,
`department_name` varchar(255) NOT NULL,
`department_description` varchar(255) NOT NULL,
PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `person` (
`person_id` int(11) NOT NULL,
`person_name` varchar(255) NOT NULL,
`person_email` varchar(255) NOT NULL,
`person_phone` varchar(255) NOT NULL,
`person_department` int(11) NOT NULL,
PRIMARY KEY (`person_id`),
FOREIGN KEY (`person_department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student` (
`student_id` int(11) NOT NULL,
`student_name` varchar(255) NOT NULL,
`student_email` varchar(255) NOT NULL,
`student_phone` varchar(255) NOT NULL,
`student_department` int(11) NOT NULL,
PRIMARY KEY (`student_id`),
FOREIGN KEY (`student_department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `professor` (
`professor_id` int(11) NOT NULL,
`professor_name` varchar(255) NOT NULL,
`professor_email` varchar(255) NOT NULL,
`professor_phone` varchar(255) NOT NULL,
`professor_department` int(11) NOT NULL,
PRIMARY KEY (`professor_id`),
FOREIGN KEY (`professor_department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
`course_id` int(11) NOT NULL,
`course_name` varchar(255) NOT NULL,
`course_description` varchar(255) NOT NULL,
`course_level` int(11) NOT NULL,
`course_professor` int(11) NOT NULL,
PRIMARY KEY (`course_id`),
FOREIGN KEY (`course_professor`) REFERENCES `professor` (`professor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advisory_relationship` (
`advisory_relationship_id` int(11) NOT NULL,
`advisory_relationship_student` int(11) NOT NULL,
`advisory_relationship_professor` int(11) NOT NULL,
PRIMARY KEY (`advisory_relationship_id`),
FOREIGN KEY (`advisory_relationship_student`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`advisory_relationship_professor`) REFERENCES `professor` (`professor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_assignment` (
`course_assignment_id` int(11) NOT NULL,
`course_assignment_course` int(11) NOT NULL,
`course_assignment_professor` int(11) NOT NULL,
PRIMARY KEY (`course_assignment_id`),
FOREIGN KEY (`course_assignment_course`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`course_assignment_professor`) REFERENCES `professor` (`professor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
