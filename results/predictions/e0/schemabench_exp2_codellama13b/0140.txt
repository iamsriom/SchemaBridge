-- Schema for database: schemabench_0140

CREATE TABLE `voyages` (
`voyage_id` int(11) NOT NULL AUTO_INCREMENT,
`vessel_name` varchar(255) NOT NULL,
`master` varchar(255) NOT NULL,
`tonnage` int(11) NOT NULL,
`type` varchar(255) NOT NULL,
`construction_details` varchar(255) NOT NULL,
`departure_date` datetime NOT NULL,
`arrival_date` datetime NOT NULL,
`cabo_stop_date` datetime NOT NULL,
`casualties` int(11) NOT NULL,
PRIMARY KEY (`voyage_id`),
FOREIGN KEY (`master`) REFERENCES `masters` (`master_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `masters` (
`master_id` int(11) NOT NULL AUTO_INCREMENT,
`master_name` varchar(255) NOT NULL,
`vessel_name` varchar(255) NOT NULL,
`tonnage` int(11) NOT NULL,
`type` varchar(255) NOT NULL,
`construction_details` varchar(255) NOT NULL,
PRIMARY KEY (`master_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
`personnel_id` int(11) NOT NULL AUTO_INCREMENT,
`voyage_id` int(11) NOT NULL,
`personnel_category` varchar(255) NOT NULL,
`status` varchar(255) NOT NULL,
`departure_date` datetime NOT NULL,
`arrival_date` datetime NOT NULL,
`cabo_stop_date` datetime NOT NULL,
`casualties` int(11) NOT NULL,
PRIMARY KEY (`personnel_id`),
FOREIGN KEY (`voyage_id`) REFERENCES `voyages` (`voyage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `invoices` (
`invoice_id` int(11) NOT NULL AUTO_INCREMENT,
`voyage_id` int(11) NOT NULL,
`chamber_details` varchar(255) NOT NULL,
PRIMARY KEY (`invoice_id`),
FOREIGN KEY (`voyage_id`) REFERENCES `voyages` (`voyage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
