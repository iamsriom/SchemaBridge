CREATE TABLE `stores` (
`store_id` int(11) NOT NULL,
`store_name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
`weather_station_id` int(11) NOT NULL,
PRIMARY KEY (`store_id`),
FOREIGN KEY (`weather_station_id`) REFERENCES `weather_stations` (`weather_station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather_stations` (
`weather_station_id` int(11) NOT NULL,
`station_name` varchar(255) NOT NULL,
`location` varchar(255) NOT NULL,
PRIMARY KEY (`weather_station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

`product_id` int(11) NOT NULL,
`product_name` varchar(255) NOT NULL,

PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
`sale_id` int(11) NOT NULL,

`product_id` int(11) NOT NULL,
`sale_date` datetime NOT NULL,
`sale_quantity` int(11) NOT NULL,
PRIMARY KEY (`sale_id`),
FOREIGN KEY (`store_id`) REFERENCES `stores` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather_data` (
`weather_data_id` int(11) NOT NULL,

`wind_speed` int(11) NOT NULL,
PRIMARY KEY (`weather_data_id`),
FOREIGN KEY (`weather_station_id`) REFERENCES `weather_stations` (`weather_station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_forecast` (
`sale_forecast_id` int(11) NOT NULL,

`forecast_quantity` int(11) NOT NULL,
PRIMARY KEY (`sale_forecast_id`),
FOREIGN KEY (`store_id`) REFERENCES `stores` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE,
