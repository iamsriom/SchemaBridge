-- Schema for database: schemabench_0145
CREATE TABLE pokemon (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  height INT(11) NOT NULL,
  weight INT(11) NOT NULL,
  base_experience INT(11) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE types (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE moves (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  power INT(11) NOT NULL,
  accuracy INT(11) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE evolutions (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  evolution_id INT(11) NOT NULL,
  evolution_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (evolution_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE abilities (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  hidden BOOLEAN NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE base_stats (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  hp INT(11) NOT NULL,
  attack INT(11) NOT NULL,
  defense INT(11) NOT NULL,
  special_attack INT(11) NOT NULL,
  special_defense INT(11) NOT NULL,
  speed INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE habitats (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE historical_data (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  version VARCHAR(255) NOT NULL,
  classification VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE pokemon_types (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  type_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (type_id) REFERENCES types (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE move_types (
  id INT(11) NOT NULL AUTO_INCREMENT,
  move_id INT(11) NOT NULL,
  type_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (move_id) REFERENCES moves (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (type_id) REFERENCES types (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE pokemon_abilities (
  id INT(11) NOT NULL AUTO_INCREMENT,

  ability_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (ability_id) REFERENCES abilities (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE pokemon_base_stats (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  base_stat_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (base_stat_id) REFERENCES base_stats (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE pokemon_habitats (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  habitat_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (habitat_id) REFERENCES habitats (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE pokemon_historical_data (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  historical_data_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (historical_data_id) REFERENCES historical_data (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE pokemon_types_types (
  id INT(11) NOT NULL AUTO_INCREMENT,

  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_type_id) REFERENCES pokemon_types (id) ON DELETE CASCADE ON UPDATE CASCADE,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
