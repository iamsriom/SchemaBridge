-- Schema for database: schemabench_0158

CREATE TABLE installations (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE laborcodes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE billingcodes (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE tickets (
  id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  labor_code INT(11) NOT NULL,
  work_hours INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (labor_code) REFERENCES laborcodes (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reports (
  id INT(11) NOT NULL AUTO_INCREMENT,
  creation_date DATETIME NOT NULL,
  labor_code INT(11) NOT NULL,
  work_hours INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (labor_code) REFERENCES laborcodes (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE shell (
  id INT(11) NOT NULL AUTO_INCREMENT,
  theme VARCHAR(255) NOT NULL,
  menu_items VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
