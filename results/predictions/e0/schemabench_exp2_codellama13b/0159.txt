-- Schema for database: schemabench_0159

CREATE TABLE films (
  film_id INT(11) NOT NULL AUTO_INCREMENT,
  title VARCHAR(255) NOT NULL,
  description TEXT,
  release_year INT(11),
  rating VARCHAR(255),
  PRIMARY KEY (film_id),
  UNIQUE KEY (title)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE actors (
  actor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (actor_id),
  UNIQUE KEY (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE filming_locations (
  location_id INT(11) NOT NULL AUTO_INCREMENT,
  city VARCHAR(255) NOT NULL,
  country VARCHAR(255) NOT NULL,
  PRIMARY KEY (location_id),
  UNIQUE KEY (city, country)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE languages (
  language_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (language_id),
  UNIQUE KEY (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_categories (
  category_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (category_id),
  UNIQUE KEY (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_actors (
  film_id INT(11) NOT NULL,
  actor_id INT(11) NOT NULL,
  PRIMARY KEY (film_id, actor_id),
  FOREIGN KEY (film_id) REFERENCES films (film_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (actor_id) REFERENCES actors (actor_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_locations (
  film_id INT(11) NOT NULL,
  location_id INT(11) NOT NULL,
  PRIMARY KEY (film_id, location_id),
  FOREIGN KEY (film_id) REFERENCES films (film_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (location_id) REFERENCES filming_locations (location_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_languages (
  film_id INT(11) NOT NULL,
  language_id INT(11) NOT NULL,
  PRIMARY KEY (film_id, language_id),
  FOREIGN KEY (film_id) REFERENCES films (film_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (language_id) REFERENCES languages (language_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE film_categories (
  film_id INT(11) NOT NULL,
  category_id INT(11) NOT NULL,
  PRIMARY KEY (film_id, category_id),
  FOREIGN KEY (film_id) REFERENCES films (film_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (category_id) REFERENCES film_categories (category_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
