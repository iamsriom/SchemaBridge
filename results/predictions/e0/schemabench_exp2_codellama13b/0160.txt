-- Schema for database: schemabench_0160
CREATE TABLE company (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  headquarters VARCHAR(255) NOT NULL,
  established DATE NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE stock (
  id INT(11) NOT NULL AUTO_INCREMENT,
  company_id INT(11) NOT NULL,
  symbol VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (company_id) REFERENCES company (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE exchange (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE symbol (
  id INT(11) NOT NULL AUTO_INCREMENT,
  stock_id INT(11) NOT NULL,
  exchange_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (stock_id) REFERENCES stock (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (exchange_id) REFERENCES exchange (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE trader (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE account (
  id INT(11) NOT NULL AUTO_INCREMENT,
  trader_id INT(11) NOT NULL,
  broker_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (trader_id) REFERENCES trader (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (broker_id) REFERENCES broker (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE broker (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE transaction (
  id INT(11) NOT NULL AUTO_INCREMENT,
  trader_id INT(11) NOT NULL,
  stock_id INT(11) NOT NULL,
  exchange_id INT(11) NOT NULL,
  symbol_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  commission DECIMAL(10,2) NOT NULL,
  timestamp TIMESTAMP NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (trader_id) REFERENCES trader (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (stock_id) REFERENCES stock (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (exchange_id) REFERENCES exchange (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (symbol_id) REFERENCES symbol (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE order (
  id INT(11) NOT NULL AUTO_INCREMENT,
  trader_id INT(11) NOT NULL,
  stock_id INT(11) NOT NULL,
  exchange_id INT(11) NOT NULL,
  symbol_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (trader_id) REFERENCES trader (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (stock_id) REFERENCES stock (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (exchange_id) REFERENCES exchange (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (symbol_id) REFERENCES symbol (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE trade (
  id INT(11) NOT NULL AUTO_INCREMENT,
  order_id INT(11) NOT NULL,
  transaction_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (order_id) REFERENCES order (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (transaction_id) REFERENCES transaction (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE trade_history (
  id INT(11) NOT NULL AUTO_INCREMENT,
  trader_id INT(11) NOT NULL,
  stock_id INT(11) NOT NULL,
  exchange_id INT(11) NOT NULL,
  symbol_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (trader_id) REFERENCES trader (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (stock_id) REFERENCES stock (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (exchange_id) REFERENCES exchange (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (symbol_id) REFERENCES symbol (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE broker_fee (
  id INT(11) NOT NULL AUTO_INCREMENT,
  broker_id INT(11) NOT NULL,
  transaction_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (broker_id) REFERENCES broker (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (transaction_id) REFERENCES transaction (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE exchange_fee (
  id INT(11) NOT NULL AUTO_INCREMENT,
  exchange_id INT(11) NOT NULL,
  transaction_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (exchange_id) REFERENCES exchange (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (transaction_id) REFERENCES transaction (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE security_fee (
  id INT(11) NOT NULL AUTO_INCREMENT,
  stock_id INT(11) NOT NULL,
  transaction_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (stock_id) REFERENCES stock (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (transaction_id) REFERENCES transaction (id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE tax (
  id INT(11) NOT NULL AUTO_INCREMENT,
  trader_id INT(11) NOT NULL,
  transaction_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (trader_id) REFERENCES trader (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (transaction_id) REFERENCES transaction (id) ON DELETE CASCADE ON UPDATE CASCADE
);
