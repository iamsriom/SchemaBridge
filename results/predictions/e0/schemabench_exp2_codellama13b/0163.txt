-- Schema for database: schemabench_0163

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
);

CREATE TABLE events (
  event_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  event_name VARCHAR(255) NOT NULL,
  event_date DATE NOT NULL,
  event_cost DECIMAL(10,2) NOT NULL,
  audience_size INT(11) NOT NULL,
  PRIMARY KEY (event_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE event_requests (
  request_id INT(11) NOT NULL AUTO_INCREMENT,
  event_id INT(11) NOT NULL,
  request_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (request_id),
  FOREIGN KEY (event_id) REFERENCES events (event_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE employees (
  employee_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (employee_id)
);

CREATE TABLE event_plans (
  plan_id INT(11) NOT NULL AUTO_INCREMENT,
  employee_id INT(11) NOT NULL,
  event_id INT(11) NOT NULL,
  plan_name VARCHAR(255) NOT NULL,
  plan_description TEXT NOT NULL,
  PRIMARY KEY (plan_id),
  FOREIGN KEY (employee_id) REFERENCES employees (employee_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (event_id) REFERENCES events (event_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE resources (
  resource_id INT(11) NOT NULL AUTO_INCREMENT,
  resource_name VARCHAR(255) NOT NULL,
  resource_description TEXT NOT NULL,
  PRIMARY KEY (resource_id)
);

CREATE TABLE resource_allocations (
  allocation_id INT(11) NOT NULL AUTO_INCREMENT,
  resource_id INT(11) NOT NULL,
  event_plan_id INT(11) NOT NULL,
  PRIMARY KEY (allocation_id),
  FOREIGN KEY (resource_id) REFERENCES resources (resource_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (event_plan_id) REFERENCES event_plans (plan_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE facilities (
  facility_id INT(11) NOT NULL AUTO_INCREMENT,
  facility_name VARCHAR(255) NOT NULL,
  facility_description TEXT NOT NULL,
  PRIMARY KEY (facility_id)
);

CREATE TABLE facility_allocations (
  allocation_id INT(11) NOT NULL AUTO_INCREMENT,
  facility_id INT(11) NOT NULL,
  event_plan_id INT(11) NOT NULL,
  PRIMARY KEY (allocation_id),
  FOREIGN KEY (facility_id) REFERENCES facilities (facility_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (event_plan_id) REFERENCES event_plans (plan_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE user_roles (
  role_id INT(11) NOT NULL AUTO_INCREMENT,
  role_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (role_id)
);

CREATE TABLE user_role_assignments (
  assignment_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  role_id INT(11) NOT NULL,
  PRIMARY KEY (assignment_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (role_id) REFERENCES user_roles (role_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE user_permissions (
  permission_id INT(11) NOT NULL AUTO_INCREMENT,
  permission_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (permission_id)
);

CREATE TABLE user_permission_assignments (
  assignment_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  permission_id INT(11) NOT NULL,
  PRIMARY KEY (assignment_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (permission_id) REFERENCES user_permissions (permission_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE user_sessions (
  session_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  session_start TIMESTAMP NOT NULL,
  session_end TIMESTAMP NOT NULL,
  PRIMARY KEY (session_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE user_activity_logs (
  log_id INT(11) NOT NULL AUTO_INCREMENT,
  user_id INT(11) NOT NULL,
  log_type VARCHAR(255) NOT NULL,
  log_description TEXT NOT NULL,
  log_timestamp TIMESTAMP NOT NULL,
  PRIMARY KEY (log_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE system_logs (
  log_id INT(11) NOT NULL AUTO_INCREMENT,
  log_type VARCHAR(255) NOT NULL,
  log_description TEXT NOT NULL,
  log_timestamp TIMESTAMP NOT NULL,
  PRIMARY KEY (log_id)
);
