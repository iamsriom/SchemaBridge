-- Schema for database: schemabench_0164
CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  loyalty_level VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
);

CREATE TABLE products (
  product_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (product_id)
);

CREATE TABLE sales (
  sale_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  sale_date DATETIME NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (sale_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE inventory (
  inventory_id INT(11) NOT NULL AUTO_INCREMENT,
  product_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (inventory_id),
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlets (
  outlet_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  size VARCHAR(255) NOT NULL,
  PRIMARY KEY (outlet_id)
);

CREATE TABLE staff (
  staff_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  role VARCHAR(255) NOT NULL,
  location VARCHAR(255) NOT NULL,
  start_date DATETIME NOT NULL,
  PRIMARY KEY (staff_id)
);

CREATE TABLE transactions (
  transaction_id INT(11) NOT NULL AUTO_INCREMENT,
  sale_id INT(11) NOT NULL,
  outlet_id INT(11) NOT NULL,
  transaction_date DATETIME NOT NULL,
  PRIMARY KEY (transaction_id),
  FOREIGN KEY (sale_id) REFERENCES sales (sale_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE customer_loyalty (
  customer_id INT(11) NOT NULL,
  loyalty_level VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE product_promotions (
  product_id INT(11) NOT NULL,
  promotion_id INT(11) NOT NULL,
  PRIMARY KEY (product_id, promotion_id),
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (promotion_id) REFERENCES promotions (promotion_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE promotions (
  promotion_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  description VARCHAR(255) NOT NULL,
  PRIMARY KEY (promotion_id)
);

CREATE TABLE waste_management (
  waste_id INT(11) NOT NULL AUTO_INCREMENT,
  product_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (waste_id),
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_targets (
  outlet_id INT(11) NOT NULL,
  target_date DATETIME NOT NULL,
  target_quantity INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, target_date),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlet_staff (
  outlet_id INT(11) NOT NULL,
  staff_id INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, staff_id),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (staff_id) REFERENCES staff (staff_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlet_products (
  outlet_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, product_id),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlet_customers (
  outlet_id INT(11) NOT NULL,
  customer_id INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, customer_id),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlet_transactions (
  outlet_id INT(11) NOT NULL,
  transaction_id INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, transaction_id),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (transaction_id) REFERENCES transactions (transaction_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlet_inventory (
  outlet_id INT(11) NOT NULL,
  inventory_id INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, inventory_id),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (inventory_id) REFERENCES inventory (inventory_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sales_outlet_waste (
  outlet_id INT(11) NOT NULL,
  waste_id INT(11) NOT NULL,
  PRIMARY KEY (outlet_id, waste_id),
  FOREIGN KEY (outlet_id) REFERENCES sales_outlets (outlet_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (waste_id) REFERENCES waste_management (waste_id) ON DELETE CASCADE ON UPDATE CASCADE
);
