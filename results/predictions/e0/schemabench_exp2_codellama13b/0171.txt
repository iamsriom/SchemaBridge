-- Schema for database: schemabench_0171

CREATE TABLE trade_data (
  trade_id INT(11) NOT NULL AUTO_INCREMENT,
  portfolio_id INT(11) NOT NULL,
  trade_date DATE NOT NULL,
  instrument_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (trade_id),
  FOREIGN KEY (portfolio_id) REFERENCES portfolio (portfolio_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (instrument_id) REFERENCES instrument (instrument_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE audit_trail (
  audit_id INT(11) NOT NULL AUTO_INCREMENT,
  trade_id INT(11) NOT NULL,
  action VARCHAR(255) NOT NULL,
  timestamp DATETIME NOT NULL,
  PRIMARY KEY (audit_id),
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE counterparty (
  counterparty_id INT(11) NOT NULL AUTO_INCREMENT,
  counterparty_name VARCHAR(255) NOT NULL,
  exposure_limit DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (counterparty_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE instrument (
  instrument_id INT(11) NOT NULL AUTO_INCREMENT,
  instrument_name VARCHAR(255) NOT NULL,
  maturity_date DATE NOT NULL,
  settlement_type VARCHAR(255) NOT NULL,
  currency_code VARCHAR(3) NOT NULL,
  PRIMARY KEY (instrument_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trader (
  trader_id INT(11) NOT NULL AUTO_INCREMENT,
  trader_name VARCHAR(255) NOT NULL,
  contact_details VARCHAR(255) NOT NULL,
  PRIMARY KEY (trader_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trade_status (
  trade_status_id INT(11) NOT NULL AUTO_INCREMENT,
  trade_id INT(11) NOT NULL,
  status VARCHAR(255) NOT NULL,
  PRIMARY KEY (trade_status_id),
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE portfolio (
  portfolio_id INT(11) NOT NULL AUTO_INCREMENT,
  portfolio_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (portfolio_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trade_portfolio (
  trade_id INT(11) NOT NULL,
  portfolio_id INT(11) NOT NULL,
  PRIMARY KEY (trade_id, portfolio_id),
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (portfolio_id) REFERENCES portfolio (portfolio_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trade_counterparty (
  trade_id INT(11) NOT NULL,
  counterparty_id INT(11) NOT NULL,
  PRIMARY KEY (trade_id, counterparty_id),
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (counterparty_id) REFERENCES counterparty (counterparty_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trade_instrument (
  trade_id INT(11) NOT NULL,
  instrument_id INT(11) NOT NULL,
  PRIMARY KEY (trade_id, instrument_id),
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (instrument_id) REFERENCES instrument (instrument_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trade_trader (
  trade_id INT(11) NOT NULL,
  trader_id INT(11) NOT NULL,
  PRIMARY KEY (trade_id, trader_id),
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (trader_id) REFERENCES trader (trader_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trade_status_history (
  trade_status_id INT(11) NOT NULL,
  trade_id INT(11) NOT NULL,
  status_date DATETIME NOT NULL,
  PRIMARY KEY (trade_status_id, trade_id),
  FOREIGN KEY (trade_status_id) REFERENCES trade_status (trade_status_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (trade_id) REFERENCES trade_data (trade_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
