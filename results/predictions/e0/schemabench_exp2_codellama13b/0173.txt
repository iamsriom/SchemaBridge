-- Schema for database: schemabench_0173

CREATE TABLE donors (
  donor_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  blood_type VARCHAR(255) NOT NULL,
  medical_history TEXT,
  PRIMARY KEY (donor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE patients (
  patient_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  blood_type VARCHAR(255) NOT NULL,
  medical_history TEXT,
  PRIMARY KEY (patient_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE hospitals (
  hospital_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (hospital_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE facilities (
  facility_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (facility_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE blood_donations (
  donation_id INT(11) NOT NULL AUTO_INCREMENT,
  donor_id INT(11) NOT NULL,
  patient_id INT(11) NOT NULL,
  hospital_id INT(11) NOT NULL,
  facility_id INT(11) NOT NULL,
  blood_type VARCHAR(255) NOT NULL,
  collection_date DATETIME NOT NULL,
  expiry_date DATETIME NOT NULL,
  contamination_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (donation_id),
  FOREIGN KEY (donor_id) REFERENCES donors(donor_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (patient_id) REFERENCES patients(patient_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (hospital_id) REFERENCES hospitals(hospital_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (facility_id) REFERENCES facilities(facility_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE blood_transfusions (
  transfusion_id INT(11) NOT NULL AUTO_INCREMENT,
  patient_id INT(11) NOT NULL,
  hospital_id INT(11) NOT NULL,
  facility_id INT(11) NOT NULL,
  blood_type VARCHAR(255) NOT NULL,
  transfusion_date DATETIME NOT NULL,
  PRIMARY KEY (transfusion_id),
  FOREIGN KEY (patient_id) REFERENCES patients(patient_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (hospital_id) REFERENCES hospitals(hospital_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (facility_id) REFERENCES facilities(facility_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
