-- Schema for database: schemabench_0179
CREATE TABLE races (
  race_id INT(11) NOT NULL AUTO_INCREMENT,
  date DATE NOT NULL,
  circuit_id INT(11) NOT NULL,
  start_time TIME NOT NULL,
  cancelled BOOLEAN NOT NULL DEFAULT FALSE,
  PRIMARY KEY (race_id),
  FOREIGN KEY (circuit_id) REFERENCES circuits (circuit_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE circuits (
  circuit_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  length INT(11) NOT NULL,
  configuration VARCHAR(255) NOT NULL,
  PRIMARY KEY (circuit_id)
);

CREATE TABLE drivers (
  driver_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  country VARCHAR(255) NOT NULL,
  PRIMARY KEY (driver_id)
);

CREATE TABLE teams (
  team_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  country VARCHAR(255) NOT NULL,
  status VARCHAR(255) NOT NULL,
  PRIMARY KEY (team_id)
);

CREATE TABLE cars (
  car_id INT(11) NOT NULL AUTO_INCREMENT,
  class VARCHAR(255) NOT NULL,
  chassis VARCHAR(255) NOT NULL,
  engine VARCHAR(255) NOT NULL,
  PRIMARY KEY (car_id)
);

CREATE TABLE tyres (
  tyre_id INT(11) NOT NULL AUTO_INCREMENT,
  brand VARCHAR(255) NOT NULL,
  PRIMARY KEY (tyre_id)
);

CREATE TABLE results (
  result_id INT(11) NOT NULL AUTO_INCREMENT,
  race_id INT(11) NOT NULL,
  driver_id INT(11) NOT NULL,
  team_id INT(11) NOT NULL,
  car_id INT(11) NOT NULL,
  tyre_id INT(11) NOT NULL,
  position INT(11) NOT NULL,
  laps INT(11) NOT NULL,
  distance INT(11) NOT NULL,
  time INT(11) NOT NULL,
  PRIMARY KEY (result_id),
  FOREIGN KEY (race_id) REFERENCES races (race_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (driver_id) REFERENCES drivers (driver_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (team_id) REFERENCES teams (team_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (car_id) REFERENCES cars (car_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (tyre_id) REFERENCES tyres (tyre_id) ON DELETE CASCADE ON UPDATE CASCADE
);
