CREATE TABLE `aircraft` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `registration` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `airline` varchar(255) NOT NULL,
  `seats` int(11) NOT NULL,
  `engine_type` varchar(255) NOT NULL,
  `engine_model` varchar(255) NOT NULL,
  `engine_manufacturer` varchar(255) NOT NULL,
  `engine_hours` int(11) NOT NULL,
  `grounded` tinyint(1) NOT NULL,
  `grounded_reason` varchar(255) DEFAULT NULL,
  `last_maintenance` datetime NOT NULL,
  `next_maintenance` datetime NOT NULL,
  `last_inspection` datetime NOT NULL,
  `next_inspection` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `registration` (`registration`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flights` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_id` int(11) NOT NULL,
  `departure_airport` varchar(255) NOT NULL,
  `arrival_airport` varchar(255) NOT NULL,
  `departure_time` datetime NOT NULL,
  `arrival_time` datetime NOT NULL,
  `flight_number` varchar(255) NOT NULL,
  `airline` varchar(255) NOT NULL,
  `seats` int(11) NOT NULL,
  `grounded` tinyint(1) NOT NULL,
  `grounded_reason` varchar(255) DEFAULT NULL,
  `last_maintenance` datetime NOT NULL,
  `next_maintenance` datetime NOT NULL,
  `last_inspection` datetime NOT NULL,
  `next_inspection` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `flight_number` (`flight_number`),
  FOREIGN KEY (`aircraft_id`) REFERENCES `aircraft` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `engine_usage` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_id` int(11) NOT NULL,
  `engine_type` varchar(255) NOT NULL,
  `engine_model` varchar(255) NOT NULL,
  `engine_manufacturer` varchar(255) NOT NULL,
  `engine_hours` int(11) NOT NULL,
  `last_maintenance` datetime NOT NULL,
  `next_maintenance` datetime NOT NULL,
  `last_inspection` datetime NOT NULL,
  `next_inspection` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `engine_type` (`engine_type`, `engine_model`, `engine_manufacturer`),
  FOREIGN KEY (`aircraft_id`) REFERENCES `aircraft` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
