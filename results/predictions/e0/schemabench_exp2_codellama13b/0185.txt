-- Schema for database: schemabench_0185
CREATE TABLE pokemon (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  type VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE ivs (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  hp INT(11) NOT NULL,
  attack INT(11) NOT NULL,
  defense INT(11) NOT NULL,
  special_attack INT(11) NOT NULL,
  special_defense INT(11) NOT NULL,
  speed INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE stats (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  hp INT(11) NOT NULL,
  attack INT(11) NOT NULL,
  defense INT(11) NOT NULL,
  special_attack INT(11) NOT NULL,
  special_defense INT(11) NOT NULL,
  speed INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE upgrades (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  level INT(11) NOT NULL,
  dust INT(11) NOT NULL,
  candy INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE trades (
  id INT(11) NOT NULL AUTO_INCREMENT,
  pokemon_id INT(11) NOT NULL,
  partner_id INT(11) NOT NULL,
  friendship INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (pokemon_id) REFERENCES pokemon(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (partner_id) REFERENCES pokemon(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
