-- Schema for database: schemabench_0187
CREATE TABLE `regions` (
  `region_id` int(11) NOT NULL AUTO_INCREMENT,
  `region_name` varchar(255) NOT NULL,
  `zip_code` varchar(255) NOT NULL,
  PRIMARY KEY (`region_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `distributors` (
  `distributor_id` int(11) NOT NULL AUTO_INCREMENT,
  `distributor_name` varchar(255) NOT NULL,
  `distributor_address` varchar(255) NOT NULL,
  `distributor_phone` varchar(255) NOT NULL,
  `distributor_email` varchar(255) NOT NULL,
  PRIMARY KEY (`distributor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dealerships` (
  `dealership_id` int(11) NOT NULL AUTO_INCREMENT,
  `dealership_name` varchar(255) NOT NULL,
  `dealership_address` varchar(255) NOT NULL,
  `dealership_phone` varchar(255) NOT NULL,
  `dealership_email` varchar(255) NOT NULL,
  `region_id` int(11) NOT NULL,
  `distributor_id` int(11) NOT NULL,
  PRIMARY KEY (`dealership_id`),
  FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`distributor_id`) REFERENCES `distributors` (`distributor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicles` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_make` varchar(255) NOT NULL,
  `vehicle_model` varchar(255) NOT NULL,
  `vehicle_body_type` varchar(255) NOT NULL,
  `vehicle_model_year` int(11) NOT NULL,
  `vehicle_price` decimal(10,2) NOT NULL,
  `dealership_id` int(11) NOT NULL,
  PRIMARY KEY (`vehicle_id`),
  FOREIGN KEY (`dealership_id`) REFERENCES `dealerships` (`dealership_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `managers` (
  `manager_id` int(11) NOT NULL AUTO_INCREMENT,
  `manager_name` varchar(255) NOT NULL,
  `manager_salary` decimal(10,2) NOT NULL,
  `dealership_id` int(11) NOT NULL,
  PRIMARY KEY (`manager_id`),
  FOREIGN KEY (`dealership_id`) REFERENCES `dealerships` (`dealership_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_agents` (
  `sales_agent_id` int(11) NOT NULL AUTO_INCREMENT,
  `sales_agent_name` varchar(255) NOT NULL,
  `sales_agent_email` varchar(255) NOT NULL,
  `sales_agent_phone` varchar(255) NOT NULL,
  `manager_id` int(11) NOT NULL,
  `dealership_id` int(11) NOT NULL,
  PRIMARY KEY (`sales_agent_id`),
  FOREIGN KEY (`manager_id`) REFERENCES `managers` (`manager_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`dealership_id`) REFERENCES `dealerships` (`dealership_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `customer_email` varchar(255) NOT NULL,
  `customer_phone` varchar(255) NOT NULL,
  `sales_agent_id` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`),
  FOREIGN KEY (`sales_agent_id`) REFERENCES `sales_agents` (`sales_agent_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `insurance_details` (
  `insurance_id` int(11) NOT NULL AUTO_INCREMENT,
  `insurance_type` varchar(255) NOT NULL,
  `insurance_renewal_date` datetime NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  PRIMARY KEY (`insurance_id`),
  FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_deals` (
  `sales_deal_id` int(11) NOT NULL AUTO_INCREMENT,
  `sales_deal_date` datetime NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `sales_agent_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `insurance_id` int(11) NOT NULL,
  PRIMARY KEY (`sales_deal_id`),
  FOREIGN KEY (`vehicle_id`) REFERENCES `vehicles` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`sales_agent_id`) REFERENCES `sales_agents` (`sales_agent_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`insurance_id`) REFERENCES `insurance_details` (`insurance_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
