-- Schema for database: schemabench_0191

CREATE TABLE bookings (
  booking_id INT(11) NOT NULL AUTO_INCREMENT,
  date DATE NOT NULL,
  table_number INT(11) NOT NULL,
  customer_name VARCHAR(255) NOT NULL,
  staff_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (booking_id),
  FOREIGN KEY (staff_name) REFERENCES staff (name) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE orders (
  order_id INT(11) NOT NULL AUTO_INCREMENT,
  order_date DATE NOT NULL,
  quantity INT(11) NOT NULL,
  total_cost DECIMAL(10,2) NOT NULL,
  delivery_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (order_id),
  FOREIGN KEY (order_id) REFERENCES bookings (booking_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE menu (
  item_id INT(11) NOT NULL AUTO_INCREMENT,
  item_name VARCHAR(255) NOT NULL,
  category VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (item_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  full_name VARCHAR(255) NOT NULL,
  contact_number VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE staff (
  staff_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  role VARCHAR(255) NOT NULL,
  salary DECIMAL(10,2) NOT NULL,
  address VARCHAR(255) NOT NULL,
  contact_number VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  PRIMARY KEY (staff_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
