-- Schema for database: schemabench_0194

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
);

CREATE TABLE sales (
  sale_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  menu_item VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  date_time DATETIME NOT NULL,
  PRIMARY KEY (sale_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE menu (
  menu_item VARCHAR(255) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (menu_item)
);

CREATE TABLE members (
  member_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  membership_type VARCHAR(255) NOT NULL,
  PRIMARY KEY (member_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE reports (
  report_id INT(11) NOT NULL AUTO_INCREMENT,
  report_type VARCHAR(255) NOT NULL,
  report_date DATETIME NOT NULL,
  PRIMARY KEY (report_id)
);

CREATE TABLE report_details (
  report_detail_id INT(11) NOT NULL AUTO_INCREMENT,
  report_id INT(11) NOT NULL,
  report_data VARCHAR(255) NOT NULL,
  PRIMARY KEY (report_detail_id),
  FOREIGN KEY (report_id) REFERENCES reports (report_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE recommendations (
  recommendation_id INT(11) NOT NULL AUTO_INCREMENT,
  recommendation_type VARCHAR(255) NOT NULL,
  recommendation_data VARCHAR(255) NOT NULL,
  PRIMARY KEY (recommendation_id)
);

CREATE TABLE user_datasets (
  dataset_id INT(11) NOT NULL AUTO_INCREMENT,
  dataset_name VARCHAR(255) NOT NULL,
  dataset_description VARCHAR(255) NOT NULL,
  PRIMARY KEY (dataset_id)
);

CREATE TABLE user_dataset_details (
  dataset_detail_id INT(11) NOT NULL AUTO_INCREMENT,
  dataset_id INT(11) NOT NULL,
  dataset_data VARCHAR(255) NOT NULL,
  PRIMARY KEY (dataset_detail_id),
  FOREIGN KEY (dataset_id) REFERENCES user_datasets (dataset_id) ON DELETE CASCADE ON UPDATE CASCADE
);
