CREATE TABLE clients (
  client_id INT(11) NOT NULL AUTO_INCREMENT,
  client_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (client_id)
);

CREATE TABLE client_interests (
  client_id INT(11) NOT NULL,
  interest_id INT(11) NOT NULL,
  PRIMARY KEY (client_id, interest_id),
  FOREIGN KEY (client_id) REFERENCES clients(client_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (interest_id) REFERENCES interests(interest_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE interests (
  interest_id INT(11) NOT NULL AUTO_INCREMENT,
  interest_name VARCHAR(255) NOT NULL,
  PRIMARY KEY (interest_id)
);

CREATE TABLE interest_metrics (
  client_id INT(11) NOT NULL,
  interest_id INT(11) NOT NULL,
  metric_type VARCHAR(255) NOT NULL,
  metric_value INT(11) NOT NULL,
  PRIMARY KEY (client_id, interest_id, metric_type),
  FOREIGN KEY (client_id) REFERENCES clients(client_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (interest_id) REFERENCES interests(interest_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE interest_rankings (
  client_id INT(11) NOT NULL,
  interest_id INT(11) NOT NULL,
  ranking INT(11) NOT NULL,
  PRIMARY KEY (client_id, interest_id, ranking),
  FOREIGN KEY (client_id) REFERENCES clients(client_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (interest_id) REFERENCES interests(interest_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE insights (
  insight_id INT(11) NOT NULL AUTO_INCREMENT,
  client_id INT(11) NOT NULL,
  insight_type VARCHAR(255) NOT NULL,
