-- Schema for database: BankingSystem

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  first_name VARCHAR(255) NOT NULL,
  last_name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE accounts (
  account_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  account_type VARCHAR(255) NOT NULL,
  balance DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (account_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE transactions (
  transaction_id INT(11) NOT NULL AUTO_INCREMENT,
  account_id INT(11) NOT NULL,
  transaction_type VARCHAR(255) NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  transaction_date DATETIME NOT NULL,
  PRIMARY KEY (transaction_id),
  FOREIGN KEY (account_id) REFERENCES accounts (account_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE balance_updates (
  balance_update_id INT(11) NOT NULL AUTO_INCREMENT,
  account_id INT(11) NOT NULL,
  balance DECIMAL(10,2) NOT NULL,
  update_date DATETIME NOT NULL,
  PRIMARY KEY (balance_update_id),
  FOREIGN KEY (account_id) REFERENCES accounts (account_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE security (
  security_id INT(11) NOT NULL AUTO_INCREMENT,
  account_id INT(11) NOT NULL,
  security_type VARCHAR(255) NOT NULL,
  security_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (security_id),
  FOREIGN KEY (account_id) REFERENCES accounts (account_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE performance (
  performance_id INT(11) NOT NULL AUTO_INCREMENT,
  account_id INT(11) NOT NULL,
  performance_type VARCHAR(255) NOT NULL,
  performance_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (performance_id),
  FOREIGN KEY (account_id) REFERENCES accounts (account_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE scalability (
  scalability_id INT(11) NOT NULL AUTO_INCREMENT,
  account_id INT(11) NOT NULL,
  scalability_type VARCHAR(255) NOT NULL,
  scalability_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (scalability_id),
  FOREIGN KEY (account_id) REFERENCES accounts (account_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reliability (
  reliability_id INT(11) NOT NULL AUTO_INCREMENT,
  account_id INT(11) NOT NULL,
  reliability_type VARCHAR(255) NOT NULL,
  reliability_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (reliability_id),
  FOREIGN KEY (account_id) REFERENCES accounts (account_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
