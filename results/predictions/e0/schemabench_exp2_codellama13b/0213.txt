-- Schema for database: schemabench_0213
CREATE TABLE orders (
  order_id INT(11) NOT NULL AUTO_INCREMENT,
  customer_id INT(11) NOT NULL,
  order_date DATETIME NOT NULL,
  shipping_mode VARCHAR(255) NOT NULL,
  shipping_cost DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (order_id),
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE customers (
  customer_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  address VARCHAR(255) NOT NULL,
  PRIMARY KEY (customer_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE products (
  product_id INT(11) NOT NULL AUTO_INCREMENT,
  product_name VARCHAR(255) NOT NULL,
  product_category VARCHAR(255) NOT NULL,
  product_subcategory VARCHAR(255) NOT NULL,
  PRIMARY KEY (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE order_items (
  order_item_id INT(11) NOT NULL AUTO_INCREMENT,
  order_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (order_item_id),
  FOREIGN KEY (order_id) REFERENCES orders (order_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE shipping_modes (
  shipping_mode_id INT(11) NOT NULL AUTO_INCREMENT,
  shipping_mode VARCHAR(255) NOT NULL,
  PRIMARY KEY (shipping_mode_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE addresses (
  address_id INT(11) NOT NULL AUTO_INCREMENT,
  address VARCHAR(255) NOT NULL,
  PRIMARY KEY (address_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales (
  sale_id INT(11) NOT NULL AUTO_INCREMENT,
  order_id INT(11) NOT NULL,
  sale_date DATETIME NOT NULL,
  discount DECIMAL(10,2) NOT NULL,
  profit DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (sale_id),
  FOREIGN KEY (order_id) REFERENCES orders (order_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE shipping_costs (
  shipping_cost_id INT(11) NOT NULL AUTO_INCREMENT,
  shipping_mode_id INT(11) NOT NULL,
  shipping_cost DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (shipping_cost_id),
  FOREIGN KEY (shipping_mode_id) REFERENCES shipping_modes (shipping_mode_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE order_addresses (
  order_address_id INT(11) NOT NULL AUTO_INCREMENT,
  order_id INT(11) NOT NULL,
  address_id INT(11) NOT NULL,
  PRIMARY KEY (order_address_id),
  FOREIGN KEY (order_id) REFERENCES orders (order_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (address_id) REFERENCES addresses (address_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_product (
  sale_id INT(11) NOT NULL,
  product_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (sale_id, product_id),
  FOREIGN KEY (sale_id) REFERENCES sales (sale_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products (product_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_customer (
  sale_id INT(11) NOT NULL,
  customer_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (sale_id, customer_id),
  FOREIGN KEY (sale_id) REFERENCES sales (sale_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (customer_id) REFERENCES customers (customer_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_shipping_mode (
  sale_id INT(11) NOT NULL,
  shipping_mode_id INT(11) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (sale_id, shipping_mode_id),
  FOREIGN KEY (sale_id) REFERENCES sales (sale_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (shipping_mode_id) REFERENCES shipping_modes (shipping_mode_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_date (
  sale_date DATE NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (sale_date),
  FOREIGN KEY (sale_date) REFERENCES sales (sale_date) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_product_category (
  product_category VARCHAR(255) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (product_category),
  FOREIGN KEY (product_category) REFERENCES products (product_category) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_product_subcategory (
  product_subcategory VARCHAR(255) NOT NULL,
  quantity INT(11) NOT NULL,
  PRIMARY KEY (product_subcategory),
  FOREIGN KEY (product_subcategory) REFERENCES products (product_subcategory) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE sales_by_customer_address (
  customer_address VARCHAR(255) NOT NULL,

  PRIMARY KEY (customer_address),

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
