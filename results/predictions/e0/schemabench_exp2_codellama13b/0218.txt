-- Schema for database: schemabench_0218

CREATE TABLE grampanchayats (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  taluka VARCHAR(255) NOT NULL,
  total_villages INT(11) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE wards (
  id INT(11) NOT NULL AUTO_INCREMENT,
  grampanchayat_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  total_villages INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (grampanchayat_id) REFERENCES grampanchayats(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE villages (
  id INT(11) NOT NULL AUTO_INCREMENT,
  ward_id INT(11) NOT NULL,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (ward_id) REFERENCES wards(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE farmers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  aadh_no VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  gender VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE machines (
  id INT(11) NOT NULL AUTO_INCREMENT,
  mach_id VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE rental_details (
  id INT(11) NOT NULL AUTO_INCREMENT,
  farmer_id INT(11) NOT NULL,
  mach_id INT(11) NOT NULL,
  rental_duration INT(11) NOT NULL,
  ward_id INT(11) NOT NULL,
  member_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (farmer_id) REFERENCES farmers(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (mach_id) REFERENCES machines(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (ward_id) REFERENCES wards(id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (member_id) REFERENCES ward_members(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE billing_details (
  id INT(11) NOT NULL AUTO_INCREMENT,
  rental_id INT(11) NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  payment_method VARCHAR(255) NOT NULL,
  payment_date DATE NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (rental_id) REFERENCES rental_details(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE ward_members (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  ward_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (ward_id) REFERENCES wards(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE grampanchayat_officials (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  grampanchayat_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (grampanchayat_id) REFERENCES grampanchayats(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
