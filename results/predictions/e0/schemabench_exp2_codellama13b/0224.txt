-- Schema for database: schemabench_0224

CREATE TABLE employees (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  age INT(11) NOT NULL,
  department VARCHAR(255) NOT NULL,
  title VARCHAR(255) NOT NULL,
  salary INT(11) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE mentorship_program (
  id INT(11) NOT NULL AUTO_INCREMENT,
  employee_id INT(11) NOT NULL,
  mentor_id INT(11) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (employee_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (mentor_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE reports (
  id INT(11) NOT NULL AUTO_INCREMENT,
  manager_id INT(11) NOT NULL,
  employee_id INT(11) NOT NULL,
  report_date DATE NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (manager_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (employee_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE mentorship_program_reports (
  id INT(11) NOT NULL AUTO_INCREMENT,
  mentorship_program_id INT(11) NOT NULL,
  report_date DATE NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (mentorship_program_id) REFERENCES mentorship_program (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE mentorship_program_reports_details (
  id INT(11) NOT NULL AUTO_INCREMENT,
  mentorship_program_report_id INT(11) NOT NULL,
  employee_id INT(11) NOT NULL,
  mentor_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (mentorship_program_report_id) REFERENCES mentorship_program_reports (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (employee_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (mentor_id) REFERENCES employees (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE mentorship_program_reports_comments (
  id INT(11) NOT NULL AUTO_INCREMENT,
  mentorship_program_report_id INT(11) NOT NULL,
  comment TEXT NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (mentorship_program_report_id) REFERENCES mentorship_program_reports (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
