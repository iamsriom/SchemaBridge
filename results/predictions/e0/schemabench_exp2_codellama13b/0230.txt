-- Schema for database: schemabench_0230

CREATE TABLE players (
  player_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  age INT(11) NOT NULL,
  team_id INT(11) NOT NULL,
  team_number INT(11) NOT NULL,
  injury_status VARCHAR(255) NOT NULL,
  PRIMARY KEY (player_id),
  FOREIGN KEY (team_id) REFERENCES teams (team_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE teams (
  team_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  manager VARCHAR(255) NOT NULL,
  team_captain VARCHAR(255) NOT NULL,
  PRIMARY KEY (team_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE matches (
  match_id INT(11) NOT NULL AUTO_INCREMENT,
  team1_id INT(11) NOT NULL,
  team2_id INT(11) NOT NULL,
  venue VARCHAR(255) NOT NULL,
  officiating_crew VARCHAR(255) NOT NULL,
  match_score INT(11) NOT NULL,
  key_players VARCHAR(255) NOT NULL,
  PRIMARY KEY (match_id),
  FOREIGN KEY (team1_id) REFERENCES teams (team_id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (team2_id) REFERENCES teams (team_id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE stadiums (
  stadium_id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  capacity INT(11) NOT NULL,
  current_condition VARCHAR(255) NOT NULL,
  PRIMARY KEY (stadium_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
