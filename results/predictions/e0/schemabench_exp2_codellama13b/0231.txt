-- Schema for database: schemabench_0231
CREATE TABLE cities (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  country VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY (name, country)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE stadiums (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  city_id INT(11) NOT NULL,
  seating_capacity INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (city_id) REFERENCES cities (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE teams (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  country VARCHAR(255) NOT NULL,
  stadium_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (stadium_id) REFERENCES stadiums (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE players (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  nationality VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  jersey_number INT(11) NOT NULL,
  position VARCHAR(255) NOT NULL,
  height INT(11) NOT NULL,
  weight INT(11) NOT NULL,
  preferred_foot VARCHAR(255) NOT NULL,
  team_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (team_id) REFERENCES teams (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE managers (
  id INT(11) NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  nationality VARCHAR(255) NOT NULL,
  team_id INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (team_id) REFERENCES teams (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE matches (
  id INT(11) NOT NULL AUTO_INCREMENT,
  season INT(11) NOT NULL,
  date_time DATETIME NOT NULL,
  home_team_id INT(11) NOT NULL,
  away_team_id INT(11) NOT NULL,
  stadium_id INT(11) NOT NULL,
  score_home INT(11) NOT NULL,
  score_away INT(11) NOT NULL,
  penalty_shootout_status TINYINT(1) NOT NULL,
  attendance INT(11) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (home_team_id) REFERENCES teams (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (away_team_id) REFERENCES teams (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (stadium_id) REFERENCES stadiums (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE goals (
  id INT(11) NOT NULL AUTO_INCREMENT,
  match_id INT(11) NOT NULL,
  scoring_player_id INT(11) NOT NULL,
  assisting_player_id INT(11) NOT NULL,
  duration INT(11) NOT NULL,
  description VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  FOREIGN KEY (match_id) REFERENCES matches (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (scoring_player_id) REFERENCES players (id) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (assisting_player_id) REFERENCES players (id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
