-- Schema for database: schemabench_0232
CREATE TABLE `communication_logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `log_date` datetime NOT NULL,
  `log_time` time NOT NULL,
  `channel_name` varchar(255) NOT NULL,
  `channel_type` varchar(255) NOT NULL,
  `ocean_region` varchar(255) NOT NULL,
  `burst_timing_offset` int(11) NOT NULL,
  `burst_frequency_offset` int(11) NOT NULL,
  `doppler_effects` int(11) NOT NULL,
  `oscillator_stability` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`channel_name`) REFERENCES `channels` (`name`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`channel_type`) REFERENCES `channel_types` (`type`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ocean_region`) REFERENCES `ocean_regions` (`region`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `channels` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `channel_types` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ocean_regions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `region` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
