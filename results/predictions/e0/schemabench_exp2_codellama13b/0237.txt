-- Schema for database: schemabench_0237

CREATE TABLE `books` (
`book_id` int(11) NOT NULL AUTO_INCREMENT,
`title` varchar(255) NOT NULL,
`author_id` int(11) NOT NULL,
`genre_id` int(11) NOT NULL,
`publisher` varchar(255) NOT NULL,
`publication_date` datetime NOT NULL,
`isbn` varchar(255) NOT NULL,
`description` text NOT NULL,
PRIMARY KEY (`book_id`),
FOREIGN KEY (`author_id`) REFERENCES `authors` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`genre_id`) REFERENCES `genres` (`genre_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `authors` (
`author_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
PRIMARY KEY (`author_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `genres` (
`genre_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
PRIMARY KEY (`genre_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patrons` (
`patron_id` int(11) NOT NULL AUTO_INCREMENT,
`name` varchar(255) NOT NULL,
`email` varchar(255) NOT NULL,
`phone` varchar(255) NOT NULL,
PRIMARY KEY (`patron_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reservations` (
`reservation_id` int(11) NOT NULL AUTO_INCREMENT,
`book_id` int(11) NOT NULL,
`patron_id` int(11) NOT NULL,
`reservation_date` datetime NOT NULL,
`return_date` datetime NOT NULL,
PRIMARY KEY (`reservation_id`),
FOREIGN KEY (`book_id`) REFERENCES `books` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (`patron_id`) REFERENCES `patrons` (`patron_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reservation_statuses` (
`reservation_status_id` int(11) NOT NULL AUTO_INCREMENT,
`reservation_id` int(11) NOT NULL,
`status` varchar(255) NOT NULL,
PRIMARY KEY (`reservation_status_id`),
FOREIGN KEY (`reservation_id`) REFERENCES `reservations` (`reservation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notifications` (
`notification_id` int(11) NOT NULL AUTO_INCREMENT,
`reservation_id` int(11) NOT NULL,
`notification_type` varchar(255) NOT NULL,
`notification_date` datetime NOT NULL,
PRIMARY KEY (`notification_id`),
FOREIGN KEY (`reservation_id`) REFERENCES `reservations` (`reservation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
