-- Schema for database: `silk_bench`

-- For table: `django_session`
CREATE TABLE `django_session` (
  `session_key` varchar(40) NOT NULL PRIMARY KEY,
  `session_data` longtext NOT NULL,
  `expire_date` datetime NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `silk_profile`
CREATE TABLE `silk_profile` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `name` varchar(255) NOT NULL,
  `user` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `request_count` int(11) NOT NULL,
  `sql_queries_count` int(11) NOT NULL,
  `data` longtext NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `silk_request`
CREATE TABLE `silk_request` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `profile_id` int(11) NOT NULL,
  `method` varchar(10) NOT NULL,
  `path` varchar(255) NOT NULL,
  `status_code` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `query_params` longtext NOT NULL,
  `data` longtext NOT NULL,
  `headers` longtext NOT NULL,
  `host` varchar(255) NOT NULL,
  `scheme` varchar(10) NOT NULL,
  `root_url` varchar(255) NOT NULL,
  `middleware_time` decimal(10, 2) NOT NULL,
  `view_time` decimal(10, 2) NOT NULL,
  `sql_queries_time` decimal(10, 2) NOT NULL,
  `template_name` varchar(255) NOT NULL,
  `template_content_type` varchar(255) NOT NULL,
  `template_length` int(11) NOT NULL,
  `template_checksum` varchar(32) NOT NULL,
  `template_rendered` longtext NOT NULL,
  `status` varchar(10) NOT NULL,
  `exception` longtext NOT NULL,
  `traceback` longtext NOT NULL,
  `meta` longtext NOT NULL,
  FOREIGN KEY (`profile_id`) REFERENCES `silk_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `silk_sql_query`
CREATE TABLE `silk_sql_query` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `request_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `sql` longtext NOT NULL,
  `duration` decimal(10, 2) NOT NULL,
  FOREIGN KEY (`request_id`) REFERENCES `silk_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `silk_profile_request`
CREATE TABLE `silk_profile_request` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `profile_id` int(11) NOT NULL,
  `request_id` int(11) NOT NULL,
  FOREIGN KEY (`profile_id`) REFERENCES `silk_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`request_id`) REFERENCES `silk_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
