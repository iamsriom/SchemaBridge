-- Schema for database: `django_auth`

-- Table: `user_profile`
CREATE TABLE `user_profile` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `user` int(11) NOT NULL,
    `phone_number` varchar(15) NOT NULL,
    `email` varchar(255) NOT NULL,
    `is_active` tinyint(1) NOT NULL DEFAULT '1',
    PRIMARY KEY (`id`),
    UNIQUE KEY `user` (`user`),
    CONSTRAINT `user_ibfk_1` FOREIGN KEY (`user`) REFERENCES `auth_user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `phone_call`
CREATE TABLE `phone_call` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `user_profile` int(11) NOT NULL,
    `call_token` varchar(255) NOT NULL,
    `expiry_date` datetime NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `user_profile` (`user_profile`),
    CONSTRAINT `user_profile_ibfk_1` FOREIGN KEY (`user_profile`) REFERENCES `user_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sms`
CREATE TABLE `sms` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `user_profile` int(11) NOT NULL,
    `sms_token` varchar(255) NOT NULL,
    `expiry_date` datetime NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `user_profile` (`user_profile`),
    CONSTRAINT `user_profile_ibfk_2` FOREIGN KEY (`user_profile`) REFERENCES `user_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `token_generator`
CREATE TABLE `token_generator` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `user_profile` int(11) NOT NULL,
    `token` varchar(255) NOT NULL,
    `expiry_date` datetime NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `user_profile` (`user_profile`),
    CONSTRAINT `user_profile_ibfk_3` FOREIGN KEY (`user_profile`) REFERENCES `user_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `yubi_key`
CREATE TABLE `yubi_key` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `user_profile` int(11) NOT NULL,
    `yubi_key_id` varchar(255) NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `user_profile` (`user_profile`),
    CONSTRAINT `user_profile_ibfk_4` FOREIGN KEY (`user_profile`) REFERENCES `user_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
