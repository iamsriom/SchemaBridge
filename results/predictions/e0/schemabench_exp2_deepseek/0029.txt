-- Schema for database: `robots_txt_manager`

-- For table: `sites`
CREATE TABLE `sites` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `domain` varchar(255) NOT NULL,
    `name` varchar(255) NOT NULL,
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `rules`
CREATE TABLE `rules` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `site_id` int(11) NOT NULL,
    `rule_type` varchar(255) NOT NULL,
    `value` varchar(255) NOT NULL,
    `crawl_delay` int(11) DEFAULT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`site_id`) REFERENCES `sites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `url_patterns`
CREATE TABLE `url_patterns` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `rule_id` int(11) NOT NULL,
    `pattern` varchar(255) NOT NULL,
    `is_allowed` tinyint(1) NOT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`rule_id`) REFERENCES `rules` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `site_associations`
CREATE TABLE `site_associations` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `site_id` int(11) NOT NULL,
    `associated_site_id` int(11) NOT NULL,
    PRIMARY KEY (`id`),
    FOREIGN KEY (`site_id`) REFERENCES `sites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`associated_site_id`) REFERENCES `sites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
