-- Schema for database: `django_filer`

-- For table: `filer_folder`

CREATE TABLE `filer_folder` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `parent_id` int(11) DEFAULT NULL,
    `owner_id` int(11) NOT NULL,
    `path` varchar(255) NOT NULL,
    `depth` int(11) NOT NULL,
    `numchild` int(11) NOT NULL,
    PRIMARY KEY (`id`),
    KEY `filer_folder_parent_id_5f3f4f4c` (`parent_id`),
    KEY `filer_folder_owner_id_5f3f4f4c` (`owner_id`),
    CONSTRAINT `filer_folder_ibfk_1` FOREIGN KEY (`parent_id`) REFERENCES `filer_folder` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `filer_folder_ibfk_2` FOREIGN KEY (`owner_id`) REFERENCES `auth_user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
