-- Schema for database: `schemabench_0062`

-- For table: `reaction_types`
CREATE TABLE `reaction_types` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `description` varchar(255) DEFAULT NULL,
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `reactables`
CREATE TABLE `reactables` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `reactable_type` varchar(255) NOT NULL,
    `reactable_id` int(11) NOT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `reactable_type_reactable_id` (`reactable_type`,`reactable_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `reactions`
CREATE TABLE `reactions` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `user_id` int(11) NOT NULL,
    `reaction_type_id` int(11) NOT NULL,
    `reactable_id` int(11) NOT NULL,
    `value` float DEFAULT NULL,
    `created_at` datetime DEFAULT NULL,
    `updated_at` datetime DEFAULT NULL,
    PRIMARY KEY (`id`),
    KEY `reactions_user_id_foreign` (`user_id`),
    KEY `reactions_reaction_type_id_foreign` (`reaction_type_id`),
    KEY `reactions_reactable_id_foreign` (`reactable_id`),
    CONSTRAINT `reactions_user_id_foreign` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `reactions_reaction_type_id_foreign` FOREIGN KEY (`reaction_type_id`) REFERENCES `reaction_types` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT `reactions_reactable_id_foreign` FOREIGN KEY (`reactable_id`) REFERENCES `reactables` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `users`
CREATE TABLE `users` (
    `id` int(11) NOT NULL AUTO_INCREMENT,
    `name` varchar(255) NOT NULL,
    `email` varchar(255) NOT NULL,
    `password` varchar(255) NOT NULL,
    `remember_token` varchar(100) DEFAULT NULL,
    `created_at` datetime DEFAULT NULL,
    `updated_at` datetime DEFAULT NULL,
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
