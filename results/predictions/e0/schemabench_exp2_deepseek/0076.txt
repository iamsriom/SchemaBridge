-- Schema for database: `schema_bench_0076`

-- For table: `server`

CREATE TABLE `server` (
  `server_id` int(11) NOT NULL AUTO_INCREMENT,
  `server_name` varchar(255) NOT NULL,
  `ssh_user` varchar(255) NOT NULL,
  `ssh_port` int(11) NOT NULL,
  `ip_address` varchar(255) NOT NULL,
  PRIMARY KEY (`server_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `disk_space`

CREATE TABLE `disk_space` (
  `check_id` int(11) NOT NULL AUTO_INCREMENT,
  `server_id` int(11) NOT NULL,
  `disk_space` decimal(5,2) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`check_id`),
  FOREIGN KEY (`server_id`) REFERENCES `server` (`server_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `process_availability`

CREATE TABLE `process_availability` (
  `check_id` int(11) NOT NULL AUTO_INCREMENT,
  `server_id` int(11) NOT NULL,
  `process_name` varchar(255) NOT NULL,
  `process_status` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`check_id`),
  FOREIGN KEY (`server_id`) REFERENCES `server` (`server_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `alert`

CREATE TABLE `alert` (
  `alert_id` int(11) NOT NULL AUTO_INCREMENT,
  `check_id` int(11) NOT NULL,
  `alert_message` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`alert_id`),
  FOREIGN KEY (`check_id`) REFERENCES `disk_space` (`check_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `check_schedule`

CREATE TABLE `check_schedule` (
  `schedule_id` int(11) NOT NULL AUTO_INCREMENT,
  `check_name` varchar(255) NOT NULL,
  `check_interval` int(11) NOT NULL,
  `enabled` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`schedule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `report`

CREATE TABLE `report` (
  `report_id` int(11) NOT NULL AUTO_INCREMENT,
  `check_id` int(11) NOT NULL,
  `report_content` text NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`report_id`),
  FOREIGN KEY (`check_id`) REFERENCES `disk_space` (`check_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
