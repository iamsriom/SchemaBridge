-- Schema for database: `schemabench_0090`

-- Table: `airlines`
CREATE TABLE `airlines` (
  `airline_id` int(11) NOT NULL,
  `airline_name` varchar(255) NOT NULL,
  PRIMARY KEY (`airline_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `airports`
CREATE TABLE `airports` (
  `airport_id` int(11) NOT NULL,
  `airport_name` varchar(255) NOT NULL,
  PRIMARY KEY (`airport_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `flights`
CREATE TABLE `flights` (
  `flight_id` int(11) NOT NULL,
  `airline_id` int(11) NOT NULL,
  `airport_id` int(11) NOT NULL,
  `flight_date` date NOT NULL,
  `flight_time` time NOT NULL,
  `flight_status` varchar(255) NOT NULL,
  PRIMARY KEY (`flight_id`),
  FOREIGN KEY (`airline_id`) REFERENCES `airlines` (`airline_id`),
  FOREIGN KEY (`airport_id`) REFERENCES `airports` (`airport_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `delays`
CREATE TABLE `delays` (
  `flight_id` int(11) NOT NULL,
  `delay_minutes` int(11) NOT NULL,
  `delay_reason` varchar(255) NOT NULL,
  PRIMARY KEY (`flight_id`),
  FOREIGN KEY (`flight_id`) REFERENCES `flights` (`flight_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `cancellations`
CREATE TABLE `cancellations` (
  `flight_id` int(11) NOT NULL,
  `cancellation_reason` varchar(255) NOT NULL,
  PRIMARY KEY (`flight_id`),
  FOREIGN KEY (`flight_id`) REFERENCES `flights` (`flight_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `diversions`
CREATE TABLE `diversions` (
  `flight_id` int(11) NOT NULL,
  `diversion_airline_id` int(11) NOT NULL,
  PRIMARY KEY (`flight_id`),
  FOREIGN KEY (`flight_id`) REFERENCES `flights` (`flight_id`),
  FOREIGN KEY (`diversion_airline_id`) REFERENCES `airlines` (`airline_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
