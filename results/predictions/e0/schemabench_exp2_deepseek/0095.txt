-- Schema for database: `carcinogenicity_prediction`

-- Table: `molecule`

CREATE TABLE `molecule` (
  `molecule_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_name` varchar(255) NOT NULL,
  `carcinogenicity_result` char(6) NOT NULL,
  PRIMARY KEY (`molecule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `atom`

CREATE TABLE `atom` (
  `atom_id` int(11) NOT NULL AUTO_INCREMENT,
  `atom_type` varchar(255) NOT NULL,
  `atom_charge` int(11) NOT NULL,
  `atom_designation` varchar(255) NOT NULL,
  `molecule_id` int(11) NOT NULL,
  PRIMARY KEY (`atom_id`),
  FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `sbond_1`
