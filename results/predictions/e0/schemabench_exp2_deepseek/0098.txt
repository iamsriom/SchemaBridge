-- Schema for database: `chess_prediction_system`

-- For table: `players`
CREATE TABLE `players` (
  `player_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `rating` int(11) NOT NULL,
  PRIMARY KEY (`player_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `events`
CREATE TABLE `events` (
  `event_id` int(11) NOT NULL AUTO_INCREMENT,
  `event_name` varchar(255) NOT NULL,
  `event_date` datetime NOT NULL,
  PRIMARY KEY (`event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `games`
CREATE TABLE `games` (
  `game_id` int(11) NOT NULL AUTO_INCREMENT,
  `player1_id` int(11) NOT NULL,
  `player2_id` int(11) NOT NULL,
  `event_id` int(11) NOT NULL,
  `result` enum('Player 1','Player 2','Draw') NOT NULL,
  PRIMARY KEY (`game_id`),
  FOREIGN KEY (`player1_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`player2_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`event_id`) REFERENCES `events` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `openings`
CREATE TABLE `openings` (
  `opening_id` int(11) NOT NULL AUTO_INCREMENT,
  `opening_name` varchar(255) NOT NULL,
  `description` text NOT NULL,
  PRIMARY KEY (`opening_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `game_openings`
CREATE TABLE `game_openings` (
  `game_id` int(11) NOT NULL,
  `opening_id` int(11) NOT NULL,
  `move_sequence` text NOT NULL,
  PRIMARY KEY (`game_id`,`opening_id`),
  FOREIGN KEY (`game_id`) REFERENCES `games` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`opening_id`) REFERENCES `openings` (`opening_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `predictions`
CREATE TABLE `predictions` (
  `game_id` int(11) NOT NULL,
  `predicted_result` enum('Player 1','Player 2','Draw') NOT NULL,
  PRIMARY KEY (`game_id`),
  FOREIGN KEY (`game_id`) REFERENCES `games` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
