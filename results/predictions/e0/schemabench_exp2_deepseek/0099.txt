-- Schema for database: `citeSeer`

-- For table: `publications`
CREATE TABLE `publications` (
  `publication_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `abstract` text,
  `authors` varchar(255) NOT NULL,
  `year` int(11) NOT NULL,
  `journal` varchar(255),
  `category` varchar(255) NOT NULL,
  PRIMARY KEY (`publication_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `citations`
CREATE TABLE `citations` (
  `citing_id` int(11) NOT NULL,
  `cited_id` int(11) NOT NULL,
  PRIMARY KEY (`citing_id`, `cited_id`),
  FOREIGN KEY (`citing_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`cited_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `word_vectors`
CREATE TABLE `word_vectors` (
  `publication_id` int(11) NOT NULL,
  `word_vector` longtext NOT NULL,
  PRIMARY KEY (`publication_id`),
  FOREIGN KEY (`publication_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
