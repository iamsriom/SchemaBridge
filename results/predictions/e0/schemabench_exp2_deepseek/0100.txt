-- Schema for database: `consumer_expenditure_survey`

-- Table: `expenditure`
CREATE TABLE `expenditure` (
  `respondent_id` int(11) NOT NULL,
  `household_id` int(11) NOT NULL,
  `expenditure_amount` int(11) NOT NULL,
  `date` datetime NOT NULL,
  PRIMARY KEY (`respondent_id`),
  FOREIGN KEY (`household_id`) REFERENCES `household`(`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `income`
CREATE TABLE `income` (
  `respondent_id` int(11) NOT NULL,
  `income_amount` int(11) NOT NULL,
  `date` datetime NOT NULL,
  PRIMARY KEY (`respondent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `demographics`
CREATE TABLE `demographics` (
  `respondent_id` int(11) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` char(6) NOT NULL,
  `race` varchar(255) NOT NULL,
  `education` varchar(255) NOT NULL,
  `income_level` varchar(255) NOT NULL,
  PRIMARY KEY (`respondent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `household`
CREATE TABLE `household` (
  `household_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `size` int(11) NOT NULL,
  PRIMARY KEY (`household_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
