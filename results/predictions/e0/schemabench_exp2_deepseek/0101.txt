-- Schema for database: `forest_area_prediction`

-- For table: `country_info`
CREATE TABLE `country_info` (
  `country_id` int(11) NOT NULL,
  `country_name` varchar(255) NOT NULL,
  `land_area` int(11) NOT NULL,
  PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `historical_data`
CREATE TABLE `historical_data` (
  `country_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `forest_area_percentage` decimal(10,2) NOT NULL,
  PRIMARY KEY (`country_id`,`year`),
  FOREIGN KEY (`country_id`) REFERENCES `country_info` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `prediction_data`
CREATE TABLE `prediction_data` (
  `country_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `forest_area_percentage` decimal(10,2) NOT NULL,
  PRIMARY KEY (`country_id`,`year`),
  FOREIGN KEY (`country_id`) REFERENCES `country_info` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
