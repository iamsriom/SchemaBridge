-- Schema for database: `schema_bench_0105`

-- For table: `sentences`
CREATE TABLE `sentences` (
  `sentence_id` int(11) NOT NULL AUTO_INCREMENT,
  `sentence` varchar(255) NOT NULL,
  `classification` varchar(255) NOT NULL,
  `created_at` datetime DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`sentence_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `terms`
CREATE TABLE `terms` (
  `term_id` int(11) NOT NULL AUTO_INCREMENT,
  `term` varchar(255) NOT NULL,
  `created_at` datetime DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`term_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `sentence_terms`
CREATE TABLE `sentence_terms` (
  `sentence_id` int(11) NOT NULL,
  `term_id` int(11) NOT NULL,
  `created_at` datetime DEFAULT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`sentence_id`,`term_id`),
  FOREIGN KEY (`sentence_id`) REFERENCES `sentences` (`sentence_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`term_id`) REFERENCES `terms` (`term_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
