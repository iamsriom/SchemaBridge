CREATE TABLE `go_1k` (
  `id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `retailer_id` int(11) NOT NULL,
  `method_id` int(11) NOT NULL,
  `date` date NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `product_id` (`product_id`),
  KEY `retailer_id` (`retailer_id`),
  KEY `method_id` (`method_id`),
  CONSTRAINT `go_1k_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `go_products` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_1k_ibfk_2` FOREIGN KEY (`retailer_id`) REFERENCES `go_retailers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_1k_ibfk_3` FOREIGN KEY (`method_id`) REFERENCES `go_methods` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_daily_sales` (
  `date` date NOT NULL,

  PRIMARY KEY (`date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_methods` (
  `id` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_products` (
  `id` int(11) NOT NULL,
