-- Schema for database: `grant_prediction`

-- For table: `grants`
CREATE TABLE `grants` (
  `grant_id` int(11) NOT NULL AUTO_INCREMENT,
  `award_amount` int(11) NOT NULL,
  `investigator_id` int(11) NOT NULL,
  `institution_id` int(11) NOT NULL,
  `program_element_id` int(11) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`grant_id`),
  KEY `investigator_id` (`investigator_id`),
  KEY `institution_id` (`institution_id`),
  KEY `program_element_id` (`program_element_id`),
  CONSTRAINT `grants_ibfk_1` FOREIGN KEY (`investigator_id`) REFERENCES `investigators` (`investigator_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grants_ibfk_2` FOREIGN KEY (`institution_id`) REFERENCES `institutions` (`institution_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grants_ibfk_3` FOREIGN KEY (`program_element_id`) REFERENCES `program_elements` (`program_element_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `investigators`
CREATE TABLE `investigators` (
  `investigator_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`investigator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `institutions`
CREATE TABLE `institutions` (
  `institution_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`institution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `program_elements`
CREATE TABLE `program_elements` (
  `program_element_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`program_element_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
