-- Schema for database: `hockey_db`

-- Table: `Players`
CREATE TABLE `Players` (
  `player_id` int(11) NOT NULL,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `birth_date` date NOT NULL,
  `position` varchar(255) NOT NULL,
  `height` int(11),
  `weight` int(11),
  `nationality` varchar(255),
  `draft_year` int(11),
  `draft_round` int(11),
  `draft_pick` int(11),
  PRIMARY KEY (`player_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `Coaches`
CREATE TABLE `Coaches` (
  `coach_id` int(11) NOT NULL,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date,
  `team_id` int(11) NOT NULL,
  PRIMARY KEY (`coach_id`),
  FOREIGN KEY (`team_id`) REFERENCES `Teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `Teams`
CREATE TABLE `Teams` (
  `team_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `league` varchar(255) NOT NULL,
  `season` int(11) NOT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `AwardsPlayers`
CREATE TABLE `AwardsPlayers` (
  `award_id` int(11) NOT NULL,
  `player_id` int(11) NOT NULL,
  `award_name` varchar(255) NOT NULL,
  `year` int(11) NOT NULL,
  PRIMARY KEY (`award_id`),
  FOREIGN KEY (`player_id`) REFERENCES `Players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `AwardsCoaches`
CREATE TABLE `AwardsCoaches` (
  `award_id` int(11) NOT NULL,
  `coach_id` int(11) NOT NULL,
  `award_name` varchar(255) NOT NULL,
  `year` int(11) NOT NULL,
  PRIMARY KEY (`award_id`),
  FOREIGN KEY (`coach_id`) REFERENCES `Coaches` (`coach_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `TeamsHalf`
CREATE TABLE `TeamsHalf` (
  `team_id` int(11) NOT NULL,
  `season` int(11) NOT NULL,
  `wins` int(11) NOT NULL,
  `losses` int(11) NOT NULL,
  `ot_losses` int(11) NOT NULL,
  `points_for` int(11) NOT NULL,
  `points_against` int(11) NOT NULL,
  PRIMARY KEY (`team_id`, `season`),
  FOREIGN KEY (`team_id`) REFERENCES `Teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `TeamsPost`
CREATE TABLE `TeamsPost` (
  `team_id` int(11) NOT NULL,
  `season` int(11) NOT NULL,
  `wins` int(11) NOT NULL,
  `losses` int(11) NOT NULL,
  `ot_losses` int(11) NOT NULL,
  `points_for` int(11) NOT NULL,
  `points_against` int(11) NOT NULL,
  PRIMARY KEY (`team_id`, `season`),
  FOREIGN KEY (`team_id`) REFERENCES `Teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
