-- Schema for database: `bulgarian_court_decisions`

-- For table: `legal_acts`
CREATE TABLE `legal_acts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `court_name` varchar(255) NOT NULL,
  `case_number` varchar(255) NOT NULL,
  `judge` varchar(255) NOT NULL,
  `decision_date` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `legal_act_link`

  `id` int(11) NOT NULL AUTO_INCREMENT,

  `legal_act_id_2` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`legal_act_id_1`) REFERENCES `legal_acts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`legal_act_id_2`) REFERENCES `legal_acts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `legal_act_people`
CREATE TABLE `legal_act_people` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `role` varchar(255) NOT NULL,
  `legal_act_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`legal_act_id`) REFERENCES `legal_acts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `scrape_fix`
