-- Schema for database: `chem_db`

-- Table: `molecules`
CREATE TABLE `molecules` (
  `molecule_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text,
  PRIMARY KEY (`molecule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `atoms`
CREATE TABLE `atoms` (
  `atom_id` int(11) NOT NULL AUTO_INCREMENT,
  `symbol` char(3) NOT NULL,
  `element_type` varchar(255) NOT NULL,
  `charge` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`atom_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `bonds`
CREATE TABLE `bonds` (
  `bond_id` int(11) NOT NULL AUTO_INCREMENT,
  `atom1_id` int(11) NOT NULL,
  `atom2_id` int(11) NOT NULL,
  `bond_type` varchar(255) NOT NULL,
  PRIMARY KEY (`bond_id`),
  FOREIGN KEY (`atom1_id`) REFERENCES `atoms` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`atom2_id`) REFERENCES `atoms` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `compounds`
CREATE TABLE `compounds` (
  `compound_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `activity` float NOT NULL,
  `logP` float NOT NULL,
  `LUMO` float NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`compound_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `compound_molecules`
CREATE TABLE `compound_molecules` (
  `compound_id` int(11) NOT NULL,
  `molecule_id` int(11) NOT NULL,
  PRIMARY KEY (`compound_id`, `molecule_id`),
  FOREIGN KEY (`compound_id`) REFERENCES `compounds` (`compound_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `compound_atoms`
CREATE TABLE `compound_atoms` (
  `compound_id` int(11) NOT NULL,
  `atom_id` int(11) NOT NULL,
  PRIMARY KEY (`compound_id`, `atom_id`),
  FOREIGN KEY (`compound_id`) REFERENCES `compounds` (`compound_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`atom_id`) REFERENCES `atoms` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
