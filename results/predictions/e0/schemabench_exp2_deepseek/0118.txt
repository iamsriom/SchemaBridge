-- For table: `countries`
CREATE TABLE `countries` (
  `country_id` int(11) NOT NULL,
  `country_name` varchar(255) NOT NULL,
  PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `country_relations`
CREATE TABLE `country_relations` (
  `relation_id` int(11) NOT NULL,
  `country_id1` int(11) NOT NULL,
  `country_id2` int(11) NOT NULL,
  `relation_type` varchar(255) NOT NULL,
  PRIMARY KEY (`relation_id`),
  FOREIGN KEY (`country_id1`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`country_id2`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `country_statistics`
CREATE TABLE `country_statistics` (
  `statistic_id` int(11) NOT NULL,
  `country_id` int(11) NOT NULL,
  `population` int(11) NOT NULL,
  `gnp` decimal(10,2) NOT NULL,
  `exports` decimal(10,2) NOT NULL,
  `imports` decimal(10,2) NOT NULL,
  `literacy_rate` decimal(4,2) NOT NULL,
  `unemployment_rate` decimal(4,2) NOT NULL,
  PRIMARY KEY (`statistic_id`),
  FOREIGN KEY (`country_id`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `political_metrics`
CREATE TABLE `political_metrics` (
  `metric_id` int(11) NOT NULL,
  `country_id` int(11) NOT NULL,
  `metric_type` varchar(255) NOT NULL,
  `metric_value` decimal(10,2) NOT NULL,
  PRIMARY KEY (`metric_id`),
  FOREIGN KEY (`country_id`) REFERENCES `countries` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
