-- Schema for database: `ncaa_tournament_analysis`

-- Table: `tournament`
CREATE TABLE `tournament` (
  `tournament_id` int(11) NOT NULL AUTO_INCREMENT,
  `tournament_name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`tournament_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `team`
CREATE TABLE `team` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_name` varchar(255) NOT NULL,
  `region` varchar(255) NOT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `match`
CREATE TABLE `match` (
  `match_id` int(11) NOT NULL AUTO_INCREMENT,
  `tournament_id` int(11) NOT NULL,
  `team_id_1` int(11) NOT NULL,
  `team_id_2` int(11) NOT NULL,
  `match_date` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `score_1` int(11) NOT NULL,
  `score_2` int(11) NOT NULL,
  `number_of_overtimes` int(11) NOT NULL,
  PRIMARY KEY (`match_id`),
  FOREIGN KEY (`tournament_id`) REFERENCES `tournament` (`tournament_id`),
  FOREIGN KEY (`team_id_1`) REFERENCES `team` (`team_id`),
  FOREIGN KEY (`team_id_2`) REFERENCES `team` (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `performance`
CREATE TABLE `performance` (
  `team_id` int(11) NOT NULL,
  `tournament_id` int(11) NOT NULL,
  `metric_1` int(11) NOT NULL,
  `metric_2` int(11) NOT NULL,
  `metric_3` int(11) NOT NULL,
  PRIMARY KEY (`team_id`,`tournament_id`),
  FOREIGN KEY (`team_id`) REFERENCES `team` (`team_id`),
  FOREIGN KEY (`tournament_id`) REFERENCES `tournament` (`tournament_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `seeds`
CREATE TABLE `seeds` (
  `tournament_id` int(11) NOT NULL,
  `team_id` int(11) NOT NULL,
  `seed` int(11) NOT NULL,
  PRIMARY KEY (`tournament_id`,`team_id`),
  FOREIGN KEY (`tournament_id`) REFERENCES `tournament` (`tournament_id`),
  FOREIGN KEY (`team_id`) REFERENCES `team` (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `slots`
CREATE TABLE `slots` (
  `tournament_id` int(11) NOT NULL,
  `team_id` int(11) NOT NULL,
  `slot` int(11) NOT NULL,
  PRIMARY KEY (`tournament_id`,`team_id`),
  FOREIGN KEY (`tournament_id`) REFERENCES `tournament` (`tournament_id`),
  FOREIGN KEY (`team_id`) REFERENCES `team` (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `prediction`
CREATE TABLE `prediction` (
  `match_id` int(11) NOT NULL,
  `prediction_1` int(11) NOT NULL,
  `prediction_2` int(11) NOT NULL,
  PRIMARY KEY (`match_id`),
  FOREIGN KEY (`match_id`) REFERENCES `match` (`match_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
