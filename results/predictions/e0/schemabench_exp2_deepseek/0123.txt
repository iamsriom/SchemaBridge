-- Schema for database: `schemabench_0123`

-- For table: `molecules`
CREATE TABLE `molecules` (
  `molecule_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `formula` varchar(255) NOT NULL,
  `smiles` varchar(255) NOT NULL,
  `inchi` varchar(255) NOT NULL,
  PRIMARY KEY (`molecule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `properties`
CREATE TABLE `properties` (
  `property_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_id` int(11) NOT NULL,
  `flexibility` float NOT NULL,
  `polarizability` float NOT NULL,
  `sigma_value` float NOT NULL,
  PRIMARY KEY (`property_id`),
  FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `activities`
CREATE TABLE `activities` (
  `activity_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_id` int(11) NOT NULL,
  `inhibition_activity` float NOT NULL,
  PRIMARY KEY (`activity_id`),
  FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
