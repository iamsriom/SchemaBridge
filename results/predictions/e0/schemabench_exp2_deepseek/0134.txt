CREATE TABLE `E_ACCOUNT_PERMISSION` (
  `USER_ID` int(11) NOT NULL,
  `PERMISSION_ID` int(11) NOT NULL,
  PRIMARY KEY (`USER_ID`,`PERMISSION_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_CUSTOMER` (
  `CUSTOMER_ID` int(11) NOT NULL PRIMARY KEY,
  `NAME` varchar(255) NOT NULL,
  `ADDRESS` varchar(255),
  `EMAIL` varchar(255),
  `PHONE` char(10)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_CUSTOMER_ACCOUNT` (
  `ACCOUNT_ID` int(11) NOT NULL PRIMARY KEY,
  `CUSTOMER_ID` int(11) NOT NULL,
  `BALANCE` decimal(10,2) NOT NULL DEFAULT 0.00,
  `LAST_UPDATED` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`CUSTOMER_ID`) REFERENCES `E_CUSTOMER` (`CUSTOMER_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_CASH_TRANSACTION` (
  `TRANSACTION_ID` int(11) NOT NULL PRIMARY KEY,
  `ACCOUNT_ID` int(11) NOT NULL,
  `AMOUNT` decimal(10,2) NOT NULL,
  `TRANSACTION_TYPE` varchar(255) NOT NULL,
  `TRANSACTION_DATE` datetime NOT NULL,
  FOREIGN KEY (`ACCOUNT_ID`) REFERENCES `E_CUSTOMER_ACCOUNT` (`ACCOUNT_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_DAILY_MARKET` (
  `MARKET_DATE` date NOT NULL PRIMARY KEY,
  `SECURITY_ID` int(11) NOT NULL,
  `OPEN_PRICE` decimal(10,2) NOT NULL,
  `CLOSE_PRICE` decimal(10,2) NOT NULL,
  `VOLUME` int(11) NOT NULL,
  FOREIGN KEY (`SECURITY_ID`) REFERENCES `E_SECURITY` (`SECURITY_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_FINANCIAL` (
  `SECURITY_ID` int(11) NOT NULL PRIMARY KEY,
  `EPS` decimal(10,2) NOT NULL,
  `DIVIDEND` decimal(10,2) NOT NULL,
  `PE_RATIO` decimal(10,2) NOT NULL,
  FOREIGN KEY (`SECURITY_ID`) REFERENCES `E_SECURITY` (`SECURITY_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_SECURITY` (
  `SECURITY_ID` int(11) NOT NULL PRIMARY KEY,
  `SECURITY_NAME` varchar(255) NOT NULL,
  `EXCHANGE_ID` int(11) NOT NULL,
  FOREIGN KEY (`EXCHANGE_ID`) REFERENCES `E_EXCHANGE` (`EXCHANGE_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_EXCHANGE` (
  `EXCHANGE_ID` int(11) NOT NULL PRIMARY KEY,
  `EXCHANGE_NAME` varchar(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_TRADE` (
  `TRADE_ID` int(11) NOT NULL PRIMARY KEY,
  `SECURITY_ID` int(11) NOT NULL,
  `TRADE_DATE` datetime NOT NULL,
  `TRADE_TYPE` varchar(255) NOT NULL,
  `SHARES` int(11) NOT NULL,
  `PRICE` decimal(10,2) NOT NULL,
  FOREIGN KEY (`SECURITY_ID`) REFERENCES `E_SECURITY` (`SECURITY_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_SETTLEMENT` (
  `SETTLEMENT_ID` int(11) NOT NULL PRIMARY KEY,
  `TRADE_ID` int(11) NOT NULL,
  `SETTLEMENT_DATE` datetime NOT NULL,
  `SETTLEMENT_TYPE` varchar(255) NOT NULL,
  `FEE` decimal(10,2) NOT NULL,
  FOREIGN KEY (`TRADE_ID`) REFERENCES `E_TRADE` (`TRADE_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_TRADE_HISTORY` (
  `HISTORY_ID` int(11) NOT NULL PRIMARY KEY,
  `TRADE_ID` int(11) NOT NULL,
  `HISTORY_DATE` datetime NOT NULL,
  `HISTORY_TYPE` varchar(255) NOT NULL,
  `HISTORY_DESCRIPTION` varchar(255) NOT NULL,
  FOREIGN KEY (`TRADE_ID`) REFERENCES `E_TRADE` (`TRADE_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_HOLDING_HISTORY` (
  `HISTORY_ID` int(11) NOT NULL PRIMARY KEY,
  `ACCOUNT_ID` int(11) NOT NULL,
  `HISTORY_DATE` datetime NOT NULL,
  `HISTORY_TYPE` varchar(255) NOT NULL,
  `HISTORY_DESCRIPTION` varchar(255) NOT NULL,
  FOREIGN KEY (`ACCOUNT_ID`) REFERENCES `E_CUSTOMER_ACCOUNT` (`ACCOUNT_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_INDUSTRY` (
  `INDUSTRY_ID` int(11) NOT NULL PRIMARY KEY,
  `INDUSTRY_NAME` varchar(255) NOT NULL,
  `INDUSTRY_PERFORMANCE` decimal(10,2) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `E_COMPANY` (
  `COMPANY_ID` int(11) NOT NULL PRIMARY KEY,
  `COMPANY_NAME` varchar(255) NOT NULL,
  `INDUSTRY_ID` int(11) NOT NULL,
  `COMPANY_PERFORMANCE` decimal(10,2) NOT NULL,
  FOREIGN KEY (`INDUSTRY_ID`) REFERENCES `E_INDUSTRY` (`INDUSTRY_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
