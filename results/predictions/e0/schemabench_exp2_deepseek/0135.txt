-- Schema for database: `schemabench_0135`

-- For table: `trains`
CREATE TABLE `trains` (
  `train_id` int(11) NOT NULL AUTO_INCREMENT,
  `direction` char(6) NOT NULL,
  PRIMARY KEY (`train_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `cars`
CREATE TABLE `cars` (
  `car_id` int(11) NOT NULL AUTO_INCREMENT,
  `train_id` int(11) NOT NULL,
  `position` int(11) NOT NULL,
  `shape` varchar(255) NOT NULL,
  `length` int(11) NOT NULL,
  `sides` int(11) NOT NULL,
  `roof_type` varchar(255) NOT NULL,
  `wheels` int(11) NOT NULL,
  PRIMARY KEY (`car_id`),
  FOREIGN KEY (`train_id`) REFERENCES `trains` (`train_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `loads`
CREATE TABLE `loads` (
  `load_id` int(11) NOT NULL AUTO_INCREMENT,
  `car_id` int(11) NOT NULL,
  `shape` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`load_id`),
  FOREIGN KEY (`car_id`) REFERENCES `cars` (`car_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
