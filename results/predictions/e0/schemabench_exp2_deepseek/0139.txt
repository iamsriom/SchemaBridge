-- Schema for database: `visual_genome`

-- For table: `ATT_CLASSES`
CREATE TABLE `ATT_CLASSES` (
  `ATTRIBUTE_ID` int(11) NOT NULL AUTO_INCREMENT,
  `ATTRIBUTE_NAME` varchar(255) NOT NULL,
  PRIMARY KEY (`ATTRIBUTE_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `OBJ_CLASSES`
CREATE TABLE `OBJ_CLASSES` (
  `OBJECT_ID` int(11) NOT NULL AUTO_INCREMENT,
  `OBJECT_NAME` varchar(255) NOT NULL,
  PRIMARY KEY (`OBJECT_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `IMG_OBJ`
CREATE TABLE `IMG_OBJ` (
  `IMAGE_ID` int(11) NOT NULL,
  `OBJECT_ID` int(11) NOT NULL,
  `X_COORDINATE` int(11) NOT NULL,
  `Y_COORDINATE` int(11) NOT NULL,
  `WIDTH` int(11) NOT NULL,
  `HEIGHT` int(11) NOT NULL,
  PRIMARY KEY (`IMAGE_ID`,`OBJECT_ID`),
  FOREIGN KEY (`IMAGE_ID`) REFERENCES `IMAGES`(`IMAGE_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`OBJECT_ID`) REFERENCES `OBJ_CLASSES`(`OBJECT_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `IMG_OBJ_ATT`
CREATE TABLE `IMG_OBJ_ATT` (
  `IMAGE_ID` int(11) NOT NULL,
  `OBJECT_ID` int(11) NOT NULL,
  `ATTRIBUTE_ID` int(11) NOT NULL,
  PRIMARY KEY (`IMAGE_ID`,`OBJECT_ID`,`ATTRIBUTE_ID`),
  FOREIGN KEY (`IMAGE_ID`) REFERENCES `IMAGES`(`IMAGE_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`OBJECT_ID`) REFERENCES `OBJ_CLASSES`(`OBJECT_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`ATTRIBUTE_ID`) REFERENCES `ATT_CLASSES`(`ATTRIBUTE_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `PRED_CLASSES`
CREATE TABLE `PRED_CLASSES` (
  `RELATION_ID` int(11) NOT NULL AUTO_INCREMENT,
  `RELATION_NAME` varchar(255) NOT NULL,
  PRIMARY KEY (`RELATION_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `IMG_REL`
CREATE TABLE `IMG_REL` (
  `IMAGE_ID` int(11) NOT NULL,
  `OBJECT_ID_1` int(11) NOT NULL,
  `OBJECT_ID_2` int(11) NOT NULL,
  `RELATION_ID` int(11) NOT NULL,
  PRIMARY KEY (`IMAGE_ID`,`OBJECT_ID_1`,`OBJECT_ID_2`,`RELATION_ID`),
  FOREIGN KEY (`IMAGE_ID`) REFERENCES `IMAGES`(`IMAGE_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`OBJECT_ID_1`) REFERENCES `OBJ_CLASSES`(`OBJECT_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`OBJECT_ID_2`) REFERENCES `OBJ_CLASSES`(`OBJECT_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`RELATION_ID`) REFERENCES `PRED_CLASSES`(`RELATION_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
