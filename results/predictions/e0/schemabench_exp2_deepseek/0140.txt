-- Schema for database: `voc_database`

-- Table: `voc_voyages`
CREATE TABLE `voc_voyages` (
  `voyage_id` int(11) NOT NULL AUTO_INCREMENT,
  `vessel_name` varchar(255) NOT NULL,
  `master` varchar(255) NOT NULL,
  `tonnage` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  `construction_details` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`voyage_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `voc_personnel`
CREATE TABLE `voc_personnel` (
  `personnel_id` int(11) NOT NULL AUTO_INCREMENT,
  `voyage_id` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  `status_at_departure` varchar(255) NOT NULL,
  `status_at_arrival` varchar(255) NOT NULL,
  `casualties` int(11) NOT NULL,
  PRIMARY KEY (`personnel_id`),
  FOREIGN KEY (`voyage_id`) REFERENCES `voc_voyages` (`voyage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `voc_invoices`
CREATE TABLE `voc_invoices` (
  `invoice_id` int(11) NOT NULL AUTO_INCREMENT,
  `voyage_id` int(11) NOT NULL,
  `chamber_details` varchar(255) NOT NULL,
  PRIMARY KEY (`invoice_id`),
  FOREIGN KEY (`voyage_id`) REFERENCES `voc_voyages` (`voyage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
