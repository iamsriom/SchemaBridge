-- Schema for database: `schmabench_0142`

-- For table: `publications`
CREATE TABLE `publications` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `abstract` text,
  `authors` varchar(255) NOT NULL,
  `class_label` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `citations`
CREATE TABLE `citations` (
  `citing_paper_id` int(11) NOT NULL,
  `cited_paper_id` int(11) NOT NULL,
  PRIMARY KEY (`citing_paper_id`, `cited_paper_id`),
  FOREIGN KEY (`citing_paper_id`) REFERENCES `publications` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`cited_paper_id`) REFERENCES `publications` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `word_vectors`
CREATE TABLE `word_vectors` (
  `paper_id` int(11) NOT NULL,
  `word` varchar(255) NOT NULL,
  `presence` char(1) NOT NULL,
  PRIMARY KEY (`paper_id`, `word`),
  FOREIGN KEY (`paper_id`) REFERENCES `publications` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
