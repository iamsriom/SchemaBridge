-- Schema for database: `BetterFiori`

-- For table: `INSTALLATIONS`
CREATE TABLE `INSTALLATIONS` (
  `INSTALLATION_ID` int(11) NOT NULL AUTO_INCREMENT,
  `ENVIRONMENT` varchar(255) NOT NULL,
  `INSTALLATION_DATE` datetime NOT NULL,
  PRIMARY KEY (`INSTALLATION_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `LABOR_CODES`
CREATE TABLE `LABOR_CODES` (
  `LABOR_CODE_ID` int(11) NOT NULL AUTO_INCREMENT,
  `LABOR_CODE` varchar(255) NOT NULL,
  PRIMARY KEY (`LABOR_CODE_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `BILLING_CODES`
CREATE TABLE `BILLING_CODES` (
  `BILLING_CODE_ID` int(11) NOT NULL AUTO_INCREMENT,
  `BILLING_CODE` varchar(255) NOT NULL,
  PRIMARY KEY (`BILLING_CODE_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `TICKETS`
CREATE TABLE `TICKETS` (
  `TICKET_ID` int(11) NOT NULL AUTO_INCREMENT,
  `CREATION_DATE` datetime NOT NULL,
  `LABOR_CODE_ID` int(11) NOT NULL,
  `WORK_HOURS` int(11) NOT NULL,
  PRIMARY KEY (`TICKET_ID`),
  FOREIGN KEY (`LABOR_CODE_ID`) REFERENCES `LABOR_CODES` (`LABOR_CODE_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `REPORTS`
CREATE TABLE `REPORTS` (
  `REPORT_ID` int(11) NOT NULL AUTO_INCREMENT,
  `CREATION_DATE` datetime NOT NULL,
  `TICKET_ID` int(11) NOT NULL,
  PRIMARY KEY (`REPORT_ID`),
  FOREIGN KEY (`TICKET_ID`) REFERENCES `TICKETS` (`TICKET_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `SHELL`
CREATE TABLE `SHELL` (
  `SHELL_ID` int(11) NOT NULL AUTO_INCREMENT,
  `THEME` varchar(255) NOT NULL,
  `MENU_ITEM` varchar(255) NOT NULL,
  PRIMARY KEY (`SHELL_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
