-- Schema for database: `stock_trading_system`

-- Table: `company`
CREATE TABLE `company` (
  `company_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `headquarters_address` varchar(255) NOT NULL,
  `established_date` datetime NOT NULL,
  PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `stock`
CREATE TABLE `stock` (
  `stock_id` int(11) NOT NULL AUTO_INCREMENT,
  `stock_code` varchar(255) NOT NULL,
  `symbol` varchar(255) NOT NULL,
  `last_trading_price` decimal(10,2) NOT NULL,
  `daily_volume` int(11) NOT NULL,
  `company_id` int(11) NOT NULL,
  PRIMARY KEY (`stock_id`),
  FOREIGN KEY (`company_id`) REFERENCES `company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `trader`
CREATE TABLE `trader` (
  `trader_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`trader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `account`
CREATE TABLE `account` (
  `account_id` int(11) NOT NULL AUTO_INCREMENT,
  `securities` decimal(10,2) NOT NULL,
  `cash` decimal(10,2) NOT NULL,
  `trader_id` int(11) NOT NULL,
  PRIMARY KEY (`account_id`),
  FOREIGN KEY (`trader_id`) REFERENCES `trader` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `broker`
CREATE TABLE `broker` (
  `broker_id` int(11) NOT NULL AUTO_INCREMENT,
  `contact_details` varchar(255) NOT NULL,
  `state` varchar(255) NOT NULL,
  PRIMARY KEY (`broker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `transaction`
CREATE TABLE `transaction` (
  `transaction_id` int(11) NOT NULL AUTO_INCREMENT,
  `shares` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `commission` decimal(10,2) NOT NULL,
  `timestamp` datetime NOT NULL,
  `account_id` int(11) NOT NULL,
  `broker_id` int(11) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  FOREIGN KEY (`account_id`) REFERENCES `account` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`broker_id`) REFERENCES `broker` (`broker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
