CREATE TABLE `customer_management` (
  `customer_id` int(11) NOT NULL,
  `loyalty_score` int(11) DEFAULT NULL,
  `preferences` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product_tracking` (
  `product_id` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `promotion` varchar(255) DEFAULT NULL,
  `description` text,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_and_inventory_analysis` (
  `transaction_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `date` date NOT NULL,
  `quantity` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  KEY `product_id` (`product_id`),
  KEY `customer_id` (`customer_id`),
  CONSTRAINT `sales_and_inventory_analysis_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `product_tracking` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_and_inventory_analysis_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer_management` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_outlet_management` (
  `outlet_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `size` int(11) NOT NULL,
  `management_details` text,
  PRIMARY KEY (`outlet_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff_management` (
  `staff_id` int(11) NOT NULL,
  `role` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `start_date` date NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_handling` (
  `transaction_id` int(11) NOT NULL,
  `outlet_id` int(11) NOT NULL,
  `date` date NOT NULL,
  `in_store_sales` decimal(10,2) DEFAULT NULL,
  `online_sales` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`transaction_id`),
  KEY `outlet_id` (`outlet_id`),
  CONSTRAINT `transaction_handling_ibfk_1` FOREIGN KEY (`outlet_id`) REFERENCES `sales_outlet_management` (`outlet_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
