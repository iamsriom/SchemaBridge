-- Schema for database: `yurt_reservation_system`

-- Table: `customers`

CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `yurts`

CREATE TABLE `yurts` (
  `yurt_id` int(11) NOT NULL AUTO_INCREMENT,
  `number_of_beds` int(11) NOT NULL,
  `maximum_occupancy` int(11) NOT NULL,
  `smoking_preference` varchar(255) NOT NULL,
  `pet_friendliness` varchar(255) NOT NULL,
  `nightly_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`yurt_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `reservations`

CREATE TABLE `reservations` (
  `reservation_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `yurt_id` int(11) NOT NULL,
  `check_in_date` date NOT NULL,
  `check_out_date` date NOT NULL,
  PRIMARY KEY (`reservation_id`),
  KEY `customer_id` (`customer_id`),
  KEY `yurt_id` (`yurt_id`),
  CONSTRAINT `reservations_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reservations_ibfk_2` FOREIGN KEY (`yurt_id`) REFERENCES `yurts` (`yurt_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
