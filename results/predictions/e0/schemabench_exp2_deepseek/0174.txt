-- Schema for database: `healthcare_management_system`

-- Table: `physicians`
CREATE TABLE `physicians` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `ssn` char(9) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `departments`
CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `head` int(11) NOT NULL,
  PRIMARY KEY (`department_id`),
  FOREIGN KEY (`head`) REFERENCES `physicians` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `medical_procedures`
CREATE TABLE `medical_procedures` (
  `code` char(5) NOT NULL,
  `name` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `patients`
CREATE TABLE `patients` (
  `ssn` char(9) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `insurance` varchar(255) NOT NULL,
  `primary_physician` int(11) NOT NULL,
  PRIMARY KEY (`ssn`),
  FOREIGN KEY (`primary_physician`) REFERENCES `physicians` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `nurses`
CREATE TABLE `nurses` (
  `employee_id` int(11) NOT NULL,
  `registration_info` varchar(255) NOT NULL,
  `department` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`),
  FOREIGN KEY (`department`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `appointments`
CREATE TABLE `appointments` (
  `appointment_id` int(11) NOT NULL AUTO_INCREMENT,
  `patient_ssn` char(9) NOT NULL,
  `nurse_id` int(11) NOT NULL,
  `physician_id` int(11) NOT NULL,
  `time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`appointment_id`),
  FOREIGN KEY (`patient_ssn`) REFERENCES `patients` (`ssn`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`nurse_id`) REFERENCES `nurses` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`physician_id`) REFERENCES `physicians` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `medications`
CREATE TABLE `medications` (
  `code` char(5) NOT NULL,
  `name` varchar(255) NOT NULL,
  `brand` varchar(255) NOT NULL,
  `description` text NOT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `rooms`
CREATE TABLE `rooms` (
  `room_id` int(11) NOT NULL,
  `block` int(11) NOT NULL,
  PRIMARY KEY (`room_id`),
  FOREIGN KEY (`block`) REFERENCES `facility_blocks` (`block_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `facility_blocks`
CREATE TABLE `facility_blocks` (
  `block_id` int(11) NOT NULL,
  `details` text NOT NULL,
  PRIMARY KEY (`block_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
