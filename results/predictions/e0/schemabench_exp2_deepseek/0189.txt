-- Schema for database: `schema_bench_0189`

-- For table: `artists`
CREATE TABLE `artists` (
  `artist_id` int(11) NOT NULL,
  `artist_name` varchar(255) NOT NULL,
  `artist_genre` varchar(255) NOT NULL,
  PRIMARY KEY (`artist_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `songs`
CREATE TABLE `songs` (
  `song_id` int(11) NOT NULL,
  `song_name` varchar(255) NOT NULL,
  `artist_id` int(11) NOT NULL,
  `album_id` int(11) NOT NULL,
  `peak_chart_position` int(11) NOT NULL,
  `weeks_on_chart` int(11) NOT NULL,
  `song_popularity` int(11) NOT NULL,
  PRIMARY KEY (`song_id`),
  FOREIGN KEY (`artist_id`) REFERENCES `artists` (`artist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `albums`
CREATE TABLE `albums` (
  `album_id` int(11) NOT NULL,
  `album_name` varchar(255) NOT NULL,
  `artist_id` int(11) NOT NULL,
  `album_release_date` date NOT NULL,
  `album_popularity` int(11) NOT NULL,
  PRIMARY KEY (`album_id`),
  FOREIGN KEY (`artist_id`) REFERENCES `artists` (`artist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `musical_attributes`
CREATE TABLE `musical_attributes` (
  `song_id` int(11) NOT NULL,
  `danceability` float NOT NULL,
  `energy` float NOT NULL,
  `loudness` float NOT NULL,
  `speechiness` float NOT NULL,
  `acousticness` float NOT NULL,
  `instrumentalness` float NOT NULL,
  `liveness` float NOT NULL,
  `valence` float NOT NULL,
  `tempo` float NOT NULL,
  PRIMARY KEY (`song_id`),
  FOREIGN KEY (`song_id`) REFERENCES `songs` (`song_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
