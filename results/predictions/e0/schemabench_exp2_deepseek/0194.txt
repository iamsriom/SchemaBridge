-- Schema for database: `danny's_diner`

-- For table: `sales`

CREATE TABLE `sales` (
  `customer_id` INT NOT NULL,
  `order_date` DATETIME NOT NULL,
  `product_id` INT NOT NULL,
  `quantity` INT NOT NULL,
  `order_id` INT NOT NULL AUTO_INCREMENT,
  PRIMARY KEY (`order_id`),
  FOREIGN KEY (`customer_id`) REFERENCES `menu` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`product_id`) REFERENCES `menu` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `menu`

CREATE TABLE `menu` (
  `product_id` INT NOT NULL AUTO_INCREMENT,
  `product_name` VARCHAR(255) NOT NULL,
  `price` DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `members`

CREATE TABLE `members` (
  `customer_id` INT NOT NULL AUTO_INCREMENT,
  `join_date` DATETIME NOT NULL,
  PRIMARY KEY (`customer_id`)
) EGINE=InnoDB DEFAULT CHARSET=utf8mb3;
