-- Schema for database: `pizza_runner`

-- For table: `runner`
CREATE TABLE `runner` (
  `runner_id` int(11) NOT NULL AUTO_INCREMENT,
  `registration_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`runner_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `customer`
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `pizza`
CREATE TABLE `pizza` (
  `pizza_id` int(11) NOT NULL AUTO_INCREMENT,
  `pizza_name` varchar(255) NOT NULL,
  PRIMARY KEY (`pizza_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `pizza_toppings`
CREATE TABLE `pizza_toppings` (
  `topping_id` int(11) NOT NULL AUTO_INCREMENT,
  `topping_name` varchar(255) NOT NULL,
  PRIMARY KEY (`topping_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `runner_pizza_orders`
CREATE TABLE `runner_pizza_orders` (
  `order_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) NOT NULL,
  `pizza_id` int(11) NOT NULL,
  `runner_id` int(11) NOT NULL,
  `order_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `pickup_time` datetime DEFAULT NULL,
  `distance` varchar(255) DEFAULT NULL,
  `duration` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  KEY `customer_id` (`customer_id`),
  KEY `pizza_id` (`pizza_id`),
  KEY `runner_id` (`runner_id`),
  CONSTRAINT `runner_pizza_orders_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `runner_pizza_orders_ibfk_2` FOREIGN KEY (`pizza_id`) REFERENCES `pizza` (`pizza_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `runner_pizza_orders_ibfk_3` FOREIGN KEY (`runner_id`) REFERENCES `runner` (`runner_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `pizza_toppings_relation`
CREATE TABLE `pizza_toppings_relation` (
  `relation_id` int(11) NOT NULL AUTO_INCREMENT,
  `pizza_id` int(11) NOT NULL,
  `topping_id` int(11) NOT NULL,
  PRIMARY KEY (`relation_id`),
  KEY `pizza_id` (`pizza_id`),
  KEY `topping_id` (`topping_id`),
  CONSTRAINT `pizza_toppings_relation_ibfk_1` FOREIGN KEY (`pizza_id`) REFERENCES `pizza` (`pizza_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `pizza_toppings_relation_ibfk_2` FOREIGN KEY (`topping_id`) REFERENCES `pizza_toppings` (`topping_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
