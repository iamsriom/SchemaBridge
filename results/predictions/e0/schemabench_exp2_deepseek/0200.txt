-- Schema for database: `fresh_segments`

-- For table: `clients`

CREATE TABLE `clients` (
  `client_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_name` varchar(255) NOT NULL,
  `industry` varchar(255) NOT NULL,
  `website` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone` char(10) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `ad_content`

CREATE TABLE `ad_content` (
  `ad_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_id` int(11) NOT NULL,
  `ad_content` text NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`ad_id`),
  FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `interest_metrics`

CREATE TABLE `interest_metrics` (
  `metric_id` int(11) NOT NULL AUTO_INCREMENT,
  `ad_id` int(11) NOT NULL,
  `interest_name` varchar(255) NOT NULL,
  `month` char(6) NOT NULL,
  `engagement_count` int(11) NOT NULL,
  PRIMARY KEY (`metric_id`),
  FOREIGN KEY (`ad_id`) REFERENCES `ad_content` (`ad_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `interest_map`

CREATE TABLE `interest_map` (
  `interest_id` int(11) NOT NULL AUTO_INCREMENT,
  `interest_name` varchar(255) NOT NULL,
  PRIMARY KEY (`interest_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `client_interest_map`

CREATE TABLE `client_interest_map` (
  `client_id` int(11) NOT NULL,
  `interest_id` int(11) NOT NULL,
  PRIMARY KEY (`client_id`,`interest_id`),
  FOREIGN KEY (`client_id`) REFERENCES `clients` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`interest_id`) REFERENCES `interest_map` (`interest_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
