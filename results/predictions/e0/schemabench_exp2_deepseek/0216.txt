-- Schema for database: `game_market_trends`

-- For table: `game_sales`
CREATE TABLE `game_sales` (
  `game_id` int(11) NOT NULL,
  `game_name` varchar(255) NOT NULL,
  `year` year NOT NULL,
  `sales_volume` int(11) NOT NULL,
  `critic_score` int(11) DEFAULT NULL,
  `user_score` int(11) DEFAULT NULL,
  PRIMARY KEY (`game_id`),
  UNIQUE KEY `game_name_year` (`game_name`,`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `critic_reviews`
CREATE TABLE `critic_reviews` (
  `review_id` int(11) NOT NULL AUTO_INCREMENT,
  `game_id` int(11) NOT NULL,
  `review_score` int(11) NOT NULL,
  PRIMARY KEY (`review_id`),
  KEY `game_id` (`game_id`),
  CONSTRAINT `critic_reviews_ibfk_1` FOREIGN KEY (`game_id`) REFERENCES `game_sales` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `user_reviews`
CREATE TABLE `user_reviews` (
  `review_id` int(11) NOT NULL AUTO_INCREMENT,
  `game_id` int(11) NOT NULL,
  `review_score` int(11) NOT NULL,
  PRIMARY KEY (`review_id`),
  KEY `game_id` (`game_id`),
  CONSTRAINT `user_reviews_ibfk_1` FOREIGN KEY (`game_id`) REFERENCES `game_sales` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
