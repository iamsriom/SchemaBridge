-- Schema for database: `schemabench_0218`

-- Table: `grampanchayat_management`
CREATE TABLE `grampanchayat_management` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `total_wards` int(11) NOT NULL,
  `taluka_name` varchar(255) NOT NULL,
  `total_villages` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `village_ward_management`
CREATE TABLE `village_ward_management` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `village_name` varchar(255) NOT NULL,
  `ward_name` varchar(255) NOT NULL,
  `representative_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`representative_id`) REFERENCES `ward_members`(`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `farmer_registration`
CREATE TABLE `farmer_registration` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aadh_no` char(12) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `machine_rental`
CREATE TABLE `machine_rental` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `mach_id` varchar(255) NOT NULL,
  `rent_duration` int(11) NOT NULL,
  `ward_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`ward_id`) REFERENCES `village_ward_management`(`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`member_id`) REFERENCES `ward_members`(`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `billing_payments`
CREATE TABLE `billing_payments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bill_id` varchar(255) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `payment_method` varchar(255) NOT NULL,
  `payment_date` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `ward_members`
CREATE TABLE `ward_members` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
