-- Schema for database: `schemabench_0221`

-- For table: `restaurants`

CREATE TABLE `restaurants` (
    `restaurant_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    `cuisine` VARCHAR(255) NOT NULL,
    `price_range` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`restaurant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `reviews`

CREATE TABLE `reviews` (
    `review_id` INT(11) NOT NULL,
    `restaurant_id` INT(11) NOT NULL,
    `review_text` TEXT NOT NULL,
    `rating` INT(11) NOT NULL,
    `date` DATETIME NOT NULL,
    PRIMARY KEY (`review_id`),
    FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `consumers`

CREATE TABLE `consumers` (
    `consumer_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `location` VARCHAR(255) NOT NULL,
    `preferred_cuisine` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`consumer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `preferences`

CREATE TABLE `preferences` (
    `consumer_id` INT(11) NOT NULL,
    `restaurant_id` INT(11) NOT NULL,
    `preference` INT(11) NOT NULL,
    PRIMARY KEY (`consumer_id`, `restaurant_id`),
    FOREIGN KEY (`consumer_id`) REFERENCES `consumers` (`consumer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (`restaurant_id`) REFERENCES `restaurants` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
