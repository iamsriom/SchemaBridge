-- Schema for database: `schemabench_0224`

-- For table: `employees`

CREATE TABLE `employees` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `hire_date` date NOT NULL,
  `title` varchar(255) NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `department` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `mentorship_program`

CREATE TABLE `mentorship_program` (
  `employee_id` int(11) NOT NULL,
  `mentor_id` int(11) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  PRIMARY KEY (`employee_id`,`mentor_id`),
  FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`mentor_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
