-- Schema for database: `schemabench_0225`

-- Table: `trainer_profiles`
CREATE TABLE `trainer_profiles` (
  `trainer_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `date_of_hire` date NOT NULL,
  `qualifications` varchar(255) NOT NULL,
  `course_assignments` varchar(255) NOT NULL,
  PRIMARY KEY (`trainer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `course_management`
CREATE TABLE `course_management` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_name` varchar(255) NOT NULL,
  `duration` int(11) NOT NULL,
  `fee` decimal(10,2) NOT NULL,
  `prerequisites` varchar(255) NOT NULL,
  `hardware_requirements` varchar(255) NOT NULL,
  `software_requirements` varchar(255) NOT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `class_scheduling`
CREATE TABLE `class_scheduling` (
  `class_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_id` int(11) NOT NULL,
  `trainer_id` int(11) NOT NULL,
  `classroom` varchar(255) NOT NULL,
  `date` date NOT NULL,
  PRIMARY KEY (`class_id`),
  FOREIGN KEY (`course_id`) REFERENCES `course_management` (`course_id`),
  FOREIGN KEY (`trainer_id`) REFERENCES `trainer_profiles` (`trainer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `client_management`
CREATE TABLE `client_management` (
  `client_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `enrollment_history` varchar(255) NOT NULL,
  `membership_type` varchar(255) NOT NULL,
  `discount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `enrollment_processing`
CREATE TABLE `enrollment_processing` (
  `enrollment_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `date_of_enrollment` date NOT NULL,
  `payment_status` varchar(255) NOT NULL,
  PRIMARY KEY (`enrollment_id`),
  FOREIGN KEY (`client_id`) REFERENCES `client_management` (`client_id`),
  FOREIGN KEY (`course_id`) REFERENCES `course_management` (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `hardware_software_tracking`
CREATE TABLE `hardware_software_tracking` (
  `track_id` int(11) NOT NULL AUTO_INCREMENT,
  `course_id` int(11) NOT NULL,
  `hardware_used` varchar(255) NOT NULL,
  `software_used` varchar(255) NOT NULL,
  PRIMARY KEY (`track_id`),
  FOREIGN KEY (`course_id`) REFERENCES `course_management` (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- Table: `book_orders`
CREATE TABLE `book_orders` (
  `order_id` int(11) NOT NULL AUTO_INCREMENT,
  `ISBN` varchar(255) NOT NULL,
  `author` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `vendor_order_id` varchar(255) NOT NULL,
  `order_status` varchar(255) NOT NULL,
  `delivery_details` varchar(255) NOT NULL,
  PRIMARY KEY (`order_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
