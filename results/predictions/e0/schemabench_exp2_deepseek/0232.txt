-- Schema for database: `schemabench_0232`

-- For table: `logs`
CREATE TABLE `logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `timestamp` datetime NOT NULL,
  `channel_name` varchar(255) NOT NULL,
  `channel_type` varchar(255) NOT NULL,
  `ocean_region` varchar(255) NOT NULL,
  `BTO` int(11) NOT NULL,
  `BFO` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `channel_name` (`channel_name`),
  KEY `channel_type` (`channel_type`),
  KEY `ocean_region` (`ocean_region`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `channels`
CREATE TABLE `channels` (
  `channel_name` varchar(255) NOT NULL,
  `channel_type` varchar(255) NOT NULL,
  PRIMARY KEY (`channel_name`,`channel_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `regions`
CREATE TABLE `regions` (
  `ocean_region` varchar(255) NOT NULL,
  PRIMARY KEY (`ocean_region`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
