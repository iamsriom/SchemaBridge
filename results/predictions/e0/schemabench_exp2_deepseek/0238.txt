-- Schema for database: `financial_management_system`

-- For table: `customers`

CREATE TABLE `customers` (
    `customer_id` INT(11) NOT NULL AUTO_INCREMENT,
    `name` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(255) NOT NULL,
    `address` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `accounts`

CREATE TABLE `accounts` (
    `account_id` INT(11) NOT NULL AUTO_INCREMENT,
    `customer_id` INT(11) NOT NULL,
    `account_type` VARCHAR(255) NOT NULL,
    `balance` DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    PRIMARY KEY (`account_id`),
    FOREIGN KEY (`customer_id`) REFERENCES `customers` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `transactions`

CREATE TABLE `transactions` (
    `transaction_id` INT(11) NOT NULL AUTO_INCREMENT,
    `account_id` INT(11) NOT NULL,
    `transaction_type` VARCHAR(255) NOT NULL,
    `amount` DECIMAL(10, 2) NOT NULL,
    `transaction_date` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`transaction_id`),
    FOREIGN KEY (`account_id`) REFERENCES `accounts` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `cards`

CREATE TABLE `cards` (
    `card_id` INT(11) NOT NULL AUTO_INCREMENT,
    `account_id` INT(11) NOT NULL,
    `card_type` VARCHAR(255) NOT NULL,
    `card_number` VARCHAR(255) NOT NULL,
    `spending_limit` DECIMAL(10, 2) NOT NULL,
    PRIMARY KEY (`card_id`),
    FOREIGN KEY (`account_id`) REFERENCES `accounts` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `loans`

CREATE TABLE `loans` (
    `loan_id` INT(11) NOT NULL AUTO_INCREMENT,
    `account_id` INT(11) NOT NULL,
    `loan_amount` DECIMAL(10, 2) NOT NULL,
    `interest_rate` DECIMAL(10, 2) NOT NULL,
    `term` INT(11) NOT NULL,
    `repayment_status` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`loan_id`),
    FOREIGN KEY (`account_id`) REFERENCES `accounts` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

-- For table: `investments`

CREATE TABLE `investments` (
    `investment_id` INT(11) NOT NULL AUTO_INCREMENT,
    `account_id` INT(11) NOT NULL,
    `investment_type` VARCHAR(255) NOT NULL,
    `investment_amount` DECIMAL(10, 2) NOT NULL,
    `current_value` DECIMAL(10, 2) NOT NULL,
    PRIMARY KEY (`investment_id`),
    FOREIGN KEY (`account_id`) REFERENCES `accounts` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
