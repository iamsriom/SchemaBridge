CREATE TABLE `taggit_tag` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `slug` varchar(100) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `taggit_tag_name_uniq` (`name`),
  UNIQUE KEY `taggit_tag_slug_uniq` (`slug`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `taggit_taggeditem` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `tag_id` int(11) NOT NULL,
  `content_type_id` int(11) NOT NULL,
  `object_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `taggit_taggeditem_tag_id_idx` (`tag_id`),
  KEY `taggit_taggeditem_content_type_id_idx` (`content_type_id`),
  KEY `taggit_taggeditem_object_id_idx` (`object_id`),
  CONSTRAINT `taggit_taggeditem_ibfk_1` FOREIGN KEY (`tag_id`) REFERENCES `taggit_tag` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
