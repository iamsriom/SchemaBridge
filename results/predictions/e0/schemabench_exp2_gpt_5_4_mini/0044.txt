-- Schema for database: schemabench_0044
CREATE TABLE `pods` (
  `pod_id` int(11) NOT NULL AUTO_INCREMENT,
  `pod_name` varchar(255) NOT NULL,
  `base_url` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`pod_id`),
  UNIQUE KEY `uniq_pods_base_url` (`base_url`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `pod_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `display_name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `password_hash` varchar(255) NOT NULL,
  `is_pseudonymous` tinyint(1) NOT NULL DEFAULT '1',
  `created_at` datetime NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `uniq_users_pod_username` (`pod_id`,`username`),
  KEY `users_ibfk_1` (`pod_id`),
  CONSTRAINT `users_ibfk_1` FOREIGN KEY (`pod_id`) REFERENCES `pods` (`pod_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `aspects` (
  `aspect_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`aspect_id`),
  UNIQUE KEY `uniq_aspects_user_name` (`user_id`,`name`),
  KEY `aspects_ibfk_1` (`user_id`),
  CONSTRAINT `aspects_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `aspect_memberships` (
  `aspect_membership_id` int(11) NOT NULL AUTO_INCREMENT,
  `aspect_id` int(11) NOT NULL,
  `member_user_id` int(11) NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`aspect_membership_id`),
  UNIQUE KEY `uniq_aspect_memberships_aspect_member` (`aspect_id`,`member_user_id`),
  KEY `aspect_memberships_ibfk_1` (`aspect_id`),
  KEY `aspect_memberships_ibfk_2` (`member_user_id`),
  CONSTRAINT `aspect_memberships_ibfk_1` FOREIGN KEY (`aspect_id`) REFERENCES `aspects` (`aspect_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `aspect_memberships_ibfk_2` FOREIGN KEY (`member_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `posts` (
  `post_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `visibility` varchar(50) NOT NULL DEFAULT 'public',
  `created_at` datetime NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  PRIMARY KEY (`post_id`),
  KEY `posts_ibfk_1` (`user_id`),
  CONSTRAINT `posts_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comments` (
  `comment_id` int(11) NOT NULL AUTO_INCREMENT,
  `post_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`comment_id`),
  KEY `comments_ibfk_1` (`post_id`),
  KEY `comments_ibfk_2` (`user_id`),
  CONSTRAINT `comments_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comments_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tags` (
  `tag_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`tag_id`),
  UNIQUE KEY `uniq_tags_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `post_tags` (
  `post_tag_id` int(11) NOT NULL AUTO_INCREMENT,
  `post_id` int(11) NOT NULL,
  `tag_id` int(11) NOT NULL,
  PRIMARY KEY (`post_tag_id`),
  UNIQUE KEY `uniq_post_tags_post_tag` (`post_id`,`tag_id`),
  KEY `post_tags_ibfk_1` (`post_id`),
  KEY `post_tags_ibfk_2` (`tag_id`),
  CONSTRAINT `post_tags_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `post_tags_ibfk_2` FOREIGN KEY (`tag_id`) REFERENCES `tags` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mentions` (
  `mention_id` int(11) NOT NULL AUTO_INCREMENT,

CREATE TABLE `reports` (
  `report_id` int(11) NOT NULL AUTO_INCREMENT,
  `reporter_user_id` int(11) NOT NULL,
  `reported_post_id` int(11) DEFAULT NULL,
  `reported_user_id` int(11) DEFAULT NULL,

  `status` varchar(50) NOT NULL DEFAULT 'open',

  PRIMARY KEY (`report_id`),
  KEY `reports_ibfk_1` (`reporter_user_id`),
  KEY `reports_ibfk_2` (`reported_post_id`),
  KEY `reports_ibfk_3` (`reported_user_id`),
  CONSTRAINT `reports_ibfk_1` FOREIGN KEY (`reporter_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reports_ibfk_2` FOREIGN KEY (`reported_post_id`) REFERENCES `posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reports_ibfk_3` FOREIGN KEY (`reported_user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account_migrations` (
  `migration_id` int(11) NOT NULL AUTO_INCREMENT,

  CONSTRAINT `account_migrations_ibfk_3` FOREIGN KEY (`to_pod_id`) REFERENCES `pods` (`pod_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
