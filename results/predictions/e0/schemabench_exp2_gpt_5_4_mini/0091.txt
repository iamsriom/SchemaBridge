-- Schema for database: schemabench_0091
CREATE TABLE `participants` (
  `participant_id` int(11) NOT NULL AUTO_INCREMENT,
  `study_code` varchar(50) NOT NULL,
  `sex` char(1) NOT NULL,
  `birth_year` int(11) DEFAULT NULL,
  `enrollment_date` date DEFAULT NULL,
  PRIMARY KEY (`participant_id`),
  UNIQUE KEY `study_code` (`study_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_history` (
  `history_id` int(11) NOT NULL AUTO_INCREMENT,
  `participant_id` int(11) NOT NULL,
  `condition_name` varchar(255) NOT NULL,
  `diagnosis_year` int(11) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`history_id`),
  KEY `participant_id` (`participant_id`),
  CONSTRAINT `medical_history_ibfk_1` FOREIGN KEY (`participant_id`) REFERENCES `participants` (`participant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `lifestyle_factors` (
  `lifestyle_id` int(11) NOT NULL AUTO_INCREMENT,
  `participant_id` int(11) NOT NULL,
  `record_date` date NOT NULL,
  `smoking_status` varchar(50) DEFAULT NULL,
  `alcohol_consumption` varchar(50) DEFAULT NULL,
  `physical_activity_level` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`lifestyle_id`),
  KEY `participant_id` (`participant_id`),
  CONSTRAINT `lifestyle_factors_ibfk_1` FOREIGN KEY (`participant_id`) REFERENCES `participants` (`participant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `health_metrics` (
  `metric_id` int(11) NOT NULL AUTO_INCREMENT,
  `participant_id` int(11) NOT NULL,
  `measurement_date` date NOT NULL,
  `cholesterol_total` decimal(5,2) DEFAULT NULL,
  `cholesterol_hdl` decimal(5,2) DEFAULT NULL,
  `cholesterol_ldl` decimal(5,2) DEFAULT NULL,
  `systolic_blood_pressure` int(11) DEFAULT NULL,
  `diastolic_blood_pressure` int(11) DEFAULT NULL,
  `body_mass_index` decimal(4,1) DEFAULT NULL,
  PRIMARY KEY (`metric_id`),
  KEY `participant_id` (`participant_id`),
  CONSTRAINT `health_metrics_ibfk_1` FOREIGN KEY (`participant_id`) REFERENCES `participants` (`participant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `atherosclerosis_outcomes` (
  `outcome_id` int(11) NOT NULL AUTO_INCREMENT,
  `participant_id` int(11) NOT NULL,
  `assessment_date` date NOT NULL,
  `atherosclerosis_present` tinyint(1) NOT NULL,
  `severity_grade` varchar(50) DEFAULT NULL,
  `comments` text DEFAULT NULL,
  PRIMARY KEY (`outcome_id`),
  KEY `participant_id` (`participant_id`),
  CONSTRAINT `atherosclerosis_outcomes_ibfk_1` FOREIGN KEY (`participant_id`) REFERENCES `participants` (`participant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
