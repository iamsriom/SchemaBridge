CREATE TABLE `schools` (
  `cds_code` varchar(14) NOT NULL,
  `school_name` varchar(255) NOT NULL,
  `district_name` varchar(255) NOT NULL,
  `county_name` varchar(255) DEFAULT NULL,
  `street_address` varchar(255) DEFAULT NULL,
  `city` varchar(100) DEFAULT NULL,
  `state` char(2) DEFAULT NULL,
  `zip_code` varchar(10) DEFAULT NULL,
  `latitude` decimal(10,7) DEFAULT NULL,
  `longitude` decimal(10,7) DEFAULT NULL,
  `school_type` varchar(100) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`cds_code`),
  KEY `schools_school_name_idx` (`school_name`),
  KEY `schools_district_name_idx` (`district_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
