-- Schema for database: schemabench_0098
CREATE TABLE `players` (
  `player_id` int(11) NOT NULL AUTO_INCREMENT,
  `player_name` varchar(255) NOT NULL,
  `country` varchar(100) DEFAULT NULL,
  `elo_rating` int(11) DEFAULT NULL,
  PRIMARY KEY (`player_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `events` (
  `event_id` int(11) NOT NULL AUTO_INCREMENT,
  `event_name` varchar(255) NOT NULL,
  `site` varchar(255) DEFAULT NULL,
  `event_date` date DEFAULT NULL,
  PRIMARY KEY (`event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `openings` (
  `opening_id` int(11) NOT NULL AUTO_INCREMENT,
  `opening_name` varchar(255) NOT NULL,
  `eco_code` char(3) DEFAULT NULL,
  `variation_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`opening_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `games` (
  `game_id` int(11) NOT NULL AUTO_INCREMENT,
  `event_id` int(11) NOT NULL,
  `white_player_id` int(11) NOT NULL,
  `black_player_id` int(11) NOT NULL,
  `opening_id` int(11) DEFAULT NULL,
  `game_date` date DEFAULT NULL,
  `round_number` varchar(50) DEFAULT NULL,
  `result` varchar(10) NOT NULL,
  `white_elo` int(11) DEFAULT NULL,
  `black_elo` int(11) DEFAULT NULL,
  `moves_pgn` text DEFAULT NULL,
  PRIMARY KEY (`game_id`),
  KEY `games_ibfk_1` (`event_id`),
  KEY `games_ibfk_2` (`white_player_id`),
  KEY `games_ibfk_3` (`black_player_id`),
  KEY `games_ibfk_4` (`opening_id`),
  CONSTRAINT `games_ibfk_1` FOREIGN KEY (`event_id`) REFERENCES `events` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `games_ibfk_2` FOREIGN KEY (`white_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `games_ibfk_3` FOREIGN KEY (`black_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `games_ibfk_4` FOREIGN KEY (`opening_id`) REFERENCES `openings` (`opening_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `predictions` (
  `prediction_id` int(11) NOT NULL AUTO_INCREMENT,
  `white_player_id` int(11) NOT NULL,
  `black_player_id` int(11) NOT NULL,
  `opening_id` int(11) DEFAULT NULL,
  `predicted_result` varchar(20) NOT NULL,
  `predicted_probability` decimal(5,4) NOT NULL,
  `white_elo` int(11) DEFAULT NULL,
  `black_elo` int(11) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`prediction_id`),
  KEY `predictions_ibfk_1` (`white_player_id`),
  KEY `predictions_ibfk_2` (`black_player_id`),
  KEY `predictions_ibfk_3` (`opening_id`),
  CONSTRAINT `predictions_ibfk_1` FOREIGN KEY (`white_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `predictions_ibfk_2` FOREIGN KEY (`black_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `predictions_ibfk_3` FOREIGN KEY (`opening_id`) REFERENCES `openings` (`opening_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
