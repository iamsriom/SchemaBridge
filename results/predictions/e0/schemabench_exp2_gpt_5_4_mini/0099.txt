-- Schema for database: schemabench_0099
CREATE TABLE `publications` (
  `publication_id` int(11) NOT NULL,
  `class_label` varchar(50) NOT NULL,
  `word_vector` text NOT NULL,
  PRIMARY KEY (`publication_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `citations` (
  `citation_id` int(11) NOT NULL AUTO_INCREMENT,
  `citing_publication_id` int(11) NOT NULL,
  `cited_publication_id` int(11) NOT NULL,
  PRIMARY KEY (`citation_id`),
  KEY `citations_ibfk_1` (`citing_publication_id`),
  KEY `citations_ibfk_2` (`cited_publication_id`),
  CONSTRAINT `citations_ibfk_1` FOREIGN KEY (`citing_publication_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `citations_ibfk_2` FOREIGN KEY (`cited_publication_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
