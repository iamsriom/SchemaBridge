CREATE TABLE `survey_year` (
  `survey_year_id` int(11) NOT NULL AUTO_INCREMENT,
  `year` int(11) NOT NULL,
  `release_name` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`survey_year_id`),
  UNIQUE KEY `uk_survey_year_year` (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `household` (
  `household_id` int(11) NOT NULL AUTO_INCREMENT,
  `survey_year_id` int(11) NOT NULL,
  `anonymized_household_code` char(12) NOT NULL,
  `region` varchar(50) DEFAULT NULL,
  `urban_rural_status` varchar(20) DEFAULT NULL,
  `household_size` int(11) DEFAULT NULL,
  `num_children` int(11) DEFAULT NULL,
  `home_ownership_status` varchar(30) DEFAULT NULL,
  `reference_person_age` int(11) DEFAULT NULL,
  `reference_person_sex` varchar(10) DEFAULT NULL,
  `reference_person_race` varchar(50) DEFAULT NULL,
  `education_level` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`household_id`),
  UNIQUE KEY `uk_household_anonymized_household_code` (`anonymized_household_code`),
  KEY `household_ibfk_1` (`survey_year_id`),
  CONSTRAINT `household_ibfk_1` FOREIGN KEY (`survey_year_id`) REFERENCES `survey_year` (`survey_year_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `household_income` (
  `household_income_id` int(11) NOT NULL AUTO_INCREMENT,
  `household_id` int(11) NOT NULL,
  `income_type` varchar(50) NOT NULL,
  `amount` decimal(12,2) NOT NULL,
  `period` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`household_income_id`),
  KEY `household_income_ibfk_1` (`household_id`),
  CONSTRAINT `household_income_ibfk_1` FOREIGN KEY (`household_id`) REFERENCES `household` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expenditure_category` (
  `expenditure_category_id` int(11) NOT NULL AUTO_INCREMENT,
  `category_name` varchar(100) NOT NULL,
  `parent_category_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`expenditure_category_id`),
  UNIQUE KEY `uk_expenditure_category_category_name` (`category_name`),
  KEY `expenditure_category_ibfk_1` (`parent_category_id`),
  CONSTRAINT `expenditure_category_ibfk_1` FOREIGN KEY (`parent_category_id`) REFERENCES `expenditure_category` (`expenditure_category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `household_expenditure` (
  `household_expenditure_id` int(11) NOT NULL AUTO_INCREMENT,
  `household_id` int(11) NOT NULL,
  `expenditure_category_id` int(11) NOT NULL,
  `amount` decimal(12,2) NOT NULL,
  `frequency` varchar(20) DEFAULT NULL,
  `reporting_period` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`household_expenditure_id`),
  KEY `household_expenditure_ibfk_1` (`household_id`),
  KEY `household_expenditure_ibfk_2` (`expenditure_category_id`),
  CONSTRAINT `household_expenditure_ibfk_1` FOREIGN KEY (`household_id`) REFERENCES `household` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `household_expenditure_ibfk_2` FOREIGN KEY (`expenditure_category_id`) REFERENCES `expenditure_category` (`expenditure_category_id`) ON DELETE CASCADE ON UPDATE CASCADE
