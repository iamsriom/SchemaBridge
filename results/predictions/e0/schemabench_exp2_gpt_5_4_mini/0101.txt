-- Schema for database: schemabench_0101
CREATE TABLE `country_forest_history` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `country_name` varchar(255) NOT NULL,
  `country_code` char(3) DEFAULT NULL,
  `year` int(11) NOT NULL,
  `forest_area_percent` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `country_year_unique` (`country_name`,`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `forest_area_prediction` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `country_name` varchar(255) NOT NULL,
  `country_code` char(3) DEFAULT NULL,
  `predicted_year` int(11) NOT NULL,
  `predicted_forest_area_percent` decimal(5,2) NOT NULL,
  `model_version` varchar(100) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `country_prediction_unique` (`country_name`,`predicted_year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
