-- Schema for database: schemabench_0102
CREATE TABLE `breweries` (
  `brewery_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `city` varchar(255) NOT NULL,
  `state` char(2) NOT NULL,
  PRIMARY KEY (`brewery_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beers` (
  `beer_id` int(11) NOT NULL AUTO_INCREMENT,
  `brewery_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `style` varchar(255) NOT NULL,
  `abv` decimal(4,2) NOT NULL,
  `ibu` int(11) DEFAULT NULL,
  `size_oz` decimal(5,2) NOT NULL,
  PRIMARY KEY (`beer_id`),
  KEY `brewery_id` (`brewery_id`),
  CONSTRAINT `beers_ibfk_1` FOREIGN KEY (`brewery_id`) REFERENCES `breweries` (`brewery_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
