-- Schema for database: schemabench_0104
CREATE TABLE `incidents` (
  `incident_id` int(11) NOT NULL AUTO_INCREMENT,
  `incident_date` date NOT NULL,
  `incident_time` time DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `city` varchar(100) DEFAULT NULL,
  `state` char(2) DEFAULT NULL,
  `subject_status` varchar(100) DEFAULT NULL,
  `weapon_info` varchar(255) DEFAULT NULL,
  `summary` text DEFAULT NULL,
  `external_document_url` varchar(500) DEFAULT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`incident_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `officers` (
  `officer_id` int(11) NOT NULL AUTO_INCREMENT,
  `incident_id` int(11) NOT NULL,
  `badge_number` varchar(50) DEFAULT NULL,
  `first_name` varchar(100) DEFAULT NULL,
  `last_name` varchar(100) DEFAULT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `race_ethnicity` varchar(100) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `rank_title` varchar(100) DEFAULT NULL,
  `role_in_incident` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`officer_id`),
  KEY `officers_ibfk_1` (`incident_id`),
  CONSTRAINT `officers_ibfk_1` FOREIGN KEY (`incident_id`) REFERENCES `incidents` (`incident_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subjects` (
  `subject_id` int(11) NOT NULL AUTO_INCREMENT,
  `incident_id` int(11) NOT NULL,
  `first_name` varchar(100) DEFAULT NULL,
  `last_name` varchar(100) DEFAULT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `race_ethnicity` varchar(100) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `armed_status` varchar(100) DEFAULT NULL,
  `injury_status` varchar(100) DEFAULT NULL,
  `outcome` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`subject_id`),
  KEY `subjects_ibfk_1` (`incident_id`),
  CONSTRAINT `subjects_ibfk_1` FOREIGN KEY (`incident_id`) REFERENCES `incidents` (`incident_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
