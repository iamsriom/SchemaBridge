-- Schema for database: schemabench_0108
CREATE TABLE `gene` (
  `gene_id` int(11) NOT NULL,
  `gene_name` varchar(255) NOT NULL,
  `essentiality` varchar(100) DEFAULT NULL,
  `phenotype` varchar(255) DEFAULT NULL,
  `function` varchar(255) DEFAULT NULL,
  `localization` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`gene_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gene_interaction` (
  `interaction_id` int(11) NOT NULL,
  `gene_id_1` int(11) NOT NULL,
  `gene_id_2` int(11) NOT NULL,
  `interaction_type` varchar(100) DEFAULT NULL,
  `expression_correlation` decimal(10,4) DEFAULT NULL,
  `notes` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`interaction_id`),
  KEY `gene_interaction_ibfk_1` (`gene_id_1`),
  KEY `gene_interaction_ibfk_2` (`gene_id_2`),
  CONSTRAINT `gene_interaction_ibfk_1` FOREIGN KEY (`gene_id_1`) REFERENCES `gene` (`gene_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `gene_interaction_ibfk_2` FOREIGN KEY (`gene_id_2`) REFERENCES `gene` (`gene_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
