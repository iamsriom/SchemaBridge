-- Schema for database: schemabench_0110
CREATE TABLE `institutions` (
  `institution_id` int(11) NOT NULL AUTO_INCREMENT,
  `institution_name` varchar(255) NOT NULL,
  `city` varchar(255) DEFAULT NULL,
  `state` varchar(100) DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`institution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_elements` (
  `program_element_id` int(11) NOT NULL AUTO_INCREMENT,
  `element_code` varchar(50) NOT NULL,
  `element_name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`program_element_id`),
  UNIQUE KEY `element_code` (`element_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grants` (
  `grant_id` int(11) NOT NULL AUTO_INCREMENT,
  `award_number` varchar(50) NOT NULL,
  `title` varchar(500) NOT NULL,
  `abstract` text DEFAULT NULL,
  `award_amount` decimal(15,2) NOT NULL,
  `award_start_date` date DEFAULT NULL,
  `award_end_date` date DEFAULT NULL,
  `fiscal_year` int(11) DEFAULT NULL,
  `institution_id` int(11) NOT NULL,
  `program_element_id` int(11) DEFAULT NULL,
  `award_type` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`grant_id`),
  UNIQUE KEY `award_number` (`award_number`),
  KEY `institution_id` (`institution_id`),
  KEY `program_element_id` (`program_element_id`),
  CONSTRAINT `grants_ibfk_1` FOREIGN KEY (`institution_id`) REFERENCES `institutions` (`institution_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grants_ibfk_2` FOREIGN KEY (`program_element_id`) REFERENCES `program_elements` (`program_element_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `investigators` (
  `investigator_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `institution_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`investigator_id`),
  KEY `institution_id` (`institution_id`),
  CONSTRAINT `investigators_ibfk_1` FOREIGN KEY (`institution_id`) REFERENCES `institutions` (`institution_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grant_investigators` (
  `grant_id` int(11) NOT NULL,
  `investigator_id` int(11) NOT NULL,
  `role` varchar(100) DEFAULT NULL,
  `is_primary` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`grant_id`,`investigator_id`),
  KEY `investigator_id` (`investigator_id`),
  CONSTRAINT `grant_investigators_ibfk_1` FOREIGN KEY (`grant_id`) REFERENCES `grants` (`grant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grant_investigators_ibfk_2` FOREIGN KEY (`investigator_id`) REFERENCES `investigators` (`investigator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `award_predictions` (
  `prediction_id` int(11) NOT NULL AUTO_INCREMENT,
  `grant_id` int(11) NOT NULL,
  `predicted_award_amount` decimal(15,2) NOT NULL,
  `model_name` varchar(255) NOT NULL,
  `model_version` varchar(100) DEFAULT NULL,
  `predicted_at` datetime NOT NULL,
  `actual_award_amount` decimal(15,2) DEFAULT NULL,
  `prediction_error` decimal(15,2) DEFAULT NULL,
  PRIMARY KEY (`prediction_id`),
  KEY `grant_id` (`grant_id`),
  CONSTRAINT `award_predictions_ibfk_1` FOREIGN KEY (`grant_id`) REFERENCES `grants` (`grant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
