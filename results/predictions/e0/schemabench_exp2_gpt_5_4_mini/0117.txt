-- Schema for database: schemabench_0117
CREATE TABLE `compound` (
  `compound_id` int(11) NOT NULL,
  `activity` varchar(50) DEFAULT NULL,
  `logp` decimal(10,4) DEFAULT NULL,
  `lumo` decimal(10,4) DEFAULT NULL,
  `is_active` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`compound_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `atom` (
  `atom_id` int(11) NOT NULL,
  `compound_id` int(11) NOT NULL,
  `element_type` varchar(10) NOT NULL,
  `charge` int(11) NOT NULL DEFAULT '0',
  `atom_type` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`atom_id`),
  KEY `compound_id` (`compound_id`),
  CONSTRAINT `atom_ibfk_1` FOREIGN KEY (`compound_id`) REFERENCES `compound` (`compound_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bond` (
  `bond_id` int(11) NOT NULL,
  `compound_id` int(11) NOT NULL,
  `atom1_id` int(11) NOT NULL,
  `atom2_id` int(11) NOT NULL,
  `bond_type` varchar(20) NOT NULL,
  PRIMARY KEY (`bond_id`),
  KEY `compound_id` (`compound_id`),
  KEY `atom1_id` (`atom1_id`),
  KEY `atom2_id` (`atom2_id`),
  CONSTRAINT `bond_ibfk_1` FOREIGN KEY (`compound_id`) REFERENCES `compound` (`compound_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bond_ibfk_2` FOREIGN KEY (`atom1_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bond_ibfk_3` FOREIGN KEY (`atom2_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ring` (
  `ring_id` int(11) NOT NULL,
  `compound_id` int(11) NOT NULL,
  `ring_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`ring_id`),
  KEY `compound_id` (`compound_id`),
  CONSTRAINT `ring_ibfk_1` FOREIGN KEY (`compound_id`) REFERENCES `compound` (`compound_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ring_atom` (
  `ring_id` int(11) NOT NULL,
  `atom_id` int(11) NOT NULL,
  PRIMARY KEY (`ring_id`,`atom_id`),
  KEY `atom_id` (`atom_id`),
  CONSTRAINT `ring_atom_ibfk_1` FOREIGN KEY (`ring_id`) REFERENCES `ring` (`ring_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ring_atom_ibfk_2` FOREIGN KEY (`atom_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
