-- Schema for database: schemabench_0123
CREATE TABLE `molecules` (
  `molecule_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_name` varchar(255) NOT NULL,
  `smiles` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`molecule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `molecule_positions` (
  `position_id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_id` int(11) NOT NULL,
  `position_number` int(11) NOT NULL,
  `substituent_name` varchar(255) DEFAULT NULL,
  `flexibility` decimal(10,4) DEFAULT NULL,
  `polarizability` decimal(10,4) DEFAULT NULL,
  `sigma_value` decimal(10,4) DEFAULT NULL,
  `activity_value` decimal(10,4) DEFAULT NULL,
  `inhibitor_activity` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`position_id`),
  KEY `molecule_positions_ibfk_1` (`molecule_id`),
  CONSTRAINT `molecule_positions_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
