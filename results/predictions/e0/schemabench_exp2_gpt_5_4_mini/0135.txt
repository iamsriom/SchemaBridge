-- Schema for database: schemabench_0135
CREATE TABLE `train` (
  `train_id` int(11) NOT NULL,
  `direction` enum('east','west') NOT NULL,
  PRIMARY KEY (`train_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `car` (
  `car_id` int(11) NOT NULL,
  `train_id` int(11) NOT NULL,
  `position_in_train` int(11) NOT NULL,
  `shape` varchar(50) DEFAULT NULL,
  `length` int(11) DEFAULT NULL,
  `sides` int(11) DEFAULT NULL,
  `roof_type` varchar(50) DEFAULT NULL,
  `wheels` int(11) DEFAULT NULL,
  `load_shape` varchar(50) DEFAULT NULL,
  `load_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`car_id`),
  KEY `train_id` (`train_id`),
  CONSTRAINT `car_ibfk_1` FOREIGN KEY (`train_id`) REFERENCES `train` (`train_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
