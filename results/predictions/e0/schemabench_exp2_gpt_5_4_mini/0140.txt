-- Schema for database: schemabench_0140
CREATE TABLE `voyages` (
  `voyage_id` int(11) NOT NULL,
  `vessel_name` varchar(255) DEFAULT NULL,
  `master_name` varchar(255) DEFAULT NULL,
  `tonnage` int(11) DEFAULT NULL,
  `vessel_type` varchar(100) DEFAULT NULL,
  `construction_place` varchar(255) DEFAULT NULL,
  `construction_year` int(11) DEFAULT NULL,
  `departure_date` date DEFAULT NULL,
  `arrival_date` date DEFAULT NULL,
  `departure_port` varchar(255) DEFAULT NULL,
  `arrival_port` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`voyage_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `invoices` (
  `invoice_id` int(11) NOT NULL,
  `voyage_id` int(11) NOT NULL,
  `chamber_name` varchar(255) DEFAULT NULL,
  `invoice_number` varchar(100) DEFAULT NULL,
  `invoice_date` date DEFAULT NULL,
  `amount` decimal(12,2) DEFAULT NULL,
  PRIMARY KEY (`invoice_id`),
  KEY `voyage_id` (`voyage_id`),
  CONSTRAINT `invoices_ibfk_1` FOREIGN KEY (`voyage_id`) REFERENCES `voyages` (`voyage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `personnel_records` (
  `personnel_id` int(11) NOT NULL,
  `voyage_id` int(11) NOT NULL,
  `person_name` varchar(255) DEFAULT NULL,
  `person_category` varchar(50) DEFAULT NULL,
  `status_departure` varchar(255) DEFAULT NULL,
  `status_arrival` varchar(255) DEFAULT NULL,
  `status_cape_stop` varchar(255) DEFAULT NULL,
  `casualty_status` varchar(255) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`personnel_id`),
  KEY `voyage_id` (`voyage_id`),
  CONSTRAINT `personnel_records_ibfk_1` FOREIGN KEY (`voyage_id`) REFERENCES `voyages` (`voyage_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
