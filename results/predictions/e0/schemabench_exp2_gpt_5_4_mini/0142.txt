-- Schema for database: schemabench_0142
CREATE TABLE `publications` (
  `publication_id` int(11) NOT NULL AUTO_INCREMENT,
  `class_label` varchar(50) NOT NULL,
  PRIMARY KEY (`publication_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `words` (
  `word_id` int(11) NOT NULL AUTO_INCREMENT,
  `word_name` varchar(255) NOT NULL,
  PRIMARY KEY (`word_id`),
  UNIQUE KEY `word_name` (`word_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publication_words` (
  `publication_id` int(11) NOT NULL,
  `word_id` int(11) NOT NULL,
  `present` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`publication_id`,`word_id`),
  KEY `word_id` (`word_id`),
  CONSTRAINT `publication_words_ibfk_1` FOREIGN KEY (`publication_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `publication_words_ibfk_2` FOREIGN KEY (`word_id`) REFERENCES `words` (`word_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `citations` (
  `citation_id` int(11) NOT NULL AUTO_INCREMENT,
  `citing_publication_id` int(11) NOT NULL,
  `cited_publication_id` int(11) NOT NULL,
  PRIMARY KEY (`citation_id`),
  UNIQUE KEY `uq_citation_pair` (`citing_publication_id`,`cited_publication_id`),
  KEY `cited_publication_id` (`cited_publication_id`),
  CONSTRAINT `citations_ibfk_1` FOREIGN KEY (`citing_publication_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `citations_ibfk_2` FOREIGN KEY (`cited_publication_id`) REFERENCES `publications` (`publication_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
