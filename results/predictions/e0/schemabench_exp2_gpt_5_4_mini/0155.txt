-- Schema for database: schemabench_0155
CREATE TABLE `person` (
  `person_id` int(11) NOT NULL AUTO_INCREMENT,
  `full_name` varchar(255) NOT NULL,
  `role` varchar(100) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`person_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `android` (
  `android_id` int(11) NOT NULL AUTO_INCREMENT,
  `android_name` varchar(255) NOT NULL,
  `model` varchar(100) DEFAULT NULL,
  `function` varchar(100) DEFAULT NULL,
  `owner_person_id` int(11) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`android_id`),
  KEY `android_ibfk_1` (`owner_person_id`),
  CONSTRAINT `android_ibfk_1` FOREIGN KEY (`owner_person_id`) REFERENCES `person` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bracelet` (
  `bracelet_id` int(11) NOT NULL AUTO_INCREMENT,
  `serial_number` varchar(100) NOT NULL,
  `person_id` int(11) DEFAULT NULL,
  `android_id` int(11) DEFAULT NULL,
  `issued_at` datetime DEFAULT NULL,
  `active` tinyint(1) NOT NULL DEFAULT '1',
  PRIMARY KEY (`bracelet_id`),
  UNIQUE KEY `serial_number` (`serial_number`),
  KEY `bracelet_ibfk_1` (`person_id`),
  KEY `bracelet_ibfk_2` (`android_id`),
  CONSTRAINT `bracelet_ibfk_1` FOREIGN KEY (`person_id`) REFERENCES `person` (`person_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bracelet_ibfk_2` FOREIGN KEY (`android_id`) REFERENCES `android` (`android_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `room` (
  `room_id` int(11) NOT NULL AUTO_INCREMENT,

  `description` text DEFAULT NULL,
  PRIMARY KEY (`room_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `camera_id` int(11) NOT NULL AUTO_INCREMENT,

  `status` varchar(50) DEFAULT NULL,

  KEY `cctv_camera_ibfk_1` (`room_id`),

  PRIMARY KEY (`movement_event_id`),
  KEY `movement_event_ibfk_1` (`bracelet_id`),
  KEY `movement_event_ibfk_2` (`camera_id`),
  KEY `movement_event_ibfk_3` (`room_id`),
  CONSTRAINT `movement_event_ibfk_1` FOREIGN KEY (`bracelet_id`) REFERENCES `bracelet` (`bracelet_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `movement_event_ibfk_2` FOREIGN KEY (`camera_id`) REFERENCES `cctv_camera` (`camera_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `movement_event_ibfk_3` FOREIGN KEY (`room_id`) REFERENCES `room` (`room_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
