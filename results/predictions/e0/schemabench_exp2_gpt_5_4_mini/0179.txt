CREATE TABLE `circuits` (
  `circuit_id` int(11) NOT NULL AUTO_INCREMENT,
  `circuit_name` varchar(255) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  `length_km` decimal(6,3) DEFAULT NULL,
  `configuration` varchar(255) DEFAULT NULL,
  `active_from_year` int(11) DEFAULT NULL,
  `active_to_year` int(11) DEFAULT NULL,
  PRIMARY KEY (`circuit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `races` (
  `race_id` int(11) NOT NULL AUTO_INCREMENT,
  `race_year` int(11) NOT NULL,
  `race_name` varchar(255) NOT NULL,
  `race_date` date NOT NULL,
  `start_time` time DEFAULT NULL,
  `circuit_id` int(11) NOT NULL,

  `cancellation_reason` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`race_id`),
  KEY `races_ibfk_1` (`circuit_id`),
  CONSTRAINT `races_ibfk_1` FOREIGN KEY (`circuit_id`) REFERENCES `circuits` (`circuit_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `driver_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `full_name` varchar(255) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`driver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `teams` (
  `team_id` int(11) NOT NULL AUTO_INCREMENT,

  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `car_id` int(11) NOT NULL AUTO_INCREMENT,
  `team_id` int(11) NOT NULL,
  `car_number` varchar(20) NOT NULL,
  `car_class` varchar(50) DEFAULT NULL,

  `engine` varchar(255) DEFAULT NULL,
  `tyre_brand` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`car_id`),
  KEY `cars_ibfk_1` (`team_id`),
  CONSTRAINT `cars_ibfk_1` FOREIGN KEY (`team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE

  KEY `race_results_ibfk_4` (`driver_id`),
  CONSTRAINT `race_results_ibfk_1` FOREIGN KEY (`race_id`) REFERENCES `races` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE,
