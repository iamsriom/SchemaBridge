-- Schema for database: schemabench_0180
CREATE TABLE `aircraft` (
  `aircraft_id` int(11) NOT NULL AUTO_INCREMENT,
  `registration_number` varchar(50) NOT NULL,
  `manufacturer` varchar(100) DEFAULT NULL,
  `model` varchar(100) DEFAULT NULL,
  `engine_manufacturer` varchar(100) DEFAULT NULL,
  `engine_model` varchar(100) DEFAULT NULL,
  `year_built` int(11) DEFAULT NULL,
  PRIMARY KEY (`aircraft_id`),
  UNIQUE KEY `registration_number` (`registration_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `flight_id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_id` int(11) NOT NULL,
  `flight_date` date NOT NULL,
  `origin_airport` varchar(10) DEFAULT NULL,
  `destination_airport` varchar(10) DEFAULT NULL,
  `flight_number` varchar(20) DEFAULT NULL,
  `flying_hours` decimal(8,2) NOT NULL,
  `operations_count` int(11) NOT NULL DEFAULT 1,
  `grounded_status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`flight_id`),
  KEY `aircraft_id` (`aircraft_id`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`aircraft_id`) REFERENCES `aircraft` (`aircraft_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
