-- Schema for database: schemabench_0198
CREATE TABLE `users` (
  `cookie_id` varchar(255) NOT NULL,
  `user_id` int(11) DEFAULT NULL,
  `first_name` varchar(255) DEFAULT NULL,
  `last_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`cookie_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `events` (
  `visit_id` int(11) NOT NULL,
  `cookie_id` varchar(255) NOT NULL,
  `event_type` int(11) NOT NULL,
  `page_id` int(11) DEFAULT NULL,
  `event_time` datetime DEFAULT NULL,
  PRIMARY KEY (`visit_id`,`event_type`),
  KEY `events_ibfk_1` (`cookie_id`),
  KEY `events_ibfk_2` (`event_type`),
  KEY `events_ibfk_3` (`page_id`),
  CONSTRAINT `events_ibfk_1` FOREIGN KEY (`cookie_id`) REFERENCES `users` (`cookie_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `events_ibfk_2` FOREIGN KEY (`event_type`) REFERENCES `event_identifier` (`event_type`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `events_ibfk_3` FOREIGN KEY (`page_id`) REFERENCES `page_hierarchy` (`page_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `event_identifier` (
  `event_type` int(11) NOT NULL,
  `event_name` varchar(255) NOT NULL,
  PRIMARY KEY (`event_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `campaign_identifier` (
  `campaign_id` int(11) NOT NULL,
  `campaign_name` varchar(255) NOT NULL,
  `start_date` datetime DEFAULT NULL,
  `end_date` datetime DEFAULT NULL,
  PRIMARY KEY (`campaign_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `page_hierarchy` (
  `page_id` int(11) NOT NULL,
  `page_name` varchar(255) NOT NULL,
  `product_category` varchar(255) DEFAULT NULL,
  `product_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`page_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
