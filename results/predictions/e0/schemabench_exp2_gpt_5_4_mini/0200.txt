-- Schema for database: schemabench_0200
CREATE TABLE `interest_map` (
  `id` int(11) NOT NULL,
  `interest_name` varchar(255) NOT NULL,
  `interest_summary` varchar(255) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  `last_modified` datetime DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `interest_metrics` (
  `month_year` char(6) NOT NULL,
  `interest_id` int(11) NOT NULL,
  `composition` decimal(5,2) DEFAULT NULL,
  `index_value` decimal(7,2) DEFAULT NULL,
  `ranking` int(11) DEFAULT NULL,
  `percentile_ranking` decimal(5,2) DEFAULT NULL,
  `created_at` datetime DEFAULT NULL,
  `last_modified` datetime DEFAULT NULL,
  PRIMARY KEY (`month_year`,`interest_id`),
  KEY `interest_metrics_ibfk_1` (`interest_id`),
  CONSTRAINT `interest_metrics_ibfk_1` FOREIGN KEY (`interest_id`) REFERENCES `interest_map` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
