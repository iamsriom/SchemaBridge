-- Schema for database: schemabench_0212
CREATE TABLE `movies` (
  `movie_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `year` int(11) DEFAULT NULL,
  `duration` int(11) DEFAULT NULL,
  `language` varchar(100) DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  `release_date` date DEFAULT NULL,
  PRIMARY KEY (`movie_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `genres` (
  `genre_id` int(11) NOT NULL,
  `genre_name` varchar(100) NOT NULL,
  PRIMARY KEY (`genre_id`),
  UNIQUE KEY `genre_name` (`genre_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `names` (
  `name_id` int(11) NOT NULL,
  `person_name` varchar(255) NOT NULL,
  `birth_year` int(11) DEFAULT NULL,
  `death_year` int(11) DEFAULT NULL,
  PRIMARY KEY (`name_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `roles` (
  `role_id` int(11) NOT NULL,
  `role_name` varchar(100) NOT NULL,
  PRIMARY KEY (`role_id`),
  UNIQUE KEY `role_name` (`role_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `directors` (
  `director_id` int(11) NOT NULL,
  `name_id` int(11) NOT NULL,
  PRIMARY KEY (`director_id`),
  KEY `name_id` (`name_id`),
  CONSTRAINT `directors_ibfk_1` FOREIGN KEY (`name_id`) REFERENCES `names` (`name_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `movie_genres` (
  `movie_id` int(11) NOT NULL,
  `genre_id` int(11) NOT NULL,
  PRIMARY KEY (`movie_id`,`genre_id`),
  KEY `genre_id` (`genre_id`),
  CONSTRAINT `movie_genres_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movies` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `movie_genres_ibfk_2` FOREIGN KEY (`genre_id`) REFERENCES `genres` (`genre_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `movie_cast` (
  `movie_id` int(11) NOT NULL,
  `name_id` int(11) NOT NULL,
  `role_id` int(11) NOT NULL,
  `billing_order` int(11) DEFAULT NULL,
  PRIMARY KEY (`movie_id`,`name_id`,`role_id`),
  KEY `name_id` (`name_id`),
  KEY `role_id` (`role_id`),
  CONSTRAINT `movie_cast_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movies` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `movie_cast_ibfk_2` FOREIGN KEY (`name_id`) REFERENCES `names` (`name_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `movie_cast_ibfk_3` FOREIGN KEY (`role_id`) REFERENCES `roles` (`role_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ratings` (
  `rating_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  `avg_rating` decimal(3,1) DEFAULT NULL,
  `total_votes` int(11) DEFAULT NULL,
  `median_rating` int(11) DEFAULT NULL,
  PRIMARY KEY (`rating_id`),
  UNIQUE KEY `movie_id` (`movie_id`),
  CONSTRAINT `ratings_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movies` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `movie_income` (
  `income_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  `budget` bigint(20) DEFAULT NULL,
  `gross_income` bigint(20) DEFAULT NULL,
  `currency` char(3) DEFAULT NULL,
  PRIMARY KEY (`income_id`),
  UNIQUE KEY `movie_id` (`movie_id`),
  CONSTRAINT `movie_income_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movies` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `audience_demographics` (
  `demographic_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  `age_group` varchar(50) DEFAULT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `region` varchar(100) DEFAULT NULL,
  `share_percentage` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`demographic_id`),
  KEY `movie_id` (`movie_id`),
  CONSTRAINT `audience_demographics_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movies` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
