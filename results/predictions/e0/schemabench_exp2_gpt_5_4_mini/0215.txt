-- Schema for database: schemabench_0215
CREATE TABLE `airport` (
  `airport_id` int(11) NOT NULL AUTO_INCREMENT,
  `airport_code` char(3) NOT NULL,
  `airport_name` varchar(255) NOT NULL,
  `city` varchar(255) NOT NULL,
  `state` varchar(255) DEFAULT NULL,
  `country` varchar(255) NOT NULL,
  PRIMARY KEY (`airport_id`),
  UNIQUE KEY `uk_airport_code` (`airport_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `passenger` (
  `passenger_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(25) DEFAULT NULL,
  `account_number` varchar(50) DEFAULT NULL,
  `frequent_flyer_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`passenger_id`),
  UNIQUE KEY `uk_passenger_email` (`email`),
  UNIQUE KEY `uk_passenger_account_number` (`account_number`),
  UNIQUE KEY `uk_passenger_frequent_flyer_number` (`frequent_flyer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `aircraft` (
  `aircraft_id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_tail_number` varchar(20) NOT NULL,
  `make` varchar(100) NOT NULL,
  `model` varchar(100) DEFAULT NULL,
  `weight_limit` decimal(10,2) NOT NULL,
  `assigned_flight_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`aircraft_id`),
  UNIQUE KEY `uk_aircraft_tail_number` (`aircraft_tail_number`),
  UNIQUE KEY `uk_aircraft_assigned_flight_number` (`assigned_flight_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_crew` (
  `crew_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `role` varchar(100) NOT NULL,
  `phone_number` varchar(25) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `assigned_airport_id` int(11) NOT NULL,
  PRIMARY KEY (`crew_id`),
  KEY `flight_crew_ibfk_1` (`assigned_airport_id`),
  CONSTRAINT `flight_crew_ibfk_1` FOREIGN KEY (`assigned_airport_id`) REFERENCES `airport` (`airport_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `flight_id` int(11) NOT NULL AUTO_INCREMENT,
  `flight_number` varchar(20) NOT NULL,
  `departure_airport_id` int(11) NOT NULL,
  `arrival_airport_id` int(11) NOT NULL,
  `aircraft_id` int(11) DEFAULT NULL,
  `departure_time` datetime NOT NULL,
  `arrival_time` datetime NOT NULL,
  `base_price` decimal(10,2) NOT NULL,
  `flight_status` varchar(50) NOT NULL,
  PRIMARY KEY (`flight_id`),
  UNIQUE KEY `uk_flight_number` (`flight_number`),
  KEY `flight_ibfk_1` (`departure_airport_id`),
  KEY `flight_ibfk_2` (`arrival_airport_id`),
  KEY `flight_ibfk_3` (`aircraft_id`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`departure_airport_id`) REFERENCES `airport` (`airport_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_2` FOREIGN KEY (`arrival_airport_id`) REFERENCES `airport` (`airport_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_3` FOREIGN KEY (`aircraft_id`) REFERENCES `aircraft` (`aircraft_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ticket` (
  `ticket_id` int(11) NOT NULL AUTO_INCREMENT,
  `ticket_number` varchar(30) NOT NULL,
  `passenger_id` int(11) NOT NULL,
  `flight_id` int(11) NOT NULL,
  `seat_number` varchar(10) NOT NULL,
  `route_details` varchar(255) DEFAULT NULL,
  `pricing` decimal(10,2) NOT NULL,
  `flight_status` varchar(50) NOT NULL,
  `issued_at` datetime NOT NULL,
  PRIMARY KEY (`ticket_id`),
  UNIQUE KEY `uk_ticket_number` (`ticket_number`),
  KEY `ticket_ibfk_1` (`passenger_id`),
  KEY `ticket_ibfk_2` (`flight_id`),
  CONSTRAINT `ticket_ibfk_1` FOREIGN KEY (`passenger_id`) REFERENCES `passenger` (`passenger_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ticket_ibfk_2` FOREIGN KEY (`flight_id`) REFERENCES `flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `luggage` (
  `luggage_id` int(11) NOT NULL AUTO_INCREMENT,
  `passenger_id` int(11) NOT NULL,
  `flight_id` int(11) NOT NULL,
  `ticket_id` int(11) DEFAULT NULL,
  `bag_tag_number` varchar(30) NOT NULL,
  `weight` decimal(10,2) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`luggage_id`),
  UNIQUE KEY `uk_luggage_bag_tag_number` (`bag_tag_number`),
  KEY `luggage_ibfk_1` (`passenger_id`),
  KEY `luggage_ibfk_2` (`flight_id`),
  KEY `luggage_ibfk_3` (`ticket_id`),
  CONSTRAINT `luggage_ibfk_1` FOREIGN KEY (`passenger_id`) REFERENCES `passenger` (`passenger_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `luggage_ibfk_2` FOREIGN KEY (`flight_id`) REFERENCES `flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `luggage_ibfk_3` FOREIGN KEY (`ticket_id`) REFERENCES `ticket` (`ticket_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
