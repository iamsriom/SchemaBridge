-- Schema for database: schemabench_0221
CREATE TABLE `consumer` (
  `consumer_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `gender` varchar(20) DEFAULT NULL,
  `city` varchar(255) DEFAULT NULL,
  `state` varchar(255) DEFAULT NULL,
  `country` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`consumer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `restaurant` (
  `restaurant_id` int(11) NOT NULL,
  `restaurant_name` varchar(255) NOT NULL,
  `city` varchar(255) DEFAULT NULL,
  `state` varchar(255) DEFAULT NULL,
  `country` varchar(255) DEFAULT NULL,
  `cuisine` varchar(255) DEFAULT NULL,
  `price_range` int(11) DEFAULT NULL,
  `rating` decimal(3,2) DEFAULT NULL,
  `sentiment_score` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`restaurant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rating` (
  `rating_id` int(11) NOT NULL,
  `consumer_id` int(11) NOT NULL,
  `restaurant_id` int(11) NOT NULL,
  `rating_value` decimal(3,2) NOT NULL,
  `review_text` text DEFAULT NULL,
  `review_date` datetime DEFAULT NULL,
  `sentiment_label` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`rating_id`),
  KEY `rating_ibfk_1` (`consumer_id`),
  KEY `rating_ibfk_2` (`restaurant_id`),
  CONSTRAINT `rating_ibfk_1` FOREIGN KEY (`consumer_id`) REFERENCES `consumer` (`consumer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rating_ibfk_2` FOREIGN KEY (`restaurant_id`) REFERENCES `restaurant` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
