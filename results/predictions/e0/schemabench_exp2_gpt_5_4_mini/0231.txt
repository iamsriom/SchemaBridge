-- Schema for database: schemabench_0231
CREATE TABLE `cities` (
  `city_id` int(11) NOT NULL,
  `city_name` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  PRIMARY KEY (`city_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stadiums` (
  `stadium_id` int(11) NOT NULL,
  `stadium_name` varchar(255) NOT NULL,
  `city_id` int(11) NOT NULL,
  `capacity` int(11) NOT NULL,
  PRIMARY KEY (`stadium_id`),
  KEY `stadiums_ibfk_1` (`city_id`),
  CONSTRAINT `stadiums_ibfk_1` FOREIGN KEY (`city_id`) REFERENCES `cities` (`city_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teams` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  `home_stadium_id` int(11) NOT NULL,
  PRIMARY KEY (`team_id`),
  KEY `teams_ibfk_1` (`home_stadium_id`),
  CONSTRAINT `teams_ibfk_1` FOREIGN KEY (`home_stadium_id`) REFERENCES `stadiums` (`stadium_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `players` (
  `player_id` int(11) NOT NULL,
  `player_name` varchar(255) NOT NULL,
  `nationality` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `jersey_number` int(11) DEFAULT NULL,
  `position` varchar(100) NOT NULL,
  `height_cm` decimal(5,2) DEFAULT NULL,
  `weight_kg` decimal(5,2) DEFAULT NULL,
  `preferred_foot` varchar(10) DEFAULT NULL,
  `team_id` int(11) NOT NULL,
  PRIMARY KEY (`player_id`),
  KEY `players_ibfk_1` (`team_id`),
  CONSTRAINT `players_ibfk_1` FOREIGN KEY (`team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `managers` (
  `manager_id` int(11) NOT NULL,
  `manager_name` varchar(255) NOT NULL,
  `nationality` varchar(255) NOT NULL,
  `team_id` int(11) NOT NULL,
  PRIMARY KEY (`manager_id`),
  KEY `managers_ibfk_1` (`team_id`),
  CONSTRAINT `managers_ibfk_1` FOREIGN KEY (`team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `matches` (
  `match_id` int(11) NOT NULL,
  `season` varchar(9) NOT NULL,
  `match_datetime` datetime NOT NULL,
  `home_team_id` int(11) NOT NULL,
  `away_team_id` int(11) NOT NULL,
  `stadium_id` int(11) NOT NULL,
  `home_score` int(11) NOT NULL,
  `away_score` int(11) NOT NULL,
  `penalty_shootout` char(1) NOT NULL,
  `attendance` int(11) DEFAULT NULL,
  PRIMARY KEY (`match_id`),
  KEY `matches_ibfk_1` (`home_team_id`),
  KEY `matches_ibfk_2` (`away_team_id`),
  KEY `matches_ibfk_3` (`stadium_id`),
  CONSTRAINT `matches_ibfk_1` FOREIGN KEY (`home_team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `matches_ibfk_2` FOREIGN KEY (`away_team_id`) REFERENCES `teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `matches_ibfk_3` FOREIGN KEY (`stadium_id`) REFERENCES `stadiums` (`stadium_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `goals` (
  `goal_id` int(11) NOT NULL,
  `match_id` int(11) NOT NULL,
  `scoring_player_id` int(11) NOT NULL,
  `assist_player_id` int(11) DEFAULT NULL,
  `goal_minute` int(11) NOT NULL,
  `description` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`goal_id`),
  KEY `goals_ibfk_1` (`match_id`),
  KEY `goals_ibfk_2` (`scoring_player_id`),
  KEY `goals_ibfk_3` (`assist_player_id`),
  CONSTRAINT `goals_ibfk_1` FOREIGN KEY (`match_id`) REFERENCES `matches` (`match_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `goals_ibfk_2` FOREIGN KEY (`scoring_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `goals_ibfk_3` FOREIGN KEY (`assist_player_id`) REFERENCES `players` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
