  KEY `taggeditem_tag_id_foreign` (`tag_id`),

  KEY `taggeditem_object_id_foreign` (`object_id`),

  CONSTRAINT `taggeditem_object_id_foreign` FOREIGN KEY (`object_id`) REFERENCES `object` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
