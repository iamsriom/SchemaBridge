CREATE TABLE `django_app` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `http_request` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `django_app_id` int(11) NOT NULL,
  `method` varchar(255) NOT NULL,
  `url` varchar(255) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `django_app_id` (`django_app_id`),
  CONSTRAINT `http_request_ibfk_1` FOREIGN KEY (`django_app_id`) REFERENCES `django_app` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `http_response` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `http_request_id` int(11) NOT NULL,
  `status_code` int(11) NOT NULL,
  `response_time` float NOT NULL,
  PRIMARY KEY (`id`),
  KEY `http_request_id` (`http_request_id`),
  CONSTRAINT `http_response_ibfk_1` FOREIGN KEY (`http_request_id`) REFERENCES `http_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sql_query` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `http_request_id` int(11) NOT NULL,
  `query` text NOT NULL,
  `execution_time` float NOT NULL,
  PRIMARY KEY (`id`),
  KEY `http_request_id` (`http_request_id`),
  CONSTRAINT `sql_query_ibfk_1` FOREIGN KEY (`http_request_id`) REFERENCES `http_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `code_block` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `http_request_id` int(11) NOT NULL,
  `start_line` int(11) NOT NULL,
  `end_line` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `http_request_id` (`http_request_id`),
  CONSTRAINT `code_block_ibfk_1` FOREIGN KEY (`http_request_id`) REFERENCES `http_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
