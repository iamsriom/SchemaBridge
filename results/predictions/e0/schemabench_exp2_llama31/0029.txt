-- Schema for database: schemabench_0029
CREATE TABLE `robots_sites` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `robots_crawlers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `robots_rules` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `site_id` int(11) NOT NULL,
  `crawler_id` int(11) NOT NULL,
  `rule_type` enum('allowed','disallowed') NOT NULL,
  `url_pattern` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `site_id` (`site_id`),
  KEY `crawler_id` (`crawler_id`),
  CONSTRAINT `robots_rules_ibfk_1` FOREIGN KEY (`site_id`) REFERENCES `robots_sites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `robots_rules_ibfk_2` FOREIGN KEY (`crawler_id`) REFERENCES `robots_crawlers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `robots_delays` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `crawler_id` int(11) NOT NULL,
  `delay` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `crawler_id` (`crawler_id`),
  CONSTRAINT `robots_delays_ibfk_1` FOREIGN KEY (`crawler_id`) REFERENCES `robots_crawlers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
