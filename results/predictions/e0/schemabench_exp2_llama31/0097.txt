-- Schema for database: schemabench_0097
CREATE TABLE `schools` (
  `cds_code` char(6) NOT NULL,
  `school_name` varchar(255) NOT NULL,
  `district_name` varchar(255) NOT NULL,
  `location` point NOT NULL,
  `latitude` decimal(10,7) NOT NULL,
  `longitude` decimal(11,8) NOT NULL,
  `admin_level` tinyint(3) unsigned NOT NULL,
  `admin_name` varchar(255) NOT NULL,
  PRIMARY KEY (`cds_code`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sat_scores` (
  `cds_code` char(6) NOT NULL,
  `year` year NOT NULL,
  `average_score` decimal(5,2) NOT NULL,
  `above_1500_percentage` decimal(5,2) NOT NULL,
  PRIMARY KEY (`cds_code`, `year`),
  FOREIGN KEY (`cds_code`) REFERENCES `schools` (`cds_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `frpm_data` (
  `cds_code` char(6) NOT NULL,
  `age_group` enum('K-5', '6-8', '9-12') NOT NULL,
  `percentage_eligible` decimal(5,2) NOT NULL,
  `count_eligible` int(11) NOT NULL,
  PRIMARY KEY (`cds_code`, `age_group`),
  FOREIGN KEY (`cds_code`) REFERENCES `schools` (`cds_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
