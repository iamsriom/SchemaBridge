-- Schema for database: schemabench_0105
CREATE TABLE `sentence` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `text` varchar(255) NOT NULL,
  `classification` enum('positive','negative') NOT NULL DEFAULT 'positive',
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `term` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sentence_term` (
  `sentence_id` int(11) NOT NULL,
  `term_id` int(11) NOT NULL,
  PRIMARY KEY (`sentence_id`, `term_id`),
  KEY `sentence_term_ibfk_1` (`sentence_id`),
  KEY `sentence_term_ibfk_2` (`term_id`),
  CONSTRAINT `sentence_term_ibfk_1` FOREIGN KEY (`sentence_id`) REFERENCES `sentence` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sentence_term_ibfk_2` FOREIGN KEY (`term_id`) REFERENCES `term` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dcg_rule` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sentence_dcg_rule` (
  `sentence_id` int(11) NOT NULL,
  `dcg_rule_id` int(11) NOT NULL,
  PRIMARY KEY (`sentence_id`, `dcg_rule_id`),
  KEY `sentence_dcg_rule_ibfk_1` (`sentence_id`),
  KEY `sentence_dcg_rule_ibfk_2` (`dcg_rule_id`),
  CONSTRAINT `sentence_dcg_rule_ibfk_1` FOREIGN KEY (`sentence_id`) REFERENCES `sentence` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sentence_dcg_rule_ibfk_2` FOREIGN KEY (`dcg_rule_id`) REFERENCES `dcg_rule` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
