-- Schema for database: schemabench_0108
CREATE TABLE `gene` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `essentiality` tinyint(1) NOT NULL,
  `phenotype` varchar(255) DEFAULT NULL,
  `function` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gene_localization` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `gene_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `gene_id` (`gene_id`),
  CONSTRAINT `gene_localization_ibfk_1` FOREIGN KEY (`gene_id`) REFERENCES `gene` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gene_interaction` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `gene1_id` int(11) NOT NULL,
  `gene2_id` int(11) NOT NULL,
  `interaction_type` varchar(255) NOT NULL,
  `expression_correlation` float DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `gene1_id` (`gene1_id`),
  KEY `gene2_id` (`gene2_id`),
  CONSTRAINT `gene_interaction_ibfk_1` FOREIGN KEY (`gene1_id`) REFERENCES `gene` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `gene_interaction_ibfk_2` FOREIGN KEY (`gene2_id`) REFERENCES `gene` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `researcher` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gene_classification` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `gene_id` int(11) NOT NULL,
  `classification` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `gene_id` (`gene_id`),
  CONSTRAINT `gene_classification_ibfk_1` FOREIGN KEY (`gene_id`) REFERENCES `gene` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `study` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `description` text,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gene_study` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `gene_id` int(11) NOT NULL,
  `study_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `gene_id` (`gene_id`),
  KEY `study_id` (`study_id`),
  CONSTRAINT `gene_study_ibfk_1` FOREIGN KEY (`gene_id`) REFERENCES `gene` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `gene_study_ibfk_2` FOREIGN KEY (`study_id`) REFERENCES `study` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_professional` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `diagnostic` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `gene_id` int(11) NOT NULL,
  `medical_professional_id` int(11) NOT NULL,
  `diagnosis` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `gene_id` (`gene_id`),
  KEY `medical_professional_id` (`medical_professional_id`),
  CONSTRAINT `diagnostic_ibfk_1` FOREIGN KEY (`gene_id`) REFERENCES `gene` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `diagnostic_ibfk_2` FOREIGN KEY (`medical_professional_id`) REFERENCES `medical_professional` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
