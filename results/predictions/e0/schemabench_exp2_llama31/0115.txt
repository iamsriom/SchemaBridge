-- Schema for database: schemabench_0115
CREATE TABLE `court` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `judge` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `legalact` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `court_id` int(11) NOT NULL,
  `case_number` varchar(255) NOT NULL,
  `judge_id` int(11) NOT NULL,
  `decision_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `legalact_ibfk_1` (`court_id`),
  KEY `legalact_ibfk_2` (`judge_id`),
  CONSTRAINT `legalact_ibfk_1` FOREIGN KEY (`court_id`) REFERENCES `court` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `legalact_ibfk_2` FOREIGN KEY (`judge_id`) REFERENCES `judge` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `legalact_link` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `legalact_id` int(11) NOT NULL,
  `linked_legalact_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `legalact_link_ibfk_1` (`legalact_id`),
  KEY `legalact_link_ibfk_2` (`linked_legalact_id`),
  CONSTRAINT `legalact_link_ibfk_1` FOREIGN KEY (`legalact_id`) REFERENCES `legalact` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `legalact_link_ibfk_2` FOREIGN KEY (`linked_legalact_id`) REFERENCES `legalact` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `legalact_people` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `legalact_id` int(11) NOT NULL,
  `person_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `legalact_people_ibfk_1` (`legalact_id`),
  KEY `legalact_people_ibfk_2` (`person_id`),
  CONSTRAINT `legalact_people_ibfk_1` FOREIGN KEY (`legalact_id`) REFERENCES `legalact` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `legalact_people_ibfk_2` FOREIGN KEY (`person_id`) REFERENCES `judge` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `scrapefix` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `legalact_id` int(11) NOT NULL,
  `contributor` varchar(255) NOT NULL,
  `fix_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `scrapefix_ibfk_1` (`legalact_id`),
  CONSTRAINT `scrapefix_ibfk_1` FOREIGN KEY (`legalact_id`) REFERENCES `legalact` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
