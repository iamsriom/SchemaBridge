-- Schema for database: schemabench_0118
CREATE TABLE `countries` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country_relations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `country1_id` int(11) NOT NULL,
  `country2_id` int(11) NOT NULL,
  `economic_ties` tinyint(1) NOT NULL DEFAULT 0,
  `political_ties` tinyint(1) NOT NULL DEFAULT 0,
  `military_ties` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  KEY `country1_id` (`country1_id`),
  KEY `country2_id` (`country2_id`),
  CONSTRAINT `country_relations_ibfk_1` FOREIGN KEY (`country1_id`) REFERENCES `countries` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `country_relations_ibfk_2` FOREIGN KEY (`country2_id`) REFERENCES `countries` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country_statistics` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `country_id` int(11) NOT NULL,
  `population` int(11) NOT NULL,
  `gnp` decimal(10,2) NOT NULL,
  `exports` decimal(10,2) NOT NULL,
  `imports` decimal(10,2) NOT NULL,
  `literacy_rate` decimal(5,2) NOT NULL,
  `unemployment_rate` decimal(5,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `country_id` (`country_id`),
  CONSTRAINT `country_statistics_ibfk_1` FOREIGN KEY (`country_id`) REFERENCES `countries` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
