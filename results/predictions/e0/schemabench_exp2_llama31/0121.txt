CREATE TABLE `customers` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `company_name` varchar(255) NOT NULL,
  `contact_name` varchar(255) NOT NULL,
  `contact_title` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `city` varchar(255) NOT NULL,
  `region` varchar(255) NOT NULL,
  `postal_code` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  `phone` varchar(255) NOT NULL,
  `fax` varchar(255) NOT NULL,

  `employee_id` int(11) NOT NULL AUTO_INCREMENT,

  `title_of_courtesy` varchar(255) NOT NULL,
  `birth_date` date NOT NULL,
  `hire_date` date NOT NULL,
  `address` varchar(255) NOT NULL,
  `city` varchar(255) NOT NULL,
  `region` varchar(255) NOT NULL,
  `postal_code` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  `extension` varchar(255) NOT NULL,
  `photo` blob,
  `notes` text,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees_territories` (
  `employee_id` int(11) NOT NULL,
  `territory_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`, `territory_id`),
  KEY `employee_id` (`employee_id`),
  KEY `territory_id` (`territory_id`),
  CONSTRAINT `employees_territories_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employees_territories_ibfk_2` FOREIGN KEY (`territory_id`) REFERENCES `territories` (`territory_id`) ON DELETE CASCADE ON UPDATE CASCADE

  `ship_country` varchar(255) NOT NULL,
  PRIMARY KEY (`order_id`),

  CONSTRAINT `orders_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employees` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `orders_ibfk_3` FOREIGN KEY (`ship_via`) REFERENCES `shippers` (`shipper_id`) ON DELETE CASCADE ON UPDATE CASCADE
