-- Schema for database: schemabench_0123
CREATE TABLE `molecules` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `formula` varchar(255) NOT NULL,
  `flexibility` float NOT NULL,
  `polarizability` float NOT NULL,
  `sigma` float NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inhibitor_activities` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `molecule_id` int(11) NOT NULL,
  `activity` float NOT NULL,
  PRIMARY KEY (`id`),
  KEY `molecule_id` (`molecule_id`),
  CONSTRAINT `inhibitor_activities_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `researchers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pharmaceutical_developers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `data_scientists` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `researcher_molecules` (
  `researcher_id` int(11) NOT NULL,
  `molecule_id` int(11) NOT NULL,
  PRIMARY KEY (`researcher_id`, `molecule_id`),
  KEY `researcher_id` (`researcher_id`),
  KEY `molecule_id` (`molecule_id`),
  CONSTRAINT `researcher_molecules_ibfk_1` FOREIGN KEY (`researcher_id`) REFERENCES `researchers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `researcher_molecules_ibfk_2` FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pharmaceutical_developer_molecules` (
  `pharmaceutical_developer_id` int(11) NOT NULL,

  PRIMARY KEY (`pharmaceutical_developer_id`, `molecule_id`),
  KEY `pharmaceutical_developer_id` (`pharmaceutical_developer_id`),
  KEY `molecule_id` (`molecule_id`),
  CONSTRAINT `pharmaceutical_developer_molecules_ibfk_1` FOREIGN KEY (`pharmaceutical_developer_id`) REFERENCES `pharmaceutical_developers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `pharmaceutical_developer_molecules_ibfk_2` FOREIGN KEY (`molecule_id`) REFERENCES `molecules` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `data_scientist_id` int(11) NOT NULL,
