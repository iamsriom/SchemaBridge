-- Schema for database: schemabench_0135
CREATE TABLE `trains` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `direction` enum('east','west') NOT NULL,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cars` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `train_id` int(11) NOT NULL,
  `position` int(11) NOT NULL,
  `shape` varchar(255) DEFAULT NULL,
  `length` decimal(10,2) DEFAULT NULL,
  `sides` int(11) DEFAULT NULL,
  `roof_type` varchar(255) DEFAULT NULL,
  `num_wheels` int(11) NOT NULL,
  `load_shape` varchar(255) DEFAULT NULL,
  `load_quantity` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `train_id` (`train_id`),
  CONSTRAINT `cars_ibfk_1` FOREIGN KEY (`train_id`) REFERENCES `trains` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `train_cars` (
  `train_id` int(11) NOT NULL,
  `car_id` int(11) NOT NULL,
  PRIMARY KEY (`train_id`, `car_id`),
  KEY `train_id` (`train_id`),
  KEY `car_id` (`car_id`),
  CONSTRAINT `train_cars_ibfk_1` FOREIGN KEY (`train_id`) REFERENCES `trains` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `train_cars_ibfk_2` FOREIGN KEY (`car_id`) REFERENCES `cars` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
