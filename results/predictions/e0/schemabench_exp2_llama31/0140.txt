-- Schema for database: schemabench_0140
CREATE TABLE `chambers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `construction_details` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `vessel_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `length` decimal(10,2) NOT NULL,
  `beam` decimal(10,2) NOT NULL,
  `tonnage` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `construction_details_ibfk_1` (`vessel_id`),
  CONSTRAINT `construction_details_ibfk_1` FOREIGN KEY (`vessel_id`) REFERENCES `vessels` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `invoices` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `voyage_id` int(11) NOT NULL,
  `chamber_id` int(11) NOT NULL,
  `date` date NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `invoices_ibfk_1` (`voyage_id`),
  KEY `invoices_ibfk_2` (`chamber_id`),
  CONSTRAINT `invoices_ibfk_1` FOREIGN KEY (`voyage_id`) REFERENCES `voyages` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `invoices_ibfk_2` FOREIGN KEY (`chamber_id`) REFERENCES `chambers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `voyage_id` int(11) NOT NULL,
  `category` enum('craftsman','impotenten','passenger','seafarer','soldier') NOT NULL,
  `status_at_departure` enum('present','absent') NOT NULL,
  `status_at_arrival` enum('present','absent') NOT NULL,
  `status_at_cape_stop` enum('present','absent') NOT NULL,
  `casualty` enum('yes','no') NOT NULL,
  PRIMARY KEY (`id`),
  KEY `personnel_ibfk_1` (`voyage_id`),
  CONSTRAINT `personnel_ibfk_1` FOREIGN KEY (`voyage_id`) REFERENCES `voyages` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vessels` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `master` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voyages` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `vessel_id` int(11) NOT NULL,
  `departure_date` date NOT NULL,
  `arrival_date` date NOT NULL,
  PRIMARY KEY (`id`),
  KEY `voyages_ibfk_1` (`vessel_id`),
  CONSTRAINT `voyages_ibfk_1` FOREIGN KEY (`vessel_id`) REFERENCES `vessels` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
