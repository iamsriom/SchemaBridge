CREATE TABLE `aircraft` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_model` varchar(255) NOT NULL,
  `aircraft_manufacturer` varchar(255) NOT NULL,
  `aircraft_type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `aircraft_engine` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_id` int(11) NOT NULL,
  `engine_model` varchar(255) NOT NULL,
  `engine_manufacturer` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `aircraft_id` (`aircraft_id`),
  CONSTRAINT `aircraft_engine_ibfk_1` FOREIGN KEY (`aircraft_id`) REFERENCES `aircraft` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_engine_id` int(11) NOT NULL,
  `flight_date` date NOT NULL,
  `flight_duration` int(11) NOT NULL,
  `flying_hours` decimal(10,2) NOT NULL,
  `grounded_status` tinyint(1) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `aircraft_engine_id` (`aircraft_engine_id`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`aircraft_engine_id`) REFERENCES `aircraft_engine` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `covid_impact` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `flight_id` int(11) NOT NULL,
  `covid_impact_date` date NOT NULL,
  `impact_type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `flight_id` (`flight_id`),
  CONSTRAINT `covid_impact_ibfk_1` FOREIGN KEY (`flight_id`) REFERENCES `flight` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
