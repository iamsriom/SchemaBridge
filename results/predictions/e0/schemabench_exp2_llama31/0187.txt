  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `distributors` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dealerships` (
  `id` int(11) NOT NULL AUTO_INCREMENT,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `dealership_id` int(11) NOT NULL,
  `distributor_id` int(11) NOT NULL,
  PRIMARY KEY (`dealership_id`, `distributor_id`),
  KEY `dealership_distributors_ibfk_1` (`dealership_id`),
  KEY `dealership_distributors_ibfk_2` (`distributor_id`),
  CONSTRAINT `dealership_distributors_ibfk_1` FOREIGN KEY (`dealership_id`) REFERENCES `dealerships` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `dealership_distributors_ibfk_2` FOREIGN KEY (`distributor_id`) REFERENCES `distributors` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
