-- Schema for database: schemabench_0216
CREATE TABLE `game` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `release_year` int(4) NOT NULL,
  `sales_volume` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `game_title_unique` (`title`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `game_id` int(11) NOT NULL,
  `critic_score` decimal(3,2) NOT NULL,
  `user_score` decimal(3,2) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `review_game_id_foreign` (`game_id`),
  CONSTRAINT `review_game_id_foreign` FOREIGN KEY (`game_id`) REFERENCES `game` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `yearly_summary` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `year` int(4) NOT NULL,
  `average_critic_score` decimal(3,2) NOT NULL,
  `average_user_score` decimal(3,2) NOT NULL,
  `top_selling_games` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `yearly_summary_year_unique` (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
