-- Schema for database: schemabench_0218
CREATE TABLE `grampanchayats` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `taluka_name` varchar(255) NOT NULL,
  `total_wards` int(11) NOT NULL,
  `total_villages` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `villages` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `grampanchayat_id` int(11) NOT NULL,
  `ward_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `villages_ibfk_1` (`grampanchayat_id`),
  KEY `villages_ibfk_2` (`ward_id`),
  CONSTRAINT `villages_ibfk_1` FOREIGN KEY (`grampanchayat_id`) REFERENCES `grampanchayats` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `villages_ibfk_2` FOREIGN KEY (`ward_id`) REFERENCES `wards` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `wards` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `grampanchayat_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `wards_ibfk_1` (`grampanchayat_id`),
  CONSTRAINT `wards_ibfk_1` FOREIGN KEY (`grampanchayat_id`) REFERENCES `grampanchayats` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `farmers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aadhaar_no` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `machines` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rentals` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `machine_id` int(11) NOT NULL,
  `farmer_id` int(11) NOT NULL,
  `ward_id` int(11) NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `rentals_ibfk_1` (`machine_id`),
  KEY `rentals_ibfk_2` (`farmer_id`),
  KEY `rentals_ibfk_3` (`ward_id`),
  CONSTRAINT `rentals_ibfk_1` FOREIGN KEY (`machine_id`) REFERENCES `machines` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rentals_ibfk_2` FOREIGN KEY (`farmer_id`) REFERENCES `farmers` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rentals_ibfk_3` FOREIGN KEY (`ward_id`) REFERENCES `wards` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bills` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `rental_id` int(11) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `payment_method` varchar(255) NOT NULL,
  `payment_date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `bills_ibfk_1` (`rental_id`),
  CONSTRAINT `bills_ibfk_1` FOREIGN KEY (`rental_id`) REFERENCES `rentals` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
