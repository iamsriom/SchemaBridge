  `id` int(11) NOT NULL AUTO_INCREMENT,

  `address` text NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `check_in_date` datetime NOT NULL,

  `board_type` enum('single','double','suite') NOT NULL,
  `payment_method` enum('cash','credit_card') NOT NULL,

  KEY `guest_id` (`guest_id`),
  KEY `hotel_id` (`hotel_id`),
  KEY `room_id` (`room_id`),
  CONSTRAINT `reservations_ibfk_1` FOREIGN KEY (`guest_id`) REFERENCES `guests` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reservations_ibfk_2` FOREIGN KEY (`hotel_id`) REFERENCES `hotels` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reservations_ibfk_3` FOREIGN KEY (`room_id`) REFERENCES `rooms` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
