-- Schema for database: schemabench_0225
CREATE TABLE `client` (
  `client_number` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone` varchar(20) NOT NULL,
  `address` varchar(255) NOT NULL,
  `membership_type` varchar(50) NOT NULL,
  `discount` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`client_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `duration` int(11) NOT NULL,
  `fee` decimal(10,2) NOT NULL,
  `prerequisites` varchar(255) DEFAULT NULL,
  `hardware_requirements` varchar(255) DEFAULT NULL,
  `software_requirements` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_book` (
  `book_id` int(11) NOT NULL AUTO_INCREMENT,
  `isbn` varchar(20) NOT NULL,
  `title` varchar(255) NOT NULL,
  `author` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `vendor_order_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`book_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_enrollment` (
  `enrollment_id` int(11) NOT NULL AUTO_INCREMENT,
  `client_number` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `enrollment_date` datetime NOT NULL,
  `attendance_confirmed` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`enrollment_id`),
  KEY `client_number` (`client_number`),
  KEY `course_id` (`course_id`),
  CONSTRAINT `course_enrollment_ibfk_1` FOREIGN KEY (`client_number`) REFERENCES `client` (`client_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_enrollment_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_trainer` (
  `course_id` int(11) NOT NULL,
  `trainer_number` int(11) NOT NULL,
  PRIMARY KEY (`course_id`, `trainer_number`),
  KEY `course_id` (`course_id`),
  KEY `trainer_number` (`trainer_number`),
  CONSTRAINT `course_trainer_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_trainer_ibfk_2` FOREIGN KEY (`trainer_number`) REFERENCES `trainer` (`trainer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `enrollment_waitlist` (
  `waitlist_id` int(11) NOT NULL AUTO_INCREMENT,
  `enrollment_id` int(11) NOT NULL,
  `waitlist_date` datetime NOT NULL,
  PRIMARY KEY (`waitlist_id`),
  KEY `enrollment_id` (`enrollment_id`),
  CONSTRAINT `enrollment_waitlist_ibfk_1` FOREIGN KEY (`enrollment_id`) REFERENCES `course_enrollment` (`enrollment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hardware` (
  `hardware_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `os` varchar(50) NOT NULL,
  `license_number` varchar(20) NOT NULL,
  PRIMARY KEY (`hardware_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `order_id` int(11) NOT NULL AUTO_INCREMENT,
  `book_id` int(11) NOT NULL,
  `vendor_order_id` int(11) NOT NULL,
  `order_date` datetime NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`order_id`),
  KEY `book_id` (`book_id`),
  KEY `vendor_order_id` (`vendor_order_id`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`book_id`) REFERENCES `course_book` (`book_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_ibfk_2` FOREIGN KEY (`vendor_order_id`) REFERENCES `course_book` (`vendor_order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trainer` (
  `trainer_number` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `employee_number` varchar(20) NOT NULL,
  `hire_date` date NOT NULL,
  `qualification` varchar(255) NOT NULL,
  PRIMARY KEY (`trainer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trainer_profile` (
  `trainer_number` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone` varchar(20) NOT NULL,
  PRIMARY KEY (`trainer_number`),
  KEY `trainer_number` (`trainer_number`),
  CONSTRAINT `trainer_profile_ibfk_1` FOREIGN KEY (`trainer_number`) REFERENCES `trainer` (`trainer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trainer_schedule` (
  `schedule_id` int(11) NOT NULL AUTO_INCREMENT,
  `trainer_number` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `classroom` varchar(50) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  PRIMARY KEY (`schedule_id`),
  KEY `trainer_number` (`trainer_number`),
  KEY `course_id` (`course_id`),
  CONSTRAINT `trainer_schedule_ibfk_1` FOREIGN KEY (`trainer_number`) REFERENCES `trainer` (`trainer_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trainer_schedule_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
