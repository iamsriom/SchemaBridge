-- Schema for database: schemabench_0232
CREATE TABLE `aircraft` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `registration` varchar(255) NOT NULL,
  `flight_number` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `channel` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inmarsat_log` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `aircraft_id` int(11) NOT NULL,
  `channel_id` int(11) NOT NULL,
  `timestamp` datetime NOT NULL,
  `burst_timing_offset` int(11) NOT NULL,
  `burst_frequency_offset` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `aircraft_id` (`aircraft_id`),
  KEY `channel_id` (`channel_id`),
  CONSTRAINT `inmarsat_log_ibfk_1` FOREIGN KEY (`aircraft_id`) REFERENCES `aircraft` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `inmarsat_log_ibfk_2` FOREIGN KEY (`channel_id`) REFERENCES `channel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ocean_region` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inmarsat_log_ocean_region` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `inmarsat_log_id` int(11) NOT NULL,
  `ocean_region_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `inmarsat_log_id` (`inmarsat_log_id`),
  KEY `ocean_region_id` (`ocean_region_id`),
  CONSTRAINT `inmarsat_log_ocean_region_ibfk_1` FOREIGN KEY (`inmarsat_log_id`) REFERENCES `inmarsat_log` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `inmarsat_log_ocean_region_ibfk_2` FOREIGN KEY (`ocean_region_id`) REFERENCES `ocean_region` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
