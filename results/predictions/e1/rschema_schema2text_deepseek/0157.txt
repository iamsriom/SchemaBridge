```python
# This is a sample Python code to create the schema described above.
# Please note that this is a simplified version and may need to be adjusted based on your specific needs.
import sqlite3
conn = sqlite3.connect('rschema_0157.db')
c = conn.cursor()
c.execute('''
CREATE TABLE School (
School_Facilities VARCHAR(255),
School_ID INTEGER PRIMARY KEY,
School_Introduction VARCHAR(255),
School_Name VARCHAR(255)
)
''')
c.execute('''
CREATE TABLE Department (
Department_ID INTEGER PRIMARY KEY,
Department_Introduction VARCHAR(255),
Department_Name VARCHAR(255)
)
''')
c.execute('''
CREATE TABLE Class (
Class_ID INTEGER PRIMARY KEY,
Class_Introduction VARCHAR(255),
Class_Name VARCHAR(255)
)
''')
c.execute('''
CREATE TABLE Teacher (
Teacher_ID INTEGER PRIMARY KEY,
Teacher_Introduction VARCHAR(255),
Teacher_Name VARCHAR(255),
Teacher_Title VARCHAR(255)
)
''')
c.execute('''
CREATE TABLE Course (
Course_Credits INTEGER,
Course_Description VARCHAR(255),
Course_ID INTEGER PRIMARY KEY,
Course_Name VARCHAR(255)
)
''')
c.execute('''
CREATE TABLE Dormitory (
Dormitory_Bed_Count INTEGER,
Dormitory_Building VARCHAR(255),
Dormitory_ID INTEGER PRIMARY KEY,
Dormitory_Room_Number INTEGER
)
''')
c.execute('''
CREATE TABLE Course_Schedule (
Class_Location VARCHAR(255),
Class_Time DATETIME,
Course_ID INTEGER NOT NULL,
Teacher_ID INTEGER NOT NULL,
PRIMARY KEY (Teacher_ID, Course_ID),
FOREIGN KEY (Course_ID) REFERENCES Course(Course_ID),
FOREIGN KEY (Teacher_ID) REFERENCES Teacher(Teacher_ID)
)
''')
c.execute('''
CREATE TABLE Student (
Student_Class INTEGER NOT NULL,
Student_Department INTEGER NOT NULL,
Student_Dormitory INTEGER NOT NULL,
Student_Gender BOOL,
Student_ID INTEGER PRIMARY KEY,
Student_Name VARCHAR(255),
FOREIGN KEY (Student_Class) REFERENCES Class(Class_ID),
FOREIGN KEY (Student_Department) REFERENCES Department(Department_ID),
FOREIGN KEY (Student_Dormitory) REFERENCES Dormitory(Dormitory_ID)
)
''')
conn.commit()
conn.close()
```
