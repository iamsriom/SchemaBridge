```python
# This is a sample Python code to create the schema using SQLAlchemy ORM
from sqlalchemy import Column, Integer, String, Float, DateTime, ForeignKey
from sqlalchemy.orm import relationship
from sqlalchemy.ext.declarative import declarative_base
Base = declarative_base()
class Customer(Base):
__tablename__ = 'Customer'
Customer_Contact_Information = Column(String(255))
Customer_ID = Column(Integer, primary_key=True)
Customer_Name = Column(String(255))
class Product(Base):
__tablename__ = 'Product'
Product_ID = Column(Integer, primary_key=True)
Product_Inventory_Quantity = Column(Integer)
Product_Name = Column(String(255))
Product_Price = Column(Float)
class Order(Base):
__tablename__ = 'Order'
Order_ID = Column(Integer, primary_key=True)
Order_Status = Column(String(255))
Ordering_Customer_ID = Column(Integer, ForeignKey('Customer.Customer_ID'))
class Ordered_Product(Base):
__tablename__ = 'Ordered_Product'
Order_ID = Column(Integer, ForeignKey('Order.Order_ID'), primary_key=True)
Ordered_Product_ID = Column(Integer, ForeignKey('Product.Product_ID'), primary_key=True)
Ordered_Quantity = Column(Integer)
class Picking_List(Base):
__tablename__ = 'Picking_List'
Order_ID = Column(Integer, ForeignKey('Order.Order_ID'))
Picking_List_ID = Column(Integer, primary_key=True)
Picking_Status = Column(String(255))
class Delivery_Order(Base):
__tablename__ = 'Delivery_Order'
Delivery_Order_ID = Column(Integer, primary_key=True)
Delivery_Status = Column(String(255))
Delivery_Time = Column(DateTime)
class Customer_Invoice(Base):
__tablename__ = 'Customer_Invoice'
Customer_Invoice_ID = Column(Integer, primary_key=True)
Invoice_Amount = Column(Float)
Invoice_Date = Column(DateTime)
Order_ID = Column(Integer, ForeignKey('Order.Order_ID'))
class Accounts_Receivable_Report(Base):
__tablename__ = 'Accounts_Receivable_Report'
Order_ID = Column(Integer, ForeignKey('Order.Order_ID'))
Receivable_Amount = Column(Float)
Report_Date = Column(DateTime)
Report_ID = Column(Integer, primary_key=True)
class Picked_Product(Base):
__tablename__ = 'Picked_Product'
Picked_Product_ID = Column(Integer, ForeignKey('Product.Product_ID'), primary_key=True)
Picked_Product_Quantity = Column(Integer)
Picking_List_ID = Column(Integer, ForeignKey('Picking_List.Picking_List_ID'), primary_key=True)
```
