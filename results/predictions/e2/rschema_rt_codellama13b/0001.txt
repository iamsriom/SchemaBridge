-- Schema for database: rschema_0001
CREATE TABLE `airplane` (
  `airplane_number` int(11) NOT NULL,
  `model` varchar(255) DEFAULT NULL,
  `passenger_capacity` int(11) DEFAULT NULL,
  PRIMARY KEY (`airplane_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crew` (
  `name` varchar(255) DEFAULT NULL,
  `personnel_id` int(11) NOT NULL,
  `phone_number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `airplane_number` int(11) NOT NULL,
  `arrival_time` datetime DEFAULT NULL,
  `captain_id` int(11) NOT NULL,
  `departure_time` datetime DEFAULT NULL,
  `destination` varchar(255) DEFAULT NULL,
  `end_time` datetime DEFAULT NULL,
  `first_officer_id` int(11) NOT NULL,
  `flight_number` int(11) NOT NULL,
  `origin` varchar(255) DEFAULT NULL,
  `start_time` datetime DEFAULT NULL,
  PRIMARY KEY (`flight_number`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`airplane_number`) REFERENCES `airplane` (`airplane_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_2` FOREIGN KEY (`captain_id`) REFERENCES `crew` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_3` FOREIGN KEY (`first_officer_id`) REFERENCES `crew` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
