-- Schema for database: rschema_0007
CREATE TABLE `advertising_client` (
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone` varchar(20) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advertisement` (
  `advertisement_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `client_id` int(11) NOT NULL,
  PRIMARY KEY (`advertisement_id`),
  CONSTRAINT `advertisement_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `advertising_client` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `client_broadcast_record` (
  `advertisement_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `broadcast_time` datetime NOT NULL,
  `broadcast_channel` varchar(255) NOT NULL,
  `operating_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`advertisement_id`, `client_id`),
  CONSTRAINT `client_broadcast_record_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `advertising_client` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `client_broadcast_record_ibfk_2` FOREIGN KEY (`advertisement_id`) REFERENCES `advertisement` (`advertisement_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
