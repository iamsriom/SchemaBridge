-- Schema for database: rschema_0010
CREATE TABLE `member` (
  `member_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `date_of_birth` date NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meeting` (
  `meeting_id` int(11) NOT NULL,
  `meeting_type` varchar(255) NOT NULL,
  `time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `meeting_content` text NOT NULL,
  `meeting_topic` varchar(255) NOT NULL,
  PRIMARY KEY (`meeting_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meeting_participation` (
  `member_id` int(11) NOT NULL,
  `meeting_id` int(11) NOT NULL,
  `remarks` text NOT NULL,
  `attendance_time` datetime NOT NULL,
  PRIMARY KEY (`member_id`, `meeting_id`),
  CONSTRAINT `meeting_participation_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `meeting_participation_ibfk_2` FOREIGN KEY (`meeting_id`) REFERENCES `meeting` (`meeting_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
