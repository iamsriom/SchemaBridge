-- Schema for database: rschema_0013
CREATE TABLE `weather_station` (
  `weather_station_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`weather_station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meteorological_data` (
  `data_id` int(11) NOT NULL,
  `humidity` float DEFAULT NULL,
  `precipitation` float DEFAULT NULL,
  `recording_date` datetime NOT NULL,
  `temperature` float DEFAULT NULL,
  `wind_direction` varchar(255) DEFAULT NULL,
  `wind_speed` float DEFAULT NULL,
  `weather_station_id` int(11) NOT NULL,
  PRIMARY KEY (`data_id`),
  CONSTRAINT `meteorological_data_ibfk_1` FOREIGN KEY (`weather_station_id`) REFERENCES `weather_station` (`weather_station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subscription_data` (
  `farm_name` varchar(255) NOT NULL,
  `subscription_date` datetime NOT NULL,
  `unsubscription_date` datetime DEFAULT NULL,
  `weather_station_id` int(11) NOT NULL,
  PRIMARY KEY (`farm_name`, `weather_station_id`),
  CONSTRAINT `subscription_data_ibfk_1` FOREIGN KEY (`weather_station_id`) REFERENCES `weather_station` (`weather_station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
