-- Schema for database: rschema_0014
CREATE TABLE `irrigation_system` (
  `system_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `installation_date` datetime NOT NULL,
  `maintenance_records` text DEFAULT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`system_id`),
  CONSTRAINT `irrigation_system_ibfk_1` FOREIGN KEY (`type`) REFERENCES `irrigation_system` (`type`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL,
  `area` float NOT NULL,
  `irrigation_status` varchar(255) NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  PRIMARY KEY (`plot_id`),
  CONSTRAINT `plot_ibfk_1` FOREIGN KEY (`soil_type`) REFERENCES `plot` (`soil_type`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `irrigation_record` (
  `irrigation_date` datetime NOT NULL,
  `system_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `amount_of_water_used` float NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`irrigation_date`, `system_id`, `plot_id`),
  CONSTRAINT `irrigation_record_ibfk_1` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `irrigation_record_ibfk_2` FOREIGN KEY (`system_id`) REFERENCES `irrigation_system` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
