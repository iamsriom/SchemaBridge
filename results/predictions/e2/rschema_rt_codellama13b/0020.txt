-- Schema for database: rschema_0020
CREATE TABLE `Supplier` (
  `Supplier_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  `Address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Supplier_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Ingredient` (
  `Ingredient_ID` int(11) NOT NULL,
  `Ingredient_Name` varchar(255) NOT NULL,
  `Ingredient_Specification` varchar(255) DEFAULT NULL,
  `Price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`Ingredient_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Purchase_Record` (
  `Purchase_ID` int(11) NOT NULL,
  `Supplier_ID` int(11) NOT NULL,
  `Ingredient_ID` int(11) NOT NULL,
  `Purchase_Date` datetime DEFAULT NULL,
  `Purchase_Quantity` int(11) NOT NULL,
  `Purchase_Amount` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`Supplier_ID`, `Ingredient_ID`),
  CONSTRAINT `purchase_record_ibfk_1` FOREIGN KEY (`Supplier_ID`) REFERENCES `Supplier` (`Supplier_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `purchase_record_ibfk_2` FOREIGN KEY (`Ingredient_ID`) REFERENCES `Ingredient` (`Ingredient_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
