-- Schema for database: rschema_0025
CREATE TABLE `tobacco_products` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `production_date` datetime NOT NULL,
  `shelf_life` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  `production_batch_number` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `total_people` int(11) NOT NULL,
  `person_in_charge_id` int(11) NOT NULL,
  `responsibilities` text NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `departments_ibfk_1` FOREIGN KEY (`person_in_charge_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone` varchar(15) NOT NULL,
  `position` varchar(255) NOT NULL,
  `joining_date` datetime NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `employees_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_records` (
  `id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  `salesperson_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  `customer_phone` varchar(15) NOT NULL,
  `customer_address` text NOT NULL,
  `sales_type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `production_records_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `tobacco_products` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_records_ibfk_2` FOREIGN KEY (`salesperson_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_record_details` (
  `id` int(11) NOT NULL,
  `producer_id` int(11) NOT NULL,
  `production_record_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `production_record_details_ibfk_1` FOREIGN KEY (`production_record_id`) REFERENCES `production_records` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_record_details_ibfk_2` FOREIGN KEY (`producer_id`) REFERENCES `employees` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
