-- Schema for database: rschema_0028
CREATE TABLE `timber` (
  `timber_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `procurement_price` decimal(10,2) NOT NULL,
  `processing_cost` decimal(10,2) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`timber_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `supervisor` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `position` varchar(255) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timber_procurement` (
  `procurement_id` int(11) NOT NULL,
  `timber_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `supplier` varchar(255) NOT NULL,
  `procurement_time` datetime NOT NULL,
  PRIMARY KEY (`procurement_id`),
  CONSTRAINT `timber_procurement_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `timber_procurement_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timber_processing` (
  `processing_id` int(11) NOT NULL,
  `timber_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `processing_time` datetime NOT NULL,
  PRIMARY KEY (`processing_id`),
  CONSTRAINT `timber_processing_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `timber_processing_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
