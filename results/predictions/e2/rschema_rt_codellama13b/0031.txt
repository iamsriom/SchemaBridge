-- Schema for database: rschema_0031
CREATE TABLE `Moving_Team` (
  `Team_ID` int(11) NOT NULL,
  `Team_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Team_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Mover` (
  `Worker_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Age` int(11) NOT NULL,
  `Gender` char(6) NOT NULL,
  `Phone_Number` varchar(15) NOT NULL,
  `Assigned_Team_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Worker_ID`),
  CONSTRAINT `Mover_ibfk_1` FOREIGN KEY (`Assigned_Team_ID`) REFERENCES `Moving_Team` (`Team_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Moving_Task` (
  `Task_ID` int(11) NOT NULL,
  `Assigned_Team_ID` int(11) DEFAULT NULL,
  `Customer_Name` varchar(255) NOT NULL,
  `Customer_Phone_Number` varchar(15) NOT NULL,
  `Customer_Feedback` text DEFAULT NULL,
  `Starting_Address` varchar(255) NOT NULL,
  `Destination_Address` varchar(255) NOT NULL,
  `Moving_Date` datetime NOT NULL,
  `Moving_Cost` decimal(10,2) NOT NULL,
  `Quantity_of_Items_Moved` int(11) NOT NULL,
  PRIMARY KEY (`Task_ID`),
  CONSTRAINT `Moving_Task_ibfk_1` FOREIGN KEY (`Assigned_Team_ID`) REFERENCES `Moving_Team` (`Team_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
