-- Schema for database: rschema_0034
CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL,
  `equipment_name` varchar(255) DEFAULT NULL,
  `model` varchar(255) DEFAULT NULL,
  `purchase_date` datetime DEFAULT NULL,
  `purchase_price` float DEFAULT NULL,
  `status` enum('active', 'inactive', 'decommissioned') DEFAULT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `creation_time` datetime DEFAULT NULL,
  `project_id` int(11) NOT NULL,
  `project_description` blob DEFAULT NULL,
  `project_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment_usage_record` (
  `equipment_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `usage_date` datetime NOT NULL,
  `usage_record_id` int(11) DEFAULT NULL,
  `usage_time` datetime DEFAULT NULL,
  `user` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`project_id`, `equipment_id`, `usage_date`),
  CONSTRAINT `equipment_usage_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `equipment_usage_record_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE RESTRICT ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
