-- Schema for database: rschema_0037
CREATE TABLE `civil_engineering_project` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  `project_manager` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `expected_completion_date` datetime DEFAULT NULL,
  `actual_completion_date` datetime DEFAULT NULL,
  `project_budget` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspector` (
  `inspector_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`inspector_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_quality_inspection` (
  `project_id` int(11) NOT NULL,
  `inspector_id` int(11) NOT NULL,
  `inspection_date` datetime NOT NULL,
  `inspection_content` text NOT NULL,
  `inspection_result` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`, `inspector_id`, `inspection_date`),
  CONSTRAINT `project_quality_inspection_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `civil_engineering_project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_quality_inspection_ibfk_2` FOREIGN KEY (`inspector_id`) REFERENCES `inspector` (`inspector_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
