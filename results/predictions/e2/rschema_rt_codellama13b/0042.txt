-- Schema for database: rschema_0042
CREATE TABLE `community` (
  `community_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `property_manager_number` int(11) NOT NULL,
  PRIMARY KEY (`community_number`),
  CONSTRAINT `community_ibfk_1` FOREIGN KEY (`property_manager_number`) REFERENCES `property_manager` (`property_manager_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property_manager` (
  `property_manager_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`property_manager_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resident` (
  `resident_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `email` varchar(255) NOT NULL,
  `community_number` int(11) NOT NULL,
  `building_number` int(11) NOT NULL,
  PRIMARY KEY (`resident_number`),
  CONSTRAINT `resident_ibfk_1` FOREIGN KEY (`community_number`) REFERENCES `community` (`community_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resident_ibfk_2` FOREIGN KEY (`building_number`) REFERENCES `building` (`building_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `maintenance_order_number` int(11) NOT NULL,
  `community_number` int(11) NOT NULL,
  `resident_number` int(11) NOT NULL,
  `maintenance_content` text NOT NULL,
  PRIMARY KEY (`maintenance_order_number`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`community_number`) REFERENCES `community` (`community_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_2` FOREIGN KEY (`resident_number`) REFERENCES `resident` (`resident_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `maintenance_personnel_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`maintenance_personnel_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `building` (
  `building_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`building_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `worker_handles_maintenance_records` (
  `maintenance_order_number` int(11) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `maintenance_staff_number` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_order_number`),
  CONSTRAINT `worker_handles_maintenance_records_ibfk_1` FOREIGN KEY (`maintenance_order_number`) REFERENCES `maintenance_record` (`maintenance_order_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `worker_handles_maintenance_records_ibfk_2` FOREIGN KEY (`maintenance_staff_number`) REFERENCES `maintenance_personnel` (`maintenance_personnel_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
