-- Schema for database: rschema_0043
CREATE TABLE `loan_application` (
  `application_number` int(11) NOT NULL,
  `property_number` varchar(255) DEFAULT NULL,
  `loan_amount` decimal(10,2) DEFAULT NULL,
  `loan_term` int(11) DEFAULT NULL,
  `approval_status` varchar(50) DEFAULT NULL,
  `applicant_name` varchar(255) DEFAULT NULL,
  `applicant_phone_number` varchar(15) DEFAULT NULL,
  `applicant_id_number` varchar(50) DEFAULT NULL,
  `application_time` datetime DEFAULT NULL,
  PRIMARY KEY (`application_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_officer` (
  `approval_officer_number` int(11) NOT NULL,
  `approval_officer_name` varchar(255) DEFAULT NULL,
  `approval_officer_phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`approval_officer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval` (
  `approval_time` datetime NOT NULL,
  `loan_application_number` int(11) NOT NULL,
  `approval_officer_number` int(11) NOT NULL,
  `approval_result` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`approval_time`, `loan_application_number`, `approval_officer_number`),
  CONSTRAINT `loan_approval_ibfk_1` FOREIGN KEY (`loan_application_number`) REFERENCES `loan_application` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_ibfk_2` FOREIGN KEY (`approval_officer_number`) REFERENCES `approval_officer` (`approval_officer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
