-- Schema for database: rschema_0050
CREATE TABLE `Movie` (
  `Movie_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `release_date` datetime NOT NULL,
  PRIMARY KEY (`Movie_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Copyright_Holder` (
  `Holder_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`Holder_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `License_Movie` (
  `Movie_ID` int(11) NOT NULL,
  `Holder_ID` int(11) NOT NULL,
  `license_type` varchar(100) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`Movie_ID`, `Holder_ID`),
  CONSTRAINT `License_Movie_ibfk_1` FOREIGN KEY (`Movie_ID`) REFERENCES `Movie` (`Movie_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `License_Movie_ibfk_2` FOREIGN KEY (`Holder_ID`) REFERENCES `Copyright_Holder` (`Holder_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
