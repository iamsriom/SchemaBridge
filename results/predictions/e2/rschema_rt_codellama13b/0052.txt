-- Schema for database: rschema_0052
CREATE TABLE `land` (
  `land_id` int(11) NOT NULL,
  `land_location` varchar(255) NOT NULL,
  `total_land_area` float NOT NULL,
  PRIMARY KEY (`land_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `purpose` varchar(255) NOT NULL,
  `responsible_person` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_change` (
  `change_id` int(11) NOT NULL,
  `change_content` varchar(255) NOT NULL,
  `change_cost` float NOT NULL,
  `change_reason` varchar(255) NOT NULL,
  `change_time` datetime NOT NULL,
  `land_id` int(11) NOT NULL,
  PRIMARY KEY (`change_id`),
  CONSTRAINT `land_change_ibfk_1` FOREIGN KEY (`land_id`) REFERENCES `land` (`land_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_usage_status` (
  `land_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `current_status` varchar(255) NOT NULL,
  `land_usage_area` float NOT NULL,
  `record_personnel` varchar(255) NOT NULL,
  `record_time` datetime NOT NULL,
  `usage_years` int(11) NOT NULL,
  PRIMARY KEY (`land_id`, `project_id`, `record_time`),
  CONSTRAINT `land_usage_status_ibfk_1` FOREIGN KEY (`land_id`) REFERENCES `land` (`land_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `land_usage_status_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
