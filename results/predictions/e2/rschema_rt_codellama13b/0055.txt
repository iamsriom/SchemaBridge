-- Schema for database: rschema_0055
CREATE TABLE `Operator` (
  `operator_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Electricity_Usage_Record` (
  `operator_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `electricity_consumption_kwh` decimal(10,2) NOT NULL,
  PRIMARY KEY (`operator_id`, `user_id`, `record_time`),
  UNIQUE KEY `unique_operator_user_time` (`operator_id`, `user_id`, `record_time`),
  CONSTRAINT `electricity_usage_record_ibfk_1` FOREIGN KEY (`operator_id`) REFERENCES `Operator` (`operator_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `electricity_usage_record_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
