-- Schema for database: rschema_0056
CREATE TABLE `Operator` (
  `Operator_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  `Department` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Operator_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  `Address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Billing_Record` (
  `User_ID` int(11) NOT NULL,
  `Bill_Date` datetime NOT NULL,
  `Bill_Amount` decimal(10,2) NOT NULL,
  `Operator_ID` int(11) NOT NULL,
  PRIMARY KEY (`User_ID`, `Bill_Date`),
  CONSTRAINT `billing_record_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `billing_record_ibfk_2` FOREIGN KEY (`Operator_ID`) REFERENCES `Operator` (`Operator_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
