-- Schema for database: rschema_0058
CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL,
  `equipment_model` varchar(255) DEFAULT NULL,
  `equipment_name` varchar(255) DEFAULT NULL,
  `installation_date` datetime DEFAULT NULL,
  `installation_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspection_personnel` (
  `inspection_personnel_id` int(11) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`inspection_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inspection_record` (
  `equipment_id` int(11) NOT NULL,
  `inspection_personnel_id` int(11) NOT NULL,
  `inspection_content` varchar(255) DEFAULT NULL,
  `inspection_date` datetime NOT NULL,
  PRIMARY KEY (`equipment_id`, `inspection_personnel_id`, `inspection_date`),
  CONSTRAINT `inspection_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `inspection_record_ibfk_2` FOREIGN KEY (`inspection_personnel_id`) REFERENCES `inspection_personnel` (`inspection_personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
