-- Schema for database: rschema_0061
CREATE TABLE `patent` (
  `patent_number` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`patent_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventor` (
  `inventor_number` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `position` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`inventor_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patent_application_record` (
  `patent_number` varchar(50) NOT NULL,
  `inventor_number` int(11) NOT NULL,
  `application_date` datetime NOT NULL,
  PRIMARY KEY (`patent_number`, `inventor_number`, `application_date`),
  CONSTRAINT `patent_application_record_ibfk_1` FOREIGN KEY (`patent_number`) REFERENCES `patent` (`patent_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patent_application_record_ibfk_2` FOREIGN KEY (`inventor_number`) REFERENCES `inventor` (`inventor_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
