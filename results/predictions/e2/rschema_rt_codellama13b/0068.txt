-- Schema for database: rschema_0068
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service` (
  `service_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `fee` varchar(50) DEFAULT NULL,
  `service_date` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`service_id`),
  CONSTRAINT `service_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `feedback` (
  `customer_id` int(11) NOT NULL,
  `service_id` int(11) NOT NULL,
  `feedback_comments` varchar(255) DEFAULT NULL,
  `rating` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `service_id`),
  CONSTRAINT `feedback_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `feedback_ibfk_2` FOREIGN KEY (`service_id`) REFERENCES `service` (`service_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
