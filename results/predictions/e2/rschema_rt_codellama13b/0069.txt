-- Schema for database: rschema_0069
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone` varchar(15) NOT NULL,
  `customer_age` int(11) NOT NULL,
  `customer_gender` char(1) NOT NULL,
  `customer_address` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_item` (
  `rental_item_id` int(11) NOT NULL,
  `item_name` varchar(255) NOT NULL,
  `item_category` varchar(100) NOT NULL,
  `single_rental_fee` decimal(10,2) NOT NULL,
  `total_inventory` int(11) NOT NULL,
  `remaining_inventory` int(11) NOT NULL,
  PRIMARY KEY (`rental_item_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `rental_record` (
  `rental_start_time` datetime NOT NULL,
  `customer_id` int(11) NOT NULL,
  `rental_item_id` int(11) NOT NULL,
  `number_of_rented_items` int(11) NOT NULL,
  `rental_payment_fee` decimal(10,2) NOT NULL,
  `rental_record_id` int(11) NOT NULL AUTO_INCREMENT,
  PRIMARY KEY (`rental_start_time`, `customer_id`, `rental_item_id`),
  CONSTRAINT `rental_record_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_record_ibfk_2` FOREIGN KEY (`rental_item_id`) REFERENCES `rental_item` (`rental_item_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
