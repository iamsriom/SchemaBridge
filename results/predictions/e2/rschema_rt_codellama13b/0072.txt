-- Schema for database: rschema_0072
CREATE TABLE `oil_and_gas_field` (
  `field_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`field_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pipeline` (
  `pipeline_id` int(11) NOT NULL,
  `starting_point` varchar(255) NOT NULL,
  `ending_point` varchar(255) NOT NULL,
  `length` float NOT NULL,
  PRIMARY KEY (`pipeline_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `maintenance_id` int(11) NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_cost` varchar(255) NOT NULL,
  `maintenance_date` varchar(255) NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `pipeline_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`pipeline_id`) REFERENCES `pipeline` (`pipeline_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
