-- Schema for database: rschema_0076
CREATE TABLE `beer_manufacturer` (
  `manufacturer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  PRIMARY KEY (`manufacturer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bar` (
  `bar_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `license` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`bar_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drinker` (
  `drinker_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`drinker_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beer` (
  `beer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `manufacturer_number` int(11) NOT NULL,
  PRIMARY KEY (`beer_number`),
  CONSTRAINT `beer_ibfk_1` FOREIGN KEY (`manufacturer_number`) REFERENCES `beer_manufacturer` (`manufacturer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `bar_number` int(11) NOT NULL,
  `beer_number` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `sales_date` datetime NOT NULL,
  PRIMARY KEY (`bar_number`, `beer_number`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`bar_number`) REFERENCES `bar` (`bar_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`beer_number`) REFERENCES `beer` (`beer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
