-- Schema for database: rschema_0089
CREATE TABLE `Block` (
  `Block_Hash` VARCHAR(255) NOT NULL,
  `Height` INT(11) DEFAULT NULL,
  `Merkle_Root` VARCHAR(255) DEFAULT NULL,
  `Previous_Block_Hash` VARCHAR(255) NOT NULL,
  `Timestamp` DATETIME DEFAULT NULL,
  PRIMARY KEY (`Block_Hash`),
  CONSTRAINT `Block_ibfk_1` FOREIGN KEY (`Previous_Block_Hash`) REFERENCES `Block` (`Block_Hash`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Account` (
  `Address` VARCHAR(255) NOT NULL,
  `Balance` FLOAT DEFAULT NULL,
  PRIMARY KEY (`Address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Transaction` (
  `Block_Hash` VARCHAR(255) NOT NULL,
  `Fee` FLOAT DEFAULT NULL,
  `Input_Address` VARCHAR(255) NOT NULL,
  `Output_Address` VARCHAR(255) NOT NULL,
  `Timestamp` DATETIME DEFAULT NULL,
  `Transaction_Hash` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`Transaction_Hash`),
  CONSTRAINT `Transaction_ibfk_1` FOREIGN KEY (`Block_Hash`) REFERENCES `Block` (`Block_Hash`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Transaction_ibfk_2` FOREIGN KEY (`Input_Address`) REFERENCES `Account` (`Address`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Transaction_ibfk_3` FOREIGN KEY (`Output_Address`) REFERENCES `Account` (`Address`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
