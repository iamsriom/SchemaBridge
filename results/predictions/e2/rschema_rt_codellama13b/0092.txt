CREATE TABLE `User` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `expiration_date` datetime DEFAULT NULL,
  `identity_status` varchar(50) NOT NULL,
  `identity_type` varchar(50) NOT NULL,
  `subordinate_user` int(11) DEFAULT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `user_ibfk_1` FOREIGN KEY (`operator_id`) REFERENCES `Operator` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_ibfk_2` FOREIGN KEY (`subordinate_user`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operator` (
  `operator_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `permissions` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Change_Superior_User` (
  `change_id` int(11) NOT NULL AUTO_INCREMENT,
  `change_time` datetime NOT NULL,
  `changed_superior_user` int(11) NOT NULL,
  `operator_id` int(11) NOT NULL,
  `previous_superior_user` int(11) NOT NULL,
  `remarks` text DEFAULT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`change_id`),
  CONSTRAINT `change_superior_user_ibfk_1` FOREIGN KEY (`changed_superior_user`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `change_superior_user_ibfk_2` FOREIGN KEY (`operator_id`) REFERENCES `Operator` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `change_superior_user_ibfk_3` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Change_Subordinate_User` (
  `change_id` int(11) NOT NULL AUTO_INCREMENT,
