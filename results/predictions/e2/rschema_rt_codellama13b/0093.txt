-- Schema for database: rschema_0093
CREATE TABLE `Book` (
  `Author` varchar(255) NOT NULL,
  `Book_Title` varchar(255) NOT NULL,
  `ISBN_Number` varchar(13) NOT NULL,
  `Publication_Year` int(11) NOT NULL,
  `Publishing_House` varchar(255) NOT NULL,
  PRIMARY KEY (`ISBN_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Type` (
  `Type_ID` int(11) NOT NULL,
  `Type_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Type_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Tag` (
  `Tag_ID` int(11) NOT NULL,
  `Tag_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Tag_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book_Type_Relation` (
  `ISBN_Number` varchar(13) NOT NULL,
  `Type_ID` int(11) NOT NULL,
  PRIMARY KEY (`ISBN_Number`, `Type_ID`),
  CONSTRAINT `Book_Type_Relation_ibfk_1` FOREIGN KEY (`ISBN_Number`) REFERENCES `Book` (`ISBN_Number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Book_Type_Relation_ibfk_2` FOREIGN KEY (`Type_ID`) REFERENCES `Type` (`Type_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book_Tag_Relation` (
  `ISBN_Number` varchar(13) NOT NULL,
  `Tag_ID` int(11) NOT NULL,
  PRIMARY KEY (`ISBN_Number`, `Tag_ID`),
  CONSTRAINT `Book_Tag_Relation_ibfk_1` FOREIGN KEY (`ISBN_Number`) REFERENCES `Book` (`ISBN_Number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Book_Tag_Relation_ibfk_2` FOREIGN KEY (`Tag_ID`) REFERENCES `Tag` (`Tag_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
