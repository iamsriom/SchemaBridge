-- Schema for database: rschema_0095
CREATE TABLE `Customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `company_name` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Employee` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `level` varchar(50) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Sales_Opportunity` (
  `customer_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `opportunity_name` varchar(255) NOT NULL,
  `opportunity_stage` varchar(50) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `estimated_amount` decimal(10,2) DEFAULT NULL,
  `estimated_transaction_date` datetime DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `employee_id`, `creation_time`),
  CONSTRAINT `sales_opportunity_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `Customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_opportunity_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `Employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Communication_Record` (
  `customer_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `communication_content` text DEFAULT NULL,
  `communication_method` varchar(50) DEFAULT NULL,
  `communication_result` varchar(50) DEFAULT NULL,
  `communication_stage` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `employee_id`, `creation_time`),
  CONSTRAINT `communication_record_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `Customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `communication_record_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `Employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Service_Request` (
  `customer_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `request_description` text DEFAULT NULL,
  `request_type` varchar(50) DEFAULT NULL,
  `solution` text DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `employee_id`, `creation_time`),
  CONSTRAINT `service_request_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `Customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_request_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `Employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
