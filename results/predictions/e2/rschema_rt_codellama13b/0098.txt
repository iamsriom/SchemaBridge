-- Schema for database: rschema_0098
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `email` varchar(255) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technical_solution` (
  `technical_solution_id` int(11) NOT NULL,
  `solution_name` varchar(255) NOT NULL,
  `solution_description` text DEFAULT NULL,
  `implementation_start_date` datetime DEFAULT NULL,
  `implementation_end_date` datetime DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`technical_solution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `email` varchar(255) NOT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `contact_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technical_solution_feedback` (
  `user_id` int(11) NOT NULL,
  `technical_solution_id` int(11) NOT NULL,
  `feedback_date` datetime NOT NULL,
  `feedback_content` text DEFAULT NULL,
  `satisfaction` int(11) DEFAULT NULL,
  `user_suggestion` text DEFAULT NULL,
  PRIMARY KEY (`user_id`, `technical_solution_id`, `feedback_date`),
  CONSTRAINT `technical_solution_feedback_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `technical_solution_feedback_ibfk_2` FOREIGN KEY (`technical_solution_id`) REFERENCES `technical_solution` (`technical_solution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `technical_solution_evaluation` (
  `technical_solution_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `evaluation_date` datetime NOT NULL,
  `evaluation_content` text DEFAULT NULL,
  `evaluation_indicators` text DEFAULT NULL,
  `evaluation_results` text DEFAULT NULL,
  PRIMARY KEY (`technical_solution_id`, `employee_id`),
  CONSTRAINT `technical_solution_evaluation_ibfk_1` FOREIGN KEY (`technical_solution_id`) REFERENCES `technical_solution` (`technical_solution_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `technical_solution_evaluation_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
