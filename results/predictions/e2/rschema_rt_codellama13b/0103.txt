-- Schema for database: rschema_0103
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_code` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salary_standard` (
  `salary_standard_id` int(11) NOT NULL,
  `standard_description` varchar(255) DEFAULT NULL,
  `standard_value` float DEFAULT NULL,
  PRIMARY KEY (`salary_standard_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `date_of_birth` datetime DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  `employee_id` int(11) NOT NULL,
  `gender` varchar(255) DEFAULT NULL,
  `hire_date` datetime DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `salary_standard_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`salary_standard_id`) REFERENCES `salary_standard` (`salary_standard_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_position_change` (
  `current_department` varchar(255) DEFAULT NULL,
  `current_position` varchar(255) DEFAULT NULL,
  `department_id` int(11) NOT NULL,
  `departure_time` datetime DEFAULT NULL,
  `employee_id` int(11) NOT NULL,
  `entry_time` datetime DEFAULT NULL,
  `original_department` varchar(255) DEFAULT NULL,
  `original_position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_id`, `department_id`),
  CONSTRAINT `employee_position_change_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_position_change_ibfk_2` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
