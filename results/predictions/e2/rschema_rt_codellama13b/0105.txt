-- Schema for database: rschema_0105
CREATE TABLE `User` (
  `user_id` int(11) NOT NULL,
  `account` varchar(255) NOT NULL,
  `birthday` datetime DEFAULT NULL,
  `email` varchar(255) NOT NULL,
  `gender` char(1) DEFAULT NULL,
  `nickname` varchar(255) DEFAULT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_dynamic_information` (
  `dynamic_id` int(11) NOT NULL,
  `dynamic_content` text NOT NULL,
  `number_of_comments` int(11) DEFAULT NULL,
  `number_of_favorites` int(11) DEFAULT NULL,
  `number_of_likes` int(11) DEFAULT NULL,
  `publisher_id` int(11) NOT NULL,
  `publishing_time` datetime NOT NULL,
  PRIMARY KEY (`dynamic_id`),
  CONSTRAINT `User_dynamic_information_ibfk_1` FOREIGN KEY (`publisher_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Add_Contact` (
  `user_id` int(11) NOT NULL,
  `added_user_id` int(11) NOT NULL,
  `time_added` datetime NOT NULL,
  `relationship_type` varchar(50) DEFAULT NULL,
  `remarks` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `added_user_id`),
  CONSTRAINT `Add_Contact_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Add_Contact_ibfk_2` FOREIGN KEY (`added_user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Social_Activity` (
  `user_id` int(11) NOT NULL,
  `dynamic_id` int(11) NOT NULL,
  `is_collection` tinyint(1) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  `creation_time` datetime NOT NULL,
  `is_like` tinyint(1) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `dynamic_id`),
  CONSTRAINT `Social_Activity_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Social_Activity_ibfk_2` FOREIGN KEY (`dynamic_id`) REFERENCES `User_dynamic_information` (`dynamic_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
