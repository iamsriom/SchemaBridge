-- Schema for database: rschema_0111
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Phone_Number` varchar(15) DEFAULT NULL,
  `Registration_Time` datetime NOT NULL,
  `Role` varchar(50) NOT NULL,
  `Update_Time` datetime DEFAULT NULL,
  `Username` varchar(255) NOT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Company` (
  `Company_ID` int(11) NOT NULL,
  `Company_Name` varchar(255) NOT NULL,
  `Company_Address` varchar(255) DEFAULT NULL,
  `Creation_Time` datetime NOT NULL,
  PRIMARY KEY (`Company_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Employee` (
  `Employee_ID` int(11) NOT NULL,
  `Employee_Name` varchar(255) NOT NULL,
  `Employee_Position` varchar(100) NOT NULL,
  `Affiliated_Company_ID` int(11) NOT NULL,
  PRIMARY KEY (`Employee_ID`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Affiliated_Company_ID`) REFERENCES `Company` (`Company_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Credit_Record` (
  `Credit_Number` int(11) NOT NULL,
  `Application_Date` datetime NOT NULL,
  `Creation_Time` datetime NOT NULL,
  `Credit_Status` varchar(50) NOT NULL,
  `Credit_Type` varchar(50) NOT NULL,
  `Deletion_Time` datetime DEFAULT NULL,
  `Employee_ID` int(11) NOT NULL,
  `User_ID` int(11) NOT NULL,
  `Monthly_Factor` decimal(10,2) DEFAULT NULL,
  `Number_of_Applications` int(11) DEFAULT NULL,
  `Update_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Credit_Number`),
  CONSTRAINT `credit_record_ibfk_1` FOREIGN KEY (`Employee_ID`) REFERENCES `Employee` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `credit_record_ibfk_2` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  UNIQUE KEY `Credit_Number_UNIQUE` (`Credit_Number`),
  INDEX `idx_employee_id` (`Employee_ID`),
  INDEX `idx_user_id` (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
