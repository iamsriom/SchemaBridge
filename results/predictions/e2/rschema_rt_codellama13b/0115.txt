-- Schema for database: rschema_0115
CREATE TABLE `benchmark_testing` (
  `benchmark_testing_id` int(11) NOT NULL,
  `benchmark_testing_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`benchmark_testing_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `unit` (
  `benchmark_testing_id` int(11) NOT NULL,
  `unit_id` int(11) NOT NULL,
  `unit_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`unit_id`),
  CONSTRAINT `unit_ibfk_1` FOREIGN KEY (`benchmark_testing_id`) REFERENCES `benchmark_testing` (`benchmark_testing_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
