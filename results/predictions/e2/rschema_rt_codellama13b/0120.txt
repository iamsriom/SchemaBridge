-- Schema for database: rschema_0120
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `active` tinyint(1) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `game_character` (
  `character_id` int(11) NOT NULL,
  `character_name` varchar(255) NOT NULL,
  `corporation` int(11) DEFAULT NULL,
  `active` tinyint(1) NOT NULL,
  `is_default` tinyint(1) NOT NULL,
  `ceo_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`character_id`),
  CONSTRAINT `game_character_ibfk_1` FOREIGN KEY (`corporation`) REFERENCES `corporation` (`corporation_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `game_character_ibfk_2` FOREIGN KEY (`ceo_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `corporation` (
  `corporation_id` int(11) NOT NULL,
  `ticker` varchar(10) NOT NULL,
  `corporation_name` varchar(255) NOT NULL,
  `ceo_id` int(11) DEFAULT NULL,
  `active` tinyint(1) NOT NULL,
  PRIMARY KEY (`corporation_id`),
  CONSTRAINT `corporation_ibfk_1` FOREIGN KEY (`ceo_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_login_record` (
  `user_id` int(11) NOT NULL,
  `source_address` varchar(255) NOT NULL,
  `browser_agent` varchar(255) NOT NULL,
  `login_result` varchar(50) NOT NULL DEFAULT 'success',
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `user_login_record_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `application` (
  `application_id` int(11) NOT NULL,
  `application_name` varchar(255) NOT NULL,
  `maintainer` int(11) DEFAULT NULL,
  `key` varchar(255) NOT NULL,
  `callback_url` varchar(255) NOT NULL,
  `active` tinyint(1) NOT NULL,
  PRIMARY KEY (`application_id`),
  CONSTRAINT `application_ibfk_1` FOREIGN KEY (`maintainer`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `application_ibfk_2` FOREIGN KEY (`key`) REFERENCES `api_key` (`key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_key` (
  `key` varchar(255) NOT NULL,
  `corporation` int(11) DEFAULT NULL,
  PRIMARY KEY (`key`),
  CONSTRAINT `api_key_ibfk_1` FOREIGN KEY (`corporation`) REFERENCES `corporation` (`corporation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
