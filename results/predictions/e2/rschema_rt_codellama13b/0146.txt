-- Schema for database: rschema_0146
CREATE TABLE `dormitory` (
  `dormitory_id` int(11) NOT NULL,
  `affiliated_college` varchar(255) DEFAULT NULL,
  `dormitory_number` varchar(255) DEFAULT NULL,
  `dormitory_type` varchar(255) DEFAULT NULL,
  `number_of_dormitory_members` int(11) DEFAULT NULL,
  PRIMARY KEY (`dormitory_id`),
  CONSTRAINT `dormitory_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `user` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `information_summary` (
  `user_id` int(11) NOT NULL,
  `dormitory_id` int(11) DEFAULT NULL,
  `summary_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `information_summary_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `gender` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `personal_profile` varchar(255) DEFAULT NULL,
  `dormitory_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `user_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
