-- Schema for database: rschema_0150
CREATE TABLE `Supplier` (
  `supplier_id` int(11) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`supplier_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Project` (
  `project_id` int(11) NOT NULL,
  `budget` decimal(10,2) DEFAULT NULL,
  `start_date` datetime DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Part` (
  `part_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `specification` varchar(255) DEFAULT NULL,
  `unit_price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`part_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Supplier_Supply_Project` (
  `supplier_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `start_time` datetime DEFAULT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`supplier_id`, `project_id`),
  CONSTRAINT `Supplier_Supply_Project_ibfk_1` FOREIGN KEY (`supplier_id`) REFERENCES `Supplier` (`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Supplier_Supply_Project_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `Project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Supplier_Supply_Part` (
  `supplier_id` int(11) NOT NULL,
  `part_id` int(11) NOT NULL,
  `supply_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`supplier_id`, `part_id`),
  CONSTRAINT `Supplier_Supply_Part_ibfk_1` FOREIGN KEY (`supplier_id`) REFERENCES `Supplier` (`supplier_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Supplier_Supply_Part_ibfk_2` FOREIGN KEY (`part_id`) REFERENCES `Part` (`part_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Project_Requires_Parts` (
  `project_id` int(11) NOT NULL,
  `part_id` int(11) NOT NULL,
  `quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`project_id`, `part_id`),
  CONSTRAINT `Project_Requires_Parts_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `Project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Project_Requires_Parts_ibfk_2` FOREIGN KEY (`part_id`) REFERENCES `Part` (`part_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Warehouse` (
  `warehouse_id` int(11) NOT NULL,
  `area` varchar(255) DEFAULT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`warehouse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Employee` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `job_title` varchar(255) DEFAULT NULL,
  `work_location` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Warehouse_Inventory_Parts` (
  `warehouse_id` int(11) NOT NULL,
  `part_id` int(11) NOT NULL,
  `inventory_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`warehouse_id`, `part_id`),
  CONSTRAINT `Warehouse_Inventory_Parts_ibfk_1` FOREIGN KEY (`warehouse_id`) REFERENCES `Warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Warehouse_Inventory_Parts_ibfk_2` FOREIGN KEY (`part_id`) REFERENCES `Part` (`part_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
