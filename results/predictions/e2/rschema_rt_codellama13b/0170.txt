-- Schema for database: rschema_0170
CREATE TABLE `store` (
  `store_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_phone_number` varchar(15) DEFAULT NULL,
  `manager` int(11) DEFAULT NULL,
  PRIMARY KEY (`store_id`),
  CONSTRAINT `store_ibfk_1` FOREIGN KEY (`manager`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `distribution_location` varchar(255) NOT NULL,
  `contact_phone_number` varchar(15) DEFAULT NULL,
  `manager` int(11) DEFAULT NULL,
  `manager_tenure` int(11) DEFAULT NULL,
  `belonging_store` int(11) NOT NULL,
  `employee_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`department_id`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`manager`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `department_ibfk_2` FOREIGN KEY (`belonging_store`) REFERENCES `store` (`store_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `salary` decimal(10,2) DEFAULT NULL,
  `emergency_contact` int(11) DEFAULT NULL,
  `belonging_department` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`emergency_contact`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`belonging_department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
