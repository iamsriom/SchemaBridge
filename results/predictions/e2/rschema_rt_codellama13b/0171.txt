-- Schema for database: rschema_0171
CREATE TABLE `parking` (
  `parking_space_id` int(11) NOT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`parking_space_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `entrance` (
  `entrance_number` int(11) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`entrance_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `exit` (
  `exit_number` int(11) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`exit_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL,
  `model` varchar(100) DEFAULT NULL,
  `license_plate_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parking_record` (
  `parking_space_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `entrance_number` int(11) NOT NULL,
  `entry_time` datetime NOT NULL,
  `exit_number` int(11) NOT NULL,
  `exit_time` datetime DEFAULT NULL,
  `parking_duration` int(11) DEFAULT NULL,
  `parking_fee` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`parking_space_id`, `vehicle_id`),
  CONSTRAINT `parking_record_ibfk_1` FOREIGN KEY (`entrance_number`) REFERENCES `entrance` (`entrance_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_2` FOREIGN KEY (`exit_number`) REFERENCES `exit` (`exit_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_3` FOREIGN KEY (`parking_space_id`) REFERENCES `parking` (`parking_space_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_4` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
