-- Schema for database: rschema_0182
CREATE TABLE `market` (
  `market_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`market_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `agricultural_products` (
  `product_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `specification` varchar(255) DEFAULT NULL,
  `price` decimal(10,2) NOT NULL,
  `current_stock` int(11) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `product_id` int(11) NOT NULL,
  `market_id` int(11) NOT NULL,
  `sales_date` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_unit_price` decimal(10,2) NOT NULL,
  `operator` varchar(255) NOT NULL,
  PRIMARY KEY (`product_id`, `sales_date`, `market_id`),
  CONSTRAINT `sales_record_ibfk_product_id` FOREIGN KEY (`product_id`) REFERENCES `agricultural_products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_market_id` FOREIGN KEY (`market_id`) REFERENCES `market` (`market_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
