-- Schema for database: rschema_0183
CREATE TABLE `tool` (
  `maintenance_count` int(11) NOT NULL,
  `tool_id` int(11) NOT NULL,
  `tool_model` varchar(255) DEFAULT NULL,
  `tool_name` varchar(255) DEFAULT NULL,
  `usage_frequency` int(11) DEFAULT NULL,
  PRIMARY KEY (`tool_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `usage_record` (
  `record_id` int(11) NOT NULL,
  `tool_id` int(11) NOT NULL,
  `usage_location` varchar(255) DEFAULT NULL,
  `usage_time` datetime DEFAULT NULL,
  `user` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `usage_record_ibfk_1` FOREIGN KEY (`tool_id`) REFERENCES `tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `maintenance_content` text DEFAULT NULL,
  `maintenance_id` int(11) NOT NULL,
  `maintenance_person` varchar(255) DEFAULT NULL,
  `maintenance_time` datetime DEFAULT NULL,
  `tool_id` int(11) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`tool_id`) REFERENCES `tool` (`tool_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
