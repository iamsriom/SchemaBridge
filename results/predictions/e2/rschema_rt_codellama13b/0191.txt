-- Schema for database: rschema_0191
CREATE TABLE `sales_team` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salesperson` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `sex` char(1) NOT NULL,
  `telephone` varchar(15) DEFAULT NULL,
  `sales_count` int(11) NOT NULL,
  `sales_cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_task` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `salesperson_id` int(11) NOT NULL,
  `customer_info` varchar(255) NOT NULL,
  `product_info` varchar(255) NOT NULL,
  `sales_info` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `sales_task_ibfk_1` FOREIGN KEY (`salesperson_id`) REFERENCES `salesperson` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluation` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `salesperson_id` int(11) NOT NULL,
  `customer_satisfaction` int(11) NOT NULL,
  `task_completion_quantity` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `performance_evaluation_ibfk_1` FOREIGN KEY (`salesperson_id`) REFERENCES `salesperson` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `sales_task_id` int(11) NOT NULL,
  `customer_evaluation` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `task_record_ibfk_1` FOREIGN KEY (`sales_task_id`) REFERENCES `sales_task` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
