-- Schema for database: rschema_0197
CREATE TABLE `Personnel` (
  `Personnel_ID` int(11) NOT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(20) DEFAULT NULL,
  `Role` varchar(100) DEFAULT NULL,
  `Project_Manager_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Personnel_ID`),
  CONSTRAINT `Personnel_ibfk_1` FOREIGN KEY (`Project_Manager_ID`) REFERENCES `Project` (`Project_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Project` (
  `Project_ID` int(11) NOT NULL,
  `Project_Name` varchar(255) DEFAULT NULL,
  `Project_Location` varchar(255) DEFAULT NULL,
  `Recording_Staff_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Project_ID`),
  CONSTRAINT `Project_ibfk_1` FOREIGN KEY (`Recording_Staff_ID`) REFERENCES `Personnel` (`Personnel_ID`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Construction_Cost_Record` (
  `Record_ID` int(11) NOT NULL,
  `Cost_Amount` decimal(10,2) DEFAULT NULL,
  `Expense_Type` varchar(100) DEFAULT NULL,
  `Project_ID` int(11) DEFAULT NULL,
  `Record_Date` datetime DEFAULT NULL,
  `Recording_Staff_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Record_ID`),
  CONSTRAINT `Construction_Cost_Record_ibfk_1` FOREIGN KEY (`Project_ID`) REFERENCES `Project` (`Project_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Construction_Cost_Record_ibfk_2` FOREIGN KEY (`Recording_Staff_ID`) REFERENCES `Personnel` (`Personnel_ID`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
