-- Schema for database: rschema_0198
CREATE TABLE `community_information` (
  `community_id` int(11) NOT NULL,
  `community_name` varchar(255) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `property_manager` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`community_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_staff_information` (
  `maintenance_staff_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `position` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`maintenance_staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `property_information` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` int(11) DEFAULT NULL,
  `community_id` int(11) NOT NULL,
  `maintenance_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`property_id`),
  CONSTRAINT `property_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community_information` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_task` (
  `task_id` int(11) NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_cost` decimal(10,2) DEFAULT NULL,
  `maintenance_time` datetime DEFAULT NULL,
  `maintenance_staff_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `task_ibfk_1` FOREIGN KEY (`maintenance_staff_id`) REFERENCES `maintenance_staff_information` (`maintenance_staff_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `task_ibfk_2` FOREIGN KEY (`property_id`) REFERENCES `property_information` (`property_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
