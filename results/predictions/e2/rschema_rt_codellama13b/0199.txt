-- Schema for database: rschema_0199
CREATE TABLE `store` (
  `store_number` varchar(255) NOT NULL,
  `store_name` varchar(255) NOT NULL,
  `store_address` varchar(255) NOT NULL,
  `store_manager` varchar(255) NOT NULL,
  PRIMARY KEY (`store_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_number` varchar(255) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `product_model` varchar(255) NOT NULL,
  `product_barcode` varchar(255) NOT NULL,
  `product_price` varchar(255) NOT NULL,
  PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `store_number` varchar(255) NOT NULL,
  `product_number` varchar(255) NOT NULL,
  `sales_time` varchar(255) NOT NULL,
  `sales_quantity` varchar(255) NOT NULL,
  `sales_amount` varchar(255) NOT NULL,
  `sales_personnel` varchar(255) NOT NULL,
  PRIMARY KEY (`sales_time`, `product_number`, `store_number`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`product_number`) REFERENCES `product` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `store_performance` (
  `store_number` varchar(255) NOT NULL,
  `sales_amount` varchar(255) NOT NULL,
  `customer_satisfaction` varchar(255) NOT NULL,
  PRIMARY KEY (`store_number`),
  CONSTRAINT `store_performance_ibfk_1` FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
