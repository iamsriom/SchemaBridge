-- Schema for database: rschema_0215
CREATE TABLE `material_administrator` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(255) NOT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `material_administrator_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mine` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `purchase_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `supplier` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_distribution_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `distribution_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `distributor_id` int(11) NOT NULL,
  `recipient` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `material_distribution_record_ibfk_1` FOREIGN KEY (`distributor_id`) REFERENCES `material_administrator` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
