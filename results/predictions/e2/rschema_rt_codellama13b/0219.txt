-- Schema for database: rschema_0219
CREATE TABLE `Card` (
  `Card_ID` int(11) NOT NULL,
  `Card_Artist` varchar(255) DEFAULT NULL,
  `Card_Effect` varchar(255) DEFAULT NULL,
  `Card_Endurance` int(11) DEFAULT NULL,
  `Card_Intelligence` int(11) DEFAULT NULL,
  `Card_Name` varchar(255) DEFAULT NULL,
  `Card_Power` int(11) DEFAULT NULL,
  `Card_Type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Card_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Card_Collection` (
  `Collection_ID` int(11) NOT NULL,
  `Collection_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Collection_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Password` varchar(255) DEFAULT NULL,
  `Phone` varchar(15) DEFAULT NULL,
  `Username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Card_Collection_Association` (
  `Card_ID` int(11) NOT NULL,
  `Collection_ID` int(11) NOT NULL,
  `Card_Rarity` varchar(255) DEFAULT NULL,
  `Collection_Number` int(11) DEFAULT NULL,
  PRIMARY KEY (`Card_ID`, `Collection_ID`),
  CONSTRAINT `Card_Collection_Association_ibfk_1` FOREIGN KEY (`Card_ID`) REFERENCES `Card` (`Card_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Card_Collection_Association_ibfk_2` FOREIGN KEY (`Collection_ID`) REFERENCES `Card_Collection` (`Collection_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Card_Exchange` (
  `User_1_ID` int(11) NOT NULL,
  `User_2_ID` int(11) NOT NULL,
  `Exchanged_Card_ID` int(11) NOT NULL,
  `Exchange_Time` datetime NOT NULL,
  PRIMARY KEY (`User_1_ID`, `User_2_ID`, `Exchange_Time`),
  CONSTRAINT `User_Card_Exchange_ibfk_1` FOREIGN KEY (`User_1_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Card_Exchange_ibfk_2` FOREIGN KEY (`User_2_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Card_Exchange_ibfk_3` FOREIGN KEY (`Exchanged_Card_ID`) REFERENCES `Card` (`Card_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
