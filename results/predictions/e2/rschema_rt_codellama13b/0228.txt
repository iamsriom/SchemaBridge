-- Schema for database: rschema_0228
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `magazine` (
  `publisher` varchar(255) NOT NULL,
  `magazine_name` varchar(255) NOT NULL,
  `theme` varchar(255) DEFAULT NULL,
  `publication_frequency` varchar(255) DEFAULT NULL,
  `website` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`publisher`, `magazine_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pricing` (
  `pricing_id` int(11) NOT NULL,
  `subscription_package` varchar(255) NOT NULL,
  `subscription_period` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `magazine_name` varchar(255) NOT NULL,
  PRIMARY KEY (`pricing_id`),
  CONSTRAINT `pricing_ibfk_1` FOREIGN KEY (`publisher`, `magazine_name`) REFERENCES `magazine` (`publisher`, `magazine_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_subscribes_to_magazine` (
  `pricing_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `subscription_start_date` datetime DEFAULT NULL,
  PRIMARY KEY (`pricing_id`, `customer_id`, `creation_time`),
  CONSTRAINT `customer_subscribes_to_magazine_ibfk_1` FOREIGN KEY (`pricing_id`) REFERENCES `pricing` (`pricing_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_subscribes_to_magazine_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
