-- Schema for database: rschema_0232
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Email` varchar(255) NOT NULL,
  `Username` varchar(255) NOT NULL,
  `Password_Hash` varchar(255) NOT NULL,
  `Activation_Token` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`),
  UNIQUE KEY `Email` (`Email`),
  UNIQUE KEY `Username` (`Username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Scholarship` (
  `Scholarship_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Scholarship_Name` varchar(255) NOT NULL,
  `Description` text DEFAULT NULL,
  `Link` varchar(255) DEFAULT NULL,
  `Amount` decimal(10,2) DEFAULT NULL,
  `Deadline` datetime DEFAULT NULL,
  PRIMARY KEY (`Scholarship_ID`),
  UNIQUE KEY `Scholarship_Name` (`Scholarship_Name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Category` (
  `Category_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Category_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Category_ID`),
  UNIQUE KEY `Category_Name` (`Category_Name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Likes_Scholarship` (
  `User_ID` int(11) NOT NULL,
  `Scholarship_ID` int(11) NOT NULL,
  `Like_Time` datetime NOT NULL,
  PRIMARY KEY (`User_ID`, `Scholarship_ID`, `Like_Time`),
  CONSTRAINT `User_Likes_Scholarship_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Likes_Scholarship_ibfk_2` FOREIGN KEY (`Scholarship_ID`) REFERENCES `Scholarship` (`Scholarship_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Scholarship_Category` (
  `Scholarship_ID` int(11) NOT NULL,
  `Category_ID` int(11) NOT NULL,
  PRIMARY KEY (`Scholarship_ID`, `Category_ID`),
  CONSTRAINT `Scholarship_Category_ibfk_1` FOREIGN KEY (`Scholarship_ID`) REFERENCES `Scholarship` (`Scholarship_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Scholarship_Category_ibfk_2` FOREIGN KEY (`Category_ID`) REFERENCES `Category` (`Category_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
