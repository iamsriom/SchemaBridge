-- Schema for database: rschema_0236
CREATE TABLE `Camera` (
  `Analysis_Time` datetime DEFAULT NULL,
  `Camera_ID` int(11) NOT NULL,
  `Location` varchar(255) DEFAULT NULL,
  `RTSP_Stream_Address` varchar(255) DEFAULT NULL,
  `Title` varchar(255) DEFAULT NULL,
  `Video_Stream_Path` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Camera_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Video_Recording` (
  `Camera_ID` int(11) NOT NULL,
  `Cover_Path` varchar(255) DEFAULT NULL,
  `Duration` int(11) DEFAULT NULL,
  `Start_Time` datetime DEFAULT NULL,
  `Video_File_Path` varchar(255) DEFAULT NULL,
  `Video_ID` int(11) NOT NULL,
  PRIMARY KEY (`Video_ID`),
  CONSTRAINT `video_recording_ibfk_1` FOREIGN KEY (`Camera_ID`) REFERENCES `Camera` (`Camera_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Alarm` (
  `Alarm_ID` int(11) NOT NULL,
  `Alarm_Level` varchar(255) DEFAULT NULL,
  `Alarm_Type` varchar(255) DEFAULT NULL,
  `End_Time` datetime DEFAULT NULL,
  `Start_Time` datetime DEFAULT NULL,
  `Video_ID` int(11) NOT NULL,
  PRIMARY KEY (`Alarm_ID`),
  CONSTRAINT `alarm_ibfk_1` FOREIGN KEY (`Video_ID`) REFERENCES `Video_Recording` (`Video_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
