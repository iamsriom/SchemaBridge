-- Schema for database: rschema_0237
CREATE TABLE `User` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `username` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Area` (
  `area_id` int(11) NOT NULL AUTO_INCREMENT,
  `area_name` varchar(255) NOT NULL,
  PRIMARY KEY (`area_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `House_Type` (
  `type_id` int(11) NOT NULL AUTO_INCREMENT,
  `type_name` varchar(255) NOT NULL,
  PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Street` (
  `street_id` int(11) NOT NULL AUTO_INCREMENT,
  `street_name` varchar(255) NOT NULL,
  `area_id` int(11) NOT NULL,
  PRIMARY KEY (`street_id`),
  CONSTRAINT `street_ibfk_1` FOREIGN KEY (`area_id`) REFERENCES `Area` (`area_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `House` (
  `house_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `type_id` int(11) NOT NULL,
  `street_id` int(11) NOT NULL,
  PRIMARY KEY (`house_id`),
  CONSTRAINT `house_ibfk_1` FOREIGN KEY (`type_id`) REFERENCES `House_Type` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `house_ibfk_2` FOREIGN KEY (`street_id`) REFERENCES `Street` (`street_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Lease` (
  `user_id` int(11) NOT NULL,
  `house_id` int(11) NOT NULL,
  `lease_date` datetime NOT NULL,
  `description` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`user_id`, `house_id`, `lease_date`),
  CONSTRAINT `lease_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `lease_ibfk_2` FOREIGN KEY (`house_id`) REFERENCES `House` (`house_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
