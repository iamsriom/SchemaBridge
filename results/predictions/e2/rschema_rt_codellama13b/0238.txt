-- Schema for database: rschema_0238
CREATE TABLE `class` (
  `class_id` int(11) NOT NULL,
  `class_name` varchar(255) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `grade` varchar(50) NOT NULL,
  PRIMARY KEY (`class_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member` (
  `member_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `work_unit` varchar(255) DEFAULT NULL,
  `phone_number` varchar(50) DEFAULT NULL,
  `class_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`member_id`),
  CONSTRAINT `member_ibfk_1` FOREIGN KEY (`class_id`) REFERENCES `class` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcement` (
  `announcement_id` int(11) NOT NULL,
  `announcement_time` datetime NOT NULL,
  `announcement_content` text NOT NULL,
  `class_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`announcement_id`),
  CONSTRAINT `announcement_ibfk_1` FOREIGN KEY (`class_id`) REFERENCES `class` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `message` (
  `message_id` int(11) NOT NULL,
  `sender_id` int(11) NOT NULL,
  `receiver_id` int(11) NOT NULL,
  `message_content` text NOT NULL,
  `send_time` datetime NOT NULL,
  `read_flag` tinyint(1) NOT NULL,
  PRIMARY KEY (`message_id`),
  CONSTRAINT `message_ibfk_1` FOREIGN KEY (`sender_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `message_ibfk_2` FOREIGN KEY (`receiver_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `log` (
  `log_id` int(11) NOT NULL,
  `log_content` text NOT NULL,
  `time` datetime NOT NULL,
  `mood` varchar(50) DEFAULT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`log_id`),
  CONSTRAINT `log_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `application` (
  `application_id` int(11) NOT NULL,
  `application_time` datetime NOT NULL,
  `approval_flag` tinyint(1) NOT NULL,
  `application_content` text NOT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`application_id`),
  CONSTRAINT `application_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_message` (
  `member_id` int(11) NOT NULL,
  `message_id` int(11) NOT NULL,
  `message_count` int(11) NOT NULL,
  PRIMARY KEY (`member_id`, `message_id`),
  CONSTRAINT `member_message_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `member_message_ibfk_2` FOREIGN KEY (`message_id`) REFERENCES `message` (`message_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_sent_message` (
  `receiver_id` int(11) NOT NULL,
  `send_time` datetime NOT NULL,
  `sender_id` int(11) NOT NULL,
  `message_content` text NOT NULL,
  `read_flag` tinyint(1) NOT NULL,
  PRIMARY KEY (`receiver_id`, `send_time`, `sender_id`),
  CONSTRAINT `member_sent_message_ibfk_1` FOREIGN KEY (`receiver_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `member_sent_message_ibfk_2` FOREIGN KEY (`sender_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
