-- Schema for database: rschema_0242
CREATE TABLE `Employee` (
  `Employee_ID` int(11) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Username` varchar(255) NOT NULL,
  PRIMARY KEY (`Employee_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Member` (
  `Member_ID` int(11) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Username` varchar(255) NOT NULL,
  `Balance` decimal(10,2) DEFAULT NULL,
  `Birth_Month_and_Year` date DEFAULT NULL,
  `Gender` char(1) DEFAULT NULL,
  `Phone_Number` varchar(15) DEFAULT NULL,
  `Administrator_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Member_ID`),
  CONSTRAINT `member_ibfk_1` FOREIGN KEY (`Administrator_ID`) REFERENCES `Administrator` (`Administrator_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Administrator` (
  `Administrator_ID` int(11) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Username` varchar(255) NOT NULL,
  PRIMARY KEY (`Administrator_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Order` (
  `Order_ID` int(11) NOT NULL,
  `Order_Time` datetime NOT NULL,
  `Employee_ID` int(11) NOT NULL,
  `Member_ID` int(11) NOT NULL,
  PRIMARY KEY (`Order_ID`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`Employee_ID`) REFERENCES `Employee` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_ibfk_2` FOREIGN KEY (`Member_ID`) REFERENCES `Member` (`Member_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
