-- Schema for database: rschema_0244
CREATE TABLE `user` (
  `account` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `user_role` varchar(50) NOT NULL,
  PRIMARY KEY (`account`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bus_route` (
  `route_id` int(11) NOT NULL,
  `route_name` varchar(255) NOT NULL,
  `start_station` int(11) NOT NULL,
  `end_station` int(11) NOT NULL,
  `passing_stations` varchar(255) DEFAULT NULL,
  `departure_time` datetime NOT NULL,
  PRIMARY KEY (`route_id`),
  CONSTRAINT `bus_route_ibfk_1` FOREIGN KEY (`start_station`) REFERENCES `station` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bus_route_ibfk_2` FOREIGN KEY (`end_station`) REFERENCES `station` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `station` (
  `station_id` int(11) NOT NULL,
  `station_name` varchar(255) NOT NULL,
  `station_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcement` (
  `announcement_id` int(11) NOT NULL,
  `announcement_title` varchar(255) NOT NULL,
  `announcement_content` text NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `publish_time` datetime NOT NULL,
  PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `question` (
  `questioner_account` varchar(255) NOT NULL,
  `question_content` text NOT NULL,
  `question_time` datetime NOT NULL,
  `answer_content` text DEFAULT NULL,
  `responder_account` varchar(255) DEFAULT NULL,
  `answer_time` datetime DEFAULT NULL,
  PRIMARY KEY (`question_time`),
  CONSTRAINT `question_ibfk_1` FOREIGN KEY (`questioner_account`) REFERENCES `user` (`account`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `question_ibfk_2` FOREIGN KEY (`responder_account`) REFERENCES `user` (`account`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
