-- Schema for database: rschema_0251
CREATE TABLE `groomer` (
  `groomer_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `rating` float NOT NULL,
  `specialization` varchar(255) NOT NULL,
  `years_of_experience` int(11) NOT NULL,
  PRIMARY KEY (`groomer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pet` (
  `pet_id` int(11) NOT NULL AUTO_INCREMENT,
  `age` int(11) NOT NULL,
  `breed` varchar(255) NOT NULL,
  `fur_characteristics` varchar(255) NOT NULL,
  `weight` float NOT NULL,
  PRIMARY KEY (`pet_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `grooming_service` (
  `groomer_id` int(11) NOT NULL,
  `pet_id` int(11) NOT NULL,
  `service_time` datetime NOT NULL,
  `customer_rating` float NOT NULL,
  `price` float NOT NULL,
  `service_items` varchar(255) NOT NULL,
  `service_market` varchar(255) NOT NULL,
  PRIMARY KEY (`groomer_id`, `pet_id`, `service_time`),
  CONSTRAINT `grooming_service_ibfk_1` FOREIGN KEY (`groomer_id`) REFERENCES `groomer` (`groomer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grooming_service_ibfk_2` FOREIGN KEY (`pet_id`) REFERENCES `pet` (`pet_id`) ON DELETE RESTRICT ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
