-- Schema for database: rschema_0265
CREATE TABLE `chessboard` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `quality_inspection_model` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `result` varchar(255) NOT NULL,
  `running_time` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chess_piece` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `color` varchar(50) NOT NULL,
  `position` varchar(10) NOT NULL,
  `chessboard_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `chess_piece_ibfk_1` FOREIGN KEY (`chessboard_id`) REFERENCES `chessboard` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `quality_inspection` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `chess_piece_id` int(11) NOT NULL,
  `inspection_id` int(11) NOT NULL,
  `inspection_result` varchar(255) NOT NULL,
  `inspection_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `quality_inspection_ibfk_1` FOREIGN KEY (`chess_piece_id`) REFERENCES `chess_piece` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `quality_alert` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `defect_type` varchar(255) NOT NULL,
  `inspection_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `quality_alert_ibfk_1` FOREIGN KEY (`inspection_id`) REFERENCES `quality_inspection` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
