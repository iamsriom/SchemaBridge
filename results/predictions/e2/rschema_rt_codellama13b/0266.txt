-- Schema for database: rschema_0266
CREATE TABLE `pipeline` (
  `pipeline_id` int(11) NOT NULL,
  `diameter` float DEFAULT NULL,
  `starting_point` varchar(255) DEFAULT NULL,
  `ending_point` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pipeline_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `maintenance_personnel_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_activity_record` (
  `pipeline_id` int(11) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_details` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pipeline_id`, `maintenance_personnel_id`, `maintenance_time`),
  CONSTRAINT `maintenance_activity_record_ibfk_1` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `maintenance_personnel` (`maintenance_personnel_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `maintenance_activity_record_ibfk_2` FOREIGN KEY (`pipeline_id`) REFERENCES `pipeline` (`pipeline_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
