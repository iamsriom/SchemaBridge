-- Schema for database: rschema_0270
CREATE TABLE `program` (
  `program_id` int(11) NOT NULL,
  `program_name` varchar(255) NOT NULL,
  `program_type` varchar(100) NOT NULL,
  `broadcast_start_time` datetime NOT NULL,
  `broadcast_end_time` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  PRIMARY KEY (`program_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff` (
  `staff_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `position` varchar(100) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_number` varchar(15) NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_production` (
  `production_staff_id` int(11) NOT NULL,
  `program_id` int(11) NOT NULL,
  `working_duration` int(11) NOT NULL,
  PRIMARY KEY (`program_id`, `production_staff_id`),
  CONSTRAINT `program_production_ibfk_1` FOREIGN KEY (`production_staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_production_ibfk_2` FOREIGN KEY (`program_id`) REFERENCES `program` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_review` (
  `program_id` int(11) NOT NULL,
  `review_result` varchar(255) NOT NULL,
  `reviewer_id` int(11) NOT NULL,
  `working_duration` int(11) NOT NULL,
  CONSTRAINT `program_review_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `program` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_review_ibfk_2` FOREIGN KEY (`reviewer_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
