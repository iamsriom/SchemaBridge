-- Schema for database: rschema_0271
CREATE TABLE `service` (
  `service_id` int(11) NOT NULL,
  `service_name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `transmission_rate` varchar(255) NOT NULL,
  `transmission_type` varchar(255) NOT NULL,
  PRIMARY KEY (`service_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `client` (
  `client_id` int(11) NOT NULL,
  `client_name` varchar(255) NOT NULL,
  `client_address` varchar(255) NOT NULL,
  `client_email` varchar(255) NOT NULL,
  `client_phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_subscription` (
  `service_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `subscription_start_time` datetime NOT NULL,
  `subscription_end_time` datetime DEFAULT NULL,
  `operator` varchar(255) NOT NULL,
  PRIMARY KEY (`service_id`, `client_id`, `subscription_start_time`),
  CONSTRAINT `service_subscription_ibfk_1` FOREIGN KEY (`service_id`) REFERENCES `service` (`service_id`) ON DELETE RESTRICT ON UPDATE RESTRICT,
  CONSTRAINT `service_subscription_ibfk_2` FOREIGN KEY (`client_id`) REFERENCES `client` (`client_id`) ON DELETE RESTRICT ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
