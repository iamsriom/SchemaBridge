-- Schema for database: rschema_0272
CREATE TABLE `Ranch` (
  `ranch_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`ranch_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Livestock` (
  `livestock_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `species` varchar(255) NOT NULL,
  `ranch_id` int(11) NOT NULL,
  PRIMARY KEY (`livestock_id`),
  CONSTRAINT `livestock_ibfk_1` FOREIGN KEY (`ranch_id`) REFERENCES `Ranch` (`ranch_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Health_Status` (
  `health_status_id` int(11) NOT NULL,
  `livestock_id` int(11) NOT NULL,
  `status` varchar(255) NOT NULL,
  `recording_time` datetime NOT NULL,
  PRIMARY KEY (`health_status_id`),
  CONSTRAINT `health_status_ibfk_1` FOREIGN KEY (`livestock_id`) REFERENCES `Livestock` (`livestock_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Vaccination_Status` (
  `vaccination_id` int(11) NOT NULL,
  `livestock_id` int(11) NOT NULL,
  `vaccination_measures` varchar(255) NOT NULL,
  `vaccination_time` datetime NOT NULL,
  PRIMARY KEY (`vaccination_id`),
  CONSTRAINT `vaccination_status_ibfk_1` FOREIGN KEY (`livestock_id`) REFERENCES `Livestock` (`livestock_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Feed_Consumption` (
  `consumption_id` int(11) NOT NULL,
  `livestock_id` int(11) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `consumption_time` datetime NOT NULL,
  `feed_type` varchar(255) NOT NULL,
  PRIMARY KEY (`consumption_id`),
  CONSTRAINT `feed_consumption_ibfk_1` FOREIGN KEY (`livestock_id`) REFERENCES `Livestock` (`livestock_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Entry_Exit_Records` (
  `record_id` int(11) NOT NULL,
  `livestock_id` int(11) NOT NULL,
  `entry_exit_type` varchar(255) NOT NULL,
  `time` datetime NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `entry_exit_records_ibfk_1` FOREIGN KEY (`livestock_id`) REFERENCES `Livestock` (`livestock_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
