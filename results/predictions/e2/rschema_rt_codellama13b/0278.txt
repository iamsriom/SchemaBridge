-- Schema for database: rschema_0278
CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `project_location` varchar(255) NOT NULL,
  `project_manager` varchar(255) NOT NULL,
  `project_budget` float DEFAULT NULL,
  `construction_start_date` datetime DEFAULT NULL,
  `actual_completion_date` datetime DEFAULT NULL,
  `expected_completion_date` datetime DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspection_record` (
  `inspection_id` int(11) NOT NULL,
  `inspection_content` varchar(255) NOT NULL,
  `inspection_date` datetime DEFAULT NULL,
  `inspection_result` varchar(255) NOT NULL,
  `inspector` varchar(255) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  CONSTRAINT `safety_inspection_record_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `unqualified_project_tracking_record` (
  `tracking_record_id` int(11) NOT NULL,
  `tracking_content` varchar(255) NOT NULL,
  `tracking_date` datetime DEFAULT NULL,
  `tracking_result` varchar(255) NOT NULL,
  `project_id` int(11) NOT NULL,
  PRIMARY KEY (`tracking_record_id`),
  CONSTRAINT `unqualified_project_tracking_record_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
