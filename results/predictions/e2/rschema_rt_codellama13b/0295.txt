-- Schema for database: rschema_0295
CREATE TABLE `trader` (
  `trader_id` int(11) NOT NULL,
  `contact_number` varchar(255) DEFAULT NULL,
  `hire_date` datetime DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`trader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `client` (
  `client_id` int(11) NOT NULL,
  `client_name` varchar(255) DEFAULT NULL,
  `contact_number` varchar(255) DEFAULT NULL,
  `id_number` varchar(255) DEFAULT NULL,
  `risk_level` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction` (
  `trader_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `futures_code` varchar(255) NOT NULL,
  `transaction_price` decimal(10,2) DEFAULT NULL,
  `transaction_quantity` int(11) DEFAULT NULL,
  `transaction_time` datetime DEFAULT NULL,
  `transaction_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`trader_id`, `client_id`, `futures_code`),
  CONSTRAINT `transaction_ibfk_1` FOREIGN KEY (`client_id`) REFERENCES `client` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_ibfk_2` FOREIGN KEY (`trader_id`) REFERENCES `trader` (`trader_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
