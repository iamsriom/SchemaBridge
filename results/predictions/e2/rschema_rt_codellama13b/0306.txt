-- Schema for database: rschema_0306
CREATE TABLE `task` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `concurrently_executable` tinyint(1) NOT NULL,
  `persistent` tinyint(1) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trigger_summary` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(50) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `scheduler` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `configuration` text DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `simple_trigger` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `trigger_id` int(11) NOT NULL,
  `repeat_count` int(11) NOT NULL,
  `repeat_interval` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `simple_trigger_ibfk_1` FOREIGN KEY (`trigger_id`) REFERENCES `trigger_summary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cron_trigger` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `trigger_id` int(11) NOT NULL,
  `cron_expression` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `cron_trigger_ibfk_1` FOREIGN KEY (`trigger_id`) REFERENCES `trigger_summary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `complex_trigger` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `trigger_id` int(11) NOT NULL,
  `repeat_count` int(11) NOT NULL,
  `repeat_interval` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `complex_trigger_ibfk_1` FOREIGN KEY (`trigger_id`) REFERENCES `trigger_summary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `blob_trigger` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `trigger_id` int(11) NOT NULL,
  `binary_data` blob NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `blob_trigger_ibfk_1` FOREIGN KEY (`trigger_id`) REFERENCES `trigger_summary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `lock` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_uses_trigger` (
  `task_id` int(11) NOT NULL,
  `trigger_id` int(11) NOT NULL,
  `priority` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `trigger_status` varchar(50) NOT NULL,
  PRIMARY KEY (`task_id`, `trigger_id`),
  CONSTRAINT `task_uses_trigger_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `task` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_uses_trigger_ibfk_2` FOREIGN KEY (`trigger_id`) REFERENCES `trigger_summary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `scheduler_schedules_task` (
  `scheduler_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `trigger_id` int(11) NOT NULL,
  PRIMARY KEY (`scheduler_id`, `task_id`),
  CONSTRAINT `scheduler_schedules_task_ibfk_1` FOREIGN KEY (`scheduler_id`) REFERENCES `scheduler` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `scheduler_schedules_task_ibfk_2` FOREIGN KEY (`task_id`) REFERENCES `task` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `scheduler_schedules_task_ibfk_3` FOREIGN KEY (`trigger_id`) REFERENCES `trigger_summary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `scheduler_acquires_lock` (
  `scheduler_id` int(11) NOT NULL,
  `lock_id` int(11) NOT NULL,
  `acquisition_time` datetime NOT NULL,
  `release_time` datetime NOT NULL,
  PRIMARY KEY (`scheduler_id`, `lock_id`),
  CONSTRAINT `scheduler_acquires_lock_ibfk_1` FOREIGN KEY (`scheduler_id`) REFERENCES `scheduler` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `scheduler_acquires_lock_ibfk_2` FOREIGN KEY (`lock_id`) REFERENCES `lock` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
