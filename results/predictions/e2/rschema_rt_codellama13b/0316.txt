-- Schema for database: rschema_0316
CREATE TABLE `category` (
  `category_id` int(11) NOT NULL,
  `category_name` varchar(255) NOT NULL,
  `display_order` int(11) NOT NULL,
  `parent_category_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`category_id`),
  CONSTRAINT `category_ibfk_1` FOREIGN KEY (`parent_category_id`) REFERENCES `category` (`category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `e_book` (
  `e_book_id` int(11) NOT NULL,
  `book_title` varchar(255) NOT NULL,
  `category_id` int(11) NOT NULL,
  `cover_url` varchar(255) DEFAULT NULL,
  `detailed_description` text DEFAULT NULL,
  PRIMARY KEY (`e_book_id`),
  CONSTRAINT `e_book_ibfk_1` FOREIGN KEY (`category_id`) REFERENCES `category` (`category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `document` (
  `document_id` int(11) NOT NULL,
  `document_content` text NOT NULL,
  `e_book_id` int(11) NOT NULL,
  PRIMARY KEY (`document_id`),
  CONSTRAINT `document_ibfk_1` FOREIGN KEY (`e_book_id`) REFERENCES `e_book` (`e_book_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `statistics` (
  `statistics_id` int(11) NOT NULL,
  `date` datetime NOT NULL,
  `e_book_id` int(11) NOT NULL,
  `like_count` int(11) NOT NULL,
  `read_count` int(11) NOT NULL,
  PRIMARY KEY (`statistics_id`),
  CONSTRAINT `statistics_ibfk_1` FOREIGN KEY (`e_book_id`) REFERENCES `e_book` (`e_book_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
