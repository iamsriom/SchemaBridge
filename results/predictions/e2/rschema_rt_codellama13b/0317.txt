-- Schema for database: rschema_0317
CREATE TABLE `service` (
  `service_id` int(11) NOT NULL,
  `base_url` varchar(255) DEFAULT NULL,
  `creation_time` datetime DEFAULT NULL,
  `health_url` varchar(255) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`service_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_description` (
  `sd_id` int(11) NOT NULL,
  `set_id` int(11) NOT NULL,
  `creation_time` datetime DEFAULT NULL,
  `description` text DEFAULT NULL,
  `service_id` int(11) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`set_id`, `sd_id`),
  KEY `service_id` (`service_id`),
  CONSTRAINT `service_description_ibfk_1` FOREIGN KEY (`service_id`) REFERENCES `service` (`service_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
