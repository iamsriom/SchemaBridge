-- Schema for database: rschema_0321
CREATE TABLE `general_user_information` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_information` (
  `member_id` int(11) NOT NULL,
  `member_level` varchar(50) NOT NULL,
  `status` varchar(50) NOT NULL,
  `validity_start_date` datetime NOT NULL,
  `validity_end_date` datetime NOT NULL,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `team_information` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) NOT NULL,
  `total_team_members` int(11) DEFAULT NULL,
  `team_leader_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`team_id`),
  CONSTRAINT `team_ibfk_1` FOREIGN KEY (`team_leader_id`) REFERENCES `general_user_information` (`user_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `football_player_information` (
  `football_player_id` int(11) NOT NULL,
  `affiliated_team_id` int(11) NOT NULL,
  PRIMARY KEY (`football_player_id`, `affiliated_team_id`),
  CONSTRAINT `player_ibfk_1` FOREIGN KEY (`affiliated_team_id`) REFERENCES `team_information` (`team_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `match_information` (
  `match_id` int(11) NOT NULL,
  `home_team_id` int(11) NOT NULL,
  `away_team_id` int(11) NOT NULL,
  `match_location` varchar(255) NOT NULL,
  `match_result` varchar(50) DEFAULT NULL,
  `match_time` datetime NOT NULL,
  PRIMARY KEY (`match_id`),
  CONSTRAINT `match_ibfk_1` FOREIGN KEY (`home_team_id`) REFERENCES `team_information` (`team_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `match_ibfk_2` FOREIGN KEY (`away_team_id`) REFERENCES `team_information` (`team_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
