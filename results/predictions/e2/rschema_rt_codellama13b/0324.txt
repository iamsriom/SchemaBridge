-- Schema for database: rschema_0324
CREATE TABLE `location_information_table` (
  `location_id` int(11) NOT NULL,
  `continent` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  `iso_code` char(6) NOT NULL,
  `is_in_eu` tinyint(1) NOT NULL,
  `detailed_description` text DEFAULT "",
  PRIMARY KEY (`location_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ip_address_table` (
  `ip_address` varchar(45) NOT NULL,
  `is_anonymous_proxy` tinyint(1) NOT NULL,
  `is_satellite_provider` tinyint(1) NOT NULL,
  `belongs_to_location` int(11) NOT NULL,
  PRIMARY KEY (`ip_address`),
  CONSTRAINT `ip_address_table_ibfk_1` FOREIGN KEY (`belongs_to_location`) REFERENCES `location_information_table` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
