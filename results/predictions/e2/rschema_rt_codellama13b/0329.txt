-- Schema for database: rschema_0329
CREATE TABLE `User` (
  `user_ID` int(11) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  `virtual_currency_balance` float DEFAULT NULL,
  PRIMARY KEY (`user_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Stock` (
  `company_name` varchar(255) DEFAULT NULL,
  `issue_date` datetime DEFAULT NULL,
  `stock_code` varchar(255) NOT NULL,
  `stock_price` float DEFAULT NULL,
  PRIMARY KEY (`stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Log` (
  `date` datetime DEFAULT NULL,
  `log_ID` int(11) NOT NULL,
  `record_value` float DEFAULT NULL,
  PRIMARY KEY (`log_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Investment_Stock` (
  `earnings_value` float DEFAULT NULL,
  `quantity_held` int(11) DEFAULT NULL,
  `stock_code` varchar(255) NOT NULL,
  `user_ID` int(11) NOT NULL,
  PRIMARY KEY (`user_ID`, `stock_code`),
  CONSTRAINT `User_Investment_Stock_ibfk_1` FOREIGN KEY (`user_ID`) REFERENCES `User` (`user_ID`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `User_Investment_Stock_ibfk_2` FOREIGN KEY (`stock_code`) REFERENCES `Stock` (`stock_code`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
