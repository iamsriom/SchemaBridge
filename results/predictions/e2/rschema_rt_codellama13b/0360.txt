-- Schema for database: rschema_0360
CREATE TABLE `Log` (
  `id` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `creation_time` datetime NOT NULL,
  `deletion_time` datetime DEFAULT NULL,
  `last_modified_time` datetime NOT NULL,
  `level` varchar(50) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operator` (
  `id` int(11) NOT NULL,
  `account` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operation_Log` (
  `log_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `operation_time` datetime NOT NULL,
  `execution_content` text NOT NULL,
  `operation_type` varchar(50) NOT NULL,
  PRIMARY KEY (`log_id`, `user_id`, `operation_time`),
  CONSTRAINT `operation_log_ibfk_1` FOREIGN KEY (`log_id`) REFERENCES `Log` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `operation_log_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `Operator` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
