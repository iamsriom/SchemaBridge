-- Schema for database: rschema_0361
CREATE TABLE `restaurant` (
  `restaurant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `average_cost` decimal(10,2) DEFAULT NULL,
  `business_hours` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`restaurant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_review` (
  `review_id` int(11) NOT NULL,
  `dining_time` datetime NOT NULL,
  `rating` int(11) NOT NULL,
  `review_content` text NOT NULL,
  `photos` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`review_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dish` (
  `dish_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `category` varchar(100) DEFAULT NULL,
  `price` decimal(10,2) NOT NULL,
  `monthly_sales` int(11) DEFAULT NULL,
  `satisfaction_rate` decimal(5,2) DEFAULT NULL,
  `dish_image_url` varchar(255) DEFAULT NULL,
  `restaurant_id` int(11) NOT NULL,
  PRIMARY KEY (`dish_id`),
  CONSTRAINT `dish_ibfk_1` FOREIGN KEY (`restaurant_id`) REFERENCES `restaurant` (`restaurant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review_reply` (
  `reply_id` int(11) NOT NULL,
  `reply_content` text NOT NULL,
  `reply_time` datetime NOT NULL,
  `review_id_being_replied_to` int(11) NOT NULL,
  PRIMARY KEY (`reply_id`),
  CONSTRAINT `review_reply_ibfk_1` FOREIGN KEY (`review_id_being_replied_to`) REFERENCES `user_review` (`review_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
