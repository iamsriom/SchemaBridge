-- Schema for database: rschema_0364
CREATE TABLE `Company` (
  `Company_ID` int(11) NOT NULL,
  `Company_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Company_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Login_Response_Type` (
  `Type_ID` int(11) NOT NULL,
  `Type_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Type_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Username` varchar(255) DEFAULT NULL,
  `Job_Title` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  `Company_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`User_ID`),
  CONSTRAINT `User_ibfk_1` FOREIGN KEY (`Company_ID`) REFERENCES `Company` (`Company_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Login_Response_Record` (
  `User_ID` int(11) NOT NULL,
  `Login_Time` datetime NOT NULL,
  `Response_Label` varchar(255) DEFAULT NULL,
  `Response_Type_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`User_ID`, `Login_Time`),
  CONSTRAINT `Login_Response_Record_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Login_Response_Record_ibfk_2` FOREIGN KEY (`Response_Type_ID`) REFERENCES `Login_Response_Type` (`Type_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
