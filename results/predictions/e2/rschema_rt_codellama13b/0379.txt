-- Schema for database: rschema_0379
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `customer_name` varchar(255) DEFAULT NULL,
  `contact_person` varchar(255) DEFAULT NULL,
  `customer_address` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `invoice` (
  `invoice_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `invoice_date` datetime DEFAULT NULL,
  `due_date` datetime DEFAULT NULL,
  `payment_date` datetime DEFAULT NULL,
  `notes` text DEFAULT NULL,
  `payment_amount` decimal(10,2) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `total_amount` decimal(10,2) DEFAULT NULL,
  `electronic_version_content` text DEFAULT NULL,
  `electronic_version_mime_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`invoice_id`),
  KEY `fk_customer` (`customer_id`),
  CONSTRAINT `invoice_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `terms` (
  `term_id` int(11) NOT NULL,
  `invoice_id` int(11) NOT NULL,
  `interest_rate` decimal(5,2) DEFAULT NULL,
  `payment_days` int(11) DEFAULT NULL,
  `term_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`term_id`),
  KEY `fk_invoice` (`invoice_id`),
  CONSTRAINT `terms_ibfk_1` FOREIGN KEY (`invoice_id`) REFERENCES `invoice` (`invoice_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `invoice_and_customer` (
  `invoice_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `due_date` datetime DEFAULT NULL,
  `invoice_date` datetime DEFAULT NULL,
  `payment_date` datetime DEFAULT NULL,
  `notes` text DEFAULT NULL,
  `payment_amount` decimal(10,2) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `total_amount` decimal(10,2) DEFAULT NULL,
  `electronic_version_content` text DEFAULT NULL,
  `electronic_version_mime_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`invoice_id`, `customer_id`),
  KEY `fk_customer_2` (`customer_id`),
  KEY `fk_invoice_2` (`invoice_id`),
  CONSTRAINT `invoice_and_customer_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `invoice_and_customer_ibfk_2` FOREIGN KEY (`invoice_id`) REFERENCES `invoice` (`invoice_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `invoice_and_terms` (
  `term_id` int(11) NOT NULL,
  `invoice_id` int(11) NOT NULL,
  `interest_rate` decimal(5,2) DEFAULT NULL,
  `payment_days` int(11) DEFAULT NULL,
  `term_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`term_id`, `invoice_id`),
  KEY `fk_invoice_3` (`invoice_id`),
  KEY `fk_terms` (`term_id`),
  CONSTRAINT `invoice_and_terms_ibfk_1` FOREIGN KEY (`invoice_id`) REFERENCES `invoice` (`invoice_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `invoice_and_terms_ibfk_2` FOREIGN KEY (`term_id`) REFERENCES `terms` (`term_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
