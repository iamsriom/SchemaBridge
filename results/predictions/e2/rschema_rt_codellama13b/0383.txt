-- Schema for database: rschema_0383
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `department_id` int(11) DEFAULT NULL,
  `enrollment_year` int(11) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `professor` (
  `professor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `office_number` varchar(50) DEFAULT NULL,
  `office_duration` int(11) DEFAULT NULL,
  `office_start_time` time DEFAULT NULL,
  PRIMARY KEY (`professor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `class_day` varchar(50) DEFAULT NULL,
  `course_duration` int(11) DEFAULT NULL,
  `semester` varchar(50) DEFAULT NULL,
  `teaching_professor_id` int(11) DEFAULT NULL,
  `year` int(11) DEFAULT NULL,
  PRIMARY KEY (`course_id`),
  CONSTRAINT `course_ibfk_1` FOREIGN KEY (`teaching_professor_id`) REFERENCES `professor` (`professor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_registration` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `registration_time` datetime NOT NULL,
  `grade` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`student_id`, `course_id`, `registration_time`),
  CONSTRAINT `course_registration_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_registration_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
