-- Schema for database: rschema_0384
CREATE TABLE `task` (
  `task_id` int(11) NOT NULL,
  `task_title` varchar(255) NOT NULL,
  `algorithm` varchar(255) DEFAULT NULL,
  `creation_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  `executable_file` varchar(255) DEFAULT NULL,
  `options` varchar(255) DEFAULT NULL,
  `query_statement` text DEFAULT NULL,
  `status_reason` varchar(255) DEFAULT NULL,
  `task_owner` varchar(255) DEFAULT NULL,
  `task_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`task_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parameter` (
  `parameter_id` int(11) NOT NULL,
  `parameter_name` varchar(255) NOT NULL,
  PRIMARY KEY (`parameter_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organism` (
  `organism_id` int(11) NOT NULL,
  `organism_name` varchar(255) NOT NULL,
  PRIMARY KEY (`organism_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_parameter` (
  `task_id` int(11) NOT NULL,
  `parameter_id` int(11) NOT NULL,
  `parameter_value` varchar(255) NOT NULL,
  PRIMARY KEY (`task_id`, `parameter_id`, `parameter_value`),
  CONSTRAINT `task_parameter_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_parameter_ibfk_2` FOREIGN KEY (`parameter_id`) REFERENCES `parameter` (`parameter_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_related_organism` (
  `task_id` int(11) NOT NULL,
  `organism_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`, `organism_id`),
  CONSTRAINT `task_related_organism_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_related_organism_ibfk_2` FOREIGN KEY (`organism_id`) REFERENCES `organism` (`organism_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_excluded_organism` (
  `task_id` int(11) NOT NULL,
  `organism_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`task_id`, `organism_id`),
  CONSTRAINT `task_excluded_organism_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_excluded_organism_ibfk_2` FOREIGN KEY (`organism_id`) REFERENCES `organism` (`organism_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
