-- Schema for database: rschema_0391
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `admission_date` datetime NOT NULL,
  `gender` char(1) NOT NULL,
  `major` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `post` (
  `post_id` int(11) NOT NULL,
  `author_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `publication_date` datetime NOT NULL,
  `topic` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`post_id`),
  CONSTRAINT `post_ibfk_1` FOREIGN KEY (`author_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
  `commenter_id` int(11) NOT NULL,
  `commented_post_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `comment_time` datetime NOT NULL,
  PRIMARY KEY (`commenter_id`, `commented_post_id`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`commented_post_id`) REFERENCES `post` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`commenter_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
