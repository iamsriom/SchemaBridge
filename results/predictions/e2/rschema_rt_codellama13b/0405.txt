-- Schema for database: rschema_0405
CREATE TABLE `drug` (
  `drug_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `specification` varchar(255) NOT NULL,
  `usage` varchar(255) NOT NULL,
  PRIMARY KEY (`drug_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor` (
  `doctor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`doctor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patient` (
  `patient_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `prescription` (
  `prescription_id` int(11) NOT NULL,
  `medical_orders` varchar(255) NOT NULL,
  `date` datetime NOT NULL,
  `category` varchar(255) NOT NULL,
  `issuing_doctor` int(11) NOT NULL,
  `remarks` varchar(255) NOT NULL,
  PRIMARY KEY (`prescription_id`),
  CONSTRAINT `prescription_ibfk_1` FOREIGN KEY (`issuing_doctor`) REFERENCES `doctor` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `outpatient_management` (
  `outpatient_id` int(11) NOT NULL,
  `doctor_number` int(11) NOT NULL,
  `outpatient_type` varchar(255) NOT NULL,
  `opening_time` datetime NOT NULL,
  PRIMARY KEY (`outpatient_id`),
  CONSTRAINT `outpatient_management_ibfk_1` FOREIGN KEY (`doctor_number`) REFERENCES `doctor` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patient_outpatient_consultation` (
  `patient_number` int(11) NOT NULL,
  `outpatient_number` int(11) NOT NULL,
  `consultation_time` datetime NOT NULL,
  `consultation_content` varchar(255) NOT NULL,
  PRIMARY KEY (`patient_number`, `outpatient_number`, `consultation_time`),
  CONSTRAINT `patient_outpatient_consultation_ibfk_1` FOREIGN KEY (`patient_number`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patient_outpatient_consultation_ibfk_2` FOREIGN KEY (`outpatient_number`) REFERENCES `outpatient_management` (`outpatient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patient_payment_prescription` (
  `patient_number` int(11) NOT NULL,
  `prescription_number` int(11) NOT NULL,
  `payment_amount` decimal(10,2) NOT NULL,
  `discount` decimal(5,2) NOT NULL,
  PRIMARY KEY (`prescription_number`, `patient_number`),
  CONSTRAINT `patient_payment_prescription_ibfk_1` FOREIGN KEY (`prescription_number`) REFERENCES `prescription` (`prescription_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patient_payment_prescription_ibfk_2` FOREIGN KEY (`patient_number`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `prescription_contains_drug` (
  `prescription_number` int(11) NOT NULL,
  `drug_number` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`prescription_number`, `drug_number`),
  CONSTRAINT `prescription_contains_drug_ibfk_1` FOREIGN KEY (`prescription_number`) REFERENCES `prescription` (`prescription_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `prescription_contains_drug_ibfk_2` FOREIGN KEY (`drug_number`) REFERENCES `drug` (`drug_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
