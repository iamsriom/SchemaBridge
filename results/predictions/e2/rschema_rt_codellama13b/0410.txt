-- Schema for database: rschema_0410
CREATE TABLE `user_information` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_information` (
  `vehicle_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_name` varchar(255) NOT NULL,
  `vehicle_category` varchar(100) NOT NULL,
  `vehicle_photo` varchar(255) DEFAULT NULL,
  `vehicle_status` varchar(50) NOT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_shop_information` (
  `repair_shop_id` int(11) NOT NULL AUTO_INCREMENT,
  `shop_name` varchar(255) NOT NULL,
  `contact_person` varchar(100) NOT NULL,
  `contact_number` varchar(15) NOT NULL,
  PRIMARY KEY (`repair_shop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcement_information` (
  `announcement_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `token_information` (
  `token_id` int(11) NOT NULL AUTO_INCREMENT,
  `token_value` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  `expiration_time` datetime NOT NULL,
  PRIMARY KEY (`token_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_maintenance` (
  `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_details` text NOT NULL,
  `is_reviewed` tinyint(1) NOT NULL,
  `review_reply` text DEFAULT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `vehicle_maintenance_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle_information` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_trip` (
  `trip_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `trip_time` datetime NOT NULL,
  `trip_remarks` text DEFAULT NULL,
  PRIMARY KEY (`trip_id`),
  CONSTRAINT `vehicle_trip_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle_information` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_maintenance_application` (
  `application_id` int(11) NOT NULL AUTO_INCREMENT,
  `vehicle_id` int(11) NOT NULL,
  `application_time` datetime NOT NULL,
  `application_reason` text NOT NULL,
  `is_reviewed` tinyint(1) NOT NULL,
  `review_reply` text DEFAULT NULL,
  PRIMARY KEY (`application_id`),
  CONSTRAINT `vehicle_maintenance_application_ibfk_1` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle_information` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
