-- Schema for database: rschema_0415
CREATE TABLE `user_table` (
  `user_id` int(11) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `user_ip_address` varchar(45) NOT NULL,
  `user_port_number` int(11) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `server_configuration_table` (
  `server_id` int(11) NOT NULL,
  `max_client_connections` int(11) DEFAULT NULL,
  `protocol` varchar(50) DEFAULT NULL,
  `receive_buffer_size` int(11) DEFAULT NULL,
  `server_ip_address` varchar(45) DEFAULT NULL,
  `server_port_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`server_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chat_record_table` (
  `message_id` int(11) NOT NULL,
  `content` text DEFAULT NULL,
  `sending_time` datetime DEFAULT NULL,
  `receiver_id` int(11) DEFAULT NULL,
  `sender_id` int(11) DEFAULT NULL,
  `user_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`message_id`),
  CONSTRAINT `chat_record_ibfk_1` FOREIGN KEY (`receiver_id`) REFERENCES `user_table` (`user_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `chat_record_ibfk_2` FOREIGN KEY (`sender_id`) REFERENCES `user_table` (`user_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `chat_record_ibfk_3` FOREIGN KEY (`user_id`) REFERENCES `user_table` (`user_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
