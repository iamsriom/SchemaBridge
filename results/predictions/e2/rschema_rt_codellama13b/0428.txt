-- Schema for database: rschema_0428
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL DEFAULT '',
  `class` varchar(255) NOT NULL DEFAULT '',
  `gender` varchar(10) NOT NULL DEFAULT '',
  `age` int(11) NOT NULL DEFAULT 0,
  `belonging_dormitory` int(11) NOT NULL,
  PRIMARY KEY (`student_id`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`belonging_dormitory`) REFERENCES `dormitory` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `super_user` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL DEFAULT '',
  `user_role` varchar(50) NOT NULL DEFAULT '',
  PRIMARY KEY (`user_id`),
  CONSTRAINT `super_user_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL,
  `course_leader` int(11) NOT NULL,
  `class_time` datetime NOT NULL,
  PRIMARY KEY (`course_id`),
  CONSTRAINT `course_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_ibfk_2` FOREIGN KEY (`course_leader`) REFERENCES `super_user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitory` (
  `dormitory_id` int(11) NOT NULL,
  `dormitory_location` varchar(255) NOT NULL DEFAULT '',
  PRIMARY KEY (`dormitory_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_course_selection` (
  `student_id` int(11) NOT NULL,
  `course_id` int(11) NOT NULL,
  `grade` varchar(2) NOT NULL DEFAULT '',
  PRIMARY KEY (`student_id`, `course_id`),
  CONSTRAINT `student_course_selection_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_course_selection_ibfk_2` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
