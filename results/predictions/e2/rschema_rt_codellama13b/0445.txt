CREATE TABLE `personnel` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(50) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `health_status` varchar(50) DEFAULT NULL,
  `entry_permission` boolean NOT NULL,
  `unit_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `personnel_ibfk_1` FOREIGN KEY (`unit_id`) REFERENCES `unit` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `unit` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(50) NOT NULL,
  `administrator_id` int(11) DEFAULT NULL,

  CONSTRAINT `unit_ibfk_1` FOREIGN KEY (`administrator_id`) REFERENCES `administrator` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `time_exit` datetime DEFAULT NULL,

  CONSTRAINT `health_record_entry_ibfk_1` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `health_record_entry_ibfk_2` FOREIGN KEY (`record_id`) REFERENCES `health_record` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
