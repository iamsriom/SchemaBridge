CREATE TABLE `supply_request` (
  `supply_request_id` int(11) NOT NULL,
  `chain_supermarket_id` int(11) NOT NULL,
  `request_time` datetime NOT NULL,
  PRIMARY KEY (`supply_request_id`),
  CONSTRAINT `supply_request_ibfk_1` FOREIGN KEY (`chain_supermarket_id`) REFERENCES `chain_supermarket` (`chain_supermarket_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `requested_product` (
  `supply_request_id` int(11) NOT NULL,
  `requested_product_id` int(11) NOT NULL,
  `requested_quantity` int(11) NOT NULL,
  PRIMARY KEY (`requested_product_id`, `supply_request_id`),
  CONSTRAINT `requested_product_ibfk_1` FOREIGN KEY (`supply_request_id`) REFERENCES `supply_request` (`supply_request_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `requested_product_ibfk_2` FOREIGN KEY (`requested_product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chain_supermarket` (
  `chain_supermarket_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  PRIMARY KEY (`chain_supermarket_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `distribution_order` (
  `distribution_order_id` int(11) NOT NULL,
  `distribution_time` datetime NOT NULL,
  `supply_request_id` int(11) NOT NULL,
  `distribution_status` varchar(50) NOT NULL,
  PRIMARY KEY (`distribution_order_id`),
  CONSTRAINT `distribution_order_ibfk_1` FOREIGN KEY (`supply_request_id`) REFERENCES `supply_request` (`supply_request_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_order` (
  `purchase_order_id` int(11) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `purchase_status` varchar(50) NOT NULL,
  PRIMARY KEY (`purchase_order_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchased_product` (
  `purchase_order_id` int(11) NOT NULL,
  `purchased_product_id` int(11) NOT NULL,
  `purchased_quantity` int(11) NOT NULL,
  `purchased_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`purchase_order_id`, `purchased_product_id`),
  CONSTRAINT `purchased_product_ibfk_1` FOREIGN KEY (`purchase_order_id`) REFERENCES `purchase_order` (`purchase_order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `purchased_product_ibfk_2` FOREIGN KEY (`purchased_product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
