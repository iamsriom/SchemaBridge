-- Schema for database: rschema_0462
CREATE TABLE `member` (
  `member_id` INTEGER NOT NULL,
  `name` VARCHAR(255) DEFAULT NULL,
  `password` VARCHAR(255) DEFAULT NULL,
  `username` VARCHAR(255) DEFAULT NULL,
  `phone` INTEGER DEFAULT NULL,
  `address` VARCHAR(255) DEFAULT NULL,
  `postal_code` INTEGER DEFAULT NULL,
  `points` FLOAT DEFAULT NULL,
  `consumption_amount` FLOAT DEFAULT NULL,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `book` (
  `isbn` VARCHAR(255) NOT NULL,
  `book_title` VARCHAR(255) DEFAULT NULL,
  `author` VARCHAR(255) DEFAULT NULL,
  `publisher` VARCHAR(255) DEFAULT NULL,
  `category_name` VARCHAR(255) DEFAULT NULL,
  `price` FLOAT DEFAULT NULL,
  `publication_date` DATETIME DEFAULT NULL,
  PRIMARY KEY (`isbn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `order_id` INTEGER NOT NULL,
  `member_id` INTEGER NOT NULL,
  `order_date` DATETIME NOT NULL,
  `sales_amount` FLOAT NOT NULL,
  `shipping_date` DATETIME NOT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase` (
  `isbn` VARCHAR(255) NOT NULL,
  `order_id` INTEGER NOT NULL,
  `quantity` INTEGER DEFAULT NULL,
  PRIMARY KEY (`isbn`, `order_id`),
  CONSTRAINT `purchase_ibfk_1` FOREIGN KEY (`isbn`) REFERENCES `book` (`isbn`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `purchase_ibfk_2` FOREIGN KEY (`order_id`) REFERENCES `order` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
