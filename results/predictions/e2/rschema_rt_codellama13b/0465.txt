-- Schema for database: rschema_0465
CREATE TABLE `user` (
  `email_address` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `account_creation_time` datetime NOT NULL,
  PRIMARY KEY (`email_address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attachment_information` (
  `attachment_id` int(11) NOT NULL AUTO_INCREMENT,
  `file_name` varchar(255) NOT NULL,
  `file_path` varchar(255) NOT NULL,
  `file_size` int(11) NOT NULL,
  `upload_time` datetime NOT NULL,
  PRIMARY KEY (`attachment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `email` (
  `email_id` int(11) NOT NULL AUTO_INCREMENT,
  `sender_email_address` varchar(255) NOT NULL,
  `recipient_email_address` varchar(255) NOT NULL,
  `email_subject` varchar(255) NOT NULL,
  `email_content` text NOT NULL,
  `delivery_time` datetime NOT NULL,
  `send_time` datetime NOT NULL,
  `attachment_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`email_id`),
  CONSTRAINT `email_ibfk_1` FOREIGN KEY (`attachment_id`) REFERENCES `attachment_information` (`attachment_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `email_ibfk_2` FOREIGN KEY (`sender_email_address`) REFERENCES `user` (`email_address`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `email_ibfk_3` FOREIGN KEY (`recipient_email_address`) REFERENCES `user` (`email_address`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
