-- Schema for database: rschema_0477
CREATE TABLE `billing_method` (
  `billing_method_id` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `calculation_function` varchar(255) NOT NULL,
  PRIMARY KEY (`billing_method_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_record` (
  `transaction_id` varchar(255) NOT NULL,
  `input_amount` varchar(255) NOT NULL,
  `billing_method_id` varchar(255) NOT NULL,
  `actual_payment_amount` varchar(255) NOT NULL,
  `transaction_time` varchar(255) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  CONSTRAINT `transaction_record_ibfk_billing_method_id` FOREIGN KEY (`billing_method_id`) REFERENCES `billing_method` (`billing_method_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
