-- Schema for database: rschema_0486
CREATE TABLE `mine` (
  `mine_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mineral_resource` (
  `resource_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `reserves` int(11) NOT NULL,
  `discovery_date` datetime NOT NULL,
  `first_discovered_mine` int(11) NOT NULL,
  PRIMARY KEY (`resource_id`),
  CONSTRAINT `mineral_resource_ibfk_1` FOREIGN KEY (`first_discovered_mine`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mining_record` (
  `record_id` int(11) NOT NULL,
  `mineral_resource` int(11) NOT NULL,
  `mining_date` datetime NOT NULL,
  `mining_volume` int(11) NOT NULL,
  `mining_cost` decimal(10,2) NOT NULL,
  `mine` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  UNIQUE KEY `mining_record_mineral_resource_unique` (`mineral_resource`),
  UNIQUE KEY `mining_record_mine_unique` (`mine`),
  KEY `mining_record_mineral_resource_index` (`mineral_resource`),
  KEY `mining_record_mine_index` (`mine`),
  CONSTRAINT `mining_record_ibfk_1` FOREIGN KEY (`mineral_resource`) REFERENCES `mineral_resource` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `mining_record_ibfk_2` FOREIGN KEY (`mine`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
