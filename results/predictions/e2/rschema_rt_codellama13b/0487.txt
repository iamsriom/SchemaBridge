CREATE TABLE `codelist` (
  `codework` varchar(255) NOT NULL,
  `code` varchar(255) NOT NULL,
  PRIMARY KEY (`codework`, `code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `codelist_relation` (
  `id` varchar(255) NOT NULL,
  `codework1` varchar(255) NOT NULL,
  `code1` varchar(255) NOT NULL,
  `codework2` varchar(255) NOT NULL,
  `code2` varchar(255) NOT NULL,
  `valid_from` varchar(255) NOT NULL,
  `valid_to` varchar(255) NOT NULL,
  `created_by` varchar(255) NOT NULL,
  `created_time` varchar(255) NOT NULL,
  `updated_by` varchar(255) NOT NULL,
  `updated_time` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `codelist_relation_ibfk_1` FOREIGN KEY (`codework1`, `code1`) REFERENCES `codelist` (`codework`, `code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `codelist_relation_ibfk_2` FOREIGN KEY (`codework2`, `code2`) REFERENCES `codelist` (`codework`, `code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
