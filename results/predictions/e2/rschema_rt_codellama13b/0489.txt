-- Schema for database: rschema_0489
CREATE TABLE `Application_Table` (
  `Application_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Application_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Table` (
  `ID` int(11) NOT NULL,
  `Username` varchar(255) NOT NULL,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Channel_Table` (
  `Belonging_Application` int(11) NOT NULL,
  `Channel_ID` int(11) NOT NULL,
  `Creation_Time` datetime NOT NULL,
  `Creator` int(11) NOT NULL,
  `Is_Private` boolean NOT NULL,
  `Status` varchar(255) NOT NULL,
  PRIMARY KEY (`Channel_ID`),
  CONSTRAINT `Channel_Table_ibfk_1` FOREIGN KEY (`Belonging_Application`) REFERENCES `Application_Table` (`Application_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Channel_Table_ibfk_2` FOREIGN KEY (`Creator`) REFERENCES `User_Table` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Channel_Event_Table` (
  `Channel_ID` int(11) NOT NULL,
  `Event_ID` int(11) NOT NULL,
  `Event_Type` varchar(255) NOT NULL,
  `Payload_Content` varchar(255) NOT NULL,
  `Timestamp` datetime NOT NULL,
  PRIMARY KEY (`Event_ID`),
  CONSTRAINT `Channel_Event_Table_ibfk_1` FOREIGN KEY (`Channel_ID`) REFERENCES `Channel_Table` (`Channel_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
