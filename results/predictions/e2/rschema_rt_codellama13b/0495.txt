-- Schema for database: rschema_0495
CREATE TABLE `ward` (
  `ward_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `number_of_beds` int(11) NOT NULL,
  `ward_level` varchar(50) NOT NULL,
  PRIMARY KEY (`ward_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medicine` (
  `medicine_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  `remarks` text DEFAULT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`medicine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patient` (
  `patient_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `ward_id` int(11) NOT NULL,
  PRIMARY KEY (`patient_id`),
  CONSTRAINT `patient_ibfk_1` FOREIGN KEY (`ward_id`) REFERENCES `ward` (`ward_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medication_dispensing` (
  `patient_id` int(11) NOT NULL,
  `medicine_id` int(11) NOT NULL,
  `medication_retrieval_time` datetime NOT NULL,
  `quantity_retrieved` int(11) NOT NULL,
  PRIMARY KEY (`patient_id`, `medicine_id`, `medication_retrieval_time`),
  CONSTRAINT `medication_dispensing_ibfk_1` FOREIGN KEY (`medicine_id`) REFERENCES `medicine` (`medicine_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `medication_dispensing_ibfk_2` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `doctor` (
  `doctor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `title` varchar(100) NOT NULL,
  `years_of_service` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`doctor_id`),
  CONSTRAINT `doctor_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_phone_number` varchar(15) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge_id` int(11) DEFAULT NULL,
  `total_personnel` int(11) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultation_record` (
  `patient_id` int(11) NOT NULL,
  `doctor_id` int(11) NOT NULL,
  `consultation_time` datetime NOT NULL,
  `diagnosis_description` text DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  `treatment_results` text DEFAULT NULL,
  PRIMARY KEY (`patient_id`, `doctor_id`, `consultation_time`),
  CONSTRAINT `consultation_record_ibfk_1` FOREIGN KEY (`doctor_id`) REFERENCES `doctor` (`doctor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `consultation_record_ibfk_2` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
