-- Schema for database: rschema_0510
CREATE TABLE `vehicle_information_table` (
  `contact_information` varchar(255) DEFAULT NULL,
  `license_plate_number` varchar(20) NOT NULL,
  `owner_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`license_plate_number`),
  CONSTRAINT `vehicle_information_table_ibfk_1` FOREIGN KEY (`owner_name`) REFERENCES `parking_space_table` (`owner`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `parking_space_table` (
  `owner` varchar(20) NOT NULL,
  `parking_space_number` int(11) NOT NULL,
  `status` enum('available', 'occupied') DEFAULT NULL,
  PRIMARY KEY (`parking_space_number`),
  CONSTRAINT `parking_space_table_ibfk_1` FOREIGN KEY (`owner`) REFERENCES `vehicle_information_table` (`license_plate_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
