-- Schema for database: rschema_0515
CREATE TABLE `User` (
  `Phone_number` varchar(255) NOT NULL,
  `Credit_status` varchar(255) DEFAULT NULL,
  `Verification_code` varchar(255) NOT NULL,
  PRIMARY KEY (`Phone_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Bike` (
  `Bike_ID` int(11) NOT NULL,
  `Bike_status` varchar(255) DEFAULT NULL,
  `Location_info` varchar(255) DEFAULT NULL,
  `QR_code` blob DEFAULT NULL,
  `Unlock_password` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Bike_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Trip` (
  `Trip_ID` int(11) NOT NULL,
  `Bike_ID` int(11) NOT NULL,
  `User_phone_number` varchar(255) NOT NULL,
  `Start_time` datetime DEFAULT NULL,
  `End_time` datetime DEFAULT NULL,
  `Fee` decimal(10,2) DEFAULT NULL,
  `Payment_status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Trip_ID`),
  CONSTRAINT `Trip_ibfk_1` FOREIGN KEY (`Bike_ID`) REFERENCES `Bike` (`Bike_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Trip_ibfk_2` FOREIGN KEY (`User_phone_number`) REFERENCES `User` (`Phone_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Malfunction` (
  `Report_time` datetime NOT NULL,
  `Bike_ID` int(11) NOT NULL,
  `User_phone_number` varchar(255) NOT NULL,
  `Malfunction_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Report_time`),
  CONSTRAINT `Malfunction_ibfk_1` FOREIGN KEY (`Bike_ID`) REFERENCES `Bike` (`Bike_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Malfunction_ibfk_2` FOREIGN KEY (`User_phone_number`) REFERENCES `User` (`Phone_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_trip` (
  `Trip` int(11) NOT NULL,
  `User_phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`Trip`, `User_phone_number`),
  CONSTRAINT `User_trip_ibfk_1` FOREIGN KEY (`Trip`) REFERENCES `Trip` (`Trip_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_trip_ibfk_2` FOREIGN KEY (`User_phone_number`) REFERENCES `User` (`Phone_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Use_bike` (
  `Bike_ID` int(11) NOT NULL,
  `Trip_ID` int(11) NOT NULL,
  PRIMARY KEY (`Bike_ID`, `Trip_ID`),
  CONSTRAINT `Use_bike_ibfk_1` FOREIGN KEY (`Bike_ID`) REFERENCES `Bike` (`Bike_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Use_bike_ibfk_2` FOREIGN KEY (`Trip_ID`) REFERENCES `Trip` (`Trip_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
