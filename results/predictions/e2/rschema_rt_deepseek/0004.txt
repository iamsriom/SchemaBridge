-- Schema for database: rschema_0004
CREATE TABLE `Customer` (
  `Address` varchar(255) DEFAULT NULL,
  `Customer_Number` varchar(255) NOT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Customer_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `Service_Plan` (
  `Plan_Name` varchar(255) DEFAULT NULL,
  `Plan_Number` int(11) NOT NULL,
  `Price` int(11) DEFAULT NULL,
  `Service_Content` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Plan_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `Subscription_Record` (
  `Cancellation_Time` datetime DEFAULT NULL,
  `Customer_Number` varchar(255) NOT NULL,
  `Operating_Personnel` varchar(255) DEFAULT NULL,
  `Plan_Number` int(11) NOT NULL,
  `Subscription_Time` datetime NOT NULL,
  PRIMARY KEY (`Customer_Number`, `Plan_Number`, `Subscription_Time`),
  CONSTRAINT `Subscription_Record_ibfk_1` FOREIGN KEY (`Customer_Number`) REFERENCES `Customer` (`Customer_Number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Subscription_Record_ibfk_2` FOREIGN KEY (`Plan_Number`) REFERENCES `Service_Plan` (`Plan_Number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
