-- Schema for database: rschema_0005
CREATE TABLE `Service` (
  `Service_ID` int(11) NOT NULL,
  `Service_description` varchar(255) DEFAULT NULL,
  `Service_name` varchar(255) NOT NULL,
  `Service_price` decimal(10,2) NOT NULL,
  `Service_type` varchar(100) NOT NULL,
  PRIMARY KEY (`Service_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `User_name` varchar(255) NOT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Service_usage_record` (
  `Operating_user_ID` int(11) NOT NULL,
  `Service_ID` int(11) NOT NULL,
  `Usage_duration` varchar(50) NOT NULL,
  `Usage_time` datetime NOT NULL,
  PRIMARY KEY (`Operating_user_ID`, `Service_ID`),
  CONSTRAINT `service_usage_record_ibfk_1` FOREIGN KEY (`Operating_user_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_usage_record_ibfk_2` FOREIGN KEY (`Service_ID`) REFERENCES `Service` (`Service_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
