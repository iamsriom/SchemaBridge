-- Schema for database: rschema_0014
CREATE TABLE `irrigation_system` (
  `system_id` int(11) NOT NULL,
  `installation_date` datetime DEFAULT NULL,
  `maintenance_records` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`system_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL,
  `area` float DEFAULT NULL,
  `current_irrigation_status` boolean DEFAULT NULL,
  `soil_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `irrigation_record` (
  `irrigation_date` datetime NOT NULL,
  `system_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `irrigation_amount` int(11) DEFAULT NULL,
  `irrigation_duration` float DEFAULT NULL,
  PRIMARY KEY (`irrigation_date`, `system_id`, `plot_id`),
  CONSTRAINT `irrigation_record_ibfk_1` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `irrigation_record_ibfk_2` FOREIGN KEY (`system_id`) REFERENCES `irrigation_system` (`system_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
