-- Schema for database: rschema_0015
CREATE TABLE `plot` (
  `plot_number` int(11) DEFAULT NULL,
  `area` float DEFAULT NULL,
  `soil_type` varchar(255) DEFAULT NULL,
  `current_pest_and_disease_status` varchar(255) DEFAULT NULL,
  `geographic_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`plot_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pest_and_disease` (
  `pest_and_disease_number` int(11) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `degree_of_harm` varchar(255) DEFAULT NULL,
  `prevention_and_control_methods` varchar(255) DEFAULT NULL,
  `season_of_occurrence` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pest_and_disease_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_number` int(11) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_record` (
  `plot_number` int(11) NOT NULL,
  `pest_number` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `monitoring_results` varchar(255) DEFAULT NULL,
  `monitoring_employee_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`pest_number`, `monitoring_time`, `plot_number`),
  CONSTRAINT `monitoring_record_ibfk_pest_number` FOREIGN KEY (`pest_number`) REFERENCES `pest_and_disease` (`pest_and_disease_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_plot_number` FOREIGN KEY (`plot_number`) REFERENCES `plot` (`plot_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
