-- Schema for database: rschema_0016
CREATE TABLE `farm` (
  `farm_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`farm_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL,
  `area` varchar(255) DEFAULT NULL,
  `soil_type` varchar(255) DEFAULT NULL,
  `current_status` boolean DEFAULT NULL,
  PRIMARY KEY (`plot_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `planting_plan` (
  `farm_id` int(11) NOT NULL,
  `plot_id` int(11) NOT NULL,
  `planting_time` datetime NOT NULL,
  `crop_type` varchar(255) DEFAULT NULL,
  `operator` varchar(255) DEFAULT NULL,
  `planting_quantity` varchar(255) DEFAULT NULL,
  `expected_harvest_time` datetime DEFAULT NULL,
  PRIMARY KEY (`farm_id`, `plot_id`, `planting_time`),
  CONSTRAINT `planting_plan_ibfk_1` FOREIGN KEY (`farm_id`) REFERENCES `farm` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `planting_plan_ibfk_2` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
