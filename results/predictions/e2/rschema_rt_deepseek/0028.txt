-- Schema for database: rschema_0028
CREATE TABLE `timber` (
  `timber_id` varchar(255) NOT NULL,
  `timber_name` varchar(255) DEFAULT NULL,
  `timber_type` varchar(255) DEFAULT NULL,
  `procurement_price` varchar(255) DEFAULT NULL,
  `processing_cost` varchar(255) DEFAULT NULL,
  `sales_price` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`timber_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` varchar(255) NOT NULL,
  `department_name` varchar(255) DEFAULT NULL,
  `supervisor` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`supervisor`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` varchar(255) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `gender` varchar(255) DEFAULT NULL,
  `phone_number` varchar(255) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timber_procurement` (
  `timber_id` varchar(255) NOT NULL,
  `employee_id` varchar(255) NOT NULL,
  `procurement_quantity` varchar(255) DEFAULT NULL,
  `supplier` varchar(255) DEFAULT NULL,
  `procurement_time` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`timber_id`, `employee_id`),
  CONSTRAINT `timber_procurement_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `timber_procurement_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `timber_processing` (
  `timber_id` varchar(255) NOT NULL,
  `employee_id` varchar(255) NOT NULL,
  `processing_quantity` varchar(255) DEFAULT NULL,
  `processing_time` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`timber_id`, `employee_id`),
  CONSTRAINT `timber_processing_ibfk_1` FOREIGN KEY (`timber_id`) REFERENCES `timber` (`timber_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `timber_processing_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
