-- Schema for database: rschema_0029
CREATE TABLE `furniture` (
  `furniture_id` int(11) NOT NULL,
  `furniture_name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `production_date` datetime NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`furniture_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `supervisor` int(11) DEFAULT NULL,
  `department_name` varchar(255) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`supervisor`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `position` varchar(255) NOT NULL,
  `department` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production` (
  `furniture_id` int(11) NOT NULL,
  `producer_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  `production_quantity` int(11) NOT NULL,
  PRIMARY KEY (`production_time`, `furniture_id`, `producer_id`),
  CONSTRAINT `production_ibfk_1` FOREIGN KEY (`furniture_id`) REFERENCES `furniture` (`furniture_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_ibfk_2` FOREIGN KEY (`producer_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `furniture_id` int(11) NOT NULL,
  `sales_representative_id` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `customer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`sales_representative_id`, `furniture_id`, `sales_time`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`furniture_id`) REFERENCES `furniture` (`furniture_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`sales_representative_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
