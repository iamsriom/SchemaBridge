-- Schema for database: rschema_0032
CREATE TABLE `Rental_Location` (
  `Rental_Location_ID` int(11) NOT NULL,
  `Rental_Location_Name` varchar(255) DEFAULT NULL,
  `Rental_Location_Address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Rental_Location_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Customer_Name` varchar(255) DEFAULT NULL,
  `Customer_Phone_Number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Vehicle` (
  `Vehicle_ID` int(11) NOT NULL,
  `Vehicle_Model` varchar(255) DEFAULT NULL,
  `Rental_Location_ID` int(11) NOT NULL,
  PRIMARY KEY (`Vehicle_ID`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`Rental_Location_ID`) REFERENCES `Rental_Location` (`Rental_Location_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Rental_Task` (
  `Customer_ID` int(11) NOT NULL,
  `Vehicle_ID` int(11) NOT NULL,
  `Rental_Date` datetime NOT NULL,
  `Rental_Duration` int(11) DEFAULT NULL,
  `Rental_Cost` decimal(10,2) DEFAULT NULL,
  `Customer_Feedback` text DEFAULT NULL,
  `Vehicle_Condition_After_Rental` text DEFAULT NULL,
  PRIMARY KEY (`Vehicle_ID`, `Customer_ID`, `Rental_Date`),
  CONSTRAINT `rental_task_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `rental_task_ibfk_2` FOREIGN KEY (`Vehicle_ID`) REFERENCES `Vehicle` (`Vehicle_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
