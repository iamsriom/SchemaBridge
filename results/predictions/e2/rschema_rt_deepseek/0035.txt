-- Schema for database: rschema_0035
CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Project` (
  `Project_ID` int(11) NOT NULL,
  `Project_Name` varchar(255) DEFAULT NULL,
  `Project_Budget` decimal(10,2) NOT NULL,
  `Start_Date` datetime NOT NULL,
  `Estimated_Completion_Date` datetime NOT NULL,
  `Actual_Completion_Date` datetime DEFAULT NULL,
  PRIMARY KEY (`Project_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer_Project_Relationship` (
  `Customer_ID` int(11) NOT NULL,
  `Project_ID` int(11) NOT NULL,
  `Feedback` text DEFAULT NULL,
  `Satisfaction_Rating` int(11) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`, `Project_ID`),
  CONSTRAINT `Customer_Project_Relationship_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Customer_Project_Relationship_ibfk_2` FOREIGN KEY (`Project_ID`) REFERENCES `Project` (`Project_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
