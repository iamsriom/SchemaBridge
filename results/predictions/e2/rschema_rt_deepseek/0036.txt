-- Schema for database: rschema_0036
CREATE TABLE `project` (
  `project_number` int(11) NOT NULL,
  `project_name` varchar(255) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `start_date` datetime DEFAULT NULL,
  `expected_completion_date` datetime DEFAULT NULL,
  `actual_completion_date` datetime DEFAULT NULL,
  `budget` float DEFAULT NULL,
  PRIMARY KEY (`project_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `contract` (
  `contract_number` int(11) NOT NULL,
  `contract_name` varchar(255) DEFAULT NULL,
  `contract_amount` float DEFAULT NULL,
  `contract_status` varchar(255) DEFAULT NULL,
  `signing_date` datetime DEFAULT NULL,
  `project_number` int(11) NOT NULL,
  PRIMARY KEY (`contract_number`),
  CONSTRAINT `contract_ibfk_1` FOREIGN KEY (`project_number`) REFERENCES `project` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_includes_contract` (
  `project_number` int(11) NOT NULL,
  `contract_number` int(11) NOT NULL,
  PRIMARY KEY (`project_number`, `contract_number`),
  CONSTRAINT `project_includes_contract_ibfk_1` FOREIGN KEY (`project_number`) REFERENCES `project` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_includes_contract_ibfk_2` FOREIGN KEY (`contract_number`) REFERENCES `contract` (`contract_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `contract_change_record` (
  `change_number` int(11) NOT NULL,
  `change_amount` int(11) DEFAULT NULL,
  `change_content` varchar(255) DEFAULT NULL,
  `change_date` datetime DEFAULT NULL,
  `contract_number` int(11) NOT NULL,
  PRIMARY KEY (`change_number`),
  CONSTRAINT `contract_change_record_ibfk_1` FOREIGN KEY (`contract_number`) REFERENCES `contract` (`contract_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
