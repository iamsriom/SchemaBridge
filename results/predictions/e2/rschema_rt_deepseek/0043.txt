-- Schema for database: rschema_0043
CREATE TABLE `loan_application` (
  `application_number` varchar(255) NOT NULL,
  `property_number` varchar(255) NOT NULL,
  `loan_amount` varchar(255) NOT NULL,
  `loan_term` varchar(255) NOT NULL,
  `approval_status` varchar(255) NOT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `applicant_phone_number` varchar(255) NOT NULL,
  `applicant_id_number` varchar(255) NOT NULL,
  `application_time` varchar(255) NOT NULL,
  PRIMARY KEY (`application_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_officer` (
  `approval_officer_number` varchar(255) NOT NULL,
  `approval_officer_name` varchar(255) NOT NULL,
  `approval_officer_phone_number` varchar(255) NOT NULL,
  PRIMARY KEY (`approval_officer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval` (
  `approval_officer_number` varchar(255) NOT NULL,
  `loan_application_number` varchar(255) NOT NULL,
  `approval_time` varchar(255) NOT NULL,
  `approval_result` varchar(255) NOT NULL,
  PRIMARY KEY (`approval_officer_number`, `loan_application_number`, `approval_time`),
  CONSTRAINT `loan_approval_ibfk_1` FOREIGN KEY (`approval_officer_number`) REFERENCES `approval_officer` (`approval_officer_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_ibfk_2` FOREIGN KEY (`loan_application_number`) REFERENCES `loan_application` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
