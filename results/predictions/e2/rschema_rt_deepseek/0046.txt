-- Schema for database: rschema_0046
CREATE TABLE `Member` (
  `Member_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Phone_Number` int(11) NOT NULL,
  `Current_Points` int(11) NOT NULL,
  PRIMARY KEY (`Member_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Shopping_Record` (
  `Member_ID` int(11) NOT NULL,
  `Shopping_Time` datetime NOT NULL,
  `Points_Change` int(11) NOT NULL,
  `Total_Price_of_Goods` float NOT NULL,
  PRIMARY KEY (`Member_ID`, `Shopping_Time`),
  CONSTRAINT `shopping_record_ibfk_1` FOREIGN KEY (`Member_ID`) REFERENCES `Member` (`Member_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Point_Redemption_Record` (
  `Member_ID` int(11) NOT NULL,
  `Redemption_Time` datetime NOT NULL,
  `Points_Consumed` int(11) NOT NULL,
  `Redeemed_Product_Name` varchar(255) NOT NULL,
  `Redemption_Quantity` int(11) NOT NULL,
  PRIMARY KEY (`Member_ID`, `Redemption_Time`),
  CONSTRAINT `point_redemption_record_ibfk_1` FOREIGN KEY (`Member_ID`) REFERENCES `Member` (`Member_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
