-- Schema for database: rschema_0049
CREATE TABLE `Program` (
  `Program_ID` int(11) NOT NULL,
  `Program_Name` varchar(255) NOT NULL,
  `Program_Type` varchar(255) NOT NULL,
  PRIMARY KEY (`Program_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Host` (
  `Host_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Contact_Information` varchar(255) NOT NULL,
  PRIMARY KEY (`Host_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Host_Program_Assignment` (
  `Host_ID` int(11) NOT NULL,
  `Program_ID` int(11) NOT NULL,
  `Time_Slot` varchar(255) NOT NULL,
  PRIMARY KEY (`Host_ID`, `Program_ID`),
  CONSTRAINT `host_program_assignment_ibfk_1` FOREIGN KEY (`Host_ID`) REFERENCES `Host` (`Host_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `host_program_assignment_ibfk_2` FOREIGN KEY (`Program_ID`) REFERENCES `Program` (`Program_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
