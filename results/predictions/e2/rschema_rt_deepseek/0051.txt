-- Schema for database: rschema_0051
CREATE TABLE `Reservoir_and_River` (
  `Resource_ID` int(11) NOT NULL,
  `Location` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Resource_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Monitoring_Personnel` (
  `Monitoring_Personnel_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone` varchar(20) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Monitoring_Personnel_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Maintenance_Personnel` (
  `Maintenance_Personnel_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone` varchar(20) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Maintenance_Personnel_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Resource_Monitoring` (
  `Resource_ID` int(11) NOT NULL,
  `Monitoring_Personnel_ID` int(11) NOT NULL,
  `Monitoring_Point` varchar(255) DEFAULT NULL,
  `Monitoring_Time` datetime DEFAULT NULL,
  `Pollution_Level` varchar(255) DEFAULT NULL,
  `Water_Quality_Indicators` varchar(255) DEFAULT NULL,
  `Water_Volume_Data` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Resource_ID`, `Monitoring_Personnel_ID`),
  CONSTRAINT `Resource_Monitoring_ibfk_1` FOREIGN KEY (`Monitoring_Personnel_ID`) REFERENCES `Monitoring_Personnel` (`Monitoring_Personnel_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Resource_Monitoring_ibfk_2` FOREIGN KEY (`Resource_ID`) REFERENCES `Reservoir_and_River` (`Resource_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Resource_Maintenance` (
  `Resource_ID` int(11) NOT NULL,
  `Maintenance_Personnel_ID` int(11) NOT NULL,
  `Maintenance_Content` varchar(255) DEFAULT NULL,
  `Maintenance_Cost` decimal(10,2) DEFAULT NULL,
  `Maintenance_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Resource_ID`, `Maintenance_Personnel_ID`),
  CONSTRAINT `Resource_Maintenance_ibfk_1` FOREIGN KEY (`Maintenance_Personnel_ID`) REFERENCES `Maintenance_Personnel` (`Maintenance_Personnel_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Resource_Maintenance_ibfk_2` FOREIGN KEY (`Resource_ID`) REFERENCES `Reservoir_and_River` (`Resource_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
