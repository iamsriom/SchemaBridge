-- Schema for database: rschema_0057
CREATE TABLE `equipment` (
  `id` int(11) NOT NULL,
  `installation_date` datetime NOT NULL,
  `installation_location` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintainer` (
  `id` int(11) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `maintainer_id` int(11) NOT NULL,
  `equipment_id` int(11) NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  PRIMARY KEY (`maintainer_id`, `equipment_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`maintainer_id`) REFERENCES `maintainer` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_2` FOREIGN KEY (`equipment_id`) REFERENCES `equipment` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
