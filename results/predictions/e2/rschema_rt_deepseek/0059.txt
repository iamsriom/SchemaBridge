-- Schema for database: rschema_0059
CREATE TABLE `Handler` (
  `Handler_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  `Department` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Handler_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Fault_Handling` (
  `Fault_ID` int(11) NOT NULL,
  `Description` varchar(255) DEFAULT NULL,
  `Occurrence_Date` datetime DEFAULT NULL,
  `Resolution_Date` datetime DEFAULT NULL,
  `Handler_ID` int(11) NOT NULL,
  `Affected_User_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Fault_ID`),
  CONSTRAINT `fault_handling_ibfk_1` FOREIGN KEY (`Handler_ID`) REFERENCES `Handler` (`Handler_ID`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
