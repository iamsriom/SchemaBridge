-- Schema for database: rschema_0062
CREATE TABLE `document` (
  `creation_date` datetime NOT NULL,
  `document_id` int(11) NOT NULL,
  `title` varchar(255) DEFAULT NULL,
  `version_number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`document_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `author` (
  `author_id` int(11) NOT NULL,
  `date_of_birth` datetime DEFAULT NULL,
  `gender` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone_number` varchar(255) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`author_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `modification_record` (
  `document_id` int(11) NOT NULL,
  `modification_content` varchar(255) DEFAULT NULL,
  `modification_date` datetime NOT NULL,
  `modifying_author_id` int(11) NOT NULL,
  `version_number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`document_id`, `modification_date`, `modifying_author_id`),
  CONSTRAINT `modification_record_ibfk_1` FOREIGN KEY (`document_id`) REFERENCES `document` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `modification_record_ibfk_2` FOREIGN KEY (`modifying_author_id`) REFERENCES `author` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
