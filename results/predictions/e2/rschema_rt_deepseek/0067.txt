-- Schema for database: rschema_0067
CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Phone` varchar(255) DEFAULT NULL,
  `Credit_score` int(11) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Rental_record` (
  `Record_ID` int(11) NOT NULL,
  `Customer_ID` int(11) NOT NULL,
  `Rental_fee` float DEFAULT NULL,
  `Rental_time` datetime DEFAULT NULL,
  `Return_time` datetime DEFAULT NULL,
  PRIMARY KEY (`Record_ID`),
  CONSTRAINT `Rental_record_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Credit_change_record` (
  `Record_ID` int(11) NOT NULL,
  `Customer_ID` int(11) NOT NULL,
  `Change_time` datetime DEFAULT NULL,
  `Change_reason` varchar(255) DEFAULT NULL,
  `Old_credit_value` float DEFAULT NULL,
  `New_credit_value` float DEFAULT NULL,
  PRIMARY KEY (`Record_ID`),
  CONSTRAINT `Credit_change_record_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
