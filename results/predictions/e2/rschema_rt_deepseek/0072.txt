-- Schema for database: rschema_0072
CREATE TABLE `oil_and_gas_field` (
  `Field_ID` int(11) NOT NULL,
  `Location` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Person_in_charge` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Field_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pipeline` (
  `Pipeline_ID` int(11) NOT NULL,
  `Starting_point` varchar(255) DEFAULT NULL,
  `Ending_point` varchar(255) DEFAULT NULL,
  `Length` float DEFAULT NULL,
  PRIMARY KEY (`Pipeline_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `Maintenance_ID` int(11) NOT NULL,
  `Maintenance_content` varchar(255) DEFAULT NULL,
  `Maintenance_cost` float DEFAULT NULL,
  `Maintenance_date` datetime DEFAULT NULL,
  `Maintenance_personnel` varchar(255) DEFAULT NULL,
  `Pipeline_ID` int(11) NOT NULL,
  PRIMARY KEY (`Maintenance_ID`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`Pipeline_ID`) REFERENCES `pipeline` (`Pipeline_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
