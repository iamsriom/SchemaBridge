-- Schema for database: rschema_0075
CREATE TABLE `Analyst` (
  `Analyst_ID` int(11) NOT NULL,
  `Contact_Number` int(11) DEFAULT NULL,
  `Date_of_Hire` datetime DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Analyst_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Stock` (
  `Current_Price` float DEFAULT NULL,
  `Stock_Code` varchar(255) NOT NULL,
  `Stock_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Stock_Code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Analysis_Record` (
  `Analysis_Reasoning` varchar(255) DEFAULT NULL,
  `Analysis_Result` varchar(255) DEFAULT NULL,
  `Analyst_ID` int(11) NOT NULL,
  `Date_of_Analysis` datetime DEFAULT NULL,
  `Stock_Code` varchar(255) NOT NULL,
  PRIMARY KEY (`Analyst_ID`, `Stock_Code`, `Date_of_Analysis`),
  CONSTRAINT `Analysis_Record_ibfk_1` FOREIGN KEY (`Analyst_ID`) REFERENCES `Analyst` (`Analyst_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Analysis_Record_ibfk_2` FOREIGN KEY (`Stock_Code`) REFERENCES `Stock` (`Stock_Code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Stock_Historical_Price` (
  `Closing_Price` float DEFAULT NULL,
  `Date` datetime NOT NULL,
  `Highest_Price` float DEFAULT NULL,
  `Lowest_Price` float DEFAULT NULL,
  `Opening_Price` float DEFAULT NULL,
  `Stock_Code` varchar(255) NOT NULL,
  `Trading_Volume` int(11) DEFAULT NULL,
  PRIMARY KEY (`Stock_Code`, `Date`),
  CONSTRAINT `Stock_Historical_Price_ibfk_1` FOREIGN KEY (`Stock_Code`) REFERENCES `Stock` (`Stock_Code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
