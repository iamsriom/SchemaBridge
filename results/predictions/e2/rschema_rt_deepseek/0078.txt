CREATE TABLE `Author` (
  `Author_ID` int(11) NOT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Gender` varchar(255) DEFAULT NULL,
  `Introduction` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Author_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Article` (
  `Article_ID` int(11) NOT NULL,
  `Article_collection_volume` int(11) DEFAULT NULL,
  `Article_comment_volume` int(11) DEFAULT NULL,
  `Article_content` int(11) DEFAULT NULL,
  `Article_like_volume` int(11) DEFAULT NULL,
  `Article_reading_volume` int(11) DEFAULT NULL,
  `Article_title` int(11) DEFAULT NULL,
  `Creation_time` int(11) DEFAULT NULL,
  `Last_modification_and_save_author_ID` int(11) NOT NULL,
  `Last_modification_and_save_time` int(11) DEFAULT NULL,
  `Total_number_of_words_in_the_article` int(11) DEFAULT NULL,
  PRIMARY KEY (`Article_ID`),
  CONSTRAINT `article_ibfk_1` FOREIGN KEY (`Last_modification_and_save_author_ID`) REFERENCES `Author` (`Author_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Author_creates_article` (
  `Article_ID` int(11) NOT NULL,
  `Author_ID` int(11) NOT NULL,
  `Author_order` varchar(255) DEFAULT NULL,
  `Author_responsible_for_part` varchar(255) DEFAULT NULL,
  `Author_role` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Article_ID`, `Author_ID`),
  CONSTRAINT `author_creates_article_ibfk_1` FOREIGN KEY (`Article_ID`) REFERENCES `Article` (`Article_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `author_creates_article_ibfk_2` FOREIGN KEY (`Author_ID`) REFERENCES `Author` (`Author_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Author_modifies_article` (
  `Article_ID` int(11) NOT NULL,
  `Author_ID` int(11) NOT NULL,
  `Modify_save_time` int(11) NOT NULL,
  `Modify_notes` varchar(255) DEFAULT NULL,
  `Modify_start_time` datetime DEFAULT NULL,
  PRIMARY KEY (`Article_ID`, `Author_ID`, `Modify_save_time`),
  CONSTRAINT `author_modifies_article_ibfk_1` FOREIGN KEY (`Article_ID`) REFERENCES `Article` (`Article_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `author_modifies_article_ibfk_2` FOREIGN KEY (`Author_ID`) REFERENCES `Author` (`Author_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `author_modifies_article_ibfk_3` FOREIGN KEY (`Modify_save_time`) REFERENCES `Author_creates_article` (`Author_order`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
