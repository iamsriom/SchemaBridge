-- Schema for database: rschema_0079
CREATE TABLE `stock` (
  `stock_code` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `market` varchar(255) NOT NULL,
  `owner` varchar(255) NOT NULL,
  `total_investment_amount` varchar(255) NOT NULL,
  `holding_shares` varchar(255) NOT NULL,
  `last_transaction_time` varchar(255) NOT NULL,
  PRIMARY KEY (`stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `order_id` varchar(255) NOT NULL,
  `order_type` varchar(255) NOT NULL,
  `creation_time` varchar(255) NOT NULL,
  `transaction_date` varchar(255) NOT NULL,
  `transaction_amount` varchar(255) NOT NULL,
  `transaction_quantity` varchar(255) NOT NULL,
  `transaction_service_fee` varchar(255) NOT NULL,
  `transaction_status` varchar(255) NOT NULL,
  PRIMARY KEY (`order_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stock_order` (
  `order_id` varchar(255) NOT NULL,
  `stock_code` varchar(255) NOT NULL,
  PRIMARY KEY (`order_id`, `stock_code`),
  CONSTRAINT `stock_order_ibfk_1` FOREIGN KEY (`order_id`) REFERENCES `order` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `stock_order_ibfk_2` FOREIGN KEY (`stock_code`) REFERENCES `stock` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
