-- Schema for database: rschema_0080
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_name` varchar(255) DEFAULT NULL,
  `customer_phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_worker` (
  `repair_worker_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`repair_worker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `part` (
  `part_serial_number` int(11) NOT NULL AUTO_INCREMENT,
  `description` varchar(255) DEFAULT NULL,
  `inventory_quantity` int(11) DEFAULT NULL,
  `price` decimal(10,2) DEFAULT NULL,
  `supplier` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`part_serial_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_task` (
  `repair_task_id` int(11) NOT NULL AUTO_INCREMENT,
  `customer_id` int(11) DEFAULT NULL,
  `task_start_date` datetime DEFAULT NULL,
  `task_completion_date` datetime DEFAULT NULL,
  `task_status` varchar(50) DEFAULT NULL,
  `vehicle_model` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`repair_task_id`),
  CONSTRAINT `repair_task_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_worker_task` (
  `repair_worker_id` int(11) NOT NULL,
  `repair_task_id` int(11) NOT NULL,
  PRIMARY KEY (`repair_worker_id`, `repair_task_id`),
  CONSTRAINT `repair_worker_task_ibfk_1` FOREIGN KEY (`repair_worker_id`) REFERENCES `repair_worker` (`repair_worker_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_worker_task_ibfk_2` FOREIGN KEY (`repair_task_id`) REFERENCES `repair_task` (`repair_task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `part_usage` (
  `part_serial_number` int(11) NOT NULL,
  `repair_task_id` int(11) NOT NULL,
  PRIMARY KEY (`part_serial_number`, `repair_task_id`),
  CONSTRAINT `part_usage_ibfk_1` FOREIGN KEY (`part_serial_number`) REFERENCES `part` (`part_serial_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `part_usage_ibfk_2` FOREIGN KEY (`repair_task_id`) REFERENCES `repair_task` (`repair_task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
