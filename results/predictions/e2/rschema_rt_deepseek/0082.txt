-- Schema for database: rschema_0082
CREATE TABLE `political_party` (
  `party_id` int(11) NOT NULL,
  `party_name` varchar(255) NOT NULL,
  `party_description` text DEFAULT NULL,
  PRIMARY KEY (`party_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `candidate` (
  `candidate_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `belonging_party_id` int(11) NOT NULL,
  `industry_connection` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`candidate_id`),
  CONSTRAINT `candidate_ibfk_1` FOREIGN KEY (`belonging_party_id`) REFERENCES `political_party` (`party_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voter` (
  `voter_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  `voted_candidate_id` int(11) DEFAULT NULL,
  `vote_time` datetime DEFAULT NULL,
  PRIMARY KEY (`voter_id`),
  CONSTRAINT `voter_ibfk_1` FOREIGN KEY (`voted_candidate_id`) REFERENCES `candidate` (`candidate_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
