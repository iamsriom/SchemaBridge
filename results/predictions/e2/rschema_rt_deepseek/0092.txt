  `creation_time` datetime NOT NULL,

  CONSTRAINT `user_ibfk_2` FOREIGN KEY (`superior_user`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
