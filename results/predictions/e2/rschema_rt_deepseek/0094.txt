-- Schema for database: rschema_0094
CREATE TABLE `guacamole_user` (
  `user_id` int(11) NOT NULL,
  `password_date` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `guacamole_user_password_history` (
  `password_history_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `password_date` datetime DEFAULT NULL,
  `password_hash` varchar(255) DEFAULT NULL,
  `password_salt` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`password_history_id`),
  CONSTRAINT `guacamole_user_password_history_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `guacamole_user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
