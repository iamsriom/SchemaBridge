-- Schema for database: rschema_0099
CREATE TABLE `speaker` (
  `speaker_id` int(11) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `surname` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`speaker_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `conference` (
  `conference_id` int(11) NOT NULL,
  `conference_time` datetime DEFAULT NULL,
  PRIMARY KEY (`conference_id`),
  CONSTRAINT `conference_ibfk_1` FOREIGN KEY (`conference_id`) REFERENCES `speaker` (`speaker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `speaker_conference` (
  `conference_id` int(11) NOT NULL,
  `speaker_id` int(11) NOT NULL,
  PRIMARY KEY (`conference_id`, `speaker_id`),
  CONSTRAINT `speaker_conference_ibfk_1` FOREIGN KEY (`conference_id`) REFERENCES `conference` (`conference_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `speaker_conference_ibfk_2` FOREIGN KEY (`speaker_id`) REFERENCES `speaker` (`speaker_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
