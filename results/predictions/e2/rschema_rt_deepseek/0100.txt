-- Schema for database: rschema_0100
CREATE TABLE `customer` (
  `creation_time` datetime DEFAULT NULL,
  `customer_id` int(11) NOT NULL,
  `deletion_time` datetime DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `creation_time` datetime DEFAULT NULL,
  `deletion_time` datetime DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  `user_id` int(11) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `creation_time` datetime DEFAULT NULL,
  `creator_user_id` int(11) DEFAULT NULL,
  `currency_type` varchar(50) DEFAULT NULL,
  `customer_id` int(11) DEFAULT NULL,
  `deletion_time` datetime DEFAULT NULL,
  `delivery_date` datetime DEFAULT NULL,
  `invoice_number` varchar(255) DEFAULT NULL,
  `order_description` text DEFAULT NULL,
  `order_id` int(11) NOT NULL,
  `order_number` varchar(255) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`creator_user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `quotation` (
  `creation_time` datetime DEFAULT NULL,
  `creator_user_id` int(11) DEFAULT NULL,
  `customer_id` int(11) DEFAULT NULL,
  `deletion_time` datetime DEFAULT NULL,
  `description` text DEFAULT NULL,
  `quotation_id` int(11) NOT NULL,
  `quotation_number` varchar(255) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`quotation_id`),
  CONSTRAINT `quotation_ibfk_1` FOREIGN KEY (`creator_user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `quotation_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order_item` (
  `creation_time` datetime DEFAULT NULL,
  `deletion_time` datetime DEFAULT NULL,
  `description` text DEFAULT NULL,
  `order_id` int(11) DEFAULT NULL,
  `order_item_id` int(11) NOT NULL,
  `quantity` int(11) DEFAULT NULL,
  `unit` varchar(50) DEFAULT NULL,
  `unit_price` decimal(10,2) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`order_item_id`),
  CONSTRAINT `order_item_ibfk_1` FOREIGN KEY (`order_id`) REFERENCES `order` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
