-- Schema for database: rschema_0107
CREATE TABLE `Game` (
  `Game_ID` int(11) NOT NULL,
  `Game_result` varchar(255) DEFAULT NULL,
  `Grid_size` float DEFAULT NULL,
  `Number_of_rounds` int(11) DEFAULT NULL,
  `Username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Game_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Spaceship` (
  `Spaceship_ID` int(11) NOT NULL,
  `Game_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  `Strength` float DEFAULT NULL,
  `Type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Spaceship_ID`),
  CONSTRAINT `spaceship_ibfk_1` FOREIGN KEY (`Game_ID`) REFERENCES `Game` (`Game_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Fighter` (
  `Fighter_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Spaceship_ID` int(11) NOT NULL,
  `Strength` float DEFAULT NULL,
  PRIMARY KEY (`Fighter_ID`),
  CONSTRAINT `fighter_ibfk_1` FOREIGN KEY (`Spaceship_ID`) REFERENCES `Spaceship` (`Spaceship_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
