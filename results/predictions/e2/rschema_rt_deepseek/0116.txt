-- Schema for database: rschema_0116
CREATE TABLE `customers_table` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone` int(11) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airlines_table` (
  `airline_id` int(11) NOT NULL,
  `airline_name` varchar(255) DEFAULT NULL,
  `route` varchar(255) DEFAULT NULL,
  `time` datetime DEFAULT NULL,
  PRIMARY KEY (`airline_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_reviews_table` (
  `review_id` int(11) NOT NULL,
  `airline_id` int(11) DEFAULT NULL,
  `customer_id` int(11) DEFAULT NULL,
  `review_content` varchar(255) DEFAULT NULL,
  `review_grade` float DEFAULT NULL,
  PRIMARY KEY (`review_id`),
  CONSTRAINT `customer_reviews_ibfk_1` FOREIGN KEY (`airline_id`) REFERENCES `airlines_table` (`airline_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_reviews_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customers_table` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_bank_accounts_table` (
  `account_id` int(11) NOT NULL,
  `account_number` varchar(255) DEFAULT NULL,
  `balance` float DEFAULT NULL,
  `customer_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`account_id`),
  CONSTRAINT `customer_bank_accounts_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customers_table` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
