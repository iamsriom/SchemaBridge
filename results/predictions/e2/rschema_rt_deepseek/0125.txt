-- Schema for database: rschema_0125
CREATE TABLE `page` (
  `page_id` int(11) NOT NULL,
  `edit_status` varchar(255) DEFAULT NULL,
  `visibility_status` varchar(255) DEFAULT NULL,
  `active_version_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`page_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
  `comment_id` int(11) NOT NULL,
  `belonging_page` int(11) DEFAULT NULL,
  `reply_comment` int(11) DEFAULT NULL,
  `sending_user` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`comment_id`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`belonging_page`) REFERENCES `page` (`page_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`reply_comment`) REFERENCES `comment` (`comment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `version_record` (
  `page_number` int(11) DEFAULT NULL,
  `version_number` int(11) DEFAULT NULL,
  `image_url` varchar(255) DEFAULT NULL,
  `description_information` text DEFAULT NULL,
  CONSTRAINT `version_record_ibfk_1` FOREIGN KEY (`page_number`) REFERENCES `page` (`page_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
