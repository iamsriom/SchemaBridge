-- Schema for database: rschema_0134
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `user_role` varchar(50) DEFAULT NULL,
  `username` varchar(100) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `houses` (
  `property_id` int(11) NOT NULL,
  `landlord_id` int(11) NOT NULL,
  `property_description` text DEFAULT NULL,
  `property_facilities` text DEFAULT NULL,
  `property_location` varchar(255) DEFAULT NULL,
  `property_price` decimal(10,2) DEFAULT NULL,
  `property_title` varchar(255) NOT NULL,
  `property_type` varchar(100) DEFAULT NULL,
  `publisher` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`property_id`),
  CONSTRAINT `houses_ibfk_1` FOREIGN KEY (`landlord_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `demands` (
  `demand_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `demand_budget` decimal(10,2) DEFAULT NULL,
  `demand_description` text DEFAULT NULL,
  `demand_location` varchar(255) DEFAULT NULL,
  `demand_title` varchar(255) NOT NULL,
  `demand_type` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`demand_id`),
  CONSTRAINT `demands_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
  `order_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `lease_start_date` datetime DEFAULT NULL,
  `lease_end_date` datetime DEFAULT NULL,
  `order_price` decimal(10,2) DEFAULT NULL,
  `order_status` varchar(50) DEFAULT NULL,
  `payment_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `orders_ibfk_2` FOREIGN KEY (`property_id`) REFERENCES `houses` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
