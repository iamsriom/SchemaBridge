-- Schema for database: rschema_0144
CREATE TABLE `Course` (
  `Course_ID` int(11) NOT NULL,
  `Course_name` varchar(255) NOT NULL,
  `Class_time` datetime NOT NULL,
  `Classroom` varchar(255) NOT NULL,
  `Credits` float DEFAULT NULL,
  PRIMARY KEY (`Course_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Class` (
  `Class_ID` int(11) NOT NULL,
  `Class_name` varchar(255) NOT NULL,
  `Specialized_direction` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Class_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `School` (
  `School_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Contact_number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`School_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Student` (
  `Student_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Age` int(11) DEFAULT NULL,
  `Gender` varchar(255) DEFAULT NULL,
  `Major` varchar(255) DEFAULT NULL,
  `Class` int(11) NOT NULL,
  PRIMARY KEY (`Student_ID`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`Class`) REFERENCES `Class` (`Class_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Teacher` (
  `Teacher_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Age` int(11) DEFAULT NULL,
  `Gender` varchar(255) DEFAULT NULL,
  `Major` varchar(255) DEFAULT NULL,
  `Professional_title` varchar(255) DEFAULT NULL,
  `School_ID` int(11) NOT NULL,
  PRIMARY KEY (`Teacher_ID`),
  CONSTRAINT `teacher_ibfk_1` FOREIGN KEY (`School_ID`) REFERENCES `School` (`School_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Student_course` (
  `Course_ID` int(11) NOT NULL,
  `Student_ID` int(11) NOT NULL,
  `Score` int(11) DEFAULT NULL,
  PRIMARY KEY (`Course_ID`, `Student_ID`),
  CONSTRAINT `student_course_ibfk_1` FOREIGN KEY (`Course_ID`) REFERENCES `Course` (`Course_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_course_ibfk_2` FOREIGN KEY (`Student_ID`) REFERENCES `Student` (`Student_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Teacher_course` (
  `Course_ID` int(11) NOT NULL,
  `Teacher_ID` int(11) NOT NULL,
  PRIMARY KEY (`Course_ID`, `Teacher_ID`),
  CONSTRAINT `teacher_course_ibfk_1` FOREIGN KEY (`Course_ID`) REFERENCES `Course` (`Course_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `teacher_course_ibfk_2` FOREIGN KEY (`Teacher_ID`) REFERENCES `Teacher` (`Teacher_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
