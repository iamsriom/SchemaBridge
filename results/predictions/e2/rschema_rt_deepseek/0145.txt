-- Schema for database: rschema_0145
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `solid_state_disk` (
  `hard_disk_id` int(11) NOT NULL,
  `brand` varchar(255) DEFAULT NULL,
  `price` float DEFAULT NULL,
  `type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`hard_disk_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `telephone` varchar(255) DEFAULT NULL,
  `belonging_department` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`belonging_department`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_order` (
  `order_id` int(11) NOT NULL,
  `placing_customer` int(11) DEFAULT NULL,
  `order_time` datetime DEFAULT NULL,
  `order_amount` float DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `sales_order_ibfk_1` FOREIGN KEY (`placing_customer`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_record` (
  `purchase_order_id` int(11) NOT NULL,
  `purchase_staff` int(11) DEFAULT NULL,
  `purchase_time` datetime DEFAULT NULL,
  `spending_amount` float DEFAULT NULL,
  PRIMARY KEY (`purchase_order_id`),
  CONSTRAINT `purchase_record_ibfk_1` FOREIGN KEY (`purchase_staff`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouse` (
  `warehouse_id` int(11) NOT NULL,
  `warehouse_name` varchar(255) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `administrator` int(11) DEFAULT NULL,
  PRIMARY KEY (`warehouse_id`),
  CONSTRAINT `warehouse_ibfk_1` FOREIGN KEY (`administrator`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_order_details` (
  `sales_order_id` int(11) NOT NULL,
  `hard_disk_id` int(11) NOT NULL,
  `purchase_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`sales_order_id`, `hard_disk_id`),
  CONSTRAINT `sales_order_details_ibfk_1` FOREIGN KEY (`sales_order_id`) REFERENCES `sales_order` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_order_details_ibfk_2` FOREIGN KEY (`hard_disk_id`) REFERENCES `solid_state_disk` (`hard_disk_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_order_details` (
  `purchase_order_id` int(11) NOT NULL,
  `hard_disk_id` int(11) NOT NULL,
  `purchase_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`purchase_order_id`, `hard_disk_id`),
  CONSTRAINT `purchase_order_details_ibfk_1` FOREIGN KEY (`purchase_order_id`) REFERENCES `purchase_record` (`purchase_order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `purchase_order_details_ibfk_2` FOREIGN KEY (`hard_disk_id`) REFERENCES `solid_state_disk` (`hard_disk_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hard_disk_storage` (
  `hard_disk_id` int(11) NOT NULL,
  `warehouse_id` int(11) NOT NULL,
  `reserve` int(11) DEFAULT NULL,
  PRIMARY KEY (`hard_disk_id`, `warehouse_id`),
  CONSTRAINT `hard_disk_storage_ibfk_1` FOREIGN KEY (`hard_disk_id`) REFERENCES `solid_state_disk` (`hard_disk_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `hard_disk_storage_ibfk_2` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
