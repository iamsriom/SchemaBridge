-- Schema for database: rschema_0156
CREATE TABLE `User` (
  `user_id` int(11) NOT NULL,
  `user_name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `delivery_address` varchar(255) DEFAULT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `registration_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Product` (
  `product_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `product_description` text DEFAULT NULL,
  `product_sales_price` decimal(10,2) DEFAULT NULL,
  `inventory_quantity` int(11) DEFAULT NULL,
  `sales_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Product_added_to_cart` (
  `user_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `add_to_cart_time` datetime NOT NULL,
  `product_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `product_id`, `add_to_cart_time`),
  CONSTRAINT `Product_added_to_cart_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Product_added_to_cart_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `Product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Purchase_product` (
  `user_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `purchase_time` datetime NOT NULL,
  `payment_method` varchar(50) DEFAULT NULL,
  `purchase_quantity` int(11) DEFAULT NULL,
  `total_order_price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `product_id`, `purchase_time`),
  CONSTRAINT `Purchase_product_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Purchase_product_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `Product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
