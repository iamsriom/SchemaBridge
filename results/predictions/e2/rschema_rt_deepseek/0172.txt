-- Schema for database: rschema_0172
CREATE TABLE `Patient` (
  `Patient_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_number` varchar(15) NOT NULL,
  `emergency_contact_number` varchar(15) NOT NULL,
  PRIMARY KEY (`Patient_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Medical_Record` (
  `Record_ID` int(11) NOT NULL,
  `Patient_ID` int(11) NOT NULL,
  `blood_pressure` varchar(10) NOT NULL,
  `heart_rate` int(11) NOT NULL,
  `temperature` decimal(5,2) NOT NULL,
  `record_time` datetime NOT NULL,
  PRIMARY KEY (`Record_ID`),
  CONSTRAINT `medical_record_ibfk_1` FOREIGN KEY (`Patient_ID`) REFERENCES `Patient` (`Patient_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Alert_Record` (
  `Alert_ID` int(11) NOT NULL,
  `Abnormal_Record_ID` int(11) NOT NULL,
  `alert_content` text NOT NULL,
  `alert_time` datetime NOT NULL,
  PRIMARY KEY (`Alert_ID`),
  CONSTRAINT `alert_record_ibfk_1` FOREIGN KEY (`Abnormal_Record_ID`) REFERENCES `Medical_Record` (`Record_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
