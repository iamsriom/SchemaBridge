-- Schema for database: rschema_0176
CREATE TABLE `Flight` (
  `Flight_ID` int(11) NOT NULL,
  `Airline` varchar(255) DEFAULT NULL,
  `Departure_Time` datetime DEFAULT NULL,
  `Arrival_Time` datetime DEFAULT NULL,
  `Departure_City` varchar(255) DEFAULT NULL,
  `Destination_City` varchar(255) DEFAULT NULL,
  `Ticket_Price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`Flight_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Passenger` (
  `Passenger_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Date_of_Birth` date DEFAULT NULL,
  `Gender` char(1) DEFAULT NULL,
  `Phone_Number` varchar(15) DEFAULT NULL,
  `VIP_Discount` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`Passenger_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Discount_Information` (
  `Discount_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Flight_ID` int(11) NOT NULL,
  `Discount_Rate` decimal(5,2) DEFAULT NULL,
  `Duration` int(11) DEFAULT NULL,
  PRIMARY KEY (`Discount_ID`),
  CONSTRAINT `discount_information_ibfk_1` FOREIGN KEY (`Flight_ID`) REFERENCES `Flight` (`Flight_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Ticket_Purchase_Record` (
  `Ticket_Order_Number` int(11) NOT NULL,
  `Flight_ID` int(11) NOT NULL,
  `Passenger_ID` int(11) NOT NULL,
  `Purchase_Amount` decimal(10,2) DEFAULT NULL,
  `Travel_Date` date DEFAULT NULL,
  PRIMARY KEY (`Ticket_Order_Number`),
  CONSTRAINT `ticket_purchase_record_ibfk_1` FOREIGN KEY (`Flight_ID`) REFERENCES `Flight` (`Flight_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ticket_purchase_record_ibfk_2` FOREIGN KEY (`Passenger_ID`) REFERENCES `Passenger` (`Passenger_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
