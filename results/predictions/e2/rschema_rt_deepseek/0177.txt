-- Schema for database: rschema_0177
CREATE TABLE `abstract_file` (
  `creation_time` datetime NOT NULL,
  `file_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `update_time` datetime NOT NULL,
  PRIMARY KEY (`file_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `directory` (
  `directory_id` int(11) NOT NULL,
  `sub_file_directory_list` blob DEFAULT NULL,
  PRIMARY KEY (`directory_id`),
  CONSTRAINT `directory_ibfk_1` FOREIGN KEY (`directory_id`) REFERENCES `abstract_file` (`file_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `file` (
  `abstract_file_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `directory_id` int(11) NOT NULL,
  `file_id` int(11) NOT NULL,
  `file_size` float DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `update_time` datetime NOT NULL,
  PRIMARY KEY (`file_id`),
  CONSTRAINT `file_ibfk_1` FOREIGN KEY (`abstract_file_id`) REFERENCES `abstract_file` (`file_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `file_ibfk_2` FOREIGN KEY (`directory_id`) REFERENCES `directory` (`directory_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
