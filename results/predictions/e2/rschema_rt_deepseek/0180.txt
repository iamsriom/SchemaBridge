-- Schema for database: rschema_0180
CREATE TABLE `Pizza` (
  `Pizza_ID` int(11) NOT NULL,
  `Pizza_Name` varchar(255) DEFAULT NULL,
  `Price` float DEFAULT NULL,
  `Stock_Quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`Pizza_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Customer_Name` varchar(255) DEFAULT NULL,
  `Customer_Address` varchar(255) DEFAULT NULL,
  `Customer_Phone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Order` (
  `Order_ID` int(11) NOT NULL,
  `Customer_ID` int(11) NOT NULL,
  `Order_Date` datetime DEFAULT NULL,
  `Expected_Delivery_Date` datetime DEFAULT NULL,
  `Order_Status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Order_ID`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Ordered_Pizza` (
  `Order_ID` int(11) NOT NULL,
  `Pizza_ID` int(11) NOT NULL,
  `Pizza_Quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`Order_ID`, `Pizza_ID`),
  CONSTRAINT `ordered_pizza_ibfk_1` FOREIGN KEY (`Order_ID`) REFERENCES `Order` (`Order_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ordered_pizza_ibfk_2` FOREIGN KEY (`Pizza_ID`) REFERENCES `Pizza` (`Pizza_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
