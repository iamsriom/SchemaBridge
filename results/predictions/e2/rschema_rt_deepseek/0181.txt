-- Schema for database: rschema_0181
CREATE TABLE `Freight_Truck` (
  `License_Plate_Number` varchar(255) NOT NULL,
  `Load_Capacity` varchar(255) DEFAULT NULL,
  `Model` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`License_Plate_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Employee` (
  `Employee_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  `Role` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Employee_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Transportation_Task` (
  `Task_Number` int(11) NOT NULL,
  `Assigned_License_Plate_Number` varchar(255) DEFAULT NULL,
  `Assistant_ID` int(11) DEFAULT NULL,
  `Cargo_Name` varchar(255) DEFAULT NULL,
  `Destination` varchar(255) DEFAULT NULL,
  `Driver_ID` int(11) DEFAULT NULL,
  `End_Time` datetime DEFAULT NULL,
  `Origin` varchar(255) DEFAULT NULL,
  `Start_Time` datetime DEFAULT NULL,
  `Transportation_Time` float DEFAULT NULL,
  `Weight` float DEFAULT NULL,
  PRIMARY KEY (`Task_Number`),
  CONSTRAINT `Transportation_Task_ibfk_1` FOREIGN KEY (`Assigned_License_Plate_Number`) REFERENCES `Freight_Truck` (`License_Plate_Number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Transportation_Task_ibfk_2` FOREIGN KEY (`Assistant_ID`) REFERENCES `Employee` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Transportation_Task_ibfk_3` FOREIGN KEY (`Driver_ID`) REFERENCES `Employee` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
