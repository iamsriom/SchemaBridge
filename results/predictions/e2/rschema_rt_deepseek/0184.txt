-- Schema for database: rschema_0184
CREATE TABLE `member` (
  `member_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone` varchar(15) DEFAULT NULL,
  `registration_date` datetime NOT NULL,
  `points` int(11) DEFAULT NULL,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `room` (
  `room_number` int(11) NOT NULL,
  `room_type` varchar(100) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`room_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `booking` (
  `booking_date` datetime NOT NULL,
  `member_id` int(11) NOT NULL,
  `room_number` int(11) NOT NULL,
  `check-in_date` datetime NOT NULL,
  `check-out_date` datetime NOT NULL,
  `booking_status` varchar(50) NOT NULL,
  `points_change` int(11) DEFAULT NULL,
  PRIMARY KEY (`booking_date`, `member_id`, `room_number`),
  CONSTRAINT `booking_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `booking_ibfk_2` FOREIGN KEY (`room_number`) REFERENCES `room` (`room_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
