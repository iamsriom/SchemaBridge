-- Schema for database: rschema_0185
CREATE TABLE `Proposal` (
  `Proposal_ID` int(11) NOT NULL,
  `Proposal_Content` text DEFAULT NULL,
  `Proposal_Time` datetime DEFAULT NULL,
  `Status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`Proposal_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Member` (
  `Member_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Date_of_Birth` date DEFAULT NULL,
  `Gender` varchar(10) DEFAULT NULL,
  `Party_Affiliation` varchar(100) DEFAULT NULL,
  `Position` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`Member_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Proposal_Submission` (
  `Member_ID` int(11) NOT NULL,
  `Proposal_ID` int(11) NOT NULL,
  PRIMARY KEY (`Member_ID`, `Proposal_ID`),
  CONSTRAINT `Proposal_Submission_ibfk_1` FOREIGN KEY (`Member_ID`) REFERENCES `Member` (`Member_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Proposal_Submission_ibfk_2` FOREIGN KEY (`Proposal_ID`) REFERENCES `Proposal` (`Proposal_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Proposal_Handling` (
  `Member_ID` int(11) NOT NULL,
  `Proposal_ID` int(11) NOT NULL,
  `Handling_Opinion` text DEFAULT NULL,
  `Handling_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Member_ID`, `Proposal_ID`),
  CONSTRAINT `Proposal_Handling_ibfk_1` FOREIGN KEY (`Member_ID`) REFERENCES `Member` (`Member_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Proposal_Handling_ibfk_2` FOREIGN KEY (`Proposal_ID`) REFERENCES `Proposal` (`Proposal_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
