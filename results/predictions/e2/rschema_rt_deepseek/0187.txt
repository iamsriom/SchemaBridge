-- Schema for database: rschema_0187
CREATE TABLE `farm` (
  `farm_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `owner_name` varchar(255) NOT NULL,
  `owner_type` varchar(255) NOT NULL,
  PRIMARY KEY (`farm_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crop` (
  `crop_id` int(11) NOT NULL,
  `crop_name` varchar(255) NOT NULL,
  `growth_cycle` varchar(255) NOT NULL,
  `suitable_climate_conditions` varchar(255) NOT NULL,
  `suitable_soil_type` varchar(255) NOT NULL,
  PRIMARY KEY (`crop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plot` (
  `plot_id` int(11) NOT NULL,
  `area` float NOT NULL,
  `current_status` varchar(255) NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `farm_id` int(11) NOT NULL,
  PRIMARY KEY (`plot_id`),
  CONSTRAINT `plot_ibfk_1` FOREIGN KEY (`farm_id`) REFERENCES `farm` (`farm_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `planting_record` (
  `plot_id` int(11) NOT NULL,
  `crop_id` int(11) NOT NULL,
  `planting_time` datetime NOT NULL,
  `expected_harvest_time` datetime DEFAULT NULL,
  `operator` varchar(255) NOT NULL,
  `planting_quantity` float NOT NULL,
  PRIMARY KEY (`plot_id`, `crop_id`, `planting_time`),
  CONSTRAINT `planting_record_ibfk_1` FOREIGN KEY (`crop_id`) REFERENCES `crop` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `planting_record_ibfk_2` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `harvest_record` (
  `plot_id` int(11) NOT NULL,
  `crop_id` int(11) NOT NULL,
  `harvest_time` datetime NOT NULL,
  `operator` varchar(255) NOT NULL,
  `harvest_quantity` float NOT NULL,
  PRIMARY KEY (`plot_id`, `crop_id`, `harvest_time`),
  CONSTRAINT `harvest_record_ibfk_1` FOREIGN KEY (`crop_id`) REFERENCES `crop` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `harvest_record_ibfk_2` FOREIGN KEY (`plot_id`) REFERENCES `plot` (`plot_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
