-- Schema for database: rschema_0190
CREATE TABLE `Workstation` (
  `Workstation_ID` int(11) NOT NULL,
  `Workstation_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Workstation_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Product` (
  `Product_ID` int(11) NOT NULL,
  `Product_Model` varchar(255) DEFAULT NULL,
  `Product_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Product_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Repair_Technician` (
  `Repair_Technician_ID` int(11) NOT NULL,
  `Repair_Technician_Name` varchar(255) DEFAULT NULL,
  `Repair_Technician_Phone` varchar(255) DEFAULT NULL,
  `Workstation_ID` int(11) NOT NULL,
  PRIMARY KEY (`Repair_Technician_ID`),
  CONSTRAINT `Repair_Technician_ibfk_1` FOREIGN KEY (`Workstation_ID`) REFERENCES `Workstation` (`Workstation_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Repair_Record` (
  `Customer_Feedback` varchar(255) DEFAULT NULL,
  `Customer_Name` varchar(255) DEFAULT NULL,
  `Customer_Phone` varchar(255) DEFAULT NULL,
  `Fault_Description` varchar(255) DEFAULT NULL,
  `Product_ID` int(11) NOT NULL,
  `Repair_Duration` float DEFAULT NULL,
  `Repair_Technician_ID` int(11) NOT NULL,
  `Repair_Time` float DEFAULT NULL,
  `Repair_Type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Repair_Technician_ID`, `Product_ID`, `Repair_Time`),
  CONSTRAINT `Repair_Record_ibfk_1` FOREIGN KEY (`Product_ID`) REFERENCES `Product` (`Product_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Repair_Record_ibfk_2` FOREIGN KEY (`Repair_Technician_ID`) REFERENCES `Repair_Technician` (`Repair_Technician_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
