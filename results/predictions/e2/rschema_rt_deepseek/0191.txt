-- Schema for database: rschema_0191
CREATE TABLE `sales_team` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `salesperson` (
  `salesperson_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `sex` varchar(255) DEFAULT NULL,
  `telephone` varchar(255) DEFAULT NULL,
  `belongs_to_team_id` int(11) NOT NULL,
  PRIMARY KEY (`salesperson_id`),
  CONSTRAINT `salesperson_ibfk_1` FOREIGN KEY (`belongs_to_team_id`) REFERENCES `sales_team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_task` (
  `task_id` int(11) NOT NULL,
  `customer_name` varchar(255) DEFAULT NULL,
  `customer_telephone` varchar(255) DEFAULT NULL,
  `product_model` varchar(255) DEFAULT NULL,
  `product_name` varchar(255) DEFAULT NULL,
  `product_price` float DEFAULT NULL,
  `sales_cost` float DEFAULT NULL,
  `sales_date` datetime DEFAULT NULL,
  `sales_quantity` int(11) DEFAULT NULL,
  `belongs_to_team_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `sales_task_ibfk_1` FOREIGN KEY (`belongs_to_team_id`) REFERENCES `sales_team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluation` (
  `evaluation_id` int(11) NOT NULL,
  `customer_satisfaction` float DEFAULT NULL,
  `salesperson_id` int(11) NOT NULL,
  `task_completion_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`evaluation_id`),
  CONSTRAINT `performance_evaluation_ibfk_1` FOREIGN KEY (`salesperson_id`) REFERENCES `salesperson` (`salesperson_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_record` (
  `record_id` int(11) NOT NULL,
  `customer_evaluation` varchar(255) DEFAULT NULL,
  `task_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `task_record_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `sales_task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
