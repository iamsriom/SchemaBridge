-- Schema for database: rschema_0192
CREATE TABLE `department` (
  `Department_Head` int(11) NOT NULL,
  `Department_ID` int(11) NOT NULL,
  `Department_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Department_ID`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`Department_Head`) REFERENCES `employee` (`Employee_Number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `Date_of_Birth` datetime DEFAULT NULL,
  `Department_ID` int(11) NOT NULL,
  `Employee_Number` int(11) NOT NULL,
  `Gender` boolean DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Employee_Number`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Department_ID`) REFERENCES `department` (`Department_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `Actual_Completion_Date` datetime DEFAULT NULL,
  `Actual_Cost` float DEFAULT NULL,
  `Estimated_Completion_Date` datetime DEFAULT NULL,
  `Project_Budget` float DEFAULT NULL,
  `Project_Location` varchar(255) DEFAULT NULL,
  `Project_Manager_ID` int(11) NOT NULL,
  `Project_Name` varchar(255) DEFAULT NULL,
  `Project_Number` int(11) NOT NULL,
  `Project_Progress` varchar(255) DEFAULT NULL,
  `Start_Date` datetime DEFAULT NULL,
  PRIMARY KEY (`Project_Number`),
  CONSTRAINT `project_ibfk_1` FOREIGN KEY (`Project_Manager_ID`) REFERENCES `employee` (`Employee_Number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_participation_record` (
  `Employee_Number` int(11) NOT NULL,
  `Participation_Time` datetime NOT NULL,
  `Project_Number` int(11) NOT NULL,
  `Role` varchar(255) DEFAULT NULL,
  `Work_Hours` float DEFAULT NULL,
  PRIMARY KEY (`Employee_Number`, `Project_Number`, `Participation_Time`),
  CONSTRAINT `project_participation_record_ibfk_1` FOREIGN KEY (`Employee_Number`) REFERENCES `employee` (`Employee_Number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_participation_record_ibfk_2` FOREIGN KEY (`Project_Number`) REFERENCES `project` (`Project_Number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
