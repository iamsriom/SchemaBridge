-- Schema for database: rschema_0196
CREATE TABLE `applicant` (
  `applicant_id` int(11) NOT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `identification_number` int(11) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`applicant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approver` (
  `approver_id` int(11) NOT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`approver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_application` (
  `application_number` int(11) NOT NULL,
  `applicant_id` int(11) NOT NULL,
  `approval_status` varchar(255) DEFAULT NULL,
  `loan_amount` float DEFAULT NULL,
  `loan_term` int(11) DEFAULT NULL,
  PRIMARY KEY (`application_number`),
  CONSTRAINT `loan_application_ibfk_1` FOREIGN KEY (`applicant_id`) REFERENCES `applicant` (`applicant_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval` (
  `approver_id` int(11) NOT NULL,
  `application_number` int(11) NOT NULL,
  `approval_date` datetime DEFAULT NULL,
  `approval_result` boolean DEFAULT NULL,
  PRIMARY KEY (`approver_id`, `application_number`, `approval_date`),
  CONSTRAINT `loan_approval_ibfk_1` FOREIGN KEY (`application_number`) REFERENCES `loan_application` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_ibfk_2` FOREIGN KEY (`approver_id`) REFERENCES `approver` (`approver_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
