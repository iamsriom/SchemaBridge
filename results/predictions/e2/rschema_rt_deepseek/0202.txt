CREATE TABLE `procured_product` (
  `procured_product_id` int(11) NOT NULL,
  `supplier_id` int(11) NOT NULL,
  `procurement_time` datetime NOT NULL,
  `procurement_quantity` int(11) NOT NULL,
  `total_procurement_price` decimal(10,2) NOT NULL,
  `payment_method` varchar(255) DEFAULT NULL,
  `payment_status` varchar(255) DEFAULT NULL,
  `delivery_address` varchar(255) DEFAULT NULL,
  `recipient_name` varchar(255) DEFAULT NULL,
  `recipient_phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`procured_product_id`, `procurement_time`, `supplier_id`),
  CONSTRAINT `procured_product_ibfk_supplier_id` FOREIGN KEY (`supplier_id`) REFERENCES `schema_3`.`supplier_id` ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `procured_product_ibfk_procured_product_id` FOREIGN KEY (`procured_product_id`) REFERENCES `schema_4`.`product_id` ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `supplier` (
  `supplier_id` int(11) NOT NULL,
  `supplier_name` varchar(255) NOT NULL,
  `supplier_address` varchar(255) DEFAULT NULL,
  `historical_procurement_quantity` int(11) DEFAULT NULL,
  `historical_procurement_amount` decimal(10,2) DEFAULT NULL,
  `credit_rating` varchar(50) DEFAULT NULL,
  `contact_person` varchar(255) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `fax` varchar(20) DEFAULT NULL,
  `company_website` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`supplier_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `product` (
  `product_id` int(11) NOT NULL,
  `product_name` varchar(255) NOT NULL,
  `product_model` varchar(255) DEFAULT NULL,
  `product_price` decimal(10,2) NOT NULL,
  `product_category` varchar(255) DEFAULT NULL,
  `shelf_life` int(11) DEFAULT NULL,
  `stock_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
