-- Schema for database: rschema_0204
CREATE TABLE `participating_teams` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) DEFAULT NULL,
  `team_size` int(11) DEFAULT NULL,
  `team_members` varchar(255) DEFAULT NULL,
  `team_leader_name` varchar(255) DEFAULT NULL,
  `team_leader_phone` varchar(20) DEFAULT NULL,
  `head_coach_name` varchar(255) DEFAULT NULL,
  `head_coach_phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `referees` (
  `referee_id` int(11) NOT NULL,
  `referee_name` varchar(255) DEFAULT NULL,
  `referee_gender` varchar(10) DEFAULT NULL,
  `referee_email` varchar(255) DEFAULT NULL,
  `referee_phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`referee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `events` (
  `event_id` int(11) NOT NULL,
  `event_name` varchar(255) DEFAULT NULL,
  `event_type` varchar(100) DEFAULT NULL,
  `start_date` datetime DEFAULT NULL,
  `end_date` datetime DEFAULT NULL,
  `referee_id` int(11) DEFAULT NULL,
  `winning_team_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`event_id`),
  CONSTRAINT `events_ibfk_1` FOREIGN KEY (`referee_id`) REFERENCES `referees` (`referee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `match_results` (
  `event_id` int(11) NOT NULL,
  `team_id` int(11) NOT NULL,
  `team_ranking` int(11) DEFAULT NULL,
  `referee_evaluation_result` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`team_id`, `event_id`),
  CONSTRAINT `match_results_ibfk_1` FOREIGN KEY (`event_id`) REFERENCES `events` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `match_results_ibfk_2` FOREIGN KEY (`team_id`) REFERENCES `participating_teams` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
