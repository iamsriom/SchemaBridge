-- Schema for database: rschema_0207
CREATE TABLE `facility` (
  `creation_date` datetime DEFAULT NULL,
  `facility_id` int(11) NOT NULL,
  `facility_location` varchar(255) DEFAULT NULL,
  `facility_name` varchar(255) DEFAULT NULL,
  `facility_responsible_person` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`facility_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `user_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `facility_maintenance` (
  `facility_id` int(11) NOT NULL,
  `maintenance_content` varchar(255) DEFAULT NULL,
  `maintenance_cost` float DEFAULT NULL,
  `maintenance_time` datetime NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`facility_id`, `user_id`, `maintenance_time`),
  CONSTRAINT `facility_maintenance_ibfk_1` FOREIGN KEY (`facility_id`) REFERENCES `facility` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `facility_maintenance_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `facility_usage` (
  `facility_id` int(11) NOT NULL,
  `usage_frequency` float DEFAULT NULL,
  `usage_time` datetime NOT NULL,
  `user_feedback` varchar(255) DEFAULT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`facility_id`, `user_id`, `usage_time`),
  CONSTRAINT `facility_usage_ibfk_1` FOREIGN KEY (`facility_id`) REFERENCES `facility` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `facility_usage_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
