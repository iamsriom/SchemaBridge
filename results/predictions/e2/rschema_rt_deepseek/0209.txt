-- Schema for database: rschema_0209
CREATE TABLE `laboratory` (
  `laboratory_number` int(11) NOT NULL,
  `laboratory_name` varchar(255) NOT NULL,
  `laboratory_location` varchar(255) NOT NULL,
  PRIMARY KEY (`laboratory_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `equipment_number` int(11) NOT NULL,
  `equipment_name` varchar(255) NOT NULL,
  `equipment_model` varchar(255) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `equipment_status` varchar(50) NOT NULL,
  `laboratory_number` int(11) NOT NULL,
  PRIMARY KEY (`equipment_number`),
  CONSTRAINT `equipment_ibfk_1` FOREIGN KEY (`laboratory_number`) REFERENCES `laboratory` (`laboratory_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `maintenance_number` int(11) NOT NULL,
  `equipment_number` int(11) NOT NULL,
  `maintenance_date` datetime NOT NULL,
  `maintenance_personnel` varchar(255) NOT NULL,
  `maintenance_content` text NOT NULL,
  PRIMARY KEY (`maintenance_number`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`equipment_number`) REFERENCES `equipment` (`equipment_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
