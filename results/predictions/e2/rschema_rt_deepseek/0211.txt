-- Schema for database: rschema_0211
CREATE TABLE `project` (
  `Project_ID` int(11) NOT NULL,
  `Client_Name` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Start_Date` datetime DEFAULT NULL,
  `End_Date` datetime DEFAULT NULL,
  PRIMARY KEY (`Project_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultant` (
  `Consultant_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Date_of_Birth` datetime DEFAULT NULL,
  `Gender` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  `Professional_Field` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Consultant_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_record` (
  `Service_ID` int(11) NOT NULL,
  `Consultant_ID` int(11) NOT NULL,
  `Project_ID` int(11) NOT NULL,
  `Service_Content` varchar(255) DEFAULT NULL,
  `Service_Date` datetime DEFAULT NULL,
  PRIMARY KEY (`Service_ID`),
  CONSTRAINT `service_record_ibfk_1` FOREIGN KEY (`Consultant_ID`) REFERENCES `consultant` (`Consultant_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_record_ibfk_2` FOREIGN KEY (`Project_ID`) REFERENCES `project` (`Project_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
