-- Schema for database: rschema_0215
CREATE TABLE `mine` (
  `mine_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`mine_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_administrator` (
  `administrator_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone` varchar(15) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `mine_id` int(11) NOT NULL,
  PRIMARY KEY (`administrator_id`),
  CONSTRAINT `material_administrator_ibfk_1` FOREIGN KEY (`mine_id`) REFERENCES `mine` (`mine_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_record` (
  `record_id` int(11) NOT NULL,
  `purchase_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `supplier` varchar(255) DEFAULT NULL,
  `purchase_cost` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`record_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_distribution_record` (
  `record_id` int(11) NOT NULL,
  `distribution_date` datetime NOT NULL,
  `material_name` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `distributor` int(11) NOT NULL,
  `recipient` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `material_distribution_record_ibfk_1` FOREIGN KEY (`distributor`) REFERENCES `material_administrator` (`administrator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
