-- Schema for database: rschema_0218
CREATE TABLE `Manufacturer` (
  `Manufacturer_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Person_in_charge` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Manufacturer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Model` (
  `Model_ID` int(11) NOT NULL,
  `Model_name` varchar(255) DEFAULT NULL,
  `Manufacturer_ID` int(11) NOT NULL,
  PRIMARY KEY (`Model_ID`),
  CONSTRAINT `Model_ibfk_1` FOREIGN KEY (`Manufacturer_ID`) REFERENCES `Manufacturer` (`Manufacturer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Printer` (
  `Printer_ID` int(11) NOT NULL,
  `Hostname` varchar(255) DEFAULT NULL,
  `Additional_driver_information` varchar(255) DEFAULT NULL,
  `Printer_location` varchar(255) DEFAULT NULL,
  `Model_ID` int(11) NOT NULL,
  PRIMARY KEY (`Printer_ID`),
  CONSTRAINT `Printer_ibfk_1` FOREIGN KEY (`Model_ID`) REFERENCES `Model` (`Model_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
