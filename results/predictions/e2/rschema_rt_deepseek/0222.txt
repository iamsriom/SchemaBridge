-- Schema for database: rschema_0222
CREATE TABLE `recipe_type_table` (
  `type_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ingredient_table` (
  `ingredient_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `image` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`ingredient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipe_table` (
  `recipe_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `image` varchar(255) DEFAULT NULL,
  `steps` text DEFAULT NULL,
  `type_id` int(11) NOT NULL,
  PRIMARY KEY (`recipe_id`),
  CONSTRAINT `recipe_table_ibfk_1` FOREIGN KEY (`type_id`) REFERENCES `recipe_type_table` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `recipe_and_ingredient_relationship_table` (
  `recipe_id` int(11) NOT NULL,
  `ingredient_id` int(11) NOT NULL,
  `quantity` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`recipe_id`, `ingredient_id`),
  CONSTRAINT `recipe_and_ingredient_relationship_table_ibfk_1` FOREIGN KEY (`recipe_id`) REFERENCES `recipe_table` (`recipe_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `recipe_and_ingredient_relationship_table_ibfk_2` FOREIGN KEY (`ingredient_id`) REFERENCES `ingredient_table` (`ingredient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
