-- Schema for database: rschema_0226
CREATE TABLE `Branch` (
  `Branch_ID` int(11) NOT NULL,
  `Location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Branch_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `First_Name` varchar(255) DEFAULT NULL,
  `Last_Name` varchar(255) DEFAULT NULL,
  `Phone` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Category` (
  `Category_ID` int(11) NOT NULL,
  `Category_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Category_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book` (
  `Book_ID` int(11) NOT NULL,
  `Book_Name` varchar(255) DEFAULT NULL,
  `Category_ID` int(11) NOT NULL,
  `Price` decimal(10,2) DEFAULT NULL,
  `Stock_Quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`Book_ID`),
  CONSTRAINT `Book_ibfk_1` FOREIGN KEY (`Category_ID`) REFERENCES `Category` (`Category_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Order` (
  `Order_ID` int(11) NOT NULL,
  `Book_ID` int(11) NOT NULL,
  `Branch_ID` int(11) NOT NULL,
  `Customer_ID` int(11) NOT NULL,
  `Order_Date` datetime DEFAULT NULL,
  `Purchase_Quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`Order_ID`),
  CONSTRAINT `Order_ibfk_1` FOREIGN KEY (`Book_ID`) REFERENCES `Book` (`Book_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Order_ibfk_2` FOREIGN KEY (`Branch_ID`) REFERENCES `Branch` (`Branch_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Order_ibfk_3` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
