-- Schema for database: rschema_0230
CREATE TABLE `vehicle_model` (
  `vehicle_model_id` int(11) NOT NULL,
  `model_name` varchar(255) NOT NULL,
  PRIMARY KEY (`vehicle_model_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mechanic` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_details` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `contact_address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `record_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `mechanic_id` int(11) NOT NULL,
  `vehicle_model_id` int(11) NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_2` FOREIGN KEY (`mechanic_id`) REFERENCES `mechanic` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_3` FOREIGN KEY (`vehicle_model_id`) REFERENCES `vehicle_model` (`vehicle_model_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `part` (
  `serial_number` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `stock_quantity` int(11) NOT NULL,
  `supplier` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`serial_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `order_id` int(11) NOT NULL,
  `dispatch_date` datetime DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`order_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order_detail` (
  `part_serial_number` int(11) NOT NULL,
  `order_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`part_serial_number`, `order_id`),
  CONSTRAINT `order_detail_ibfk_1` FOREIGN KEY (`part_serial_number`) REFERENCES `part` (`serial_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_detail_ibfk_2` FOREIGN KEY (`order_id`) REFERENCES `order` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
