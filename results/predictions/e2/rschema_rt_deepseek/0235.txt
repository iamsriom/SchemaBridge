-- Schema for database: rschema_0235
CREATE TABLE `User` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `platform` varchar(50) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  `update_time` datetime DEFAULT NULL,
  `username` varchar(100) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Product` (
  `product_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `price` decimal(10,2) NOT NULL,
  `currency` varchar(10) NOT NULL,
  `inventory_unit` int(11) NOT NULL,
  `platform` varchar(50) DEFAULT NULL,
  `creation_time` datetime NOT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Receipt` (
  `receipt_id` int(11) NOT NULL AUTO_INCREMENT,
  `hash_value` varchar(255) NOT NULL,
  `receipt_data` text DEFAULT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`receipt_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Subscribed_Products` (
  `user_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `subscription_time` datetime NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime DEFAULT NULL,
  `trial_period` int(11) DEFAULT NULL,
  `promotional_period` int(11) DEFAULT NULL,
  `renewal_status` varchar(20) DEFAULT NULL,
  `subscription_status` varchar(20) DEFAULT NULL,
  `validity_type` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `product_id`, `subscription_time`),
  CONSTRAINT `User_Subscribed_Products_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Subscribed_Products_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `Product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Refunded_Products` (
  `user_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `refund_time` datetime NOT NULL,
  `refund_amount` decimal(10,2) NOT NULL,
  `refund_date` datetime NOT NULL,
  `refund_status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `product_id`, `refund_time`),
  CONSTRAINT `User_Refunded_Products_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Refunded_Products_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `Product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Purchased_Products` (
  `user_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `receipt_id` int(11) NOT NULL,
  `order_number` varchar(50) NOT NULL,
  `purchase_time` datetime NOT NULL,
  `purchase_price` decimal(10,2) NOT NULL,
  `purchase_platform` varchar(50) DEFAULT NULL,
  `hash_value` varchar(255) DEFAULT NULL,
  `refund_status` varchar(20) DEFAULT NULL,
  `subscription_status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `product_id`, `order_number`),
  CONSTRAINT `User_Purchased_Products_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Purchased_Products_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `Product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `User_Purchased_Products_ibfk_3` FOREIGN KEY (`receipt_id`) REFERENCES `Receipt` (`receipt_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
