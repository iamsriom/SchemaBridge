-- Schema for database: rschema_0241
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `class` varchar(255) DEFAULT NULL,
  `date_of_birth` datetime DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `gender` boolean DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL,
  `course_name` varchar(255) DEFAULT NULL,
  `credits` float DEFAULT NULL,
  `teacher` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_course_selection` (
  `course_id` int(11) NOT NULL,
  `student_id` int(11) NOT NULL,
  `credits` float DEFAULT NULL,
  `score` float DEFAULT NULL,
  PRIMARY KEY (`course_id`, `student_id`),
  CONSTRAINT `student_course_selection_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_course_selection_ibfk_2` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
