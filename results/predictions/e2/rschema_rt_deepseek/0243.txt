-- Schema for database: rschema_0243
CREATE TABLE `Reader` (
  `Reader_Email` varchar(255) DEFAULT NULL,
  `Reader_ID` int(11) NOT NULL,
  `Reader_Name` varchar(255) DEFAULT NULL,
  `Reader_Phone` varchar(255) DEFAULT NULL,
  `Registration_Date` datetime DEFAULT NULL,
  `Update_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Reader_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book` (
  `Book_Author` varchar(255) DEFAULT NULL,
  `Book_ID` int(11) NOT NULL,
  `Book_ISBN` varchar(255) DEFAULT NULL,
  `Book_Publication_Date` datetime DEFAULT NULL,
  `Book_Stock` int(11) DEFAULT NULL,
  `Book_Title` varchar(255) DEFAULT NULL,
  `Book_Type` varchar(255) DEFAULT NULL,
  `Update_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Book_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Borrowing` (
  `Book_ID` int(11) NOT NULL,
  `Borrow_Date` datetime DEFAULT NULL,
  `Overdue_Fine` float DEFAULT NULL,
  `Reader_ID` int(11) NOT NULL,
  `Renewal_Count` int(11) DEFAULT NULL,
  `Return_Date` datetime DEFAULT NULL,
  PRIMARY KEY (`Reader_ID`, `Book_ID`),
  CONSTRAINT `borrowing_ibfk_1` FOREIGN KEY (`Reader_ID`) REFERENCES `Reader` (`Reader_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `borrowing_ibfk_2` FOREIGN KEY (`Book_ID`) REFERENCES `Book` (`Book_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
