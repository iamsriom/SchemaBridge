-- Schema for database: rschema_0245
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Username` varchar(255) DEFAULT NULL,
  `Email` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Creation_time` datetime NOT NULL,
  `Update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Novel` (
  `Novel_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Novel_title` varchar(255) NOT NULL,
  `Novel_description` text DEFAULT NULL,
  `Author` varchar(255) NOT NULL,
  `Publication_date` date NOT NULL,
  `Creation_time` datetime NOT NULL,
  `Update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`Novel_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Bookshelf` (
  `Bookshelf_ID` int(11) NOT NULL AUTO_INCREMENT,
  `User_ID` int(11) NOT NULL,
  `Novel_ID` int(11) NOT NULL,
  PRIMARY KEY (`Bookshelf_ID`),
  CONSTRAINT `bookshelf_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bookshelf_ibfk_2` FOREIGN KEY (`Novel_ID`) REFERENCES `Novel` (`Novel_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book_review` (
  `Book_review_ID` int(11) NOT NULL AUTO_INCREMENT,
  `User_ID` int(11) NOT NULL,
  `Novel_ID` int(11) NOT NULL,
  `Rating` int(11) NOT NULL,
  `Review_content` text DEFAULT NULL,
  `Creation_time` datetime NOT NULL,
  PRIMARY KEY (`Book_review_ID`),
  CONSTRAINT `book_review_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `book_review_ibfk_2` FOREIGN KEY (`Novel_ID`) REFERENCES `Novel` (`Novel_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Novel_added_to_Bookshelf` (
  `Novel_ID` int(11) NOT NULL,
  `Bookshelf_ID` int(11) NOT NULL,
  PRIMARY KEY (`Novel_ID`, `Bookshelf_ID`),
  CONSTRAINT `novel_added_to_bookshelf_ibfk_1` FOREIGN KEY (`Novel_ID`) REFERENCES `Novel` (`Novel_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `novel_added_to_bookshelf_ibfk_2` FOREIGN KEY (`Bookshelf_ID`) REFERENCES `Bookshelf` (`Bookshelf_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
