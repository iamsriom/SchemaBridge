-- Schema for database: rschema_0246
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) DEFAULT NULL,
  `login_status` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crew` (
  `crew_id` int(11) NOT NULL AUTO_INCREMENT,
  `crew_information` varchar(255) DEFAULT NULL,
  `crew_members` varchar(255) DEFAULT NULL,
  `crew_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`crew_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `film_information` (
  `information_id` int(11) NOT NULL AUTO_INCREMENT,
  `information_title` varchar(255) DEFAULT NULL,
  `information_content` varchar(255) DEFAULT NULL,
  `publish_time` datetime DEFAULT NULL,
  PRIMARY KEY (`information_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
  `film_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `comment_time` datetime NOT NULL,
  `comment_content` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`film_id`, `user_id`, `comment_time`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`film_id`) REFERENCES `film_information` (`information_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `film_ranking` (
  `ranking_id` int(11) NOT NULL AUTO_INCREMENT,
  `film_id` int(11) NOT NULL,
  `ranking_position` varchar(255) DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`ranking_id`),
  CONSTRAINT `film_ranking_ibfk_1` FOREIGN KEY (`film_id`) REFERENCES `film_information` (`information_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
