-- Schema for database: rschema_0251
CREATE TABLE `Groomer` (
  `Groomer_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Rating` float DEFAULT NULL,
  `Specialization` varchar(255) DEFAULT NULL,
  `Years_of_experience` int(11) DEFAULT NULL,
  PRIMARY KEY (`Groomer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Pet` (
  `Pet_ID` int(11) NOT NULL,
  `Age` int(11) DEFAULT NULL,
  `Breed` varchar(255) DEFAULT NULL,
  `Fur_characteristics` varchar(255) DEFAULT NULL,
  `Weight` float DEFAULT NULL,
  PRIMARY KEY (`Pet_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Grooming_service` (
  `Groomer_ID` int(11) NOT NULL,
  `Pet_ID` int(11) NOT NULL,
  `Service_time` float NOT NULL,
  `Customer_rating` float DEFAULT NULL,
  `Price` float DEFAULT NULL,
  `Service_items` varchar(255) DEFAULT NULL,
  `Service_market` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Groomer_ID`, `Pet_ID`, `Service_time`),
  CONSTRAINT `grooming_service_ibfk_1` FOREIGN KEY (`Groomer_ID`) REFERENCES `Groomer` (`Groomer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `grooming_service_ibfk_2` FOREIGN KEY (`Pet_ID`) REFERENCES `Pet` (`Pet_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
