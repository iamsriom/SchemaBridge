-- Schema for database: rschema_0254
CREATE TABLE `user` (
  `user_name` varchar(255) NOT NULL,
  `affiliated_earthquake_monitoring_center_station` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `seismometer_calibration_information` (
  `calibration_time` datetime NOT NULL,
  `amplitude` float DEFAULT NULL,
  `calibration_coil_sensitivity` float DEFAULT NULL,
  `equivalent_pendulum_length` float DEFAULT NULL,
  `mechanical_cycle` int(11) DEFAULT NULL,
  `moment_of_inertia` float DEFAULT NULL,
  `ratio_test_result` varchar(255) DEFAULT NULL,
  `vibration_table_test_result` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`calibration_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `system_management_information` (
  `personnel_information` varchar(255) NOT NULL,
  `equivalent_pendulum_length` float DEFAULT NULL,
  `moment_of_inertia` float DEFAULT NULL,
  PRIMARY KEY (`personnel_information`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_role_access_control` (
  `user_name` varchar(255) NOT NULL,
  `calibration_time` datetime NOT NULL,
  `role` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_name`, `calibration_time`),
  CONSTRAINT `user_role_access_control_ibfk_1` FOREIGN KEY (`user_name`) REFERENCES `user` (`user_name`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_role_access_control_ibfk_2` FOREIGN KEY (`calibration_time`) REFERENCES `seismometer_calibration_information` (`calibration_time`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
