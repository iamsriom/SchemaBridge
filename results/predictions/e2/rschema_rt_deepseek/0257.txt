-- Schema for database: rschema_0257
CREATE TABLE `Customer` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `id_number` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `contact_information` varchar(255) NOT NULL,
  `booking_date` datetime NOT NULL,
  `check_in_time` datetime NOT NULL,
  `check_out_time` datetime NOT NULL,
  `type` varchar(50) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Flight` (
  `flight_id` int(11) NOT NULL,
  `flight_name` varchar(255) NOT NULL,
  `airline_name` varchar(255) NOT NULL,
  `departure_location` varchar(255) NOT NULL,
  `destination` varchar(255) NOT NULL,
  `flight_date` date NOT NULL,
  `departure_time` datetime NOT NULL,
  `arrival_time` datetime NOT NULL,
  PRIMARY KEY (`flight_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Flight_Booking_Record` (
  `booking_record_id` int(11) NOT NULL,
  `booked_flight_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `discount_rate` decimal(5,2) DEFAULT NULL,
  `ticket_order_number` varchar(255) NOT NULL,
  PRIMARY KEY (`booking_record_id`),
  CONSTRAINT `Flight_Booking_Record_ibfk_1` FOREIGN KEY (`booked_flight_id`) REFERENCES `Flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Flight_Booking_Record_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `Customer` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Order` (
  `order_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `order_flight_id` int(11) NOT NULL,
  `discount_rate` decimal(5,2) DEFAULT NULL,
  `ticket_order_number` varchar(255) NOT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `Order_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `Customer` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Order_ibfk_2` FOREIGN KEY (`order_flight_id`) REFERENCES `Flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
