-- Schema for database: rschema_0266
CREATE TABLE `Pipeline` (
  `Pipeline_ID` int(11) NOT NULL,
  `Diameter` float NOT NULL,
  `Starting_Point` varchar(255) DEFAULT NULL,
  `Ending_Point` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Pipeline_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Maintenance_Personnel` (
  `Maintenance_Personnel_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Phone` varchar(255) NOT NULL,
  PRIMARY KEY (`Maintenance_Personnel_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Maintenance_Activity_Record` (
  `Maintenance_Activity_Record_ID` int(11) NOT NULL,
  `Maintenance_Details` varchar(255) DEFAULT NULL,
  `Maintenance_Time` datetime DEFAULT NULL,
  `Maintenance_Personnel_ID` int(11) NOT NULL,
  `Pipeline_ID` int(11) NOT NULL,
  PRIMARY KEY (`Maintenance_Activity_Record_ID`),
  CONSTRAINT `Maintenance_Activity_Record_ibfk_1` FOREIGN KEY (`Maintenance_Personnel_ID`) REFERENCES `Maintenance_Personnel` (`Maintenance_Personnel_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Maintenance_Activity_Record_ibfk_2` FOREIGN KEY (`Pipeline_ID`) REFERENCES `Pipeline` (`Pipeline_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
