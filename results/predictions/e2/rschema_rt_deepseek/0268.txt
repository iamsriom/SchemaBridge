-- Schema for database: rschema_0268
CREATE TABLE `Restaurant` (
  `Restaurant_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Location` varchar(255) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Operating_Hours` varchar(255) NOT NULL,
  PRIMARY KEY (`Restaurant_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Dish` (
  `Dish_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Category` varchar(255) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`Dish_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Name` varchar(255) NOT NULL,
  `Phone_Number` varchar(15) NOT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Restaurant_Dish` (
  `Restaurant_ID` int(11) NOT NULL,
  `Dish_ID` int(11) NOT NULL,
  PRIMARY KEY (`Restaurant_ID`, `Dish_ID`),
  CONSTRAINT `restaurant_dish_ibfk_1` FOREIGN KEY (`Restaurant_ID`) REFERENCES `Restaurant` (`Restaurant_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `restaurant_dish_ibfk_2` FOREIGN KEY (`Dish_ID`) REFERENCES `Dish` (`Dish_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer_Reservation` (
  `Reservation_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Customer_ID` int(11) NOT NULL,
  `Restaurant_ID` int(11) NOT NULL,
  `Number_of_Diners` int(11) NOT NULL,
  `Reservation_Time` datetime NOT NULL,
  PRIMARY KEY (`Reservation_ID`),
  CONSTRAINT `customer_reservation_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_reservation_ibfk_2` FOREIGN KEY (`Restaurant_ID`) REFERENCES `Restaurant` (`Restaurant_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
