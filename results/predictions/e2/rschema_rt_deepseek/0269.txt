-- Schema for database: rschema_0269
CREATE TABLE `Location` (
  `Table_ID` int(11) NOT NULL,
  `Location` varchar(255) NOT NULL,
  `Seating_Capacity` int(11) NOT NULL,
  PRIMARY KEY (`Table_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Phone` varchar(255) NOT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Reservation` (
  `Customer_ID` int(11) NOT NULL,
  `Table_ID` int(11) NOT NULL,
  `Reservation_Time` datetime NOT NULL,
  `Number_of_Diners` int(11) NOT NULL,
  `Reservation_Status` varchar(255) DEFAULT 'Pending',
  PRIMARY KEY (`Customer_ID`, `Table_ID`, `Reservation_Time`),
  CONSTRAINT `reservation_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reservation_ibfk_2` FOREIGN KEY (`Table_ID`) REFERENCES `Location` (`Table_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
