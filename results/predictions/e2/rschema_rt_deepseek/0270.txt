-- Schema for database: rschema_0270
CREATE TABLE `program` (
  `program_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `type` varchar(255) DEFAULT NULL,
  `broadcast_start` datetime DEFAULT NULL,
  `broadcast_end` datetime DEFAULT NULL,
  `duration` int(11) DEFAULT NULL,
  PRIMARY KEY (`program_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff` (
  `staff_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_production` (
  `program_id` int(11) NOT NULL,
  `production_staff_id` int(11) NOT NULL,
  `working_duration` int(11) DEFAULT NULL,
  PRIMARY KEY (`program_id`, `production_staff_id`),
  CONSTRAINT `program_production_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `program` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_production_ibfk_2` FOREIGN KEY (`production_staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `program_review` (
  `program_id` int(11) NOT NULL,
  `reviewer_id` int(11) NOT NULL,
  `working_duration` int(11) DEFAULT NULL,
  `review_result` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`program_id`, `reviewer_id`),
  CONSTRAINT `program_review_ibfk_1` FOREIGN KEY (`program_id`) REFERENCES `program` (`program_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `program_review_ibfk_2` FOREIGN KEY (`reviewer_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
