-- Schema for database: rschema_0271
CREATE TABLE `Service` (
  `Service_ID` int(11) NOT NULL,
  `Service_Name` varchar(255) NOT NULL,
  `Price` decimal(10,2) NOT NULL,
  `Transmission_Rate` varchar(255) DEFAULT NULL,
  `Transmission_Type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Service_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Client` (
  `Client_ID` int(11) NOT NULL,
  `Client_Name` varchar(255) NOT NULL,
  `Client_Email` varchar(255) DEFAULT NULL,
  `Client_Phone` varchar(20) DEFAULT NULL,
  `Client_Address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Client_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Service_Subscription` (
  `Service_ID` int(11) NOT NULL,
  `Client_ID` int(11) NOT NULL,
  `Subscription_Start_Time` datetime NOT NULL,
  `Subscription_End_Time` datetime DEFAULT NULL,
  `Operator` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Service_ID`, `Client_ID`, `Subscription_Start_Time`),
  CONSTRAINT `service_subscription_ibfk_1` FOREIGN KEY (`Client_ID`) REFERENCES `Client` (`Client_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_subscription_ibfk_2` FOREIGN KEY (`Service_ID`) REFERENCES `Service` (`Service_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
