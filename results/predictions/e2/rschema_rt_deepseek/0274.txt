-- Schema for database: rschema_0274
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Contact_number` int(11) DEFAULT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer_service_representative` (
  `Employee_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Contact_information` varchar(255) DEFAULT NULL,
  `Work_status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Employee_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Feedback` (
  `Feedback_ID` int(11) NOT NULL,
  `Feedback_content` varchar(255) DEFAULT NULL,
  `Feedback_time` datetime DEFAULT NULL,
  `Processing_status` varchar(255) DEFAULT NULL,
  `User_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Feedback_ID`),
  CONSTRAINT `feedback_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Feedback_result` (
  `Feedback_ID` int(11) NOT NULL,
  `Processing_result` varchar(255) DEFAULT NULL,
  `Processing_time` datetime DEFAULT NULL,
  `Processor_employee_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Feedback_ID`),
  CONSTRAINT `feedback_result_ibfk_1` FOREIGN KEY (`Feedback_ID`) REFERENCES `Feedback` (`Feedback_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `feedback_result_ibfk_2` FOREIGN KEY (`Processor_employee_ID`) REFERENCES `Customer_service_representative` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
