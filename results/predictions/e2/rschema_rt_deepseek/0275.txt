-- Schema for database: rschema_0275
CREATE TABLE `Ecosystem` (
  `System_ID` int(11) NOT NULL,
  `Current_Status` varchar(255) DEFAULT NULL,
  `Location` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`System_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operator` (
  `Employee_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Gender` varchar(10) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Employee_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Monitoring_Record` (
  `System_ID` int(11) NOT NULL,
  `Employee_ID` int(11) NOT NULL,
  `Monitoring_Date` datetime NOT NULL,
  `Humidity` float DEFAULT NULL,
  `Soil_Moisture` float DEFAULT NULL,
  `Temperature` float DEFAULT NULL,
  `Plant_Growth` float DEFAULT NULL,
  PRIMARY KEY (`System_ID`, `Monitoring_Date`),
  CONSTRAINT `monitoring_record_ibfk_1` FOREIGN KEY (`Employee_ID`) REFERENCES `Operator` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_2` FOREIGN KEY (`System_ID`) REFERENCES `Ecosystem` (`System_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
