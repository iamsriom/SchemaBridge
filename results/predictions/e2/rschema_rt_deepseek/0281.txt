-- Schema for database: rschema_0281
CREATE TABLE `Exhibition` (
  `Exhibition_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`Exhibition_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Curator` (
  `Curator_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`Curator_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Exhibit` (
  `Exhibit_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text NOT NULL,
  `image_path` varchar(255) DEFAULT NULL,
  `source` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Exhibit_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Exhibition_Management` (
  `Curator_ID` int(11) NOT NULL,
  `Exhibition_ID` int(11) NOT NULL,
  PRIMARY KEY (`Curator_ID`, `Exhibition_ID`),
  CONSTRAINT `exhibition_management_ibfk_1` FOREIGN KEY (`Curator_ID`) REFERENCES `Curator` (`Curator_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `exhibition_management_ibfk_2` FOREIGN KEY (`Exhibition_ID`) REFERENCES `Exhibition` (`Exhibition_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Exhibit_Association` (
  `Exhibit_ID` int(11) NOT NULL,
  `Exhibition_ID` int(11) NOT NULL,
  PRIMARY KEY (`Exhibit_ID`, `Exhibition_ID`),
  CONSTRAINT `exhibit_association_ibfk_1` FOREIGN KEY (`Exhibit_ID`) REFERENCES `Exhibit` (`Exhibit_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `exhibit_association_ibfk_2` FOREIGN KEY (`Exhibition_ID`) REFERENCES `Exhibition` (`Exhibition_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
