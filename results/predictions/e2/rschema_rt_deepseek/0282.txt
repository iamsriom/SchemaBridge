-- Schema for database: rschema_0282
CREATE TABLE `book` (
  `book_title` varchar(255) DEFAULT NULL,
  `category` varchar(255) DEFAULT NULL,
  `isbn` varchar(255) NOT NULL,
  `publication_date` datetime DEFAULT NULL,
  PRIMARY KEY (`isbn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `author` (
  `author_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`author_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `editor` (
  `editor_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `office_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`editor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `book_author` (
  `author_id` int(11) NOT NULL,
  `isbn` varchar(255) NOT NULL,
  PRIMARY KEY (`author_id`, `isbn`),
  CONSTRAINT `book_author_ibfk_1` FOREIGN KEY (`author_id`) REFERENCES `author` (`author_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `book_author_ibfk_2` FOREIGN KEY (`isbn`) REFERENCES `book` (`isbn`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `book_editor` (
  `editor_id` int(11) NOT NULL,
  `isbn` varchar(255) NOT NULL,
  PRIMARY KEY (`editor_id`, `isbn`),
  CONSTRAINT `book_editor_ibfk_1` FOREIGN KEY (`editor_id`) REFERENCES `editor` (`editor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `book_editor_ibfk_2` FOREIGN KEY (`isbn`) REFERENCES `book` (`isbn`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
