-- Schema for database: rschema_0284
CREATE TABLE `Advertisement` (
  `Advertisement_ID` int(11) NOT NULL,
  `Content` text DEFAULT NULL,
  `Name` text DEFAULT NULL,
  PRIMARY KEY (`Advertisement_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Name` text DEFAULT NULL,
  `Contact_Information` text DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Broadcast_Record` (
  `Advertisement_ID` int(11) NOT NULL,
  `Time_Slot` datetime NOT NULL,
  PRIMARY KEY (`Advertisement_ID`, `Time_Slot`),
  CONSTRAINT `broadcast_record_ibfk_1` FOREIGN KEY (`Advertisement_ID`) REFERENCES `Advertisement` (`Advertisement_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Advertisement_Push` (
  `Advertisement_ID` int(11) NOT NULL,
  `User_ID` int(11) NOT NULL,
  PRIMARY KEY (`Advertisement_ID`, `User_ID`),
  CONSTRAINT `advertisement_push_ibfk_1` FOREIGN KEY (`Advertisement_ID`) REFERENCES `Advertisement` (`Advertisement_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `advertisement_push_ibfk_2` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
