-- Schema for database: rschema_0285
CREATE TABLE `Advertisement` (
  `Advertisement_ID` int(11) NOT NULL,
  `Advertisement_Content` varchar(255) DEFAULT NULL,
  `Advertisement_Duration` datetime DEFAULT NULL,
  `Advertisement_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Advertisement_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Client` (
  `Client_ID` int(11) NOT NULL,
  `Client_Name` varchar(255) DEFAULT NULL,
  `Client_Email` varchar(255) DEFAULT NULL,
  `Client_Phone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Client_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Play_Advertisement` (
  `Advertisement_ID` int(11) NOT NULL,
  `Client_ID` int(11) NOT NULL,
  `Start_Time` datetime DEFAULT NULL,
  `End_Time` datetime DEFAULT NULL,
  `Play_Duration` datetime DEFAULT NULL,
  `Viewership_Rating` float DEFAULT NULL,
  PRIMARY KEY (`Advertisement_ID`, `Client_ID`),
  CONSTRAINT `play_advertisement_ibfk_1` FOREIGN KEY (`Advertisement_ID`) REFERENCES `Advertisement` (`Advertisement_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `play_advertisement_ibfk_2` FOREIGN KEY (`Client_ID`) REFERENCES `Client` (`Client_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
