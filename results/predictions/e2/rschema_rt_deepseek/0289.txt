-- Schema for database: rschema_0289
CREATE TABLE `contact_information` (
  `contact_information` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`contact_information`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_process` (
  `operator_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `process_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`operator_id`),
  CONSTRAINT `production_process_ibfk_1` FOREIGN KEY (`operator_id`) REFERENCES `contact_information` (`contact_information`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_record` (
  `production_process_id` int(11) NOT NULL,
  `recording_time` datetime NOT NULL,
  `monitoring_parameters` varchar(255) NOT NULL,
  PRIMARY KEY (`production_process_id`, `recording_time`),
  CONSTRAINT `monitoring_record_ibfk_1` FOREIGN KEY (`production_process_id`) REFERENCES `production_process` (`process_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
