-- Schema for database: rschema_0290
CREATE TABLE `Project` (
  `Project_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Budget` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`Project_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Fund_Flow_Record` (
  `Flow_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Project_ID` int(11) NOT NULL,
  `Amount` decimal(10,2) DEFAULT NULL,
  `Date` datetime DEFAULT NULL,
  `Fund_Type` varchar(255) DEFAULT NULL,
  `Remarks` text DEFAULT NULL,
  PRIMARY KEY (`Flow_ID`),
  CONSTRAINT `Fund_Flow_Record_ibfk_1` FOREIGN KEY (`Project_ID`) REFERENCES `Project` (`Project_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
