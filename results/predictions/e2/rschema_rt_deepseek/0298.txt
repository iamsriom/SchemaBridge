-- Schema for database: rschema_0298
CREATE TABLE `Salary_Level` (
  `Salary_Level_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Maximum_Salary` float NOT NULL,
  `Minimum_Salary` float NOT NULL,
  PRIMARY KEY (`Salary_Level_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Department` (
  `Department_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Department_Location` varchar(255) NOT NULL,
  `Department_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Department_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Employee` (
  `Employee_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Name` varchar(255) NOT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  `Department_ID` int(11) NOT NULL,
  `Salary_Level_ID` int(11) NOT NULL,
  `Supervisor_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Employee_ID`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Department_ID`) REFERENCES `Department` (`Department_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`Salary_Level_ID`) REFERENCES `Salary_Level` (`Salary_Level_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_3` FOREIGN KEY (`Supervisor_ID`) REFERENCES `Employee` (`Employee_ID`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
