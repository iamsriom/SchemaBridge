-- Schema for database: rschema_0299
CREATE TABLE `City` (
  `city_ID` int(11) NOT NULL,
  `city_name` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  `latitude` decimal(9,6) NOT NULL,
  `longitude` decimal(9,6) NOT NULL,
  `population` int(11) DEFAULT NULL,
  `timezone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`city_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Weather_Description` (
  `weather_type_ID` int(11) NOT NULL,
  `detailed_description` varchar(255) NOT NULL,
  `icon` varchar(255) DEFAULT NULL,
  `main_description` varchar(255) NOT NULL,
  PRIMARY KEY (`weather_type_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Weather_Forecast` (
  `forecast_ID` int(11) NOT NULL,
  `morning_temperature` decimal(5,2) DEFAULT NULL,
  `afternoon_temperature` decimal(5,2) DEFAULT NULL,
  `noon_temperature` decimal(5,2) DEFAULT NULL,
  `humidity` int(11) DEFAULT NULL,
  `atmospheric_pressure` decimal(7,2) DEFAULT NULL,
  `cloud_cover` int(11) DEFAULT NULL,
  `date` date NOT NULL,
  `sunrise_time` datetime DEFAULT NULL,
  `sunset_time` datetime DEFAULT NULL,
  `wind_direction` varchar(255) DEFAULT NULL,
  `wind_speed` decimal(5,2) DEFAULT NULL,
  `city_ID` int(11) NOT NULL,
  `weather_type_ID` int(11) NOT NULL,
  PRIMARY KEY (`forecast_ID`),
  CONSTRAINT `weather_forecast_ibfk_1` FOREIGN KEY (`city_ID`) REFERENCES `City` (`city_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `weather_forecast_ibfk_2` FOREIGN KEY (`weather_type_ID`) REFERENCES `Weather_Description` (`weather_type_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
