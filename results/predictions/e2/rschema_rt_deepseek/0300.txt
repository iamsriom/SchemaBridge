-- Schema for database: rschema_0300
CREATE TABLE `material_table` (
  `material_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `unit_price` float DEFAULT NULL,
  PRIMARY KEY (`material_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_table` (
  `project_id` int(11) NOT NULL,
  `creation_date` datetime DEFAULT NULL,
  `close_date` datetime DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bill_table` (
  `bill_id` int(11) NOT NULL,
  `creation_time` datetime DEFAULT NULL,
  `modification_time` datetime DEFAULT NULL,
  `project_id` int(11) NOT NULL,
  `total_sale_amount` float DEFAULT NULL,
  `unit` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`bill_id`),
  CONSTRAINT `bill_table_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project_table` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bill_details_table` (
  `bill_id` int(11) NOT NULL,
  `material_id` int(11) NOT NULL,
  `quantity` int(11) DEFAULT NULL,
  `selling_price` float DEFAULT NULL,
  PRIMARY KEY (`bill_id`, `material_id`),
  CONSTRAINT `bill_details_table_ibfk_1` FOREIGN KEY (`bill_id`) REFERENCES `bill_table` (`bill_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bill_details_table_ibfk_2` FOREIGN KEY (`material_id`) REFERENCES `material_table` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
