-- Schema for database: rschema_0301
CREATE TABLE `Seat_Template` (
  `template_ID` int(11) NOT NULL,
  `template_name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`template_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `user_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `contact_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`user_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Seat` (
  `seat_ID` int(11) NOT NULL,
  `row` int(11) NOT NULL,
  `column` int(11) NOT NULL,
  `location_X` int(11) NOT NULL,
  `location_Y` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `type` varchar(50) DEFAULT NULL,
  `flag` varchar(50) DEFAULT NULL,
  `creation_time` datetime DEFAULT NULL,
  `update_time` datetime DEFAULT NULL,
  `reservedPurchasedUserID` int(11) NOT NULL,
  PRIMARY KEY (`seat_ID`),
  CONSTRAINT `seat_ibfk_1` FOREIGN KEY (`reservedPurchasedUserID`) REFERENCES `User` (`user_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Seat_Template_Layout` (
  `seat_ID` int(11) NOT NULL,
  `template_ID` int(11) NOT NULL,
  `occasion` varchar(255) DEFAULT NULL,
  `time_period` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`seat_ID`, `template_ID`),
  CONSTRAINT `seat_template_layout_ibfk_1` FOREIGN KEY (`seat_ID`) REFERENCES `Seat` (`seat_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `seat_template_layout_ibfk_2` FOREIGN KEY (`template_ID`) REFERENCES `Seat_Template` (`template_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
