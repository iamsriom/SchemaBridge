-- Schema for database: rschema_0307
CREATE TABLE `companies_table` (
  `company_id` int(11) NOT NULL,
  `company_name` varchar(255) NOT NULL,
  `company_website` varchar(255) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_keys_table` (
  `key_id` int(11) NOT NULL,
  `key_name` varchar(255) NOT NULL,
  `private_key` varchar(255) DEFAULT NULL,
  `uri` varchar(255) DEFAULT NULL,
  `application_reason` varchar(255) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`key_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employees_table` (
  `employee_id` int(11) NOT NULL,
  `company_id` int(11) DEFAULT NULL,
  `employee_name` varchar(255) NOT NULL,
  `position` varchar(255) DEFAULT NULL,
  `salary` varchar(255) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employees_table_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `companies_table` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sites_table` (
  `site_id` int(11) NOT NULL,
  `site_name` varchar(255) NOT NULL,
  `site_abbreviation` varchar(255) DEFAULT NULL,
  `hostname` varchar(255) DEFAULT NULL,
  `database_name` varchar(255) DEFAULT NULL,
  `is_deprecated` boolean DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`site_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `devices_table` (
  `device_id` int(11) NOT NULL,
  `device_name` varchar(255) NOT NULL,
  `associated_employee_id` int(11) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`device_id`),
  CONSTRAINT `devices_table_ibfk_1` FOREIGN KEY (`associated_employee_id`) REFERENCES `employees_table` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users_table` (
  `user_id` int(11) NOT NULL,
  `company_id` int(11) DEFAULT NULL,
  `site_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `display_name_usage` varchar(255) DEFAULT NULL,
  `email` varchar(255) NOT NULL,
  `is_admin` boolean DEFAULT NULL,
  `is_developer` boolean DEFAULT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `password` varchar(255) NOT NULL,
  `password_salt` varchar(255) DEFAULT NULL,
  `password_type` varchar(255) DEFAULT NULL,
  `real_name` varchar(255) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `users_table_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `companies_table` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `users_table_ibfk_2` FOREIGN KEY (`site_id`) REFERENCES `sites_table` (`site_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
