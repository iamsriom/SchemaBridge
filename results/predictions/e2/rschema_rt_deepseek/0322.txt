-- Schema for database: rschema_0322
CREATE TABLE `company` (
  `company_id` varchar(255) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `website` varchar(255) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` varchar(255) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `salary` decimal(10,2) DEFAULT NULL,
  `belonging_company` varchar(255) NOT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`belonging_company`) REFERENCES `company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device` (
  `device_id` varchar(255) NOT NULL,
  `device_name` varchar(255) DEFAULT NULL,
  `belonging_employee` varchar(255) NOT NULL,
  PRIMARY KEY (`device_id`),
  CONSTRAINT `device_ibfk_1` FOREIGN KEY (`belonging_employee`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` varchar(255) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `last_login_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `site` (
  `site_name` varchar(255) NOT NULL,
  `hostname` varchar(255) DEFAULT NULL,
  `site_status` varchar(255) DEFAULT NULL,
  `belonging_user` varchar(255) NOT NULL,
  PRIMARY KEY (`site_name`),
  CONSTRAINT `site_ibfk_1` FOREIGN KEY (`belonging_user`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_uses_device` (
  `employee_id` varchar(255) NOT NULL,
  `device_id` varchar(255) NOT NULL,
  `start_usage_time` datetime DEFAULT NULL,
  `end_usage_time` datetime DEFAULT NULL,
  PRIMARY KEY (`employee_id`, `device_id`),
  CONSTRAINT `employee_uses_device_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_uses_device_ibfk_2` FOREIGN KEY (`device_id`) REFERENCES `device` (`device_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_creates_site` (
  `username` varchar(255) NOT NULL,
  `site_name` varchar(255) NOT NULL,
  `creation_time` datetime DEFAULT NULL,
  PRIMARY KEY (`username`, `site_name`),
  CONSTRAINT `user_creates_site_ibfk_1` FOREIGN KEY (`username`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_creates_site_ibfk_2` FOREIGN KEY (`site_name`) REFERENCES `site` (`site_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
