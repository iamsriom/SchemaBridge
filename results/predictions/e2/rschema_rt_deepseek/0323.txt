-- Schema for database: rschema_0323
CREATE TABLE `Salesperson` (
  `Salesperson_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Salesperson_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Sales_record` (
  `Salesperson_ID` int(11) DEFAULT NULL,
  `Customer_ID` int(11) DEFAULT NULL,
  `Sales_date` datetime DEFAULT NULL,
  `Sales_amount` decimal(10,2) NOT NULL,
  `Transaction_quantity` int(11) DEFAULT NULL,
  `Visit_count` int(11) DEFAULT NULL,
  PRIMARY KEY (`Salesperson_ID`, `Customer_ID`, `Sales_date`),
  CONSTRAINT `Sales_record_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Sales_record_ibfk_2` FOREIGN KEY (`Salesperson_ID`) REFERENCES `Salesperson` (`Salesperson_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
