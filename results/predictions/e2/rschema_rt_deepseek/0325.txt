-- Schema for database: rschema_0325
CREATE TABLE `Class` (
  `Class_ID` int(11) NOT NULL,
  `Class_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Class_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Character` (
  `Character_ID` int(11) NOT NULL,
  `Character_Name` varchar(255) DEFAULT NULL,
  `Class_ID` int(11) NOT NULL,
  `Rarity_ID` int(11) DEFAULT NULL,
  `Star_Level` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Character_ID`),
  CONSTRAINT `Character_ibfk_1` FOREIGN KEY (`Class_ID`) REFERENCES `Class` (`Class_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
