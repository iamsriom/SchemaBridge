-- Schema for database: rschema_0327
CREATE TABLE `Address` (
  `Address_ID` int(11) NOT NULL,
  `City` varchar(255) NOT NULL,
  `Country` varchar(255) NOT NULL,
  `District` varchar(255) NOT NULL,
  `Postal_Code` varchar(20) NOT NULL,
  `Street_Address` varchar(255) NOT NULL,
  PRIMARY KEY (`Address_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Resource` (
  `Resource_ID` int(11) NOT NULL,
  `Resource_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Resource_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `First_Name` varchar(255) NOT NULL,
  `Last_Name` varchar(255) NOT NULL,
  `Phone_Number` varchar(20) NOT NULL,
  `Address_ID` int(11) NOT NULL,
  `Resource_ID` int(11) NOT NULL,
  PRIMARY KEY (`Customer_ID`),
  CONSTRAINT `customer_ibfk_1` FOREIGN KEY (`Address_ID`) REFERENCES `Address` (`Address_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_ibfk_2` FOREIGN KEY (`Resource_ID`) REFERENCES `Resource` (`Resource_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Order` (
  `Order_ID` int(11) NOT NULL,
  `Amount` decimal(10,2) NOT NULL,
  `Currency` varchar(10) NOT NULL,
  `Start_Time` datetime NOT NULL,
  `Type` varchar(50) NOT NULL,
  `Customer_ID` int(11) NOT NULL,
  `Resource_ID` int(11) NOT NULL,
  PRIMARY KEY (`Order_ID`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_ibfk_2` FOREIGN KEY (`Resource_ID`) REFERENCES `Resource` (`Resource_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
