-- Schema for database: rschema_0328
CREATE TABLE `administrator` (
  `administrator_id` int(11) NOT NULL,
  `account` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`administrator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `survey` (
  `survey_id` int(11) NOT NULL,
  `survey_title` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `background_image_url` varchar(255) DEFAULT NULL,
  `survey_logo_url` varchar(255) DEFAULT NULL,
  `start_time` datetime DEFAULT NULL,
  `end_time` datetime DEFAULT NULL,
  `survey_status` varchar(50) DEFAULT NULL,
  `access_restriction` varchar(50) DEFAULT NULL,
  `access_rule` varchar(50) DEFAULT NULL,
  `anonymous` tinyint(1) DEFAULT NULL,
  PRIMARY KEY (`survey_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `question` (
  `question_id` int(11) NOT NULL,
  `question_title` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `display_order` int(11) DEFAULT NULL,
  `display_style` varchar(50) DEFAULT NULL,
  `evaluation_score` int(11) DEFAULT NULL,
  `for_evaluation` tinyint(1) DEFAULT NULL,
  `required` tinyint(1) DEFAULT NULL,
  `question_type` varchar(50) DEFAULT NULL,
  `validation_type` varchar(50) DEFAULT NULL,
  `survey_id` int(11) NOT NULL,
  PRIMARY KEY (`question_id`),
  CONSTRAINT `question_ibfk_1` FOREIGN KEY (`survey_id`) REFERENCES `survey` (`survey_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `option` (
  `option_id` int(11) NOT NULL,
  `option_content` text DEFAULT NULL,
  `display_order` int(11) DEFAULT NULL,
  `is_correct_answer` tinyint(1) DEFAULT NULL,
  `question_id` int(11) NOT NULL,
  PRIMARY KEY (`option_id`),
  CONSTRAINT `option_ibfk_1` FOREIGN KEY (`question_id`) REFERENCES `question` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `text_answer_record` (
  `answer_id` int(11) NOT NULL,
  `answer_content` text DEFAULT NULL,
  `question_id` int(11) NOT NULL,
  `respondent_information` varchar(255) DEFAULT NULL,
  `submission_time` datetime DEFAULT NULL,
  PRIMARY KEY (`answer_id`),
  CONSTRAINT `text_answer_record_ibfk_1` FOREIGN KEY (`question_id`) REFERENCES `question` (`question_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `choice_answer_record` (
  `answer_id` int(11) NOT NULL,
  `answer_type` varchar(50) DEFAULT NULL,
  `option_id` int(11) NOT NULL,
  `respondent_information` varchar(255) DEFAULT NULL,
  `submission_time` datetime DEFAULT NULL,
  PRIMARY KEY (`answer_id`),
  CONSTRAINT `choice_answer_record_ibfk_1` FOREIGN KEY (`option_id`) REFERENCES `option` (`option_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
