-- Schema for database: rschema_0334
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `avatar_path` varchar(255) DEFAULT NULL,
  `date_of_birth` datetime DEFAULT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `first_name` varchar(255) DEFAULT NULL,
  `last_name` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Post` (
  `Post_ID` int(11) NOT NULL,
  `post_content` varchar(255) DEFAULT NULL,
  `post_title` varchar(255) DEFAULT NULL,
  `Poster_User_ID` int(11) NOT NULL,
  PRIMARY KEY (`Post_ID`),
  CONSTRAINT `Post_ibfk_1` FOREIGN KEY (`Poster_User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Comment` (
  `Comment_ID` int(11) NOT NULL,
  `comment_content` varchar(255) DEFAULT NULL,
  `Commented_Post_ID` int(11) NOT NULL,
  `Commenter_User_ID` int(11) NOT NULL,
  PRIMARY KEY (`Comment_ID`),
  CONSTRAINT `Comment_ibfk_1` FOREIGN KEY (`Commented_Post_ID`) REFERENCES `Post` (`Post_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Comment_ibfk_2` FOREIGN KEY (`Commenter_User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
