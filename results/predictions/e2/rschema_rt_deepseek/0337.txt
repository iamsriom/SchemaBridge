-- Schema for database: rschema_0337
CREATE TABLE `Exam` (
  `Exam_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Exam_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Student` (
  `Student_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Student_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Practice_Question` (
  `Question_ID` int(11) NOT NULL,
  `Exam_ID_it_belongs_to` int(11) NOT NULL,
  `Question_Content` varchar(255) DEFAULT NULL,
  `Score` float DEFAULT NULL,
  PRIMARY KEY (`Question_ID`),
  CONSTRAINT `Practice_Question_ibfk_1` FOREIGN KEY (`Exam_ID_it_belongs_to`) REFERENCES `Exam` (`Exam_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Submission_Record` (
  `Submission_ID` int(11) NOT NULL,
  `Exam_ID` int(11) NOT NULL,
  `Student_ID` int(11) NOT NULL,
  `Submission_Status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Submission_ID`),
  CONSTRAINT `Submission_Record_ibfk_1` FOREIGN KEY (`Exam_ID`) REFERENCES `Exam` (`Exam_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Submission_Record_ibfk_2` FOREIGN KEY (`Student_ID`) REFERENCES `Student` (`Student_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Answer_Record` (
  `Answer_Record_ID` int(11) NOT NULL,
  `Answer_Content` varchar(255) DEFAULT NULL,
  `Practice_Question_ID_it_belongs_to` int(11) NOT NULL,
  `Submission_ID_it_belongs_to` int(11) NOT NULL,
  PRIMARY KEY (`Answer_Record_ID`),
  CONSTRAINT `Answer_Record_ibfk_1` FOREIGN KEY (`Practice_Question_ID_it_belongs_to`) REFERENCES `Practice_Question` (`Question_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Answer_Record_ibfk_2` FOREIGN KEY (`Submission_ID_it_belongs_to`) REFERENCES `Submission_Record` (`Submission_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
