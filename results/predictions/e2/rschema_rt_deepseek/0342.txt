-- Schema for database: rschema_0342
CREATE TABLE `CORS_Station` (
  `CORS_Station_ID` int(11) NOT NULL,
  `CORS_Station_Name` varchar(255) DEFAULT NULL,
  `CORS_Station_Description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`CORS_Station_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `CORS_Network` (
  `CORS_Network_ID` int(11) NOT NULL,
  `CORS_Network_Name` varchar(255) DEFAULT NULL,
  `CORS_Network_Description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`CORS_Network_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Station_Network_Relationship` (
  `CORS_Network_ID` int(11) NOT NULL,
  `CORS_Station_ID` int(11) NOT NULL,
  `Effective_Start_Time` datetime DEFAULT NULL,
  `Effective_End_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`CORS_Network_ID`, `CORS_Station_ID`),
  CONSTRAINT `Station_Network_Relationship_ibfk_1` FOREIGN KEY (`CORS_Network_ID`) REFERENCES `CORS_Network` (`CORS_Network_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Station_Network_Relationship_ibfk_2` FOREIGN KEY (`CORS_Station_ID`) REFERENCES `CORS_Station` (`CORS_Station_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
