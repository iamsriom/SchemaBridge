-- Schema for database: rschema_0346
CREATE TABLE `account` (
  `account` varchar(255) NOT NULL,
  `affiliated_unit` varchar(255) DEFAULT NULL,
  `user` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`account`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expense` (
  `record_id` int(11) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `associated_account` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  `description` text NOT NULL,
  `due_time` datetime NOT NULL,
  `payment_time` datetime NOT NULL,
  `status` varchar(50) NOT NULL DEFAULT 'Pending',
  PRIMARY KEY (`record_id`),
  CONSTRAINT `expense_ibfk_1` FOREIGN KEY (`associated_account`) REFERENCES `account` (`account`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
