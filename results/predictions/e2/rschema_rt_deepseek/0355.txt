-- Schema for database: rschema_0355
CREATE TABLE `Race` (
  `Race_ID` int(11) NOT NULL,
  `Race_Name` varchar(255) NOT NULL,
  `Race_Result` varchar(255) DEFAULT NULL,
  `Start_Time` datetime NOT NULL,
  `End_Time` datetime NOT NULL,
  PRIMARY KEY (`Race_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Horse` (
  `Horse_ID` int(11) NOT NULL,
  `Horse_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Horse_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Bettor` (
  `Bettor_ID` int(11) NOT NULL,
  `Identity_Information` varchar(255) NOT NULL,
  `Phone_Number` varchar(255) NOT NULL,
  PRIMARY KEY (`Bettor_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Horse_Participation` (
  `Horse_ID` int(11) NOT NULL,
  `Race_ID` int(11) NOT NULL,
  `Horse_Ranking` int(11) NOT NULL,
  `Race_Time` datetime NOT NULL,
  PRIMARY KEY (`Horse_ID`, `Race_ID`),
  CONSTRAINT `horse_participation_ibfk_1` FOREIGN KEY (`Horse_ID`) REFERENCES `Horse` (`Horse_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `horse_participation_ibfk_2` FOREIGN KEY (`Race_ID`) REFERENCES `Race` (`Race_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Bettor_Betting_Race` (
  `Bettor_ID` int(11) NOT NULL,
  `Horse_ID` int(11) NOT NULL,
  `Race_ID` int(11) NOT NULL,
  `Betting_Amount` float NOT NULL,
  `Betting_Time` datetime NOT NULL,
  `Expected_Ranking` int(11) NOT NULL,
  `Actual_Result` varchar(255) DEFAULT NULL,
  `Winning_Status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Bettor_ID`, `Horse_ID`, `Race_ID`),
  CONSTRAINT `bettor_betting_race_ibfk_1` FOREIGN KEY (`Bettor_ID`) REFERENCES `Bettor` (`Bettor_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bettor_betting_race_ibfk_2` FOREIGN KEY (`Horse_ID`) REFERENCES `Horse` (`Horse_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bettor_betting_race_ibfk_3` FOREIGN KEY (`Race_ID`) REFERENCES `Race` (`Race_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
