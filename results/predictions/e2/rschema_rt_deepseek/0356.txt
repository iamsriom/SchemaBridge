-- Schema for database: rschema_0356
CREATE TABLE `Work_Item` (
  `creation_Date` DATETIME NOT NULL,
  `description` VARCHAR(255) NOT NULL,
  `update_Date` DATETIME NOT NULL,
  `workItem_ID` INT(11) NOT NULL,
  `workItem_Name` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`workItem_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Work_Item_Attributes` (
  `Attributes_ID` INT(11) NOT NULL,
  `Attributes_Name` VARCHAR(255) NOT NULL,
  `Attributes_Value` FLOAT NOT NULL,
  `link_Status` VARCHAR(255) NOT NULL,
  `workItem_ID` INT(11) NOT NULL,
  PRIMARY KEY (`Attributes_ID`, `workItem_ID`),
  CONSTRAINT `Work_Item_Attributes_ibfk_1` FOREIGN KEY (`workItem_ID`) REFERENCES `Work_Item` (`workItem_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
