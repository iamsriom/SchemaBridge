-- Schema for database: rschema_0359
CREATE TABLE `user` (
  `account` varchar(255) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `avatar_url` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`account`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `event` (
  `event_id` int(11) NOT NULL AUTO_INCREMENT,
  `event_name` varchar(255) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `time` datetime DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `event_type` varchar(255) DEFAULT NULL,
  `creator` varchar(255) NOT NULL,
  PRIMARY KEY (`event_id`),
  CONSTRAINT `event_ibfk_1` FOREIGN KEY (`creator`) REFERENCES `user` (`account`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `message` (
  `message_id` int(11) NOT NULL AUTO_INCREMENT,
  `associated_event` int(11) NOT NULL,
  `message_content` text DEFAULT NULL,
  `sender` varchar(255) NOT NULL,
  PRIMARY KEY (`message_id`),
  CONSTRAINT `message_ibfk_1` FOREIGN KEY (`associated_event`) REFERENCES `event` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `message_ibfk_2` FOREIGN KEY (`sender`) REFERENCES `user` (`account`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
