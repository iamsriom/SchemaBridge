-- Schema for database: rschema_0360
CREATE TABLE `log` (
  `log_id` int(11) NOT NULL,
  `log_category` varchar(255) DEFAULT NULL,
  `log_content` varchar(255) DEFAULT NULL,
  `log_creation_time` datetime DEFAULT NULL,
  `log_deletion_time` datetime DEFAULT NULL,
  `log_last_modified_time` datetime DEFAULT NULL,
  PRIMARY KEY (`log_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operator` (
  `id` int(11) NOT NULL,
  `account` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operation_log` (
  `log_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `operation_time` datetime NOT NULL,
  `execution_content` varchar(255) DEFAULT NULL,
  `operation_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`log_id`, `user_id`, `operation_time`),
  CONSTRAINT `operation_log_ibfk_1` FOREIGN KEY (`log_id`) REFERENCES `log` (`log_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `operation_log_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `operator` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
