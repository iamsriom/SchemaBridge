-- Schema for database: rschema_0361
CREATE TABLE `Restaurant` (
  `Restaurant_ID` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `average_cost` float DEFAULT NULL,
  `business_hours` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Restaurant_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_review` (
  `Review_ID` int(11) NOT NULL,
  `content` varchar(255) DEFAULT NULL,
  `dining_time` datetime DEFAULT NULL,
  `rating` float DEFAULT NULL,
  `photos` blob DEFAULT NULL,
  PRIMARY KEY (`Review_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Dish` (
  `Dish_ID` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `description` varchar(255) DEFAULT NULL,
  `category` varchar(255) DEFAULT NULL,
  `price` float DEFAULT NULL,
  `monthly_sales` float DEFAULT NULL,
  `satisfaction_rate` float DEFAULT NULL,
  `Restaurant_ID` int(11) NOT NULL,
  `image_url` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Dish_ID`),
  CONSTRAINT `Dish_ibfk_1` FOREIGN KEY (`Restaurant_ID`) REFERENCES `Restaurant` (`Restaurant_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Review_reply` (
  `Reply_ID` int(11) NOT NULL,
  `content` varchar(255) DEFAULT NULL,
  `reply_time` datetime DEFAULT NULL,
  `Review_ID_being_replied_to` int(11) NOT NULL,
  PRIMARY KEY (`Reply_ID`),
  CONSTRAINT `Review_reply_ibfk_1` FOREIGN KEY (`Review_ID_being_replied_to`) REFERENCES `User_review` (`Review_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
