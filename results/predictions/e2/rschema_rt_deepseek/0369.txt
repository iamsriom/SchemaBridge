-- Schema for database: rschema_0369
CREATE TABLE `content` (
  `content_id` int(11) NOT NULL,
  `content_description` varchar(255) DEFAULT NULL,
  `content_type` varchar(255) DEFAULT NULL,
  `creation_time` datetime DEFAULT NULL,
  `title` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`content_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `role` varchar(255) DEFAULT NULL,
  `username` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `operation_log` (
  `log_id` int(11) NOT NULL,
  `content_id` int(11) NOT NULL,
  `operation_comment` varchar(255) DEFAULT NULL,
  `operation_time` datetime DEFAULT NULL,
  `operation_type` varchar(255) DEFAULT NULL,
  `operation_user` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`log_id`),
  CONSTRAINT `operation_log_ibfk_1` FOREIGN KEY (`content_id`) REFERENCES `content` (`content_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_content_modification` (
  `user_id` int(11) NOT NULL,
  `content_id` int(11) NOT NULL,
  `modification_time` datetime NOT NULL,
  `modified_content` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `content_id`, `modification_time`),
  CONSTRAINT `user_content_modification_ibfk_1` FOREIGN KEY (`content_id`) REFERENCES `content` (`content_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_content_modification_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
