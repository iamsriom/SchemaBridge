-- Schema for database: rschema_0377
CREATE TABLE `article` (
  `post_id` varchar(255) NOT NULL,
  `article_title` varchar(255) NOT NULL,
  PRIMARY KEY (`article_title`, `post_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tag` (
  `tag_id` varchar(255) NOT NULL,
  `tag_category` varchar(255) NOT NULL,
  PRIMARY KEY (`tag_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `article_tag_association` (
  `post_id` varchar(255) NOT NULL,
  `tag_id` varchar(255) NOT NULL,
  `created_at` datetime DEFAULT NULL,
  `fk_post_id` varchar(255) NOT NULL,
  `fk_tag_id` varchar(255) NOT NULL,
  PRIMARY KEY (`tag_id`, `post_id`),
  CONSTRAINT `article_tag_association_ibfk_1` FOREIGN KEY (`fk_post_id`) REFERENCES `article` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `article_tag_association_ibfk_2` FOREIGN KEY (`fk_tag_id`) REFERENCES `tag` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
