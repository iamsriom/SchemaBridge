-- Schema for database: rschema_0381
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `News` (
  `News_ID` int(11) NOT NULL,
  `News_Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`News_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Subscription` (
  `User_ID` int(11) NOT NULL,
  `News_ID` int(11) NOT NULL,
  `Creation_Time` datetime DEFAULT NULL,
  `Modification_Time` datetime DEFAULT NULL,
  `Subscription_IP` varchar(255) DEFAULT NULL,
  `Subscription_Status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`, `News_ID`),
  CONSTRAINT `subscription_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `subscription_ibfk_2` FOREIGN KEY (`News_ID`) REFERENCES `News` (`News_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
