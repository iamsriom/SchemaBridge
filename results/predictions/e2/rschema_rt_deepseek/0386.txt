-- Schema for database: rschema_0386
CREATE TABLE `User` (
  `ID` int(11) NOT NULL,
  `Username` varchar(255) DEFAULT NULL,
  `Public_Key` varchar(255) DEFAULT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Creation_Time` timestamp DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Block` (
  `Block_ID` int(11) NOT NULL,
  `Block_Hash` varchar(255) DEFAULT NULL,
  `Block_Size` int(11) DEFAULT NULL,
  `Creation_Time` timestamp DEFAULT CURRENT_TIMESTAMP,
  `Previous_Block_ID` int(11) DEFAULT NULL,
  PRIMARY KEY (`Block_ID`),
  CONSTRAINT `Block_ibfk_1` FOREIGN KEY (`Previous_Block_ID`) REFERENCES `Block` (`Block_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Transaction_Record` (
  `Sender_ID` int(11) NOT NULL,
  `Receiver_ID` int(11) NOT NULL,
  `Transaction_Time` timestamp NOT NULL,
  `Block_ID` int(11) NOT NULL,
  `Operation_Type` varchar(255) DEFAULT NULL,
  `Operation_Content` varchar(255) DEFAULT NULL,
  `Transaction_Amount` int(11) DEFAULT NULL,
  PRIMARY KEY (`Sender_ID`, `Receiver_ID`, `Transaction_Time`, `Block_ID`),
  CONSTRAINT `Transaction_Record_ibfk_1` FOREIGN KEY (`Block_ID`) REFERENCES `Block` (`Block_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Transaction_Record_ibfk_2` FOREIGN KEY (`Receiver_ID`) REFERENCES `User` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Transaction_Record_ibfk_3` FOREIGN KEY (`Sender_ID`) REFERENCES `User` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
