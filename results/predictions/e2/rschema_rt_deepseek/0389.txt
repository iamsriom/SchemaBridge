-- Schema for database: rschema_0389
CREATE TABLE `Student` (
  `Student_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` varchar(255) NOT NULL,
  `Phone` varchar(255) DEFAULT NULL,
  `Age` int(11) NOT NULL,
  PRIMARY KEY (`Student_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Teacher` (
  `Teacher_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Gender` varchar(255) NOT NULL,
  `Phone` varchar(255) DEFAULT NULL,
  `Title` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Teacher_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Course` (
  `Course_ID` int(11) NOT NULL,
  `Class_Hours` float NOT NULL,
  `Class_Location` varchar(255) NOT NULL,
  `Credits` float NOT NULL,
  `Number_of_Students_Enrolled` int(11) NOT NULL,
  `Teaching_Teacher_ID` int(11) NOT NULL,
  PRIMARY KEY (`Course_ID`),
  CONSTRAINT `course_ibfk_1` FOREIGN KEY (`Teaching_Teacher_ID`) REFERENCES `Teacher` (`Teacher_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Student_Course_Selection` (
  `Student_ID` int(11) NOT NULL,
  `Course_ID` int(11) NOT NULL,
  `Grades` int(11) NOT NULL,
  `Selection_Time` datetime NOT NULL,
  PRIMARY KEY (`Student_ID`, `Course_ID`, `Selection_Time`),
  CONSTRAINT `student_course_selection_ibfk_1` FOREIGN KEY (`Student_ID`) REFERENCES `Student` (`Student_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_course_selection_ibfk_2` FOREIGN KEY (`Course_ID`) REFERENCES `Course` (`Course_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
