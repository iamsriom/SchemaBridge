-- Schema for database: rschema_0390
CREATE TABLE `Consumer` (
  `Consumer_ID` int(11) NOT NULL,
  `Address_Information` varchar(255) DEFAULT NULL,
  `Phone_Number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Consumer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Courier_Company` (
  `Company_Name` varchar(255) NOT NULL,
  `Arrival_Time` datetime DEFAULT NULL,
  `Charging_Price` float DEFAULT NULL,
  PRIMARY KEY (`Company_Name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Courier_Tracking` (
  `Courier_Tracking_Number` varchar(255) NOT NULL,
  `Recipient_ID` int(11) DEFAULT NULL,
  `Recipient_Name` varchar(255) DEFAULT NULL,
  `Recipient_Phone_Number` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Courier_Tracking_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Administrator` (
  `Administrator_ID` int(11) NOT NULL,
  `Courier_Username` varchar(255) DEFAULT NULL,
  `Login_Password` varchar(255) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Administrator_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Cooperation` (
  `Administrator_ID` int(11) NOT NULL,
  `Company_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Administrator_ID`, `Company_Name`),
  CONSTRAINT `cooperation_ibfk_1` FOREIGN KEY (`Administrator_ID`) REFERENCES `Administrator` (`Administrator_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `cooperation_ibfk_2` FOREIGN KEY (`Company_Name`) REFERENCES `Courier_Company` (`Company_Name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
