-- Schema for database: rschema_0391
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `admission_date` datetime NOT NULL,
  `gender` varchar(10) NOT NULL,
  `major` varchar(100) NOT NULL,
  `name` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `post` (
  `post_id` int(11) NOT NULL,
  `author` int(11) NOT NULL,
  `content` text NOT NULL,
  `publication_date` datetime NOT NULL,
  `topic` varchar(100) NOT NULL,
  PRIMARY KEY (`post_id`),
  CONSTRAINT `post_ibfk_1` FOREIGN KEY (`author`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
  `commented_post` int(11) NOT NULL,
  `commenter` int(11) NOT NULL,
  `comment_content` text NOT NULL,
  `comment_time` datetime NOT NULL,
  PRIMARY KEY (`commented_post`, `commenter`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`commented_post`) REFERENCES `post` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`commenter`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
