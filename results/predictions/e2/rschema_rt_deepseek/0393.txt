-- Schema for database: rschema_0393
CREATE TABLE `patient` (
  `patient_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `gender` bool NOT NULL,
  `treatment_information` varchar(255) DEFAULT NULL,
  `registration_category_id` int(11) DEFAULT NULL,
  `ward_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`patient_id`),
  CONSTRAINT `patient_ibfk_1` FOREIGN KEY (`registration_category_id`) REFERENCES `registration_category` (`registration_category_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patient_ibfk_2` FOREIGN KEY (`ward_id`) REFERENCES `ward` (`ward_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ward` (
  `ward_id` int(11) NOT NULL,
  `ward_information` varchar(255) NOT NULL,
  PRIMARY KEY (`ward_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `registration_category` (
  `registration_category_id` int(11) NOT NULL,
  `registration_information` varchar(255) NOT NULL,
  PRIMARY KEY (`registration_category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `drug` (
  `drug_id` int(11) NOT NULL,
  `drug_information` varchar(255) NOT NULL,
  PRIMARY KEY (`drug_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `prescription` (
  `drug_id` int(11) NOT NULL,
  `patient_id` int(11) NOT NULL,
  PRIMARY KEY (`drug_id`, `patient_id`),
  CONSTRAINT `prescription_ibfk_1` FOREIGN KEY (`drug_id`) REFERENCES `drug` (`drug_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `prescription_ibfk_2` FOREIGN KEY (`patient_id`) REFERENCES `patient` (`patient_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
