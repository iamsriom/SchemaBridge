-- Schema for database: rschema_0395
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meal_card` (
  `card_id` int(11) NOT NULL,
  `balance` float DEFAULT NULL,
  `student_id` int(11) NOT NULL,
  PRIMARY KEY (`card_id`),
  CONSTRAINT `meal_card_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consumption_record` (
  `record_id` int(11) NOT NULL,
  `amount` int(11) DEFAULT NULL,
  `card_id` int(11) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `time` datetime DEFAULT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `consumption_record_ibfk_1` FOREIGN KEY (`card_id`) REFERENCES `meal_card` (`card_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
