-- Schema for database: rschema_0401
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `grade` varchar(50) NOT NULL,
  `department` int(11) NOT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `enrollment_time` datetime DEFAULT NULL,
  PRIMARY KEY (`student_id`),
  CONSTRAINT `students_ibfk_department` FOREIGN KEY (`department`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classes` (
  `class_id` int(11) NOT NULL,
  `class_name` varchar(255) NOT NULL,
  `grade` varchar(50) NOT NULL,
  `major` varchar(255) DEFAULT NULL,
  `head_teacher` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  `class_monitor` int(11) NOT NULL,
  PRIMARY KEY (`class_id`),
  CONSTRAINT `classes_ibfk_head_teacher` FOREIGN KEY (`head_teacher`) REFERENCES `teachers` (`teacher_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `classes_ibfk_department` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `classes_ibfk_class_monitor` FOREIGN KEY (`class_monitor`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `department_size` int(11) DEFAULT NULL,
  `subordinate_majors` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teachers` (
  `teacher_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `grade` varchar(50) NOT NULL,
  `department` int(11) NOT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `entry_time` datetime DEFAULT NULL,
  PRIMARY KEY (`teacher_id`),
  CONSTRAINT `teachers_ibfk_department` FOREIGN KEY (`department`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
