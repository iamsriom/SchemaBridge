-- Schema for database: rschema_0415
CREATE TABLE `user_table` (
  `user_id` int(11) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `time_of_users_last_login` datetime DEFAULT NULL,
  `user_ip_address` varchar(255) DEFAULT NULL,
  `user_port_number` int(11) DEFAULT NULL,
  `username` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `server_configuration_table` (
  `server_id` int(11) NOT NULL,
  `maximum_client_connections` int(11) DEFAULT NULL,
  `protocol_used` varchar(255) DEFAULT NULL,
  `receive_buffer_size` int(11) DEFAULT NULL,
  `server_ip_address` varchar(255) DEFAULT NULL,
  `server_port_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`server_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chat_record_table` (
  `message_id` int(11) NOT NULL,
  `message_content_and_sending_time` varchar(255) NOT NULL,
  `receiver_id` int(11) NOT NULL,
  `sender_id` int(11) NOT NULL,
  `user_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`message_id`),
  CONSTRAINT `chat_record_table_ibfk_1` FOREIGN KEY (`receiver_id`) REFERENCES `user_table` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `chat_record_table_ibfk_2` FOREIGN KEY (`sender_id`) REFERENCES `user_table` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
