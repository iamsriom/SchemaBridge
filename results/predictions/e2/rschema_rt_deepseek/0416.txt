-- Schema for database: rschema_0416
CREATE TABLE `User` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  `nickname` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `avatar` varchar(255) DEFAULT NULL,
  `registration_time` datetime DEFAULT NULL,
  `role` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Company` (
  `company_id` int(11) NOT NULL,
  `company_name` varchar(255) DEFAULT NULL,
  `company_address` varchar(255) DEFAULT NULL,
  `company_profile` text DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`company_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Conversation` (
  `conversation_id` int(11) NOT NULL,
  PRIMARY KEY (`conversation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Message_Board` (
  `message_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `content` text DEFAULT NULL,
  `timestamp` datetime DEFAULT NULL,
  PRIMARY KEY (`message_id`),
  CONSTRAINT `message_board_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Job_Info` (
  `job_info_id` int(11) NOT NULL,
  `company_id` int(11) NOT NULL,
  `title` varchar(255) DEFAULT NULL,
  `content` text DEFAULT NULL,
  `requirements` text DEFAULT NULL,
  `salary` decimal(10,2) DEFAULT NULL,
  `education_level` varchar(50) DEFAULT NULL,
  `job_category` varchar(50) DEFAULT NULL,
  `work_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`job_info_id`),
  CONSTRAINT `job_info_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `Company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_and_Conversation` (
  `user_id` int(11) NOT NULL,
  `conversation_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`, `conversation_id`),
  CONSTRAINT `user_and_conversation_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_and_conversation_ibfk_2` FOREIGN KEY (`conversation_id`) REFERENCES `Conversation` (`conversation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_and_Message` (
  `user_id` int(11) NOT NULL,
  `message_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`, `message_id`),
  CONSTRAINT `user_and_message_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_and_message_ibfk_2` FOREIGN KEY (`message_id`) REFERENCES `Message_Board` (`message_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Resume_Submission` (
  `submission_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `job_info_id` int(11) NOT NULL,
  PRIMARY KEY (`submission_id`),
  CONSTRAINT `resume_submission_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resume_submission_ibfk_2` FOREIGN KEY (`job_info_id`) REFERENCES `Job_Info` (`job_info_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Resume_Save` (
  `save_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `job_info_id` int(11) NOT NULL,
  PRIMARY KEY (`save_id`),
  CONSTRAINT `resume_save_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resume_save_ibfk_2` FOREIGN KEY (`job_info_id`) REFERENCES `Job_Info` (`job_info_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Company_and_Job_Info` (
  `company_id` int(11) NOT NULL,
  `job_info_id` int(11) NOT NULL,
  PRIMARY KEY (`company_id`, `job_info_id`),
  CONSTRAINT `company_and_job_info_ibfk_1` FOREIGN KEY (`company_id`) REFERENCES `Company` (`company_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `company_and_job_info_ibfk_2` FOREIGN KEY (`job_info_id`) REFERENCES `Job_Info` (`job_info_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
