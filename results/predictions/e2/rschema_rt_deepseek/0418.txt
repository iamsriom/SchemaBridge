-- Schema for database: rschema_0418
CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Login_Name` varchar(255) DEFAULT NULL,
  `Password` varchar(255) DEFAULT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Age` int(11) DEFAULT NULL,
  `Gender` varchar(10) DEFAULT NULL,
  `Security_Question` varchar(255) DEFAULT NULL,
  `Answer` varchar(255) DEFAULT NULL,
  `Status` varchar(50) DEFAULT NULL,
  `User_Type` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Announcement` (
  `Announcement_ID` int(11) NOT NULL,
  `Title` varchar(255) DEFAULT NULL,
  `Content` text DEFAULT NULL,
  `Publication_Time` datetime DEFAULT NULL,
  `Status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`Announcement_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Type` (
  `Type_ID` int(11) NOT NULL,
  `Type_Name` varchar(255) DEFAULT NULL,
  `Status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`Type_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Income_Expense_Record` (
  `Record_ID` int(11) NOT NULL,
  `Record_Name` varchar(255) DEFAULT NULL,
  `Amount` decimal(10,2) DEFAULT NULL,
  `Time` datetime DEFAULT NULL,
  `Remark` text DEFAULT NULL,
  `Status` varchar(50) DEFAULT NULL,
  `Type_ID` int(11) NOT NULL,
  `User_ID` int(11) NOT NULL,
  PRIMARY KEY (`Record_ID`),
  CONSTRAINT `Income_Expense_Record_ibfk_1` FOREIGN KEY (`Type_ID`) REFERENCES `Type` (`Type_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Income_Expense_Record_ibfk_2` FOREIGN KEY (`User_ID`) REFERENCES `User` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
