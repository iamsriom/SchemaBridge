-- Schema for database: rschema_0422
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `phone` varchar(255) DEFAULT NULL,
  `registration_time` datetime DEFAULT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `role` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) DEFAULT NULL,
  `introduction` varchar(255) DEFAULT NULL,
  `creation_time` datetime DEFAULT NULL,
  `project_status` varchar(255) DEFAULT NULL,
  `responsible_person_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_version` (
  `version_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `version_name` varchar(255) DEFAULT NULL,
  `version_description` varchar(255) DEFAULT NULL,
  `release_date` date DEFAULT NULL,
  `is_stable_version` tinyint(1) DEFAULT NULL,
  PRIMARY KEY (`version_id`, `project_id`),
  CONSTRAINT `project_version_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `code_submission` (
  `user_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `submission_time` datetime NOT NULL,
  `code_path` varchar(255) DEFAULT NULL,
  `submission_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`, `project_id`, `submission_time`),
  CONSTRAINT `code_submission_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `code_submission_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task` (
  `project_id` int(11) NOT NULL,
  `assigned_to_user_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `task_name` varchar(255) DEFAULT NULL,
  `task_description` varchar(255) DEFAULT NULL,
  `task_status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`project_id`, `assigned_to_user_id`, `start_time`),
  CONSTRAINT `task_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_ibfk_2` FOREIGN KEY (`assigned_to_user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
