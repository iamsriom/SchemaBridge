-- Schema for database: rschema_0441
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `permission` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `budget` float NOT NULL,
  `remaining_funds` float NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `accounting_record` (
  `record_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `transaction_operator` int(11) NOT NULL,
  `purpose_description` varchar(255) NOT NULL,
  `record_time` datetime NOT NULL,
  `transaction_amount` float NOT NULL,
  PRIMARY KEY (`record_id`),
  CONSTRAINT `accounting_record_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `accounting_record_ibfk_2` FOREIGN KEY (`transaction_operator`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
