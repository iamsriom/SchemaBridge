-- Schema for database: rschema_0443
CREATE TABLE `user_information` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `registered_phone_number` varchar(20) DEFAULT NULL,
  `role_permissions` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `asset_information` (
  `device_id` int(11) NOT NULL,
  `device_name` varchar(255) NOT NULL,
  `device_number` varchar(50) NOT NULL,
  `device_model` varchar(255) DEFAULT NULL,
  `device_status` varchar(50) DEFAULT NULL,
  `calibration_date` datetime DEFAULT NULL,
  `calibration_validity` datetime DEFAULT NULL,
  `calibration_result` varchar(255) DEFAULT NULL,
  `device_location` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  `responsible_person` varchar(255) DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`device_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status_log_information` (
  `log_id` int(11) NOT NULL,
  `device_number` varchar(50) NOT NULL,
  `borrow_receiver` varchar(255) DEFAULT NULL,
  `borrow_time` datetime DEFAULT NULL,
  `borrow_device_photo` varchar(255) DEFAULT NULL,
  `iou_photo` varchar(255) DEFAULT NULL,
  `receiver` varchar(255) DEFAULT NULL,
  `receiving_area` varchar(255) DEFAULT NULL,
  `receiving_time` datetime DEFAULT NULL,
  `receiving_device_photo` varchar(255) DEFAULT NULL,
  `return_person` varchar(255) DEFAULT NULL,
  `return_time` datetime DEFAULT NULL,
  `return_device_photo` varchar(255) DEFAULT NULL,
  `data_monitoring` text DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`log_id`),
  CONSTRAINT `status_log_information_ibfk_1` FOREIGN KEY (`device_number`) REFERENCES `asset_information` (`device_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account_dynamic_log_information` (
  `dynamic_log_id` int(11) NOT NULL,
  `device_number` varchar(50) NOT NULL,
  `borrow_operation_account` varchar(255) DEFAULT NULL,
  `receive_operation_account` varchar(255) DEFAULT NULL,
  `return_operation_account` varchar(255) DEFAULT NULL,
  `remarks` text DEFAULT NULL,
  PRIMARY KEY (`dynamic_log_id`),
  CONSTRAINT `account_dynamic_log_information_ibfk_1` FOREIGN KEY (`device_number`) REFERENCES `asset_information` (`device_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_asset_relationship` (
  `user_id` int(11) NOT NULL,
  `device_number` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`, `device_number`),
  CONSTRAINT `user_asset_relationship_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user_information` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_asset_relationship_ibfk_2` FOREIGN KEY (`device_number`) REFERENCES `asset_information` (`device_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `device_number_log_id` (
  `device_number` varchar(50) NOT NULL,
  `log_id` int(11) NOT NULL,
  PRIMARY KEY (`device_number`, `log_id`),
  CONSTRAINT `device_number_log_id_ibfk_1` FOREIGN KEY (`log_id`) REFERENCES `status_log_information` (`log_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `device_number_log_id_ibfk_2` FOREIGN KEY (`device_number`) REFERENCES `asset_information` (`device_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `asset_account_dynamic_log_relationship` (
  `device_number` varchar(50) NOT NULL,
  `dynamic_log_id` int(11) NOT NULL,
  PRIMARY KEY (`device_number`, `dynamic_log_id`),
  CONSTRAINT `asset_account_dynamic_log_relationship_ibfk_1` FOREIGN KEY (`dynamic_log_id`) REFERENCES `account_dynamic_log_information` (`dynamic_log_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `asset_account_dynamic_log_relationship_ibfk_2` FOREIGN KEY (`device_number`) REFERENCES `asset_information` (`device_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
