-- Schema for database: rschema_0447
CREATE TABLE `user` (
  `user_id` varchar(255) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `phone_number` varchar(255) DEFAULT NULL,
  `role` varchar(255) DEFAULT NULL,
  `associated_unit` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `financial_document` (
  `document_id` varchar(255) NOT NULL,
  `document_title` varchar(255) DEFAULT NULL,
  `document_number` varchar(255) DEFAULT NULL,
  `document_content` varchar(255) DEFAULT NULL,
  `document_category` varchar(255) DEFAULT NULL,
  `document_level` varchar(255) DEFAULT NULL,
  `issuing_unit` varchar(255) DEFAULT NULL,
  `issue_date` varchar(255) DEFAULT NULL,
  `remarks` varchar(255) DEFAULT NULL,
  `validity_status` varchar(255) DEFAULT NULL,
  `attachment_path` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`document_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `notification` (
  `notification_id` varchar(255) NOT NULL,
  `notification_title` varchar(255) DEFAULT NULL,
  `notification_content` varchar(255) DEFAULT NULL,
  `issue_date` varchar(255) DEFAULT NULL,
  `notification_status` varchar(255) DEFAULT NULL,
  `user_id` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`notification_id`),
  CONSTRAINT `notification_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `system_configuration` (
  `configuration_id` varchar(255) NOT NULL,
  `system_name` varchar(255) DEFAULT NULL,
  `data_type` varchar(255) DEFAULT NULL,
  `database_directory` varchar(255) DEFAULT NULL,
  `file_upload_directory` varchar(255) DEFAULT NULL,
  `admin_id` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`configuration_id`),
  CONSTRAINT `system_configuration_ibfk_1` FOREIGN KEY (`admin_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `document_management` (
  `admin_id` varchar(255) NOT NULL,
  `document_id` varchar(255) NOT NULL,
  `operation_time` varchar(255) DEFAULT NULL,
  `operation_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`admin_id`, `document_id`),
  CONSTRAINT `document_management_ibfk_1` FOREIGN KEY (`admin_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `document_management_ibfk_2` FOREIGN KEY (`document_id`) REFERENCES `financial_document` (`document_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
