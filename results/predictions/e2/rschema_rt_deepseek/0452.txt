-- Schema for database: rschema_0452
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `role` varchar(50) NOT NULL,
  `account_creation_time` datetime NOT NULL,
  `account_update_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `student_name` varchar(255) NOT NULL,
  `major` varchar(100) NOT NULL,
  `information_creation_time` datetime NOT NULL,
  `information_update_time` datetime DEFAULT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`student_id`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `certificate` (
  `certificate_id` int(11) NOT NULL,
  `certificate_name` varchar(255) NOT NULL,
  `issuing_institution` varchar(255) NOT NULL,
  `issuance_date` date NOT NULL,
  `certificate_description` text DEFAULT NULL,
  `certificate_status` varchar(50) NOT NULL,
  `information_creation_time` datetime NOT NULL,
  `information_update_time` datetime DEFAULT NULL,
  `hash_value` varchar(255) NOT NULL,
  `associated_student` int(11) NOT NULL,
  PRIMARY KEY (`certificate_id`),
  CONSTRAINT `certificate_ibfk_1` FOREIGN KEY (`associated_student`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
