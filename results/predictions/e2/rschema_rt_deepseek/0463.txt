-- Schema for database: rschema_0463
CREATE TABLE `Mall` (
  `Mall_ID` int(11) NOT NULL,
  `Mall_Name` varchar(255) NOT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Contact_Number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`Mall_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Department` (
  `Department_ID` int(11) NOT NULL,
  `Department_Name` varchar(255) NOT NULL,
  `Location_Distribution` varchar(255) DEFAULT NULL,
  `Contact_Number` varchar(15) DEFAULT NULL,
  `Mall_ID` int(11) NOT NULL,
  `Manager_Employee_ID` int(11) DEFAULT NULL,
  `Tenure` int(11) DEFAULT NULL,
  PRIMARY KEY (`Department_ID`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`Mall_ID`) REFERENCES `Mall` (`Mall_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `department_ibfk_2` FOREIGN KEY (`Manager_Employee_ID`) REFERENCES `Employee` (`Employee_ID`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Employee` (
  `Employee_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Phone_Number` varchar(15) DEFAULT NULL,
  `Position` varchar(255) DEFAULT NULL,
  `Salary` decimal(10,2) DEFAULT NULL,
  `Emergency_Contact_Name` varchar(255) DEFAULT NULL,
  `Emergency_Contact_Phone` varchar(15) DEFAULT NULL,
  `Department_ID` int(11) NOT NULL,
  PRIMARY KEY (`Employee_ID`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Department_ID`) REFERENCES `Department` (`Department_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
