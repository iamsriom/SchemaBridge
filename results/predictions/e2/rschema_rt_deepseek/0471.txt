-- Schema for database: rschema_0471
CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Contact_number` varchar(20) DEFAULT NULL,
  `Application_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Credit_card` (
  `Card_number` varchar(20) NOT NULL,
  `Customer_ID` int(11) NOT NULL,
  `Type` varchar(50) DEFAULT NULL,
  `Activation_status` varchar(50) DEFAULT NULL,
  `Credit_limit` decimal(10,2) DEFAULT NULL,
  `Validity_period` date DEFAULT NULL,
  PRIMARY KEY (`Card_number`),
  CONSTRAINT `Credit_card_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Transaction_record` (
  `Transaction_ID` int(11) NOT NULL,
  `Card_number` varchar(20) NOT NULL,
  `Transaction_amount` decimal(10,2) DEFAULT NULL,
  `Transaction_date` datetime DEFAULT NULL,
  `Transaction_description` varchar(255) DEFAULT NULL,
  `Transaction_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`Transaction_ID`),
  CONSTRAINT `Transaction_record_ibfk_1` FOREIGN KEY (`Card_number`) REFERENCES `Credit_card` (`Card_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
