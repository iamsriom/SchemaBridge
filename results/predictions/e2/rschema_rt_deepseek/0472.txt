-- Schema for database: rschema_0472
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone` varchar(255) DEFAULT NULL,
  `class` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `instructor` (
  `instructor_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `course` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`instructor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `homework` (
  `homework_id` int(11) NOT NULL,
  `homework_content` varchar(255) DEFAULT NULL,
  `grading_status` varchar(255) DEFAULT NULL,
  `submission_time` datetime DEFAULT NULL,
  `submitting_student_id` int(11) NOT NULL,
  PRIMARY KEY (`homework_id`),
  CONSTRAINT `homework_ibfk_1` FOREIGN KEY (`submitting_student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
