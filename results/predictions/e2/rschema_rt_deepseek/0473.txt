-- Schema for database: rschema_0473
CREATE TABLE `Publication` (
  `Publication_ID` int(11) NOT NULL,
  `Title` varchar(255) NOT NULL,
  `Author` varchar(255) NOT NULL,
  `Price` float NOT NULL,
  `Type` varchar(255) NOT NULL,
  PRIMARY KEY (`Publication_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer` (
  `Customer_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Phone` varchar(255) NOT NULL,
  `Registration_Status` varchar(255) NOT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Administrator` (
  `Administrator_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Email` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Phone` varchar(255) NOT NULL,
  PRIMARY KEY (`Administrator_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Order` (
  `Customer_ID` int(11) NOT NULL,
  `Publication_ID` int(11) NOT NULL,
  `Order_Time` datetime NOT NULL,
  `Order_Status` varchar(255) NOT NULL,
  `Payment_Method` varchar(255) NOT NULL,
  `Shipping_Address` varchar(255) NOT NULL,
  PRIMARY KEY (`Customer_ID`, `Publication_ID`, `Order_Time`),
  CONSTRAINT `order_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_ibfk_2` FOREIGN KEY (`Publication_ID`) REFERENCES `Publication` (`Publication_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
