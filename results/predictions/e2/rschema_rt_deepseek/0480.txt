-- Schema for database: rschema_0480
CREATE TABLE `Property` (
  `Property_ID` int(11) NOT NULL,
  `Address` varchar(255) DEFAULT NULL,
  `Built_area` float DEFAULT NULL,
  `Construction_year` int(11) DEFAULT NULL,
  `Is_apartment` tinyint(1) DEFAULT NULL,
  `Property_certificate` varchar(255) DEFAULT NULL,
  `Property_photos` varchar(255) DEFAULT NULL,
  `Sale_price` decimal(10,2) DEFAULT NULL,
  `Sales_status` varchar(50) DEFAULT 'For Sale',
  PRIMARY KEY (`Property_ID`),
  UNIQUE KEY `Property_certificate` (`Property_certificate`),
  UNIQUE KEY `Property_photos` (`Property_photos`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Agent` (
  `Agent_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Contact_number` varchar(50) DEFAULT NULL,
  `Home_address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Agent_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `User_ID` int(11) NOT NULL,
  `Username` varchar(255) DEFAULT NULL,
  `Password` varchar(255) DEFAULT NULL,
  `Role` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Responsibility` (
  `Agent_ID` int(11) NOT NULL,
  `Property_ID` int(11) NOT NULL,
  `Responsibility_start_time` datetime DEFAULT NULL,
  `Responsibility_end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`Agent_ID`, `Property_ID`),
  CONSTRAINT `Responsibility_ibfk_1` FOREIGN KEY (`Agent_ID`) REFERENCES `Agent` (`Agent_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Responsibility_ibfk_2` FOREIGN KEY (`Property_ID`) REFERENCES `Property` (`Property_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
