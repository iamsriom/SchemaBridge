-- Schema for database: rschema_0493
CREATE TABLE `enterprise` (
  `Enterprise_ID` int(11) NOT NULL,
  `Enterprise_Name` varchar(255) NOT NULL,
  PRIMARY KEY (`Enterprise_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role` (
  `Role_ID` int(11) NOT NULL,
  `Daily_Working_Hours` float DEFAULT NULL,
  `Hourly_Wage` float DEFAULT NULL,
  `Lunch_Time` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Role_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `Employee_ID` int(11) NOT NULL,
  `Creation_Time` datetime DEFAULT NULL,
  `Email` varchar(255) DEFAULT NULL,
  `Enterprise_ID` int(11) NOT NULL,
  `Identification_Number` varchar(255) DEFAULT NULL,
  `Name` varchar(255) NOT NULL,
  `Password` varchar(255) NOT NULL,
  `Role_ID` int(11) NOT NULL,
  `Update_Time` datetime DEFAULT NULL,
  PRIMARY KEY (`Employee_ID`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Enterprise_ID`) REFERENCES `enterprise` (`Enterprise_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`Role_ID`) REFERENCES `role` (`Role_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendance_record` (
  `Employee_ID` int(11) NOT NULL,
  `Check_in_Time` datetime NOT NULL,
  PRIMARY KEY (`Employee_ID`, `Check_in_Time`),
  CONSTRAINT `attendance_record_ibfk_1` FOREIGN KEY (`Employee_ID`) REFERENCES `employee` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `to_do_item` (
  `Employee_ID` int(11) NOT NULL,
  `Creation_Time` datetime NOT NULL,
  PRIMARY KEY (`Employee_ID`, `Creation_Time`),
  CONSTRAINT `to_do_item_ibfk_1` FOREIGN KEY (`Employee_ID`) REFERENCES `employee` (`Employee_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
