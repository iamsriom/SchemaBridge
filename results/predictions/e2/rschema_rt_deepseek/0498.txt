-- Schema for database: rschema_0498
CREATE TABLE `student_table` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `dormitory_id` int(11) DEFAULT NULL,
  `class_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`student_id`),
  CONSTRAINT `student_table_ibfk_1` FOREIGN KEY (`dormitory_id`) REFERENCES `dormitory_table` (`dormitory_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_table_ibfk_2` FOREIGN KEY (`class_id`) REFERENCES `class_table` (`class_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `class_table` (
  `class_id` int(11) NOT NULL,
  `class_teacher` varchar(255) DEFAULT NULL,
  `number_of_students` int(11) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`class_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sports_team_table` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) DEFAULT NULL,
  `activity_time` varchar(255) DEFAULT NULL,
  `coach` varchar(255) DEFAULT NULL,
  `activity_venue` varchar(255) DEFAULT NULL,
  `affiliated_school` int(11) DEFAULT NULL,
  PRIMARY KEY (`team_id`),
  CONSTRAINT `sports_team_table_ibfk_1` FOREIGN KEY (`affiliated_school`) REFERENCES `school_table` (`school_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `school_table` (
  `school_id` int(11) NOT NULL,
  `school_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`school_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `dormitory_table` (
  `dormitory_id` int(11) NOT NULL,
  `building_number` varchar(255) DEFAULT NULL,
  `administrator` varchar(255) DEFAULT NULL,
  `dormitory_capacity` int(11) DEFAULT NULL,
  PRIMARY KEY (`dormitory_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
