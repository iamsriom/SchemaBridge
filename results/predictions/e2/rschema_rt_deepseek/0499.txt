-- Schema for database: rschema_0499
CREATE TABLE `Class` (
  `Class_ID` int(11) NOT NULL,
  `Class_name` varchar(255) DEFAULT NULL,
  `Classroom` varchar(255) DEFAULT NULL,
  `Major` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Class_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Sports_team` (
  `Team_ID` int(11) NOT NULL,
  `Activity_time` varchar(255) DEFAULT NULL,
  `Coach` varchar(255) DEFAULT NULL,
  `Team_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Team_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Dormitory` (
  `Building_number` int(11) NOT NULL,
  `Dormitory_number` int(11) NOT NULL,
  `Number_of_residents` int(11) DEFAULT NULL,
  PRIMARY KEY (`Building_number`, `Dormitory_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Student` (
  `Student_ID` int(11) NOT NULL,
  `Age` int(11) DEFAULT NULL,
  `Building_number` int(11) NOT NULL,
  `Class_ID` int(11) NOT NULL,
  `Dormitory_number` int(11) NOT NULL,
  `Gender` tinyint(1) DEFAULT NULL,
  `Name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Student_ID`),
  CONSTRAINT `student_ibfk_1` FOREIGN KEY (`Building_number`, `Dormitory_number`) REFERENCES `Dormitory` (`Building_number`, `Dormitory_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_ibfk_2` FOREIGN KEY (`Class_ID`) REFERENCES `Class` (`Class_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Team_membership` (
  `Student_ID` int(11) NOT NULL,
  `Team_ID` int(11) NOT NULL,
  PRIMARY KEY (`Student_ID`, `Team_ID`),
  CONSTRAINT `team_membership_ibfk_1` FOREIGN KEY (`Student_ID`) REFERENCES `Student` (`Student_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `team_membership_ibfk_2` FOREIGN KEY (`Team_ID`) REFERENCES `Sports_team` (`Team_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
