-- Schema for database: rschema_0501
CREATE TABLE `Aircraft` (
  `aircraft_id` int(11) NOT NULL,
  `affiliated_airline` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `manufacture_date` date NOT NULL,
  `max_passenger_capacity` int(11) NOT NULL,
  PRIMARY KEY (`aircraft_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Crew_Member` (
  `crew_member_id` int(11) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`crew_member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Passenger` (
  `passenger_id` int(11) NOT NULL,
  `identification_number` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  PRIMARY KEY (`passenger_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Flight` (
  `flight_id` int(11) NOT NULL,
  `aircraft_id` int(11) NOT NULL,
  `departure_airport` varchar(255) NOT NULL,
  `departure_time` datetime NOT NULL,
  `landing_airport` varchar(255) NOT NULL,
  `landing_time` datetime NOT NULL,
  `captain_id` int(11) NOT NULL,
  PRIMARY KEY (`flight_id`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`aircraft_id`) REFERENCES `Aircraft` (`aircraft_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_2` FOREIGN KEY (`captain_id`) REFERENCES `Crew_Member` (`crew_member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Maintenance_Record` (
  `maintenance_id` int(11) NOT NULL,
  `aircraft_id` int(11) NOT NULL,
  `maintenance_date` date NOT NULL,
  `description` text NOT NULL,
  `personnel_id` int(11) NOT NULL,
  `type` varchar(100) NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`aircraft_id`) REFERENCES `Aircraft` (`aircraft_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_record_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `Crew_Member` (`crew_member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Ticket_Order` (
  `ticket_order_id` int(11) NOT NULL,
  `booking_time` datetime NOT NULL,
  `flight_id` int(11) NOT NULL,
  `passenger_id` int(11) NOT NULL,
  `seat_number` varchar(10) NOT NULL,
  `ticket_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`ticket_order_id`),
  CONSTRAINT `ticket_order_ibfk_1` FOREIGN KEY (`flight_id`) REFERENCES `Flight` (`flight_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ticket_order_ibfk_2` FOREIGN KEY (`passenger_id`) REFERENCES `Passenger` (`passenger_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
