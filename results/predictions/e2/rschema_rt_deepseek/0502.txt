-- Schema for database: rschema_0502
CREATE TABLE `venue` (
  `venue_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area_size` int(11) NOT NULL,
  `maximum_capacity` int(11) NOT NULL,
  `type` varchar(100) NOT NULL,
  `status` varchar(50) NOT NULL,
  `creation_time` datetime NOT NULL,
  `last_maintenance_time` datetime DEFAULT NULL,
  PRIMARY KEY (`venue_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `equipment_id` int(11) NOT NULL,
  `equipment_name` varchar(255) NOT NULL,
  `last_maintenance_time` datetime DEFAULT NULL,
  `start_usage_time` datetime DEFAULT NULL,
  `venue_id` int(11) NOT NULL,
  PRIMARY KEY (`equipment_id`),
  CONSTRAINT `equipment_ibfk_1` FOREIGN KEY (`venue_id`) REFERENCES `venue` (`venue_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `venue_reservation` (
  `user_id` int(11) NOT NULL,
  `venue_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `reservation_status` varchar(50) NOT NULL,
  `reservation_success_time` datetime NOT NULL,
  `reservation_cancellation_time` datetime DEFAULT NULL,
  PRIMARY KEY (`user_id`, `venue_id`, `reservation_success_time`),
  CONSTRAINT `venue_reservation_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `venue_reservation_ibfk_2` FOREIGN KEY (`venue_id`) REFERENCES `venue` (`venue_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `venue_evaluation` (
  `user_id` int(11) NOT NULL,
  `venue_id` int(11) NOT NULL,
  `evaluation_time` datetime NOT NULL,
  `rating` int(11) NOT NULL,
  `evaluation_content` text NOT NULL,
  PRIMARY KEY (`user_id`, `venue_id`, `evaluation_time`),
  CONSTRAINT `venue_evaluation_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `venue_evaluation_ibfk_2` FOREIGN KEY (`venue_id`) REFERENCES `venue` (`venue_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
