-- Schema for database: rschema_0503
CREATE TABLE `material` (
  `material_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `source_location` varchar(255) DEFAULT NULL,
  `specification` varchar(255) DEFAULT NULL,
  `type` varchar(255) DEFAULT NULL,
  `volume` float DEFAULT NULL,
  `weight` float DEFAULT NULL,
  PRIMARY KEY (`material_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `warehouse` (
  `warehouse_id` int(11) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `current_inventory_level` int(11) DEFAULT NULL,
  `maximum_capacity` int(11) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `remaining_capacity` int(11) DEFAULT NULL,
  PRIMARY KEY (`warehouse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `driver` (
  `driver_id` int(11) NOT NULL,
  `age` int(11) DEFAULT NULL,
  `driving_license_number` varchar(255) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`driver_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_storage` (
  `material_id` int(11) NOT NULL,
  `warehouse_id` int(11) NOT NULL,
  `storage_time` datetime NOT NULL,
  `notes` varchar(255) DEFAULT NULL,
  `storage_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`material_id`, `warehouse_id`, `storage_time`),
  CONSTRAINT `material_storage_ibfk_1` FOREIGN KEY (`material_id`) REFERENCES `material` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_storage_ibfk_2` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_dispatch` (
  `material_id` int(11) NOT NULL,
  `warehouse_id` int(11) NOT NULL,
  `dispatch_time` datetime NOT NULL,
  `notes` varchar(255) DEFAULT NULL,
  `dispatch_quantity` int(11) DEFAULT NULL,
  PRIMARY KEY (`material_id`, `warehouse_id`, `dispatch_time`),
  CONSTRAINT `material_dispatch_ibfk_1` FOREIGN KEY (`material_id`) REFERENCES `material` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_dispatch_ibfk_2` FOREIGN KEY (`warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL,
  `license_plate_number` varchar(20) DEFAULT NULL,
  `maximum_load` float DEFAULT NULL,
  `current_load` float DEFAULT NULL,
  `vehicle_type` varchar(255) DEFAULT NULL,
  `assigned_driver_id` int(11) DEFAULT NULL,
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`assigned_driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `material_transportation` (
  `material_id` int(11) NOT NULL,
  `driver_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `destination` varchar(255) DEFAULT NULL,
  `transportation_start_time` datetime DEFAULT NULL,
  CONSTRAINT `material_transportation_ibfk_1` FOREIGN KEY (`material_id`) REFERENCES `material` (`material_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_transportation_ibfk_2` FOREIGN KEY (`driver_id`) REFERENCES `driver` (`driver_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `material_transportation_ibfk_3` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
