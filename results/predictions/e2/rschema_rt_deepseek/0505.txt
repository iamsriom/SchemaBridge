-- Schema for database: rschema_0505
CREATE TABLE `User` (
  `Account` varchar(255) NOT NULL,
  `Password` varchar(255) DEFAULT NULL,
  `Role` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Account`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Student` (
  `Student_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Major` varchar(255) DEFAULT NULL,
  `Average_Grade` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`Student_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Teacher` (
  `Teacher_ID` int(11) NOT NULL,
  `Name` varchar(255) DEFAULT NULL,
  `Age` int(11) DEFAULT NULL,
  `Gender` varchar(10) DEFAULT NULL,
  `Job_Title` varchar(255) DEFAULT NULL,
  `Contact_Information` varchar(255) DEFAULT NULL,
  `Work_History` text DEFAULT NULL,
  PRIMARY KEY (`Teacher_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Log_Information` (
  `Detailed_Description` text DEFAULT NULL,
  `Operation_Time` datetime DEFAULT NULL,
  `Operation_Type` varchar(255) DEFAULT NULL,
  `User_Account` varchar(255) DEFAULT NULL,
  CONSTRAINT `log_information_ibfk_1` FOREIGN KEY (`User_Account`) REFERENCES `User` (`Account`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Research_Project` (
  `Project_ID` int(11) NOT NULL,
  `Project_Name` varchar(255) DEFAULT NULL,
  `Funding` decimal(15,2) DEFAULT NULL,
  `Outcome` text DEFAULT NULL,
  `Responsible_Teacher` int(11) DEFAULT NULL,
  PRIMARY KEY (`Project_ID`),
  CONSTRAINT `research_project_ibfk_1` FOREIGN KEY (`Responsible_Teacher`) REFERENCES `Teacher` (`Teacher_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Teacher_Responsible_for_Research_Project` (
  `Project_ID` int(11) NOT NULL,
  `Teacher_ID` int(11) NOT NULL,
  CONSTRAINT `teacher_responsible_for_research_project_ibfk_1` FOREIGN KEY (`Project_ID`) REFERENCES `Research_Project` (`Project_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `teacher_responsible_for_research_project_ibfk_2` FOREIGN KEY (`Teacher_ID`) REFERENCES `Teacher` (`Teacher_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
