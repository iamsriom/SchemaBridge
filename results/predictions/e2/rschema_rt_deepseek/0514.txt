-- Schema for database: rschema_0514
CREATE TABLE `Customer` (
  `Account_Opening_Date` DATETIME NOT NULL,
  `Contact_Number` VARCHAR(255) NOT NULL,
  `Customer_ID` INT(11) NOT NULL,
  `Email` VARCHAR(255) NOT NULL,
  `Name` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`Customer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `Account` (
  `Account_ID` INT(11) NOT NULL,
  `Balance` VARCHAR(255) DEFAULT NULL,
  `Customer_ID` INT(11) DEFAULT NULL,
  PRIMARY KEY (`Account_ID`),
  CONSTRAINT `account_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `Transaction` (
  `Customer_ID` INT(11) DEFAULT NULL,
  `Transaction_Amount` INT(11) NOT NULL,
  `Transaction_ID` INT(11) NOT NULL,
  `Transaction_Status` VARCHAR(255) DEFAULT NULL,
  `Transaction_Time` DATETIME NOT NULL,
  `Transaction_Type` VARCHAR(255) DEFAULT NULL,
  PRIMARY KEY (`Transaction_ID`),
  CONSTRAINT `transaction_ibfk_1` FOREIGN KEY (`Customer_ID`) REFERENCES `Customer` (`Customer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
