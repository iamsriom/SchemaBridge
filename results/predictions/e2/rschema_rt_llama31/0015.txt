-- Schema for database: rschema_0015
CREATE TABLE `plot` (
  `plot_number` int(11) NOT NULL,
  `area` float NOT NULL,
  `soil_type` varchar(255) NOT NULL,
  `pest_and_disease_status` varchar(255) DEFAULT NULL,
  `geographic_location` varchar(255) NOT NULL,
  PRIMARY KEY (`plot_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pest_and_disease` (
  `pest_and_disease_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `degree_of_harm` varchar(255) NOT NULL,
  `prevention_and_control_methods` varchar(255) NOT NULL,
  `season_of_occurrence` varchar(255) NOT NULL,
  PRIMARY KEY (`pest_and_disease_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `age` int(11) NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_record` (
  `plot_number` int(11) NOT NULL,
  `pest_and_disease_number` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `monitoring_results` varchar(255) DEFAULT NULL,
  `employee_number` int(11) NOT NULL,
  PRIMARY KEY (`plot_number`, `pest_and_disease_number`, `monitoring_time`),
  CONSTRAINT `monitoring_record_ibfk_1` FOREIGN KEY (`plot_number`) REFERENCES `plot` (`plot_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_2` FOREIGN KEY (`pest_and_disease_number`) REFERENCES `pest_and_disease` (`pest_and_disease_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_record_ibfk_3` FOREIGN KEY (`employee_number`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
