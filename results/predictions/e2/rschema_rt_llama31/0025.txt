-- Schema for database: rschema_0025
CREATE TABLE `tobacco_products` (
  `product_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `production_date` datetime NOT NULL,
  `shelf_life` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  `production_batch_number` varchar(255) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `total_employees` int(11) NOT NULL,
  `person_in_charge` varchar(255) NOT NULL,
  `responsibilities` text NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(255) NOT NULL,
  `joining_date` datetime NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_record` (
  `salesperson_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `sales_price` decimal(10,2) NOT NULL,
  `customer_phone_number` varchar(15) NOT NULL,
  `customer_address` text NOT NULL,
  `sales_type` varchar(255) NOT NULL,
  PRIMARY KEY (`salesperson_id`, `product_id`, `sales_time`),
  CONSTRAINT `production_record_ibfk_1` FOREIGN KEY (`product_id`) REFERENCES `tobacco_products` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_record_details` (
  `producer_id` int(11) NOT NULL,
  `production_record_id` int(11) NOT NULL,
  PRIMARY KEY (`producer_id`, `production_record_id`),
  CONSTRAINT `production_record_details_ibfk_1` FOREIGN KEY (`production_record_id`) REFERENCES `production_record` (`salesperson_id`, `product_id`, `sales_time`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
