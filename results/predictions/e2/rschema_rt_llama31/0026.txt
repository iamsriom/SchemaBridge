-- Schema for database: rschema_0026
CREATE TABLE `textile` (
  `textile_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `production_date` datetime NOT NULL,
  `type` varchar(100) NOT NULL,
  PRIMARY KEY (`textile_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `manager` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventory` (
  `inventory_id` int(11) NOT NULL AUTO_INCREMENT,
  `textile_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `stocked_time` datetime NOT NULL,
  PRIMARY KEY (`inventory_id`),
  CONSTRAINT `inventory_ibfk_1` FOREIGN KEY (`textile_id`) REFERENCES `textile` (`textile_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_id` int(11) NOT NULL AUTO_INCREMENT,
  `department_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `position` varchar(100) NOT NULL,
  `gender` char(1) NOT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_relationship` (
  `production_id` int(11) NOT NULL AUTO_INCREMENT,
  `textile_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  `quantity_produced` int(11) NOT NULL,
  PRIMARY KEY (`production_id`),
  CONSTRAINT `production_relationship_ibfk_1` FOREIGN KEY (`textile_id`) REFERENCES `textile` (`textile_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_relationship_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_relationship` (
  `sales_id` int(11) NOT NULL AUTO_INCREMENT,
  `textile_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `customer` varchar(255) NOT NULL,
  `sales_time` datetime NOT NULL,
  `quantity_sold` int(11) NOT NULL,
  PRIMARY KEY (`sales_id`),
  CONSTRAINT `sales_relationship_ibfk_1` FOREIGN KEY (`textile_id`) REFERENCES `textile` (`textile_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_relationship_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_evaluation` (
  `performance_id` int(11) NOT NULL AUTO_INCREMENT,
  `employee_id` int(11) NOT NULL,
  `evaluation_date` datetime NOT NULL,
  `score` int(11) NOT NULL,
  PRIMARY KEY (`performance_id`),
  CONSTRAINT `performance_evaluation_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
