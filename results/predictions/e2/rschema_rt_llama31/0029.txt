-- Schema for database: rschema_0029
CREATE TABLE `furniture` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `production_date` datetime NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `id` int(11) NOT NULL,
  `supervisor_employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `department_ibfk_1` FOREIGN KEY (`supervisor_employee_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `id` int(11) NOT NULL,
  `gender` char(1) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `position` varchar(255) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production` (
  `furniture_id` int(11) NOT NULL,
  `producer_employee_id` int(11) NOT NULL,
  `production_time` datetime NOT NULL,
  `production_quantity` int(11) NOT NULL,
  PRIMARY KEY (`furniture_id`, `producer_employee_id`),
  CONSTRAINT `production_ibfk_1` FOREIGN KEY (`furniture_id`) REFERENCES `furniture` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `production_ibfk_2` FOREIGN KEY (`producer_employee_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `furniture_id` int(11) NOT NULL,
  `sales_rep_employee_id` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  `sales_quantity` int(11) NOT NULL,
  `customer` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`furniture_id`, `sales_rep_employee_id`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`furniture_id`) REFERENCES `furniture` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`sales_rep_employee_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
