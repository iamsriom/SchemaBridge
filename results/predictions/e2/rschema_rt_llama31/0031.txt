-- Schema for database: rschema_0031
CREATE TABLE `moving_team` (
  `team_id` int(11) NOT NULL,
  `team_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `mover` (
  `worker_id` int(11) NOT NULL,
  `name` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `assigned_team_id` int(11) NOT NULL,
  PRIMARY KEY (`worker_id`),
  CONSTRAINT `mover_ibfk_1` FOREIGN KEY (`assigned_team_id`) REFERENCES `moving_team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `moving_task` (
  `task_id` int(11) NOT NULL,
  `assigned_team_id` int(11) NOT NULL,
  `customer_feedback` text DEFAULT NULL,
  `customer_name` varchar(255) DEFAULT NULL,
  `customer_phone_number` varchar(15) DEFAULT NULL,
  `destination_address` varchar(255) DEFAULT NULL,
  `starting_address` varchar(255) DEFAULT NULL,
  `moving_cost` decimal(10,2) DEFAULT NULL,
  `moving_date` datetime DEFAULT NULL,
  `quantity_of_items` int(11) DEFAULT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `moving_task_ibfk_1` FOREIGN KEY (`assigned_team_id`) REFERENCES `moving_team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
