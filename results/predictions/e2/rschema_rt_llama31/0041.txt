-- Schema for database: rschema_0041
CREATE TABLE `property_manager` (
  `property_manager_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`property_manager_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_staff` (
  `repair_staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`repair_staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `community` (
  `community_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `property_manager_id` int(11) NOT NULL,
  PRIMARY KEY (`community_id`),
  CONSTRAINT `community_ibfk_1` FOREIGN KEY (`property_manager_id`) REFERENCES `property_manager` (`property_manager_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resident` (
  `resident_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `community_id` int(11) NOT NULL,
  PRIMARY KEY (`resident_id`),
  CONSTRAINT `resident_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `management` (
  `property_manager_id` int(11) NOT NULL,
  `community_id` int(11) NOT NULL,
  PRIMARY KEY (`property_manager_id`, `community_id`),
  CONSTRAINT `management_ibfk_1` FOREIGN KEY (`property_manager_id`) REFERENCES `property_manager` (`property_manager_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `management_ibfk_2` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_order` (
  `repair_order_id` int(11) NOT NULL AUTO_INCREMENT,
  `community_id` int(11) NOT NULL,
  `description` text NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `request_date` datetime NOT NULL,
  `resident_id` int(11) NOT NULL,
  `repair_staff_id` int(11) NOT NULL,
  PRIMARY KEY (`repair_order_id`),
  CONSTRAINT `repair_order_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_order_ibfk_2` FOREIGN KEY (`resident_id`) REFERENCES `resident` (`resident_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_order_ibfk_3` FOREIGN KEY (`repair_staff_id`) REFERENCES `repair_staff` (`repair_staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `residence` (
  `resident_id` int(11) NOT NULL,
  `community_id` int(11) NOT NULL,
  PRIMARY KEY (`resident_id`, `community_id`),
  CONSTRAINT `residence_ibfk_1` FOREIGN KEY (`resident_id`) REFERENCES `resident` (`resident_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `residence_ibfk_2` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `execution` (
  `repair_order_id` int(11) NOT NULL,
  `repair_staff_id` int(11) NOT NULL,
  PRIMARY KEY (`repair_order_id`, `repair_staff_id`),
  CONSTRAINT `execution_ibfk_1` FOREIGN KEY (`repair_order_id`) REFERENCES `repair_order` (`repair_order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `execution_ibfk_2` FOREIGN KEY (`repair_staff_id`) REFERENCES `repair_staff` (`repair_staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
