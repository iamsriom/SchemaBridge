-- Schema for database: rschema_0043
CREATE TABLE `loan_application` (
  `application_number` int(11) NOT NULL,
  `property_number` varchar(255) NOT NULL,
  `loan_amount` decimal(10,2) NOT NULL,
  `loan_term` int(11) NOT NULL,
  `approval_status` varchar(50) NOT NULL,
  `applicant_name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `id_number` varchar(50) NOT NULL,
  `submitted_time` datetime NOT NULL,
  PRIMARY KEY (`application_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `approval_officer` (
  `officer_number` int(11) NOT NULL,
  `officer_name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  PRIMARY KEY (`officer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `loan_approval` (
  `officer_number` int(11) NOT NULL,
  `approval_time` datetime NOT NULL,
  `application_number` int(11) NOT NULL,
  `approval_result` varchar(50) NOT NULL,
  PRIMARY KEY (`officer_number`, `approval_time`, `application_number`),
  CONSTRAINT `loan_approval_ibfk_1` FOREIGN KEY (`application_number`) REFERENCES `loan_application` (`application_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `loan_approval_ibfk_2` FOREIGN KEY (`officer_number`) REFERENCES `approval_officer` (`officer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
