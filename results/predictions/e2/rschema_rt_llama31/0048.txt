-- Schema for database: rschema_0048
CREATE TABLE `reporter` (
  `reporter_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`reporter_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `editor` (
  `editor_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`editor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publisher` (
  `publisher_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `section` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`publisher_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `manuscript` (
  `manuscript_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`manuscript_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `writing_record` (
  `reporter_id` int(11) NOT NULL,
  `manuscript_id` int(11) NOT NULL,
  `submission_time` datetime NOT NULL,
  `responsible_part` varchar(255) NOT NULL,
  PRIMARY KEY (`reporter_id`, `manuscript_id`),
  CONSTRAINT `writing_record_ibfk_1` FOREIGN KEY (`reporter_id`) REFERENCES `reporter` (`reporter_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `writing_record_ibfk_2` FOREIGN KEY (`manuscript_id`) REFERENCES `manuscript` (`manuscript_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `review_record` (
  `editor_id` int(11) NOT NULL,
  `manuscript_id` int(11) NOT NULL,
  `review_time` datetime NOT NULL,
  `review_part` varchar(255) NOT NULL,
  PRIMARY KEY (`editor_id`, `manuscript_id`),
  CONSTRAINT `review_record_ibfk_1` FOREIGN KEY (`editor_id`) REFERENCES `editor` (`editor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `review_record_ibfk_2` FOREIGN KEY (`manuscript_id`) REFERENCES `manuscript` (`manuscript_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publication_record` (
  `manuscript_id` int(11) NOT NULL,
  `publisher_id` int(11) NOT NULL,
  `publication_time` datetime NOT NULL,
  PRIMARY KEY (`manuscript_id`, `publisher_id`),
  CONSTRAINT `publication_record_ibfk_1` FOREIGN KEY (`manuscript_id`) REFERENCES `manuscript` (`manuscript_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `publication_record_ibfk_2` FOREIGN KEY (`publisher_id`) REFERENCES `publisher` (`publisher_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
