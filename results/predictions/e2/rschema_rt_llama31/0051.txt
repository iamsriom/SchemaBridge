-- Schema for database: rschema_0051
CREATE TABLE `reservoir_and_river` (
  `resource_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  PRIMARY KEY (`resource_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_personnel` (
  `monitoring_personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`monitoring_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel` (
  `maintenance_personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(100) NOT NULL,
  PRIMARY KEY (`maintenance_personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_monitoring` (
  `resource_id` int(11) NOT NULL,
  `monitoring_personnel_id` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `pollution_level` float DEFAULT NULL,
  `water_quality_indicator` float DEFAULT NULL,
  `water_volume` float DEFAULT NULL,
  PRIMARY KEY (`resource_id`, `monitoring_personnel_id`, `monitoring_time`),
  CONSTRAINT `resource_monitoring_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `reservoir_and_river` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resource_monitoring_ibfk_2` FOREIGN KEY (`monitoring_personnel_id`) REFERENCES `monitoring_personnel` (`monitoring_personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resource_maintenance` (
  `resource_id` int(11) NOT NULL,
  `maintenance_personnel_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `cost` float DEFAULT NULL,
  PRIMARY KEY (`resource_id`, `maintenance_personnel_id`, `maintenance_time`),
  CONSTRAINT `resource_maintenance_ibfk_1` FOREIGN KEY (`resource_id`) REFERENCES `reservoir_and_river` (`resource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `resource_maintenance_ibfk_2` FOREIGN KEY (`maintenance_personnel_id`) REFERENCES `maintenance_personnel` (`maintenance_personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
