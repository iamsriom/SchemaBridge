-- Schema for database: rschema_0053
CREATE TABLE `processing_pool` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `responsible_person_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `processing_pool_ibfk_1` FOREIGN KEY (`responsible_person_id`) REFERENCES `staff` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `responsibilities` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_pool_record` (
  `processing_pool_id` int(11) NOT NULL,
  `recording_time` datetime NOT NULL,
  `recording_staff_id` int(11) NOT NULL,
  `processing_capacity` int(11) NOT NULL,
  `efficiency` float NOT NULL,
  PRIMARY KEY (`processing_pool_id`, `recording_time`, `recording_staff_id`),
  CONSTRAINT `processing_pool_record_ibfk_1` FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pool` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `processing_pool_record_ibfk_2` FOREIGN KEY (`recording_staff_id`) REFERENCES `staff` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `processing_pool_maintenance` (
  `processing_pool_id` int(11) NOT NULL,
  `maintenance_staff_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`processing_pool_id`, `maintenance_staff_id`, `maintenance_time`),
  CONSTRAINT `processing_pool_maintenance_ibfk_1` FOREIGN KEY (`processing_pool_id`) REFERENCES `processing_pool` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `processing_pool_maintenance_ibfk_2` FOREIGN KEY (`maintenance_staff_id`) REFERENCES `staff` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
