-- Schema for database: rschema_0059
CREATE TABLE `handler` (
  `handler_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `department` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`handler_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fault_handling` (
  `fault_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `description` varchar(255) NOT NULL,
  `date_occurred` datetime NOT NULL,
  `date_resolved` datetime DEFAULT NULL,
  `handler_id` int(11) NOT NULL,
  PRIMARY KEY (`fault_id`),
  CONSTRAINT `fault_handling_ibfk_1` FOREIGN KEY (`handler_id`) REFERENCES `handler` (`handler_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
