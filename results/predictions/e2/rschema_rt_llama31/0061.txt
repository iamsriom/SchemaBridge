-- Schema for database: rschema_0061
CREATE TABLE `patent` (
  `patent_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `application_date` varchar(255) NOT NULL,
  `status` varchar(255) NOT NULL,
  PRIMARY KEY (`patent_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `inventor` (
  `inventor_number` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` varchar(255) NOT NULL,
  `gender` varchar(255) NOT NULL,
  `phone_number` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`inventor_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `patent_application_record` (
  `inventor_number` varchar(255) NOT NULL,
  `application_date` varchar(255) NOT NULL,
  `patent_number` varchar(255) NOT NULL,
  PRIMARY KEY (`inventor_number`, `application_date`, `patent_number`),
  CONSTRAINT `patent_application_record_ibfk_1` FOREIGN KEY (`inventor_number`) REFERENCES `inventor` (`inventor_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `patent_application_record_ibfk_2` FOREIGN KEY (`patent_number`) REFERENCES `patent` (`patent_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
