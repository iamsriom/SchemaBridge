-- Schema for database: rschema_0062
CREATE TABLE `document` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `creation_date` datetime NOT NULL,
  `title` varchar(255) NOT NULL,
  `version_number` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `author` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `position` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `modification_record` (
  `document_id` int(11) NOT NULL,
  `modification_date` datetime NOT NULL,
  `modifying_author_id` int(11) NOT NULL,
  `modification_content` text DEFAULT NULL,
  `version_number` int(11) NOT NULL,
  PRIMARY KEY (`document_id`, `modification_date`, `modifying_author_id`),
  CONSTRAINT `modification_record_ibfk_1` FOREIGN KEY (`document_id`) REFERENCES `document` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `modification_record_ibfk_2` FOREIGN KEY (`modifying_author_id`) REFERENCES `author` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
