-- Schema for database: rschema_0063
CREATE TABLE `scientific_and_technological_achievement` (
  `achievement_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_researched` datetime NOT NULL,
  `technical_field` varchar(255) NOT NULL,
  PRIMARY KEY (`achievement_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_number` int(11) NOT NULL,
  `barcode` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `achievement_transformation` (
  `achievement_number` int(11) NOT NULL,
  `product_number` int(11) NOT NULL,
  `date_of_transformation` datetime NOT NULL,
  PRIMARY KEY (`achievement_number`, `product_number`),
  CONSTRAINT `achievement_transformation_ibfk_1` FOREIGN KEY (`achievement_number`) REFERENCES `scientific_and_technological_achievement` (`achievement_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `achievement_transformation_ibfk_2` FOREIGN KEY (`product_number`) REFERENCES `product` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
