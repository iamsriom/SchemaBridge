-- Schema for database: rschema_0064
CREATE TABLE `project` (
  `project_number` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `application_date` datetime NOT NULL,
  `unit` varchar(255) NOT NULL,
  PRIMARY KEY (`project_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expert` (
  `expert_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `unit` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`expert_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project_review_record` (
  `project_number` int(11) NOT NULL,
  `expert_number` int(11) NOT NULL,
  `review_date` datetime NOT NULL,
  `opinion` text DEFAULT NULL,
  PRIMARY KEY (`project_number`, `expert_number`),
  CONSTRAINT `project_review_record_ibfk_1` FOREIGN KEY (`project_number`) REFERENCES `project` (`project_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `project_review_record_ibfk_2` FOREIGN KEY (`expert_number`) REFERENCES `expert` (`expert_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
