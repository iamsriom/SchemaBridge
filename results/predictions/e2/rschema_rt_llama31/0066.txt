-- Schema for database: rschema_0066
CREATE TABLE `community` (
  `community_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `manager` varchar(255) NOT NULL,
  PRIMARY KEY (`community_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `owner` (
  `owner_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `house_number` varchar(255) NOT NULL,
  `community_id` int(11) NOT NULL,
  PRIMARY KEY (`owner_id`),
  CONSTRAINT `owner_ibfk_1` FOREIGN KEY (`community_id`) REFERENCES `community` (`community_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `house` (
  `house_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`house_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_application_record` (
  `owner_id` int(11) NOT NULL,
  `house_id` int(11) NOT NULL,
  `application_time` datetime NOT NULL,
  `repair_content` text NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`owner_id`, `house_id`),
  CONSTRAINT `repair_application_record_ibfk_1` FOREIGN KEY (`owner_id`) REFERENCES `owner` (`owner_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_application_record_ibfk_2` FOREIGN KEY (`house_id`) REFERENCES `house` (`house_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
