-- Schema for database: rschema_0071
CREATE TABLE `mine` (
  `mine_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `person_in_charge` int(11) NOT NULL,
  PRIMARY KEY (`mine_number`),
  CONSTRAINT `mine_ibfk_1` FOREIGN KEY (`person_in_charge`) REFERENCES `safety_officer` (`officer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_officer` (
  `officer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `telephone_number` varchar(15) NOT NULL,
  `position` varchar(255) NOT NULL,
  `mine_number` int(11) NOT NULL,
  PRIMARY KEY (`officer_number`),
  CONSTRAINT `safety_officer_ibfk_1` FOREIGN KEY (`mine_number`) REFERENCES `mine` (`mine_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `safety_inspection_record` (
  `inspection_date` datetime NOT NULL,
  `mine_number` int(11) NOT NULL,
  `content` text NOT NULL,
  `results` varchar(255) NOT NULL,
  `inspector` int(11) NOT NULL,
  PRIMARY KEY (`mine_number`, `inspection_date`),
  CONSTRAINT `safety_inspection_record_ibfk_1` FOREIGN KEY (`mine_number`) REFERENCES `mine` (`mine_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
