-- Schema for database: rschema_0079
CREATE TABLE `stock` (
  `stock_code` varchar(20) NOT NULL,
  `type` varchar(50) DEFAULT NULL,
  `market` varchar(50) DEFAULT NULL,
  `owner` varchar(100) DEFAULT NULL,
  `total_investment_amount` decimal(15,2) DEFAULT NULL,
  `number_of_holding_shares` int(11) DEFAULT NULL,
  `last_transaction_time` datetime DEFAULT NULL,
  `transaction_status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`stock_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `order` (
  `order_id` int(11) NOT NULL AUTO_INCREMENT,
  `type` varchar(50) DEFAULT NULL,
  `creation_time` datetime DEFAULT NULL,
  `transaction_date` datetime DEFAULT NULL,
  `amount` decimal(15,2) DEFAULT NULL,
  `quantity` int(11) DEFAULT NULL,
  `service_fee` decimal(15,2) DEFAULT NULL,
  `status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`order_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stock_order` (
  `order_id` int(11) NOT NULL,
  `stock_code` varchar(20) NOT NULL,
  PRIMARY KEY (`order_id`, `stock_code`),
  CONSTRAINT `stock_order_ibfk_1` FOREIGN KEY (`order_id`) REFERENCES `order` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `stock_order_ibfk_2` FOREIGN KEY (`stock_code`) REFERENCES `stock` (`stock_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
