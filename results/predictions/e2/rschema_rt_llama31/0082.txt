-- Schema for database: rschema_0082
CREATE TABLE `political_party` (
  `party_id` int(11) NOT NULL AUTO_INCREMENT,
  `party_name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`party_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `candidate` (
  `candidate_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `party_id` int(11) NOT NULL,
  `industry_connections` text DEFAULT NULL,
  PRIMARY KEY (`candidate_id`),
  CONSTRAINT `candidate_ibfk_1` FOREIGN KEY (`party_id`) REFERENCES `political_party` (`party_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `voter` (
  `voter_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `candidate_id` int(11) NOT NULL,
  `vote_time` datetime NOT NULL,
  PRIMARY KEY (`voter_id`),
  CONSTRAINT `voter_ibfk_1` FOREIGN KEY (`candidate_id`) REFERENCES `candidate` (`candidate_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
