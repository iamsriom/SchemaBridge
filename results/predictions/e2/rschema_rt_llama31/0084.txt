-- Schema for database: rschema_0084
CREATE TABLE `land` (
  `land_id` int(11) NOT NULL,
  `location` varchar(255) DEFAULT NULL,
  `total_area` float DEFAULT NULL,
  PRIMARY KEY (`land_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `person_in_charge` varchar(255) DEFAULT NULL,
  `project_location` varchar(255) DEFAULT NULL,
  `project_name` varchar(255) DEFAULT NULL,
  `purpose` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `land_usage_record` (
  `land_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `current_status` varchar(255) DEFAULT NULL,
  `personnel` varchar(255) DEFAULT NULL,
  `duration_of_usage` float DEFAULT NULL,
  `area_used` float DEFAULT NULL,
  PRIMARY KEY (`land_id`, `project_id`, `record_time`),
  CONSTRAINT `land_usage_record_ibfk_1` FOREIGN KEY (`land_id`) REFERENCES `land` (`land_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `land_usage_record_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
