-- Schema for database: rschema_0089
CREATE TABLE `block` (
  `hash` varchar(64) NOT NULL,
  `height` int(11) NOT NULL,
  `merkle_root` varchar(64) NOT NULL,
  `previous_block_hash` varchar(64) DEFAULT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`hash`),
  CONSTRAINT `block_ibfk_1` FOREIGN KEY (`previous_block_hash`) REFERENCES `block` (`hash`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account` (
  `address` varchar(42) NOT NULL,
  `balance` decimal(20,8) NOT NULL,
  PRIMARY KEY (`address`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction` (
  `transaction_hash` varchar(64) NOT NULL,
  `block_hash` varchar(64) NOT NULL,
  `fee` decimal(20,8) NOT NULL,
  `input_address` varchar(42) NOT NULL,
  `output_address` varchar(42) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`transaction_hash`),
  CONSTRAINT `transaction_ibfk_1` FOREIGN KEY (`block_hash`) REFERENCES `block` (`hash`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_ibfk_2` FOREIGN KEY (`input_address`) REFERENCES `account` (`address`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_ibfk_3` FOREIGN KEY (`output_address`) REFERENCES `account` (`address`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
