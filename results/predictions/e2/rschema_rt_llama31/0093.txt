-- Schema for database: rschema_0093
CREATE TABLE `Book` (
  `isbn` varchar(13) NOT NULL,
  `title` varchar(255) NOT NULL,
  `publication_year` int(4) NOT NULL,
  `publishing_house` varchar(255) NOT NULL,
  PRIMARY KEY (`isbn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Type` (
  `type_id` int(11) NOT NULL AUTO_INCREMENT,
  `type_name` varchar(100) NOT NULL,
  PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Tag` (
  `tag_id` int(11) NOT NULL AUTO_INCREMENT,
  `tag_name` varchar(100) NOT NULL,
  PRIMARY KEY (`tag_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book_Type_Relation` (
  `isbn` varchar(13) NOT NULL,
  `type_id` int(11) NOT NULL,
  PRIMARY KEY (`isbn`, `type_id`),
  CONSTRAINT `Book_Type_Relation_ibfk_1` FOREIGN KEY (`isbn`) REFERENCES `Book` (`isbn`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Book_Type_Relation_ibfk_2` FOREIGN KEY (`type_id`) REFERENCES `Type` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Book_Tag_Relation` (
  `isbn` varchar(13) NOT NULL,
  `tag_id` int(11) NOT NULL,
  PRIMARY KEY (`isbn`, `tag_id`),
  CONSTRAINT `Book_Tag_Relation_ibfk_1` FOREIGN KEY (`isbn`) REFERENCES `Book` (`isbn`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Book_Tag_Relation_ibfk_2` FOREIGN KEY (`tag_id`) REFERENCES `Tag` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
