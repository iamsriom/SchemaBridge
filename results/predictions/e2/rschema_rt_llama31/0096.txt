-- Schema for database: rschema_0096
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `hair_color` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hairstylist` (
  `hairstylist_id` int(11) NOT NULL,
  `expertise` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`hairstylist_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service` (
  `customer_id` int(11) NOT NULL,
  `hairstylist_id` int(11) NOT NULL,
  `appointment_time` datetime NOT NULL,
  `service_content` varchar(255) DEFAULT NULL,
  `service_time` int(11) DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `hairstylist_id`, `appointment_time`),
  CONSTRAINT `service_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_ibfk_2` FOREIGN KEY (`hairstylist_id`) REFERENCES `hairstylist` (`hairstylist_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
