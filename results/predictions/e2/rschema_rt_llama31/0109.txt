-- Schema for database: rschema_0109
CREATE TABLE `country` (
  `country_code` char(3) NOT NULL,
  `country_name` varchar(255) NOT NULL,
  PRIMARY KEY (`country_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member` (
  `member_id` int(11) NOT NULL AUTO_INCREMENT,
  `company_name` varchar(255) DEFAULT NULL,
  `country_of_origin` char(3) NOT NULL,
  `name` varchar(255) NOT NULL,
  `name_pronunciation` varchar(255) DEFAULT NULL,
  `public_status` boolean NOT NULL DEFAULT TRUE,
  PRIMARY KEY (`member_id`),
  CONSTRAINT `member_ibfk_1` FOREIGN KEY (`country_of_origin`) REFERENCES `country` (`country_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `address` (
  `address_id` int(11) NOT NULL AUTO_INCREMENT,
  `contact_number` varchar(20) DEFAULT NULL,
  `country_code` char(3) NOT NULL,
  `postal_code` varchar(10) DEFAULT NULL,
  `specific_address` varchar(255) NOT NULL,
  PRIMARY KEY (`address_id`),
  CONSTRAINT `address_ibfk_1` FOREIGN KEY (`country_code`) REFERENCES `country` (`country_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member_address_management` (
  `address_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`address_id`, `member_id`),
  CONSTRAINT `member_address_management_ibfk_1` FOREIGN KEY (`address_id`) REFERENCES `address` (`address_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `member_address_management_ibfk_2` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
