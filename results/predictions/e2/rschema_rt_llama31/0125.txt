-- Schema for database: rschema_0125
CREATE TABLE `page` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `edit_status` varchar(50) NOT NULL,
  `visibility_status` varchar(50) NOT NULL,
  `active_version` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `belonging_page` int(11) NOT NULL,
  `parent_comment` int(11) DEFAULT NULL,
  `content` text NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`belonging_page`) REFERENCES `page` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`parent_comment`) REFERENCES `comment` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `version_record` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `page_number` int(11) NOT NULL,
  `version_number` int(11) NOT NULL,
  `image_url` varchar(255) DEFAULT NULL,
  `description` text NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `version_record_ibfk_1` FOREIGN KEY (`page_number`) REFERENCES `page` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
