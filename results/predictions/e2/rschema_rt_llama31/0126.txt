-- Schema for database: rschema_0126
CREATE TABLE `Inspector` (
  `Inspector_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Email` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Inspector_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Death_Record` (
  `Death_Record_ID` int(11) NOT NULL,
  `Inspector_ID` int(11) NOT NULL,
  `Last_Modification` datetime NOT NULL,
  `Deceased_Name` varchar(255) NOT NULL,
  `Unique_ID` varchar(255) NOT NULL,
  PRIMARY KEY (`Death_Record_ID`),
  CONSTRAINT `death_record_ibfk_1` FOREIGN KEY (`Inspector_ID`) REFERENCES `Inspector` (`Inspector_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Confirmed_Death_Record` (
  `Inspector_ID` int(11) NOT NULL,
  `Death_Record_ID` int(11) NOT NULL,
  PRIMARY KEY (`Inspector_ID`, `Death_Record_ID`),
  CONSTRAINT `confirmed_death_record_ibfk_1` FOREIGN KEY (`Inspector_ID`) REFERENCES `Inspector` (`Inspector_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `confirmed_death_record_ibfk_2` FOREIGN KEY (`Death_Record_ID`) REFERENCES `Death_Record` (`Death_Record_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
