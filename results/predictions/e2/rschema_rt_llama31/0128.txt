-- Schema for database: rschema_0128
CREATE TABLE `task_batch` (
  `batch_id` int(11) NOT NULL,
  `batch_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  PRIMARY KEY (`batch_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_execution_record` (
  `execution_id` int(11) NOT NULL,
  `task_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `exit_code` int(11) NOT NULL,
  `exit_message` varchar(255) DEFAULT NULL,
  `error_message` varchar(255) DEFAULT NULL,
  `last_update_time` datetime NOT NULL,
  `external_execution_id` int(11) DEFAULT NULL,
  `parent_execution_id` int(11) DEFAULT NULL,
  `batch_id` int(11) NOT NULL,
  PRIMARY KEY (`execution_id`),
  CONSTRAINT `task_execution_record_ibfk_1` FOREIGN KEY (`batch_id`) REFERENCES `task_batch` (`batch_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_execution_record_ibfk_2` FOREIGN KEY (`external_execution_id`) REFERENCES `task_execution_record` (`execution_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `task_execution_record_ibfk_3` FOREIGN KEY (`parent_execution_id`) REFERENCES `task_execution_record` (`execution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task_parameter` (
  `execution_id` int(11) NOT NULL,
  `parameter_name` varchar(255) NOT NULL,
  `parameter_value` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`execution_id`, `parameter_name`),
  CONSTRAINT `task_parameter_ibfk_1` FOREIGN KEY (`execution_id`) REFERENCES `task_execution_record` (`execution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
