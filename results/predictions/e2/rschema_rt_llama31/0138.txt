-- Schema for database: rschema_0138
CREATE TABLE `employee` (
  `employee_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `performance` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`employee_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `store` (
  `store_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`store_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product_table` (
  `product_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `specifications` varchar(255) DEFAULT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employment` (
  `employee_number` int(11) NOT NULL,
  `store_number` int(11) NOT NULL,
  `employment_period` varchar(255) DEFAULT NULL,
  `monthly_salary` decimal(10,2) NOT NULL,
  PRIMARY KEY (`employee_number`, `store_number`),
  CONSTRAINT `employment_ibfk_1` FOREIGN KEY (`employee_number`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employment_ibfk_2` FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `product_number` int(11) NOT NULL,
  `store_number` int(11) NOT NULL,
  `monthly_sales_volume` int(11) NOT NULL,
  PRIMARY KEY (`product_number`, `store_number`),
  CONSTRAINT `sales_ibfk_1` FOREIGN KEY (`product_number`) REFERENCES `product_table` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_ibfk_2` FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
