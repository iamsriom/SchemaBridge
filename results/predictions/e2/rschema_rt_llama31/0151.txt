-- Schema for database: rschema_0151
CREATE TABLE `customer` (
  `identity_card_number` varchar(20) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`identity_card_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `savings_institution` (
  `savings_institution_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`savings_institution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_record` (
  `identity_card_number` varchar(20) NOT NULL,
  `savings_institution_id` int(11) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_amount` decimal(10,2) NOT NULL,
  `interest_rate` decimal(5,2) DEFAULT NULL,
  `maturity_date` date DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`identity_card_number`, `savings_institution_id`, `transaction_time`),
  CONSTRAINT `transaction_record_ibfk_1` FOREIGN KEY (`identity_card_number`) REFERENCES `customer` (`identity_card_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_record_ibfk_2` FOREIGN KEY (`savings_institution_id`) REFERENCES `savings_institution` (`savings_institution_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
