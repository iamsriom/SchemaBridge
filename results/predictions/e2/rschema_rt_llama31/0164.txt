-- Schema for database: rschema_0164
CREATE TABLE `warehouse` (
  `warehouse_id` int(11) NOT NULL AUTO_INCREMENT,
  `creation_date` datetime NOT NULL,
  `update_date` datetime NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) DEFAULT NULL,
  `image_path` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`warehouse_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `role` varchar(50) DEFAULT NULL,
  `username` varchar(100) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stack_position` (
  `stack_position_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` varchar(255) DEFAULT NULL,
  `image_path` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) DEFAULT NULL,
  `belonging_warehouse_id` int(11) NOT NULL,
  PRIMARY KEY (`stack_position_id`),
  CONSTRAINT `stack_position_ibfk_1` FOREIGN KEY (`belonging_warehouse_id`) REFERENCES `warehouse` (`warehouse_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sensor` (
  `sensor_id` int(11) NOT NULL AUTO_INCREMENT,
  `collected_data_value` float NOT NULL,
  `collection_time` datetime NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `type` varchar(100) DEFAULT NULL,
  `stack_position_id` int(11) NOT NULL,
  PRIMARY KEY (`sensor_id`),
  CONSTRAINT `sensor_ibfk_1` FOREIGN KEY (`stack_position_id`) REFERENCES `stack_position` (`stack_position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_operation_stack_position` (
  `operation_user_id` int(11) NOT NULL,
  `stack_position_id` int(11) NOT NULL,
  `operation_description` varchar(255) DEFAULT NULL,
  `operation_time` datetime NOT NULL,
  `operation_type` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`operation_user_id`, `stack_position_id`),
  CONSTRAINT `user_operation_stack_position_ibfk_1` FOREIGN KEY (`operation_user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_operation_stack_position_ibfk_2` FOREIGN KEY (`stack_position_id`) REFERENCES `stack_position` (`stack_position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stack_position_alert` (
  `stack_position_id` int(11) NOT NULL,
  `alert_time` datetime NOT NULL,
  `alert_description` varchar(255) DEFAULT NULL,
  `alert_level` varchar(50) DEFAULT NULL,
  `alert_status` varchar(50) DEFAULT NULL,
  `alert_type` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`stack_position_id`, `alert_time`),
  CONSTRAINT `stack_position_alert_ibfk_1` FOREIGN KEY (`stack_position_id`) REFERENCES `stack_position` (`stack_position_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
