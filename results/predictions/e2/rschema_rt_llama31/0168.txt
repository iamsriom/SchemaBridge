-- Schema for database: rschema_0168
CREATE TABLE `electrical_equipment` (
  `equipment_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(255) NOT NULL,
  `installation_date` datetime NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `log_record` (
  `equipment_id` int(11) NOT NULL,
  `record_time` datetime NOT NULL,
  `description` text NOT NULL,
  PRIMARY KEY (`equipment_id`, `record_time`),
  CONSTRAINT `log_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `electrical_equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_record` (
  `equipment_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `personnel` varchar(255) NOT NULL,
  `maintenance_type` varchar(255) NOT NULL,
  PRIMARY KEY (`equipment_id`, `maintenance_time`),
  CONSTRAINT `maintenance_record_ibfk_1` FOREIGN KEY (`equipment_id`) REFERENCES `electrical_equipment` (`equipment_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
