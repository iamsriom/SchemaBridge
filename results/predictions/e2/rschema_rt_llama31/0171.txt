-- Schema for database: rschema_0171
CREATE TABLE `Parking` (
  `Parking_Space_ID` int(11) NOT NULL,
  `Parking_Space_Status` varchar(50) NOT NULL,
  PRIMARY KEY (`Parking_Space_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Entrance` (
  `Entrance_Number` int(11) NOT NULL,
  `Location` varchar(255) NOT NULL,
  PRIMARY KEY (`Entrance_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Exit` (
  `Exit_Number` int(11) NOT NULL,
  `Location` varchar(255) NOT NULL,
  PRIMARY KEY (`Exit_Number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Vehicle` (
  `Vehicle_ID` int(11) NOT NULL,
  `License_Plate` varchar(20) NOT NULL,
  `Model` varchar(100) NOT NULL,
  PRIMARY KEY (`Vehicle_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Parking_Record` (
  `Parking_Space_ID` int(11) NOT NULL,
  `Vehicle_ID` int(11) NOT NULL,
  `Entry_Time` datetime NOT NULL,
  `Exit_Time` datetime DEFAULT NULL,
  `Duration` int(11) DEFAULT NULL,
  `Fee` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`Parking_Space_ID`, `Vehicle_ID`),
  CONSTRAINT `parking_record_ibfk_1` FOREIGN KEY (`Parking_Space_ID`) REFERENCES `Parking` (`Parking_Space_ID`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_2` FOREIGN KEY (`Vehicle_ID`) REFERENCES `Vehicle` (`Vehicle_ID`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_3` FOREIGN KEY (`Entrance_Number`) REFERENCES `Entrance` (`Entrance_Number`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `parking_record_ibfk_4` FOREIGN KEY (`Exit_Number`) REFERENCES `Exit` (`Exit_Number`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
