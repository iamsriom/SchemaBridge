-- Schema for database: rschema_0174
CREATE TABLE `patient` (
  `medical_record_number` varchar(20) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `id_card_number` varchar(20) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `admission_time` datetime DEFAULT NULL,
  PRIMARY KEY (`medical_record_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vital_signs` (
  `feature_name` varchar(50) NOT NULL,
  `lower_limit` decimal(5,2) NOT NULL,
  `upper_limit` decimal(5,2) NOT NULL,
  PRIMARY KEY (`feature_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring` (
  `medical_record_number` varchar(20) NOT NULL,
  `feature_name` varchar(50) NOT NULL,
  `value` decimal(5,2) NOT NULL,
  `recording_time` datetime NOT NULL,
  PRIMARY KEY (`medical_record_number`, `feature_name`),
  CONSTRAINT `monitoring_ibfk_1` FOREIGN KEY (`medical_record_number`) REFERENCES `patient` (`medical_record_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `monitoring_ibfk_2` FOREIGN KEY (`feature_name`) REFERENCES `vital_signs` (`feature_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
