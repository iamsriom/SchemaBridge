-- Schema for database: rschema_0185
CREATE TABLE `proposal` (
  `proposal_id` int(11) NOT NULL,
  `content` text DEFAULT NULL,
  `submitted_at` datetime DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`proposal_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `member` (
  `member_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date DEFAULT NULL,
  `gender` char(1) DEFAULT NULL,
  `party_affiliation` varchar(100) DEFAULT NULL,
  `position` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`member_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `proposal_submission` (
  `proposal_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  PRIMARY KEY (`proposal_id`, `member_id`),
  CONSTRAINT `proposal_submission_ibfk_1` FOREIGN KEY (`proposal_id`) REFERENCES `proposal` (`proposal_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `proposal_submission_ibfk_2` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `proposal_handling` (
  `proposal_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  `opinion` text DEFAULT NULL,
  `handled_at` datetime DEFAULT NULL,
  PRIMARY KEY (`proposal_id`, `member_id`),
  CONSTRAINT `proposal_handling_ibfk_1` FOREIGN KEY (`proposal_id`) REFERENCES `proposal` (`proposal_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `proposal_handling_ibfk_2` FOREIGN KEY (`member_id`) REFERENCES `member` (`member_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
