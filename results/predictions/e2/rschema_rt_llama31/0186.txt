-- Schema for database: rschema_0186
CREATE TABLE `insured_person` (
  `social_security_id` varchar(20) NOT NULL,
  `name` varchar(255) NOT NULL,
  `identity_card_number` varchar(50) NOT NULL,
  `gender` char(1) NOT NULL,
  `date_of_birth` date NOT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`social_security_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `social_security_type` (
  `type_id` int(11) NOT NULL AUTO_INCREMENT,
  `type_name` varchar(100) NOT NULL,
  PRIMARY KEY (`type_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `insured_person_payment` (
  `payment_id` int(11) NOT NULL,
  `social_security_id` varchar(20) NOT NULL,
  `type_id` int(11) NOT NULL,
  `payer` varchar(255) NOT NULL,
  `payment_amount` decimal(10,2) NOT NULL,
  `payment_time` datetime NOT NULL,
  PRIMARY KEY (`payment_id`, `social_security_id`, `type_id`),
  CONSTRAINT `insured_person_payment_ibfk_1` FOREIGN KEY (`social_security_id`) REFERENCES `insured_person` (`social_security_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `insured_person_payment_ibfk_2` FOREIGN KEY (`type_id`) REFERENCES `social_security_type` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `insured_person_receipt` (
  `receipt_id` int(11) NOT NULL,
  `social_security_id` varchar(20) NOT NULL,
  `type_id` int(11) NOT NULL,
  `recipient` varchar(255) NOT NULL,
  `receipt_amount` decimal(10,2) NOT NULL,
  `receipt_time` datetime NOT NULL,
  PRIMARY KEY (`receipt_id`, `social_security_id`, `type_id`),
  CONSTRAINT `insured_person_receipt_ibfk_1` FOREIGN KEY (`social_security_id`) REFERENCES `insured_person` (`social_security_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `insured_person_receipt_ibfk_2` FOREIGN KEY (`type_id`) REFERENCES `social_security_type` (`type_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
