-- Schema for database: rschema_0190
CREATE TABLE `workstation` (
  `workstation_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`workstation_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_id` int(11) NOT NULL,
  `model` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`product_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_technician` (
  `repair_technician_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `workstation_id` int(11) NOT NULL,
  PRIMARY KEY (`repair_technician_id`),
  CONSTRAINT `repair_technician_ibfk_1` FOREIGN KEY (`workstation_id`) REFERENCES `workstation` (`workstation_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `repair_record` (
  `repair_technician_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `repair_time` datetime NOT NULL,
  `customer_name` varchar(255) NOT NULL,
  `customer_phone` varchar(15) DEFAULT NULL,
  `feedback` text DEFAULT NULL,
  `fault_description` text DEFAULT NULL,
  `duration` int(11) DEFAULT NULL,
  `repair_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`repair_technician_id`, `product_id`, `repair_time`),
  CONSTRAINT `repair_record_ibfk_1` FOREIGN KEY (`repair_technician_id`) REFERENCES `repair_technician` (`repair_technician_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `repair_record_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`product_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
