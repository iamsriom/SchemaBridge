-- Schema for database: rschema_0195
CREATE TABLE `property` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` float NOT NULL,
  `sale_price` decimal(10,2) NOT NULL,
  `sale_status` varchar(50) NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_personnel` (
  `employee_id` int(11) NOT NULL,
  `commission_rate` decimal(5,2) NOT NULL,
  `contact_phone` varchar(15) NOT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `client` (
  `client_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_phone` varchar(15) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `property_id` int(11) NOT NULL,
  `employee_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `sale_date` datetime NOT NULL,
  `sale_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`property_id`, `employee_id`, `client_id`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `sales_personnel` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_3` FOREIGN KEY (`client_id`) REFERENCES `client` (`client_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
