-- Schema for database: rschema_0199
CREATE TABLE `store` (
  `store_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `manager` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`store_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `product_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `model` varchar(255) DEFAULT NULL,
  `barcode` varchar(255) DEFAULT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales_record` (
  `product_number` int(11) NOT NULL,
  `store_number` int(11) NOT NULL,
  `sales_time` datetime NOT NULL,
  `quantity_sold` int(11) NOT NULL,
  `total_sales_amount` decimal(10,2) NOT NULL,
  `sales_personnel` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`product_number`, `store_number`, `sales_time`),
  CONSTRAINT `sales_record_ibfk_1` FOREIGN KEY (`product_number`) REFERENCES `product` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sales_record_ibfk_2` FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `store_performance` (
  `store_number` int(11) NOT NULL,
  `total_sales_amount` decimal(10,2) NOT NULL,
  `customer_satisfaction_level` int(11) DEFAULT NULL,
  PRIMARY KEY (`store_number`),
  CONSTRAINT `store_performance_ibfk_1` FOREIGN KEY (`store_number`) REFERENCES `store` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
