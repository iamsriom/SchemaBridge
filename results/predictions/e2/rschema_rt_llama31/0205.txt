-- Schema for database: rschema_0205
CREATE TABLE `performance_event` (
  `performance_event_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `event_time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`performance_event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `actor` (
  `actor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`actor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `performance_list` (
  `actor_id` int(11) NOT NULL,
  `performance_event_id` int(11) NOT NULL,
  PRIMARY KEY (`actor_id`, `performance_event_id`),
  CONSTRAINT `performance_list_ibfk_1` FOREIGN KEY (`actor_id`) REFERENCES `actor` (`actor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `performance_list_ibfk_2` FOREIGN KEY (`performance_event_id`) REFERENCES `performance_event` (`performance_event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
