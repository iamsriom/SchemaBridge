-- Schema for database: rschema_0206
CREATE TABLE `Region` (
  `Region_ID` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `administrator` varchar(255) DEFAULT NULL,
  `geographical_location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Region_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Flora_and_Fauna_Record` (
  `Record_ID` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `observation_time` datetime NOT NULL,
  `observer` varchar(255) NOT NULL,
  `species_type` varchar(255) NOT NULL,
  `Region_ID` int(11) NOT NULL,
  PRIMARY KEY (`Record_ID`),
  CONSTRAINT `Flora_and_Fauna_Record_ibfk_1` FOREIGN KEY (`Region_ID`) REFERENCES `Region` (`Region_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Environmental_Change_Record` (
  `Change_ID` int(11) NOT NULL,
  `change_time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `Region_ID` int(11) NOT NULL,
  PRIMARY KEY (`Change_ID`),
  CONSTRAINT `Environmental_Change_Record_ibfk_1` FOREIGN KEY (`Region_ID`) REFERENCES `Region` (`Region_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
