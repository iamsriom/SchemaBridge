-- Schema for database: rschema_0210
CREATE TABLE `conference` (
  `conference_id` int(11) NOT NULL,
  `date` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `theme` varchar(255) NOT NULL,
  PRIMARY KEY (`conference_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendee_information` (
  `attendee_id` int(11) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `name` varchar(255) NOT NULL,
  `organization` varchar(255) NOT NULL,
  `position` varchar(255) NOT NULL,
  PRIMARY KEY (`attendee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attendance` (
  `attendance_id` int(11) NOT NULL,
  `attendee_id` int(11) NOT NULL,
  `conference_id` int(11) NOT NULL,
  PRIMARY KEY (`attendance_id`),
  CONSTRAINT `attendance_ibfk_1` FOREIGN KEY (`attendee_id`) REFERENCES `attendee_information` (`attendee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `attendance_ibfk_2` FOREIGN KEY (`conference_id`) REFERENCES `conference` (`conference_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `speaker_presentation` (
  `presentation_id` int(11) NOT NULL,
  `attendee_id` int(11) NOT NULL,
  `conference_id` int(11) NOT NULL,
  `presentation_address` varchar(255) NOT NULL,
  `presentation_time` datetime NOT NULL,
  `topic` varchar(255) NOT NULL,
  PRIMARY KEY (`presentation_id`),
  CONSTRAINT `speaker_presentation_ibfk_1` FOREIGN KEY (`attendee_id`) REFERENCES `attendee_information` (`attendee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `speaker_presentation_ibfk_2` FOREIGN KEY (`conference_id`) REFERENCES `conference` (`conference_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
