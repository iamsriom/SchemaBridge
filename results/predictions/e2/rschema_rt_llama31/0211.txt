-- Schema for database: rschema_0211
CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `project_name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `consultant` (
  `consultant_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `professional_field` varchar(255) NOT NULL,
  PRIMARY KEY (`consultant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `service_record` (
  `service_id` int(11) NOT NULL,
  `consultant_id` int(11) NOT NULL,
  `project_id` int(11) NOT NULL,
  `service_content` text NOT NULL,
  `service_date` datetime NOT NULL,
  PRIMARY KEY (`service_id`),
  CONSTRAINT `service_record_ibfk_1` FOREIGN KEY (`consultant_id`) REFERENCES `consultant` (`consultant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `service_record_ibfk_2` FOREIGN KEY (`project_id`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
