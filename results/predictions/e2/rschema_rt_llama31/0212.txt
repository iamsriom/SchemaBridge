-- Schema for database: rschema_0212
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `branch` (
  `branch_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `income` decimal(10,2) NOT NULL,
  `manager_name` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`branch_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` int(11) NOT NULL,
  `branch_id` int(11) NOT NULL,
  `license_plate` varchar(20) NOT NULL,
  `model` varchar(255) NOT NULL,
  `rental_rate` decimal(10,2) NOT NULL,
  `year_of_manufacture` int(4) NOT NULL,
  PRIMARY KEY (`vehicle_id`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`branch_id`) REFERENCES `branch` (`branch_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_vehicle_rental` (
  `customer_id` int(11) NOT NULL,
  `vehicle_id` int(11) NOT NULL,
  `rental_fee` decimal(10,2) NOT NULL,
  `rental_time` datetime NOT NULL,
  `return_time` datetime DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `vehicle_id`),
  CONSTRAINT `customer_vehicle_rental_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_vehicle_rental_ibfk_2` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`vehicle_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
