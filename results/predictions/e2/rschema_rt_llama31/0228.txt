-- Schema for database: rschema_0228
CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `magazine` (
  `magazine_name` varchar(255) NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `theme` varchar(255) DEFAULT NULL,
  `publication_frequency` varchar(255) DEFAULT NULL,
  `website` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`magazine_name`, `publisher`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pricing` (
  `pricing_id` int(11) NOT NULL,
  `magazine_name` varchar(255) NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `package_name` varchar(255) NOT NULL,
  `period` varchar(255) DEFAULT NULL,
  `price` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`pricing_id`),
  CONSTRAINT `pricing_ibfk_1` FOREIGN KEY (`magazine_name`, `publisher`) REFERENCES `magazine` (`magazine_name`, `publisher`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_subscribes_to_magazine` (
  `customer_id` int(11) NOT NULL,
  `pricing_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `start_date` date DEFAULT NULL,
  PRIMARY KEY (`customer_id`, `creation_time`, `pricing_id`),
  CONSTRAINT `customer_subscribes_to_magazine_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `customer_subscribes_to_magazine_ibfk_2` FOREIGN KEY (`pricing_id`) REFERENCES `pricing` (`pricing_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
