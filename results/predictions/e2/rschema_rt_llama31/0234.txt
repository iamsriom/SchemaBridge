-- Schema for database: rschema_0234
CREATE TABLE `category` (
  `category_code` varchar(50) NOT NULL,
  `category_name` varchar(255) NOT NULL,
  PRIMARY KEY (`category_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tag` (
  `tag_id` int(11) NOT NULL,
  `primary_category` varchar(50) NOT NULL,
  `secondary_category` varchar(50) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`tag_id`),
  CONSTRAINT `tag_ibfk_1` FOREIGN KEY (`primary_category`) REFERENCES `category` (`category_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `tag_ibfk_2` FOREIGN KEY (`secondary_category`) REFERENCES `category` (`category_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
