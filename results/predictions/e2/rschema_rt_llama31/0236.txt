-- Schema for database: rschema_0236
CREATE TABLE `camera` (
  `camera_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `rtsp_stream_address` varchar(255) NOT NULL,
  `title` varchar(255) NOT NULL,
  `video_stream_path` varchar(255) NOT NULL,
  PRIMARY KEY (`camera_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `video_recording` (
  `video_id` int(11) NOT NULL,
  `camera_id` int(11) NOT NULL,
  `cover_path` varchar(255) NOT NULL,
  `duration` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `video_file_path` varchar(255) NOT NULL,
  PRIMARY KEY (`video_id`),
  CONSTRAINT `video_recording_ibfk_1` FOREIGN KEY (`camera_id`) REFERENCES `camera` (`camera_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `alarm` (
  `alarm_id` int(11) NOT NULL,
  `alarm_level` varchar(50) NOT NULL,
  `alarm_type` varchar(50) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `video_id` int(11) NOT NULL,
  PRIMARY KEY (`alarm_id`),
  CONSTRAINT `alarm_ibfk_1` FOREIGN KEY (`video_id`) REFERENCES `video_recording` (`video_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
