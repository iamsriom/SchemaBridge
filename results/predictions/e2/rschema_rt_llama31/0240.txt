-- Schema for database: rschema_0240
CREATE TABLE `Fleet` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `base_location` varchar(255) NOT NULL,
  `commander_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `Fleet_ibfk_1` FOREIGN KEY (`commander_id`) REFERENCES `Crew_Member` (`id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Ship` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `construction_date` datetime NOT NULL,
  `service_status` varchar(50) NOT NULL,
  `type` varchar(100) NOT NULL,
  `fleet_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `Ship_ibfk_1` FOREIGN KEY (`fleet_id`) REFERENCES `Fleet` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Crew_Member` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `length_of_service` int(11) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `position` varchar(100) NOT NULL,
  `fleet_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `Crew_Member_ibfk_1` FOREIGN KEY (`fleet_id`) REFERENCES `Fleet` (`id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Fleet_Mission_Execution` (
  `crew_member_id` int(11) NOT NULL,
  `fleet_id` int(11) NOT NULL,
  `execution_time` datetime NOT NULL,
  `is_primary` tinyint(1) NOT NULL,
  PRIMARY KEY (`crew_member_id`, `fleet_id`, `execution_time`),
  CONSTRAINT `Fleet_Mission_Execution_ibfk_1` FOREIGN KEY (`crew_member_id`) REFERENCES `Crew_Member` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Fleet_Mission_Execution_ibfk_2` FOREIGN KEY (`fleet_id`) REFERENCES `Fleet` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Ship_Supply` (
  `responsible_crew_member_id` int(11) NOT NULL,
  `ship_id` int(11) NOT NULL,
  `supply_items` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `supply_time` datetime NOT NULL,
  PRIMARY KEY (`responsible_crew_member_id`, `ship_id`, `supply_time`),
  CONSTRAINT `Ship_Supply_ibfk_1` FOREIGN KEY (`responsible_crew_member_id`) REFERENCES `Crew_Member` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Ship_Supply_ibfk_2` FOREIGN KEY (`ship_id`) REFERENCES `Ship` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
