-- Schema for database: rschema_0244
CREATE TABLE `users` (
  `account_name` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `user_role` varchar(50) NOT NULL,
  PRIMARY KEY (`account_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bus_routes` (
  `route_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `start_station` int(11) NOT NULL,
  `end_station` int(11) NOT NULL,
  `passing_stations` text DEFAULT NULL,
  `departure_time` datetime NOT NULL,
  PRIMARY KEY (`route_id`),
  CONSTRAINT `bus_routes_ibfk_1` FOREIGN KEY (`start_station`) REFERENCES `stations` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bus_routes_ibfk_2` FOREIGN KEY (`end_station`) REFERENCES `stations` (`station_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stations` (
  `station_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`station_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `announcements` (
  `announcement_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `publisher` varchar(255) NOT NULL,
  `publish_time` datetime NOT NULL,
  PRIMARY KEY (`announcement_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `questions` (
  `questioner_account` varchar(255) NOT NULL,
  `question_time` datetime NOT NULL,
  `question_content` text NOT NULL,
  `answer_content` text DEFAULT NULL,
  `responder_account` varchar(255) DEFAULT NULL,
  `answer_time` datetime DEFAULT NULL,
  PRIMARY KEY (`questioner_account`, `question_time`),
  CONSTRAINT `questions_ibfk_1` FOREIGN KEY (`questioner_account`) REFERENCES `users` (`account_name`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `questions_ibfk_2` FOREIGN KEY (`responder_account`) REFERENCES `users` (`account_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
