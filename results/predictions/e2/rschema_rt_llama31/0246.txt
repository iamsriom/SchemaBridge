-- Schema for database: rschema_0246
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `login_status` boolean NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crew` (
  `crew_id` int(11) NOT NULL AUTO_INCREMENT,
  `crew_name` varchar(255) NOT NULL,
  `crew_info` text DEFAULT NULL,
  `crew_members` text DEFAULT NULL,
  `login_status` boolean NOT NULL,
  PRIMARY KEY (`crew_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `film_information` (
  `info_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `content` text DEFAULT NULL,
  `publication_time` datetime NOT NULL,
  `publisher` varchar(255) NOT NULL,
  PRIMARY KEY (`info_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comment` (
  `film_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `comment_time` datetime NOT NULL,
  `comment_content` text NOT NULL,
  PRIMARY KEY (`film_id`, `user_id`, `comment_time`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`film_id`) REFERENCES `film_information` (`info_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `film_ranking` (
  `ranking_id` int(11) NOT NULL AUTO_INCREMENT,
  `film_id` int(11) NOT NULL,
  `ranking_position` int(11) NOT NULL,
  `update_time` datetime NOT NULL,
  PRIMARY KEY (`ranking_id`),
  CONSTRAINT `film_ranking_ibfk_1` FOREIGN KEY (`film_id`) REFERENCES `film_information` (`info_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
