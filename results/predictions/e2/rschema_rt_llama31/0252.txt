-- Schema for database: rschema_0252
CREATE TABLE `player` (
  `player_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  PRIMARY KEY (`player_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `game_record` (
  `game_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `rules` text NOT NULL,
  `max_players` int(11) NOT NULL,
  PRIMARY KEY (`game_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `player_communication` (
  `player_id_1` int(11) NOT NULL,
  `player_id_2` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `content` text NOT NULL,
  PRIMARY KEY (`player_id_1`, `player_id_2`, `start_time`),
  CONSTRAINT `player_communication_ibfk_1` FOREIGN KEY (`player_id_1`) REFERENCES `player` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `player_communication_ibfk_2` FOREIGN KEY (`player_id_2`) REFERENCES `player` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `single_player_game` (
  `player_id` int(11) NOT NULL,
  `game_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`player_id`, `game_id`, `start_time`),
  CONSTRAINT `single_player_game_ibfk_1` FOREIGN KEY (`player_id`) REFERENCES `player` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `single_player_game_ibfk_2` FOREIGN KEY (`game_id`) REFERENCES `game_record` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `two_player_battle_game` (
  `player_id_1` int(11) NOT NULL,
  `player_id_2` int(11) NOT NULL,
  `game_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `result` varchar(50) NOT NULL,
  PRIMARY KEY (`player_id_1`, `player_id_2`, `game_id`, `start_time`),
  CONSTRAINT `two_player_battle_game_ibfk_1` FOREIGN KEY (`player_id_1`) REFERENCES `player` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `two_player_battle_game_ibfk_2` FOREIGN KEY (`player_id_2`) REFERENCES `player` (`player_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `two_player_battle_game_ibfk_3` FOREIGN KEY (`game_id`) REFERENCES `game_record` (`game_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
