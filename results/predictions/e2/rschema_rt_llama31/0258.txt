-- Schema for database: rschema_0258
CREATE TABLE `donor` (
  `donor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` char(1) NOT NULL,
  `identity_card_number` varchar(50) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `address` varchar(255) NOT NULL,
  `donor_type` varchar(50) NOT NULL,
  PRIMARY KEY (`donor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fundraising_event` (
  `event_id` int(11) NOT NULL,
  `event_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `address` varchar(255) NOT NULL,
  `current_amount` decimal(10,2) NOT NULL,
  `total_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donation_record` (
  `donor_id` int(11) NOT NULL,
  `event_id` int(11) NOT NULL,
  `donation_time` datetime NOT NULL,
  `amount_donated` decimal(10,2) NOT NULL,
  `donation_method` varchar(50) NOT NULL,
  `message` text DEFAULT NULL,
  PRIMARY KEY (`donor_id`, `event_id`, `donation_time`),
  CONSTRAINT `donation_record_ibfk_1` FOREIGN KEY (`donor_id`) REFERENCES `donor` (`donor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `donation_record_ibfk_2` FOREIGN KEY (`event_id`) REFERENCES `fundraising_event` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
