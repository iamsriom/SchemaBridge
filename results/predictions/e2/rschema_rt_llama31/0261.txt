-- Schema for database: rschema_0261
CREATE TABLE `Address` (
  `Address_ID` int(11) NOT NULL AUTO_INCREMENT,
  `address_description` varchar(255) NOT NULL,
  `address_type` varchar(100) NOT NULL,
  PRIMARY KEY (`Address_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `City` (
  `City_ID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `region` varchar(100) NOT NULL,
  PRIMARY KEY (`City_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `District` (
  `District_ID` int(11) NOT NULL AUTO_INCREMENT,
  `City_ID` int(11) NOT NULL,
  `name` varchar(100) NOT NULL,
  `type` varchar(100) NOT NULL,
  PRIMARY KEY (`District_ID`),
  CONSTRAINT `district_ibfk_1` FOREIGN KEY (`City_ID`) REFERENCES `City` (`City_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Voter` (
  `Voter_ID` int(11) NOT NULL AUTO_INCREMENT,
  `Address_ID` int(11) NOT NULL,
  `Registration_district_ID` int(11) NOT NULL,
  `name` varchar(100) NOT NULL,
  `registration_status` varchar(50) NOT NULL,
  PRIMARY KEY (`Voter_ID`),
  CONSTRAINT `voter_ibfk_1` FOREIGN KEY (`Address_ID`) REFERENCES `Address` (`Address_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `voter_ibfk_2` FOREIGN KEY (`Registration_district_ID`) REFERENCES `District` (`District_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Town` (
  `Town_ID` int(11) NOT NULL AUTO_INCREMENT,
  `District_ID` int(11) NOT NULL,
  `name` varchar(100) NOT NULL,
  `region` varchar(100) NOT NULL,
  PRIMARY KEY (`Town_ID`),
  CONSTRAINT `town_ibfk_1` FOREIGN KEY (`District_ID`) REFERENCES `District` (`District_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
