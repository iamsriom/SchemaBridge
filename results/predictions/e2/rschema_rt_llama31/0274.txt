-- Schema for database: rschema_0274
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `contact_number` varchar(15) DEFAULT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `username` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_service_representative` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(100) DEFAULT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `work_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `feedback` (
  `feedback_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `feedback_content` text DEFAULT NULL,
  `submitted_time` datetime DEFAULT NULL,
  `processing_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`feedback_id`),
  CONSTRAINT `feedback_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `feedback_result` (
  `feedback_result_id` int(11) NOT NULL,
  `feedback_id` int(11) NOT NULL,
  `processing_result` varchar(255) DEFAULT NULL,
  `processed_time` datetime DEFAULT NULL,
  `processor_employee_id` int(11) NOT NULL,
  PRIMARY KEY (`feedback_result_id`),
  CONSTRAINT `feedback_result_ibfk_1` FOREIGN KEY (`feedback_id`) REFERENCES `feedback` (`feedback_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `feedback_result_ibfk_2` FOREIGN KEY (`processor_employee_id`) REFERENCES `customer_service_representative` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
