-- Schema for database: rschema_0276
CREATE TABLE `maintenance_technician` (
  `technician_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(100) NOT NULL,
  `employment_date` datetime NOT NULL,
  `team_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`technician_id`),
  CONSTRAINT `maintenance_technician_ibfk_1` FOREIGN KEY (`team_id`) REFERENCES `maintenance_team` (`team_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_team` (
  `team_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `team_leader_id` int(11) DEFAULT NULL,
  `responsibilities` text NOT NULL,
  `total_members` int(11) NOT NULL,
  PRIMARY KEY (`team_id`),
  CONSTRAINT `maintenance_team_ibfk_1` FOREIGN KEY (`team_leader_id`) REFERENCES `maintenance_technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_task` (
  `task_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  `maintenance_cost` decimal(10,2) NOT NULL,
  `duration` int(11) NOT NULL,
  `result` text NOT NULL,
  `time` datetime NOT NULL,
  `technician_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `maintenance_task_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_task_ibfk_2` FOREIGN KEY (`technician_id`) REFERENCES `maintenance_technician` (`technician_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_feedback` (
  `feedback_id` int(11) NOT NULL,
  `task_id` int(11) NOT NULL,
  `feedback_content` text NOT NULL,
  `resolution_result` varchar(255) NOT NULL,
  `customer_satisfaction` int(11) NOT NULL,
  PRIMARY KEY (`feedback_id`),
  CONSTRAINT `maintenance_feedback_ibfk_1` FOREIGN KEY (`task_id`) REFERENCES `maintenance_task` (`task_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
