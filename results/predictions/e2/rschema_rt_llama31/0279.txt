-- Schema for database: rschema_0279
CREATE TABLE `property_table` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `area` float NOT NULL,
  `sale_date` datetime DEFAULT NULL,
  `sale_price` float NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `analysis_report_table` (
  `report_id` int(11) NOT NULL,
  `report_date` datetime NOT NULL,
  `report_type` varchar(255) NOT NULL,
  `property_id` int(11) NOT NULL,
  PRIMARY KEY (`report_id`),
  CONSTRAINT `analysis_report_table_ibfk_1` FOREIGN KEY (`property_id`) REFERENCES `property_table` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
