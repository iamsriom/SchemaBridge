-- Schema for database: rschema_0283
CREATE TABLE `Book` (
  `BookID` int(11) NOT NULL,
  `ISBN` varchar(20) DEFAULT NULL,
  `category` varchar(100) DEFAULT NULL,
  `title` varchar(255) DEFAULT NULL,
  `publication_date` datetime DEFAULT NULL,
  `AuthorID` int(11) NOT NULL,
  `EditorID` int(11) NOT NULL,
  PRIMARY KEY (`BookID`),
  CONSTRAINT `book_ibfk_1` FOREIGN KEY (`AuthorID`) REFERENCES `Author` (`AuthorID`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `book_ibfk_2` FOREIGN KEY (`EditorID`) REFERENCES `Editor` (`EditorID`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Author` (
  `AuthorID` int(11) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `name` varchar(100) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`AuthorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Editor` (
  `EditorID` int(11) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `name` varchar(100) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`EditorID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
