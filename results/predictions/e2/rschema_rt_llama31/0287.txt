-- Schema for database: rschema_0287
CREATE TABLE `park` (
  `park_id` int(11) NOT NULL,
  `area` float NOT NULL,
  `location` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `maintenance_person` varchar(255) NOT NULL,
  PRIMARY KEY (`park_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `park_maintenance` (
  `maintenance_id` int(11) NOT NULL,
  `park_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `personnel` varchar(255) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  PRIMARY KEY (`maintenance_id`),
  CONSTRAINT `park_maintenance_ibfk_1` FOREIGN KEY (`park_id`) REFERENCES `park` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `park_ecological_record` (
  `record_number` int(11) NOT NULL,
  `park_id` int(11) NOT NULL,
  PRIMARY KEY (`record_number`),
  CONSTRAINT `park_ecological_record_ibfk_1` FOREIGN KEY (`park_id`) REFERENCES `park` (`park_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plant` (
  `plant_id` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`plant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `plant_growth_record` (
  `plant_id` int(11) NOT NULL,
  `record_number` int(11) NOT NULL,
  `growth_status` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`plant_id`, `record_number`),
  CONSTRAINT `plant_growth_record_ibfk_1` FOREIGN KEY (`plant_id`) REFERENCES `plant` (`plant_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `plant_growth_record_ibfk_2` FOREIGN KEY (`record_number`) REFERENCES `park_ecological_record` (`record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
