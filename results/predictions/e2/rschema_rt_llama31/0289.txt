-- Schema for database: rschema_0289
CREATE TABLE `operator` (
  `operator_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `department` varchar(255) NOT NULL,
  `contact_details` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`operator_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `production_process` (
  `process_id` int(11) NOT NULL AUTO_INCREMENT,
  `process_name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime DEFAULT NULL,
  `operator_id` int(11) NOT NULL,
  PRIMARY KEY (`process_id`),
  CONSTRAINT `production_process_ibfk_1` FOREIGN KEY (`operator_id`) REFERENCES `operator` (`operator_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `monitoring_record` (
  `process_id` int(11) NOT NULL,
  `recording_time` datetime NOT NULL,
  `parameter` varchar(255) NOT NULL,
  PRIMARY KEY (`process_id`, `recording_time`),
  CONSTRAINT `monitoring_record_ibfk_1` FOREIGN KEY (`process_id`) REFERENCES `production_process` (`process_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
