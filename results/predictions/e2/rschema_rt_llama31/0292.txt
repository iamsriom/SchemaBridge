-- Schema for database: rschema_0292
CREATE TABLE `facility_table` (
  `facility_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`facility_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_personnel_table` (
  `personnel_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `facility_maintenance_table` (
  `personnel_id` int(11) NOT NULL,
  `facility_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `description` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  PRIMARY KEY (`personnel_id`, `facility_id`, `maintenance_time`),
  CONSTRAINT `facility_maintenance_ibfk_1` FOREIGN KEY (`facility_id`) REFERENCES `facility_table` (`facility_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `facility_maintenance_ibfk_2` FOREIGN KEY (`personnel_id`) REFERENCES `maintenance_personnel_table` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
