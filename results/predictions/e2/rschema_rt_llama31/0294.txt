-- Schema for database: rschema_0294
CREATE TABLE `branch_office` (
  `branch_office_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  PRIMARY KEY (`branch_office_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer` (
  `customer_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `id_number` varchar(50) NOT NULL,
  PRIMARY KEY (`customer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `account` (
  `account_number` int(11) NOT NULL,
  `account_type` varchar(50) NOT NULL,
  `balance` decimal(10,2) NOT NULL,
  `opening_date` datetime NOT NULL,
  `branch_office_id` int(11) NOT NULL,
  `customer_id` int(11) NOT NULL,
  PRIMARY KEY (`account_number`),
  CONSTRAINT `account_ibfk_1` FOREIGN KEY (`branch_office_id`) REFERENCES `branch_office` (`branch_office_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `account_ibfk_2` FOREIGN KEY (`customer_id`) REFERENCES `customer` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_record` (
  `transaction_id` int(11) NOT NULL,
  `account_number` int(11) NOT NULL,
  `transaction_amount` decimal(10,2) NOT NULL,
  `location` varchar(255) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `transaction_type` varchar(50) NOT NULL,
  PRIMARY KEY (`transaction_id`),
  CONSTRAINT `transaction_record_ibfk_1` FOREIGN KEY (`account_number`) REFERENCES `account` (`account_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
