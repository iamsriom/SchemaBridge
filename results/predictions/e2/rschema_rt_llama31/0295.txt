-- Schema for database: rschema_0295
CREATE TABLE `trader` (
  `trader_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `hire_date` datetime NOT NULL,
  `contact_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`trader_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `client` (
  `client_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_number` varchar(15) DEFAULT NULL,
  `id_number` varchar(50) NOT NULL,
  `risk_level` varchar(50) NOT NULL,
  PRIMARY KEY (`client_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction` (
  `trader_id` int(11) NOT NULL,
  `client_id` int(11) NOT NULL,
  `futures_code` varchar(50) NOT NULL,
  `transaction_price` decimal(10,2) NOT NULL,
  `quantity` int(11) NOT NULL,
  `time` datetime NOT NULL,
  `type` varchar(50) NOT NULL,
  PRIMARY KEY (`trader_id`, `client_id`, `futures_code`),
  CONSTRAINT `transaction_ibfk_1` FOREIGN KEY (`trader_id`) REFERENCES `trader` (`trader_id`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `transaction_ibfk_2` FOREIGN KEY (`client_id`) REFERENCES `client` (`client_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
