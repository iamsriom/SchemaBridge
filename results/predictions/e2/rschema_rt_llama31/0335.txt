-- Schema for database: rschema_0335
CREATE TABLE `bucket` (
  `bucket_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`bucket_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `permission_access` (
  `permission_id` int(11) NOT NULL,
  `access_account` varchar(255) NOT NULL,
  `access_password` varchar(255) DEFAULT NULL,
  `bucket_id` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `creator` varchar(255) NOT NULL,
  `description` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `type` varchar(50) NOT NULL,
  `tenant` varchar(255) NOT NULL,
  PRIMARY KEY (`permission_id`),
  CONSTRAINT `permission_access_ibfk_1` FOREIGN KEY (`bucket_id`) REFERENCES `bucket` (`bucket_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
