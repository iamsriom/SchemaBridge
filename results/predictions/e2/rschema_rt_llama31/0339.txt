-- Schema for database: rschema_0339
CREATE TABLE `project` (
  `project_id` int(11) NOT NULL,
  `avatar_url` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`project_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api` (
  `identifier` int(11) NOT NULL,
  `path` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `belonging_project` int(11) NOT NULL,
  PRIMARY KEY (`identifier`),
  CONSTRAINT `api_ibfk_1` FOREIGN KEY (`belonging_project`) REFERENCES `project` (`project_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `api_response_record` (
  `api_identifier` int(11) NOT NULL,
  `call_time` datetime NOT NULL,
  `response_time` int(11) NOT NULL,
  PRIMARY KEY (`api_identifier`, `call_time`),
  CONSTRAINT `api_response_record_ibfk_1` FOREIGN KEY (`api_identifier`) REFERENCES `api` (`identifier`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
