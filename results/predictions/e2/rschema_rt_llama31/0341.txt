-- Schema for database: rschema_0341
CREATE TABLE `Anonymous_Comment` (
  `Comment_ID` int(11) NOT NULL AUTO_INCREMENT,
  `comment_content` text NOT NULL,
  `user_url` varchar(255) DEFAULT NULL,
  `publication_time` datetime NOT NULL,
  `storage_format` varchar(50) NOT NULL,
  PRIMARY KEY (`Comment_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Information` (
  `User_ID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`User_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Normal_Comment` (
  `Comment_ID` int(11) NOT NULL AUTO_INCREMENT,
  `comment_content` text NOT NULL,
  `publication_time` datetime NOT NULL,
  `storage_format` varchar(50) NOT NULL,
  `User_ID` int(11) NOT NULL,
  PRIMARY KEY (`Comment_ID`),
  CONSTRAINT `normal_comment_ibfk_1` FOREIGN KEY (`User_ID`) REFERENCES `User_Information` (`User_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
