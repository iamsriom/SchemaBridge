-- Schema for database: rschema_0345
CREATE TABLE `subject` (
  `subject_code` varchar(10) NOT NULL,
  `name` varchar(255) NOT NULL,
  `credits` int(11) NOT NULL,
  PRIMARY KEY (`subject_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `professor` (
  `professor_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  PRIMARY KEY (`professor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student` (
  `student_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `average_grade` decimal(5,2) DEFAULT NULL,
  `credits_completed` int(11) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subject_equivalence` (
  `subject_code_1` varchar(10) NOT NULL,
  `subject_code_2` varchar(10) NOT NULL,
  PRIMARY KEY (`subject_code_1`, `subject_code_2`),
  CONSTRAINT `subject_equivalence_ibfk_1` FOREIGN KEY (`subject_code_1`) REFERENCES `subject` (`subject_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `subject_equivalence_ibfk_2` FOREIGN KEY (`subject_code_2`) REFERENCES `subject` (`subject_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subject_prerequisites` (
  `subject_code` varchar(10) NOT NULL,
  `prerequisite_code` varchar(10) NOT NULL,
  PRIMARY KEY (`subject_code`, `prerequisite_code`),
  CONSTRAINT `subject_prerequisites_ibfk_1` FOREIGN KEY (`subject_code`) REFERENCES `subject` (`subject_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `subject_prerequisites_ibfk_2` FOREIGN KEY (`prerequisite_code`) REFERENCES `subject` (`subject_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_code` varchar(10) NOT NULL,
  `name` varchar(255) NOT NULL,
  `professor_id` int(11) NOT NULL,
  `total_credits` int(11) NOT NULL,
  PRIMARY KEY (`course_code`),
  CONSTRAINT `course_ibfk_1` FOREIGN KEY (`professor_id`) REFERENCES `professor` (`professor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course_assignment` (
  `course_code` varchar(10) NOT NULL,
  `subject_code` varchar(10) NOT NULL,
  `semester` char(6) NOT NULL,
  PRIMARY KEY (`course_code`, `subject_code`, `semester`),
  CONSTRAINT `course_assignment_ibfk_1` FOREIGN KEY (`course_code`) REFERENCES `course` (`course_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `course_assignment_ibfk_2` FOREIGN KEY (`subject_code`) REFERENCES `subject` (`subject_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `student_course_enrollment` (
  `student_id` int(11) NOT NULL,
  `course_code` varchar(10) NOT NULL,
  PRIMARY KEY (`student_id`, `course_code`),
  CONSTRAINT `student_course_enrollment_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `student_course_enrollment_ibfk_2` FOREIGN KEY (`course_code`) REFERENCES `course` (`course_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
