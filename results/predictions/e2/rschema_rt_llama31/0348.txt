-- Schema for database: rschema_0348
CREATE TABLE `plugin_execution_record` (
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `plugin_version_number` varchar(255) NOT NULL,
  `sql_record_number` int(11) NOT NULL,
  PRIMARY KEY (`start_time`, `end_time`, `plugin_version_number`),
  CONSTRAINT `plugin_execution_record_ibfk_1` FOREIGN KEY (`sql_record_number`) REFERENCES `sql_record` (`sql_record_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sql_record` (
  `sql_record_number` int(11) NOT NULL AUTO_INCREMENT,
  `sql_statement` text NOT NULL,
  PRIMARY KEY (`sql_record_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
