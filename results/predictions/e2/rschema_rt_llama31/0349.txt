-- Schema for database: rschema_0349
CREATE TABLE `file_collection` (
  `file_collection_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`file_collection_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `file` (
  `file_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `size` int(11) NOT NULL,
  `status` varchar(50) DEFAULT NULL,
  `storage_location` varchar(255) DEFAULT NULL,
  `type` varchar(50) DEFAULT NULL,
  `version_number` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `last_update_time` datetime NOT NULL,
  `file_collection_id` int(11) NOT NULL,
  PRIMARY KEY (`file_id`),
  CONSTRAINT `file_ibfk_1` FOREIGN KEY (`file_collection_id`) REFERENCES `file_collection` (`file_collection_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `replica` (
  `replica_id` int(11) NOT NULL AUTO_INCREMENT,
  `replica_device_name` varchar(255) NOT NULL,
  `replica_index` int(11) NOT NULL,
  `replica_last_update_time` datetime NOT NULL,
  `replica_node_name` varchar(255) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `version_number` int(11) NOT NULL,
  `file_id` int(11) NOT NULL,
  PRIMARY KEY (`replica_id`),
  CONSTRAINT `replica_ibfk_1` FOREIGN KEY (`file_id`) REFERENCES `file` (`file_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
