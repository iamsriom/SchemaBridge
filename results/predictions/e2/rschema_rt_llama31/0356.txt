-- Schema for database: rschema_0356
CREATE TABLE `work_item` (
  `work_item_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `description` text DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`work_item_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `work_item_attributes` (
  `attributes_id` int(11) NOT NULL,
  `work_item_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `value` text DEFAULT NULL,
  `link_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`attributes_id`, `work_item_id`),
  CONSTRAINT `work_item_attributes_ibfk_1` FOREIGN KEY (`work_item_id`) REFERENCES `work_item` (`work_item_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
