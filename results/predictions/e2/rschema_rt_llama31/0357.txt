-- Schema for database: rschema_0357
CREATE TABLE `rental_location` (
  `location_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `business_hours` varchar(255) NOT NULL,
  PRIMARY KEY (`location_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_category` (
  `category_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `weekday_rental_price` decimal(10,2) NOT NULL,
  `weekend_rental_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`category_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `license_plate_number` varchar(15) NOT NULL,
  `brand` varchar(255) NOT NULL,
  `model` varchar(255) NOT NULL,
  `photo_path` varchar(255) DEFAULT NULL,
  `location_id` int(11) NOT NULL,
  `category_id` int(11) NOT NULL,
  PRIMARY KEY (`license_plate_number`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`location_id`) REFERENCES `rental_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `vehicle_ibfk_2` FOREIGN KEY (`category_id`) REFERENCES `vehicle_category` (`category_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `email` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `postal_code` varchar(10) NOT NULL,
  `city` varchar(255) NOT NULL,
  `credit_card_info` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_vehicle_reservation` (
  `reservation_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `vehicle_id` varchar(15) NOT NULL,
  `reservation_date` datetime NOT NULL,
  `return_date` datetime NOT NULL,
  PRIMARY KEY (`reservation_id`),
  CONSTRAINT `user_vehicle_reservation_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_vehicle_reservation_ibfk_2` FOREIGN KEY (`vehicle_id`) REFERENCES `vehicle` (`license_plate_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `news_and_featured_content` (
  `content_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `release_date` datetime NOT NULL,
  PRIMARY KEY (`content_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle_current_location` (
  `license_plate_number` varchar(15) NOT NULL,
  `record_time` datetime NOT NULL,
  `location_id` int(11) NOT NULL,
  PRIMARY KEY (`license_plate_number`, `record_time`),
  CONSTRAINT `vehicle_current_location_ibfk_1` FOREIGN KEY (`license_plate_number`) REFERENCES `vehicle` (`license_plate_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `vehicle_current_location_ibfk_2` FOREIGN KEY (`location_id`) REFERENCES `rental_location` (`location_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
