-- Schema for database: rschema_0358
CREATE TABLE `scheduled_task` (
  `task_id` int(11) NOT NULL,
  `revision_number` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  `lock_expiration_time` datetime DEFAULT NULL,
  `lock_owner` varchar(255) DEFAULT NULL,
  `execution_id` int(11) NOT NULL,
  `process_instance_id` int(11) NOT NULL,
  `process_definition_id` int(11) NOT NULL,
  `retries` int(11) DEFAULT NULL,
  `exception_stack_id` int(11) DEFAULT NULL,
  `exception_message` text DEFAULT NULL,
  `due_date` datetime DEFAULT NULL,
  `repeat_frequency` int(11) DEFAULT NULL,
  `handler_type` varchar(255) DEFAULT NULL,
  `handler_configuration` text DEFAULT NULL,
  `tenant_id` int(11) NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `scheduled_task_ibfk_1` FOREIGN KEY (`execution_id`) REFERENCES `execution_instance` (`execution_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `scheduled_task_ibfk_2` FOREIGN KEY (`process_instance_id`) REFERENCES `execution_instance` (`process_instance_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `execution_instance` (
  `execution_id` int(11) NOT NULL,
  `process_instance_id` int(11) NOT NULL,
  `process_definition_id` int(11) NOT NULL,
  PRIMARY KEY (`execution_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
