-- Schema for database: rschema_0359
CREATE TABLE `user` (
  `account_name` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_info` varchar(255) NOT NULL,
  `avatar_url` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`account_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `event` (
  `event_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text NOT NULL,
  `event_time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `event_type` varchar(255) NOT NULL,
  `creator` varchar(255) NOT NULL,
  PRIMARY KEY (`event_id`),
  CONSTRAINT `event_ibfk_1` FOREIGN KEY (`creator`) REFERENCES `user` (`account_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `message` (
  `message_id` int(11) NOT NULL AUTO_INCREMENT,
  `event_id` int(11) NOT NULL,
  `content` text NOT NULL,
  `sender` varchar(255) NOT NULL,
  PRIMARY KEY (`message_id`),
  CONSTRAINT `message_ibfk_1` FOREIGN KEY (`event_id`) REFERENCES `event` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `message_ibfk_2` FOREIGN KEY (`sender`) REFERENCES `user` (`account_name`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
