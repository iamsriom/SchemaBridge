-- Schema for database: rschema_0360
CREATE TABLE `Log` (
  `Log_ID` int(11) NOT NULL AUTO_INCREMENT,
  `category` varchar(255) NOT NULL,
  `content` text NOT NULL,
  `created_at` datetime NOT NULL,
  `last_modified` datetime NOT NULL,
  `severity` varchar(50) NOT NULL,
  PRIMARY KEY (`Log_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operator` (
  `ID` int(11) NOT NULL AUTO_INCREMENT,
  `account_name` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operation_Log` (
  `Log_ID` int(11) NOT NULL,
  `User_ID` int(11) NOT NULL,
  `Operation_Time` datetime NOT NULL,
  `operation_content` text NOT NULL,
  `operation_type` varchar(100) NOT NULL,
  PRIMARY KEY (`Log_ID`, `User_ID`, `Operation_Time`),
  CONSTRAINT `operation_log_ibfk_1` FOREIGN KEY (`Log_ID`) REFERENCES `Log` (`Log_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `operation_log_ibfk_2` FOREIGN KEY (`User_ID`) REFERENCES `Operator` (`ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
