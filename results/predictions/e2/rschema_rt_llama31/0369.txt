-- Schema for database: rschema_0369
CREATE TABLE `Content` (
  `content_id` int(11) NOT NULL AUTO_INCREMENT,
  `description` text DEFAULT NULL,
  `type` varchar(50) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime DEFAULT NULL,
  `title` varchar(255) NOT NULL,
  PRIMARY KEY (`content_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `email` varchar(255) NOT NULL,
  `role` varchar(50) DEFAULT NULL,
  `username` varchar(100) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Operation_Log` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `content_id` int(11) NOT NULL,
  `comment` text DEFAULT NULL,
  `operation_time` datetime NOT NULL,
  `operation_type` varchar(50) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`log_id`),
  CONSTRAINT `operation_log_ibfk_1` FOREIGN KEY (`content_id`) REFERENCES `Content` (`content_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `operation_log_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User_Content_Modification` (
  `content_id` int(11) NOT NULL,
  `modification_time` datetime NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`content_id`, `modification_time`, `user_id`),
  CONSTRAINT `user_content_modification_ibfk_1` FOREIGN KEY (`content_id`) REFERENCES `Content` (`content_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_content_modification_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `User` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
