-- Schema for database: rschema_0377
CREATE TABLE `article` (
  `post_id` int(11) NOT NULL,
  `article_title` varchar(255) NOT NULL,
  PRIMARY KEY (`post_id`, `article_title`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tag` (
  `tag_id` int(11) NOT NULL,
  `tag_category` varchar(255) NOT NULL,
  PRIMARY KEY (`tag_id`, `tag_category`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `article_tag_association` (
  `post_id` int(11) NOT NULL,
  `tag_id` int(11) NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`post_id`, `tag_id`),
  CONSTRAINT `article_tag_association_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `article` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `article_tag_association_ibfk_2` FOREIGN KEY (`tag_id`) REFERENCES `tag` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
