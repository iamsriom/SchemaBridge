-- Schema for database: rschema_0386
CREATE TABLE `user` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `public_key` varchar(255) NOT NULL,
  `username` varchar(255) NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `block` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `block_hash` varchar(255) NOT NULL,
  `size` int(11) NOT NULL,
  `creation_time` datetime NOT NULL,
  `previous_block_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `block_ibfk_1` FOREIGN KEY (`previous_block_id`) REFERENCES `block` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `transaction_record` (
  `sender_id` int(11) NOT NULL,
  `receiver_id` int(11) NOT NULL,
  `block_id` int(11) NOT NULL,
  `transaction_time` datetime NOT NULL,
  `operation_type` varchar(255) NOT NULL,
  `operation_content` text NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`sender_id`, `receiver_id`, `transaction_time`, `block_id`),
  CONSTRAINT `transaction_record_ibfk_1` FOREIGN KEY (`sender_id`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_record_ibfk_2` FOREIGN KEY (`receiver_id`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `transaction_record_ibfk_3` FOREIGN KEY (`block_id`) REFERENCES `block` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
