-- Schema for database: rschema_0396
CREATE TABLE `user_table` (
  `username` varchar(255) NOT NULL,
  `nickname` varchar(255) DEFAULT NULL,
  `password` varchar(255) DEFAULT NULL,
  `profile_picture` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `guide_table` (
  `guide_title` varchar(255) NOT NULL,
  `category` varchar(255) DEFAULT NULL,
  `content` text DEFAULT NULL,
  `tags` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`guide_title`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `restaurant_table` (
  `restaurant_name` varchar(255) NOT NULL,
  `image` varchar(255) DEFAULT NULL,
  `introduction` text DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`restaurant_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hotel_table` (
  `hotel_name` varchar(255) NOT NULL,
  `image` varchar(255) DEFAULT NULL,
  `introduction` text DEFAULT NULL,
  `location` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`hotel_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `scenic_spot_table` (
  `scenic_spot_name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `days_of_operation` varchar(255) DEFAULT NULL,
  `image` varchar(255) DEFAULT NULL,
  `introduction` text DEFAULT NULL,
  `opening_hours` varchar(255) DEFAULT NULL,
  `tips` text DEFAULT NULL,
  PRIMARY KEY (`scenic_spot_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `route_table` (
  `route_name` varchar(255) NOT NULL,
  `content` text DEFAULT NULL,
  `introduction` text DEFAULT NULL,
  `tags` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`route_name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `write_guide_table` (
  `username` varchar(255) NOT NULL,
  `guide_title` varchar(255) NOT NULL,
  PRIMARY KEY (`username`, `guide_title`),
  CONSTRAINT `write_guide_ibfk_1` FOREIGN KEY (`username`) REFERENCES `user_table` (`username`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `write_guide_ibfk_2` FOREIGN KEY (`guide_title`) REFERENCES `guide_table` (`guide_title`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
