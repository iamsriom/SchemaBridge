-- Schema for database: rschema_0401
CREATE TABLE `students` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `grade_level` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `enrollment_time` datetime NOT NULL,
  PRIMARY KEY (`student_id`),
  CONSTRAINT `students_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `classes` (
  `class_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `grade_level` int(11) NOT NULL,
  `major` varchar(255) NOT NULL,
  `head_teacher_id` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  `class_monitor_id` int(11) NOT NULL,
  PRIMARY KEY (`class_id`),
  CONSTRAINT `classes_ibfk_1` FOREIGN KEY (`head_teacher_id`) REFERENCES `teachers` (`teacher_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `classes_ibfk_2` FOREIGN KEY (`class_monitor_id`) REFERENCES `students` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `classes_ibfk_3` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `departments` (
  `department_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `size` int(11) NOT NULL,
  `subordinate_majors` varchar(255) NOT NULL,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `teachers` (
  `teacher_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) NOT NULL,
  `grade_level` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `entry_time` datetime NOT NULL,
  PRIMARY KEY (`teacher_id`),
  CONSTRAINT `teachers_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `departments` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
