-- Schema for database: rschema_0426
CREATE TABLE `user` (
  `account` varchar(255) NOT NULL,
  `bio` text DEFAULT NULL,
  `nickname` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`account`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `attraction` (
  `attraction_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `picture_url` varchar(255) DEFAULT NULL,
  `reference_price` decimal(10,2) NOT NULL,
  `text_description` text DEFAULT NULL,
  PRIMARY KEY (`attraction_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_bookmark` (
  `account` varchar(255) NOT NULL,
  `attraction_id` int(11) NOT NULL,
  PRIMARY KEY (`account`, `attraction_id`),
  CONSTRAINT `user_bookmark_ibfk_1` FOREIGN KEY (`account`) REFERENCES `user` (`account`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_bookmark_ibfk_2` FOREIGN KEY (`attraction_id`) REFERENCES `attraction` (`attraction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ticket_reservation_record` (
  `account` varchar(255) NOT NULL,
  `attraction_id` int(11) NOT NULL,
  `number_of_tickets` int(11) NOT NULL,
  `reservation_time` datetime NOT NULL,
  `ticket_purchase_time` datetime NOT NULL,
  PRIMARY KEY (`account`, `attraction_id`),
  CONSTRAINT `ticket_reservation_record_ibfk_1` FOREIGN KEY (`account`) REFERENCES `user` (`account`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ticket_reservation_record_ibfk_2` FOREIGN KEY (`attraction_id`) REFERENCES `attraction` (`attraction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_comment` (
  `account` varchar(255) NOT NULL,
  `attraction_id` int(11) NOT NULL,
  `comment_content` text NOT NULL,
  `status` varchar(50) NOT NULL,
  `comment_time` datetime NOT NULL,
  PRIMARY KEY (`account`, `attraction_id`),
  CONSTRAINT `user_comment_ibfk_1` FOREIGN KEY (`account`) REFERENCES `user` (`account`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_comment_ibfk_2` FOREIGN KEY (`attraction_id`) REFERENCES `attraction` (`attraction_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
