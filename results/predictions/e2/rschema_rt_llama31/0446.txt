-- Schema for database: rschema_0446
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `registration_time` datetime NOT NULL,
  `last_login_time` datetime DEFAULT NULL,
  `role` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sensors` (
  `sensor_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`sensor_id`),
  CONSTRAINT `sensors_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `meteorological_data` (
  `data_id` int(11) NOT NULL AUTO_INCREMENT,
  `sensor_id` int(11) NOT NULL,
  `temperature` float NOT NULL,
  `humidity` float NOT NULL,
  `rainfall` float NOT NULL,
  `wind_speed` float NOT NULL,
  `light_intensity` float NOT NULL,
  PRIMARY KEY (`data_id`),
  CONSTRAINT `meteorological_data_ibfk_1` FOREIGN KEY (`sensor_id`) REFERENCES `sensors` (`sensor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crops` (
  `crop_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `model_id` int(11) NOT NULL,
  PRIMARY KEY (`crop_id`),
  CONSTRAINT `crops_ibfk_1` FOREIGN KEY (`model_id`) REFERENCES `growth_model` (`model_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `growth_model` (
  `model_id` int(11) NOT NULL AUTO_INCREMENT,
  `schema` varchar(255) NOT NULL,
  `parameters` text NOT NULL,
  `creation_time` datetime NOT NULL,
  PRIMARY KEY (`model_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_operation` (
  `operation_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `crop_id` int(11) NOT NULL,
  `operation_type` varchar(50) NOT NULL,
  `operation_time` datetime NOT NULL,
  PRIMARY KEY (`operation_id`),
  CONSTRAINT `user_operation_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_operation_ibfk_2` FOREIGN KEY (`crop_id`) REFERENCES `crops` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `system_performance_evaluation` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `crop_id` int(11) NOT NULL,
  `yield` float NOT NULL,
  `evaluation_time` datetime NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  CONSTRAINT `system_performance_evaluation_ibfk_1` FOREIGN KEY (`crop_id`) REFERENCES `crops` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `model_performance_evaluation` (
  `evaluation_id` int(11) NOT NULL AUTO_INCREMENT,
  `model_id` int(11) NOT NULL,
  `accuracy` float NOT NULL,
  `evaluation_time` datetime NOT NULL,
  PRIMARY KEY (`evaluation_id`),
  CONSTRAINT `model_performance_evaluation_ibfk_1` FOREIGN KEY (`model_id`) REFERENCES `growth_model` (`model_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user_water_and_fertilizer_supply` (
  `supply_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `crop_id` int(11) NOT NULL,
  `sensor_id` int(11) NOT NULL,
  `supply_type` varchar(50) NOT NULL,
  `supply_time` datetime NOT NULL,
  PRIMARY KEY (`supply_id`),
  CONSTRAINT `user_water_and_fertilizer_supply_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_water_and_fertilizer_supply_ibfk_2` FOREIGN KEY (`crop_id`) REFERENCES `crops` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `user_water_and_fertilizer_supply_ibfk_3` FOREIGN KEY (`sensor_id`) REFERENCES `sensors` (`sensor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
