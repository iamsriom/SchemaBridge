CREATE TABLE `supply_request` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `chain_supermarket_id` int(11) NOT NULL,
  `requested_product_id` int(11) NOT NULL,
  `request_time` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `supply_request_ibfk_1` FOREIGN KEY (`chain_supermarket_id`) REFERENCES `chain_supermarket` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `supply_request_ibfk_2` FOREIGN KEY (`requested_product_id`) REFERENCES `requested_product` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `product` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `inventory_quantity` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `requested_product` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `supply_request_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `requested_product_ibfk_1` FOREIGN KEY (`supply_request_id`) REFERENCES `supply_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `requested_product_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `chain_supermarket` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_details` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `distribution_order` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `supply_request_id` int(11) NOT NULL,
  `order_time` datetime NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `distribution_order_ibfk_1` FOREIGN KEY (`supply_request_id`) REFERENCES `supply_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchase_order` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `order_date` datetime NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `purchased_product` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `purchase_order_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `purchased_product_ibfk_1` FOREIGN KEY (`purchase_order_id`) REFERENCES `purchase_order` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `purchased_product_ibfk_2` FOREIGN KEY (`product_id`) REFERENCES `product` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
