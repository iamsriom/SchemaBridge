-- Schema for database: rschema_0455
CREATE TABLE `parking_space` (
  `parking_space_id` int(11) NOT NULL,
  `occupied_vehicle` varchar(20) DEFAULT NULL,
  `entry_id` int(11) DEFAULT NULL,
  `exit_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`parking_space_id`),
  CONSTRAINT `parking_space_ibfk_1` FOREIGN KEY (`entry_id`) REFERENCES `entry` (`entry_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_space_ibfk_2` FOREIGN KEY (`exit_id`) REFERENCES `exit` (`exit_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `parking_space_ibfk_3` FOREIGN KEY (`occupied_vehicle`) REFERENCES `vehicle` (`license_plate_number`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `entry` (
  `entry_id` int(11) NOT NULL,
  `status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`entry_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `exit` (
  `exit_id` int(11) NOT NULL,
  `status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`exit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `vehicle` (
  `license_plate_number` varchar(20) NOT NULL,
  `entry_time` datetime DEFAULT NULL,
  `exit_time` datetime DEFAULT NULL,
  PRIMARY KEY (`license_plate_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
