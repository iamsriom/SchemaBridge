-- Schema for database: rschema_0478
CREATE TABLE `user` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `phone_number` varchar(15) NOT NULL,
  `name` varchar(255) NOT NULL,
  `credit` decimal(10,2) NOT NULL DEFAULT 0.00,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bicycle` (
  `bicycle_id` int(11) NOT NULL AUTO_INCREMENT,
  `location` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `unlock_code` varchar(50) NOT NULL,
  PRIMARY KEY (`bicycle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `trip_record` (
  `bicycle_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `payment_status` varchar(50) NOT NULL,
  `billing_rule_id` int(11) NOT NULL,
  PRIMARY KEY (`bicycle_id`, `user_id`, `start_time`),
  CONSTRAINT `trip_record_ibfk_1` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycle` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trip_record_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `trip_record_ibfk_3` FOREIGN KEY (`billing_rule_id`) REFERENCES `billing_rule` (`rule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fault_record` (
  `bicycle_id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `reporting_time` datetime NOT NULL,
  `fault_description` text NOT NULL,
  PRIMARY KEY (`bicycle_id`, `user_id`, `reporting_time`),
  CONSTRAINT `fault_record_ibfk_1` FOREIGN KEY (`bicycle_id`) REFERENCES `bicycle` (`bicycle_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fault_record_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `user` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `billing_rule` (
  `rule_id` int(11) NOT NULL AUTO_INCREMENT,
  `billing_method` varchar(255) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  `minimum_fee` decimal(10,2) NOT NULL,
  PRIMARY KEY (`rule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
