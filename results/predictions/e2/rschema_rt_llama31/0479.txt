-- Schema for database: rschema_0479
CREATE TABLE `subsidiary` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_address` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `manager_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `subsidiary_ibfk_1` FOREIGN KEY (`manager_id`) REFERENCES `employee` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `business_department` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `contact_address` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `subsidiary_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `business_department_ibfk_1` FOREIGN KEY (`subsidiary_id`) REFERENCES `subsidiary` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `family_member_name` varchar(255) DEFAULT NULL,
  `relationship` varchar(50) DEFAULT NULL,
  `phone_number` varchar(15) NOT NULL,
  `position` varchar(100) NOT NULL,
  `department_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `business_department` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
