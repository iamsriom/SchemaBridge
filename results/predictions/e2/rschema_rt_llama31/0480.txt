-- Schema for database: rschema_0480
CREATE TABLE `property` (
  `property_id` int(11) NOT NULL,
  `address` varchar(255) NOT NULL,
  `built_area` float NOT NULL,
  `construction_year` int(11) NOT NULL,
  `is_apartment` tinyint(1) NOT NULL,
  `property_certificate` varchar(255) DEFAULT NULL,
  `photos` text DEFAULT NULL,
  `sale_price` decimal(10,2) NOT NULL,
  `sales_status` varchar(50) NOT NULL,
  PRIMARY KEY (`property_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `agent` (
  `agent_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `contact_number` varchar(20) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  PRIMARY KEY (`agent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role` varchar(50) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `responsibility` (
  `agent_id` int(11) NOT NULL,
  `property_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  PRIMARY KEY (`agent_id`, `property_id`),
  CONSTRAINT `responsibility_ibfk_1` FOREIGN KEY (`agent_id`) REFERENCES `agent` (`agent_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `responsibility_ibfk_2` FOREIGN KEY (`property_id`) REFERENCES `property` (`property_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
