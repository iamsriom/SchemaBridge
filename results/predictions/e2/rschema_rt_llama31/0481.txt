-- Schema for database: rschema_0481
CREATE TABLE `crop` (
  `crop_id` int(11) NOT NULL,
  `crop_name` varchar(255) NOT NULL,
  `crop_type` varchar(255) NOT NULL,
  PRIMARY KEY (`crop_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sensor` (
  `sensor_id` int(11) NOT NULL,
  `location` varchar(255) NOT NULL,
  `light_intensity` float DEFAULT NULL,
  `rainfall` float DEFAULT NULL,
  `temperature` float DEFAULT NULL,
  PRIMARY KEY (`sensor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sensor_monitoring_crop` (
  `crop_id` int(11) NOT NULL,
  `sensor_id` int(11) NOT NULL,
  `collection_time` datetime NOT NULL,
  `data_type` varchar(255) NOT NULL,
  `value` float NOT NULL,
  PRIMARY KEY (`crop_id`, `sensor_id`),
  CONSTRAINT `sensor_monitoring_crop_ibfk_1` FOREIGN KEY (`crop_id`) REFERENCES `crop` (`crop_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `sensor_monitoring_crop_ibfk_2` FOREIGN KEY (`sensor_id`) REFERENCES `sensor` (`sensor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
