-- Schema for database: rschema_0482
CREATE TABLE `client` (
  `client_number` int(11) NOT NULL,
  `contact_person` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `type` varchar(100) NOT NULL,
  `contact_phone_number` varchar(15) NOT NULL,
  `discount_rate` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`client_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `license_plate_number` varchar(10) NOT NULL,
  `client_number` int(11) NOT NULL,
  `category` varchar(100) NOT NULL,
  `color` varchar(50) NOT NULL,
  `model` varchar(100) NOT NULL,
  PRIMARY KEY (`license_plate_number`),
  CONSTRAINT `vehicle_ibfk_1` FOREIGN KEY (`client_number`) REFERENCES `client` (`client_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `department` (
  `department_number` int(11) NOT NULL,
  `department_name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `supervisor_employee_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`department_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `phone_number` varchar(15) NOT NULL,
  `home_address` varchar(255) NOT NULL,
  `position` varchar(100) NOT NULL,
  `department_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`employee_number`),
  CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`department_number`) REFERENCES `department` (`department_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `maintenance_task` (
  `order_number` int(11) NOT NULL,
  `license_plate_number` varchar(10) NOT NULL,
  `maintenance_content` varchar(255) NOT NULL,
  `man_hours` decimal(5,2) NOT NULL,
  `employee_number` int(11) NOT NULL,
  PRIMARY KEY (`order_number`),
  CONSTRAINT `maintenance_task_ibfk_1` FOREIGN KEY (`license_plate_number`) REFERENCES `vehicle` (`license_plate_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `maintenance_task_ibfk_2` FOREIGN KEY (`employee_number`) REFERENCES `employee` (`employee_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
