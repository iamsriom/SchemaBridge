-- Schema for database: rschema_0488
CREATE TABLE `department` (
  `department_id` int(11) NOT NULL,
  `description` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `email_address` varchar(255) DEFAULT NULL,
  `allows_customer_work_orders` tinyint(1) NOT NULL DEFAULT 0,
  `hides_department` tinyint(1) NOT NULL DEFAULT 0,
  `requires_feedback` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`department_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_table` (
  `employee_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `gender` char(1) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`employee_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `email_configuration` (
  `configuration_id` int(11) NOT NULL,
  `department_id` int(11) NOT NULL,
  `host_address` varchar(255) NOT NULL,
  `login_password` varchar(255) NOT NULL,
  `login_username` varchar(255) NOT NULL,
  `port_number` int(11) NOT NULL,
  PRIMARY KEY (`configuration_id`),
  CONSTRAINT `email_configuration_ibfk_1` FOREIGN KEY (`department_id`) REFERENCES `department` (`department_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_maintenance_email_server` (
  `employee_id` int(11) NOT NULL,
  `maintenance_time` datetime NOT NULL,
  `server_id` int(11) NOT NULL,
  PRIMARY KEY (`employee_id`, `maintenance_time`, `server_id`),
  CONSTRAINT `employee_maintenance_email_server_ibfk_1` FOREIGN KEY (`employee_id`) REFERENCES `employee_table` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_maintenance_email_server_ibfk_2` FOREIGN KEY (`server_id`) REFERENCES `email_configuration` (`configuration_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
