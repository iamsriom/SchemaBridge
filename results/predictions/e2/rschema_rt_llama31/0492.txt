-- Schema for database: rschema_0492
CREATE TABLE `department_table` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `position_table` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `salary` decimal(10,2) NOT NULL,
  `belonging_department` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `position_table_ibfk_1` FOREIGN KEY (`belonging_department`) REFERENCES `department_table` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee_table` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `position_id` int(11) NOT NULL,
  `superior_employee` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `employee_table_ibfk_1` FOREIGN KEY (`position_id`) REFERENCES `position_table` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `employee_table_ibfk_2` FOREIGN KEY (`superior_employee`) REFERENCES `employee_table` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
