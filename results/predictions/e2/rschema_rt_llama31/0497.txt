-- Schema for database: rschema_0497
CREATE TABLE `student` (
  `student_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `class` varchar(255) DEFAULT NULL,
  `contact_information` varchar(255) DEFAULT NULL,
  `gender` char(6) DEFAULT NULL,
  `grade` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `event` (
  `event_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `event_time` datetime DEFAULT NULL,
  `competition_rules` text DEFAULT NULL,
  `number_of_participants` int(11) DEFAULT NULL,
  PRIMARY KEY (`event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `competition_status` (
  `student_id` int(11) NOT NULL,
  `event_id` int(11) NOT NULL,
  `registration_time` datetime NOT NULL,
  `competition_result` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`student_id`, `event_id`),
  CONSTRAINT `competition_status_ibfk_1` FOREIGN KEY (`student_id`) REFERENCES `student` (`student_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `competition_status_ibfk_2` FOREIGN KEY (`event_id`) REFERENCES `event` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
