-- Schema for database: rschema_0504
CREATE TABLE `cultural_relic` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `category` varchar(255) NOT NULL,
  `excavation_site` varchar(255) NOT NULL,
  `year_of_origin` int(11) NOT NULL,
  `last_repair_time` datetime DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staff` (
  `staff_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  PRIMARY KEY (`staff_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cultural_relic_exhibition` (
  `relic_id` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  PRIMARY KEY (`relic_id`, `start_time`),
  CONSTRAINT `cultural_relic_exhibition_ibfk_1` FOREIGN KEY (`relic_id`) REFERENCES `cultural_relic` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cultural_relic_repair_record` (
  `staff_id` int(11) NOT NULL,
  `relic_id` int(11) NOT NULL,
  `repair_start_time` datetime NOT NULL,
  `repair_end_time` datetime NOT NULL,
  `method` varchar(255) NOT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`staff_id`, `relic_id`, `repair_start_time`),
  CONSTRAINT `cultural_relic_repair_record_ibfk_1` FOREIGN KEY (`staff_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `cultural_relic_repair_record_ibfk_2` FOREIGN KEY (`relic_id`) REFERENCES `cultural_relic` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cultural_relic_monitoring_record` (
  `relic_id` int(11) NOT NULL,
  `monitoring_time` datetime NOT NULL,
  `person_in_charge_id` int(11) NOT NULL,
  `air_quality` varchar(255) NOT NULL,
  `humidity` float NOT NULL,
  `light` float NOT NULL,
  `temperature` float NOT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`relic_id`, `monitoring_time`, `person_in_charge_id`),
  CONSTRAINT `cultural_relic_monitoring_record_ibfk_1` FOREIGN KEY (`relic_id`) REFERENCES `cultural_relic` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `cultural_relic_monitoring_record_ibfk_2` FOREIGN KEY (`person_in_charge_id`) REFERENCES `staff` (`staff_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
