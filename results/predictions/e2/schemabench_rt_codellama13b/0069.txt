-- Schema for database: schemabench_0069
CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `oauth_applications` (
  `application_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `uid` varchar(255) NOT NULL,
  `secret` varchar(255) NOT NULL,
  `redirect_uri` varchar(255) NOT NULL,
  `scopes` varchar(255) NOT NULL,
  `confidentiality` boolean NOT NULL,
  `creation_date` datetime DEFAULT NULL,
  `update_date` datetime DEFAULT NULL,
  PRIMARY KEY (`application_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `oauth_access_grants` (
  `grant_id` int(11) NOT NULL,
  `resource_owner_id` int(11) NOT NULL,
  `application_id` int(11) NOT NULL,
  `token` varchar(255) NOT NULL,
  `expiration_date` datetime NOT NULL,
  `redirect_uri` varchar(255) DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `revocation_date` datetime DEFAULT NULL,
  `scopes` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`grant_id`),
  CONSTRAINT `oauth_access_grants_ibfk_1` FOREIGN KEY (`resource_owner_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `oauth_access_grants_ibfk_2` FOREIGN KEY (`application_id`) REFERENCES `oauth_applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `oauth_access_tokens` (
  `token_id` int(11) NOT NULL,
  `resource_owner_id` int(11) DEFAULT NULL,
  `application_id` int(11) DEFAULT NULL,
  `token` varchar(255) DEFAULT NULL,
  `refresh_token` varchar(255) DEFAULT NULL,
  `expiration_date` datetime DEFAULT NULL,
  `revocation_date` datetime DEFAULT NULL,
  `creation_date` datetime DEFAULT NULL,
  `scopes` varchar(255) DEFAULT NULL,
  `previous_refresh_token` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`token_id`),
  CONSTRAINT `oauth_access_tokens_ibfk_1` FOREIGN KEY (`resource_owner_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `oauth_access_tokens_ibfk_2` FOREIGN KEY (`application_id`) REFERENCES `oauth_applications` (`application_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
