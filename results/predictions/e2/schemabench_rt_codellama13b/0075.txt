  `id` int(11) NOT NULL AUTO_INCREMENT,
  `details` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organizations_organization` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `created_at` datetime NOT NULL,
  `modified_at` datetime NOT NULL,
  `slug` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organizations_organizationowner` (
  `organization_id` int(11) NOT NULL,
  `owner_id` int(11) NOT NULL,
  `created_at` datetime NOT NULL,
  `modified_at` datetime NOT NULL,
  PRIMARY KEY (`organization_id`, `owner_id`),
  CONSTRAINT `organizations_organizationowner_ibfk_1` FOREIGN KEY (`organization_id`) REFERENCES `organizations_organization` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `organizations_organizationowner_ibfk_2` FOREIGN KEY (`owner_id`) REFERENCES `auth_user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organizations_organizationuser` (
  `organization_id` int(11) NOT NULL,

  `modified_at` datetime NOT NULL,
  PRIMARY KEY (`organization_id`, `user_id`),
  CONSTRAINT `organizations_organizationuser_ibfk_1` FOREIGN KEY (`organization_id`) REFERENCES `organizations_organization` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
