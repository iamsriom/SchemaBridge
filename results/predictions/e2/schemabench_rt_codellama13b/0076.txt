-- Schema for database: schemabench_0076
CREATE TABLE `hosts` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `ssh_user` varchar(255) DEFAULT NULL,
  `port` int(11) DEFAULT NULL,
  `ip` varchar(255) DEFAULT NULL,
  `custom_properties` json NOT NULL,
  `created_at` timestamp NOT NULL,
  `updated_at` timestamp NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `checks` (
  `host_id` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  `status` varchar(255) DEFAULT NULL,
  `enabled` boolean NOT NULL,
  `last_run_message` text DEFAULT NULL,
  `last_run_output` json DEFAULT NULL,
  `last_ran_at` time DEFAULT NULL,
  `next_run_in_minutes` int(11) DEFAULT NULL,
  `started_throttling_failing_notifications_at` timestamp DEFAULT NULL,
  `custom_properties` json DEFAULT NULL,
  `created_at` timestamp NOT NULL,
  `updated_at` timestamp NOT NULL,
  PRIMARY KEY (`host_id`, `type`),
  CONSTRAINT `checks_ibfk_1` FOREIGN KEY (`host_id`) REFERENCES `hosts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
