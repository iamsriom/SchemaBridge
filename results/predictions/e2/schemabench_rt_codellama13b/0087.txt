CREATE TABLE `customers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `billable_id` int(11) NOT NULL,
  `billable_type` varchar(255) NOT NULL,
  `trial_end_date` datetime DEFAULT NULL,
  `updated_at` datetime NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`billable_id`) REFERENCES `users_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subscriptions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `billable_id` int(11) NOT NULL,
  `billable_type` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `paddle_id` varchar(255) NOT NULL,
  `paddle_status` varchar(255) NOT NULL,
  `paddle_plan` varchar(255) NOT NULL,
  `quantity` int(11) NOT NULL,
  `trial_end_date` datetime DEFAULT NULL,
  `paused_from_date` datetime DEFAULT NULL,
  `ends_at_date` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `subscriptions_ibfk_1` FOREIGN KEY (`billable_id`) REFERENCES `users_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `receipts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `billable_id` int(11) NOT NULL,
  `billable_type` varchar(255) NOT NULL,
  `paddle_subscription_id` int(11) NOT NULL,
  `checkout_id` varchar(255) NOT NULL,
  `order_id` varchar(255) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `tax` decimal(10,2) NOT NULL,
  `currency` varchar(10) NOT NULL,
  `quantity` int(11) NOT NULL,
  `receipt_url` varchar(255) NOT NULL,
  `paid_at` datetime DEFAULT NULL,
  `updated_at` datetime NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `receipts_ibfk_1` FOREIGN KEY (`billable_id`) REFERENCES `users_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users_teams` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
