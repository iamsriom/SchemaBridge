-- Schema for database: schemabench_0094
CREATE TABLE `atom` (
  `id` int(11) NOT NULL,
  `molecule_id` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `atom_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `atom_ibfk_2` FOREIGN KEY (`id`) REFERENCES `bond` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bond` (
  `id` int(11) NOT NULL,
  `atom1_id` int(11) NOT NULL,
  `atom2_id` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `bond_ibfk_1` FOREIGN KEY (`atom1_id`) REFERENCES `atom` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bond_ibfk_2` FOREIGN KEY (`atom2_id`) REFERENCES `atom` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gmember` (
  `id` int(11) NOT NULL,
  `atom_id` int(11) NOT NULL,
  `group_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `gmember_ibfk_1` FOREIGN KEY (`atom_id`) REFERENCES `atom` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `gmember_ibfk_2` FOREIGN KEY (`group_id`) REFERENCES `group` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `group` (
  `id` int(11) NOT NULL,
  `type` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `molecule` (
  `id` int(11) NOT NULL,
  `activity` varchar(255) NOT NULL,
  `logP` float NOT NULL,
  `molecular_weight` float NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
