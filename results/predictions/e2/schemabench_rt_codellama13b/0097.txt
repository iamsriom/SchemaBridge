-- Schema for database: schemabench_0097
CREATE TABLE `schools` (
  `school_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_info` varchar(255) DEFAULT NULL,
  `status` enum('open', 'closed') NOT NULL,
  `funding_type` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`school_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `frpm` (
  `frpm_id` int(11) NOT NULL AUTO_INCREMENT,
  `school_id` int(11) NOT NULL,
  `students_eligible` int(11) NOT NULL,
  `students_receiving` int(11) NOT NULL,
  `percentage_eligible` decimal(5,2) NOT NULL,
  PRIMARY KEY (`frpm_id`),
  CONSTRAINT `frpm_ibfk_1` FOREIGN KEY (`school_id`) REFERENCES `schools` (`school_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `satscores` (
  `satscore_id` int(11) NOT NULL AUTO_INCREMENT,
  `school_id` int(11) NOT NULL,
  `students_taking` int(11) NOT NULL,
  `average_reading` decimal(5,2) NOT NULL,
  `average_math` decimal(5,2) NOT NULL,
  `average_writing` decimal(5,2) NOT NULL,
  `percentage_above` decimal(5,2) NOT NULL,
  `percentage_below` decimal(5,2) NOT NULL,
  PRIMARY KEY (`satscore_id`),
  CONSTRAINT `satscores_ibfk_1` FOREIGN KEY (`school_id`) REFERENCES `schools` (`school_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
