-- Schema for database: schemabench_0102
CREATE TABLE `breweries` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `city` varchar(255) NOT NULL,
  `state` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `id` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beers` (
  `id` int(11) NOT NULL,
  `brewery_id` int(11) NOT NULL,
  `abv` float DEFAULT NULL,
  `ibu` decimal(10,2) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `style` varchar(255) NOT NULL,
  `ounces` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `beers_ibfk_1` FOREIGN KEY (`brewery_id`) REFERENCES `breweries` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
