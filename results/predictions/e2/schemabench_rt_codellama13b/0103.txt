  `category_number` int(11) NOT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `charge` (
  `charge_number` int(11) NOT NULL,

  `provider_number` int(11) NOT NULL,
  `category_number` int(11) NOT NULL,
  `charge_date` datetime NOT NULL,
  `charge_amount` decimal(10,2) NOT NULL,
  `statement_number` int(11) NOT NULL,
  `charge_code` varchar(50) NOT NULL,
  PRIMARY KEY (`charge_number`),
  CONSTRAINT `charge_ibfk_1` FOREIGN KEY (`member_number`) REFERENCES `member` (`member_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `charge_ibfk_2` FOREIGN KEY (`provider_number`) REFERENCES `provider` (`provider_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `charge_ibfk_3` FOREIGN KEY (`category_number`) REFERENCES `category` (`category_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `charge_ibfk_4` FOREIGN KEY (`statement_number`) REFERENCES `statement` (`statement_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
