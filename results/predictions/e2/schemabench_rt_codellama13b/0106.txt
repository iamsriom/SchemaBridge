CREATE TABLE `circuits` (
  `circuit_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `country` varchar(255) NOT NULL,
  `latitude` decimal(9,6) NOT NULL,
  `longitude` decimal(9,6) NOT NULL,
  `altitude` int(11) DEFAULT NULL,
  PRIMARY KEY (`circuit_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `result_id` int(11) NOT NULL AUTO_INCREMENT,

  `race_id` int(11) NOT NULL,
  `points` decimal(5,2) NOT NULL,
  `status` varchar(50) NOT NULL,
  PRIMARY KEY (`result_id`),
  CONSTRAINT `constructor_results_ibfk_1` FOREIGN KEY (`constructor_id`) REFERENCES `constructors` (`constructor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `constructor_results_ibfk_2` FOREIGN KEY (`race_id`) REFERENCES `races` (`race_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
