-- Schema for database: schemabench_0108
CREATE TABLE `Classification` (
  `GeneID` int(11) NOT NULL,
  `Localization` varchar(255) DEFAULT NULL,
  `Essential` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`GeneID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Genes` (
  `GeneID` int(11) NOT NULL,
  `Localization` varchar(255) NOT NULL,
  `Class` varchar(255) DEFAULT NULL,
  `Complex` varchar(255) DEFAULT NULL,
  `Phenotype` varchar(255) DEFAULT NULL,
  `Motif` varchar(255) DEFAULT NULL,
  `Chromosome` varchar(255) DEFAULT NULL,
  `Function` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`GeneID`, `Localization`),
  UNIQUE KEY `GeneID_UNIQUE` (`GeneID`),
  CONSTRAINT `genes_ibfk_1` FOREIGN KEY (`GeneID`) REFERENCES `Classification` (`GeneID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Interactions` (
  `GeneID1` int(11) NOT NULL,
  `GeneID2` int(11) NOT NULL,
  `InteractionType` varchar(255) DEFAULT NULL,
  `ExpressionCorrelation` float DEFAULT NULL,
  PRIMARY KEY (`GeneID1`, `GeneID2`),
  UNIQUE KEY `GeneID1_GeneID2_UNIQUE` (`GeneID1`, `GeneID2`),
  CONSTRAINT `interactions_ibfk_1` FOREIGN KEY (`GeneID1`) REFERENCES `Classification` (`GeneID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `interactions_ibfk_2` FOREIGN KEY (`GeneID2`) REFERENCES `Classification` (`GeneID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
