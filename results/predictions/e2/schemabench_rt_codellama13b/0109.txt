-- Schema for database: schemabench_0109
CREATE TABLE `go_methods` (
  `order_method_code` varchar(10) NOT NULL,
  `order_method_type` varchar(255) NOT NULL,
  PRIMARY KEY (`order_method_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_retailers` (
  `retailer_code` varchar(10) NOT NULL,
  `retailer_name` varchar(255) NOT NULL,
  `type` varchar(50) NOT NULL,
  `country` varchar(100) NOT NULL,
  PRIMARY KEY (`retailer_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_products` (
  `product_number` varchar(10) NOT NULL,
  `product_line` varchar(100) NOT NULL,
  `product_type` varchar(100) NOT NULL,
  `product` varchar(255) NOT NULL,
  `product_brand` varchar(100) NOT NULL,
  `product_color` varchar(50) NOT NULL,
  `unit_cost` decimal(10,2) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`product_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_daily_sales` (
  `retailer_code` varchar(10) NOT NULL,
  `product_number` varchar(10) NOT NULL,
  `order_method_code` varchar(10) NOT NULL,
  `date` date NOT NULL,
  `quantity` int(11) NOT NULL,
  `unit_price` decimal(10,2) NOT NULL,
  `unit_sale_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`retailer_code`, `product_number`),
  CONSTRAINT `go_daily_sales_ibfk_1` FOREIGN KEY (`order_method_code`) REFERENCES `go_methods` (`order_method_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_daily_sales_ibfk_2` FOREIGN KEY (`retailer_code`) REFERENCES `go_retailers` (`retailer_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_1k` (
  `retailer_code` varchar(10) NOT NULL,
  `product_number` varchar(10) NOT NULL,
  `date` date NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`retailer_code`, `product_number`, `date`),
  CONSTRAINT `go_1k_ibfk_1` FOREIGN KEY (`retailer_code`) REFERENCES `go_retailers` (`retailer_code`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_1k_ibfk_2` FOREIGN KEY (`product_number`) REFERENCES `go_products` (`product_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
