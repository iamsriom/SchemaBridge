CREATE TABLE `awards` (
  `award_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `effective_date` datetime NOT NULL,
  `expiration_date` datetime NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `instrument` varchar(255) NOT NULL,
  `organization_code` varchar(50) NOT NULL,
  `program_officer` varchar(255) NOT NULL,
  `abstract_narration` text DEFAULT NULL,
  `min_amd_letter_date` date DEFAULT NULL,
  `max_amd_letter_date` date DEFAULT NULL,
  `arra_amount` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`award_id`),
  CONSTRAINT `awards_ibfk_1` FOREIGN KEY (`organization_code`) REFERENCES `organization` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `foa_info` (
  `code` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `foa_info_awards` (
  `award_id` int(11) NOT NULL,
  `funding_agency_code` varchar(50) NOT NULL,
  PRIMARY KEY (`award_id`, `funding_agency_code`),
  CONSTRAINT `foa_info_awards_ibfk_1` FOREIGN KEY (`award_id`) REFERENCES `awards` (`award_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `foa_info_awards_ibfk_2` FOREIGN KEY (`funding_agency_code`) REFERENCES `foa_info` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `institution` (
  `name` varchar(255) NOT NULL,
  `city` varchar(100) NOT NULL,
  `zip_code` varchar(20) NOT NULL,
  `contact` varchar(255) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `country` varchar(100) NOT NULL,
  `state` varchar(100) NOT NULL,
  `state_code` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`name`, `zip_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `institution_awards` (
  `award_id` int(11) NOT NULL,
  `institution_name` varchar(255) NOT NULL,
  `institution_zip_code` varchar(20) NOT NULL,
  PRIMARY KEY (`award_id`, `institution_name`, `institution_zip_code`),
  CONSTRAINT `institution_awards_ibfk_1` FOREIGN KEY (`award_id`) REFERENCES `awards` (`award_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `institution_awards_ibfk_2` FOREIGN KEY (`institution_name`, `institution_zip_code`) REFERENCES `institution` (`name`, `zip_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
