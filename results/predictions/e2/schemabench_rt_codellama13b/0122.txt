CREATE TABLE `authors` (
  `au_id` int(11) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `first_name` varchar(255) NOT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `address` varchar(255) DEFAULT NULL,
  `city` varchar(100) DEFAULT NULL,
  `state` char(2) DEFAULT NULL,
  `zip` varchar(10) DEFAULT NULL,
  `contract` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`au_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `discounts` (
  `discount_id` int(11) NOT NULL,
  `discount_type` varchar(100) NOT NULL,
  `store_id` int(11) NOT NULL,
  `low_quantity` int(11) NOT NULL,
  `high_quantity` int(11) NOT NULL,
  `discount_percentage` decimal(5,2) NOT NULL,
  PRIMARY KEY (`discount_id`),
  FOREIGN KEY (`store_id`) REFERENCES `stores` (`stor_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `emp_id` int(11) NOT NULL,
  `first_name` varchar(255) NOT NULL,
  `middle_initial` char(1) DEFAULT NULL,
  `last_name` varchar(255) NOT NULL,
  `job_id` int(11) NOT NULL,
  `job_level` int(11) NOT NULL,
  `publisher_id` int(11) NOT NULL,
  `hire_date` datetime DEFAULT NULL,
  `job_description` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`emp_id`),
  FOREIGN KEY (`job_id`) REFERENCES `jobs` (`job_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`publisher_id`) REFERENCES `publishers` (`pub_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `jobs` (
  `job_id` int(11) NOT NULL,
  `job_description` varchar(255) NOT NULL,
  `min_level` int(11) NOT NULL,
  `max_level` int(11) NOT NULL,
  PRIMARY KEY (`job_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pub_info` (
  `pub_id` int(11) NOT NULL,
  `logo` varchar(255) DEFAULT NULL,
  `publication_info` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pub_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `publishers` (
  `pub_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `city` varchar(100) DEFAULT NULL,
  `state` char(2) DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`pub_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `roysched` (
  `title_id` int(11) NOT NULL,
  `low_range` decimal(10,2) NOT NULL,
  `high_range` decimal(10,2) NOT NULL,
  `royalty_percentage` decimal(5,2) NOT NULL,
  PRIMARY KEY (`title_id`),
  FOREIGN KEY (`title_id`) REFERENCES `titles` (`title_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `sales` (
  `stor_id` int(11) NOT NULL,
  `ord_num` int(11) NOT NULL,
  `order_date` datetime NOT NULL,
  `quantity` int(11) NOT NULL,
  `payment_terms` varchar(100) DEFAULT NULL,
  `title_id` int(11) NOT NULL,
  `total_price` decimal(10,2) NOT NULL,
  PRIMARY KEY (`stor_id`, `ord_num`, `title_id`),
  FOREIGN KEY (`stor_id`) REFERENCES `stores` (`stor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`title_id`) REFERENCES `titles` (`title_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stores` (
  `stor_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `city` varchar(100) DEFAULT NULL,
  `state` char(2) DEFAULT NULL,
  `zip` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`stor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `titleauthor` (
  `au_id` int(11) NOT NULL,
  `title_id` int(11) NOT NULL,
  `author_order` int(11) NOT NULL,
  `royalty_percentage` decimal(5,2) NOT NULL,
  PRIMARY KEY (`au_id`, `title_id`),
  FOREIGN KEY (`au_id`) REFERENCES `authors` (`au_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`title_id`) REFERENCES `titles` (`title_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `titles` (
  `title_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `type` varchar(100) DEFAULT NULL,
  `publisher_id` int(11) NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `advance` decimal(10,2) DEFAULT NULL,
  `royalty` decimal(10,2) DEFAULT NULL,
  `ytd_sales` decimal(10,2) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  `publication_date` datetime DEFAULT NULL,
  PRIMARY KEY (`title_id`),
  FOREIGN KEY (`publisher_id`) REFERENCES `publishers` (`pub_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
