-- Schema for database: schemabench_0128
CREATE TABLE `businesses` (
  `business_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `city` varchar(100) NOT NULL,
  `postal_code` varchar(20) NOT NULL,
  `latitude` decimal(10,6) DEFAULT NULL,
  `longitude` decimal(10,6) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `tax_code` varchar(50) DEFAULT NULL,
  `business_certificate` varchar(50) DEFAULT NULL,
  `application_date` datetime DEFAULT NULL,
  `owner_name` varchar(255) NOT NULL,
  `owner_address` varchar(255) NOT NULL,
  `owner_city` varchar(100) NOT NULL,
  `owner_state` varchar(100) NOT NULL,
  `owner_zip` varchar(20) NOT NULL,
  PRIMARY KEY (`business_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `inspections` (
  `business_id` int(11) NOT NULL,
  `score` int(11) DEFAULT NULL,
  `date` datetime DEFAULT NULL,
  `type` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`business_id`),
  CONSTRAINT `inspections_ibfk_1` FOREIGN KEY (`business_id`) REFERENCES `businesses` (`business_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `violations` (
  `business_id` int(11) NOT NULL,
  `date` datetime DEFAULT NULL,
  `violation_type_id` int(11) DEFAULT NULL,
  `risk_category` varchar(100) DEFAULT NULL,
  `description` text DEFAULT NULL,
  PRIMARY KEY (`business_id`),
  CONSTRAINT `violations_ibfk_1` FOREIGN KEY (`business_id`) REFERENCES `businesses` (`business_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
