CREATE TABLE `customers` (
  `customer_key` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `nation_key` int(11) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `account_balance` decimal(10,2) DEFAULT NULL,
  `market_segment` varchar(50) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`customer_key`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`nation_key`) REFERENCES `nations` (`nation_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `line_items` (
  `order_key` int(11) NOT NULL,
  `part_key` int(11) NOT NULL,
  `supplier_key` int(11) NOT NULL,
  `line_number` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `extended_price` decimal(10,2) DEFAULT NULL,
  `discount` decimal(5,2) DEFAULT NULL,
  `tax` decimal(5,2) DEFAULT NULL,
  `return_flag` char(1) DEFAULT NULL,
  `line_status` varchar(20) DEFAULT NULL,
  `ship_date` datetime DEFAULT NULL,
  `commit_date` datetime DEFAULT NULL,
  `receipt_date` datetime DEFAULT NULL,
  `ship_instruction` text DEFAULT NULL,
  `ship_mode` varchar(50) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`order_key`, `line_number`),
  CONSTRAINT `line_items_ibfk_1` FOREIGN KEY (`order_key`) REFERENCES `orders` (`order_key`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `line_items_ibfk_2` FOREIGN KEY (`part_key`) REFERENCES `parts` (`part_key`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `line_items_ibfk_3` FOREIGN KEY (`supplier_key`) REFERENCES `suppliers` (`supplier_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `nations` (
  `nation_key` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `region_key` int(11) NOT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`nation_key`),
  CONSTRAINT `nations_ibfk_1` FOREIGN KEY (`region_key`) REFERENCES `regions` (`region_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
  `order_key` int(11) NOT NULL,
  `customer_key` int(11) NOT NULL,
  `order_status` varchar(50) DEFAULT NULL,
  `total_price` decimal(10,2) DEFAULT NULL,
  `order_date` datetime DEFAULT NULL,
  `order_priority` varchar(20) DEFAULT NULL,
  `clerk` varchar(255) DEFAULT NULL,
  `ship_priority` int(11) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`order_key`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`customer_key`) REFERENCES `customers` (`customer_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `parts` (
  `part_key` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `manufacturer` varchar(255) DEFAULT NULL,
  `brand` varchar(255) DEFAULT NULL,
  `type` varchar(50) DEFAULT NULL,
  `size` varchar(50) DEFAULT NULL,
  `container` varchar(50) DEFAULT NULL,
  `retail_price` decimal(10,2) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`part_key`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `part_suppliers` (
  `part_key` int(11) NOT NULL,
  `supplier_key` int(11) NOT NULL,
  `available_quantity` int(11) DEFAULT NULL,
  `supply_cost` decimal(10,2) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`part_key`, `supplier_key`),
  CONSTRAINT `part_suppliers_ibfk_1` FOREIGN KEY (`part_key`) REFERENCES `parts` (`part_key`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `part_suppliers_ibfk_2` FOREIGN KEY (`supplier_key`) REFERENCES `suppliers` (`supplier_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `regions` (
  `region_key` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`region_key`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `suppliers` (
  `supplier_key` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `nation_key` int(11) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `account_balance` decimal(10,2) DEFAULT NULL,
  `comment` text DEFAULT NULL,
  PRIMARY KEY (`supplier_key`),
  CONSTRAINT `suppliers_ibfk_1` FOREIGN KEY (`nation_key`) REFERENCES `nations` (`nation_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
