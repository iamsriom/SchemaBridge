CREATE TABLE `key` (
  `store_number` int(11) NOT NULL,
  `station_number` int(11) NOT NULL,
  PRIMARY KEY (`store_number`),
  CONSTRAINT `key_ibfk_1` FOREIGN KEY (`station_number`) REFERENCES `station` (`station_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `station` (
  `station_number` int(11) NOT NULL,
  PRIMARY KEY (`station_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `train` (
  `store_number` int(11) NOT NULL,
  `date` date NOT NULL,
  `item_number` int(11) NOT NULL,
  `units_sold` int(11) NOT NULL,
  PRIMARY KEY (`store_number`, `date`),
  CONSTRAINT `train_ibfk_1` FOREIGN KEY (`store_number`) REFERENCES `key` (`store_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `date` date NOT NULL,

  `min_temperature` float DEFAULT NULL,

  `dew_point` float DEFAULT NULL,
  `wet_bulb_temperature` float DEFAULT NULL,
  `heat_index` float DEFAULT NULL,
  `cool_index` float DEFAULT NULL,
  `sunrise` datetime DEFAULT NULL,
  `sunset` datetime DEFAULT NULL,
  `precipitation` float DEFAULT NULL,
  `snowfall` float DEFAULT NULL,
  PRIMARY KEY (`station_number`, `date`),
  CONSTRAINT `weather_ibfk_1` FOREIGN KEY (`station_number`) REFERENCES `station` (`station_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
