-- Schema for database: schemabench_0143
CREATE TABLE `City` (
  `ID` int(11) NOT NULL AUTO_INCREMENT,
  `Name` varchar(255) NOT NULL,
  `CountryCode` varchar(3) NOT NULL,
  `District` varchar(255) NOT NULL,
  `Population` int(11) DEFAULT NULL,
  PRIMARY KEY (`ID`),
  CONSTRAINT `city_ibfk_1` FOREIGN KEY (`CountryCode`) REFERENCES `Country` (`Code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Country` (
  `Code` varchar(3) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `Continent` varchar(255) NOT NULL,
  `Region` varchar(255) NOT NULL,
  `SurfaceArea` float DEFAULT NULL,
  `IndepYear` int(11) DEFAULT NULL,
  `Population` int(11) DEFAULT NULL,
  `LifeExpectancy` float DEFAULT NULL,
  `GNP` float DEFAULT NULL,
  `GNPOld` float DEFAULT NULL,
  `LocalName` varchar(255) DEFAULT NULL,
  `GovernmentForm` varchar(255) DEFAULT NULL,
  `HeadOfState` varchar(255) DEFAULT NULL,
  `Capital` int(11) DEFAULT NULL,
  `Code2` varchar(2) DEFAULT NULL,
  PRIMARY KEY (`Code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `CountryLanguage` (
  `CountryCode` varchar(3) NOT NULL,
  `Language` varchar(255) NOT NULL,
  `IsOfficial` varchar(1) NOT NULL,
  `Percentage` float DEFAULT NULL,
  PRIMARY KEY (`CountryCode`, `Language`),
  CONSTRAINT `countrylanguage_ibfk_1` FOREIGN KEY (`CountryCode`) REFERENCES `Country` (`Code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
