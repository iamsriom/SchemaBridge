CREATE TABLE `products` (
  `id_product` char(11) NOT NULL,
  `category` varchar(50) DEFAULT NULL,
  `subcategory` varchar(50) DEFAULT NULL,
  `product_name` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_product`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cities` (
  `id_city` int(11) NOT NULL AUTO_INCREMENT,
  `city` varchar(50) DEFAULT NULL,
  `state` varchar(50) DEFAULT NULL,
  `country` varchar(50) DEFAULT NULL,
  `region` varchar(50) DEFAULT NULL,
  `market` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_city`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `id_customer` varchar(50) NOT NULL,
  `customer_name` varchar(50) DEFAULT NULL,
  `segment` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_customer`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `orders` (
  `id_row` int(11) NOT NULL AUTO_INCREMENT,
  `id_order` varchar(50) NOT NULL,
  `order_date` varchar(50) DEFAULT NULL,
  `ship_date` varchar(50) DEFAULT NULL,
  `ship_mode` varchar(50) DEFAULT NULL,
  `order_priority` varchar(50) DEFAULT NULL,
  `id_customer` varchar(50) NOT NULL,
  `id_city` int(11) NOT NULL,
  `id_product` char(11) NOT NULL,
  `unit_price` varchar(50) DEFAULT NULL,
  `unit_cost` varchar(50) DEFAULT NULL,
  `quantity` varchar(50) DEFAULT NULL,
  `discount` varchar(50) DEFAULT NULL,
  `shipping_cost` varchar(50) DEFAULT NULL,
  `returned` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`id_row`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`id_customer`) REFERENCES `customers` (`id_customer`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `orders_ibfk_2` FOREIGN KEY (`id_city`) REFERENCES `cities` (`id_city`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `orders_ibfk_3` FOREIGN KEY (`id_product`) REFERENCES `products` (`id_product`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
