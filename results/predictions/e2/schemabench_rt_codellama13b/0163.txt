CREATE TABLE `customer` (
  `custno` int(11) NOT NULL,
  `custname` varchar(40) NOT NULL,
  `address` varchar(40) NOT NULL,
  `internal` char(1) NOT NULL,
  `contact` varchar(40) NOT NULL,
  `phone` varchar(11) NOT NULL,
  `city` varchar(20) NOT NULL,
  `state` char(2) NOT NULL,
  `zip` int(11) NOT NULL,
  PRIMARY KEY (`custno`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `facility` (
  `facno` int(11) NOT NULL,
  `facname` char(40) NOT NULL,
  PRIMARY KEY (`facno`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `location` (
  `locno` int(11) NOT NULL,
  `facno` int(11) NOT NULL,
  `locname` varchar(40) NOT NULL,
  PRIMARY KEY (`locno`),
  CONSTRAINT `location_ibfk_1` FOREIGN KEY (`facno`) REFERENCES `facility` (`facno`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `eventrequest` (
  `eventno` int(11) NOT NULL,
  `dateheld` date NOT NULL,
  `datereq` date NOT NULL,
  `facno` int(11) NOT NULL,
  `custno` int(11) NOT NULL,
  `dateauth` date DEFAULT NULL,
  `status` varchar(8) NOT NULL,
  `estcost` decimal(7,2) NOT NULL,
  `estaudience` int(11) NOT NULL,
  `budno` varchar(5) DEFAULT NULL,
  PRIMARY KEY (`eventno`),
  CONSTRAINT `eventrequest_ibfk_1` FOREIGN KEY (`facno`) REFERENCES `facility` (`facno`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `eventrequest_ibfk_2` FOREIGN KEY (`custno`) REFERENCES `customer` (`custno`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `empno` int(11) NOT NULL,
  `empname` varchar(40) NOT NULL,
  `department` varchar(25) NOT NULL,
  `email` varchar(30) NOT NULL,
  `phone` char(6) NOT NULL,
  PRIMARY KEY (`empno`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `eventplan` (
  `planno` int(11) NOT NULL,
  `eventno` int(11) NOT NULL,
  `workdate` date NOT NULL,
  `notes` varchar(25) DEFAULT NULL,
  `activity` varchar(9) NOT NULL,
  `empno` int(11) NOT NULL,
  PRIMARY KEY (`planno`),
  CONSTRAINT `eventplan_ibfk_1` FOREIGN KEY (`eventno`) REFERENCES `eventrequest` (`eventno`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `eventplan_ibfk_2` FOREIGN KEY (`empno`) REFERENCES `employee` (`empno`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resourcetbl` (
