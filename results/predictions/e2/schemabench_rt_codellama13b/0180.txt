-- Schema for database: schemabench_0180
CREATE TABLE `aircraft` (
  `db_key` int(11) NOT NULL AUTO_INCREMENT,
  `registration_number` varchar(20) NOT NULL,
  `manufacturer` varchar(100) NOT NULL,
  `model` varchar(100) NOT NULL,
  `type_code` varchar(10) NOT NULL,
  `year_of_manufacture` int(4) DEFAULT NULL,
  `capacity` int(11) DEFAULT NULL,
  `range` int(11) DEFAULT NULL,
  `max_speed` int(11) DEFAULT NULL,
  `wing_span` float DEFAULT NULL,
  `height` float DEFAULT NULL,
  `length` float DEFAULT NULL,
  `empty_weight` float DEFAULT NULL,
  `max_takeoff_weight` float DEFAULT NULL,
  `fuel_capacity` float DEFAULT NULL,
  `engine_type` varchar(50) DEFAULT NULL,
  `crew_required` int(11) DEFAULT NULL,
  `service_ceiling` int(11) DEFAULT NULL,
  `stall_speed` float DEFAULT NULL,
  `cruise_speed` float DEFAULT NULL,
  `icao24` char(6) NOT NULL,
  `status` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`db_key`),
  UNIQUE KEY `unique_registration_number` (`registration_number`),
  CONSTRAINT `aircraft_ibfk_1` FOREIGN KEY (`icao24`) REFERENCES `traffic` (`icao24`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `traffic` (
  `db_key` int(11) NOT NULL AUTO_INCREMENT,
  `icao24` char(6) NOT NULL,
  `callsign` varchar(20) NOT NULL,
  `origin` varchar(100) NOT NULL,
  `destination` varchar(100) NOT NULL,
  `departure_time` datetime DEFAULT NULL,
  `arrival_time` datetime DEFAULT NULL,
  `flight_number` varchar(20) DEFAULT NULL,
  `status` varchar(20) DEFAULT NULL,
  `altitude` int(11) DEFAULT NULL,
  `speed` int(11) DEFAULT NULL,
  `heading` int(11) DEFAULT NULL,
  PRIMARY KEY (`db_key`),
  CONSTRAINT `traffic_ibfk_1` FOREIGN KEY (`icao24`) REFERENCES `aircraft` (`icao24`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
