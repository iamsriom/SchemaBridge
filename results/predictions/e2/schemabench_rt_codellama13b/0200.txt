-- Schema for database: schemabench_0200
CREATE TABLE `interest_map` (
  `id` int(11) NOT NULL,
  `interest_name` varchar(255) NOT NULL,
  `interest_summary` varchar(255) NOT NULL,
  `created_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `last_modified` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `interest_metrics` (
  `_month` varchar(6) NOT NULL,
  `_year` varchar(4) NOT NULL,
  `month_year` varchar(10) NOT NULL,
  `interest_id` int(11) NOT NULL,
  `composition` float NOT NULL,
  `index_value` float NOT NULL,
  `ranking` float NOT NULL,
  `percentile_ranking` float NOT NULL,
  PRIMARY KEY (`_month`, `_year`, `interest_id`),
  CONSTRAINT `interest_metrics_ibfk_1` FOREIGN KEY (`interest_id`) REFERENCES `interest_map` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
