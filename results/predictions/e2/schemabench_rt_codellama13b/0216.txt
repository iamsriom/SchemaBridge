-- Schema for database: schemabench_0216
CREATE TABLE `game_sales` (
  `game` varchar(255) NOT NULL,
  `platform` varchar(100) DEFAULT NULL,
  `publisher` varchar(255) DEFAULT NULL,
  `developer` varchar(255) DEFAULT NULL,
  `number_of_games_sold` int(11) DEFAULT NULL,
  `year` int(4) DEFAULT NULL,
  PRIMARY KEY (`game`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviews` (
  `game` varchar(255) NOT NULL,
  `critic_score` float DEFAULT NULL,
  `user_score` float DEFAULT NULL,
  PRIMARY KEY (`game`),
  CONSTRAINT `reviews_ibfk_1` FOREIGN KEY (`game`) REFERENCES `game_sales` (`game`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_critic_years` (
  `year` int(4) NOT NULL,
  `average_critic_score` float DEFAULT NULL,
  PRIMARY KEY (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_critic_years_more_than_four_games` (
  `year` int(4) NOT NULL,
  `average_critic_score` float DEFAULT NULL,
  PRIMARY KEY (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_user_years_more_than_four_games` (
  `year` int(4) NOT NULL,
  `number_of_games` int(11) DEFAULT NULL,
  `average_user_score` float DEFAULT NULL,
  PRIMARY KEY (`year`),
  CONSTRAINT `top_user_years_ibfk_1` FOREIGN KEY (`year`) REFERENCES `top_critic_years` (`year`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
