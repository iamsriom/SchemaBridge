CREATE TABLE `airline` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `headquarters` varchar(255) NOT NULL,
  `logo` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airport_abbreviation` (
  `abbreviation` char(3) NOT NULL,
  PRIMARY KEY (`abbreviation`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airport` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `abbreviation` char(3) NOT NULL,
  `name` varchar(255) NOT NULL,
  `location` varchar(255) NOT NULL,
  `radio_operator` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`abbreviation`) REFERENCES `airport_abbreviation` (`abbreviation`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airport_airline` (
  `airport_id` int(11) NOT NULL,
  `airline_id` int(11) NOT NULL,
  PRIMARY KEY (`airport_id`, `airline_id`),
  FOREIGN KEY (`airport_id`) REFERENCES `airport` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`airline_id`) REFERENCES `airline` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `amenity` (
  `code` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_crew` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `crew_size` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_schedule` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `flight_number` varchar(20) NOT NULL,
  `airline_id` int(11) NOT NULL,
  `departure_time` datetime NOT NULL,
  `arrival_time` datetime NOT NULL,
  `crew_id` int(11) NOT NULL,
  `origin_airport_id` int(11) NOT NULL,
  `destination_airport_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`airline_id`) REFERENCES `airline` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`crew_id`) REFERENCES `flight_crew` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`origin_airport_id`) REFERENCES `airport` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`destination_airport_id`) REFERENCES `airport` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `model` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `model_name` varchar(255) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `passenger_capacity` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airplane` (
  `tail_number` varchar(20) NOT NULL,
  `airline_id` int(11) NOT NULL,
  `model_id` int(11) NOT NULL,
  `manufacturer` varchar(255) NOT NULL,
  `nickname` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`tail_number`),
  FOREIGN KEY (`airline_id`) REFERENCES `airline` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`model_id`) REFERENCES `model` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `crew_member` (
  `faa_number` varchar(10) NOT NULL,

  `sex` char(1) NOT NULL,
  `role_id` int(11) NOT NULL,
  PRIMARY KEY (`faa_number`),
  FOREIGN KEY (`role_id`) REFERENCES `role` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_crew_crew_member` (
  `crew_id` int(11) NOT NULL,
  `faa_number` varchar(10) NOT NULL,
  `present_role` int(11) NOT NULL,
  PRIMARY KEY (`crew_id`, `faa_number`),
  FOREIGN KEY (`crew_id`) REFERENCES `flight_crew` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`faa_number`) REFERENCES `crew_member` (`faa_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`present_role`) REFERENCES `role` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `status` (
  `id` int(11) NOT NULL AUTO_INCREMENT,

  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight_instance` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `flight_schedule_id` int(11) NOT NULL,
  `tail_number` varchar(20) NOT NULL,
  `status_id` int(11) NOT NULL,
  `departure_date` date NOT NULL,
  `departure_time` time NOT NULL,
  `arrival_time` time NOT NULL,
  `registered_passengers` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`flight_schedule_id`) REFERENCES `flight_schedule` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`tail_number`) REFERENCES `airplane` (`tail_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`status_id`) REFERENCES `status` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
