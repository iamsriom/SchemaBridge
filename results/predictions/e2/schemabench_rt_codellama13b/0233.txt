-- Schema for database: schemabench_0233
CREATE TABLE `incident` (
  `incident_id` char(10) NOT NULL,
  `incident_type` varchar(10) DEFAULT NULL,
  `incident_location` varchar(25) DEFAULT NULL,
  `incident_description` varchar(15) DEFAULT NULL,
  `incident_date` date DEFAULT NULL,
  `personnel_id` char(10) NOT NULL,
  PRIMARY KEY (`incident_id`),
  CONSTRAINT `incident_ibfk_1` FOREIGN KEY (`personnel_id`) REFERENCES `personnel` (`personnel_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `personnel` (
  `personnel_id` char(10) NOT NULL,
  `personnel_name` varchar(10) DEFAULT NULL,
  `personnel_contact` varchar(15) DEFAULT NULL,
  `personnel_designation` varchar(10) DEFAULT NULL,
  `personnel_shift` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`personnel_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `vehicle` (
  `vehicle_id` char(10) NOT NULL,
  `vehicle_type` varchar(10) DEFAULT NULL,
  `vehicle_model` int(4) DEFAULT NULL,
  `vehicle_availability` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`vehicle_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `equipment` (
  `equipment_id` char(10) NOT NULL,
  `equipment_type` varchar(10) DEFAULT NULL,
  `equipment_quantity` int(5) DEFAULT NULL,
  `equipment_condition` varchar(20) DEFAULT NULL,
  `equipment_description` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`equipment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `victim` (
  `victim_cnic` char(15) NOT NULL,
  `victim_name` varchar(10) DEFAULT NULL,
  `victim_age` int(2) DEFAULT NULL,
  PRIMARY KEY (`victim_cnic`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `task` (
  `task_id` char(10) NOT NULL,
  `task_type` varchar(10) DEFAULT NULL,
  `task_completion` varchar(15) DEFAULT NULL,
  `incident_id` char(10) NOT NULL,
  PRIMARY KEY (`task_id`),
  CONSTRAINT `task_ibfk_1` FOREIGN KEY (`incident_id`) REFERENCES `incident` (`incident_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medical_record` (
  `record_id` char(10) NOT NULL,
  `record_treatment` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`record_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
