-- Schema for database: schemabench_0007
CREATE TABLE `tags` (
  `id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `created_at` int(11) NOT NULL,
  `updated_at` int(11) NOT NULL,
  `taggings_count` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tagger` (
  `id` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `taggings` (
  `id` int(11) NOT NULL,
  `taggable_id` int(11) NOT NULL,
  `taggable_type` int(11) NOT NULL,
  `tag_id` int(11) NOT NULL,
  `context` varchar(255) NOT NULL,
  `tagger_id` int(11) NOT NULL,
  `tagger_type` int(11) NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `taggings_ibfk_1` FOREIGN KEY (`tag_id`) REFERENCES `tags` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `taggings_ibfk_2` FOREIGN KEY (`tagger_id`) REFERENCES `tagger` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  INDEX `idx_taggings` (`taggable_id`, `taggable_type`, `tag_id`, `tagger_id`, `tagger_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
