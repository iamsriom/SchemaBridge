CREATE TABLE `silk_profile` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `time_taken` float NOT NULL,
  `file_path` varchar(255) NOT NULL,
  `line_num` int(11) NOT NULL,
  `end_line_num` int(11) NOT NULL,
  `func_name` varchar(255) NOT NULL,
  `exception_raised` varchar(255) DEFAULT NULL,
  `dynamic` tinyint(1) NOT NULL,
  `request_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `silk_profile_ibfk_1` FOREIGN KEY (`request_id`) REFERENCES `silk_request` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `silk_profile_queries` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `profile_id` int(11) NOT NULL,
  `sqlquery_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `silk_profile_queries_ibfk_1` FOREIGN KEY (`profile_id`) REFERENCES `silk_profile` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `silk_profile_queries_ibfk_2` FOREIGN KEY (`sqlquery_id`) REFERENCES `silk_sqlquery` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `id` int(11) NOT NULL AUTO_INCREMENT,

  `query_params` text DEFAULT NULL,
  `raw_body` text DEFAULT NULL,
  `body` text DEFAULT NULL,
  `method` varchar(10) NOT NULL,
  `start_time` datetime NOT NULL,
  `view_name` varchar(255) NOT NULL,
  `end_time` datetime NOT NULL,
  `time_taken` float NOT NULL,
  `encoded_headers` text DEFAULT NULL,
  `meta_time` float NOT NULL,
  `meta_num_queries` int(11) NOT NULL,

  `pyprofile` text DEFAULT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `id` int(11) NOT NULL AUTO_INCREMENT,
