CREATE TABLE `users` (
  `github_id` bigint(20) NOT NULL,
  `github_login` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `last_synced_at` datetime DEFAULT NULL,
  `refresh_interval` int(11) DEFAULT NULL,
  `api_token` varchar(255) DEFAULT NULL,
  `sync_job_id` bigint(20) DEFAULT NULL,
  `encrypted_access_token` varchar(255) DEFAULT NULL,
  `encrypted_access_token_iv` varchar(255) DEFAULT NULL,
  `encrypted_personal_access_token` varchar(255) DEFAULT NULL,
  `encrypted_personal_access_token_iv` varchar(255) DEFAULT NULL,
  `encrypted_app_token` varchar(255) DEFAULT NULL,
  `encrypted_app_token_iv` varchar(255) DEFAULT NULL,
  `theme` varchar(50) DEFAULT NULL,
  `display_comments` boolean DEFAULT NULL,
  PRIMARY KEY (`github_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `app_installations` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `github_id` bigint(20) NOT NULL,
  `app_id` bigint(20) NOT NULL,
  `account_login` varchar(255) NOT NULL,
  `account_id` bigint(20) NOT NULL,
  `account_type` varchar(50) NOT NULL,
  `target_type` varchar(50) NOT NULL,
  `target_id` bigint(20) NOT NULL,
  `permission_statuses` json DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `app_installation_permissions` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `user_github_id` bigint(20) NOT NULL,
  `app_installation_id` bigint(20) NOT NULL,
  `permissions` json DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `app_installation_permissions_ibfk_1` FOREIGN KEY (`user_github_id`) REFERENCES `users` (`github_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `app_installation_permissions_ibfk_2` FOREIGN KEY (`app_installation_id`) REFERENCES `app_installations` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
