-- Schema for database: schemabench_0048
CREATE TABLE `account` (
  `id` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `date_created` timestamp DEFAULT NULL,
  `stripe_customer_id` varchar(255) NOT NULL,
  `stripe_subscription_id` varchar(255) NOT NULL,
  `plan` varchar(255) NOT NULL,
  `referrer` varchar(255) NOT NULL,
  `active` tinyint DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `invite` (
  `id` int(11) NOT NULL,
  `invite_id` varchar(255) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `account_id` varchar(255) DEFAULT NULL,
  `date_sent` timestamp DEFAULT NULL,
  `used` timestamp DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `invite_ibfk_1` FOREIGN KEY (`account_id`) REFERENCES `account` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `user` (
  `id` varchar(255) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `date_created` timestamp DEFAULT NULL,
  `last_login` timestamp DEFAULT NULL,
  `permission` varchar(255) DEFAULT NULL,
  `push_token` varchar(255) DEFAULT NULL,
  `account_id` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `user_ibfk_1` FOREIGN KEY (`account_id`) REFERENCES `account` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
