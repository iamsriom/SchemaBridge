-- Schema for database: schemabench_0059
CREATE TABLE `websites` (
  `id` bigint(20) NOT NULL,
  `uuid` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `deleted_at` datetime DEFAULT NULL,
  `managed_by_database_connection` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hostnames` (
  `id` bigint(20) NOT NULL,
  `fqdn` varchar(255) NOT NULL,
  `redirect_to` varchar(255) DEFAULT NULL,
  `force_https` tinyint(1) NOT NULL,
  `under_maintenance_since` timestamp NULL DEFAULT NULL,
  `website_id` bigint(20) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `deleted_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `hostnames_ibfk_1` FOREIGN KEY (`website_id`) REFERENCES `websites` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
