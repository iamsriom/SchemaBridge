-- Schema for database: schemabench_0068
CREATE TABLE `flipper_features` (
  `id` bigint(20) NOT NULL,
  `key` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flipper_gates` (
  `id` bigint(20) NOT NULL,
  `feature_key` varchar(255) NOT NULL,
  `key` varchar(255) NOT NULL,
  `value` varchar(255) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`, `feature_key`, `key`),
  CONSTRAINT `flipper_gates_ibfk_1` FOREIGN KEY (`feature_key`) REFERENCES `flipper_features` (`key`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flipper_gates_ibfk_2` FOREIGN KEY (`key`) REFERENCES `flipper_features` (`key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
