CREATE TABLE `clicks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_ip` varchar(45) DEFAULT NULL,
  `user_country` varchar(100) DEFAULT NULL,
  `referer_url` varchar(255) DEFAULT NULL,
  `referer_host` varchar(255) DEFAULT NULL,
  `user_agent` varchar(255) DEFAULT NULL,
  `link_id` int(11) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `clicks_ibfk_1` FOREIGN KEY (`link_id`) REFERENCES `links` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `links` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `short_url` varchar(255) NOT NULL,
  `original_url` varchar(255) NOT NULL,
  `creator_ip` varchar(45) DEFAULT NULL,
  `creator` int(11) NOT NULL,
  `click_count` int(11) DEFAULT NULL,
  `secret_key` varchar(255) DEFAULT NULL,
  `is_disabled` tinyint(1) DEFAULT NULL,
  `is_custom` tinyint(1) DEFAULT NULL,
  `is_api` tinyint(1) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `links_ibfk_1` FOREIGN KEY (`creator`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `migrations` (
  `migration` varchar(255) NOT NULL,
  `batch` int(11) NOT NULL,
  PRIMARY KEY (`migration`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `ip_address` varchar(45) DEFAULT NULL,
  `recovery_key` varchar(255) DEFAULT NULL,
  `role` varchar(50) DEFAULT NULL,
  `is_active` tinyint(1) NOT NULL,
  `api_key` varchar(255) DEFAULT NULL,
  `is_api_active` tinyint(1) DEFAULT NULL,
  `api_quota` int(11) DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `deleted_at` datetime DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
