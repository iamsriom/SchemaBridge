) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customers` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `billable_id` bigint(20) unsigned NOT NULL,
  `billable_type` bigint(20) unsigned NOT NULL,
  `trial_ends_at` timestamp NOT NULL,
  `updated_at` timestamp NOT NULL,
  `created_at` timestamp NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `customers_ibfk_1` FOREIGN KEY (`billable_id`) REFERENCES `users_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `name` varchar(255) NOT NULL,
  `paddle_id` varchar(255) NOT NULL,
  `paddle_status` varchar(255) NOT NULL,
  `paddle_plan` int(11) NOT NULL,
  `quantity` int(11) NOT NULL,
  `trial_ends_at` datetime NOT NULL,
  `paused_from` datetime NOT NULL,
  `ends_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  `created_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `subscriptions_ibfk_1` FOREIGN KEY (`billable_id`) REFERENCES `users_teams` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
