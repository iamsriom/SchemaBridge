-- Schema for database: schemabench_0100
CREATE TABLE `households` (
  `household_id` int(11) NOT NULL,
  `year` int(11) DEFAULT NULL,
  `income_rank` int(11) DEFAULT NULL,
  `income_rank_2` int(11) DEFAULT NULL,
  `income_rank_3` int(11) DEFAULT NULL,
  `income_rank_4` int(11) DEFAULT NULL,
  `income_rank_5` int(11) DEFAULT NULL,
  `income_rank_mean` float DEFAULT NULL,
  PRIMARY KEY (`household_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expenditures` (
  `expenditure_id` int(11) NOT NULL,
  `household_id` int(11) NOT NULL,
  `year` int(11) DEFAULT NULL,
  `month` int(11) DEFAULT NULL,
  `product_code` varchar(255) DEFAULT NULL,
  `cost` float DEFAULT NULL,
  `gift` float DEFAULT NULL,
  `is_training` tinyint(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (`expenditure_id`),
  CONSTRAINT `expenditures_ibfk_1` FOREIGN KEY (`household_id`) REFERENCES `households` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `household_members` (
  `household_id` int(11) NOT NULL,
  `year` int(11) DEFAULT NULL,
  `marital` varchar(255) DEFAULT NULL,
  `sex` varchar(10) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `work_status` varchar(255) DEFAULT NULL,
  CONSTRAINT `household_members_ibfk_1` FOREIGN KEY (`household_id`) REFERENCES `households` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
