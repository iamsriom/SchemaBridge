-- Schema for database: schemabench_0104
CREATE TABLE `incidents` (
  `case_number` varchar(255) NOT NULL,
  `date` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `subject_statuses` varchar(255) NOT NULL,
  `subject_weapon` varchar(255) NOT NULL,
  `subjects` int(11) NOT NULL,
  `subject_count` int(11) NOT NULL,
  `officers` int(11) NOT NULL,
  `officer_count` int(11) NOT NULL,
  `grand_jury_disposition` varchar(255) NOT NULL,
  `attorney_general_forms_url` varchar(255) DEFAULT NULL,
  `summary_url` varchar(255) DEFAULT NULL,
  `summary_text` text DEFAULT NULL,
  `latitude` decimal(9,6) DEFAULT NULL,
  `longitude` decimal(9,6) DEFAULT NULL,
  PRIMARY KEY (`case_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `officers` (
  `case_number` varchar(255) NOT NULL,
  `race` varchar(50) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `first_name` varchar(255) NOT NULL,
  `full_name` varchar(255) NOT NULL,
  PRIMARY KEY (`case_number`, `last_name`, `first_name`),
  CONSTRAINT `officers_ibfk_1` FOREIGN KEY (`case_number`) REFERENCES `incidents` (`case_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `subjects` (
  `case_number` varchar(255) NOT NULL,
  `race` varchar(50) NOT NULL,
  `gender` varchar(10) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `first_name` varchar(255) NOT NULL,
  `full_name` varchar(255) NOT NULL,
  PRIMARY KEY (`case_number`, `last_name`, `first_name`),
  CONSTRAINT `subjects_ibfk_1` FOREIGN KEY (`case_number`) REFERENCES `incidents` (`case_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
