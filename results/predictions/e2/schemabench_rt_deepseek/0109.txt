-- Schema for database: schemabench_0109
CREATE TABLE `go_retailers` (
  `retailercode` varchar(50) NOT NULL,
  `retailername` varchar(255) NOT NULL,
  `type` varchar(100) DEFAULT NULL,
  `country` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`retailercode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_products` (
  `productnumber` varchar(50) NOT NULL,
  `productline` varchar(100) DEFAULT NULL,
  `producttype` varchar(100) DEFAULT NULL,
  `product` varchar(255) DEFAULT NULL,
  `productbrand` varchar(100) DEFAULT NULL,
  `productcolor` varchar(100) DEFAULT NULL,
  `unitcost` decimal(10,2) DEFAULT NULL,
  `unitprice` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`productnumber`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_methods` (
  `ordermethodcode` varchar(50) NOT NULL,
  `ordermethodtype` varchar(100) DEFAULT NULL,
  PRIMARY KEY (`ordermethodcode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_1k` (
  `retailercode` varchar(50) NOT NULL,
  `productnumber` varchar(50) NOT NULL,
  `date` date NOT NULL,
  `quantity` int(11) NOT NULL,
  PRIMARY KEY (`retailercode`, `productnumber`, `date`),
  CONSTRAINT `go_1k_ibfk_1` FOREIGN KEY (`retailercode`) REFERENCES `go_retailers` (`retailercode`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `go_1k_ibfk_2` FOREIGN KEY (`productnumber`) REFERENCES `go_products` (`productnumber`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `go_daily_sales` (
  `retailercode` varchar(50) NOT NULL,
  `productnumber` varchar(50) NOT NULL,
  `ordermethodcode` varchar(50) NOT NULL,
  `date` date NOT NULL,
  `quantity` int(11) NOT NULL,
  `unitprice` decimal(10,2) NOT NULL,
  `unitsaleprice` decimal(10,2) NOT NULL,
  PRIMARY KEY (`retailercode`, `productnumber`),
  CONSTRAINT `go_daily_sales_ibfk_1` FOREIGN KEY (`ordermethodcode`) REFERENCES `go_methods` (`ordermethodcode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
