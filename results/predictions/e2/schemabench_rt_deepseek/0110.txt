-- Schema for database: schemabench_0110
CREATE TABLE `awards` (
  `award_id` int(11) NOT NULL,
  `award_title` varchar(255) NOT NULL,
  `effective_date` datetime NOT NULL,
  `expiration_date` datetime NOT NULL,
  `award_amount` decimal(10,2) NOT NULL,
  `award_instrument` varchar(255) NOT NULL,
  `organization_code` varchar(50) NOT NULL,
  `program_officer` varchar(255) NOT NULL,
  `abstract_narration` text DEFAULT NULL,
  `min_amd_letter_date` date DEFAULT NULL,
  `max_amd_letter_date` date DEFAULT NULL,
  `arra_amount` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`award_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `foa_info` (
  `foa_code` varchar(50) NOT NULL,
  PRIMARY KEY (`foa_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `foa_info_awards` (
  `award_id` int(11) NOT NULL,
  `foa_code` varchar(50) NOT NULL,
  PRIMARY KEY (`award_id`, `foa_code`),
  CONSTRAINT `foa_info_awards_ibfk_1` FOREIGN KEY (`award_id`) REFERENCES `awards` (`award_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `foa_info_awards_ibfk_2` FOREIGN KEY (`foa_code`) REFERENCES `foa_info` (`foa_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `institution` (
  `institution_name` varchar(255) NOT NULL,
  `zipcode` varchar(10) NOT NULL,
  `city_name` varchar(255) NOT NULL,
  `contact_number` varchar(20) DEFAULT NULL,
  `address` text DEFAULT NULL,
  `country_name` varchar(255) DEFAULT NULL,
  `state_name` varchar(255) DEFAULT NULL,
  `state_code` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`institution_name`, `zipcode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `institution_awards` (
  `award_id` int(11) NOT NULL,
  `institution_name` varchar(255) NOT NULL,
  `zipcode` varchar(10) NOT NULL,
  PRIMARY KEY (`award_id`, `institution_name`, `zipcode`),
  CONSTRAINT `institution_awards_ibfk_1` FOREIGN KEY (`award_id`) REFERENCES `awards` (`award_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `institution_awards_ibfk_2` FOREIGN KEY (`institution_name`, `zipcode`) REFERENCES `institution` (`institution_name`, `zipcode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `investigator` (
  `email_id` varchar(255) NOT NULL,

  PRIMARY KEY (`email_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `award_id` int(11) NOT NULL,

  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  `role_code` varchar(50) NOT NULL,
  PRIMARY KEY (`award_id`, `email_id`),
  CONSTRAINT `investigator_awards_ibfk_1` FOREIGN KEY (`award_id`) REFERENCES `awards` (`award_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `investigator_awards_ibfk_2` FOREIGN KEY (`email_id`) REFERENCES `investigator` (`email_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `organization` (
