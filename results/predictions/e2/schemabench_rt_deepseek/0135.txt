-- Schema for database: schemabench_0135
CREATE TABLE `trains` (
  `id` int(11) NOT NULL,
  `direction` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cars` (
  `id` int(11) NOT NULL,
  `train_id` int(11) DEFAULT NULL,
  `position` varchar(255) DEFAULT NULL,
  `shape` varchar(255) DEFAULT NULL,
  `length` int(11) DEFAULT NULL,
  `sides` int(11) DEFAULT NULL,
  `roof` varchar(255) DEFAULT NULL,
  `wheels` int(11) DEFAULT NULL,
  `load_shape` varchar(255) DEFAULT NULL,
  `load_number` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `cars_ibfk_1` FOREIGN KEY (`train_id`) REFERENCES `trains` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
