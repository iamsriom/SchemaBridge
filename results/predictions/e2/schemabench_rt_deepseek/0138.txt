-- Schema for database: schemabench_0138
CREATE TABLE `person` (
  `p_id` int(11) NOT NULL,
  `professor` tinyint(1) DEFAULT '0',
  `student` tinyint(1) DEFAULT '0',
  `hasPosition` tinyint(1) DEFAULT '0',
  `inPhase` tinyint(1) DEFAULT NULL,
  `yearsInProgram` tinyint(1) DEFAULT NULL,
  PRIMARY KEY (`p_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `course` (
  `course_id` int(11) NOT NULL,
  `courseLevel` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `advisedBy` (
  `p_id` int(11) NOT NULL,
  `p_id_dummy` int(11) NOT NULL,
  PRIMARY KEY (`p_id_dummy`, `p_id`),
  CONSTRAINT `advisedBy_ibfk_1` FOREIGN KEY (`p_id`) REFERENCES `person` (`p_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `advisedBy_ibfk_2` FOREIGN KEY (`p_id_dummy`) REFERENCES `person` (`p_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `taughtBy` (
  `course_id` int(11) NOT NULL,
  `p_id` int(11) NOT NULL,
  PRIMARY KEY (`course_id`, `p_id`),
  CONSTRAINT `taughtBy_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `course` (`course_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `taughtBy_ibfk_2` FOREIGN KEY (`p_id`) REFERENCES `person` (`p_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
