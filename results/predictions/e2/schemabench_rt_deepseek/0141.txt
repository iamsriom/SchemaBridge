-- Schema for database: schemabench_0141
CREATE TABLE `key` (
  `store_nbr` int(11) NOT NULL,
  `station_nbr` int(11) NOT NULL,
  PRIMARY KEY (`store_nbr`),
  CONSTRAINT `key_ibfk_1` FOREIGN KEY (`station_nbr`) REFERENCES `station` (`station_nbr`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `station` (
  `station_nbr` int(11) NOT NULL,
  PRIMARY KEY (`station_nbr`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `train` (
  `store_nbr` int(11) NOT NULL,
  `date` int(11) NOT NULL,
  `item_nbr` int(11) NOT NULL,
  PRIMARY KEY (`store_nbr`, `date`, `item_nbr`),
  CONSTRAINT `train_ibfk_1` FOREIGN KEY (`store_nbr`) REFERENCES `key` (`store_nbr`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `weather` (
  `station_nbr` int(11) NOT NULL,
  `date` int(11) NOT NULL,
  PRIMARY KEY (`station_nbr`, `date`),
  CONSTRAINT `weather_ibfk_1` FOREIGN KEY (`station_nbr`) REFERENCES `station` (`station_nbr`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
