-- Schema for database: schemabench_0158
CREATE TABLE IF NOT EXISTS `installations` (
  `installationid` int(11) NOT NULL,
  `installationname` varchar(255) NOT NULL,
  PRIMARY KEY (`installationid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE IF NOT EXISTS `laborcodes` (
  `laborcodeid` int(11) NOT NULL,
  `laborcodename` varchar(255) NOT NULL,
  PRIMARY KEY (`laborcodeid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE IF NOT EXISTS `billingcodes` (
  `billingcodeid` varchar(255) NOT NULL,
  `billingcode` varchar(255) NOT NULL,
  PRIMARY KEY (`billingcodeid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE IF NOT EXISTS `tickets` (
  `ticketid` int(11) NOT NULL AUTO_INCREMENT,
  `ticketnumber` int(11) NOT NULL,
  `ticketdesc` int(11) NOT NULL,
  `laborcode` int(11) NOT NULL,
  `dateofcreation` int(11) NOT NULL,
  `installation` int(11) NOT NULL,
  `isclosed` int(11) NOT NULL DEFAULT 0,
  PRIMARY KEY (`ticketid`),
  CONSTRAINT `tickets_ibfk_1` FOREIGN KEY (`installation`) REFERENCES `installations` (`installationid`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `tickets_ibfk_2` FOREIGN KEY (`laborcode`) REFERENCES `laborcodes` (`laborcodeid`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE IF NOT EXISTS `reports` (
  `reportid` int(11) NOT NULL AUTO_INCREMENT,
  `dateofreport` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `billingcode` varchar(255) NOT NULL DEFAULT 'B05',
  `ticketid` int(11) NOT NULL,
  `arrivaltime` int(11) NOT NULL,
  `exittime` int(11) NOT NULL,
  `activity` int(11) NOT NULL,
  PRIMARY KEY (`reportid`),
  CONSTRAINT `reports_ibfk_1` FOREIGN KEY (`ticketid`) REFERENCES `tickets` (`ticketid`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE IF NOT EXISTS `shell` (
  `component` text,
  `title` text,
  `icon` text,
  `menu_item` text,
  `theme` text
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
