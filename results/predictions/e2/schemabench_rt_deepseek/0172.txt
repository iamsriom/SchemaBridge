-- Schema for database: schemabench_0172
CREATE TABLE `Publisher` (
  `publisherID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`publisherID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `User` (
  `userID` int(11) NOT NULL AUTO_INCREMENT,
  `phoneNumber` varchar(15) NOT NULL UNIQUE,
  `gender` char(1) NOT NULL CHECK (`gender` IN ('M', 'F')),
  `birthDate` date NOT NULL,
  `age` int(11) NOT NULL CHECK (`age` >= 13),
  `address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`userID`),
  CONSTRAINT `user_ibfk_1` FOREIGN KEY (`userID`) REFERENCES `Publisher` (`publisherID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Friendship` (
  `senderID` int(11) NOT NULL,
  `receiverID` int(11) NOT NULL,
  `state` tinyint(1) NOT NULL CHECK (`state` BETWEEN 1 AND 3),
  `date` datetime NOT NULL,
  PRIMARY KEY (`senderID`, `receiverID`),
  CONSTRAINT `friendship_ibfk_1` FOREIGN KEY (`senderID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `friendship_ibfk_2` FOREIGN KEY (`receiverID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Page` (
  `pageID` int(11) NOT NULL AUTO_INCREMENT,
  `website` varchar(255) DEFAULT NULL,
  `adminID` int(11) NOT NULL,
  PRIMARY KEY (`pageID`),
  CONSTRAINT `page_ibfk_1` FOREIGN KEY (`adminID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `PageFollower` (
  `followerID` int(11) NOT NULL,
  `pageID` int(11) NOT NULL,
  PRIMARY KEY (`followerID`, `pageID`),
  CONSTRAINT `pagefollower_ibfk_1` FOREIGN KEY (`followerID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `pagefollower_ibfk_2` FOREIGN KEY (`pageID`) REFERENCES `Page` (`pageID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Group` (
  `groupID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `adminID` int(11) NOT NULL,
  PRIMARY KEY (`groupID`),
  CONSTRAINT `group_ibfk_1` FOREIGN KEY (`adminID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `GroupMember` (
  `memberID` int(11) NOT NULL,
  `groupID` int(11) NOT NULL,
  PRIMARY KEY (`memberID`, `groupID`),
  CONSTRAINT `groupmember_ibfk_1` FOREIGN KEY (`memberID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `groupmember_ibfk_2` FOREIGN KEY (`groupID`) REFERENCES `Group` (`groupID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Chat` (
  `chatID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`chatID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ChatParticipant` (
  `participantID` int(11) NOT NULL,
  `chatID` int(11) NOT NULL,
  PRIMARY KEY (`participantID`, `chatID`),
  CONSTRAINT `chatparticipant_ibfk_1` FOREIGN KEY (`participantID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `chatparticipant_ibfk_2` FOREIGN KEY (`chatID`) REFERENCES `Chat` (`chatID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Multimedia` (
  `multimediaID` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) DEFAULT NULL,
  `uri` varchar(255) DEFAULT NULL,
  `size` int(11) NOT NULL CHECK (`size` > 0),
  `format` varchar(5) NOT NULL CHECK (`format` IN ('.mp3', '.jpg', '.png', '.wav', '.mp4')),
  `length` int(11) NOT NULL CHECK (`length` > 0),
  PRIMARY KEY (`multimediaID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Activity` (
  `activityID` int(11) NOT NULL AUTO_INCREMENT,
  `activityText` text DEFAULT NULL,
  `activityDate` datetime DEFAULT NULL,
  PRIMARY KEY (`activityID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Message` (
  `messageID` int(11) NOT NULL AUTO_INCREMENT,
  `dateSent` datetime NOT NULL,
  `multimediaID` int(11) DEFAULT NULL,
  `authorID` int(11) NOT NULL,
  `chatID` int(11) NOT NULL,
  PRIMARY KEY (`messageID`),
  CONSTRAINT `message_ibfk_1` FOREIGN KEY (`multimediaID`) REFERENCES `Multimedia` (`multimediaID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `message_ibfk_2` FOREIGN KEY (`authorID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `message_ibfk_3` FOREIGN KEY (`chatID`) REFERENCES `Chat` (`chatID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Post` (
  `postID` int(11) NOT NULL AUTO_INCREMENT,
  `publisherID` int(11) NOT NULL,
  `multimediaID` int(11) DEFAULT NULL,
  `pageID` int(11) DEFAULT NULL,
  `groupID` int(11) DEFAULT NULL,
  PRIMARY KEY (`postID`),
  CONSTRAINT `post_ibfk_1` FOREIGN KEY (`publisherID`) REFERENCES `Publisher` (`publisherID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `post_ibfk_2` FOREIGN KEY (`multimediaID`) REFERENCES `Multimedia` (`multimediaID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `post_ibfk_3` FOREIGN KEY (`pageID`) REFERENCES `Page` (`pageID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `post_ibfk_4` FOREIGN KEY (`groupID`) REFERENCES `Group` (`groupID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Comment` (
  `commentID` int(11) NOT NULL AUTO_INCREMENT,
  `authorID` int(11) NOT NULL,
  `postID` int(11) NOT NULL,
  PRIMARY KEY (`commentID`),
  CONSTRAINT `comment_ibfk_1` FOREIGN KEY (`authorID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comment_ibfk_2` FOREIGN KEY (`postID`) REFERENCES `Post` (`postID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Reaction` (
  `activityID` int(11) NOT NULL,
  `userID` int(11) NOT NULL,
  `type` tinyint(1) NOT NULL CHECK (`type` BETWEEN 1 AND 6),
  PRIMARY KEY (`activityID`, `userID`),
  CONSTRAINT `reaction_ibfk_1` FOREIGN KEY (`activityID`) REFERENCES `Activity` (`activityID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `reaction_ibfk_2` FOREIGN KEY (`userID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Event` (
  `eventID` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `description` text NOT NULL,
  `occurrenceDate` datetime NOT NULL,
  `creatorID` int(11) NOT NULL,
  PRIMARY KEY (`eventID`),
  CONSTRAINT `event_ibfk_1` FOREIGN KEY (`creatorID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `EventParticipant` (
  `participantID` int(11) NOT NULL,
  `eventID` int(11) NOT NULL,
  PRIMARY KEY (`participantID`, `eventID`),
  CONSTRAINT `eventparticipant_ibfk_1` FOREIGN KEY (`participantID`) REFERENCES `User` (`userID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `eventparticipant_ibfk_2` FOREIGN KEY (`eventID`) REFERENCES `Event` (`eventID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
