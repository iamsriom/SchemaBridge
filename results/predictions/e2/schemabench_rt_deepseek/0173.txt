) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `hospital` (
  `hospital_id` int(11) NOT NULL AUTO_INCREMENT,
  `hospital_name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_number` varchar(15) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `head_doctor_name` varchar(255) DEFAULT NULL,
  `services_offered` varchar(255) DEFAULT NULL,
  `emergency_services_availability` tinyint(1) NOT NULL,
  PRIMARY KEY (`hospital_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `private_facility` (
  `private_facility_id` int(11) NOT NULL AUTO_INCREMENT,
  `hospital_name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `contact_number` varchar(15) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `head_doctor_name` varchar(255) DEFAULT NULL,
  `services_offered` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`private_facility_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donor` (
  `donor_id` int(11) NOT NULL AUTO_INCREMENT,
  `first_name` varchar(255) NOT NULL,
  `last_name` varchar(255) NOT NULL,
  `date_of_birth` date NOT NULL,
  `gender` varchar(10) NOT NULL,
  `contact_number` varchar(15) NOT NULL,
  `email` varchar(255) DEFAULT NULL,
  `home_address` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`donor_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `donor_medical` (
  `donor_medical_id` int(11) NOT NULL AUTO_INCREMENT,
  `donor_id` int(11) NOT NULL,
  `allergies` varchar(255) DEFAULT NULL,
  `weight` decimal(5,2) DEFAULT NULL,
  `blood_type` varchar(10) NOT NULL,
  `medical_condition` varchar(255) DEFAULT NULL,
  `medical_history` text DEFAULT NULL,
  `insurance_details` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`donor_medical_id`),
  CONSTRAINT `donor_medical_ibfk_1` FOREIGN KEY (`donor_id`) REFERENCES `donor` (`donor_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `donor_medical_ibfk_2` FOREIGN KEY (`blood_type`) REFERENCES `blood` (`blood_type`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
