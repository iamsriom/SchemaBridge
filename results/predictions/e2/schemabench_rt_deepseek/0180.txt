-- Schema for database: schemabench_0180
CREATE TABLE `aircraft` (
  `db_key` INT(11) NOT NULL AUTO_INCREMENT,
  `registration` VARCHAR(255) DEFAULT NULL,
  `manufacturer` VARCHAR(255) DEFAULT NULL,
  `model` VARCHAR(255) DEFAULT NULL,
  `type_code` VARCHAR(50) DEFAULT NULL,
  `serial_number` VARCHAR(50) DEFAULT NULL,
  `line_number` VARCHAR(50) DEFAULT NULL,
  `aircraft_type` VARCHAR(50) DEFAULT NULL,
  `operator` VARCHAR(255) DEFAULT NULL,
  `test_registration` BOOLEAN DEFAULT NULL,
  `built_date` TIMESTAMP DEFAULT NULL,
  `first_flight_date` TIMESTAMP DEFAULT NULL,
  `seat_configuration` VARCHAR(255) DEFAULT NULL,
  `engine_details` VARCHAR(255) DEFAULT NULL,
  `modes` VARCHAR(255) DEFAULT NULL,
  `adsb_capability` BOOLEAN DEFAULT NULL,
  `acars_capability` BOOLEAN DEFAULT NULL,
  `notes` TEXT DEFAULT NULL,
  PRIMARY KEY (`db_key`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `traffic` (
  `db_key` INT(11) NOT NULL AUTO_INCREMENT,
  `callsign` VARCHAR(50) DEFAULT NULL,
  `number` VARCHAR(50) DEFAULT NULL,
  `registration` VARCHAR(255) DEFAULT NULL,
  `type_code` VARCHAR(50) DEFAULT NULL,
  `origin` VARCHAR(255) DEFAULT NULL,
  `destination` VARCHAR(255) DEFAULT NULL,
  `first_seen` TIMESTAMP DEFAULT NULL,
  `last_seen` TIMESTAMP DEFAULT NULL,
  `day` DATE DEFAULT NULL,
  `latitude` DOUBLE PRECISION DEFAULT NULL,
  `longitude` DOUBLE PRECISION DEFAULT NULL,
  `altitude` DOUBLE PRECISION DEFAULT NULL,
  PRIMARY KEY (`db_key`),
  CONSTRAINT `traffic_ibfk_1` FOREIGN KEY (`db_key`) REFERENCES `aircraft` (`db_key`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
