CREATE TABLE `runners` (
  `runner_id` int(11) NOT NULL,
  `registration_date` date DEFAULT NULL,
  PRIMARY KEY (`runner_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_orders` (
  `order_id` int(11) NOT NULL,
  `customer_id` int(11) DEFAULT NULL,
  `pizza_id` int(11) NOT NULL,
  `exclusions` varchar(255) DEFAULT NULL,
  `extras` varchar(255) DEFAULT NULL,
  `order_time` timestamp DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `customer_orders_ibfk_1` FOREIGN KEY (`pizza_id`) REFERENCES `pizza_recipes` (`pizza_id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `runner_orders` (
  `order_id` int(11) NOT NULL,
  `runner_id` int(11) NOT NULL,
  `pickup_time` varchar(255) DEFAULT NULL,
  `distance` varchar(255) DEFAULT NULL,
  `duration` varchar(255) DEFAULT NULL,
  `cancellation` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`order_id`),
  CONSTRAINT `runner_orders_ibfk_1` FOREIGN KEY (`runner_id`) REFERENCES `runners` (`runner_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `runner_orders_ibfk_2` FOREIGN KEY (`order_id`) REFERENCES `customer_orders` (`order_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pizza_recipes` (
  `pizza_id` int(11) NOT NULL,
  `toppings` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pizza_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pizza_names` (
  `pizza_id` int(11) NOT NULL,
  `pizza_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`pizza_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `pizza_toppings` (
  `topping_id` int(11) NOT NULL,

  PRIMARY KEY (`topping_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
