-- Schema for database: schemabench_0197
CREATE TABLE `regions` (
  `region_id` int(11) NOT NULL,
  `region_name` varchar(9) DEFAULT NULL,
  PRIMARY KEY (`region_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_nodes` (
  `customer_id` int(11) NOT NULL,
  `region_id` int(11) NOT NULL,
  `node_id` int(11) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  PRIMARY KEY (`customer_id`, `region_id`, `node_id`),
  CONSTRAINT `customer_nodes_ibfk_1` FOREIGN KEY (`region_id`) REFERENCES `regions` (`region_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `customer_transactions` (
  `customer_id` int(11) NOT NULL,
  `txn_date` date NOT NULL,
  `txn_type` varchar(10) DEFAULT NULL,
  `txn_amount` int(11) NOT NULL,
  PRIMARY KEY (`customer_id`),
  CONSTRAINT `customer_transactions_ibfk_1` FOREIGN KEY (`customer_id`) REFERENCES `customer_nodes` (`customer_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
