-- Schema for database: schemabench_0198
CREATE TABLE `event_identifier` (
  `event_type` int(11) NOT NULL,
  `event_name` varchar(13) DEFAULT NULL,
  PRIMARY KEY (`event_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `campaign_identifier` (
  `campaign_id` int(11) NOT NULL,
  `products` varchar(3) DEFAULT NULL,
  `campaign_name` varchar(33) DEFAULT NULL,
  `start_date` timestamp DEFAULT NULL,
  `end_date` timestamp DEFAULT NULL,
  PRIMARY KEY (`campaign_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `page_hierarchy` (
  `page_id` int(11) NOT NULL,
  `page_name` varchar(14) DEFAULT NULL,
  `product_category` varchar(9) DEFAULT NULL,
  `product_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`page_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL,
  `cookie_id` varchar(6) DEFAULT NULL,
  `start_date` timestamp DEFAULT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `events` (
  `visit_id` varchar(6) DEFAULT NULL,
  `cookie_id` varchar(6) DEFAULT NULL,
  `page_id` int(11) NOT NULL,
  `event_type` int(11) NOT NULL,
  `sequence_number` int(11) NOT NULL,
  `event_time` timestamp NOT NULL,
  PRIMARY KEY (`visit_id`, `sequence_number`),
  CONSTRAINT `events_ibfk_1` FOREIGN KEY (`page_id`) REFERENCES `page_hierarchy` (`page_id`) ON DELETE CASCADE ON UPDATE RESTRICT,
  CONSTRAINT `events_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE CASCADE ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
