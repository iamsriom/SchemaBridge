-- Schema for database: schemabench_0212
CREATE TABLE `movie` (
  `id` int(11) DEFAULT NULL,
  `title` varchar(255) DEFAULT NULL,
  `year` int(11) DEFAULT NULL,
  `date_published` date DEFAULT NULL,
  `duration` int(11) DEFAULT NULL,
  `country` varchar(255) DEFAULT NULL,
  `worldwide_gross_income` decimal(15,2) DEFAULT NULL,
  `languages_spoken` varchar(255) DEFAULT NULL,
  `production_company` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `genre` (
  `movie_id` int(11) NOT NULL,
  `genre_name` varchar(255) NOT NULL,
  PRIMARY KEY (`movie_id`),
  CONSTRAINT `genre_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movie` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `director_mapping` (
  `movie_id` int(11) NOT NULL,
  `director_name` varchar(255) NOT NULL,
  PRIMARY KEY (`movie_id`),
  CONSTRAINT `director_mapping_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movie` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role_mapping` (
  `movie_id` int(11) NOT NULL,
  `name_id` int(11) NOT NULL,
  `category` varchar(255) NOT NULL,
  PRIMARY KEY (`movie_id`, `name_id`),
  CONSTRAINT `role_mapping_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movie` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `role_mapping_ibfk_2` FOREIGN KEY (`name_id`) REFERENCES `names` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `names` (
  `id` int(11) DEFAULT NULL,
  `name` varchar(255) DEFAULT NULL,
  `height` decimal(5,2) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ratings` (
  `movie_id` int(11) NOT NULL,
  `average_rating` decimal(3,2) DEFAULT NULL,
  `total_votes` int(11) DEFAULT NULL,
  `median_rating` decimal(3,2) DEFAULT NULL,
  PRIMARY KEY (`movie_id`),
  CONSTRAINT `ratings_ibfk_1` FOREIGN KEY (`movie_id`) REFERENCES `movie` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
