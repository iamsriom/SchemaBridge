-- Schema for database: schemabench_0215
CREATE TABLE `passenger` (
  `account_id` int(11) NOT NULL,
  `passenger_name` varchar(255) DEFAULT NULL,
  `passenger_age` int(11) DEFAULT NULL,
  `frequentflyer_id` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`account_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `airport` (
  `airport_code` varchar(8) NOT NULL,
  `airport_name` varchar(255) DEFAULT NULL,
  `airport_city` varchar(255) DEFAULT NULL,
  `airport_state` varchar(255) DEFAULT NULL,
  `airport_address` varchar(255) DEFAULT NULL,
  `classification` varchar(255) DEFAULT NULL,
  `time_zone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`airport_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `aircraft` (
  `airplane_number` varchar(30) NOT NULL,
  `airplane_make` varchar(255) DEFAULT NULL,
  `weight_limit` int(11) DEFAULT NULL,
  `flight_number` varchar(30) DEFAULT NULL,
  `airport_code` varchar(8) NOT NULL,
  PRIMARY KEY (`airplane_number`),
  CONSTRAINT `aircraft_ibfk_1` FOREIGN KEY (`airport_code`) REFERENCES `airport` (`airport_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flightcrew` (
  `employee_id` int(11) NOT NULL,
  `employee_name` varchar(255) DEFAULT NULL,
  `employee_title` varchar(255) DEFAULT NULL,
  `employee_age` int(11) DEFAULT NULL,
  `employee_street_address` varchar(255) DEFAULT NULL,
  `employee_zip_code` varchar(20) DEFAULT NULL,
  `employee_city` varchar(255) DEFAULT NULL,
  `employee_state` varchar(255) DEFAULT NULL,
  `airport_code` varchar(8) NOT NULL,
  PRIMARY KEY (`employee_id`),
  CONSTRAINT `flightcrew_ibfk_1` FOREIGN KEY (`airport_code`) REFERENCES `airport` (`airport_code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `flight` (
  `flight_number` varchar(30) NOT NULL,
  `departure_time` datetime DEFAULT NULL,
  `arrival_time` datetime DEFAULT NULL,
  `flight_price` decimal(10,2) DEFAULT NULL,
  `airport_code` varchar(8) NOT NULL,
  `gate_number` varchar(10) DEFAULT NULL,
  `airplane_number` varchar(30) NOT NULL,
  `employee_id` int(11) NOT NULL,
  PRIMARY KEY (`flight_number`),
  CONSTRAINT `flight_ibfk_1` FOREIGN KEY (`airplane_number`) REFERENCES `aircraft` (`airplane_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `flight_ibfk_2` FOREIGN KEY (`employee_id`) REFERENCES `flightcrew` (`employee_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ticket` (
  `ticket_id` varchar(20) NOT NULL,
  `from_location` varchar(255) DEFAULT NULL,
  `to_location` varchar(255) DEFAULT NULL,
  `seat` varchar(10) DEFAULT NULL,
  `price` decimal(10,2) DEFAULT NULL,
  `flight_status` varchar(50) DEFAULT NULL,
  `class` varchar(50) DEFAULT NULL,
  `account_id` int(11) NOT NULL,
  `flight_number` varchar(30) NOT NULL,
  PRIMARY KEY (`ticket_id`),
  CONSTRAINT `ticket_ibfk_1` FOREIGN KEY (`account_id`) REFERENCES `passenger` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `ticket_ibfk_2` FOREIGN KEY (`flight_number`) REFERENCES `flight` (`flight_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `luggage` (
  `luggage_id` varchar(20) NOT NULL,
  `carousel_number` varchar(10) DEFAULT NULL,
  `weight` decimal(10,2) DEFAULT NULL,
  `account_id` int(11) NOT NULL,
  `airplane_number` varchar(30) NOT NULL,
  PRIMARY KEY (`luggage_id`),
  CONSTRAINT `luggage_ibfk_1` FOREIGN KEY (`account_id`) REFERENCES `passenger` (`account_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `luggage_ibfk_2` FOREIGN KEY (`airplane_number`) REFERENCES `aircraft` (`airplane_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
