-- Schema for database: schemabench_0216
CREATE TABLE `game_sales` (
  `game` varchar(100) NOT NULL,
  `platform` varchar(64) DEFAULT NULL,
  `publisher` varchar(64) DEFAULT NULL,
  `developer` varchar(64) DEFAULT NULL,
  `games_sold` decimal(10,5) DEFAULT NULL,
  `year` int(11) DEFAULT NULL,
  PRIMARY KEY (`game`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviews` (
  `game` varchar(100) NOT NULL,
  `critic_score` float DEFAULT NULL,
  `user_score` float DEFAULT NULL,
  PRIMARY KEY (`game`),
  CONSTRAINT `reviews_ibfk_1` FOREIGN KEY (`game`) REFERENCES `game_sales` (`game`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_critic_years` (
  `year` int(11) NOT NULL,
  `avg_critic_score` decimal(5,4) DEFAULT NULL,
  PRIMARY KEY (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_critic_years_more_than_four_games` (
  `year` int(11) NOT NULL,
  `num_games` int(11) DEFAULT NULL,
  `avg_critic_score` decimal(5,4) DEFAULT NULL,
  PRIMARY KEY (`year`),
  CONSTRAINT `top_critic_years_more_than_four_games_ibfk_1` FOREIGN KEY (`year`) REFERENCES `top_critic_years` (`year`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_user_years_more_than_four_games` (
  `year` int(11) NOT NULL,
  `num_games` int(11) DEFAULT NULL,
  `avg_user_score` decimal(5,4) DEFAULT NULL,
  PRIMARY KEY (`year`),
  CONSTRAINT `top_user_years_more_than_four_games_ibfk_1` FOREIGN KEY (`year`) REFERENCES `top_critic_years` (`year`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
