CREATE TABLE `Customer_Ratings` (
  `Consumer_ID` int(11) NOT NULL,
  `Restaurant_ID` int(11) NOT NULL,
  `Overall_Rating` float DEFAULT NULL,
  `Food_Rating` float DEFAULT NULL,
  `Service_Rating` float DEFAULT NULL,
  PRIMARY KEY (`Consumer_ID`, `Restaurant_ID`),
  CONSTRAINT `Customer_Ratings_ibfk_1` FOREIGN KEY (`Consumer_ID`) REFERENCES `Customer_Details` (`Consumer_ID`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `Customer_Ratings_ibfk_2` FOREIGN KEY (`Restaurant_ID`) REFERENCES `Restaurants` (`Restaurant_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer_Details` (
  `Consumer_ID` int(11) NOT NULL,
  `City` varchar(255) DEFAULT NULL,
  `State` varchar(255) DEFAULT NULL,
  `Country` varchar(255) DEFAULT NULL,
  `Latitude` float DEFAULT NULL,
  `Longitude` float DEFAULT NULL,
  `Smoker` varchar(10) DEFAULT NULL,
  `Drink_Level` varchar(10) DEFAULT NULL,
  `Transportation_Method` varchar(255) DEFAULT NULL,
  `Marital_Status` varchar(10) DEFAULT NULL,
  `Children` int(11) DEFAULT NULL,
  `Age` int(11) DEFAULT NULL,
  `Occupation` varchar(255) DEFAULT NULL,
  `Budget` float DEFAULT NULL,
  PRIMARY KEY (`Consumer_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Customer_Preference` (
  `Consumer_ID` int(11) NOT NULL,
  `Preferred_Cuisine` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`Consumer_ID`),
  CONSTRAINT `Customer_Preference_ibfk_1` FOREIGN KEY (`Consumer_ID`) REFERENCES `Customer_Details` (`Consumer_ID`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `Restaurants` (
  `Restaurant_ID` int(11) NOT NULL,
  `Name` varchar(255) NOT NULL,
  `City` varchar(255) DEFAULT NULL,
  `State` varchar(255) DEFAULT NULL,
  `Country` varchar(255) DEFAULT NULL,
  `Zip_Code` varchar(10) DEFAULT NULL,
  `Latitude` float DEFAULT NULL,
  `Longitude` float DEFAULT NULL,
  `Alcohol_Service` varchar(10) DEFAULT NULL,
  `Smoking_Allowed` varchar(10) DEFAULT NULL,
  `Price` float DEFAULT NULL,
  `Franchise` varchar(10) DEFAULT NULL,
  `Area` varchar(255) DEFAULT NULL,
  `Parking` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`Restaurant_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `restaurant_cuisines` (
  `Restaurant_ID` int(11) NOT NULL,

  PRIMARY KEY (`Restaurant_ID`),
  CONSTRAINT `restaurant_cuisines_ibfk_1` FOREIGN KEY (`Restaurant_ID`) REFERENCES `Restaurants` (`Restaurant_ID`) ON DELETE CASCADE ON UPDATE CASCADE
