-- Schema for database: schemabench_0230
CREATE TABLE `player` (
  `id` int(11) NOT NULL,
  `first_name` varchar(255) DEFAULT NULL,
  `last_name` varchar(255) DEFAULT NULL,
  `age` int(11) DEFAULT NULL,
  `team` varchar(255) DEFAULT NULL,
  `team_number` int(11) DEFAULT NULL,
  `injured` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `player_ibfk_1` FOREIGN KEY (`team_number`) REFERENCES `team` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `team` (
  `id` int(11) NOT NULL,
  `num_of_players` int(11) DEFAULT NULL,
  `manager` varchar(255) DEFAULT NULL,
  `captain` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `match` (
  `id` int(11) NOT NULL,
  `team1_id` int(11) DEFAULT NULL,
  `team2_id` int(11) DEFAULT NULL,
  `stadium_id` int(11) DEFAULT NULL,
  `commentator1` varchar(255) DEFAULT NULL,
  `commentator2` varchar(255) DEFAULT NULL,
  `head_referee` varchar(255) DEFAULT NULL,
  `asst_referee1` varchar(255) DEFAULT NULL,
  `asst_referee2` varchar(255) DEFAULT NULL,
  `man_of_the_match` int(11) DEFAULT NULL,
  `home_score` int(11) DEFAULT NULL,
  `away_score` int(11) DEFAULT NULL,
  `captain_home` int(11) DEFAULT NULL,
  `captain_away` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `match_ibfk_1` FOREIGN KEY (`team1_id`) REFERENCES `team` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `match_ibfk_2` FOREIGN KEY (`team2_id`) REFERENCES `team` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `match_ibfk_3` FOREIGN KEY (`stadium_id`) REFERENCES `stadium` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stadium` (
  `id` int(11) NOT NULL,
  `capacity` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
