-- Schema for database: schemabench_0032
CREATE TABLE `axes_accessattempt` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_agent` varchar(255) DEFAULT NULL,
  `ip_address` varchar(45) DEFAULT NULL,
  `username` varchar(255) DEFAULT NULL,
  `http_accept_header` varchar(255) NOT NULL,
  `path_info` varchar(255) NOT NULL,
  `get_data` text NOT NULL,
  `post_data` text NOT NULL,
  `failures_since_start` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `axes_accesslog` (
  `id` int(11) NOT NULL,
  `user_agent` varchar(255) DEFAULT NULL,
  `ip_address` varchar(45) DEFAULT NULL,
  `username` varchar(255) DEFAULT NULL,
  `http_accept_header` varchar(255) NOT NULL,
  `path_info` varchar(255) NOT NULL,
  `logout_time` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `axes_accesslog_ibfk_1` FOREIGN KEY (`id`) REFERENCES `axes_accessattempt` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
