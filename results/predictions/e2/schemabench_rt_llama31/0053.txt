-- Schema for database: schemabench_0053
CREATE TABLE `canvas_users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL UNIQUE,
  `password` varchar(255) NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_posts` (
  `post_id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `slug` varchar(255) NOT NULL UNIQUE,
  `title` varchar(255) NOT NULL,
  `body` text NOT NULL,
  `featured_image` varchar(255) DEFAULT NULL,
  `summary` text DEFAULT NULL,
  PRIMARY KEY (`post_id`),
  CONSTRAINT `canvas_posts_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `canvas_users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_tags` (
  `tag_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL UNIQUE,
  PRIMARY KEY (`tag_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_topics` (
  `topic_id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL UNIQUE,
  PRIMARY KEY (`topic_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_posts_tags` (
  `post_id` int(11) NOT NULL,
  `tag_id` int(11) NOT NULL,
  PRIMARY KEY (`post_id`, `tag_id`),
  CONSTRAINT `canvas_posts_tags_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `canvas_posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `canvas_posts_tags_ibfk_2` FOREIGN KEY (`tag_id`) REFERENCES `canvas_tags` (`tag_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_posts_topics` (
  `post_id` int(11) NOT NULL,
  `topic_id` int(11) NOT NULL,
  PRIMARY KEY (`post_id`, `topic_id`),
  CONSTRAINT `canvas_posts_topics_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `canvas_posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `canvas_posts_topics_ibfk_2` FOREIGN KEY (`topic_id`) REFERENCES `canvas_topics` (`topic_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_user_meta` (
  `user_id` int(11) NOT NULL,
  `username` varchar(255) DEFAULT NULL,
  `avatar` varchar(255) DEFAULT NULL,
  `locale` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  CONSTRAINT `canvas_user_meta_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `canvas_users` (`user_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_views` (
  `view_id` int(11) NOT NULL AUTO_INCREMENT,
  `post_id` int(11) NOT NULL,
  `viewed_at` datetime NOT NULL,
  PRIMARY KEY (`view_id`),
  CONSTRAINT `canvas_views_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `canvas_posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `canvas_visits` (
  `visit_id` int(11) NOT NULL AUTO_INCREMENT,
  `post_id` int(11) NOT NULL,
  `visited_at` datetime NOT NULL,
  PRIMARY KEY (`visit_id`),
  CONSTRAINT `canvas_visits_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `canvas_posts` (`post_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
