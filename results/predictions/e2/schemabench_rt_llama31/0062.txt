CREATE TABLE `love_reactants` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `type` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `love_reacters` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `type` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `love_reaction_types` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `mass` float NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `love_reactions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `reactant_id` int(11) NOT NULL,
  `reacter_id` int(11) NOT NULL,
  `reaction_type_id` int(11) NOT NULL,
  `rate` float NOT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `love_reactions_ibfk_1` FOREIGN KEY (`reactant_id`) REFERENCES `love_reactants` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `love_reactions_ibfk_2` FOREIGN KEY (`reacter_id`) REFERENCES `love_reacters` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `love_reactions_ibfk_3` FOREIGN KEY (`reaction_type_id`) REFERENCES `love_reaction_types` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `love_reactant_reaction_counters` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `reaction_id` int(11) NOT NULL,
  `reactant_id` int(11) NOT NULL,
  `count` int(11) NOT NULL,
  `weight` float NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `love_reactant_reaction_counters_ibfk_1` FOREIGN KEY (`reaction_id`) REFERENCES `love_reactions` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `love_reactant_reaction_counters_ibfk_2` FOREIGN KEY (`reactant_id`) REFERENCES `love_reactants` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `love_reactant_reaction_totals` (
  `id` int(11) NOT NULL AUTO_INCREMENT,

  `total_count` int(11) NOT NULL,
  `total_weight` float NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `love_reactant_reaction_totals_ibfk_1` FOREIGN KEY (`reactant_id`) REFERENCES `love_reactants` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
