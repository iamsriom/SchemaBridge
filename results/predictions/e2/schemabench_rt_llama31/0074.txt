CREATE TABLE `clicks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `ip_address` varchar(45) NOT NULL,
  `country` varchar(100) NOT NULL,
  `referer` varchar(255) DEFAULT NULL,
  `user_agent` varchar(255) NOT NULL,
  `link_id` int(11) NOT NULL,
  `timestamp` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `clicks_ibfk_1` FOREIGN KEY (`link_id`) REFERENCES `links` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `links` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `short_url` varchar(255) NOT NULL,
  `long_url` varchar(2048) NOT NULL,
  `creator` int(11) NOT NULL,
  `ip_address` varchar(45) NOT NULL,
  `clicks_count` int(11) DEFAULT 0,
  `is_disabled` tinyint(1) DEFAULT 0,
  `is_custom` tinyint(1) DEFAULT 0,
  `is_api_created` tinyint(1) DEFAULT 0,
  PRIMARY KEY (`id`),
  CONSTRAINT `links_ibfk_1` FOREIGN KEY (`creator`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `migrations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `migration` varchar(255) NOT NULL,
  `batch` int(11) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `email` varchar(255) NOT NULL,
  `ip_address` varchar(45) NOT NULL,
  `recovery_key` varchar(255) DEFAULT NULL,
  `role` varchar(50) NOT NULL,
  `api_key` varchar(255) DEFAULT NULL,
  `is_active` tinyint(1) DEFAULT 1,
  `is_api_key_active` tinyint(1) DEFAULT 0,
  `api_key_quota` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
