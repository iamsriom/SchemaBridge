-- Schema for database: schemabench_0076
CREATE TABLE `hosts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `ssh_user` varchar(255) NOT NULL,
  `port` int(11) NOT NULL,
  `ip_address` varchar(45) NOT NULL,
  `custom_properties` json DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `checks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `host_id` int(11) NOT NULL,
  `check_type` varchar(255) NOT NULL,
  `status` varchar(255) NOT NULL,
  `is_enabled` boolean NOT NULL,
  `last_run_message` varchar(255) DEFAULT NULL,
  `last_run_output` text DEFAULT NULL,
  `last_run_timestamp` datetime DEFAULT NULL,
  `next_run_timestamp` datetime DEFAULT NULL,
  `custom_properties` json DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `checks_ibfk_1` FOREIGN KEY (`host_id`) REFERENCES `hosts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
