CREATE TABLE `billable` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `processor` varchar(255) NOT NULL,
  `processor_id` varchar(255) NOT NULL,
  `trial_end_date` datetime DEFAULT NULL,
  `card_type` varchar(50) NOT NULL,
  `last_four_digits` char(4) NOT NULL,
  `expiration_month` int(2) NOT NULL,
  `expiration_year` int(4) NOT NULL,
  `extra_billing_info` text DEFAULT NULL,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `processor` varchar(255) NOT NULL,
  `processor_id` varchar(255) NOT NULL,
  `amount` decimal(10,2) NOT NULL,
  `amount_refunded` decimal(10,2) DEFAULT NULL,
  `card_type` varchar(50) NOT NULL,
  `last_four_digits` char(4) NOT NULL,
  `expiration_month` int(2) NOT NULL,
  `expiration_year` int(4) NOT NULL,
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  CONSTRAINT `pay_charges_ibfk_1` FOREIGN KEY (`owner_id`) REFERENCES `billable` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
