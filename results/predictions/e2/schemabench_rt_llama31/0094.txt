CREATE TABLE `atom` (
  `atom_id` int(11) NOT NULL,
  `molecule_id` int(11) NOT NULL,
  `atom_type` varchar(255) NOT NULL,
  PRIMARY KEY (`atom_id`),
  CONSTRAINT `atom_ibfk_1` FOREIGN KEY (`molecule_id`) REFERENCES `molecule` (`molecule_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `bond` (
  `bond_id` int(11) NOT NULL,
  `atom_id_1` int(11) NOT NULL,
  `atom_id_2` int(11) NOT NULL,
  `bond_type` varchar(255) NOT NULL,
  PRIMARY KEY (`bond_id`),
  CONSTRAINT `bond_ibfk_1` FOREIGN KEY (`atom_id_1`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `bond_ibfk_2` FOREIGN KEY (`atom_id_2`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `gmember` (
  `gmember_id` int(11) NOT NULL,
  `atom_id` int(11) NOT NULL,
  `group_id` int(11) NOT NULL,
  PRIMARY KEY (`gmember_id`),
  CONSTRAINT `gmember_ibfk_1` FOREIGN KEY (`atom_id`) REFERENCES `atom` (`atom_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `gmember_ibfk_2` FOREIGN KEY (`group_id`) REFERENCES `group` (`group_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `group` (
  `group_id` int(11) NOT NULL,
  `group_type` varchar(255) NOT NULL,
  PRIMARY KEY (`group_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `molecule` (
  `molecule_id` int(11) NOT NULL,
  `activity` varchar(255) DEFAULT NULL,
  `logp` float DEFAULT NULL,
  `molecular_weight` float DEFAULT NULL,
  PRIMARY KEY (`molecule_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
