-- Schema for database: schemabench_0097
CREATE TABLE `schools` (
  `cdscode` char(14) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `website` varchar(255) DEFAULT NULL,
  `admin_contacts` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`cdscode`),
  UNIQUE KEY `cdscode_UNIQUE` (`cdscode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `frpm` (
  `cdscode` char(14) NOT NULL,
  `enrollment` int(11) NOT NULL,
  `meal_counts` int(11) NOT NULL,
  `eligibility_percentage_k12` decimal(5,2) NOT NULL,
  `eligibility_percentage_5_17` decimal(5,2) NOT NULL,
  PRIMARY KEY (`cdscode`),
  CONSTRAINT `frpm_ibfk_1` FOREIGN KEY (`cdscode`) REFERENCES `schools` (`cdscode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `satscores` (
  `cdscode` char(14) NOT NULL,
  `students_took_test` int(11) NOT NULL,
  `average_reading_score` decimal(5,2) NOT NULL,
  `average_math_score` decimal(5,2) NOT NULL,
  `average_writing_score` decimal(5,2) NOT NULL,
  `percentage_above_1500` decimal(5,2) NOT NULL,
  PRIMARY KEY (`cdscode`),
  CONSTRAINT `satscores_ibfk_1` FOREIGN KEY (`cdscode`) REFERENCES `schools` (`cdscode`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
