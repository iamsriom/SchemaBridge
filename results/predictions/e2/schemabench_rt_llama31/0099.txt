-- Schema for database: schemabench_0099
CREATE TABLE `paper` (
  `paper_id` int(11) NOT NULL,
  `class_label` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`paper_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `content` (
  `word_id` int(11) NOT NULL,
  `word` varchar(255) NOT NULL,
  `paper_id` int(11) NOT NULL,
  PRIMARY KEY (`word_id`),
  CONSTRAINT `content_ibfk_1` FOREIGN KEY (`paper_id`) REFERENCES `paper` (`paper_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `cites` (
  `citing_paper_id` int(11) NOT NULL,
  `cited_paper_id` int(11) NOT NULL,
  PRIMARY KEY (`citing_paper_id`, `cited_paper_id`),
  CONSTRAINT `cites_ibfk_1` FOREIGN KEY (`citing_paper_id`) REFERENCES `paper` (`paper_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `cites_ibfk_2` FOREIGN KEY (`cited_paper_id`) REFERENCES `paper` (`paper_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
