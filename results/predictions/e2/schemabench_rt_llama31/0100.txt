-- Schema for database: schemabench_0100
CREATE TABLE `households` (
  `household_id` char(10) NOT NULL,
  `year` int(11) NOT NULL,
  `income_rank` int(11) NOT NULL,
  `income_rank_category_1` int(11) NOT NULL,
  `income_rank_category_2` int(11) NOT NULL,
  `income_rank_category_3` int(11) NOT NULL,
  `reference_person_age` int(11) NOT NULL,
  PRIMARY KEY (`household_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `expenditures` (
  `expenditure_id` char(11) NOT NULL,
  `household_id` char(10) NOT NULL,
  `year` int(11) NOT NULL,
  `month` int(11) NOT NULL,
  `product_code` varchar(255) NOT NULL,
  `cost` decimal(10,2) NOT NULL,
  `is_gift` int(11) DEFAULT 1,
  PRIMARY KEY (`expenditure_id`),
  CONSTRAINT `expenditures_ibfk_1` FOREIGN KEY (`household_id`) REFERENCES `households` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `household_members` (
  `household_id` char(10) NOT NULL,
  `year` int(11) NOT NULL,
  `marital_status` varchar(50) NOT NULL,
  `sex` varchar(10) NOT NULL,
  `age` int(11) NOT NULL,
  `work_status` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`household_id`, `year`),
  CONSTRAINT `household_members_ibfk_1` FOREIGN KEY (`household_id`) REFERENCES `households` (`household_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
