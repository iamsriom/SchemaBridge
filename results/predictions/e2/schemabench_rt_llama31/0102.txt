-- Schema for database: schemabench_0102
CREATE TABLE `breweries` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) DEFAULT NULL,
  `city` varchar(255) DEFAULT NULL,
  `state` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `beers` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `brewery_id` int(11) NOT NULL,
  `abv` decimal(5,2) DEFAULT NULL,
  `ibu` int(11) DEFAULT NULL,
  `name` varchar(255) NOT NULL,
  `style` varchar(255) DEFAULT NULL,
  `ounces` decimal(5,2) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `beers_ibfk_1` FOREIGN KEY (`brewery_id`) REFERENCES `breweries` (`id`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
