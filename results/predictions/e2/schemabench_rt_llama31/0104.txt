-- Schema for database: schemabench_0104
CREATE TABLE `incidents` (
  `case_number` varchar(20) NOT NULL,
  `incident_date` datetime NOT NULL,
  `location` varchar(255) NOT NULL,
  `status` varchar(50) NOT NULL,
  `weapon_type` varchar(50) DEFAULT NULL,
  `num_subjects` int(11) NOT NULL,
  `num_officers` int(11) NOT NULL,
  `grand_jury_disposition` varchar(255) DEFAULT NULL,
  `attorney_general_forms` varchar(255) DEFAULT NULL,
  `summary` text DEFAULT NULL,
  `latitude` decimal(9,6) DEFAULT NULL,
  `longitude` decimal(9,6) DEFAULT NULL,
  PRIMARY KEY (`case_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `officers` (
  `case_number` varchar(20) NOT NULL,
  `race` varchar(50) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `full_name` varchar(200) NOT NULL,
  PRIMARY KEY (`case_number`, `full_name`),
  CONSTRAINT `officers_ibfk_1` FOREIGN KEY (`case_number`) REFERENCES `incidents` (`case_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `subjects` (
  `case_number` varchar(20) NOT NULL,
  `race` varchar(50) DEFAULT NULL,
  `gender` varchar(10) DEFAULT NULL,
  `first_name` varchar(100) NOT NULL,
  `last_name` varchar(100) NOT NULL,
  `full_name` varchar(200) NOT NULL,
  PRIMARY KEY (`case_number`, `full_name`),
  CONSTRAINT `subjects_ibfk_1` FOREIGN KEY (`case_number`) REFERENCES `incidents` (`case_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
