-- Schema for database: schemabench_0108
CREATE TABLE `classification` (
  `gene_id` int(11) NOT NULL,
  `localization` varchar(255) NOT NULL,
  PRIMARY KEY (`gene_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `genes` (
  `gene_id` int(11) NOT NULL,
  `essentiality` varchar(255) DEFAULT NULL,
  `class` varchar(255) DEFAULT NULL,
  `complex` varchar(255) DEFAULT NULL,
  `phenotype` varchar(255) DEFAULT NULL,
  `motif` varchar(255) DEFAULT NULL,
  `chromosome` varchar(255) DEFAULT NULL,
  `function` varchar(255) DEFAULT NULL,
  FOREIGN KEY (`gene_id`) REFERENCES `classification` (`gene_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `interactions` (
  `gene_id_1` int(11) NOT NULL,
  `gene_id_2` int(11) NOT NULL,
  `type` varchar(255) DEFAULT NULL,
  `expression_correlation` float DEFAULT NULL,
  PRIMARY KEY (`gene_id_1`, `gene_id_2`),
  FOREIGN KEY (`gene_id_1`) REFERENCES `classification` (`gene_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`gene_id_2`) REFERENCES `classification` (`gene_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
