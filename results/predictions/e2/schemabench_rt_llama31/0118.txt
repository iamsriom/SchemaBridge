-- Schema for database: schemabench_0118
CREATE TABLE `country` (
  `country_id` int(11) NOT NULL,
  `country_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`country_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `relation` (
  `country_id1` int(11) NOT NULL,
  `country_id2` int(11) NOT NULL,
  `relation` varchar(255) NOT NULL,
  PRIMARY KEY (`country_id1`, `country_id2`, `relation`),
  CONSTRAINT `relation_ibfk_1` FOREIGN KEY (`country_id1`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `relation_ibfk_2` FOREIGN KEY (`country_id2`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `stat` (
  `country_id` int(11) NOT NULL,
  `telephone` int(11) DEFAULT NULL,
  `agricultural_population` int(11) DEFAULT NULL,
  `energy_consumption` int(11) DEFAULT NULL,
  PRIMARY KEY (`country_id`),
  CONSTRAINT `stat_ibfk_1` FOREIGN KEY (`country_id`) REFERENCES `country` (`country_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
