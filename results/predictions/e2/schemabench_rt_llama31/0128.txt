-- Schema for database: schemabench_0128
CREATE TABLE `businesses` (
  `business_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `city` varchar(100) NOT NULL,
  `postal_code` varchar(20) NOT NULL,
  `latitude` decimal(10,8) NOT NULL,
  `longitude` decimal(11,8) NOT NULL,
  `phone_number` varchar(20) NOT NULL,
  `tax_code` varchar(50) NOT NULL,
  `business_certificate_number` varchar(50) NOT NULL,
  `application_date` datetime NOT NULL,
  `owner_name` varchar(255) NOT NULL,
  `owner_address` varchar(255) NOT NULL,
  `owner_city` varchar(100) NOT NULL,
  `owner_state` varchar(100) NOT NULL,
  `owner_zip_code` varchar(20) NOT NULL,
  PRIMARY KEY (`business_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `inspections` (
  `inspection_id` int(11) NOT NULL,
  `business_id` int(11) NOT NULL,
  `inspection_score` int(11) NOT NULL,
  `inspection_date` datetime NOT NULL,
  `inspection_type` varchar(100) NOT NULL,
  PRIMARY KEY (`inspection_id`),
  CONSTRAINT `inspections_ibfk_1` FOREIGN KEY (`business_id`) REFERENCES `businesses` (`business_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
CREATE TABLE `violations` (
  `violation_id` int(11) NOT NULL,
  `business_id` int(11) NOT NULL,
  `violation_date` datetime NOT NULL,
  `violation_type_id` int(11) NOT NULL,
  `risk_category` varchar(100) NOT NULL,
  `violation_description` text NOT NULL,
  PRIMARY KEY (`violation_id`),
  CONSTRAINT `violations_ibfk_1` FOREIGN KEY (`business_id`) REFERENCES `businesses` (`business_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
