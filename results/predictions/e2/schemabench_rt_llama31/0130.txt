CREATE TABLE `badges` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `date` datetime NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `badges_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `comments` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `post_id` int(11) NOT NULL,
  `score` int(11) NOT NULL,
  `text` text NOT NULL,
  `creation_date` datetime NOT NULL,
  `user_id` int(11) NOT NULL,
  `user_display_name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `comments_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `comments_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `postHistory` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `post_id` int(11) NOT NULL,
  `revision_guid` char(36) NOT NULL,
  `creation_date` datetime NOT NULL,
  `user_id` int(11) NOT NULL,
  `text` text NOT NULL,
  `comment` text DEFAULT NULL,
  `user_display_name` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `postHistory_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `postHistory_ibfk_2` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `postLinks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `creation_date` datetime NOT NULL,
  `post_id` int(11) NOT NULL,
  `related_post_id` int(11) NOT NULL,
  `link_type_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `postLinks_ibfk_1` FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `postLinks_ibfk_2` FOREIGN KEY (`related_post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `posts` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `post_type_id` int(11) NOT NULL,
  `accepted_answer_id` int(11) DEFAULT NULL,
  `creation_date` datetime NOT NULL,
  `score` int(11) NOT NULL,
  `view_count` int(11) NOT NULL,
  `body` text NOT NULL,
  `owner_user_id` int(11) NOT NULL,
  `last_activity_date` datetime NOT NULL,
  `title` varchar(255) NOT NULL,
  `tags` varchar(255) NOT NULL,
  `answer_count` int(11) NOT NULL,
  `comment_count` int(11) NOT NULL,
  `favorite_count` int(11) NOT NULL,
  `last_editor_user_id` int(11) DEFAULT NULL,
  `last_edit_date` datetime DEFAULT NULL,
  `community_owned_date` datetime DEFAULT NULL,
  `parent_post_id` int(11) DEFAULT NULL,
  `closed_date` datetime DEFAULT NULL,
  `owner_display_name` varchar(255) NOT NULL,
  `last_editor_display_name` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `posts_ibfk_1` FOREIGN KEY (`accepted_answer_id`) REFERENCES `posts` (`id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `posts_ibfk_2` FOREIGN KEY (`owner_user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `posts_ibfk_3` FOREIGN KEY (`last_editor_user_id`) REFERENCES `users` (`id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `posts_ibfk_4` FOREIGN KEY (`parent_post_id`) REFERENCES `posts` (`id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `tags` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `tag_name` varchar(255) NOT NULL,
  `count` int(11) NOT NULL,
  `excerpt_post_id` int(11) DEFAULT NULL,
  `wiki_post_id` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `tags_ibfk_1` FOREIGN KEY (`excerpt_post_id`) REFERENCES `posts` (`id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `tags_ibfk_2` FOREIGN KEY (`wiki_post_id`) REFERENCES `posts` (`id`) ON DELETE SET NULL ON UPDATE CASCADE

  `about_me` text DEFAULT NULL,
