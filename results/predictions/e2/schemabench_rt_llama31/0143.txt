-- Schema for database: schemabench_0143
CREATE TABLE `country` (
  `code` char(3) NOT NULL,
  `name` varchar(255) NOT NULL,
  `continent` varchar(255) NOT NULL,
  `region` varchar(255) NOT NULL,
  `surface_area` float NOT NULL,
  `independence_year` int(11) DEFAULT NULL,
  `population` int(11) NOT NULL,
  `life_expectancy` float DEFAULT NULL,
  `gnp` float DEFAULT NULL,
  `gnp_old` float DEFAULT NULL,
  `local_name` varchar(255) NOT NULL,
  `government_form` varchar(255) NOT NULL,
  `head_of_state` varchar(255) NOT NULL,
  `capital` int(11) DEFAULT NULL,
  `alternative_code` char(3) DEFAULT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `city` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `country_code` char(3) NOT NULL,
  `district` varchar(255) NOT NULL,
  `population` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `city_ibfk_1` FOREIGN KEY (`country_code`) REFERENCES `country` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `country_language` (
  `country_code` char(3) NOT NULL,
  `language` varchar(255) NOT NULL,
  `is_official` enum('T','F') NOT NULL,
  `percentage` float NOT NULL,
  PRIMARY KEY (`country_code`, `language`),
  CONSTRAINT `country_language_ibfk_1` FOREIGN KEY (`country_code`) REFERENCES `country` (`code`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
