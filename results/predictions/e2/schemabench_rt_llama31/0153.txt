-- Schema for database: schemabench_0153
CREATE TABLE `books` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(255) NOT NULL,
  `author` varchar(255) DEFAULT NULL,
  `publisher` varchar(255) DEFAULT NULL,
  `publication_date` date DEFAULT NULL,
  `isbn` varchar(13) DEFAULT NULL,
  `total_copies` int(11) NOT NULL,
  `available_copies` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `books_check_author_or_publisher` CHECK (`author` IS NOT NULL OR `publisher` IS NOT NULL)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `staffs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `date_of_joining` date DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `members` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `membership_status` varchar(50) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `issued_books` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `member_id` int(11) NOT NULL,
  `book_id` int(11) NOT NULL,
  `issue_date` date NOT NULL,
  `due_date` date NOT NULL,
  `staff_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `issued_books_ibfk_1` FOREIGN KEY (`member_id`) REFERENCES `members` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `issued_books_ibfk_2` FOREIGN KEY (`book_id`) REFERENCES `books` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `issued_books_ibfk_3` FOREIGN KEY (`staff_id`) REFERENCES `staffs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `return_books` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `issued_book_id` int(11) NOT NULL,
  `return_date` date NOT NULL,
  `staff_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `return_books_ibfk_1` FOREIGN KEY (`issued_book_id`) REFERENCES `issued_books` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `return_books_ibfk_2` FOREIGN KEY (`staff_id`) REFERENCES `staffs` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `fines` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `return_id` int(11) NOT NULL,
  `issued_book_id` int(11) NOT NULL,
  `member_id` int(11) NOT NULL,
  `fine_amount` decimal(10,2) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `fines_ibfk_1` FOREIGN KEY (`return_id`) REFERENCES `return_books` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fines_ibfk_2` FOREIGN KEY (`issued_book_id`) REFERENCES `issued_books` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fines_ibfk_3` FOREIGN KEY (`member_id`) REFERENCES `members` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
