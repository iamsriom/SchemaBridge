CREATE TABLE `company_info` (

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `trader_id` int(11) NOT NULL AUTO_INCREMENT,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `account_id` int(11) NOT NULL AUTO_INCREMENT,
  `cash_balance` decimal(10,2) NOT NULL,

  CONSTRAINT `broker_exchange_deals_ibfk_1` FOREIGN KEY (`broker_id`) REFERENCES `broker_info` (`broker_id`) ON DELETE CASCADE ON UPDATE CASCADE,
