CREATE TABLE `customer` (
  `customer_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `address` varchar(255) DEFAULT NULL,
  `internal_status` varchar(50) DEFAULT NULL,
  `contact_details` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  `city` varchar(100) DEFAULT NULL,
  `state` varchar(100) DEFAULT NULL,
  `zip_code` varchar(10) DEFAULT NULL,
  PRIMARY KEY (`customer_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `facility` (
  `facility_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  PRIMARY KEY (`facility_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `location` (
  `location_number` int(11) NOT NULL,
  `facility_number` int(11) NOT NULL,
  `location_name` varchar(255) NOT NULL,
  PRIMARY KEY (`location_number`),
  CONSTRAINT `location_ibfk_1` FOREIGN KEY (`facility_number`) REFERENCES `facility` (`facility_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `eventrequest` (
  `event_number` int(11) NOT NULL,
  `date_held` datetime NOT NULL,
  `date_requested` datetime NOT NULL,
  `facility_number` int(11) NOT NULL,
  `customer_number` int(11) NOT NULL,
  `authorization_date` datetime DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `estimated_cost` decimal(10,2) DEFAULT NULL,
  `estimated_audience` int(11) DEFAULT NULL,
  `budget_number` varchar(50) DEFAULT NULL,
  PRIMARY KEY (`event_number`),
  CONSTRAINT `eventrequest_ibfk_1` FOREIGN KEY (`facility_number`) REFERENCES `facility` (`facility_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `eventrequest_ibfk_2` FOREIGN KEY (`customer_number`) REFERENCES `customer` (`customer_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `employee` (
  `employee_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `department` varchar(100) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone_number` varchar(15) DEFAULT NULL,
  PRIMARY KEY (`employee_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `eventplan` (
  `plan_number` int(11) NOT NULL,
  `event_number` int(11) NOT NULL,
  `work_date` datetime NOT NULL,
  `notes` text DEFAULT NULL,
  `activity` varchar(255) DEFAULT NULL,
  `employee_number` int(11) NOT NULL,
  PRIMARY KEY (`plan_number`),
  CONSTRAINT `eventplan_ibfk_1` FOREIGN KEY (`event_number`) REFERENCES `eventrequest` (`event_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `eventplan_ibfk_2` FOREIGN KEY (`employee_number`) REFERENCES `employee` (`employee_number`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `resourcetbl` (
  `resource_number` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `rate` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`resource_number`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `eventplanline` (
  `plan_number` int(11) NOT NULL,
  `line_number` int(11) NOT NULL,
  `start_time` datetime NOT NULL,
  `end_time` datetime NOT NULL,
  `number_of_fields` int(11) DEFAULT NULL,
  `location_number` int(11) NOT NULL,
  `resource_number` int(11) NOT NULL,
  PRIMARY KEY (`plan_number`, `line_number`),
  CONSTRAINT `eventplanline_ibfk_1` FOREIGN KEY (`plan_number`) REFERENCES `eventplan` (`plan_number`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `eventplanline_ibfk_2` FOREIGN KEY (`location_number`) REFERENCES `location` (`location_number`) ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `eventplanline_ibfk_3` FOREIGN KEY (`resource_number`) REFERENCES `resourcetbl` (`resource_number`) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
