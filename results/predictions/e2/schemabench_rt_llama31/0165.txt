CREATE TABLE `athlete` (
  `code` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `function` varchar(100) NOT NULL,
  `country` varchar(100) NOT NULL,
  `height` float DEFAULT NULL,
  `weight` float DEFAULT NULL,
  `birth_date` date DEFAULT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `coach` (
  `code` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `function` varchar(100) NOT NULL,

  `disciplines` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `event` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `tag` varchar(50) NOT NULL,
  `sport` varchar(100) NOT NULL,
  `sport_code` varchar(50) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medalist` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `date_won` date NOT NULL,
  `medal_type` varchar(50) NOT NULL,
  `name` varchar(255) NOT NULL,
  `country` varchar(100) NOT NULL,
  `event_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `medalist_ibfk_1` FOREIGN KEY (`event_id`) REFERENCES `event` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `medal` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `medal_type` varchar(50) NOT NULL,
  `code` varchar(50) NOT NULL,
  `date` date NOT NULL,
  `event_id` int(11) NOT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `medal_ibfk_1` FOREIGN KEY (`event_id`) REFERENCES `event` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

  `country` varchar(100) NOT NULL,

  PRIMARY KEY (`country`)
