-- Schema for database: schemabench_0180
CREATE TABLE `aircraft` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `registration_number` varchar(50) NOT NULL,
  `manufacturer` varchar(100) NOT NULL,
  `model` varchar(100) NOT NULL,
  `owner` varchar(100) DEFAULT NULL,
  `operator` varchar(100) DEFAULT NULL,
  `status` varchar(50) DEFAULT NULL,
  `serial_number` varchar(50) DEFAULT NULL,
  `engine_type` varchar(50) DEFAULT NULL,
  `ads_b_equipped` tinyint(1) DEFAULT NULL,
  `acars_equipped` tinyint(1) DEFAULT NULL,
  `registration_date` datetime DEFAULT NULL,
  `deregistration_date` datetime DEFAULT NULL,
  `first_flight_date` datetime DEFAULT NULL,
  `built_date` datetime DEFAULT NULL,
  `category_description` varchar(255) DEFAULT NULL,
  `notes` text DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `traffic` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `callsign` varchar(50) NOT NULL,
  `flight_number` varchar(50) NOT NULL,
  `type_code` varchar(50) DEFAULT NULL,
  `current_latitude` decimal(9,6) DEFAULT NULL,
  `current_longitude` decimal(9,6) DEFAULT NULL,
  `previous_latitude` decimal(9,6) DEFAULT NULL,
  `previous_longitude` decimal(9,6) DEFAULT NULL,
  `altitude` int(11) DEFAULT NULL,
  `origin` varchar(100) DEFAULT NULL,
  `destination` varchar(100) DEFAULT NULL,
  `first_sighting` datetime DEFAULT NULL,
  `last_sighting` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `traffic_ibfk_1` FOREIGN KEY (`flight_number`) REFERENCES `aircraft` (`registration_number`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
