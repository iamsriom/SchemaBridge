-- Schema for database: schemabench_0198
CREATE TABLE `event_identifier` (
  `event_id` int(11) NOT NULL AUTO_INCREMENT,
  `event_type` varchar(255) NOT NULL,
  `event_name` varchar(255) NOT NULL,
  PRIMARY KEY (`event_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `campaign_identifier` (
  `campaign_id` int(11) NOT NULL AUTO_INCREMENT,
  `product_promoted` varchar(255) NOT NULL,
  `campaign_name` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  `end_date` datetime NOT NULL,
  PRIMARY KEY (`campaign_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `page_hierarchy` (
  `page_id` int(11) NOT NULL AUTO_INCREMENT,
  `page_name` varchar(255) NOT NULL,
  `product_category` varchar(255) NOT NULL,
  `product_id` int(11) NOT NULL,
  PRIMARY KEY (`page_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `users` (
  `user_id` int(11) NOT NULL AUTO_INCREMENT,
  `cookie_id` varchar(255) NOT NULL,
  `start_date` datetime NOT NULL,
  PRIMARY KEY (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `events` (
  `visit_id` int(11) NOT NULL AUTO_INCREMENT,
  `cookie_id` varchar(255) NOT NULL,
  `page_id` int(11) NOT NULL,
  `event_type_id` int(11) NOT NULL,
  `sequence_number` int(11) NOT NULL,
  `event_timestamp` datetime NOT NULL,
  PRIMARY KEY (`visit_id`),
  CONSTRAINT `events_ibfk_1` FOREIGN KEY (`cookie_id`) REFERENCES `users` (`cookie_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `events_ibfk_2` FOREIGN KEY (`page_id`) REFERENCES `page_hierarchy` (`page_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `events_ibfk_3` FOREIGN KEY (`event_type_id`) REFERENCES `event_identifier` (`event_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
