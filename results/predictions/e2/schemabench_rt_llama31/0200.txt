-- Schema for database: schemabench_0200
CREATE TABLE `interest_map` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `summary` text DEFAULT NULL,
  `created_at` datetime NOT NULL,
  `updated_at` datetime NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `interest_metrics` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `month` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `combined_month_year` char(7) NOT NULL,
  `interest_id` int(11) NOT NULL,
  `composition` float DEFAULT NULL,
  `index_value` float DEFAULT NULL,
  `ranking` int(11) DEFAULT NULL,
  `percentile_ranking` float DEFAULT NULL,
  PRIMARY KEY (`id`),
  CONSTRAINT `interest_metrics_ibfk_1` FOREIGN KEY (`interest_id`) REFERENCES `interest_map` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
