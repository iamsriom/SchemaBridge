-- Schema for database: schemabench_0212
CREATE TABLE `movie` (
  `movie_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `release_year` int(4) NOT NULL,
  `publication_date` datetime NOT NULL,
  `duration` int(11) NOT NULL,
  `country_of_origin` varchar(100) NOT NULL,
  `worldwide_box_office` decimal(15,2) DEFAULT NULL,
  `languages_spoken` varchar(255) NOT NULL,
  `production_company` varchar(255) NOT NULL,
  PRIMARY KEY (`movie_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `genre` (
  `genre_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  PRIMARY KEY (`genre_id`),
  FOREIGN KEY (`movie_id`) REFERENCES `movie` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `names` (
  `name_id` int(11) NOT NULL,
  `name` varchar(255) NOT NULL,
  `height` decimal(5,2) DEFAULT NULL,
  `date_of_birth` date DEFAULT NULL,
  `notable_movies` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`name_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `director_mapping` (
  `director_mapping_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  `name_id` int(11) NOT NULL,
  PRIMARY KEY (`director_mapping_id`),
  FOREIGN KEY (`movie_id`) REFERENCES `movie` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`name_id`) REFERENCES `names` (`name_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `role_mapping` (
  `role_mapping_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  `name_id` int(11) NOT NULL,
  `role` varchar(100) NOT NULL,
  PRIMARY KEY (`role_mapping_id`),
  FOREIGN KEY (`movie_id`) REFERENCES `movie` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  FOREIGN KEY (`name_id`) REFERENCES `names` (`name_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `ratings` (
  `rating_id` int(11) NOT NULL,
  `movie_id` int(11) NOT NULL,
  `average_rating` decimal(3,2) DEFAULT NULL,
  `total_rating` int(11) DEFAULT NULL,
  `median_rating` decimal(3,2) DEFAULT NULL,
  PRIMARY KEY (`rating_id`),
  FOREIGN KEY (`movie_id`) REFERENCES `movie` (`movie_id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
