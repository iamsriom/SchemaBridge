-- Schema for database: schemabench_0216
CREATE TABLE `game_sales` (
  `title` varchar(255) NOT NULL,
  `platform` varchar(100) DEFAULT NULL,
  `publisher` varchar(100) DEFAULT NULL,
  `developer` varchar(100) DEFAULT NULL,
  `copies_sold` int(11) DEFAULT NULL,
  `year_of_release` int(11) DEFAULT NULL,
  PRIMARY KEY (`title`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `reviews` (
  `title` varchar(255) NOT NULL,
  `user_score` float DEFAULT NULL,
  `critic_score` float DEFAULT NULL,
  PRIMARY KEY (`title`),
  CONSTRAINT `reviews_ibfk_1` FOREIGN KEY (`title`) REFERENCES `game_sales` (`title`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_critic_years` (
  `year` int(11) NOT NULL,
  `average_critic_score` float DEFAULT NULL,
  PRIMARY KEY (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_critic_years_more_than_four_games` (
  `year` int(11) NOT NULL,
  `average_critic_score` float DEFAULT NULL,
  `number_of_games` int(11) DEFAULT NULL,
  PRIMARY KEY (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;

CREATE TABLE `top_user_years_more_than_four_games` (
  `year` int(11) NOT NULL,
  `average_user_score` float DEFAULT NULL,
  `number_of_games` int(11) DEFAULT NULL,
  PRIMARY KEY (`year`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3;
